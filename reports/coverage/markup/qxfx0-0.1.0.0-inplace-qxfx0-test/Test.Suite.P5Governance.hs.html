  199 testAppendIdempotenceAndConflict :: Test
  200 testAppendIdempotenceAndConflict = TestCase $ do
  201   let (event, _) = promotionEvent 1 Nothing
  202   case appendGovernanceEventToHistory event [] of
  203     Left err -> assertFailure ("initial append failed: " <> T.unpack err)
  204     Right history1 -> do
  205       assertEqual "initial append stores one event" 1 (length history1)
  206       assertEqual "duplicate append is idempotent" (Right history1) (appendGovernanceEventToHistory event history1)
  207       let conflicting = event
  208             { geEnvelope = (geEnvelope event)
  209                 { geeReason = GovernanceReason "conflicting_duplicate" ["same id, changed reason"]
  210                 , geeEventHash = Nothing
  211                 , geePayloadHash = Nothing
  212                 }
  213             }
  214       case appendGovernanceEventToHistory conflicting history1 of
  215         Left _ -> pure ()
  216         Right _ -> assertFailure "conflicting duplicate unexpectedly accepted"

  627 promotionEvent :: Int -> Maybe GovernanceEvent -> (GovernanceEvent, PerspectiveRegistry)
  628 promotionEvent sequenceNo previousEvent =
  629   let bundle = mkPerspectiveBundleWithThesis sequenceNo "initial promoted thesis"
  630       candidate = opinionCore bundle
  631       registry = applyPerspectiveDecision sequenceNo defaultPerspectiveRegistry bundle candidate PpdPromoteEndorsed
  632       event = buildPerspectiveGovernanceEvent
  633         ActorRuntime
  634         sequenceNo
  635         "session-p5"
  636         previousEvent
  637         defaultPerspectiveRegistry
  638         registry
  639         bundle
  640         candidate
  641         PerspectiveAdmissibleAccepted
  642         (Just PpdPromoteEndorsed)
  643         PpdPromoteEndorsed
  644         (buildPerspectiveProjection registry (pibScope bundle))
  645   in (event, registry)
  646 
  647 revisionEvent :: Int -> Maybe GovernanceEvent -> PerspectiveRegistry -> (GovernanceEvent, PerspectiveRegistry)
  648 revisionEvent sequenceNo previousEvent beforeRegistry =
  649   let bundle = mkPerspectiveBundleWithThesis sequenceNo "revised promoted thesis"
  650       candidate = opinionCore bundle
  651       registry = applyPerspectiveDecision sequenceNo beforeRegistry bundle candidate PpdReviseActive
  652       event = buildPerspectiveGovernanceEvent
  653         ActorRuntime
  654         sequenceNo
  655         "session-p5"
  656         previousEvent
  657         beforeRegistry
  658         registry
  659         bundle
  660         candidate
  661         PerspectiveAdmissibleAccepted
  662         (Just PpdReviseActive)
  663         PpdReviseActive
  664         (buildPerspectiveProjection registry (pibScope bundle))
  665   in (event, registry)
