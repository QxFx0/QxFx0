  157 -- 4. JSON backward compatibility: MorphologyData can be parsed from JSON without mdFormsBySurface field
  158 
  159 testJsonBackwardCompatibility :: Assertion
  160 testJsonBackwardCompatibility = do
  161   let jsonWithoutFormsBySurface = T.unlines
  162         [ "{"
  163         , "  \"mdPrepositional\": {\"свобода\": \"свободе\"},"
  164         , "  \"mdGenitive\": {\"свобода\": \"свободы\"},"
  165         , "  \"mdNominative\": {\"свободе\": \"свобода\"}"
  166         , "}"
  167         ]
  168       jsonWithEmptyFormsBySurface = T.unlines
  169         [ "{"
  170         , "  \"mdPrepositional\": {\"свобода\": \"свободе\"},"
  171         , "  \"mdGenitive\": {\"свобода\": \"свободы\"},"
  172         , "  \"mdNominative\": {\"свободе\": \"свобода\"},"
  173         , "  \"mdFormsBySurface\": {}"
  174         , "}"
  175         ]
  176   case eitherDecodeStrict' (TE.encodeUtf8 jsonWithoutFormsBySurface) of
  177     Left err -> assertFailure ("MorphologyData should parse without mdFormsBySurface: " <> err)
  178     Right (md :: MorphologyData) -> do
  179       assertEqual "mdNominative should be populated" (Map.fromList [("свободе", "свобода")]) (mdNominative md)
  180       assertEqual "mdGenitive should be populated" (Map.fromList [("свобода", "свободы")]) (mdGenitive md)
  181       assertEqual "mdPrepositional should be populated" (Map.fromList [("свобода", "свободе")]) (mdPrepositional md)
  182       assertEqual "mdFormsBySurface should default to empty" Map.empty (mdFormsBySurface md)
  183   case eitherDecodeStrict' (TE.encodeUtf8 jsonWithEmptyFormsBySurface) of
  184     Left err -> assertFailure ("MorphologyData should parse with empty mdFormsBySurface: " <> err)
  185     Right (md :: MorphologyData) ->
  186       assertEqual "mdFormsBySurface should be empty map" Map.empty (mdFormsBySurface md)
