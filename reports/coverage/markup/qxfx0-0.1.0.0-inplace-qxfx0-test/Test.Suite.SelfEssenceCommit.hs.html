  305 -- | E6 — once committed, never reverts; commitment invariant.
  306 --
  307 -- Replaces the previous tautological constructor-match with a
  308 -- property that runs N continuations through the actual 'witness'
  309 -- morphism and verifies the commitment remains invariant.
  310 propStickyCommitment :: Property
  311 propStickyCommitment =
  312   forAll arbitraryEssenceCommitment $ \commitment ->
  313   forAll arbitraryEssenceTrajectory $ \trajBase ->
  314   forAll (choose (1, 20)) $ \n ->
  315   forAll (vectorOf n arbitraryStepInput) $ \steps ->
  316     let initial = EssenceCommitted trajBase commitment
  317         step e (ord, (ce, fd, delib)) =
  318           case e of
  319             EssenceCommitted t c ->
  320               EssenceCommitted
  321                 (witness defaultEssenceModulation ord ce fd delib t)
  322                 c
  323             EssenceUncommitted _ -> e  -- unreachable from this seed
  324         final = foldl' step initial (zip [1..] steps)
  325     in case final of
  326          EssenceCommitted _ c' -> c' === commitment
  327          EssenceUncommitted _  -> counterexample "commitment lost" False
  328   where
  329     arbitraryStepInput =
  330       (,,) <$> arbitraryConatusEnergy <*> arbitraryField <*> arbitraryDeliberation

  344 -- | Q7 — sliding-window exact semantics.
  345 propSlidingWindowExactSemantics :: Property
  346 propSlidingWindowExactSemantics =
  347   forAll (choose (1, 6)) $ \window ->
  348   forAll (vectorOf window arbitraryEssenceWitness) $ \ws ->
  349   forAll arbitraryUnitDouble $ \angst ->
  350     let em = defaultEssenceModulation { emConatusFloorWindow = window
  351                                       , emConatusStructuralFloor = 0.5
  352                                       , emAngstCommitmentThreshold = 1.1
  353                                       }
  354         traj = EssenceTrajectory
  355           { etWitnesses    = Seq.fromList ws
  356           , etAngstLevel   = angst  -- below threshold so angst never fires
  357           , etConatusFloor = 0.0
  358           , etCapacity     = emTrajectoryCapacity em
  359           }
  360         allSubFloor = all (\w -> ewConatusScalar w < 0.5) ws
  361         lenOK       = length ws >= window
  362     in shouldCommit em traj == if allSubFloor && lenOK
  363                                  then Just TriggerConatusErosion
  364                                  else Nothing

  366 -- | C1 — reconcile courtesy never widens admissible family set.
  367 --
  368 -- When 'RuleTiedFallback' fires and a courtesy predicate is
  369 -- supplied, the fallback may switch to the holistic plan, but
  370 -- ONLY if it is admissible under the predicate.  This property
  371 -- verifies that the switch never occurs to an inadmissible plan.
  372 -- Cases where RuleTiedFallback does not fire are discarded; cases
  373 -- where it fires but the output remains formal (or the families
  374 -- coincide) pass trivially.
  375 propReconcileCourtesyNeverWidens :: Property
  376 propReconcileCourtesyNeverWidens =
  377   forAll arbitraryEssenceCommitment $ \c ->
  378   forAll arbitrarySalience $ \sal ->
  379   forAll arbitraryPlan $ \hPlan ->
  380   forAll arbitraryPlan $ \fPlan ->
  381   forAll arbitraryField $ \fd ->
  382     let adm = isAdmissible c
  383         hp  = holisticProposal hPlan fd
  384         fp  = formalProposal (const fPlan)
  385         result = reconcile (Just adm) sal hp fp fd
  386         out    = delibReconciled result
  387         rule   = dtRule (delibTrace result)
  388         hAdmissible = adm hPlan
  389     in rule == RuleTiedFallback ==>
  390          (if planFamily out == planFamily hPlan && planFamily hPlan /= planFamily fPlan
  391             then hAdmissible
  392             else True)
  393   where
  394     isAdmissible commitment plan =
  395       case validatePlan commitment plan of
  396         Right _ -> True
  397         Left  _ -> False

  425 testFlagOnCommitmentFires :: IO ()
  426 testFlagOnCommitmentFires = do
  427   let em = defaultEssenceModulation
  428         { emAngstCommitmentThreshold = 0.5
  429         , emAngstAccrualRate = 0.1
  430         }
  431       -- Build a trajectory with angst just below threshold
  432       traj0 = emptyTrajectory { etAngstLevel = 0.45 }
  433       fd = emptyField
  434       ce = ConatusEnergy 0.5 (ConatusComponents 0 0 0 0)
  435       -- One divergence witness to push angst over threshold.
  436       -- dtAgreement must NOT be Agree, otherwise extractMode counts
  437       -- the witness as Integrative regardless of dtRule.
  438       delib = defaultDeliberation
  439         { delibTrace = (delibTrace defaultDeliberation)
  440             { dtRule = RuleHolisticAdvantage
  441             , dtAgreement = DivergeOnFamily
  442             , dtDivergence = 0.6
  443             }
  444         }
  445       traj1 = witness em 1 ce fd delib traj0
  446   -- After one witness with high divergence, angst should jump to 0.55
  447   assertBool "angst above threshold"
  448     (etAngstLevel traj1 >= emAngstCommitmentThreshold em)
  449   case shouldCommit em traj1 of
  450     Just TriggerAngstThreshold -> pure ()
  451     other -> assertFailure
  452       ("expected Just TriggerAngstThreshold, got " ++ show other)
  453   let committed = commit 1 TriggerAngstThreshold traj1
  454   -- With one non-Agree HolisticAdvantage witness, extractMode counts
  455   -- it toward Dialogical (hRate = 1), so commit produces Dialogical.
  456   assertEqual "extractMode on single HolisticAdvantage"
  457     EssenceDialogical (extractMode traj1)
  458   assertEqual "committed mode from extractMode"
  459     EssenceDialogical (ecMode committed)

  461 testFlagOnRuptureThrows :: IO ()
  462 testFlagOnRuptureThrows = do
  463   let commitment = EssenceCommitment
  464         { ecMode = EssenceContemplative
  465         , ecTrigger = TriggerAngstThreshold
  466         , ecCommittedAt = 1
  467         , ecWitnessHash = TrajectoryHash "sha256:test-rupture"
  468         }
  469       -- A Plan with family CMContact is NOT admissible for Contemplative
  470       badPlan = defaultPlan { planFamily = CMContact }
  471       violation = validatePlan commitment badPlan
  472   -- Verify the violation is the exact family mismatch we expect.
  473   case violation of
  474     Left (ViolationFamilyMismatch EssenceContemplative CMContact) -> pure ()
  475     other -> assertFailure
  476       ("expected family mismatch violation, got " ++ show other)
  477   -- Verify the violation is a Left (would become EssenceRupture in
  478   -- resolveFinalizeCommit before persistence).
  479   assertBool "violation is Left" (isLeft violation)

  485 -- | §4 calibration — corrected Conatus floor unit mismatch.
  486 --
  487 -- 'computeConatusEnergy' produces a log-scale unbounded scalar
  488 -- (healthy range ~14-15).  The Phase 9 floor of 0.5 was calibrated
  489 -- against 'arbitraryUnitDouble' ([0,1]) generators and could never
  490 -- fire in production.  The corrected floor (7.0, ≈ half healthy)
  491 -- triggers on realistic post-violation values (~5).  This test also
  492 -- regression-locks against reverting to the buggy Phase 9 floor.
  493 testConatusErosionFiresUnderCorrectedFloor :: IO ()
  494 testConatusErosionFiresUnderCorrectedFloor = do
  495   let -- 8 witnesses (== 'emConatusFloorWindow') all with
  496       -- 'ewConatusScalar' = 5.0 (post-violation state).
  497       mkWitness scalar = EssenceWitness
  498         { ewTurnOrdinal     = 1
  499         , ewSalienceDriver  = DrivenByDefault
  500         , ewReconcileRule   = RuleAgreement
  501         , ewAgreement       = Agree
  502         , ewDivergence      = 0.0
  503         , ewConatusScalar   = scalar
  504         , ewFieldSignature  = fieldSignature defaultEssenceModulation emptyField
  505         }
  506       ws = replicate 8 (mkWitness 5.0)
  507       traj = emptyTrajectory { etWitnesses = Seq.fromList ws }
  508   -- With 'defaultEssenceModulation' (floor = 7.0), all witnesses
  509   -- are sub-floor → Conatus erosion fires.
  510   case shouldCommit defaultEssenceModulation traj of
  511     Just TriggerConatusErosion -> pure ()
  512     other -> assertFailure
  513       ("expected Just TriggerConatusErosion under corrected floor, got "
  514        ++ show other)
  515   -- With 'phase9EssenceModulation' (floor = 0.5), 5.0 is NOT sub-floor
  516   -- → no trigger.  Regression lock against accidental reversion.
  517   case shouldCommit phase9EssenceModulation traj of
  518     Nothing -> pure ()
  519     other -> assertFailure
  520       ("Phase 9 buggy floor should NOT trigger on scalar 5.0, got "
  521        ++ show other)
