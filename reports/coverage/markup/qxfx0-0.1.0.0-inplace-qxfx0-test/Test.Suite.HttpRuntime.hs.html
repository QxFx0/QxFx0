  951 queryInt :: NSQL.Database -> Text -> IO Int
  952 queryInt db sql = do
  953   mStmt <- NSQL.prepare db sql
  954   case mStmt of
  955     Left _ -> pure 0
  956     Right stmt -> do
  957       hasRow <- NSQL.stepRow stmt
  958       result <- if hasRow then NSQL.columnInt stmt 0 else pure 0
  959       NSQL.finalize stmt
  960       pure result

 1001 removePathIfExists :: FilePath -> IO ()

 1003   isDir <- doesDirectoryExist path
 1004   if isDir
 1005     then removePathForcibly path

 1008 prepareInstalledArtifactRoot :: FilePath -> FilePath -> IO ()
 1009 prepareInstalledArtifactRoot sourceRoot artifactRoot = do
 1010   removePathIfExists artifactRoot
 1011   createDirectoryIfMissing True artifactRoot
 1012   mapM_
 1013     (\relative -> copyTree (sourceRoot </> relative) (artifactRoot </> relative))
 1014     [ "semantics"

 1016     , "spec"

 1018     , "scripts"

 1021 copyTree :: FilePath -> FilePath -> IO ()
 1022 copyTree source destination = do
 1023   isDir <- doesDirectoryExist source
 1024   if isDir
 1025     then do
 1026       createDirectoryIfMissing True destination
 1027       entries <- listDirectory source
 1028       mapM_ (\entry -> copyTree (source </> entry) (destination </> entry)) entries
 1029     else do
 1030       createDirectoryIfMissing True (takeDirectory destination)
 1031       copyFile source destination
