  665 postTurnRawBodyWithHeaders port headerMutators body = do

  693         Nothing -> assertFailure (label <> " missing field: " <> T.unpack fieldName) >> fail "unreachable"
  694     _ -> assertFailure (label <> " expected JSON object") >> fail "unreachable"

  733 runtimeSessionsSnapshot :: FilePath -> IO (Bool, Int)
  734 runtimeSessionsSnapshot dbPath = do
  735   exists <- doesFileExist dbPath
  736   if not exists
  737     then pure (False, 0)
  738     else do
  739       mDb <- NSQL.open dbPath
  740       case mDb of
  741         Left _ -> pure (True, 0)
  742         Right db -> do
  743           tableCount <- queryInt db "SELECT count(*) FROM sqlite_master WHERE type='table' AND name='runtime_sessions'"
  744           rowCount <- if tableCount > 0 then queryInt db "SELECT count(*) FROM runtime_sessions" else pure 0
  745           NSQL.close db
  746           pure (True, rowCount)

  748 queryInt :: NSQL.Database -> Text -> IO Int
  749 queryInt db sql = do
  750   mStmt <- NSQL.prepare db sql
  751   case mStmt of
  752     Left _ -> pure 0
  753     Right stmt -> do
  754       hasRow <- NSQL.stepRow stmt
  755       result <- if hasRow then NSQL.columnInt stmt 0 else pure 0
  756       NSQL.finalize stmt
  757       pure result
