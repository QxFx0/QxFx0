   30 testGraftFruitCreatesBranchAndCount :: Test
   31 testGraftFruitCreatesBranchAndCount = TestCase $ do
   32   let fruit = mkFruit "freedom requires responsibility" "свобода" SourceInternal True 0.5 0.3

   66 testPruneFruitsDropsInvalidAndNegative :: Test
   67 testPruneFruitsDropsInvalidAndNegative = TestCase $ do
   68   let validFruit = mkFruit "valid" "свобода" SourceInternal True 0.5 0.3
   69       invalidFruit = mkFruit "invalid" "ошибка" SourceLLM False 0.0 0.0
   70       negativeFruit = mkFruit "negative" "риск" SourceLLM True (-0.5) 0.0
   71       tree0 = emptyKnowledgeTree
   72         { ktBranches = M.singleton "agreement"
   73             [ Branch
   74                 { brRule = "agreement"
   75                 , brFruits = [validFruit, invalidFruit, negativeFruit]
   76                 , brHealth = 0.0
   77                 , brCreatedTurn = 1
   78                 }
   79             ]
   80         , ktQuarantine = [invalidFruit]
   81         }
   82       (tree1, dropped) = pruneFruits 1 tree0
   83       remaining = concatMap brFruits (concat (M.elems (ktBranches tree1)))
   84   assertEqual "invalid branch fruit and invalid quarantine fruit must be pruned"
   85     3 dropped
   86   assertEqual "only validated non-degrading fruit should remain grafted"
   87     [validFruit] remaining
   88   assertEqual "invalid quarantine entries must be removed"
   89     [] (ktQuarantine tree1)

  155 -- | Regression lock for audit-round3 P0 #3 (partial mitigation): the
  156 -- 'pruneFruits' pass must drain unvalidated quarantine entries even
  157 -- when the quarantine has accreted in bulk.  Without this cleanup,
  158 -- @ktQuarantine@ grows without bound across long-running sessions.
  159 -- The current code path keeps validated entries (still a known
  160 -- residual gap requiring a hard cap), but unvalidated entries must
  161 -- not survive a single prune cycle regardless of how many accumulate.
  162 testPruneFruitsCleansUnvalidatedQuarantineUnderBulkInsertion :: Test
  163 testPruneFruitsCleansUnvalidatedQuarantineUnderBulkInsertion = TestCase $ do
  164   let bulkUnvalidated =
  165         [ mkFruit ("u" <> T.pack (show i)) "x" SourceLLM False 0.0 0.0
  166         | i <- [1 .. 500 :: Int]
  167         ]
  168       tree0 = foldl' (flip quarantineFruit) emptyKnowledgeTree bulkUnvalidated
  169   assertEqual "bulk insertion must accrete in quarantine before prune"
  170     500 (length (ktQuarantine tree0))
  171   let (tree1, dropped) = pruneFruits 100 tree0
  172   assertEqual "all unvalidated quarantine entries must be dropped on prune"
  173     500 dropped
  174   assertEqual "no unvalidated quarantine entries must remain after prune"
  175     [] (ktQuarantine tree1)

  177 -- | Regression lock G1: 'rootStressSignal' must remain in [0, 1] and
  178 -- never propagate NaN/Infinity even when fed a tree whose components
  179 -- could otherwise produce non-finite arithmetic (e.g. degenerate
  180 -- branch health values).  The NaN-safe 'clampUnit' in
  181 -- "QxFx0.Learning.KnowledgeTree" guarantees a finite signal at the
  182 -- calibration boundary.
  183 testRootStressSignalIsNaNSafe :: Test
  184 testRootStressSignalIsNaNSafe = TestCase $ do
  185   let nanBranch = Branch
  186         { brRule = "agreement"
  187         , brFruits = [mkFruit "p" "" SourceInternal True 0.0 0.0]
  188         , brHealth = 0/0
  189         , brCreatedTurn = 1
  190         }
  191       infBranch = Branch
  192         { brRule = "repair"
  193         , brFruits = [mkFruit "q" "" SourceInternal True 0.0 0.0]
  194         , brHealth = (-1)/0
  195         , brCreatedTurn = 1
  196         }
  197       tree = emptyKnowledgeTree
  198         { ktBranches = M.fromList [("agreement", [nanBranch]), ("repair", [infBranch])]
  199         , ktQuarantinedCount = 1
  200         , ktGraftedCount = 1
  201         }
  202       stress = rootStressSignal tree
  203   assertBool "root stress must not be NaN under non-finite branch health"
  204     (not (isNaN stress))
  205   assertBool "root stress must not be Infinity under non-finite branch health"
  206     (not (isInfinite stress))
  207   assertBool "root stress must stay bounded to the unit interval"
  208     (stress >= 0.0 && stress <= 1.0)

  210 mkFruit :: Text -> Text -> KnowledgeSource -> Bool -> Double -> Double -> KnowledgeFruit
  211 mkFruit prop word src valid cDelta pDelta =
  212   KnowledgeFruit
  213     { kfProposition = prop
  214     , kfWord = word
  215     , kfSource = src
  216     , kfValidated = valid
  217     , kfConatusDelta = cDelta
  218     , kfPredictiveDelta = pDelta
  219     , kfGraftedTurn = Nothing
  220     , kfObservedTurn = 1
  221     }
