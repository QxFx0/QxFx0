   56 testOutcomeLearningRingBufferIsBounded :: Test
   57 testOutcomeLearningRingBufferIsBounded = TestCase $ do
   58   let samples =
   59         [ mkSampleAcceptedStrong i DialogueOutcomeSuccess "topic"
   60         | i <- [1 .. 50 :: Int]
   61         ]
   62       state = foldl' (flip updateDialogueOutcomeLearning) emptyDialogueOutcomeLearningState samples
   63   assertBool "dolRecentOutcomes must stay bounded to maxRecentOutcomeSamples"
   64     (length (dolRecentOutcomes state) <= 12)
   65   assertEqual "dolRecentOutcomes must hit the cap exactly under sustained insertion"
   66     12 (length (dolRecentOutcomes state))
   67   assertEqual "dolSuccessCount must reflect every accepted-strong success update"
   68     50 (dolSuccessCount state)

   98 testSpeechPolicyOnlyMutatesOnAcceptedStrongUpdate :: Test
   99 testSpeechPolicyOnlyMutatesOnAcceptedStrongUpdate = TestCase $ do
  100   -- ADR-0032 §2.2: weak/observational outcomes must not mutate
  101   -- speech policy.  Only AdaptiveAccepted + dosStrongUpdate triggers
  102   -- the bounded delta.
  103   let weakSample = mkSampleObservationalOnly 1 DialogueOutcomeSuccess "x"
  104       strongSample = mkSampleAcceptedStrong 1 DialogueOutcomeSuccess "x"
  105       afterWeak = updateSpeechPolicy weakSample StyleStandard emptySpeechPolicyState
  106       afterStrong = updateSpeechPolicy strongSample StyleStandard emptySpeechPolicyState
  107   assertEqual "weak / observational sample must not mutate speech policy"
  108     emptySpeechPolicyState afterWeak
  109   assertBool "accepted + strong-update sample must mutate speech policy"
  110     (afterStrong /= emptySpeechPolicyState)
  111   assertEqual "successful update must record the turn it occurred on"
  112     1 (spsLastUpdatedTurn afterStrong)

  114 testSpeechPolicyPatternMapsBounded :: Test
  115 testSpeechPolicyPatternMapsBounded = TestCase $ do
  116   -- regression: even if a state arrives with oversized pattern maps
  117   -- (e.g. legacy snapshot before bounding was introduced), a single
  118   -- update pass through 'updateSpeechPolicy' must normalize them back
  119   -- to at most 'maxSpeechPatternEntries' (8) entries.
  120   let oversized =
  121         emptySpeechPolicyState
  122           { spsFailedPatterns =
  123               M.fromList [("k" <> T.pack (show i), 1) | i <- [1 .. 20 :: Int]]
  124           }
  125       sample = mkSampleAcceptedStrong 1 DialogueOutcomeSuccess "topic"
  126       next = updateSpeechPolicy sample StyleStandard oversized
  127   assertBool "spsFailedPatterns must be capped to 8 after any update"
  128     (M.size (spsFailedPatterns next) <= 8)
  129   assertBool "spsSuccessfulPatterns must remain bounded"
  130     (M.size (spsSuccessfulPatterns next) <= 8)

  132 testRenderStylePreservesForcedRecovery :: Test
  133 testRenderStylePreservesForcedRecovery = TestCase $ do
  134   -- ADR-0032 §2.2: speech policy may bias future render style but must
  135   -- never override an already forced StyleRecovery.
  136   let aggressivePolicy = emptySpeechPolicyState
  137         { spsRepairBias = 1.0
  138         , spsAmbiguityTolerance = 0.0
  139         }
  140   assertEqual "forced StyleRecovery must survive aggressive policy bias"
  141     StyleRecovery
  142     (adjustRenderStyleForSpeechPolicy aggressivePolicy StyleRecovery)
  143   assertEqual "forced StyleRecovery must survive empty policy"
  144     StyleRecovery
  145     (adjustRenderStyleForSpeechPolicy emptySpeechPolicyState StyleRecovery)

  171 testIsWeakAcknowledgementCoversCommonPhrases :: Test
  172 testIsWeakAcknowledgementCoversCommonPhrases = TestCase $ do
  173   -- Cue list (DialogueDevelopment.hs:401): ack, понял, поняла, ясно,
  174   -- спасибо, thank you, thanks, i understand, makes sense.
  175   assertBool "Russian \"понял\" must be detected as weak acknowledgement"
  176     (isWeakAcknowledgementText "понял")
  177   assertBool "Russian \"ясно\" must be detected as weak acknowledgement"
  178     (isWeakAcknowledgementText "ясно")
  179   assertBool "English \"thanks\" must be detected as weak acknowledgement"
  180     (isWeakAcknowledgementText "thanks")
  181   assertBool "phrase \"makes sense\" must be detected as weak acknowledgement"
  182     (isWeakAcknowledgementText "yes, that makes sense")
  183   assertBool "leading and trailing whitespace must not block detection"
  184     (isWeakAcknowledgementText "   спасибо   ")
  185   assertBool "non-acknowledgement assertion must not match"
  186     (not (isWeakAcknowledgementText "это утверждение требует доказательства"))
  187   assertBool "empty text must not match"
  188     (not (isWeakAcknowledgementText ""))

  190 testNormalizeDialogueTextFoldsAndStrips :: Test
  191 testNormalizeDialogueTextFoldsAndStrips = TestCase $ do
  192   assertEqual "normalization must lower-case, strip, and collapse whitespace"
  193     "привет мир"
  194     (normalizeDialogueText "  ПриВЕТ   МИР  ")
  195   -- NOTE: 'normalizeDialogueText' replaces "ё" before toLower, so it
  196   -- only folds *lowercase* ё.  Locking that exact contract here so a
  197   -- future fix that also handles capital Ё forces an intentional
  198   -- update to this regression.
  199   assertEqual "normalization must collapse lowercase \"ё\" to \"е\""
  200     "ежик"
  201     (normalizeDialogueText "ёжик")
  202   assertEqual "normalization of empty input must yield empty"
  203     ""
  204     (normalizeDialogueText "   ")

  221 mkSampleAcceptedStrong :: Int -> DialogueOutcomeKind -> Text -> DialogueOutcomeSample
  222 mkSampleAcceptedStrong turn kind topic = DialogueOutcomeSample
  223   { dosTurn = turn
  224   , dosKind = kind
  225   , dosTopic = topic
  226   , dosSignals = ["test"]
  227   , dosEvidenceStrength = EvidenceStrong
  228   , dosStrongUpdate = True
  229   , dosDecisionRecord = AdaptiveDecisionRecord
  230       { adrTurn = turn
  231       , adrCause = "test"
  232       , adrEvidence = ["test"]
  233       , adrConfidence = 0.8
  234       , adrBoundedDelta = ["recent_outcomes<=12"]
  235       , adrDecision = AdaptiveAccepted
  236       , adrTargets = [MutDialogueOutcome]
  237       , adrMutationRecords = []
  238       }
  239   }

  241 mkSampleObservationalOnly :: Int -> DialogueOutcomeKind -> Text -> DialogueOutcomeSample
  242 mkSampleObservationalOnly turn kind topic =
  243   let base = mkSampleAcceptedStrong turn kind topic
  244       decision = (dosDecisionRecord base) { adrDecision = AdaptiveObserved }
  245   in base
  246        { dosEvidenceStrength = EvidenceObservational
  247        , dosStrongUpdate = False
  248        , dosDecisionRecord = decision
  249        }
