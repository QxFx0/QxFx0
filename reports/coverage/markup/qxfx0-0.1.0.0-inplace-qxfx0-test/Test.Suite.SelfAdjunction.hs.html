    4 {-|
    5 Module      : Test.Suite.SelfAdjunction
    6 Description : Property tests for the Phase-3 @Holistic ⊣ Formal@ adjunction.
    7 
    8 Verifies, by QuickCheck, the laws asserted in
    9 @docs\/adr\/0008-left-right-adjunction.md@ §10:
   10 
   11  * the hom-set isomorphism round-trips
   12    (@leftAdjunct@ ∘ @rightAdjunct@ ≡ id, and the reverse);
   13  * the left triangle identity on 'Holistic'
   14    (@counit . fmap unit@ ≡ id);
   15  * the right triangle identity on 'Formal'
   16    (@fmap counit . unit@ ≡ id, pointwise on probe fields);
   17  * value-level coherence of the derived combinators
   18    ('groundIn' / 'rebroaden');
   19  * the 'Functor' identity laws on both functors as a sanity check.
   20 
   21 Functions cannot be compared structurally in Haskell, so equality on
   22 'Formal' values is established pointwise: two formal commitments are
   23 considered equal here iff they agree on every member of a small,
   24 randomly-generated probe set of 'Field' values. Combined with
   25 parametricity, this is sufficient to verify the implementation is
   26 faithful to the categorical laws on the test family.
   27 -}

   77 selfAdjunctionTests :: [Test]
   78 selfAdjunctionTests =
   79   [ TestLabel "leftAdjunct ∘ rightAdjunct ≡ id (hom-set round-trip, l→r→l)" $
   80       quickCheckProperty "leftAdjunct . rightAdjunct ≡ id"  propLeftRightId
   81   , TestLabel "rightAdjunct ∘ leftAdjunct ≡ id (hom-set round-trip, r→l→r)" $
   82       quickCheckProperty "rightAdjunct . leftAdjunct ≡ id" propRightLeftId
   83   , TestLabel "Left triangle identity: counit . fmap unit ≡ id on Holistic" $
   84       quickCheckProperty "left triangle"  propLeftTriangle
   85   , TestLabel "Right triangle identity: fmap counit . unit ≡ id on Formal (pointwise)" $
   86       quickCheckProperty "right triangle" propRightTriangle
   87   , TestLabel "groundIn ≡ probe ∘ rebroaden (value-level coherence)" $
   88       quickCheckProperty "groundIn / rebroaden coherence" propGroundRebroaden
   89   , TestLabel "Functor identity on Holistic: fmap id ≡ id" $
   90       quickCheckProperty "Functor id on Holistic" propFunctorIdHolistic
   91   , TestLabel "Functor identity on Formal: fmap id ≡ id (pointwise)" $
   92       quickCheckProperty "Functor id on Formal" propFunctorIdFormal
   93   , TestLabel "arbitraryField smallest shrink is well-formed (regression lock)" $
   94       TestCase testArbitraryFieldSmallestShrinkIsWellFormed
   95   ]

  101 quickCheckProperty :: String -> Property -> Test
  102 quickCheckProperty label prop = TestCase $ do
  103   result <- quickCheckWithResult stdArgs { maxSuccess = 200 } prop
  104   if isSuccess result
  105     then pure ()
  106     else assertFailure ("Property failed: " ++ label)

  112 -- | A composite generator for 'Field': each component is drawn
  113 -- from its valid range and run through its smart constructor.
  114 -- 'fieldConfidence' is then re-derived so the generated value is
  115 -- internally consistent with the other four components.
  116 arbitraryField :: Gen Field
  117 arbitraryField = do
  118   r  <- choose (0.0, 1.0)
  119   v  <- choose (-1.0, 1.0)
  120   ar <- choose (0.0, 1.0)
  121   c  <- choose (0.0, 1.0)
  122   cf <- choose (0.0, 1.0)
  123   let scaffold =
  124         Field
  125           { fieldResonance      = mkResonance r
  126           , fieldAtmosphere     = mkAtmosphere v ar
  127           , fieldConfidence     = mkFieldConfidence 0.0
  128           , fieldConsolidation  = mkConsolidation c
  129           , fieldCounterfactual = mkCounterfactual cf
  130           }
  131       withConf = scaffold { fieldConfidence = deriveFieldConfidence scaffold }
  132   pure withConf

  134 -- | Single-scalar projection used by the affine test functions
  135 -- below. We use 'Resonance' as the canonical probe channel; any
  136 -- other component would do equally well — the laws under test are
  137 -- natural in 'Field' and do not depend on which scalar we pick.
  138 fieldProbeScalar :: Field -> Double
  139 fieldProbeScalar = unResonance . fieldResonance
  140 
  141 arbitraryInt :: Gen Int
  142 arbitraryInt = choose (-100, 100)
  143 
  144 arbitraryHolisticInt :: Gen (Holistic Int)
  145 arbitraryHolisticInt =
  146   (\a fd -> Holistic (a, fd)) <$> arbitraryInt <*> arbitraryField
  147 
  148 -- | A small, randomly-chosen set of probe fields used to compare two
  149 -- 'Formal' values for pointwise agreement.
  150 probeFields :: Gen [Field]
  151 probeFields = vectorOf 6 arbitraryField
  152 
  153 -- | Build a 'Formal' @Int@ from two integer seeds. The affine
  154 -- shape @\\(Field x) -> k + m * round x@ is sufficient to exercise
  155 -- the categorical laws because they are natural in the type
  156 -- parameter; the only structure being tested is the function
  157 -- shape itself.
  158 --
  159 -- We expose the seeds rather than the constructed function so
  160 -- QuickCheck can 'Show' counter-examples (function types have no
  161 -- meaningful 'Show').
  162 formalFromSeeds :: Int -> Int -> Formal Int
  163 formalFromSeeds k m = Formal (\fd -> k + m * round (fieldProbeScalar fd))
  164 
  165 -- | Pointwise equality of two 'Formal' values on a given probe set.
  166 formalEqOn :: Eq a => [Field] -> Formal a -> Formal a -> Bool
  167 formalEqOn probes f g =
  168   all (\fd -> probe f fd == probe g fd) probes
  169 
  170 -- ---------------------------------------------------------------------------
  171 -- Properties
  172 -- ---------------------------------------------------------------------------
  173 
  174 -- | Hom-set round-trip in the @(Holistic a -> b) → (a -> Formal b)@ direction:
  175 -- @rightAdjunct (leftAdjunct g) ≡ g@ for every @g :: Holistic a -> b@.
  176 --
  177 -- We sample @g@ from a family parameterised by an integer seed so that
  178 -- QuickCheck can drive it.
  179 propLeftRightId :: Property
  180 propLeftRightId =
  181   forAll arbitraryInt $ \seed ->
  182   forAll arbitraryHolisticInt $ \h ->
  183     let g :: Holistic Int -> Int
  184         g (Holistic (a, fd)) = a + seed + round (fieldProbeScalar fd)
  185         g' = rightAdjunct (leftAdjunct g)
  186     in g h == g' h
  187 
  188 -- | Hom-set round-trip in the @(a -> Formal b) → (Holistic a -> b)@ direction:
  189 -- @leftAdjunct (rightAdjunct k) ≡ k@ for every @k :: a -> Formal b@.
  190 --
  191 -- Equality on the @Formal@ output is checked pointwise on a probe set.
  192 propRightLeftId :: Property
  193 propRightLeftId =
  194   forAll arbitraryInt $ \seed ->
  195   forAll arbitraryInt $ \a ->
  196   forAll probeFields $ \probes ->
  197     let k :: Int -> Formal Int
  198         k x = Formal (\fd -> x * seed + round (fieldProbeScalar fd))
  199         k' :: Int -> Formal Int
  200         k' = leftAdjunct (rightAdjunct k)
  201     in formalEqOn probes (k a) (k' a)
  202 
  203 -- | Left triangle identity on @Holistic@:
  204 --
  205 -- @
  206 -- counit . fmap unit ≡ id        on Holistic
  207 -- @
  208 --
  209 -- Equality is structural since 'Holistic' has an 'Eq' instance.
  210 propLeftTriangle :: Property
  211 propLeftTriangle =
  212   forAll arbitraryHolisticInt $ \h ->
  213     counit (fmap unit h) == h
  214 
  215 -- | Right triangle identity on @Formal@:
  216 --
  217 -- @
  218 -- fmap counit . unit ≡ id        on Formal
  219 -- @
  220 --
  221 -- Tested pointwise on the probe set (function equality). The
  222 -- formal value is built from two integer seeds so QuickCheck can
  223 -- present a meaningful counter-example if the property fails.
  224 propRightTriangle :: Property
  225 propRightTriangle =
  226   forAll arbitraryInt $ \k ->
  227   forAll arbitraryInt $ \m ->
  228   forAll probeFields $ \probes ->
  229     let f       = formalFromSeeds k m
  230         stepped :: Formal (Holistic (Formal Int))
  231         stepped = unit f
  232         lhs :: Formal Int
  233         lhs     = fmap counit stepped
  234     in formalEqOn probes lhs f
  235 
  236 -- | Value-level coherence between 'groundIn' and 'rebroaden':
  237 --
  238 -- @groundIn (Holistic (a, fd)) ≡ probe (rebroaden a) fd@
  239 --
  240 -- Both sides equal @a@ by definition; the property guards against
  241 -- accidental drift of either combinator from this contract.
  242 propGroundRebroaden :: Property
  243 propGroundRebroaden =
  244   forAll arbitraryInt $ \a ->
  245   forAll arbitraryField $ \fd ->
  246     groundIn (Holistic (a, fd)) == probe (rebroaden a) fd
  247 
  248 -- | Sanity: 'Functor' identity law on 'Holistic'.
  249 propFunctorIdHolistic :: Property
  250 propFunctorIdHolistic =
  251   forAll arbitraryHolisticInt $ \h ->
  252     fmap id h == h
  253 
  254 -- | Sanity: 'Functor' identity law on 'Formal' (pointwise).
  255 propFunctorIdFormal :: Property
  256 propFunctorIdFormal =
  257   forAll arbitraryInt $ \k ->
  258   forAll arbitraryInt $ \m ->
  259   forAll probeFields $ \probes ->
  260     let f = formalFromSeeds k m
  261     in formalEqOn probes (fmap id f) f
  262 
  263 -- | F3-lock (regression): 'arbitraryField' must always produce a
  264 -- well-formed Field with finite components and confidence in [0,1].
  265 -- If the generator still contains 'error "unimplemented"', this
  266 -- test crashes immediately — catching the regression without
  267 -- dependence on QuickCheck seeds.
  268 testArbitraryFieldSmallestShrinkIsWellFormed :: IO ()
  269 testArbitraryFieldSmallestShrinkIsWellFormed = do
  270   fd <- generate arbitraryField
  271   let r  = unResonance (fieldResonance fd)
  272       v  = atmosphereValence (fieldAtmosphere fd)
  273       a  = atmosphereArousal (fieldAtmosphere fd)
  274       c  = unFieldConfidence (fieldConfidence fd)
  275       co = unConsolidation (fieldConsolidation fd)
  276       cf = unCounterfactual (fieldCounterfactual fd)
  277   assertBool "resonance must be finite" (not (isNaN r) && not (isInfinite r))
  278   assertBool "atmosphere valence must be finite" (not (isNaN v) && not (isInfinite v))
  279   assertBool "atmosphere arousal must be finite" (not (isNaN a) && not (isInfinite a))
  280   assertBool "confidence must be finite" (not (isNaN c) && not (isInfinite c))
  281   assertBool "confidence must be in [0,1]" (c >= 0.0 && c <= 1.0)
  282   assertBool "consolidation must be finite" (not (isNaN co) && not (isInfinite co))
  283   assertBool "counterfactual must be finite" (not (isNaN cf) && not (isInfinite cf))
