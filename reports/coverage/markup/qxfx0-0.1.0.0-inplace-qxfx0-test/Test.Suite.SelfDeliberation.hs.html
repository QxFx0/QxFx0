   94 quickCheckProperty :: String -> Property -> Test
   95 quickCheckProperty label prop = TestCase $ do
   96   result <- quickCheckWithResult stdArgs { maxSuccess = 100 } prop
   97   if isSuccess result
   98     then pure ()
   99     else assertFailure ("Property failed: " ++ label)

  105 arbitraryField :: Gen Field
  106 arbitraryField = do
  107   r  <- choose (0.0, 1.0)
  108   v  <- choose (-1.0, 1.0)
  109   ar <- choose (0.0, 1.0)
  110   c  <- choose (0.0, 1.0)
  111   cf <- choose (0.0, 1.0)
  112   pure $ Field
  113     { fieldResonance      = mkResonance r
  114     , fieldAtmosphere     = mkAtmosphere v ar
  115     , fieldConfidence     = mkFieldConfidence c
  116     , fieldConsolidation  = mkConsolidation cf
  117     , fieldCounterfactual = mkCounterfactual cf

  120 arbitraryPlan :: Gen Plan
  121 arbitraryPlan = do
  122   family  <- elements [CMGround, CMDefine, CMDistinguish, CMReflect, CMDescribe
  123                       , CMPurpose, CMHypothesis, CMRepair, CMContact, CMAnchor
  124                       , CMClarify, CMDeepen, CMConfront, CMNextStep]
  125   style   <- elements [StyleFormal, StyleWarm, StyleDirect, StylePoetic
  126                       , StyleClinical, StyleCautious, StyleRecovery, StyleStandard]
  127   mCause  <- elements [Nothing, Just RecoveryLowLegitimacy
  128                       , Just RecoveryParserLowConfidence
  129                       , Just RecoveryConatusGate]
  130   tone    <- elements [NarrativeNeutral, NarrativeWarm, NarrativeFormal
  131                       , NarrativeTerse, NarrativeRecovery]
  132   conf    <- choose (0.0, 1.0)
  133   pure $ Plan
  134     { planFamily        = family
  135     , planRenderStyle   = style
  136     , planRecoveryCause = mCause
  137     , planNarrativeTone = tone
  138     , planConfidence    = conf
  139     }

  185 -- B.6.1
  186 propConatusOverridesEverything :: Property
  187 propConatusOverridesEverything =
  188   forAll arbitraryPlan $ \hPlan ->
  189   forAll arbitraryPlan $ \fPlan ->
  190   forAll arbitraryField $ \fd ->
  191   forAll arbitrarySalience $ \salience ->
  192     let conatusSalience = salience { salienceDriver = DrivenByConatusGate }
  193         hp = holisticProposal hPlan fd
  194         fp = formalProposal (\_ -> fPlan)
  195         d  = reconcile Nothing conatusSalience hp fp fd
  196         r  = delibReconciled d
  197         t  = delibTrace d
  198      in planRenderStyle r == StyleRecovery
  199         && planRecoveryCause r == Just RecoveryConatusGate
  200         && planNarrativeTone r == NarrativeRecovery
  201         && planConfidence r == 1.0
  202         && dtRule t == RuleConatusOverride
  203 
  204 -- B.6.2 — when both proposals are identical and the salience
  205 -- driver is not 'DrivenByConatusGate' (which legitimately
  206 -- rewrites the plan into recovery shape), 'reconcile' returns
  207 -- the same plan and tags 'RuleAgreement' with 'Agree'.
  208 propAgreementIsIdempotent :: Property
  209 propAgreementIsIdempotent =
  210   forAll arbitraryPlan $ \plan ->
  211   forAll arbitraryField $ \fd ->
  212   forAll arbitraryNonConatusSalience $ \salience ->
  213     let hp = holisticProposal plan fd
  214         fp = formalProposal (\_ -> plan)
  215         d  = reconcile Nothing salience hp fp fd
  216         r  = delibReconciled d
  217         t  = delibTrace d
  218      in r == plan
  219         && dtAgreement t == Agree
  220         && dtRule t == RuleAgreement
  221 
  222 -- B.6.3
  223 propRecoveryNeverSilenced :: Property
  224 propRecoveryNeverSilenced =
  225   forAll arbitraryPlan $ \hPlan ->
  226   forAll arbitraryPlan $ \fPlan ->
  227   forAll arbitraryField $ \fd ->
  228   forAll arbitrarySalience $ \salience ->
  229     let hHasRecovery = planRecoveryCause hPlan /= Nothing
  230         fHasRecovery = planRecoveryCause fPlan /= Nothing
  231         hp = holisticProposal hPlan fd
  232         fp = formalProposal (\_ -> fPlan)
  233         d  = reconcile Nothing salience hp fp fd
  234         r  = delibReconciled d
  235      in (hHasRecovery || fHasRecovery)
  236         ==> (planRecoveryCause r /= Nothing)
  237 
  238 -- B.6.4
  239 propTiedFallbackIsFormal :: Property
  240 propTiedFallbackIsFormal =
  241   forAll arbitraryPlan $ \hPlan ->
  242   forAll arbitraryPlan $ \fPlan ->
  243   forAll arbitraryField $ \fd ->
  244     let salience = Salience
  245           { salienceHolisticBias = 0.5
  246           , salienceConfidence   = 0.0
  247           , salienceDriver       = DrivenByDefault
  248           }
  249         hp = holisticProposal hPlan fd
  250         fp = formalProposal (\_ -> fPlan)
  251         d  = reconcile Nothing salience hp fp fd
  252         r  = delibReconciled d
  253         t  = delibTrace d
  254      in dtRule t == RuleTiedFallback
  255         ==> (r == fPlan { planRecoveryCause = planRecoveryCause r })
  256 
  257 -- B.6.5
  258 propDivergenceBounded :: Property
  259 propDivergenceBounded =
  260   forAll arbitraryPlan $ \hPlan ->
  261   forAll arbitraryPlan $ \fPlan ->
  262   forAll arbitraryField $ \fd ->
  263   forAll arbitrarySalience $ \salience ->
  264     let hp = holisticProposal hPlan fd
  265         fp = formalProposal (\_ -> fPlan)
  266         d  = reconcile Nothing salience hp fp fd
  267         divg = dtDivergence (delibTrace d)
  268      in divg >= 0.0 && divg <= 1.0
  269 
  270 -- B.6.6
  271 propDeterminism :: Property
  272 propDeterminism =
  273   forAll arbitraryPlan $ \hPlan ->
  274   forAll arbitraryPlan $ \fPlan ->
  275   forAll arbitraryField $ \fd ->
  276   forAll arbitrarySalience $ \salience ->
  277     let hp = holisticProposal hPlan fd
  278         fp = formalProposal (\_ -> fPlan)
  279         d1 = reconcile Nothing salience hp fp fd
  280         d2 = reconcile Nothing salience hp fp fd
  281      in d1 == d2

  283 -- Extra: renderAgreement is total (no pattern-match failure)
  284 propRenderAgreementTotal :: Property
  285 propRenderAgreementTotal =
