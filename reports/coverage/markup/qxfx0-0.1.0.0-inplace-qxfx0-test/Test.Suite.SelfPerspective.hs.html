   66 testPerspectiveAdmissibilityRejectsInsufficientEvidence :: Test
   67 testPerspectiveAdmissibilityRejectsInsufficientEvidence = TestCase $ do
   68   let bundle = mkPerspectiveBundle [] [] [] defaultNormativeProfile []
   69       candidate = opinionCore bundle
   70   assertEqual "candidate without evidence must be inadmissible"
   71     (PerspectiveInadmissible "insufficient_evidence")
   72     (evaluatePerspectiveAdmissibility bundle candidate)

   84 testPerspectivePromotionRequiresStabilityAndStrongEvidence :: Test
   85 testPerspectivePromotionRequiresStabilityAndStrongEvidence = TestCase $ do
   86   let weakBundle = mkPerspectiveBundle [EvidenceRef "thanks_ack"] [] [] defaultNormativeProfile []
   87       weakCandidate = opinionCore weakBundle
   88       stableBundle = mkPerspectiveBundle
   89         [ EvidenceRef "knowledge:freedom is bounded agency"
   90         , EvidenceRef "dialogue:DialogueOutcomeSuccess:freedom"
   91         ]
   92         [ClaimStanceRef "stance:BeliefAffirmed:freedom requires responsibility"]
   93         []
   94         defaultNormativeProfile
   95         []
   96       stableCandidate = opinionCore stableBundle
   97       registry = defaultPerspectiveRegistry
   98   assertEqual "weak evidence can be admissible but must not promote"
   99     PpdObserveOnly
  100     (decidePerspectivePromotion registry weakBundle weakCandidate PerspectiveAdmissibleAccepted)
  101   assertEqual "stable corroborated candidate can be promoted"
  102     PpdPromoteEndorsed
  103     (decidePerspectivePromotion registry stableBundle stableCandidate (evaluatePerspectiveAdmissibility stableBundle stableCandidate))

  141 testPerspectiveRollbackRestoresPriorProjection :: Test
  142 testPerspectiveRollbackRestoresPriorProjection = TestCase $ do
  143   let bundle = mkPerspectiveBundle
  144         [EvidenceRef "knowledge:freedom is bounded agency", EvidenceRef "dialogue:DialogueOutcomeSuccess:freedom"]
  145         [ClaimStanceRef "stance:BeliefAffirmed:freedom requires responsibility"]
  146         []
  147         defaultNormativeProfile
  148         []
  149       candidate = opinionCore bundle
  150       registry1 = applyPerspectiveDecision 1 defaultPerspectiveRegistry bundle candidate PpdPromoteEndorsed
  151       bundle2 = bundle { pibRevisionLineage = maybe [] ptRevisionHistory (M.lookup (pibScope bundle) (prThreads registry1)) }
  152       candidate2 = (opinionCore bundle2) { pcThesis = "revision one" }
  153       registry2 = applyPerspectiveDecision 2 registry1 bundle2 candidate2 PpdReviseActive
  154       rollbackBundle = bundle2 { pibCounterarguments = [CounterargumentRef "counter:a", CounterargumentRef "counter:b"] }
  155       rollbackCandidate = (opinionCore rollbackBundle) { pcCounterargumentPressure = 0.50 }
  156       decision = decidePerspectivePromotion registry2 rollbackBundle rollbackCandidate PerspectiveAdmissibleQuarantined
  157       registry3 = applyPerspectiveDecision 3 registry2 rollbackBundle rollbackCandidate decision
  158   assertEqual "rollback must be reachable from promotion policy"
  159     PpdRollbackPrior decision
  160   case buildPerspectiveProjection registry3 (pibScope bundle) of
  161     Nothing -> assertFailure "expected restored prior projection"
  162     Just projection -> assertBool "rollback must not expose withdrawn active version"
  163       (ppSummary projection /= "revision one")

  165 testPerspectiveProjectionIsExplainableAndSafe :: Test
  166 testPerspectiveProjectionIsExplainableAndSafe = TestCase $ do
  167   let bundle = mkPerspectiveBundle
  168         [ EvidenceRef "knowledge:freedom is bounded agency"
  169         , EvidenceRef "dialogue:DialogueOutcomeSuccess:freedom"
  170         ]
  171         [ClaimStanceRef "stance:BeliefAffirmed:freedom requires responsibility"]
  172         [CounterargumentRef "counter:freedom can conflict with safety"]
  173         defaultNormativeProfile
  174         []
  175       candidate = opinionCore bundle
  176       registry = applyPerspectiveDecision 1 defaultPerspectiveRegistry bundle candidate PpdPromoteEndorsed
  177   case buildPerspectiveProjection registry (pibScope bundle) of
  178     Nothing -> assertFailure "expected safe perspective projection"
  179     Just projection -> do
  180       assertEqual "projection exposes profile provenance"
  181         1 (ppNormativeProfileVersion projection)
  182       assertBool "projection exposes explanation handle"
  183         (not (T.null (ppExplanationHandle projection)))
  184       assertEqual "projection keeps counterarguments as counts only"
  185         1 (ppCounterargumentCount projection)

  187 -- | Regression lock G1: 'opinionCore' must not propagate NaN or
  188 -- Infinity from a degenerate 'ConatusSlice' into the candidate's
  189 -- confidence, counterargument pressure, normative alignment, or
  190 -- internal tension surfaces.  The NaN-safe 'clampUnit' in
  191 -- "QxFx0.Self.Perspective" guarantees a finite candidate at the
  192 -- perspective-evaluation boundary.
  193 testPerspectiveCandidateIsNaNSafe :: Test
  194 testPerspectiveCandidateIsNaNSafe = TestCase $ do
  195   let bundle0 = mkPerspectiveBundle
  196         [EvidenceRef "knowledge:freedom is bounded agency"]
  197         [ClaimStanceRef "stance:BeliefAffirmed:freedom requires responsibility"]
  198         [CounterargumentRef "counter:contested"]
  199         defaultNormativeProfile
  200         []
  201       bundle = bundle0
  202         { pibConatusSlice = (pibConatusSlice bundle0)
  203             { csEnergy          = 0/0
  204             , csFieldConfidence = 1/0
  205             , csStability       = (-1)/0
  206             }
  207         }
  208       candidate = opinionCore bundle
  209   assertBool "candidate confidence must not be NaN under non-finite Conatus slice"
  210     (not (isNaN (pcConfidence candidate)))
  211   assertBool "candidate confidence must not be Infinity under non-finite Conatus slice"
  212     (not (isInfinite (pcConfidence candidate)))
  213   assertBool "candidate counterargument pressure must remain finite"
  214     (not (isNaN (pcCounterargumentPressure candidate)) && not (isInfinite (pcCounterargumentPressure candidate)))
  215   assertBool "candidate normative alignment must remain finite"
  216     (not (isNaN (pcNormativeAlignment candidate)) && not (isInfinite (pcNormativeAlignment candidate)))
  217   assertBool "candidate internal tension must remain finite"
  218     (not (isNaN (pcInternalTension candidate)) && not (isInfinite (pcInternalTension candidate)))
  219   assertBool "all candidate scores must stay bounded to the unit interval"
  220     ( pcConfidence candidate              >= 0.0 && pcConfidence candidate              <= 1.0
  221    && pcCounterargumentPressure candidate >= 0.0 && pcCounterargumentPressure candidate <= 1.0
  222    && pcNormativeAlignment candidate      >= 0.0 && pcNormativeAlignment candidate      <= 1.0
  223    && pcInternalTension candidate         >= 0.0 && pcInternalTension candidate         <= 1.0
  224     )
