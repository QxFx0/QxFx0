   28 vecPropertiesTests :: [Test]
   29 vecPropertiesTests =
   30   [ TestLabel "vec norm is non-negative" $
   31       quickCheckProperty "vecNorm non-negative" $
   32         forAll arbitraryCoreVec $ \v ->
   33           vecNorm v >= 0
   34   , TestLabel "clampVecNorm respects cap" $
   35       quickCheckProperty "clampVecNorm bounded" $
   36         forAll arbitraryCoreVec $ \v ->
   37           forAll (choose (0.01, 5.0)) $ \cap ->
   38             let clamped = clampVecNorm cap v
   39             in vecNorm clamped <= cap + 1e-8
   40   , TestLabel "vecAdd is commutative" $
   41       quickCheckProperty "vecAdd commutative" $
   42         forAll arbitraryCoreVec $ \a ->
   43           forAll arbitraryCoreVec $ \b ->
   44             vecAdd a b == vecAdd b a
   45   , TestLabel "vecSub cancels vecAdd" $
   46       quickCheckProperty "vecAdd vecSub identity" $
   47         forAll arbitraryCoreVec $ \v ->
   48           vecSub (vecAdd v zeroVec) v == zeroVec
   49   ]

   59 quickCheckProperty :: String -> Property -> Test
   60 quickCheckProperty label prop =
   61   TestCase $ do
   62     result <- quickCheckWithResult stdArgs { maxSuccess = 200 } prop
   63     unless (isSuccess result) $
   64       assertFailure (label <> ": QuickCheck failed")
   65   where
   66     unless p action = if p then pure () else action
