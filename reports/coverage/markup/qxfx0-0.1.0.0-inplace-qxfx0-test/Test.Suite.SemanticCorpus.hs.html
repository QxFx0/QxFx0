   51 semanticCorpusTests :: [Test]
   52 semanticCorpusTests =
   53   [ TestLabel "semantic corpus file has expected shape" (TestCase testSemanticCorpusShape)
   54   , TestLabel "semantic corpus P0/P1 pure invariants hold" (TestCase testSemanticCorpusPureInvariants)
   55   , TestLabel "semantic corpus NixGuard degradation is not semantic repair" (TestCase testNixGuardCorpusDiagnostics)

   58 testSemanticCorpusShape :: Assertion
   59 testSemanticCorpusShape = do
   60   corpusCases <- readSemanticCorpus
   61   assertEqual "semantic corpus case count" 110 (length corpusCases)
   62   assertEqual "P0 case count" 5 (tierCount "P0" corpusCases)
   63   assertEqual "P1 case count" 88 (tierCount "P1" corpusCases)
   64   assertEqual "P2 case count" 17 (tierCount "P2" corpusCases)
   65   forM_ corpusCases $ \c -> do
   66     assertBool (caseLabel c <> " has id") (not (T.null (sccId c)))
   67     assertBool (caseLabel c <> " has lang") (sccLang c `elem` ["ru", "en"])
   68     assertBool (caseLabel c <> " has class") (not (T.null (sccClass c)))
   69     assertBool (caseLabel c <> " requires replay envelope") (sccRequiresReplayEnvelope c)
   70 
   71 testSemanticCorpusPureInvariants :: Assertion
   72 testSemanticCorpusPureInvariants = do
   73   corpusCases <- readSemanticCorpus
   74   forM_ (filter isEnforcedTier corpusCases) $ \c -> do
   75     let frame = parseProposition (sccInput c)
   76         focus = normalizeFocus (firstNonEmpty [ipfFocusNominative frame, ipfFocusEntity frame])
   77         atoms = collectAtoms (sccInput c) semanticCorpusClusters
   78         families = map (textShow . fst) (runSemanticLogic atoms)
   79         canonicalFamily = textShow (ipfCanonicalFamily frame)
   80     forM_ (sccMustNotFocus c) $ \badFocus ->
   81       unless (T.null focus) $
   82         assertBool
   83           (caseLabel c <> " focus must not be " <> T.unpack badFocus <> "; actual=" <> T.unpack focus)
   84           (focus /= normalizeFocus badFocus)
   85     forM_ (sccMustNotFamily c) $ \badFamily ->
   86       assertBool
   87         (caseLabel c <> " semantic logic must not rank " <> T.unpack badFamily <> "; ranked=" <> show families)
   88         (badFamily `notElem` families)
   89     when (sccTier c == "P0" && not (null (sccExpectedFamilyAnyOf c))) $
   90       assertBool
   91         (caseLabel c <> " canonical family must stay in expected routing band; actual=" <> T.unpack canonicalFamily)
   92         (canonicalFamily `elem` sccExpectedFamilyAnyOf c)
   93 
   94 testNixGuardCorpusDiagnostics :: Assertion
   95 testNixGuardCorpusDiagnostics = do
   96   corpusCases <- readSemanticCorpus
   97   forM_ (filter ((== "P0") . sccTier) corpusCases) $ \c -> do
   98     let frame = parseProposition (sccInput c)
   99         concept = firstNonEmpty [ipfFocusNominative frame, ipfFocusEntity frame]
  100     status <- NixGuard.checkConstitution "semantics/concepts.nix" concept 0.0 0.0
  101     let rendered = T.toLower (textShow status)
  102     forM_ (sccMustNotGuardReasonContains c) $ \needle ->
  103       assertBool
  104         (caseLabel c <> " guard diagnostic must not contain " <> T.unpack needle <> "; status=" <> T.unpack rendered)
  105         (not (T.toLower needle `T.isInfixOf` rendered))
  106     case status of
  107       Blocked reason ->
  108         assertBool
  109           (caseLabel c <> " explicit policy block must not be caused by unsupported ordinary prose; reason=" <> T.unpack reason)
  110           (not ("unsupported" `T.isInfixOf` T.toLower reason))
  111       _ -> pure ()

  113 readSemanticCorpus :: IO [SemanticCorpusCase]
  114 readSemanticCorpus = do
  115   raw <- TIO.readFile corpusPath
  116   let nonEmptyLines = filter (not . T.null . T.strip) (T.lines raw)
  117   case traverse decodeLine (zip [(1 :: Int)..] nonEmptyLines) of
  118     Left err -> assertFailure err >> pure []
  119     Right corpusCases -> pure corpusCases
  120   where
  121     decodeLine (lineNo, lineText) =
  122       case eitherDecodeStrict' (TE.encodeUtf8 lineText) of
  123         Left err -> Left ("invalid semantic corpus JSONL line " <> show lineNo <> ": " <> err)
  124         Right c -> Right c
  125

  127 corpusPath = "test" </> "golden" </> "semantic_corpus.jsonl"

  188 isEnforcedTier :: SemanticCorpusCase -> Bool
  189 isEnforcedTier c = sccTier c `elem` ["P0", "P1"]
  190 
  191 tierCount :: Text -> [SemanticCorpusCase] -> Int
  192 tierCount tier = length . filter ((== tier) . sccTier)
  193 
  194 normalizeFocus :: Text -> Text
  195 normalizeFocus raw =
  196   case simpleTokens raw of
  197     (token:_) -> token
  198     [] -> T.toLower (T.strip raw)
  199 
  200 simpleTokens :: Text -> [Text]
  201 simpleTokens =
  202   filter (not . T.null)
  203     . T.words
  204     . T.map (\c -> if isAlphaNum c || c == '_' || c == '-' then c else ' ')
  205     . T.toLower
  206 
  207 firstNonEmpty :: [Text] -> Text
  208 firstNonEmpty = foldr (\candidate acc -> if T.null (T.strip candidate) then acc else candidate) ""
  209 
  210 caseLabel :: SemanticCorpusCase -> String
  211 caseLabel c = T.unpack (sccId c)
  212 
  213 textShow :: Show a => a -> Text
  214 textShow = T.pack . show
