   49 testGenerativeDiversity :: Test
   50 testGenerativeDiversity = TestCase $ do
   51   md <- loadMorphologyData
   52   let prompts =
   53         [ "скажи интересную мысль"
   54         , "а ещё одну мысль"
   55         , "скажи логичную мысль"

   57       rendered = map (renderForPrompt md) prompts
   58       uniqueRendered = Set.fromList rendered

   74 testTaggedFallbackReason :: Test
   75 testTaggedFallbackReason = TestCase $ do
   76   md <- loadMorphologyData
   77   let frame = emptyInputPropositionFrame
   78         { ipfRawText = "ты работаешь?"
   79         , ipfPropositionType = "OperationalStatusQ"
   80         , ipfFocusEntity = "система"
   81         , ipfSemanticSubject = "система"
   82         , ipfCanonicalFamily = CMGround
   83         }
   84       rmp0 = TurnPlanning.buildRMP CMGround emptyDialogueCommitmentLedger Exploring emptyDialogueThread frame emptySenseVector "система" emptyEgoState emptyAtomTrace True
   85       rcp = TurnPlanning.buildRCP CMGround rmp0
   86       rmpBroken = rmp0 { rmpPrimaryClaimAst = Just (ClaimPurpose "") }
   87       artifact = renderDialogueArtifact frame rmpBroken rcp "система" [] md
   88   assertBool "forced broken AST should trigger linearization fallback"
   89     (not (draLinearizationOk artifact))
   90   assertEqual "fallback reason should preserve exact branch tag"
   91     (Just "gf_linearization_failed:operational_status")
   92     (draFallbackReason artifact)

  110 assertPromptRenders :: MorphologyData -> Text -> Assertion
  111 assertPromptRenders md prompt = do
  112   let frame = Proposition.parseProposition prompt
  113       family = ipfCanonicalFamily frame
  114       topic = nonEmpty (ipfFocusEntity frame) "тема"
  115       rmp = TurnPlanning.buildRMP family emptyDialogueCommitmentLedger Exploring emptyDialogueThread frame emptySenseVector topic emptyEgoState emptyAtomTrace True
  116       rcp = TurnPlanning.buildRCP family rmp
  117       artifact = renderDialogueArtifact frame rmp rcp topic [] md
  118       rendered = draRenderedText artifact
  119   assertBool ("rendered text must be non-empty for prompt: " <> T.unpack prompt)
  120     (not (T.null (T.strip rendered)))
  121   assertBool ("structured prompts must avoid hard no-output fallback: " <> T.unpack prompt)
  122     (not (hasStructuredDialogueSurface frame && draFallbackReason artifact == Just "gf_no_output"))
  123 
  124 assertAssemblyNotEmpty :: MorphologyData -> Text -> Assertion
  125 assertAssemblyNotEmpty md prompt = do
  126   let frame = Proposition.parseProposition prompt
  127       family = ipfCanonicalFamily frame
  128       topic = nonEmpty (ipfFocusEntity frame) "тема"
  129       rmp = TurnPlanning.buildRMP family emptyDialogueCommitmentLedger Exploring emptyDialogueThread frame emptySenseVector topic emptyEgoState emptyAtomTrace True
  130       rcp = TurnPlanning.buildRCP family rmp
  131       artifact =
  132         renderArtifactViaAssembly
  133           emptyRuntimeParadigms
  134           emptySystemState
  135           frame
  136           rmp
  137           rcp
  138           topic
  139           []
  140           md
  141           (rcpStyle rcp)
  142           (emptyParsedInput prompt)
  143           Nothing
  144           Nothing
  145   assertBool ("assembly output must be non-empty for prompt: " <> T.unpack prompt)
  146     (not (T.null (T.strip (draRenderedText artifact))))
  147   assertBool ("assembly path must not end with gf_no_output for prompt: " <> T.unpack prompt)
  148     (draFallbackReason artifact /= Just "gf_no_output")

  150 renderForPrompt :: MorphologyData -> Text -> Text
  151 renderForPrompt md prompt =
  152   let frame = Proposition.parseProposition prompt
  153       family = ipfCanonicalFamily frame
  154       topic = nonEmpty (ipfFocusEntity frame) "тема"
  155       rmp = TurnPlanning.buildRMP family emptyDialogueCommitmentLedger Exploring emptyDialogueThread frame emptySenseVector topic emptyEgoState emptyAtomTrace True
  156       rcp = TurnPlanning.buildRCP family rmp
  157       artifact = renderDialogueArtifact frame rmp rcp topic [] md
  158   in T.strip (draRenderedText artifact)

  160 nonEmpty :: Text -> Text -> Text
  161 nonEmpty preferred fallback
  162   | T.null (T.strip preferred) = fallback
  163   | otherwise = preferred
