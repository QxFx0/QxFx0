   26 testRecordProposalSubmissionTracksQuarantine :: Test
   27 testRecordProposalSubmissionTracksQuarantine = TestCase $ do
   28   let proposalId = CalibrationId 7
   29       gs = recordProposalSubmission emptyGuardrailState 5 proposalId
   30   assertEqual "submission must record last proposal turn"
   31     5 (gsLastProposalTurn gs)
   32   assertEqual "submission must increment proposals in current window"
   33     1 (gsProposalsThisWindow gs)
   34   assertEqual "submission must quarantine the proposal id"
   35     [(5, proposalId)] (gsQuarantine gs)

   59 testGuardrailCircuitBreakerOpensAfterRejections :: Test
   60 testGuardrailCircuitBreakerOpensAfterRejections = TestCase $ do
   61   let gs0 = emptyGuardrailState { gsConsecutiveRejections = 2 }
   62       gs1 = recordRejection gs0 10
   63   assertEqual "rejection must increment consecutive rejection counter"
   64     3 (gsConsecutiveRejections gs1)
   65   assertEqual "threshold rejection must open cooldown window"
   66     15 (gsCooldownExpiry gs1)
   67   assertBool "circuit breaker must be open during cooldown"
   68     (not (canSubmitProposal gs1 11))

   99 testGuardrailStateRoundTripsThroughJson :: Test
  100 testGuardrailStateRoundTripsThroughJson = TestCase $ do
  101   let gs = GuardrailState
  102         { gsLastProposalTurn = 7

  104         , gsWindowStart = 3

  106         , gsCooldownExpiry = 15

  113 -- | Regression lock for ADR-0030 §5.2.  A previously shipped bug treated
  114 -- @gsCooldownExpiry = 0@ (the empty-state sentinel meaning "no active
  115 -- cooldown") as an open circuit on turn 0, blocking the first
  116 -- 'canSubmitProposal' call of every fresh session.  This test pins the
  117 -- contract that a fresh 'emptyGuardrailState' permits submissions on
  118 -- turn 0 and on any later turn while no cooldown has been recorded.
  119 testCanSubmitOnTurnZeroWithEmptyState :: Test
  120 testCanSubmitOnTurnZeroWithEmptyState = TestCase $ do
  121   assertBool "fresh guardrail state must allow submission on turn 0"
  122     (canSubmitProposal emptyGuardrailState 0)
  123   assertBool "fresh guardrail state must allow submission on a later turn"
  124     (canSubmitProposal emptyGuardrailState 100)
  125   assertBool "fresh guardrail state must allow submission far ahead in time"
  126     (canSubmitProposal emptyGuardrailState 1000000)
