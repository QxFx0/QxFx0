   44 -- | Approximate floating-point equality with a small absolute
   45 -- tolerance. Conatus arithmetic uses 'log' and divisions, so we do
   46 -- not expect bit-exact results but expect agreement to several
   47 -- decimal places.
   48 approxEqual :: Double -> Double -> Double -> Bool
   49 approxEqual tol x y = abs (x - y) <= tol
   50 
   51 eps :: Double
   52 eps = 1e-9
   53 
   54 -- | Baseline blanket used as the start of monotonicity comparisons.
   55 baseBlanket :: SelfBlanket
   56 baseBlanket = SelfBlanket
   57   { sbSessionId           = "demo"
   58   , sbMorphologyTotalSize = 10
   59   , sbIdentityClaimsCount = 2
   60   , sbTurnCount           = 5
   61   }
   62 
   63 selfConatusTests :: [Test]
   64 selfConatusTests =
   65   [ TestLabel "scalar equals sum of components (no violations)" $
   66       TestCase $ do
   67         let ce    = computeConatusEnergy baseBlanket []
   68             comps = ceComponents ce
   69             sumC  = ccMorphology comps + ccIdentity comps + ccTurns comps + ccPenalty comps
   70         assertBool
   71           ("scalar/component disagreement: scalar=" <> show (ceScalar ce)
   72             <> " sum=" <> show sumC)
   73           (approxEqual eps (ceScalar ce) sumC)
   74 
   75   , TestLabel "scalar equals sum of components (with violations)" $
   76       TestCase $ do
   77         let vs    = [BlanketEmptySession, BlanketTurnRegressed 3 1]
   78             ce    = computeConatusEnergy baseBlanket vs
   79             comps = ceComponents ce
   80             sumC  = ccMorphology comps + ccIdentity comps + ccTurns comps + ccPenalty comps
   81         assertBool
   82           "scalar must equal sum of all four components in the presence of violations"
   83           (approxEqual eps (ceScalar ce) sumC)
   84 
   85   , TestLabel "degenerate blanket (0,0,0) gives zero smooth scalar" $
   86       TestCase $ do
   87         let degenerate = baseBlanket
   88               { sbMorphologyTotalSize = 0
   89               , sbIdentityClaimsCount = 0
   90               , sbTurnCount           = 0
   91               }
   92             ce = computeConatusEnergy degenerate []
   93         assertBool
   94           ("degenerate scalar must equal 0, got " <> show (ceScalar ce))
   95           (approxEqual eps (ceScalar ce) 0)
   96 
   97   , TestLabel "scalar is monotone in morphology size" $
   98       TestCase $ do
   99         let smaller = baseBlanket { sbMorphologyTotalSize = 10 }
  100             larger  = baseBlanket { sbMorphologyTotalSize = 100 }
  101             ceS     = computeConatusEnergy smaller []
  102             ceL     = computeConatusEnergy larger  []
  103         assertBool
  104           "more morphology => greater Conatus"
  105           (ceScalar ceL > ceScalar ceS)
  106 
  107   , TestLabel "scalar is monotone in identity claims count" $
  108       TestCase $ do
  109         let smaller = baseBlanket { sbIdentityClaimsCount = 0 }
  110             larger  = baseBlanket { sbIdentityClaimsCount = 20 }
  111             ceS     = computeConatusEnergy smaller []
  112             ceL     = computeConatusEnergy larger  []
  113         assertBool
  114           "more identity claims => greater Conatus"
  115           (ceScalar ceL > ceScalar ceS)
  116 
  117   , TestLabel "scalar is monotone in turn count" $
  118       TestCase $ do
  119         let smaller = baseBlanket { sbTurnCount = 1 }
  120             larger  = baseBlanket { sbTurnCount = 200 }
  121             ceS     = computeConatusEnergy smaller []
  122             ceL     = computeConatusEnergy larger  []
  123         assertBool
  124           "more turns => greater Conatus"
  125           (ceScalar ceL > ceScalar ceS)
  126 
  127   , TestLabel "each violation drops scalar by exactly the violation weight" $
  128       TestCase $ do
  129         let clean      = computeConatusEnergy baseBlanket []
  130             withOne    = computeConatusEnergy baseBlanket [BlanketEmptySession]
  131             withTwo    = computeConatusEnergy baseBlanket
  132                           [BlanketEmptySession, BlanketEmptyMorphology]
  133             withThree  = computeConatusEnergy baseBlanket
  134                           [ BlanketEmptySession
  135                           , BlanketEmptyMorphology
  136                           , BlanketTurnRegressed 4 2
  137                           ]
  138             lam        = cwViolation defaultConatusWeights
  139         assertBool "one violation drops scalar by lambda"
  140           (approxEqual eps (ceScalar clean - ceScalar withOne) lam)
  141         assertBool "two violations drop scalar by 2*lambda"
  142           (approxEqual eps (ceScalar clean - ceScalar withTwo) (2 * lam))
  143         assertBool "three violations drop scalar by 3*lambda"
  144           (approxEqual eps (ceScalar clean - ceScalar withThree) (3 * lam))
  145 
  146   , TestLabel "violation penalty helper matches inline computation" $
  147       TestCase $ do
  148         let vs   = [BlanketEmptySession, BlanketIdentityErased 5 3]
  149             comp = ceComponents (computeConatusEnergy baseBlanket vs)
  150             via  = conatusViolationPenalty defaultConatusWeights vs
  151         assertBool
  152           "conatusViolationPenalty must agree with ccPenalty"
  153           (approxEqual eps via (ccPenalty comp))
  154 
  155   , TestLabel "violations do not affect gradient" $
  156       TestCase $ do
  157         let g  = computeConatusGradient baseBlanket
  158             -- Gradient does not take violations; this test
  159             -- documents the structural invariant that the gradient
  160             -- depends only on the blanket axes.
  161             allPositive = cgMorphology g > 0 && cgIdentity g > 0 && cgTurns g > 0
  162         assertBool
  163           "all gradient components must be strictly positive on a non-degenerate blanket"
  164           allPositive
  165 
  166   , TestLabel "gradient is strictly positive on the degenerate blanket" $
  167       TestCase $ do
  168         let degenerate = baseBlanket
  169               { sbMorphologyTotalSize = 0
  170               , sbIdentityClaimsCount = 0
  171               , sbTurnCount           = 0
  172               }
  173             g = computeConatusGradient degenerate
  174         assertBool
  175           "even the degenerate blanket has a strictly positive gradient (no recovery direction missing)"
  176           (cgMorphology g > 0 && cgIdentity g > 0 && cgTurns g > 0)
  177 
  178   , TestLabel "gradient exhibits diminishing returns in each axis" $
  179       TestCase $ do
  180         let small    = baseBlanket { sbMorphologyTotalSize = 1 }
  181             large    = baseBlanket { sbMorphologyTotalSize = 1000 }
  182             gSmall   = computeConatusGradient small
  183             gLarge   = computeConatusGradient large
  184         assertBool
  185           "gradient w.r.t. morphology must decrease as morphology grows"
  186           (cgMorphology gSmall > cgMorphology gLarge)
  187 
  188   , TestLabel "gradient magnitude is strictly positive" $
  189       TestCase $ do
  190         let g = computeConatusGradient baseBlanket
  191         assertBool
  192           "magnitude must be > 0 for any reachable blanket under default weights"
  193           (gradientMagnitude g > 0)
  194 
  195   , TestLabel "gradientNormalize produces a unit-magnitude vector" $
  196       TestCase $ do
  197         let g = computeConatusGradient baseBlanket
  198         case gradientNormalize g of
  199           Nothing ->
  200             assertFailure
  201               "default-weighted gradient must normalise; got Nothing"
  202           Just n  ->
  203             assertBool
  204               ("normalised gradient must have magnitude 1; got "
  205                 <> show (gradientMagnitude n))
  206               (approxEqual 1e-12 (gradientMagnitude n) 1.0)
  207 
  208   , TestLabel "gradientNormalize returns Nothing on a zero gradient" $
  209       TestCase $ do
  210         let zeroWeights = ConatusWeights 0 0 0 0
  211             g           = computeConatusGradientWith zeroWeights baseBlanket
  212         case gradientNormalize g of
  213           Nothing -> pure ()
  214           Just _  ->
  215             assertFailure "zero gradient must not be normalisable"
  216 
  217   , TestLabel "tuned weights produce proportional scalars" $
  218       TestCase $ do
  219         -- Doubling all smooth weights and zeroing the penalty must
  220         -- double the scalar of a violation-free blanket. This
  221         -- documents the linearity-in-weights property the recovery
  222         -- driver will eventually depend on.
  223         let w     = defaultConatusWeights
  224             w2    = ConatusWeights
  225                       (2 * cwMorphology w)
  226                       (2 * cwIdentity   w)
  227                       (2 * cwTurns      w)
  228                       0
  229             wZeroPen = w { cwViolation = 0 }
  230             ce1   = computeConatusEnergyWith wZeroPen baseBlanket []
  231             ce2   = computeConatusEnergyWith w2       baseBlanket []
  232         assertBool
  233           ("doubled smooth weights must double the scalar; got "
  234             <> show (ceScalar ce1) <> " and " <> show (ceScalar ce2))
  235           (approxEqual 1e-9 (ceScalar ce2) (2 * ceScalar ce1))
  236   ]
