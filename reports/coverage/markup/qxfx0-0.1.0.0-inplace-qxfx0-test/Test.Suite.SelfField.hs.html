  177 quickCheckProperty :: String -> Property -> Test
  178 quickCheckProperty label prop = TestCase $ do
  179   result <- quickCheckWithResult stdArgs { maxSuccess = 200 } prop
  180   if isSuccess result
  181     then pure ()
  182     else assertFailure ("Property failed: " ++ label)

  338 -- ---------------------------------------------------------------------------
  339 -- Field-level combinator commutativity
  340 -- ---------------------------------------------------------------------------
  341 
  342 -- | Strict structural equality is sufficient for commutativity:
  343 -- max, min, average and weighted-average with weight 0.5 are all
  344 -- commutative on Double, and the construction of 'Field' preserves
  345 -- this pointwise.
  346 propCombineMaximaCommutative :: Property
  347 propCombineMaximaCommutative =
  348   forAll arbitraryField $ \f1 ->
  349   forAll arbitraryField $ \f2 ->
  350     combineField CombineMaxima f1 f2 == combineField CombineMaxima f2 f1
  351 
  352 propCombineAverageCommutative :: Property
  353 propCombineAverageCommutative =
  354   forAll arbitraryField $ \f1 ->
  355   forAll arbitraryField $ \f2 ->
  356     combineField CombineAverage f1 f2 == combineField CombineAverage f2 f1
  357 
  358 -- | All non-confidence components of CombineAccumulate are
  359 -- commutative ('combineConsolidation' is commutative because
  360 -- addition is). 'combineFieldConfidence' is also commutative
  361 -- ('min'). Hence the whole field combine is commutative.
  362 propCombineAccumulateNonConfCommutative :: Property
  363 propCombineAccumulateNonConfCommutative =
  364   forAll arbitraryField $ \f1 ->
  365   forAll arbitraryField $ \f2 ->
  366     combineField CombineAccumulate f1 f2 == combineField CombineAccumulate f2 f1

  368 -- ---------------------------------------------------------------------------
  369 -- emptyField identity
  370 -- ---------------------------------------------------------------------------
  371 
  372 -- | For 'CombineMaxima', the four non-confidence components of
  373 -- 'emptyField' are zero (or identity-shaped for 'Atmosphere'),
  374 -- and combining with them leaves the other field's non-confidence
  375 -- components untouched. We check that explicitly per component.
  376 --
  377 -- 'fieldConfidence' is not preserved because 'emptyField' has
  378 -- 'FieldConfidence 1.0' and the combine takes 'min'; that is the
  379 -- documented behaviour and we simply do not assert on it.
  380 propEmptyFieldIdentityForMaxima :: Property
  381 propEmptyFieldIdentityForMaxima =
  382   forAll arbitraryField $ \f ->
  383     let merged = combineField CombineMaxima f emptyField
  384     in fieldResonance     merged == fieldResonance     f
  385        && fieldConsolidation  merged == fieldConsolidation  f
  386        && fieldCounterfactual merged == fieldCounterfactual f
  387        && atmosphereValence (fieldAtmosphere merged)
  388             == 0.5 * atmosphereValence (fieldAtmosphere f)
  389        && atmosphereArousal (fieldAtmosphere merged)
  390             == 0.5 * atmosphereArousal (fieldAtmosphere f)
  391 
  392 -- ---------------------------------------------------------------------------
  393 -- deriveFieldConfidence boundary cases

  396 -- | A field whose four scalarised components are all equal is
  397 -- maximally coherent and should receive 'FieldConfidence 1.0'.
  398 propUniformFieldMaxConfidence :: Property
  399 propUniformFieldMaxConfidence =
  400   forAll arbitraryUnitDouble $ \x ->
  401     let scaffold =
  402           Field
  403             { fieldResonance      = mkResonance x
  404             , fieldAtmosphere     = mkAtmosphere 0.0 x
  405             , fieldConfidence     = mkFieldConfidence 1.0
  406             , fieldConsolidation  = mkConsolidation x
  407             , fieldCounterfactual = mkCounterfactual x
  408             }
  409         FieldConfidence c = deriveFieldConfidence scaffold
  410     in abs (c - 1.0) < 1e-9

  423 -- | Sanity check that the Phase-3 'Holistic' \/ 'Formal'
  424 -- adjunction continues to compose against the substantive 'Field'.
  425 -- We exercise 'unit' followed by 'counit' on a Holistic value and
  426 -- expect to get back the original.
  427 propAdjunctionSmoke :: Property
  428 propAdjunctionSmoke =
  429   forAll arbitraryField $ \fd ->
  430   forAll (choose (-100, 100 :: Int)) $ \a ->
  431     let h        = Holistic (a, fd)
  432         rebuilt  = counit (fmap unit h)
  433     in h == rebuilt
