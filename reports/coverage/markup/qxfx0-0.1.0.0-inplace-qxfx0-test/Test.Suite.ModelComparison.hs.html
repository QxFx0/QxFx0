   38 -- | Build a mock table covering all first-word prefixes used by
   39 -- 'deterministicCorpus'.  Each row maps a prefix to a response.
   40 mkModelMockTable :: [(Text, Either ExternalQueryError Text)] -> MockTable
   41 mkModelMockTable overrides =
   42   let defaults =
   43         [ ("llm-augment", "NeedLexiconExtension", "что",        Right perfectBody)
   44         , ("llm-augment", "NeedLexiconExtension", "тема",       Right perfectBody)
   45         , ("llm-augment", "NeedLexiconExtension", "как",        Right perfectBody)
   46         , ("llm-augment", "NeedLexiconExtension", "Explore",    Right perfectBody)
   47         , ("llm-augment", "NeedLexiconExtension", "fail",       Right perfectBody)
   48         , ("llm-augment", "NeedLexiconExtension", "определение",Right perfectBody)
   49         , ("llm-augment", "NeedLexiconExtension", "What",       Right perfectBody)
   50         , ("llm-augment", "NeedLexiconExtension", "Explain",    Right perfectBody)
   51         , ("llm-augment", "NeedLexiconExtension", "Describe",   Right perfectBody)
   52         , ("llm-augment", "NeedLexiconExtension", "Analyze",    Right perfectBody)
   53         , ("llm-augment", "NeedLexiconExtension", "Discuss",    Right perfectBody)
   54         , ("llm-augment", "NeedLexiconExtension", "Evaluate",   Right perfectBody)
   55         , ("llm-augment", "NeedLexiconExtension", "Interpret",  Right perfectBody)
   56         ]
   57       applyOverride (prefix, result) table =
   58         map (\(t,n,p,r) -> if p == prefix then (t,n,p,result) else (t,n,p,r)) table
   59   in foldr applyOverride defaults overrides
   60

   78 mkTurnResult :: ModelId -> Text -> Maybe Text -> Maybe Int -> Either ExternalQueryError ExternalQueryResponse -> TurnResult
   79 mkTurnResult modelId status mReject mGraft outcome =
   80   TurnResult
   81     { trModelId     = modelId
   82     , trPromptIndex = 1
   83     , trOutcome     = outcome
   84     , trLatencyMs   = 0
   85     , trTelemetry   = emptyLearningTelemetry
   86         { ltValidationStatus = status
   87         , ltRejectReason     = mReject
   88         , ltGraftTurn        = mGraft
   89         }
   90     }

  153 testAggregateSession :: Test
  154 testAggregateSession = TestCase $ do
  155   let results =
  156         [ mkTurnResult "m" "accept" Nothing (Just 1) (Right (ExternalQueryResponse "" "" "" 0))
  157         , mkTurnResult "m" "accept" Nothing (Just 2) (Right (ExternalQueryResponse "" "" "" 0))
  158         , mkTurnResult "m" "transport_error" (Just "network") Nothing (Left (EqeServerError "mock"))
  159         , mkTurnResult "m" "invalid_response" (Just "parser") Nothing (Right (ExternalQueryResponse "" "" "" 0))
  160         , mkTurnResult "m" "validation_reject" (Just "conflict") Nothing (Right (ExternalQueryResponse "" "" "" 0))
  161         , mkTurnResult "m" "sandbox_reject" (Just "degrading_conatus") Nothing (Right (ExternalQueryResponse "" "" "" 0))
  162         , mkTurnResult "m" "accept" Nothing (Just 7) (Right (ExternalQueryResponse "" "" "" 0))
  163         ]
  164       so = aggregateSession "m" 1 results
  165   assertEqual "total turns" 7 (soTotalTurns so)
  166   assertEqual "successes" 3 (soSuccessCount so)
  167   assertEqual "transport errors" 1 (soTransportErrorCount so)
  168   assertEqual "parse rejects" 1 (soParseRejectCount so)
  169   assertEqual "validation rejects" 1 (soValidationRejectCount so)
  170   assertEqual "sandbox rejects" 1 (soSandboxRejectCount so)
  171   assertEqual "sandbox accepts" 3 (soSandboxAcceptCount so)

  201 testComparisonRunThreeModels :: Test
  202 testComparisonRunThreeModels = TestCase $ do
  203   let models =
  204         [ ("glm",      MockTransport perfectMockTable   Nothing)
  205         , ("deepseek", MockTransport errorMockTable       Nothing)
  206         , ("kimi",     MockTransport degradingMockTable   Nothing)
  207         ]
  208   run <- runComparison "test-run-001" Sequential models deterministicCorpus emptySystemState
  209   assertEqual "3 model outcomes" 3 (length (crModelOutcomes run))
  210   let outcomes = crModelOutcomes run
  211       glm = case filter ((== "glm") . moModelId) outcomes of
  212         []    -> error "testComparisonRunThreeModels: glm outcome not found"
  213         (x:_) -> x
  214       ds  = case filter ((== "deepseek") . moModelId) outcomes of
  215         []    -> error "testComparisonRunThreeModels: deepseek outcome not found"
  216         (x:_) -> x
  217       km  = case filter ((== "kimi") . moModelId) outcomes of
  218         []    -> error "testComparisonRunThreeModels: kimi outcome not found"
  219         (x:_) -> x
  220   assertEqual "glm perfect success rate" 1.0 (moAvgSuccessRate glm)
  221   assertEqual "glm zero incidents" 0 (moTotalIncidents glm)
  222   assertBool "deepseek has reduced success rate" (moAvgSuccessRate ds < 1.0)
  223   assertEqual "deepseek validator accept rate still 1.0 (non-transport responses pass)"
  224     1.0 (moAvgValidatorAcceptRate ds)
  225   assertEqual "kimi sandbox pass rate 0.0 (all degraded)" 0.0 (moAvgSandboxPassRate km)
  226   assertBool "kimi has incidents" (moTotalIncidents km > 0)

  228 testInterleavedCountsMatch :: Test
  229 testInterleavedCountsMatch = TestCase $ do
  230   let transport = MockTransport perfectMockTable Nothing
  231   (_ssSeq, seqResults) <- runModelSession 1 "m" transport deterministicCorpus emptySystemState
  232   interleaved <- runInterleavedSession 1 [("m", transport)] deterministicCorpus emptySystemState
  233   let intResults = case lookup "m" interleaved of Just r -> r; Nothing -> []
  234   assertEqual "interleaved count matches sequential" (length seqResults) (length intResults)
  235   assertEqual "interleaved accept count matches sequential"
  236     (length (filter ((== "accept") . ltValidationStatus . trTelemetry) seqResults))
  237     (length (filter ((== "accept") . ltValidationStatus . trTelemetry) intResults))
