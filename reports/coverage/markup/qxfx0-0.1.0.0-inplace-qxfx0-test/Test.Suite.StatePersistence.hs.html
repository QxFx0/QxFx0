  315 injectLegacyNumericTrajectoryHash :: Value -> Value
  316 injectLegacyNumericTrajectoryHash value =
  317   case value of
  318     Object root ->
  319       Object (updateKeyMap (Key.fromText "essence") updateEssence root)
  320     _ -> value
  321   where
  322     updateEssence essenceVal =
  323       case essenceVal of
  324         Object essenceObj ->
  325           Object (updateKeyMap (Key.fromText "contents") updateCommitmentContents essenceObj)
  326         _ -> essenceVal
  327     updateCommitmentContents contentsVal =
  328       case contentsVal of
  329         Array xs
  330           | length xs >= 2 ->
  331               let updated = xs V.// [(1, updateCommitment (xs V.! 1))]
  332               in Array updated
  333         _ -> contentsVal
  334     updateCommitment commitmentVal =
  335       case commitmentVal of
  336         Object commitmentObj ->
  337           Object (KeyMap.insert (Key.fromText "ecWitnessHash") (Number 0) commitmentObj)
  338         _ -> commitmentVal
  339 
  340 updateKeyMap :: Key.Key -> (Value -> Value) -> KeyMap.KeyMap Value -> KeyMap.KeyMap Value
  341 updateKeyMap key f km =
  342   case KeyMap.lookup key km of
  343     Just current -> KeyMap.insert key (f current) km
  344     Nothing -> km

  347 injectNonAuthoritativeTruthContract value =
  348   case value of
  349     Object root -> Object (KeyMap.insert (Key.fromText "truthContractStatus") (String "LegacyIncompleteSurface") root)
  350     _ -> value

  352 writeNonAuthoritativePersistedTruthBlob :: Runtime.RuntimeContext -> T.Text -> IO ()
  353 writeNonAuthoritativePersistedTruthBlob rt sessionId =
  354   Runtime.withRuntimeDb rt $ \db -> do
  355     let selectSql = "SELECT value FROM dialogue_state WHERE session_id = ? AND key = ?"
  356     mSelect <- NSQL.prepare db selectSql
  357     selectStmt <- case mSelect of
  358       Left err -> assertFailure ("failed to prepare non-authoritative persisted-state select: " <> T.unpack err) >> fail "unreachable"
  359       Right stmt -> pure stmt
  360     _ <- NSQL.bindText selectStmt 1 sessionId
  361     _ <- NSQL.bindText selectStmt 2 "__system_state__"
  362     hasRow <- NSQL.stepRow selectStmt
  363     blob <- if hasRow then NSQL.columnText selectStmt 0 else assertFailure "missing persisted state blob for non-authoritative fixture" >> fail "unreachable"
  364     NSQL.finalize selectStmt
  365     mutatedBlob <-
  366       case (Aeson.decode (BL.fromStrict (encodeUtf8 blob)) :: Maybe Value) of
  367         Nothing -> assertFailure "failed to decode persisted state JSON for non-authoritative fixture" >> fail "unreachable"
  368         Just value -> pure (TE.decodeUtf8 . BL.toStrict . Aeson.encode $ injectNonAuthoritativeTruthContract value)
  369     let updateSql = "INSERT OR REPLACE INTO dialogue_state(session_id, key, value, updated_at) VALUES(?, ?, ?, datetime('now'))"
  370     mUpdate <- NSQL.prepare db updateSql
  371     updateStmt <- case mUpdate of
  372       Left err -> assertFailure ("failed to prepare non-authoritative persisted-state update: " <> T.unpack err) >> fail "unreachable"
  373       Right stmt -> pure stmt
  374     _ <- NSQL.bindText updateStmt 1 sessionId
  375     _ <- NSQL.bindText updateStmt 2 "__system_state__"
  376     _ <- NSQL.bindText updateStmt 3 mutatedBlob
  377     _ <- NSQL.step updateStmt
  378     NSQL.finalize updateStmt

  678 fetchLatestReplayTraceJson :: NSQL.Database -> T.Text -> IO T.Text

  680   mStmt <- NSQL.prepare db "SELECT replay_trace_json FROM turn_quality WHERE session_id = ? ORDER BY turn DESC LIMIT 1"

  682     Left err -> assertFailure ("Failed to prepare replay_trace_json query: " <> T.unpack err) >> fail "unreachable"
  683     Right s -> pure s
  684   _ <- NSQL.bindText stmt 1 sessionId

  687   NSQL.finalize stmt

  690 normalizeReplayTraceJson :: String -> T.Text -> IO Value
  691 normalizeReplayTraceJson label payload =
  692   case eitherDecodeStrict' (encodeUtf8 payload) of
  693     Left err -> assertFailure ("Failed to decode replay trace JSON for " <> label <> ": " <> err) >> fail "unreachable"
  694     Right value -> pure (normalizeReplayTraceValue value)
  695 
  696 normalizeReplayTraceValue :: Value -> Value
  697 normalizeReplayTraceValue (Object objectValue) =
  698   Object
  699     ( KeyMap.insert "trcSessionId" (String "<normalized-session>")
  700     $ KeyMap.insert "trcRequestId" (String "<normalized-request>") objectValue
  701     )
  702 normalizeReplayTraceValue other = other
