  139 testMigrationMatchesCanonicalSpec :: Test
  140 testMigrationMatchesCanonicalSpec = TestCase $ do
  141   root <- getCurrentDirectory
  142   schemaSql <- TIO.readFile (root </> "spec" </> "sql" </> "schema.sql")
  143   migrationV1Sql <- TIO.readFile (root </> "migrations" </> "001_initial_schema.sql")
  144   migrationV2Sql <- TIO.readFile (root </> "migrations" </> "002_turn_quality_trace_columns.sql")
  145   -- Cumulative migrations (001 + 002) must produce the same schema as spec/sql/schema.sql
  146   dbMigrationsPath <- freshTestDbPath "migration_check.db"
  147   dbCanonicalPath <- freshTestDbPath "canonical_check.db"
  148   dbMigrations <- NSQL.open dbMigrationsPath
  149   dbCanonical <- NSQL.open dbCanonicalPath
  150   case (dbMigrations, dbCanonical) of
  151     (Left err, _) -> assertFailure ("Cannot open migration check DB: " <> T.unpack err)
  152     (_, Left err) -> assertFailure ("Cannot open canonical check DB: " <> T.unpack err)
  153     (Right dbM, Right dbC) -> do
  154       _ <- NSQL.execSql dbM migrationV1Sql
  155       _ <- NSQL.execSql dbM migrationV2Sql
  156       _ <- NSQL.execSql dbC schemaSql
  157       migSig <- dumpSchemaSignature dbM
  158       canSig <- dumpSchemaSignature dbC
  159       NSQL.close dbM
  160       NSQL.close dbC
  161       let normCan = map (\(a,b,c,d) -> (a,b,c,normalizeSql d)) canSig
  162           normMig = map (\(a,b,c,d) -> (a,b,c,normalizeSql d)) migSig
  163       assertEqual "cumulative migrations must match canonical schema" normCan normMig
  164   where
  165     dumpSchemaSignature db = do
  166       mStmt <- NSQL.prepare db "SELECT type, name, tbl_name, sql FROM sqlite_master WHERE type IN ('table','index','trigger','view') AND name NOT LIKE 'sqlite_%' ORDER BY type, name"
  167       case mStmt of
  168         Left err -> assertFailure ("Prepare failed: " <> T.unpack err) >> fail "unreachable"
  169         Right stmt -> go stmt []
  170       where
  171         go stmt acc = do
  172           hasRow <- NSQL.stepRow stmt
  173           if not hasRow
  174             then do
  175               _ <- NSQL.finalize stmt
  176               pure (reverse acc)
  177             else do
  178               t <- NSQL.columnText stmt 0
  179               n <- NSQL.columnText stmt 1
  180               tbl <- NSQL.columnText stmt 2
  181               sql_ <- NSQL.columnText stmt 3
  182               go stmt ((T.strip t, T.strip n, T.strip tbl, T.strip sql_) : acc)
  183     normalizeSql :: T.Text -> T.Text
  184     normalizeSql = T.unwords . T.words
