   88 phaseM2dTests :: [Test]
   89 phaseM2dTests =
   90   [ -- Phase 5.5e — renderSalienceDriver coverage
   91     TestLabel "renderSalienceDriver covers all 7 variants with stable snake_case tags" $
   92       TestCase $ do
   93         renderSalienceDriver DrivenByResonance       @?= "resonance"
   94         renderSalienceDriver DrivenByAtmosphere      @?= "atmosphere"
   95         renderSalienceDriver DrivenByConsolidation   @?= "consolidation"
   96         renderSalienceDriver DrivenByCounterfactual  @?= "counterfactual"
   97         renderSalienceDriver DrivenByFieldConfidence @?= "field_confidence"
   98         renderSalienceDriver DrivenByConatusGate     @?= "conatus_gate"
   99         renderSalienceDriver DrivenByDefault         @?= "default"
  100 
  101     -- RecoveryConatusGate — text rendering and JSON tag
  102   , TestLabel "renderLocalRecoveryCause RecoveryConatusGate renders to conatus_gate" $
  103       TestCase $
  104         renderLocalRecoveryCause RecoveryConatusGate @?= "conatus_gate"
  105 
  106   , TestLabel "ToJSON RecoveryConatusGate produces String \"conatus_gate\"" $
  107       TestCase $
  108         toJSON RecoveryConatusGate @?= String "conatus_gate"
  109 
  110   , TestLabel "renderLocalRecoveryCause RecoveryRuntimeDegraded preserves runtime_degraded after split" $
  111       TestCase $
  112         renderLocalRecoveryCause RecoveryRuntimeDegraded @?= "runtime_degraded"
  113 
  114     -- Phase 2.5 (M2d) — psConatusEnergy invariant
  115   , TestLabel "buildPrepareEffectPlan stores psConatusEnergy matching direct computation" $
  116       TestCase $ do
  117         let plan       = buildPrepareEffectPlan viableSystemState "пробный ввод" testEpochZero
  118             blanket    = computeSelfBlanket viableSystemState
  119             violations = checkInitialBlanket blanket
  120             expected   = computeConatusEnergy blanket violations
  121         psConatusEnergy (pepStatic plan) @?= expected
  122 
  123   , TestLabel "buildPrepareEffectPlan threads conatusEnergy into PrepareReqConsciousness" $
  124       TestCase $ do
  125         let plan     = buildPrepareEffectPlan viableSystemState "пробный ввод" testEpochZero
  126             expected = psConatusEnergy (pepStatic plan)
  127         case pepConsciousnessRequest plan of
  128           PrepareReqConsciousness _ _ _ ce _ ->
  129             ce @?= expected
  130           _ ->
  131             assertFailure "pepConsciousnessRequest must be PrepareReqConsciousness"
  132 
  133   , TestLabel "buildPrepareEffectPlan threads conatusEnergy into PrepareReqIntuition" $
  134       TestCase $ do
  135         let plan     = buildPrepareEffectPlan viableSystemState "пробный ввод" testEpochZero
  136             expected = psConatusEnergy (pepStatic plan)
  137         case pepIntuitionRequest plan of
  138           PrepareReqIntuition _ _ _ _ ce _ _ ->
  139             ce @?= expected
  140           _ ->
  141             assertFailure "pepIntuitionRequest must be PrepareReqIntuition"
  142 
  143   , TestLabel "viable SystemState yields a healthy ConatusEnergy (gate does not fire)" $
  144       TestCase $ do
  145         let plan = buildPrepareEffectPlan viableSystemState "пробный ввод" testEpochZero
  146             ce   = psConatusEnergy (pepStatic plan)
  147         assertBool
  148           ("ceScalar should be >= 0 on a viable state, got " ++ show (ceScalar ce))
  149           (ceScalar ce >= 0.0)
  150   ]
  151 
  152 testEpochZero :: UTCTime
  153 testEpochZero = UTCTime (ModifiedJulianDay 0) 0
