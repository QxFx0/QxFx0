   56 testObserveOnlyUpdatesTurnWithoutThread :: Test
   57 testObserveOnlyUpdatesTurnWithoutThread = TestCase $ do
   58   let scope = ScopeTopic "freedom"
   59       registry = applyDecision 9 PpdObserveOnly defaultPerspectiveRegistry scope "observed only"
   60   assertEqual "observe-only must not create a lineage thread"
   61     M.empty (prThreads registry)
   62   assertEqual "observe-only still records the last evaluated turn"
   63     9 (prLastUpdatedTurn registry)
   64   assertEqual "observe-only must not allocate a perspective id"
   65     1 (prNextPerspectiveOrdinal registry)
   66   assertEqual "observe-only must not allocate a version id"
   67     1 (prNextVersionOrdinal registry)
   68   assertEqual "observe-only must not expose a projection"
   69     Nothing (buildPerspectiveProjection registry scope)

   71 testPromotionCreatesCanonicalActiveThread :: Test
   72 testPromotionCreatesCanonicalActiveThread = TestCase $ do
   73   let scope = ScopeTopic "freedom"
   74       registry = applyDecision 1 PpdPromoteEndorsed defaultPerspectiveRegistry scope "initial promoted thesis"
   75   thread <- expectJust "expected promoted perspective thread" (M.lookup scope (prThreads registry))
   76   active <- expectJust "expected active endorsed perspective" (activeEndorsedPerspective thread)
   77   projection <- expectJust "expected active projection" (buildPerspectiveProjection registry scope)
   78   assertEqual "first promoted scope receives version 1"
   79     (Just (PerspectiveVersionId 1)) (ptActiveVersion thread)
   80   assertEqual "thread status is active"
   81     PerspectiveActive (ptStatus thread)
   82   assertEqual "active endorsed perspective is version 1"
   83     (PerspectiveVersionId 1) (epVersion active)
   84   assertEqual "active endorsed perspective status is active"
   85     PerspectiveActive (epStatus active)
   86   assertEqual "promotion records one revision entry"
   87     1 (length (ptRevisionHistory thread))
   88   assertEqual "projection points at the active version"
   89     (PerspectiveVersionId 1) (ppPerspectiveVersion projection)
   90   assertEqual "projection is scoped through the registry thread"
   91     scope (ppScope projection)
   92   assertEqual "perspective ordinal advances after first new thread"
   93     2 (prNextPerspectiveOrdinal registry)
   94   assertEqual "version ordinal advances after first endorsed version"
   95     2 (prNextVersionOrdinal registry)

   97 testRevisionMarksPriorActiveVersion :: Test
   98 testRevisionMarksPriorActiveVersion = TestCase $ do
   99   let scope = ScopeTopic "freedom"
  100       registry1 = applyDecision 1 PpdPromoteEndorsed defaultPerspectiveRegistry scope "initial promoted thesis"
  101       registry2 = applyDecision 2 PpdReviseActive registry1 scope "revision one"
  102   thread <- expectJust "expected revised perspective thread" (M.lookup scope (prThreads registry2))
  103   projection <- expectJust "expected revised active projection" (buildPerspectiveProjection registry2 scope)
  104   assertEqual "revision makes version 2 active"
  105     (Just (PerspectiveVersionId 2)) (ptActiveVersion thread)
  106   assertBool "prior version remains in lineage as revised"
  107     (any (hasVersionStatus (PerspectiveVersionId 1) PerspectiveRevised) (ptVersions thread))
  108   assertBool "new version is the only active version"
  109     (oneActiveVersion (ptVersions thread))
  110   assertEqual "projection exposes the revised thesis"
  111     "revision one" (ppSummary projection)
  112   assertEqual "revision history keeps newest revision first"
  113     [PerspectiveVersionId 2, PerspectiveVersionId 1] (map prrToVersion (ptRevisionHistory thread))

  115 testRevisionHistoryIsBounded :: Test
  116 testRevisionHistoryIsBounded = TestCase $ do
  117   let scope = ScopeTopic "freedom"
  118       cappedRegistry = defaultPerspectiveRegistry { prMaxRevisionsPerScope = 2 }
  119       registry = applyMany scope cappedRegistry
  120         [ (1, PpdPromoteEndorsed, "initial")
  121         , (2, PpdReviseActive, "revision one")
  122         , (3, PpdReviseActive, "revision two")
  123         , (4, PpdReviseActive, "revision three")
  124         ]
  125   thread <- expectJust "expected bounded revision thread" (M.lookup scope (prThreads registry))
  126   assertEqual "revision history respects registry cap"
  127     2 (length (ptRevisionHistory thread))
  128   assertEqual "bounded history keeps the two newest versions"
  129     [PerspectiveVersionId 4, PerspectiveVersionId 3] (map prrToVersion (ptRevisionHistory thread))

  149 testActiveCapSuspendsOlderScope :: Test
  150 testActiveCapSuspendsOlderScope = TestCase $ do
  151   let scopeA = ScopeTopic "freedom"
  152       scopeB = ScopeTopic "responsibility"
  153       cappedRegistry = defaultPerspectiveRegistry { prMaxActivePerspectives = 1 }
  154       registry1 = applyDecision 1 PpdPromoteEndorsed cappedRegistry scopeA "freedom thesis"
  155       registry2 = applyDecision 2 PpdPromoteEndorsed registry1 scopeB "responsibility thesis"
  156   threadA <- expectJust "expected older thread to remain present" (M.lookup scopeA (prThreads registry2))
  157   threadB <- expectJust "expected newer thread to remain present" (M.lookup scopeB (prThreads registry2))
  158   assertEqual "newer scope is the only active projection scope under cap"
  159     [scopeB] (activePerspectiveProjectionScopes registry2)
  160   assertEqual "older thread active pointer is cleared, not deleted"
  161     Nothing (ptActiveVersion threadA)
  162   assertEqual "older thread is suspended by active cap"
  163     PerspectiveSuspended (ptStatus threadA)
  164   assertEqual "newer thread remains active"
  165     (Just (PerspectiveVersionId 2)) (ptActiveVersion threadB)

  199 testProjectionCapsPublicSurface :: Test
  200 testProjectionCapsPublicSurface = TestCase $ do

  202       bundle = mkRichPerspectiveBundle scope

  205   thread <- expectJust "expected rich projection thread" (M.lookup scope (prThreads registry))

  207   projection <- expectJust "expected rich active projection" (buildPerspectiveProjection registry scope)

  210   assertEqual "projection evidence count reflects capped support surface"

  212   assertEqual "projection counterargument count reflects capped counterargument surface"

  214   assertEqual "endorsed support claims are capped in canonical registry"
  215     8 (length (epSupportingClaims active))
  216   assertEqual "endorsed counterarguments are capped in canonical registry"

  219 applyMany :: PerspectiveScope -> PerspectiveRegistry -> [(Int, PerspectivePromotionDecision, T.Text)] -> PerspectiveRegistry
  220 applyMany scope = foldl step
  221   where
  222     step registry (turn, decision, thesis) = applyDecision turn decision registry scope thesis
  223 
  224 applyDecision :: Int -> PerspectivePromotionDecision -> PerspectiveRegistry -> PerspectiveScope -> T.Text -> PerspectiveRegistry
  225 applyDecision turn decision registry scope thesis =
  226   let bundle = mkPerspectiveBundle scope (lineageForScope scope registry)
  227       candidate = (opinionCore bundle) { pcThesis = thesis }
  228   in applyPerspectiveDecision turn registry bundle candidate decision

  255 mkRichPerspectiveBundle :: PerspectiveScope -> PerspectiveInputBundle
  256 mkRichPerspectiveBundle scope = (mkPerspectiveBundle scope [])
  257   { pibEvidence = map (EvidenceRef . ("evidence-" <>) . T.pack . show) ([1..12] :: [Int])
  258   , pibStanceSlice = map (ClaimStanceRef . ("stance-" <>) . T.pack . show) ([1..12] :: [Int])
  259   , pibCounterarguments = map (CounterargumentRef . ("counter-" <>) . T.pack . show) ([1..12] :: [Int])

  262 lineageForScope :: PerspectiveScope -> PerspectiveRegistry -> [PerspectiveRevisionRecord]
  263 lineageForScope scope registry =
  264   maybe [] ptRevisionHistory (M.lookup scope (prThreads registry))

  266 hasVersionStatus :: PerspectiveVersionId -> PerspectiveStatus -> EndorsedPerspective -> Bool
  267 hasVersionStatus version status endorsed =
  268   epVersion endorsed == version && epStatus endorsed == status
  269 
  270 oneActiveVersion :: [EndorsedPerspective] -> Bool
  271 oneActiveVersion versions =
  272   length (filter ((== PerspectiveActive) . epStatus) versions) == 1
