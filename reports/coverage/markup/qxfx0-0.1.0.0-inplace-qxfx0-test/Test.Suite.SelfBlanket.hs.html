    3 {-|
    4 Module      : Test.Suite.SelfBlanket
    5 Description : Unit tests for the Phase-1 Self-layer (SelfBlanket invariants).

    8 'checkInitialBlanket', and 'checkBlanketTransition' over manually
    9 constructed 'SelfBlanket' values and a minimally-populated
   10 'SystemState'. No IO, no fixtures.

   14 module Test.Suite.SelfBlanket
   15   ( selfBlanketTests
   16   ) where
   17 
   18 import qualified Data.Map.Strict as Map
   19 import qualified Data.Text as T
   20 import Test.HUnit (Test (..), (@?=), assertBool)
   21 
   22 import QxFx0.Self.Blanket (computeSelfBlanket)
   23 import QxFx0.Self.Invariants
   24   ( checkBlanketTransition
   25   , checkInitialBlanket
   26   , renderBlanketViolations
   27   )
   28 import QxFx0.Self.Types

   38 -- | A minimally valid baseline blanket for transition tests.
   39 validBlanket :: SelfBlanket
   40 validBlanket = SelfBlanket
   41   { sbSessionId           = "demo"
   42   , sbMorphologyTotalSize = 100
   43   , sbIdentityClaimsCount = 0
   44   , sbTurnCount           = 0
   45   }
   46 
   47 -- | Populate the empty system state with the minimum needed for a
   48 -- valid self-blanket: a session id and a non-empty morphology.
   49 viableSystemState :: SystemState
   50 viableSystemState =
   51   let base = emptySystemState
   52       morph = MorphologyData
   53         (Map.singleton "о" "preposition")  -- one prepositional form
   54         Map.empty
   55         Map.empty
   56         Map.empty
   57    in base
   58         { ssSessionId  = "demo"
   59         , ssMorphology = morph
   60         }
   61 
   62 selfBlanketTests :: [Test]
   63 selfBlanketTests =
   64   [ TestLabel "computeSelfBlanket projects expected fields from SystemState" $
   65       TestCase $ do
   66         let sb = computeSelfBlanket viableSystemState
   67         sbSessionId sb           @?= "demo"
   68         sbMorphologyTotalSize sb @?= 1
   69         sbIdentityClaimsCount sb @?= 0
   70         sbTurnCount sb           @?= 0
   71 
   72   , TestLabel "checkInitialBlanket accepts a valid blanket" $
   73       TestCase $
   74         checkInitialBlanket validBlanket @?= []
   75 
   76   , TestLabel "checkInitialBlanket rejects an empty session id" $
   77       TestCase $ do
   78         let sb = validBlanket { sbSessionId = "" }
   79         checkInitialBlanket sb @?= [BlanketEmptySession]
   80 
   81   , TestLabel "checkInitialBlanket rejects empty morphology" $
   82       TestCase $ do
   83         let sb = validBlanket { sbMorphologyTotalSize = 0 }
   84         checkInitialBlanket sb @?= [BlanketEmptyMorphology]
   85 
   86   , TestLabel "checkInitialBlanket reports both violations when both fail" $
   87       TestCase $ do
   88         let sb = validBlanket
   89               { sbSessionId           = ""
   90               , sbMorphologyTotalSize = 0
   91               }
   92         checkInitialBlanket sb
   93           @?= [BlanketEmptySession, BlanketEmptyMorphology]
   94 
   95   , TestLabel "checkBlanketTransition accepts a benign turn increment" $
   96       TestCase $ do
   97         let prev = validBlanket
   98             cur  = validBlanket { sbTurnCount = 1 }
   99         checkBlanketTransition prev cur @?= []
  100 
  101   , TestLabel "checkBlanketTransition accepts identity-claim growth" $
  102       TestCase $ do
  103         let prev = validBlanket
  104             cur  = validBlanket { sbIdentityClaimsCount = 3, sbTurnCount = 1 }
  105         checkBlanketTransition prev cur @?= []
  106 
  107   , TestLabel "checkBlanketTransition rejects a session-id change" $
  108       TestCase $ do
  109         let prev = validBlanket
  110             cur  = validBlanket { sbSessionId = "other" }
  111         checkBlanketTransition prev cur
  112           @?= [BlanketSessionChanged "demo" "other"]
  113 
  114   , TestLabel "checkBlanketTransition rejects a turn-count regression" $
  115       TestCase $ do
  116         let prev = validBlanket { sbTurnCount = 5 }
  117             cur  = validBlanket { sbTurnCount = 3 }
  118         checkBlanketTransition prev cur
  119           @?= [BlanketTurnRegressed 5 3]
  120 
  121   , TestLabel "checkBlanketTransition rejects identity-claim erasure" $
  122       TestCase $ do
  123         let prev = validBlanket { sbIdentityClaimsCount = 4 }
  124             cur  = validBlanket { sbIdentityClaimsCount = 2 }
  125         checkBlanketTransition prev cur
  126           @?= [BlanketIdentityErased 4 2]
  127 
  128   , TestLabel "checkBlanketTransition reports compound violations together" $
  129       TestCase $ do
  130         let prev = validBlanket
  131               { sbTurnCount           = 5
  132               , sbIdentityClaimsCount = 4
  133               }
  134             cur  = SelfBlanket
  135               { sbSessionId           = "other"
  136               , sbMorphologyTotalSize = 0
  137               , sbIdentityClaimsCount = 2
  138               , sbTurnCount           = 3
  139               }
  140         let violations = checkBlanketTransition prev cur
  141         -- Order: initial-blanket checks first, then transition checks.
  142         violations @?=
  143           [ BlanketEmptyMorphology
  144           , BlanketSessionChanged "demo" "other"
  145           , BlanketTurnRegressed 5 3
  146           , BlanketIdentityErased 4 2
  147           ]
  148 
  149   , TestLabel "renderBlanketViolations produces a non-empty diagnostic" $
  150       TestCase $ do
  151         let rendered =
  152               renderBlanketViolations
  153                 [ BlanketEmptySession
  154                 , BlanketTurnRegressed 2 1
  155                 ]
  156         assertBool "rendering is non-empty"     (not (T.null rendered))
  157         assertBool "rendering mentions session" ("session" `T.isInfixOf` rendered)
  158         assertBool "rendering mentions turn"    ("turn"    `T.isInfixOf` rendered)
  159   ]
