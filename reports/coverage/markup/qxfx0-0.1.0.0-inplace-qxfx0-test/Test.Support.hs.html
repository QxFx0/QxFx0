   51 assertExec :: NSQL.Database -> FilePath -> T.Text -> IO ()
   52 assertExec db label sql = do
   53   result <- NSQL.execSql db sql
   54   case result of
   55     Left err -> assertFailure ("SQL apply failed for " <> label <> ": " <> T.unpack err)
   56     Right _ -> pure ()
   57 
   58 queryCount :: NSQL.Database -> T.Text -> IO Int
   59 queryCount db sql = do
   60   stmtRes <- NSQL.prepare db sql
   61   stmt <- case stmtRes of
   62     Left err -> assertFailure ("Prepare failed: " <> T.unpack err) >> fail "unreachable"
   63     Right s -> pure s
   64   hasRow <- NSQL.stepRow stmt
   65   count <- if hasRow then NSQL.columnInt stmt 0 else pure 0
   66   NSQL.finalize stmt
   67   pure count

   69 withFakeSouffle :: IO a -> IO a
   70 withFakeSouffle action = do
   71   oldPath <- lookupEnv "PATH"
   72   binDir <- freshTestPath "qxfx0_fake_souffle_bin"
   73   let cleanup = removeDirIfExists binDir
   74       scriptPath = binDir </> "souffle"

   76   bracket_ (createDirectoryIfMissing True binDir) cleanup $ do

  129 withRuntimeEnv :: FilePath -> IO a -> IO a
  130 withRuntimeEnv dbName action = do
  131   root <- getCurrentDirectory
  132   dbPath <- freshTestDbPath dbName
  133   stateDir <- freshTestPath (dbName <> ".state")
  134   bracket_
  135     (do
  136       mapM_ removeIfExists (runtimeArtifacts dbPath)
  137       removeDirIfExists stateDir)
  138     (do
  139       mapM_ removeIfExists (runtimeArtifacts dbPath)
  140       removeDirIfExists stateDir)
  141     $ withEnvVar "QXFX0_ROOT" (Just root) $
  142         withEnvVar "QXFX0_DB" (Just dbPath) $
  143           withEnvVar "QXFX0_STATE_DIR" (Just stateDir) $

  147 withStrictRuntimeEnv :: FilePath -> IO a -> IO a
  148 withStrictRuntimeEnv dbName action = do
  149   root <- getCurrentDirectory
  150   dbPath <- freshTestDbPath dbName
  151   witnessPath <- freshTestPath (dbName <> ".agda-witness.json")
  152   stateDir <- freshTestPath (dbName <> ".state")
  153   let cleanup = witnessPath : runtimeArtifacts dbPath
  154   bracket_
  155     (do
  156       mapM_ removeIfExists cleanup
  157       removeDirIfExists stateDir)
  158     (do
  159       mapM_ removeIfExists cleanup
  160       removeDirIfExists stateDir)
  161     $ withFakeSouffle $
  162         withFakeNixInstantiateOk $
  163           withEnvVar "QXFX0_ROOT" (Just root) $
  164             withEnvVar "QXFX0_DB" (Just dbPath) $
  165               withEnvVar "QXFX0_STATE_DIR" (Just stateDir) $
  166                 withEnvVar "QXFX0_RUNTIME_MODE" (Just "strict") $
  167                   withEnvVar "QXFX0_EMBEDDING_BACKEND" (Just "local-deterministic") $
  168                     withEnvVar "EMBEDDING_API_URL" Nothing $
  169                       withEnvVar "QXFX0_AGDA_WITNESS" (Just witnessPath) $ do
  170                         mAgda <- findExecutable "agda"
  171                         case mAgda of
  172                           Just _  -> do
  173                             _ <- Runtime.writeAgdaWitness
  174                             action
  175                           Nothing -> do
  176                             AW.writeStubAgdaWitness witnessPath
  177                             action

  179 withEnvVar :: String -> Maybe String -> IO a -> IO a
  180 withEnvVar key mValue action = do
  181   old <- lookupEnv key
  182   let apply = case mValue of
  183         Just value -> setEnv key value
  184         Nothing -> unsetEnv key
  185       restore = case old of
  186         Just value -> setEnv key value
  187         Nothing -> unsetEnv key
  188   bracket_ apply restore action

  190 removeIfExists :: FilePath -> IO ()
  191 removeIfExists filePath = do
  192   exists <- doesFileExist filePath
  193   if exists then removeFile filePath else pure ()
  194 
  195 removeDirIfExists :: FilePath -> IO ()
  196 removeDirIfExists dirPath = do
  197   exists <- doesDirectoryExist dirPath
  198   if exists then removePathForcibly dirPath else pure ()
  199 
  200 withCleanFiles :: [FilePath] -> IO a -> IO a
  201 withCleanFiles paths =
  202   bracket_
  203     (mapM_ removeIfExists paths)
  204     (mapM_ removeIfExists paths)
