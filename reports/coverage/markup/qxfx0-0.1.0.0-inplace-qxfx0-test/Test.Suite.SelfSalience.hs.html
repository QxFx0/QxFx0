  154 quickCheckProperty :: String -> Property -> Test
  155 quickCheckProperty label prop = TestCase $ do
  156   result <- quickCheckWithResult stdArgs { maxSuccess = 200 } prop
  157   if isSuccess result
  158     then pure ()
  159     else assertFailure ("Property failed: " ++ label)

  169 arbitraryValenceDouble = choose (-1.0, 1.0)

  171 -- | A 'Field' whose every component is drawn from its smart
  172 -- constructor, with @fieldConfidence@ re-derived for internal
  173 -- consistency.
  174 arbitraryField :: Gen Field
  175 arbitraryField = do
  176   r  <- arbitraryUnitDouble
  177   v  <- arbitraryValenceDouble
  178   ar <- arbitraryUnitDouble
  179   c  <- arbitraryUnitDouble
  180   cf <- arbitraryUnitDouble
  181   let scaffold = Field
  182         { fieldResonance      = mkResonance r
  183         , fieldAtmosphere     = mkAtmosphere v ar
  184         , fieldConfidence     = mkFieldConfidence 1.0
  185         , fieldConsolidation  = mkConsolidation c
  186         , fieldCounterfactual = mkCounterfactual cf
  187         }
  188   pure scaffold { fieldConfidence = deriveFieldConfidence scaffold }

  190 -- | A 'ConatusEnergy' whose @ceScalar@ is drawn from a wide
  191 -- positive range so the Conatus gate (default threshold @0.0@) is

  193 -- only reads @ceScalar@.

  253   forAll arbitraryHealthyConatus $ \ce ->

  272 -- ---------------------------------------------------------------------------
  273 -- Monotonicity along each rule direction
  274 --
  275 -- We check that increasing a positive-direction component does not
  276 -- /decrease/ the holistic bias, and that increasing a negative-
  277 -- direction component does not /increase/ it. These are
  278 -- non-strict inequalities because the sigmoid squash and
  279 -- floating-point arithmetic admit ties on near-equal raw scores.
  280 -- ---------------------------------------------------------------------------

  282 propMonotoneResonance :: Property
  283 propMonotoneResonance =
  284   forAll arbitraryHealthyConatus $ \ce ->
  285   forAll arbitraryField $ \f ->
  286   forAll arbitraryUnitDouble $ \lo ->
  287   forAll (choose (0.0, 1.0)) $ \delta ->
  288     let hi = min 1.0 (lo + delta)
  289         bLo = salienceHolisticBias (computeSalience defaultSalienceWeights ce (withResonance lo f))
  290         bHi = salienceHolisticBias (computeSalience defaultSalienceWeights ce (withResonance hi f))
  291     in bHi >= bLo - 1e-12

  349 -- | Build a 'Salience' with a chosen bias for verdict-band tests.
  350 synthSalience :: Double -> Salience
  351 synthSalience b = Salience
  352   { salienceHolisticBias = b
  353   , salienceConfidence   = 1.0
  354   , salienceDriver       = DrivenByDefault
  355   }

  357 isTied :: SalienceVerdict -> Bool
  358 isTied Tied = True
  359 isTied _    = False
  360 
  361 isPreferHolistic :: SalienceVerdict -> Bool
  362 isPreferHolistic (PreferHolistic _) = True
  363 isPreferHolistic _                  = False
  364 
  365 isPreferFormal :: SalienceVerdict -> Bool
  366 isPreferFormal (PreferFormal _) = True
  367 isPreferFormal _                = False
  368 
  369 propTiedBand :: Property
  370 propTiedBand =
  371   forAll (choose (-1.0, 1.0)) $ \scaled ->
  372     let t   = verdictThreshold defaultSalienceWeights
  373         b   = 0.5 + scaled * t                          -- in [0.5 - t, 0.5 + t]
  374         b'  = max 0.0 (min 1.0 b)
  375         v   = salienceVerdict defaultSalienceWeights (synthSalience b')
  376     in isTied v
  377 
  378 propPreferHolisticAboveBand :: Property
  379 propPreferHolisticAboveBand =
  380   forAll (choose (0.0, 1.0)) $ \jitter ->
  381     let t = verdictThreshold defaultSalienceWeights
  382         b = min 1.0 (0.5 + t + 1e-3 + jitter * (0.5 - t - 1e-3))
  383         v = salienceVerdict defaultSalienceWeights (synthSalience b)
  384     in isPreferHolistic v
  385 
  386 propPreferFormalBelowBand :: Property
  387 propPreferFormalBelowBand =
  388   forAll (choose (0.0, 1.0)) $ \jitter ->
  389     let t = verdictThreshold defaultSalienceWeights
  390         b = max 0.0 (0.5 - t - 1e-3 - jitter * (0.5 - t - 1e-3))
  391         v = salienceVerdict defaultSalienceWeights (synthSalience b)
  392     in isPreferFormal v

  394 -- ---------------------------------------------------------------------------
  395 -- chooseBranch dispatch
  396 --
  397 -- The two test branches are constructed so that their outputs are
  398 -- distinguishable: the holistic branch tags its output with @+1@,
  399 -- the formal branch tags it with @+100@ (the latter will reach
  400 -- the caller via @rightAdjunct@ inside @chooseBranch@).
  401 -- ---------------------------------------------------------------------------
  402 
  403 holisticBranchTag :: Holistic Int -> Int
  404 holisticBranchTag (Holistic (a, _fd)) = a + 1
  405 
  406 formalBranchTag :: Int -> Formal Int
  407 formalBranchTag a = Formal (\_fd -> a + 100)
  408 
  409 propDispatchHolistic :: Property
  410 propDispatchHolistic =
  411   forAll (choose (-50, 50 :: Int)) $ \magInput ->
  412   forAll arbitraryField $ \fd ->
  413   forAll (choose (1, 100 :: Int)) $ \a ->
  414     let mag = fromIntegral magInput / 100.0  -- magnitude (sign is irrelevant for verdict tag)
  415         v   = PreferHolistic (abs mag + 0.1)
  416         f   = chooseBranch v holisticBranchTag formalBranchTag
  417     in f (Holistic (a, fd)) == a + 1
  418 
  419 propDispatchFormal :: Property
  420 propDispatchFormal =
  421   forAll (choose (-50, 50 :: Int)) $ \magInput ->
  422   forAll arbitraryField $ \fd ->
  423   forAll (choose (1, 100 :: Int)) $ \a ->
  424     let mag = fromIntegral magInput / 100.0
  425         v   = PreferFormal (abs mag + 0.1)
  426         f   = chooseBranch v holisticBranchTag formalBranchTag
  427     in f (Holistic (a, fd)) == a + 100
  428 
  429 propDispatchTied :: Property
  430 propDispatchTied =
  431   forAll arbitraryField $ \fd ->
  432   forAll (choose (1, 100 :: Int)) $ \a ->
  433     let f = chooseBranch Tied holisticBranchTag formalBranchTag
  434     in f (Holistic (a, fd)) == a + 100

  478 -- | Atmosphere arousal has a non-negative weight, so increasing it

  488         bHi = salienceHolisticBias (computeSalience defaultSalienceWeights ce (withAtmosphereArousal hi f))
  489     in bHi >= bLo - 1e-12

  491 -- | Atmosphere valence is not read by the salience controller;
  492 -- changing it while holding arousal fixed leaves the verdict
  493 -- unchanged.
  494 propAtmosphereValenceIrrelevant :: Property
  495 propAtmosphereValenceIrrelevant =
  496   forAll arbitraryHealthyConatus $ \ce ->
  497   forAll arbitraryField $ \f ->
  498   forAll arbitraryValenceDouble $ \v1 ->
  499   forAll arbitraryValenceDouble $ \v2 ->
  500     let f1 = f { fieldAtmosphere = (fieldAtmosphere f) { atmosphereValence = v1 } }
  501         f2 = f { fieldAtmosphere = (fieldAtmosphere f) { atmosphereValence = v2 } }
  502         s1 = computeSalience defaultSalienceWeights ce f1
  503         s2 = computeSalience defaultSalienceWeights ce f2
  504     in s1 == s2

  528 arbitrarySalienceWeights :: Gen SalienceWeights
  529 arbitrarySalienceWeights = do
  530   wr  <- choose (0.0, 2.0)

  540 propAdaptSalienceIdentity :: Property
  541 propAdaptSalienceIdentity =
  542   forAll arbitrarySalienceWeights $ \w ->
  543     adaptSalienceWeights 0.0 w == w

  545 propAdaptSalienceClamp :: Property
  546 propAdaptSalienceClamp =
  547   forAll arbitrarySalienceWeights $ \w ->
  548   forAll (choose (-10.0, 10.0)) $ \signal ->
  549     let adapted = adaptSalienceWeights signal w
  550     in all (\x -> x >= 0.0 && x <= 2.0)
  551          [ weightResonance adapted
  552          , weightAtmosphere adapted
  553          , weightConsolidation adapted
  554          , weightCounterfactual adapted
  555          , weightFieldConfidence adapted
  556          ]
  557 
  558 propAdaptSalienceAntiDrift :: Property
  559 propAdaptSalienceAntiDrift =
  560   forAll arbitrarySalienceWeights $ \w ->
  561   forAll (choose (-10.0, 10.0)) $ \signal ->
  562     let adapted = adaptSalienceWeights signal w
  563         def = defaultSalienceWeights
  564         withinDrift x target = abs (x - target) <= 1.0 + 1e-12
  565     in withinDrift (weightResonance adapted)       (weightResonance def)
  566     && withinDrift (weightAtmosphere adapted)      (weightAtmosphere def)
  567     && withinDrift (weightConsolidation adapted)   (weightConsolidation def)
  568     && withinDrift (weightCounterfactual adapted)  (weightCounterfactual def)
  569     && withinDrift (weightFieldConfidence adapted) (weightFieldConfidence def)
