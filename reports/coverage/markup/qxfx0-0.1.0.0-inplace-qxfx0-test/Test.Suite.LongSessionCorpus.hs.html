   45 runLongSessionFixture :: FilePath -> IO ()
   46 runLongSessionFixture filename = do
   47   root <- getCurrentDirectory
   48   let fixturePath = root </> "test/fixtures/long-sessions" </> filename
   49   inputs <- fmap T.pack . lines <$> readFile fixturePath
   50   withRuntimeEnv ("long-session-" <> filename) $
   51     do
   52       session0 <- Runtime.bootstrapSession True "long-session-test"
   53       let step (turnIdx, sess) input = do
   54             result <- try (Runtime.runTurnInSession sess input)
   55                           :: IO (Either QxFx0Exception (Runtime.Session, T.Text))
   56             case result of
   57               Left err ->
   58                 assertFailure ("turn " ++ show turnIdx ++ " raised: " ++ show err)
   59               Right (sess', _) -> do
   60                 let ss = Runtime.sessSystemState sess'
   61                     angst = case ssEssence ss of
   62                               EssenceUncommitted t -> etAngstLevel t
   63                               EssenceCommitted t _ -> etAngstLevel t
   64                     wits  = case ssEssence ss of
   65                               EssenceUncommitted t -> length (etWitnesses t)
   66                               EssenceCommitted t _ -> length (etWitnesses t)
   67                 assertBool ("angst out of [0,1] at turn " ++ show turnIdx)
   68                   (angst >= 0.0 && angst <= 1.0)
   69                 assertBool ("witness count must increase at turn " ++ show turnIdx)
   70                   (wits == turnIdx)
   71                 pure (turnIdx + 1, sess')
   72       (_, finalSession) <- foldM step (1, session0) inputs
   73       let finalAngst = case ssEssence (Runtime.sessSystemState finalSession) of
   74                          EssenceUncommitted t -> etAngstLevel t
   75                          EssenceCommitted t _ -> etAngstLevel t
   76       assertBool "final angst must be non-negative" (finalAngst >= 0.0)
