   16 testSelfLayerStaysFoundational :: Test
   17 testSelfLayerStaysFoundational = TestCase $ do
   18   root <- getCurrentDirectory
   19   files <- listHsFilesRecursive (root </> "src" </> "QxFx0" </> "Self")
   20   violations <- fmap sort (concat <$> mapM forbiddenImportsInFile files)
   21   assertEqual
   22     "QxFx0.Self modules must not import runtime, bridge, render, CLI, or app layers"
   23     []

   26 forbiddenImportsInFile :: FilePath -> IO [String]
   27 forbiddenImportsInFile filePath = do
   28   contents <- lines <$> readFile filePath
   29   pure
   30     [ filePath <> ":" <> show lineNo <> ": " <> line
   31     | (lineNo, line) <- zip [1 :: Int ..] contents
   32     , let trimmed = dropWhile (`elem` [' ', '\t']) line
   33     , any (`isPrefixOf` trimmed) forbiddenImportPrefixes
   34     ]
   35 
   36 listHsFilesRecursive :: FilePath -> IO [FilePath]
   37 listHsFilesRecursive dir = do
   38   entries <- sort <$> listDirectory dir
   39   fmap concat $ forM entries $ \entry -> do
   40     let entryPath = dir </> entry
   41     isDir <- doesDirectoryExist entryPath
   42     if isDir
   43       then listHsFilesRecursive entryPath
   44       else pure [entryPath | takeExtension entryPath == ".hs"]
   45
