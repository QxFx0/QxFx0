  185 testProtocolInterpreter :: TurnEffectRequest -> IO TurnEffectResult
  186 testProtocolInterpreter request =
  187   case request of
  188     TurnReqEmbedding inputText ->
  189       TurnResEmbedding <$> Emb.textToEmbeddingResult inputText
  190     TurnReqNixGuard _ _ _ ->
  191       pure (TurnResNixGuard Allowed)
  192     TurnReqConsciousness semanticInput humanTheta resonance _conatusEnergy _salienceWeights -> do
  193       let (loop1, fragment) = CLoop.runConsciousnessLoop CLoop.initialLoop semanticInput humanTheta resonance
  194       pure (TurnResConsciousness loop1 (CLoop.clLastNarrative loop1) (if T.null fragment then Nothing else Just fragment))
  195     TurnReqIntuition inputText resonance tension turnNumber _conatusEnergy _salienceWeights _semanticConfig -> do
  196       let (mFlash, intuitionState) =
  197             Intuition.checkIntuitionWithInput inputText resonance tension turnNumber Intuition.defaultIntuitiveState
  198       pure (TurnResIntuition mFlash (Intuition.effectivePosterior intuitionState) intuitionState)
  199     TurnReqApiHealth ->
  200       pure (TurnResApiHealth True)
  201     TurnReqShadow family force _ ->
  202       pure (TurnResShadow (Just (family, force)) ShadowMatch emptyShadowDivergence (ShadowSnapshotId "shadow:test_protocol") [])
  203     TurnReqAgdaVerify ->
  204       pure (TurnResAgdaVerify AgdaVerified)
  205     TurnReqCurrentTime ->
  206       pure (TurnResCurrentTime protocolFixedTime)
  207     TurnReqRequestId ->
  208       pure (TurnResRequestId "request-id-protocol")
  209     TurnReqReadEnv _ ->
  210       pure (TurnResReadEnv Nothing)
  211     TurnReqTestMarkOnceFile _ ->
  212       pure (TurnResTestMarkOnceFile False)
  213     TurnReqSemanticIntrospectionEnv ->
  214       pure (TurnResSemanticIntrospectionEnv False)
  215     TurnReqCommitRuntimeState _ _ _ ->
  216       pure TurnResCommitRuntimeState
  217     TurnReqSaveState ss _ _ ->
  218       pure (TurnResSaveState (Right ss))
  219     TurnReqRollbackTurnProjections _ _ ->
  220       pure (TurnResRollbackTurnProjections (Right ()))
  221     TurnReqCheckpoint _ ->
  222       pure TurnResCheckpointCompleted
  223     TurnReqLinearizeClaimAst _ _ _ ->
  224       pure (TurnResLinearizeClaimAst (Left "pgf_unavailable_test_protocol"))
  225     TurnReqLinearizeDialogAtoms _ _ _ ->
  226       pure (TurnResLinearizeDialogAtoms (Left "pgf_unavailable_test_protocol"))
  227     TurnReqExternalQuery tool need queryText -> do
  228       transport <- buildTransportFromConfig explicitMockExternalQueryConfig
  229       result <- queryExternalTool transport tool need queryText
  230       pure (TurnResExternalQuery result)
