  125 quickCheckProperty :: String -> Property -> Test
  126 quickCheckProperty label prop = TestCase $ do
  127   result <- quickCheckWithResult stdArgs { maxSuccess = 200 } prop
  128   if isSuccess result
  129     then pure ()
  130     else assertFailure ("Property failed: " ++ label)

  241 arbitraryEssenceTrajectory :: Gen EssenceTrajectory
  242 arbitraryEssenceTrajectory = do
  243   n        <- choose (0, 12)
  244   witnesses <- vectorOf n arbitraryEssenceWitness
  245   angst    <- arbitraryUnitDouble
  246   floor_   <- arbitraryUnitDouble
  247   pure $ EssenceTrajectory
  248     { etWitnesses    = Seq.fromList witnesses
  249     , etAngstLevel   = angst
  250     , etConatusFloor = floor_
  251     , etCapacity     = emTrajectoryCapacity defaultEssenceModulation

  254 arbitraryTriggeredTrajectory :: Gen EssenceTrajectory
  255 arbitraryTriggeredTrajectory = do
  256   trigger <- elements [TriggerAngstThreshold, TriggerConatusErosion]
  257   case trigger of
  258     TriggerAngstThreshold -> do
  259       let window = emConatusFloorWindow defaultEssenceModulation
  260       witnessCount <- choose (window, window + 4)
  261       witnesses <- vectorOf witnessCount arbitraryEssenceWitness
  262       floor_ <- arbitraryUnitDouble
  263       pure EssenceTrajectory
  264         { etWitnesses = Seq.fromList witnesses
  265         , etAngstLevel = emAngstCommitmentThreshold defaultEssenceModulation
  266         , etConatusFloor = floor_
  267         , etCapacity = emTrajectoryCapacity defaultEssenceModulation
  268         }
  269     TriggerConatusErosion -> do
  270       let window = emConatusFloorWindow defaultEssenceModulation
  271           lowConatus = max 0.0 (emConatusStructuralFloor defaultEssenceModulation - 0.1)
  272           mkLowWitness n = (emptyWitness n) { ewConatusScalar = lowConatus }
  273       angst <- choose (0.0, emAngstCommitmentThreshold defaultEssenceModulation - 0.05)
  274       pure EssenceTrajectory
  275         { etWitnesses = Seq.fromList (map mkLowWitness [1 .. window])
  276         , etAngstLevel = angst
  277         , etConatusFloor = lowConatus
  278         , etCapacity = emTrajectoryCapacity defaultEssenceModulation
  279         }

  285 -- | E1 — extractMode determinism + non-Witnessing.
  286 propExtractModeDeterministic :: Property
  287 propExtractModeDeterministic =
  288   forAll arbitraryEssenceTrajectory $ \t ->
  289     let m1 = extractMode t
  290         m2 = extractMode t
  291     in m1 == m2 && m1 /= EssenceWitnessing
  292 
  293 -- | E2 — angst decays under FullAgreement with zero divergence.
  294 propAngstDecaysUnderAgreement :: Property
  295 propAngstDecaysUnderAgreement =
  296   forAll arbitraryEssenceTrajectory $ \traj ->
  297     forAll arbitraryField $ \fd ->
  298     forAll arbitraryConatusEnergy $ \ce ->
  299     let deliberation = (defaultDeliberation :: Deliberation)
  300           { delibTrace = (delibTrace defaultDeliberation)
  301               { dtAgreement  = Agree
  302               , dtDivergence = 0.0
  303               , dtRule       = RuleAgreement
  304               }
  305           }
  306         traj' = witness defaultEssenceModulation 1 ce fd deliberation traj
  307         a0 = etAngstLevel traj
  308         a1 = etAngstLevel traj'
  309     in a1 <= a0 || a1 == 0.0
  310 
  311 -- | E3 — angst accrues under hemispheric advantage with divergence.
  312 propAngstAccruesUnderDivergence :: Property
  313 propAngstAccruesUnderDivergence =
  314   forAll arbitraryEssenceTrajectory $ \traj ->
  315     etAngstLevel traj < 1.0 ==>  -- only meaningful when there is room to accrue
  316     forAll arbitraryField $ \fd ->
  317     forAll arbitraryConatusEnergy $ \ce ->
  318     forAll (choose (0.5, 1.0)) $ \divg ->
  319     forAll (elements [RuleHolisticAdvantage, RuleFormalAdvantage]) $ \rule ->
  320     let deliberation = (defaultDeliberation :: Deliberation)
  321           { delibTrace = (delibTrace defaultDeliberation)
  322               { dtDivergence = divg
  323               , dtRule       = rule
  324               }
  325           }
  326         traj' = witness defaultEssenceModulation 1 ce fd deliberation traj
  327         a0 = etAngstLevel traj
  328         a1 = etAngstLevel traj'
  329     in a1 >= a0 || a1 == 1.0
  330 
  331 -- | E4 — shouldCommit monotone once fired.
  332 -- Vacuously true for most random trajectories because default
  333 -- thresholds (0.75 angst, 8 witnesses) are rarely hit.
  334 propShouldCommitMonotone :: Property
  335 propShouldCommitMonotone =
  336   checkCoverage $
  337   forAll arbitraryTriggeredTrajectory $ \traj ->
  338     forAll arbitraryField $ \fd ->
  339     forAll arbitraryConatusEnergy $ \ce ->
  340     forAll arbitraryDeliberation $ \delib ->
  341     let m0 = shouldCommit defaultEssenceModulation traj
  342         traj' = witness defaultEssenceModulation 1 ce fd delib traj
  343         m1 = shouldCommit defaultEssenceModulation traj'
  344         firedByAngst = m0 == Just TriggerAngstThreshold
  345         firedByConatus = m0 == Just TriggerConatusErosion
  346     in cover 40 firedByAngst "angst-triggered"
  347        . cover 40 firedByConatus "conatus-triggered"
  348        $ case m1 of
  349            Nothing -> False
  350            Just _ -> True
  351 
  352 -- | E5 — trigger priority: Angst beats Conatus when both fire.
  353 propTriggerPriorityAngst :: Property
  354 propTriggerPriorityAngst =
  355   let traj = EssenceTrajectory
  356         { etWitnesses    = Seq.fromList (replicate 8 (emptyWitness 1))
  357         , etAngstLevel   = 0.80
  358         , etConatusFloor = 0.40
  359         , etCapacity     = emTrajectoryCapacity defaultEssenceModulation
  360         }
  361   in property (shouldCommit defaultEssenceModulation traj == Just TriggerAngstThreshold)
  362 
  363 emptyWitness :: Int -> EssenceWitness
  364 emptyWitness n = EssenceWitness
  365   { ewTurnOrdinal     = n
  366   , ewSalienceDriver  = DrivenByDefault
  367   , ewReconcileRule   = RuleSalienceLead
  368   , ewAgreement       = Agree
  369   , ewDivergence      = 0.0
  370   , ewConatusScalar   = 1.0
  371   , ewFieldSignature  = fieldSignature defaultEssenceModulation emptyField
  372   }

  378 testEmptyEssenceShape :: IO ()
  379 testEmptyEssenceShape = do
  380   let e = emptyEssence
  381   case e of
  382     EssenceUncommitted t -> do
  383       assertEqual "emptyEssence witnesses" Seq.empty (etWitnesses t)
  384       assertEqual "emptyEssence angst" 0.0 (etAngstLevel t)
  385       assertEqual "emptyEssence floor" 1.0 (etConatusFloor t)
  386       assertEqual "emptyEssence capacity"
  387         (emTrajectoryCapacity defaultEssenceModulation)
  388         (etCapacity t)
  389     EssenceCommitted _ _ ->
  390       assertBool "emptyEssence should be Uncommitted" False
  391 
  392 testFieldSignatureTotal :: IO ()
  393 testFieldSignatureTotal = do
  394   let sig = fieldSignature defaultEssenceModulation emptyField
  395   -- Totality: no exception, and all bands are valid inhabitants.
  396   assertBool "resonance band valid" (isValidFieldBand (fsResonance sig))
  397   assertBool "arousal band valid"   (isValidFieldBand (fsArousal sig))
  398   assertBool "valence band valid"   (isValidValenceBand (fsValence sig))
  399   assertBool "consolidation band valid" (isValidFieldBand (fsConsolidation sig))
  400   assertBool "counterfactual band valid" (isValidFieldBand (fsCounterfactual sig))
  401 
  402 isValidFieldBand :: FieldBand -> Bool
  403 isValidFieldBand b = b `elem` [BandLow, BandMid, BandHigh]
  404 
  405 isValidValenceBand :: ValenceBand -> Bool
  406 isValidValenceBand b = b `elem` [ValenceNegative, ValenceNeutral, ValencePositive]
  407 
  408 testShouldCommitAngstBoundary :: IO ()
  409 testShouldCommitAngstBoundary = do
  410   let window = emConatusFloorWindow defaultEssenceModulation
  411       traj = EssenceTrajectory
  412         { etWitnesses = Seq.fromList (replicate window (emptyWitness 1))
  413         , etAngstLevel = emAngstCommitmentThreshold defaultEssenceModulation
  414         , etConatusFloor = 1.0
  415         , etCapacity = emTrajectoryCapacity defaultEssenceModulation
  416         }
  417   assertEqual "angst threshold should fire commitment"
  418     (Just TriggerAngstThreshold)
  419     (shouldCommit defaultEssenceModulation traj)

  421 testShouldCommitConatusWindowBoundary :: IO ()
  422 testShouldCommitConatusWindowBoundary = do
  423   let window = emConatusFloorWindow defaultEssenceModulation
  424       lowConatus = max 0.0 (emConatusStructuralFloor defaultEssenceModulation - 0.1)
  425       traj = EssenceTrajectory
  426         { etWitnesses = Seq.fromList [ (emptyWitness n) { ewConatusScalar = lowConatus } | n <- [1 .. window] ]
  427         , etAngstLevel = emAngstCommitmentThreshold defaultEssenceModulation - 0.05
  428         , etConatusFloor = lowConatus
  429         , etCapacity = emTrajectoryCapacity defaultEssenceModulation
  430         }
  431   assertEqual "sub-floor window should fire conatus erosion commitment"
  432     (Just TriggerConatusErosion)
  433     (shouldCommit defaultEssenceModulation traj)

  435 testShouldCommitRemainsFiredAfterWitness :: IO ()
  436 testShouldCommitRemainsFiredAfterWitness = do
  437   let traj = EssenceTrajectory
  438         { etWitnesses = Seq.fromList (replicate 8 (emptyWitness 1))
  439         , etAngstLevel = emAngstCommitmentThreshold defaultEssenceModulation
  440         , etConatusFloor = 1.0
  441         , etCapacity = emTrajectoryCapacity defaultEssenceModulation
  442         }
  443       traj' = witness defaultEssenceModulation 9 (ConatusEnergy 0.9 (ConatusComponents 0 0 0 0)) emptyField defaultDeliberation traj
  444   assertBool "fired commitment should remain fired after another witness"
  445     (case shouldCommit defaultEssenceModulation traj' of
  446        Nothing -> False
  447        Just _ -> True)
