   47 testIsVapidTopic :: Test
   48 testIsVapidTopic = TestCase $ do
   49   assertBool "'не' should be vapid" (isVapidTopic "не")
   50   assertBool "'это' should be vapid" (isVapidTopic "это")
   51 
   52 testIsVapidTopicFalse :: Test
   53 testIsVapidTopicFalse = TestCase $ do
   54   assertBool "'право' should not be vapid" (not (isVapidTopic "право"))
   55 
   56 testCleanTopic :: Test
   57 testCleanTopic = TestCase $ do
   58   assertEqual "strips only" "ПРАВО" (cleanTopic "  ПРАВО  ")
   59 
   60 testStancePrefixAll :: Test
   61 testStancePrefixAll = TestCase $ do
   62   -- Calling stancePrefix for all constructors exercises the mapping.
   63   -- Commit and Observe intentionally return empty text per design.
   64   mapM_ (\sm -> assertBool ("stancePrefix should not crash for " ++ show sm)
   65              (let _ = stancePrefix sm in True))
   66     [Commit, Observe, Explore, HoldBack, Firm, Honest, Tentative, Curated]

   68 testMoveToTextGroundKnown :: Test
   69 testMoveToTextGroundKnown = TestCase $ do
   70   let txt = moveToText MoveGroundKnown "право" emptyMorphologyData
   71   assertBool "GroundKnown should produce non-empty text" (not (T.null txt))
