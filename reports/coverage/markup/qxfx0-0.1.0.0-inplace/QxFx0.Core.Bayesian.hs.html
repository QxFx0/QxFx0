   30 bayesianUpdate :: SemanticConfig -> BeliefState -> ParsedInput -> BeliefState
   31 bayesianUpdate cfg prior obs =
   32   let unnorm = M.mapWithKey (\h p -> p * likelihood cfg h obs) prior
   33       total  = sum (M.elems unnorm)
   34   in if total > 1e-9 then M.map (/ total) unnorm else prior
   35 
   36 bayesianUpdateFromText :: SemanticConfig -> BeliefState -> Text -> BeliefState
   37 bayesianUpdateFromText cfg prior raw =
   38   let stress = simpleTextStress cfg raw
   39       unnorm = M.mapWithKey (\h p -> p * textLikelihood cfg h raw stress) prior
   40       total  = sum (M.elems unnorm)
   41   in if total > 1e-9 then M.map (/ total) unnorm else prior
   42 
   43 simpleTextStress :: SemanticConfig -> Text -> Double
   44 simpleTextStress cfg raw =
   45   let w = T.words (T.toLower raw)
   46       dCount = fromIntegral (length (filter (`elem` scTensionDistressLemmas cfg) w))
   47       nCount = fromIntegral (length (filter (== "не") w))
   48       defCount = fromIntegral (length (filter (`elem` scBayesianDefineLemmas cfg) w))
   49   in dCount + nCount * 0.5 + defCount * 0.3

   51 textLikelihood :: SemanticConfig -> HiddenBelief -> Text -> Double -> Double
   52 textLikelihood cfg UserWantsDefine raw _ =
   53   let w = T.words (T.toLower raw)
   54       defCount = fromIntegral (length (filter (`elem` scBayesianDefineLemmas cfg) w))
   55   in 0.1 + 0.15 * defCount
   56 textLikelihood cfg UserWantsCompare raw _ =
   57   let w = T.words (T.toLower raw)
   58       cmpCount = fromIntegral (length (filter (`elem` scBayesianCompareLemmas cfg) w))
   59   in 0.1 + 0.15 * cmpCount
   60 textLikelihood cfg UserWantsConfront raw _ =
   61   let w = T.words (T.toLower raw)
   62       cfmCount = fromIntegral (length (filter (`elem` scBayesianConfrontLemmas cfg) w))
   63   in 0.05 + 0.2 * cfmCount
   64 textLikelihood cfg UserSeeksSupport raw _ =
   65   let w = T.words (T.toLower raw)
   66       supCount = fromIntegral (length (filter (`elem` scBayesianSupportLemmas cfg) w))
   67   in 0.05 + 0.15 * supCount
   68 textLikelihood cfg UserIsConfused raw _ =
   69   let w = T.words (T.toLower raw)
   70       cnfCount = fromIntegral (length (filter (`elem` scBayesianConfusedLemmas cfg) w))
   71   in 0.05 + 0.2 * cnfCount
   72 textLikelihood _cfg UserIsDistressed _ stress = 0.05 + 0.2 * stress

   74 likelihood :: SemanticConfig -> HiddenBelief -> ParsedInput -> Double
   75 likelihood cfg UserWantsDefine pi =
   76   let defCount = fromIntegral (length [t | t <- piTokens pi, ptLemma t `elem` scBayesianDefineLemmas cfg])
   77   in 0.1 + 0.15 * defCount
   78 likelihood cfg UserWantsCompare pi =
   79   let cmpCount = fromIntegral (length [t | t <- piTokens pi, ptLemma t `elem` scBayesianCompareLemmas cfg])
   80   in 0.1 + 0.15 * cmpCount
   81 likelihood cfg UserWantsConfront pi =
   82   let cfmCount = fromIntegral (length [t | t <- piTokens pi, ptLemma t `elem` scBayesianConfrontLemmas cfg])
   83   in 0.05 + 0.2 * cfmCount
   84 likelihood cfg UserSeeksSupport pi =
   85   let supCount = fromIntegral (length [t | t <- piTokens pi, ptLemma t `elem` scBayesianSupportLemmas cfg])
   86   in 0.05 + 0.15 * supCount
   87 likelihood cfg UserIsConfused pi =
   88   let cnfCount = fromIntegral (length [t | t <- piTokens pi, ptLemma t `elem` scBayesianConfusedLemmas cfg])
   89   in 0.05 + 0.2 * cnfCount
   90 likelihood cfg UserIsDistressed pi =
   91   let dWords = fromIntegral (length [t | t <- piTokens pi, ptLemma t `elem` scTensionDistressLemmas cfg])
   92   in 0.05 + 0.2 * dWords

   94 detectInsight :: BeliefState -> Maybe (HiddenBelief, Double)
   95 detectInsight bs =
   96   let above = filter (\(_, p) -> p >= 0.95) (M.toList bs)
   97   in listToMaybe above

   99 maxBelief :: BeliefState -> Double
  100 maxBelief bs = if M.null bs then 0.0 else maximum (M.elems bs)
