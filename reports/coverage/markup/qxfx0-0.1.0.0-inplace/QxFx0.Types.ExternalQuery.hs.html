    7 {-|
    8 Module      : QxFx0.Types.ExternalQuery
    9 Description : Types for external tool query transport (Phase 8).
   10 
   11 Typed error taxonomy and response envelope for LLM / mentor / script
   12 queries.  Kept pure and serialisable so the sandbox gate can replay
   13 responses deterministically.
   14 -}
   15 module QxFx0.Types.ExternalQuery
   16   ( ExternalQueryError(..)
   17   , ExternalQueryResponse(..)
   18   , ExternalQueryConfig(..)
   19   , TransportFallbackReason(..)
   20   , renderExternalQueryError
   21   , renderFallbackReason
   22   ) where
   23 
   24 import Control.DeepSeq (NFData)
   25 import Data.Aeson (FromJSON, ToJSON)
   26 import Data.Text (Text)
   27 import qualified Data.Text as T
   28 import GHC.Generics (Generic)
   29 
   30 -- | Typed error taxonomy for external query failures.
   31 --
   32 -- Network and transport errors are distinguished from content-level
   33 -- errors so the learning loop can decide whether to retry, fallback,
   34 -- or reject.
   35 data ExternalQueryError
   36   = EqeNetworkUnavailable !Text
   37     -- ^ DNS timeout, TLS failure, or no route to host.
   38   | EqeAuthFailure !Text
   39     -- ^ 401 / 403 or missing API key.
   40   | EqeRateLimited !Text
   41     -- ^ 429 or quota exceeded; back-off recommended.
   42   | EqeServerError !Text
   43     -- ^ 5xx from upstream.
   44   | EqeTimeout !Text
   45     -- ^ In-application deadline exceeded.
   46   | EqeInvalidResponse !Text
   47     -- ^ Response body is not valid JSON / schema.
   48   | EqeFallback !TransportFallbackReason
   49     -- ^ The runtime intentionally fell back instead of attempting an
   50     --   authoritative upstream query. This is non-authoritative and
   51     --   must not be treated as learning success.
   52   | EqeEmptyResponse
   53     -- ^ Response body is empty after trimming.
   54   deriving stock (Eq, Show, Generic)
   55     deriving anyclass (NFData, FromJSON, ToJSON)
   56 
   57 -- | Normalised response from an external tool.
   58 --
   59 -- The raw body is kept for telemetry / debugging, but the
   60 -- 'eqrStructured' field is what the validator and parser consume.
   61 data ExternalQueryResponse = ExternalQueryResponse
   62   { eqrRawBody      :: !Text
   63     -- ^ Original response body (truncated at 4 KiB by transport).
   64   , eqrStructured   :: !Text
   65     -- ^ Extracted structured payload (JSON schema or constrained text).
   66   , eqrToolName     :: !Text
   67     -- ^ Canonical tool identifier for reliability tracking.
   68   , eqrLatencyMs    :: !Int
   69     -- ^ Round-trip latency in milliseconds (for telemetry).
   70   }
   71   deriving stock (Eq, Show, Generic)
   72     deriving anyclass (NFData, FromJSON, ToJSON)
   73 
   74 -- | Explicit configuration contract for external query transport.
   75 -- All fields are optional with safe defaults; the builder in
   76 -- 'Bridge.ExternalLLM' fills them from env vars.
   77 data ExternalQueryConfig = ExternalQueryConfig
   78   { eqcTransportMode   :: !Text
   79     -- ^ "mock" | "mistral"
   80   , eqcApiKey          :: !(Maybe Text)
   81     -- ^ Redacted in 'Show' / logs.
   82   , eqcModel           :: !Text
   83   , eqcEndpoint        :: !Text
   84   , eqcTimeoutMs       :: !Int
   85   , eqcFallbackReason  :: !(Maybe TransportFallbackReason)
   86     -- ^ If transport fell back to mock, why.
   87   }
   88   deriving stock (Eq, Generic)
   89     deriving anyclass (NFData, FromJSON, ToJSON)
   90 
   91 instance Show ExternalQueryConfig where
   92   show cfg =
   93     "ExternalQueryConfig { transportMode=" ++ show (eqcTransportMode cfg)
   94     ++ ", apiKey=<REDACTED>"
   95     ++ ", model=" ++ show (eqcModel cfg)
   96     ++ ", endpoint=" ++ show (eqcEndpoint cfg)
   97     ++ ", timeoutMs=" ++ show (eqcTimeoutMs cfg)
   98     ++ ", fallbackReason=" ++ show (eqcFallbackReason cfg)
   99     ++ " }"
  100 
  101 -- | Why the transport fell back to mock instead of using the real
  102 -- Mistral API.  Used for telemetry and operator diagnosis.
  103 data TransportFallbackReason
  104   = TfrEnvNotSet
  105     -- ^ QXFX0_LLM_TRANSPORT is not "mistral".
  106   | TfrKeyMissing
  107     -- ^ QXFX0_MISTRAL_API_KEY is absent or empty.
  108   | TfrExplicitMock
  109     -- ^ User explicitly requested mock mode.
  110   | TfrUnsafeEndpoint
  111     -- ^ Endpoint does not use HTTPS.
  112   | TfrBlockedHost
  113     -- ^ Endpoint host is not in the official allowlist and untrusted-host
  114     -- override is not enabled.
  115   | TfrUntrustedOverrideRejected
  116     -- ^ Untrusted-host override was requested but the runtime was not in an
  117     -- explicit dev/test or double-confirmed mode.
  118   deriving stock (Eq, Show, Generic)
  119     deriving anyclass (NFData, FromJSON, ToJSON)
  120 
  121 renderFallbackReason :: TransportFallbackReason -> Text
  122 renderFallbackReason TfrEnvNotSet    = "env_not_set"
  123 renderFallbackReason TfrKeyMissing = "key_missing"
  124 renderFallbackReason TfrExplicitMock = "explicit_mock"
  125 renderFallbackReason TfrUnsafeEndpoint = "unsafe_endpoint"
  126 renderFallbackReason TfrBlockedHost = "blocked_host"
  127 renderFallbackReason TfrUntrustedOverrideRejected = "untrusted_override_rejected"
  128 
  129 renderExternalQueryError :: ExternalQueryError -> Text
  130 renderExternalQueryError err =
  131   case err of
  132     EqeNetworkUnavailable t -> T.concat ["network_unavailable:", t]
  133     EqeAuthFailure t        -> T.concat ["auth_failure:", t]
  134     EqeRateLimited t        -> T.concat ["rate_limited:", t]
  135     EqeServerError t        -> T.concat ["server_error:", t]
  136     EqeTimeout t            -> T.concat ["timeout:", t]
  137     EqeInvalidResponse t      -> T.concat ["invalid_response:", t]
  138     EqeFallback reason      -> T.concat ["fallback:", renderFallbackReason reason]
  139     EqeEmptyResponse        -> "empty_response"
