  349 instance FromJSON SystemState where
  350   parseJSON = withObject "SystemState" $ \o -> do
  351     schemaVersion <- o .:? "schemaVersion" .!= 1
  352     let requiredTopLevelFields
  353           | schemaVersion >= currentSystemStateSchemaVersion =
  354               [ "morphology"
  355               , "salienceWeights"
  356               , "fieldHeuristics"
  357               , "learningNeedState"
  358               , "knowledgeTree"
  359               , "truthContractStatus"
  360               , "dialogueOutcomeLearning"
  361               , "dialogueThread"
  362               , "dialogueCommitmentLedger"
  363               , "dialoguePhase"
  364               , "speechPolicyState"
  365               , "beliefStore"
  366               , "governanceHistory"
  367               ]
  368           | otherwise =
  369               [ "morphology"
  370               , "salienceWeights"
  371               , "fieldHeuristics"
  372               , "learningNeedState"
  373               , "knowledgeTree"
  374               , "truthContractStatus"
  375               ]
  376         missingTopLevel = filter (\k -> not (KM.member (AK.fromText k) o)) requiredTopLevelFields
  377     when (not (null missingTopLevel)) $
  378       fail ("missing required top-level fields: " <> show missingTopLevel)
  379     ds <- DialogueState
  380       <$> o .: "history"
  381       <*> o .: "rawInputHistory"
  382       <*> o .: "turnCount"
  383       <*> o .: "lastTopic"
  384       <*> o .: "lastFamily"

  495 ssDiscourse :: SystemState -> DiscourseState
  496 ssDiscourse ss =
  497   let history = dsHistory (ssDialogue ss)
  498       turnCount = dsTurnCount (ssDialogue ss)
  499       lastTopic = dsLastTopic (ssDialogue ss)
  500       turnMemory = if turnCount <= 0
  501         then Seq.empty
  502         else Seq.singleton TurnMemory
  503           { tmrTurnIndex = max 0 (turnCount - 1)
  504           , tmrTopic = lastTopic
  505           , tmrFamily = dsLastFamily (ssDialogue ss)
  506           , tmrRendered = maybe "" id (Seq.lookup (max 0 (Seq.length history - 1)) history)
  507           }
  508   in recomputeDiscourse emptyDiscourseState
  509        { dscTurnMemory = turnMemory
  510        , dscTopicChain = filter (not . T.null) [lastTopic]
  511        }
