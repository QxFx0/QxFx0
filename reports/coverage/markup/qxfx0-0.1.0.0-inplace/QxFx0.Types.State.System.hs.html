  137 instance FromJSON SystemState where
  138   parseJSON = withObject "SystemState" $ \o -> do
  139     ds <- DialogueState
  140       <$> o .: "history"
  141       <*> o .: "rawInputHistory"
  142       <*> o .: "turnCount"
  143       <*> o .: "lastTopic"
  144       <*> o .: "lastFamily"
  145       <*> o .: "lastForce"
  146       <*> o .: "lastLayer"
  147       <*> o .: "lastEmbedding"
  148       <*> o .: "consecutiveReflect"
  149       <*> o .: "recentFamilies"
  150       <*> o .: "activeScene"
  151       <*> o .: "userState"
  152     ids <- IdentityState
  153       <$> o .: "ego"
  154       <*> o .: "identityClaims"
  155       <*> o .: "orbitalMemory"
  156       <*> o .:? "lastGuardReport" .!= Nothing
  157     sem <- SemanticState
  158       <$> o .: "trace"
  159       <*> o .: "meaningGraph"
  160       <*> o .: "kernelPulse"
  161       <*> o .: "blockedConcepts"
  162       <*> o .: "clusters"
  163       <*> o .:? "dreamState" .!= emptyDreamState zeroVec
  164       <*> o .:? "intuitionState" .!= Just defaultIntuitiveState
  165       <*> o .:? "semanticAnchor" .!= Nothing
  166       <*> o .:? "lastTurnDecision" .!= Nothing
  167       <*> o .: "intuitConfidence"
  168     SystemState ds ids sem
  169       <$> o .: "sessionId"
  170       <*> (parseDialogueOutputMode <$> o .: "outputMode")
  171       <*> o .:? "morphology" .!= MorphologyData M.empty M.empty M.empty M.empty
  172       <*> o .: "observability"
