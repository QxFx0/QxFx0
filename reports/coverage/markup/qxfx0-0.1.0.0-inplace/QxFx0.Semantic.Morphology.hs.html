   90 data MorphToken = MorphToken
   91   { mtSurface :: !Text
   92   , mtLemma :: !Text
   93   , mtPOS :: !POS
   94   , mtCase :: !(Maybe Case)
   95   , mtNumber :: !(Maybe Number)
   96   , mtGender :: !(Maybe Gender)
   97   , mtTense :: !(Maybe Tense)
   98   , mtMood :: !(Maybe Mood)
   99   , mtPerson :: !(Maybe Person)
  100   } deriving stock (Eq, Show)
  101 
  102 type MorphDict = Map Text MorphToken
  103 
  104 generatedMorphDict :: MorphDict
  105 generatedMorphDict = foldl insertLexeme M.empty generatedLexemeEntries
  106   where
  107     insertLexeme :: MorphDict -> (Text, Text, Text, Text) -> MorphDict
  108     insertLexeme acc (surface, lemma, posTag, caseTag) =
  109       let key = T.toLower surface
  110           token = mkToken surface lemma posTag caseTag
  111       in M.insertWith pickPreferred key token acc
  112 
  113     pickPreferred :: MorphToken -> MorphToken -> MorphToken
  114     pickPreferred new old
  115       | tokenPriority new > tokenPriority old = new
  116       | otherwise = old
  117 
  118     tokenPriority :: MorphToken -> Int
  119     tokenPriority tok = case mtCase tok of
  120       Just Nominative -> 3
  121       Just Genitive -> 2
  122       Just Prepositional -> 1
  123       _ -> 0
  124 
  125     parsePOS :: Text -> POS
  126     parsePOS tag = case T.toLower tag of
  127       "noun" -> Noun
  128       "verb" -> Verb
  129       "adj" -> Adj
  130       "adjective" -> Adj
  131       "adv" -> Adv
  132       _ -> UnknownPOS
  133 
  134     parseCase :: Text -> Maybe Case
  135     parseCase tag = case T.toLower tag of
  136       "nominative" -> Just Nominative
  137       "genitive" -> Just Genitive
  138       "prepositional" -> Just Prepositional
  139       _ -> Nothing
  140 
  141     parseTense :: POS -> Maybe Tense
  142     parseTense Verb = Just Present
  143     parseTense _ = Nothing
  144 
  145     mkToken :: Text -> Text -> Text -> Text -> MorphToken
  146     mkToken surface lemma posTag caseTag =
  147       let pos = parsePOS posTag
  148       in MorphToken
  149         { mtSurface = surface
  150         , mtLemma = lemma
  151         , mtPOS = pos
  152         , mtCase = parseCase caseTag
  153         , mtNumber = Just Singular
  154         , mtGender = Nothing
  155         , mtTense = parseTense pos
  156         , mtMood = Nothing
  157         , mtPerson = Nothing
  158         }

  160 analyzeMorph :: Text -> MorphToken
  161 analyzeMorph word =
  162   let low = T.toLower word
  163   in case M.lookup low generatedMorphDict of
  164     Just tok -> tok { mtSurface = word }
  165     Nothing -> guessMorph word

  167 resolveMorphBackend :: IO MorphBackend
  168 resolveMorphBackend = do
  169   raw <- lookupEnv "QXFX0_MORPH_BACKEND"
  170   pure $ case fmap (T.toLower . T.pack) raw of
  171     Just "remote" -> MorphBackendRemote
  172     _ -> MorphBackendLocal
  173 
  174 analyzeMorphWithBackend :: MorphBackend -> Text -> IO MorphToken
  175 analyzeMorphWithBackend backend word =
  176   case backend of
  177     MorphBackendLocal -> pure (analyzeMorph word)
  178     -- Local-first runtime: remote morphology flag is accepted but still
  179     -- resolved to local analysis in the semantic layer.
  180     MorphBackendRemote -> pure (analyzeMorph word)

  182 guessMorph :: Text -> MorphToken
  183 guessMorph word =
  184   let w = T.toLower word
  185   in MorphToken word w (guessPOS w) (guessCase w) (guessNumber w) (guessGender w) Nothing Nothing Nothing
  186 
  187 guessPOS :: Text -> POS
  188 guessPOS w
  189   | T.isSuffixOf morphVerbSuffixT w || T.isSuffixOf morphVerbSuffixTi w = Verb
  190   | T.isSuffixOf morphAdjSuffixYj w || T.isSuffixOf morphAdjSuffixIj w
  191     || T.isSuffixOf morphAdjSuffixOj w || T.isSuffixOf morphAdjSuffixYj w = Adj
  192   | T.isSuffixOf morphAdvSuffixO w && T.length w > 3 = Adv
  193   | T.isSuffixOf morphNounSuffixOst w || T.isSuffixOf morphNounSuffixNost w
  194     || T.isSuffixOf morphNounSuffixEnie w || T.isSuffixOf morphNounSuffixNnie w = Noun
  195   | T.isSuffixOf morphNounSuffixA w || T.isSuffixOf morphNounSuffixIya w = Noun
  196   | otherwise = Noun
  197 
  198 guessCase :: Text -> Maybe Case
  199 guessCase w
  200   | T.isSuffixOf morphInstrSuffixOm w && T.length w > 3 = Just Instrumental
  201   | T.isSuffixOf morphGenSuffixI w && not (T.isSuffixOf morphVerbSuffixTi w) = Just Genitive
  202   | T.isSuffixOf morphPrepSuffixE w && T.length w > 4 = Just Prepositional
  203   | T.isSuffixOf morphDatSuffixU w = Just Dative
  204   | T.isSuffixOf morphAccSuffixUyu w = Just Accusative
  205   | otherwise = Just Nominative
  206 
  207 guessNumber :: Text -> Maybe Number
  208 guessNumber w
  209   | T.isSuffixOf morphPluralSuffixY w || T.isSuffixOf morphPluralSuffixI w || T.isSuffixOf morphPluralSuffixAmi w = Just Plural
  210   | otherwise = Just Singular
  211 
  212 guessGender :: Text -> Maybe Gender
  213 guessGender w
  214   | T.isSuffixOf morphFemSuffixA w || T.isSuffixOf morphFemSuffixYa w = Just Feminine
  215   | T.isSuffixOf morphNeutSuffixO w || T.isSuffixOf morphNeutSuffixE w = Just Neuter
  216   | otherwise = Just Masculine
  217 
  218 nounFromVerb :: Text -> Maybe Text
  219 nounFromVerb v
  220   | T.isSuffixOf morphVerbDerivOvat v = Just $ T.dropEnd 3 v <> morphDerivEnie
  221   | T.isSuffixOf morphVerbDerivT v = Just $ T.dropEnd 2 v <> morphDerivEnie
  222   | T.isSuffixOf morphVerbDerivTi v = Just $ T.dropEnd 2 v <> morphDerivEnie
  223   | otherwise = Nothing
  224 
  225 nounFromAdj :: Text -> Maybe Text
  226 nounFromAdj a
  227   | T.isSuffixOf morphAdjDerivYj a = Just $ T.dropEnd 2 a <> morphDerivOst
  228   | T.isSuffixOf morphAdjDerivIj a = Just $ T.dropEnd 2 a <> morphDerivOst
  229   | T.isSuffixOf morphAdjSuffixOj a = Just $ T.dropEnd 2 a <> morphDerivOst
  230   | otherwise = Nothing

  232 extractContentNouns :: Text -> [Text]
  233 extractContentNouns input =
  234   let words' = map normalizeLexeme (T.words input)
  235       tokens = map analyzeMorph (filter (not . T.null) words')
  236   in
  237     [ mtLemma t
  238     | t <- tokens
  239     , isContentNoun t
  240     ]
  241 
  242 normalizeLexeme :: Text -> Text
  243 normalizeLexeme = T.dropAround (\ch -> not (isLetter ch) && ch /= '-')
  244 
  245 isContentNoun :: MorphToken -> Bool
  246 isContentNoun token =
  247   let lemma = T.toLower (mtLemma token)
  248   in mtPOS token == Noun
  249       && T.length lemma > 3
  250       && lemma `notElem` contentStopwords
  251 
  252 contentStopwords :: [Text]
  253 contentStopwords =
  254   [ "что", "кто", "как", "где", "когда", "зачем", "почему"
  255   , "такое", "такой", "такая", "такие", "таков", "такова"
  256   , "это", "этот", "эта", "эти", "того", "этому"
  257   , "какой", "какая", "какие", "каково"
  258   , "ничего", "нечто"
  259   ]

  261 buildMorphologyData :: [MorphToken] -> MorphologyData
  262 buildMorphologyData tokens =
  263   let prepPairs = [ (mtSurface t, mtLemma t) | t <- tokens, mtCase t == Just Prepositional ]
  264       genPairs  = [ (mtSurface t, mtLemma t) | t <- tokens, mtCase t == Just Genitive ]
  265       nomPairs  = [ (mtSurface t, mtLemma t) | t <- tokens, mtCase t == Just Nominative ]
  266   in MorphologyData
  267     { mdPrepositional = M.fromList prepPairs
  268     , mdGenitive = M.fromList genPairs
  269     , mdNominative = M.fromList nomPairs
  270     , mdFormsBySurface = generatedCandidateForms
  271     }
  272 
  273 hasKnownMorphologyForm :: MorphologyData -> Text -> Bool
  274 hasKnownMorphologyForm md w =
  275   let lower = T.toLower w
  276       present dict = M.member w dict || M.member lower dict
  277       presentSurfaceForms = M.member w (mdFormsBySurface md) || M.member lower (mdFormsBySurface md)
  278   in present (mdNominative md)
  279       || present (mdGenitive md)
  280       || present (mdPrepositional md)
  281       || presentSurfaceForms
