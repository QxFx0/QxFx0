    8 {-|
    9 Module      : QxFx0.Self.Types
   10 Description : Pure type definitions for the Self layer (SelfBlanket invariants).
   11 
   12 This module defines the typed snapshot of a system's structural
   13 self-identity: the Markov-blanket-style record whose simultaneous
   14 preservation across operations defines "this system, persisting".
   15 
   16 The types here are intentionally minimal in P1. The full P2+ design
   17 will extend this with conatus-functional carriers and
   18 adjunction-pair witnesses; for now we record only what is needed to
   19 detect identity rupture under existing operations.
   20 
   21 See @docs\/THEORY.md@ §4.1 and @docs\/adr\/0007-dual-mode-conatus.md@.
   22 -}

   24   ( SelfBlanket (..)
   25   , BlanketViolation (..)
   26   , renderBlanketViolation
   27   ) where

   29 import Control.DeepSeq (NFData)
   30 import Data.Aeson (FromJSON, ToJSON, object, (.=))
   31 import qualified Data.Aeson as Aeson
   32 import Data.Text (Text)
   33 import qualified Data.Text as T
   34 import GHC.Generics (Generic)

   36 -- | A snapshot of the system's structural self-identity at a single
   37 -- point in time. Together the fields here form the Markov blanket
   38 -- distinguishing /this system/ from any other system that happens
   39 -- to share state shape.
   40 --
   41 -- The blanket is computed from a 'QxFx0.Types.State.SystemState'
   42 -- (see 'QxFx0.Self.Blanket.computeSelfBlanket') and is intended to
   43 -- be checked at two transition points in the runtime:
   44 --
   45 -- * after session bootstrap (initial blanket must satisfy
   46 --   'QxFx0.Self.Invariants.checkInitialBlanket'); and
   47 -- * after each commit (previous and current blankets together must
   48 --   satisfy 'QxFx0.Self.Invariants.checkBlanketTransition').
   49 data SelfBlanket = SelfBlanket
   50   { sbSessionId           :: Text
   51     -- ^ Session identifier the blanket was taken under. Must remain
   52     -- stable for the lifetime of a single session: a session
   53     -- identity change without an explicit re-bootstrap is a
   54     -- categorical failure ('BlanketSessionChanged').
   55   , sbMorphologyTotalSize :: Int
   56     -- ^ Total entry count across all morphology dictionaries
   57     -- (prepositional + genitive + nominative + forms-by-surface).
   58     -- Must be strictly positive: a system without lexical material
   59     -- cannot produce dialogue and so is not /this system/
   60     -- ('BlanketEmptyMorphology').
   61   , sbIdentityClaimsCount :: Int
   62     -- ^ Number of identity claims attached to the session. Must be
   63     -- non-decreasing across commits: silent erasure of claims is a
   64     -- categorical failure ('BlanketIdentityErased'). The count
   65     -- /may/ legitimately grow as new claims are discovered, and
   66     -- /may/ legitimately start at zero on a fresh session.
   67   , sbTurnCount           :: Int
   68     -- ^ Current turn count. Must be non-decreasing across commits
   69     -- ('BlanketTurnRegressed'). A turn-count regression indicates
   70     -- either persistence corruption or recovery to a state strictly
   71     -- older than the committed state.
   72   }
   73   deriving stock (Eq, Show, Generic)
   74   deriving anyclass (NFData)
   75 
   76 instance ToJSON SelfBlanket where
   77   toJSON sb = object
   78     [ "sessionId"            .= sbSessionId sb
   79     , "morphologyTotalSize"  .= sbMorphologyTotalSize sb
   80     , "identityClaimsCount"  .= sbIdentityClaimsCount sb
   81     , "turnCount"            .= sbTurnCount sb
   82     ]
   83 
   84 instance FromJSON SelfBlanket where
   85   parseJSON = Aeson.withObject "SelfBlanket" $ \o -> SelfBlanket
   86     <$> o Aeson..: "sessionId"
   87     <*> o Aeson..: "morphologyTotalSize"
   88     <*> o Aeson..: "identityClaimsCount"
   89     <*> o Aeson..: "turnCount"

   91 -- | A categorical failure of structural self-identity. Each
   92 -- constructor names a single invariant that may be violated; a
   93 -- single transition may emit zero, one, or several violations.
   94 --
   95 -- Violations are /not/ recoverable runtime errors in the
   96 -- 'QxFx0.ExceptionPolicy' sense. They indicate that the system
   97 -- has ceased to be /this system/. The corresponding exception
   98 -- is 'QxFx0.ExceptionPolicy.IdentityRupture'.
   99 data BlanketViolation
  100   = BlanketEmptySession
  101     -- ^ The session identifier in the blanket is the empty string.
  102   | BlanketEmptyMorphology
  103     -- ^ The morphology dictionaries are collectively empty.
  104   | BlanketSessionChanged !Text !Text
  105     -- ^ Session identifier changed between blankets:
  106     -- @BlanketSessionChanged previousSessionId currentSessionId@.
  107   | BlanketTurnRegressed !Int !Int
  108     -- ^ Turn count went down:
  109     -- @BlanketTurnRegressed previousTurnCount currentTurnCount@
  110     -- where @currentTurnCount < previousTurnCount@.
  111   | BlanketIdentityErased !Int !Int
  112     -- ^ Identity-claim count decreased:
  113     -- @BlanketIdentityErased previousCount currentCount@ where
  114     -- @currentCount < previousCount@.
  115   deriving stock (Eq, Show, Generic)
  116   deriving anyclass (NFData)
  117 
  118 -- | Render a violation to a human-readable diagnostic, suitable
  119 -- for inclusion in 'QxFx0.ExceptionPolicy.IdentityRupture' payloads
  120 -- and observability logs.
  121 renderBlanketViolation :: BlanketViolation -> Text
  122 renderBlanketViolation = \case
  123   BlanketEmptySession ->
  124     "self-blanket: session identifier is empty"
  125   BlanketEmptyMorphology ->
  126     "self-blanket: morphology dictionaries are empty"
  127   BlanketSessionChanged prev cur ->
  128     "self-blanket: session identifier changed (was=\""
  129       <> prev
  130       <> "\" now=\""
  131       <> cur
  132       <> "\")"
  133   BlanketTurnRegressed prev cur ->
  134     "self-blanket: turn count regressed (was="
  135       <> T.pack (show prev)
  136       <> " now="
  137       <> T.pack (show cur)
  138       <> ")"
  139   BlanketIdentityErased prev cur ->
  140     "self-blanket: identity claim count erased (was="
  141       <> T.pack (show prev)
  142       <> " now="
  143       <> T.pack (show cur)
  144       <> ")"
