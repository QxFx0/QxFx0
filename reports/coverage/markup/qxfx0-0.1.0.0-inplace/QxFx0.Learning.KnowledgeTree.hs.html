    7 {-|
    8 Module      : QxFx0.Learning.KnowledgeTree
    9 Description : WP1 — Rooted knowledge tree with grafting, pruning,
   10               and anti-dogmatism quarantine.
   11 
   12 A 'KnowledgeTree' grows from a single root ('EssenceCommitment').
   13 Every 'Branch' is keyed by a 'ReconcileRule' (as rendered text) and
   14 holds 'KnowledgeFruit' that have passed root-connection, validation,
   15 and simulation gates.
   16 
   17 Fruits that are valid but marginal (weak simulation deltas) enter
   18 a quarantine bucket rather than being grafted immediately — the
   19 anti-dogmatism mechanism.  After a monitoring window they are either
   20 promoted to the tree or pruned.
   21 -}

   58 -- | A single unit of learned knowledge.
   59 data KnowledgeFruit = KnowledgeFruit
   60   { kfProposition     :: !Text
   61     -- ^ The learned statement / rule / concept.
   62   , kfWord            :: !Text
   63     -- ^ The key surface word / concept this fruit answers.
   64     --   Backward-compatible: old JSON without this field loads as "".
   65   , kfSource          :: !KnowledgeSource
   66     -- ^ Where it came from.
   67   , kfValidated       :: !Bool
   68     -- ^ Passed basic syntactic / range verification.
   69   , kfConatusDelta    :: !Double
   70     -- ^ Estimated impact on Conatus energy.  Positive = strengthening.
   71   , kfPredictiveDelta :: !Double
   72     -- ^ Estimated impact on repair-loop / uncertainty reduction.
   73     --   Positive = improves prediction quality.
   74   , kfGraftedTurn     :: !(Maybe Int)
   75     -- ^ Turn when grafted into a branch (Nothing if still quarantined).
   76   , kfObservedTurn    :: !Int
   77     -- ^ Turn when the fruit was first observed / proposed.
   78   }
   79   deriving stock (Eq, Show, Generic)
   80     deriving anyclass (NFData)

  136 -- | The rooted knowledge tree.
  137 data KnowledgeTree = KnowledgeTree
  138   { ktRootMode       :: !Text
  139     -- ^ 'renderEssenceMode' of the committed essence serving as root.
  140   , ktRootTrigger    :: !Text
  141     -- ^ 'renderCommitmentTrigger' of the committed essence.
  142   , ktBranches       :: !(M.Map Text [Branch])
  143     -- ^ Branches keyed by reconcile-rule name.
  144   , ktQuarantine     :: ![KnowledgeFruit]

  148   , ktGraftedCount   :: !Int

  152   }

  189 -- | Threshold for root stress signal.
  190 defaultRootStressThreshold :: Double
  191 defaultRootStressThreshold = 0.7
  192 
  193 -- | Compute a root stress signal in [0, 1].
  194 -- High stress means many valid fruits are sitting in quarantine
  195 -- (not grafting) or branches are systemically unhealthy — a signal
  196 -- that the root may be under tension without auto-switching it.
  197 rootStressSignal :: KnowledgeTree -> Double
  198 rootStressSignal t =
  199   let totalFruits = ktGraftedCount t + ktQuarantinedCount t
  200       quarantineRatio =
  201         if totalFruits == 0
  202            then 0.0
  203            else fromIntegral (ktQuarantinedCount t) / fromIntegral totalFruits
  204       -- Average branch health (0 if no branches)
  205       allBranches = concat (M.elems (ktBranches t))
  206       avgHealth =
  207         if null allBranches
  208            then 0.0
  209            else sum (map brHealth allBranches) / fromIntegral (length allBranches)
  210       -- Health inversion: negative health → high stress
  211       healthStress = max 0.0 (-avgHealth)
  212   in clampUnit (0.5 * quarantineRatio + 0.5 * healthStress)

  214 -- | Graft a validated + simulated fruit into a branch.
  215 -- If the branch for the rule does not exist, it is created.
  216 graftFruit :: Text -> KnowledgeFruit -> KnowledgeTree -> KnowledgeTree
  217 graftFruit rule fruit t =
  218   let fruit' = fruit { kfGraftedTurn = Just (kfObservedTurn fruit) }
  219       newBranch = Branch
  220         { brRule = rule
  221         , brFruits = [fruit']
  222         , brHealth = 0.0
  223         , brCreatedTurn = kfObservedTurn fruit
  224         }
  225       updateBranches [] = [newBranch]
  226       updateBranches (b:bs)
  227         | brRule b == rule =
  228             b { brFruits = fruit' : brFruits b
  229               , brHealth = min 1.0 (brHealth b + 0.1)
  230               } : bs
  231         | otherwise = b : updateBranches bs
  232       branches' = M.insertWith (\_ old -> updateBranches old) rule (updateBranches []) (ktBranches t)
  233   in t
  234        { ktBranches = branches'
  235        , ktGraftedCount = ktGraftedCount t + 1
  236        }

  238 -- | Place a fruit into quarantine (anti-dogmatism).
  239 quarantineFruit :: KnowledgeFruit -> KnowledgeTree -> KnowledgeTree
  240 quarantineFruit fruit t =
  241   t { ktQuarantine = fruit : ktQuarantine t
  242     , ktQuarantinedCount = ktQuarantinedCount t + 1
  243     }

  245 -- | Promote quarantined fruits that have aged enough and show
  246 -- positive deltas.  Returns (updated tree, promoted count, remaining
  247 -- quarantined).
  248 promoteFromQuarantine
  249   :: Int        -- ^ current turn
  250   -> Int        -- ^ minimum quarantine age (turns)
  251   -> Text       -- ^ rule to graft under
  252   -> KnowledgeTree
  253   -> (KnowledgeTree, Int, Int)
  254 promoteFromQuarantine currentTurn minAge rule t =
  255   let (ripe, stillQuarantined) =
  256         partitionQuarantine currentTurn minAge (ktQuarantine t)
  257       -- Only promote validated fruits with non-negative net delta
  258       (promotable, reject) =
  259         span (\f -> kfValidated f && netDelta f >= 0.0) ripe
  260       netDelta f = kfConatusDelta f + kfPredictiveDelta f
  261       t' = foldl' (\acc f -> graftFruit rule f acc)
  262               (t { ktQuarantine = stillQuarantined ++ reject
  263                  , ktQuarantinedCount = max 0 (ktQuarantinedCount t - length promotable)
  264                  })
  265               promotable
  266       promoted = length promotable
  267       rejected = length reject
  268   in (t', promoted, rejected)
  269 
  270 partitionQuarantine :: Int -> Int -> [KnowledgeFruit] -> ([KnowledgeFruit], [KnowledgeFruit])
  271 partitionQuarantine currentTurn minAge = go []
  272   where
  273     go acc [] = (acc, [])
  274     go acc (f:fs)
  275       | currentTurn - kfObservedTurn f >= minAge =
  276           go (f : acc) fs
  277       | otherwise =
  278           (acc, f : fs)

  280 -- | Prune branches that have been unhealthy for K consecutive turns.
  281 -- Returns (updated tree, pruned branch count).
  282 pruneBranches :: Int -> Double -> Int -> KnowledgeTree -> (KnowledgeTree, Int)
  283 pruneBranches currentTurn healthThreshold minUnhealthyTurns t =
  284   let allBranches = concat (M.elems (ktBranches t))
  285       -- A branch is pruned if its health has been below threshold
  286       -- and it has existed long enough
  287       (survivors, pruned) =
  288         partitionBranches currentTurn healthThreshold minUnhealthyTurns allBranches
  289       survivorsMap = foldl' insertBranch M.empty survivors
  290       insertBranch m b = M.insertWith (++) (brRule b) [b] m
  291       prunedFruitCount = sum (map (length . brFruits) pruned)
  292   in ( t { ktBranches = survivorsMap
  293          , ktPrunedCount = ktPrunedCount t + prunedFruitCount
  294          }
  295      , length pruned
  296      )
  297 
  298 partitionBranches
  299   :: Int -> Double -> Int -> [Branch] -> ([Branch], [Branch])
  300 partitionBranches currentTurn threshold minAge bs =
  301   let pred b = brHealth b >= threshold
  302             || currentTurn - brCreatedTurn b < minAge
  303   in L.partition pred bs

  305 -- | Prune individual fruits that are unvalidated or have persistently
  306 -- negative deltas.  Also cleans quarantine of unvalidated items.
  307 pruneFruits :: Int -> KnowledgeTree -> (KnowledgeTree, Int)
  308 pruneFruits _currentTurn t =
  309   let pruneBranch b =
  310         let kept = filter (\f -> kfValidated f && netDelta f >= (-0.3)) (brFruits b)
  311             dropped = length (brFruits b) - length kept
  312             health' = brHealth b - 0.05 * fromIntegral dropped
  313         in (b { brFruits = kept, brHealth = max (-1.0) health' }, dropped)
  314       allBranches = concat (M.elems (ktBranches t))
  315       prunedBranches = map pruneBranch allBranches
  316       newMap = foldl' (\m (b, _) -> M.insertWith (++) (brRule b) [b] m) M.empty prunedBranches
  317       totalDropped = sum (map snd prunedBranches)
  318       -- Clean unvalidated quarantine items
  319       cleanQuarantine = filter kfValidated (ktQuarantine t)
  320       quarantineDropped = length (ktQuarantine t) - length cleanQuarantine
  321   in ( t { ktBranches = newMap
  322          , ktQuarantine = cleanQuarantine
  323          , ktPrunedCount = ktPrunedCount t + totalDropped + quarantineDropped
  324          }
  325      , totalDropped + quarantineDropped
  326      )
  327   where
  328     netDelta f = kfConatusDelta f + kfPredictiveDelta f

  335         then 0.0
  336         else sum (map brHealth allBranches) / fromIntegral (length allBranches)

  347 -- | Check whether a surface term is already represented by a validated,
  348 -- grafted fruit in any branch.  Searches by kfWord (if present) and falls
  349 -- back to substring match on kfProposition for backward compatibility.
  350 isTermKnownInKnowledgeTree :: Text -> KnowledgeTree -> Bool
  351 isTermKnownInKnowledgeTree term tree =
  352   let lower = T.toLower term
  353       allFruits = concatMap brFruits (concat (M.elems (ktBranches tree)))
  354       matchesWord f = not (T.null (kfWord f)) && T.toLower (kfWord f) == lower
  355       matchesProp f = T.isInfixOf lower (T.toLower (kfProposition f))
  356   in any (\f -> kfValidated f && (matchesWord f || matchesProp f)) allFruits

  358 -- | Clamp to [0, 1].  NaN inputs collapse to the lower bound so that
  359 -- 'rootStressSignal' (and any other consumer) cannot leak non-finite
  360 -- doubles into the calibration signal pipeline.  See regression G1.
  361 clampUnit :: Double -> Double
  362 clampUnit x
  363   | isNaN x   = 0.0
  364   | otherwise = max 0.0 (min 1.0 x)
