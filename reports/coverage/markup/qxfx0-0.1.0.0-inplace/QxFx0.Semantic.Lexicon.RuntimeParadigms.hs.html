   64 data ParadigmEntry = ParadigmEntry
   65   { pePos     :: !Text
   66   , peGender  :: !(Maybe Text)
   67   , peAnimacy :: !(Maybe Text)
   68   , peAspect  :: !(Maybe Text)
   69   , peTransitivity :: !(Maybe Text)
   70   , peForms   :: !(Map Text Text)
   71   } deriving stock (Eq, Show, Generic)
   72     deriving anyclass (NFData)

   74 instance FromJSON ParadigmEntry where
   75   parseJSON = genericParseJSON defaultOptions
   76     { fieldLabelModifier = \lbl -> case lbl of
   77         ('p':'e':rest) -> map toLower rest
   78         _              -> map toLower lbl
   79     }

   90 loadRuntimeParadigms :: BS.ByteString -> Either String RuntimeParadigms
   91 loadRuntimeParadigms bs = do
   92   paradigms <- eitherDecodeStrict bs
   93   pure RuntimeParadigms { rpMap = paradigms, rpExceptions = mempty }

   95 emptyRuntimeParadigms :: RuntimeParadigms
   96 emptyRuntimeParadigms = RuntimeParadigms mempty mempty

   98 -- | Load paradigms + exceptions from default paths. Pure IO — no global cache.
   99 loadDefaultRuntimeParadigms :: IO RuntimeParadigms
  100 loadDefaultRuntimeParadigms = do
  101   paradigms <- loadFile "resources/morphology/paradigms.json" "paradigms"
  102   exceptions <- loadFile "resources/morphology/exceptions.json" "exceptions"
  103   pure RuntimeParadigms { rpMap = paradigms, rpExceptions = exceptions }
  104   where
  105     loadFile path label = do
  106       exists <- doesFileExist path
  107       if not exists
  108         then do
  109           hPutStrLn stderr $ "[RuntimeParadigms] " <> label <> " file not found: " <> path
  110           pure mempty
  111         else do
  112           bs <- BS.readFile path
  113           case eitherDecodeStrict bs of
  114             Left err -> do
  115               hPutStrLn stderr $ "[RuntimeParadigms] " <> label <> " parse error: " <> err
  116               pure mempty
  117             Right mp -> pure (mp :: Map Text ParadigmEntry)

  123 data NounCase = Nom | Gen | Dat | Acc | Ins | Loc
  124   deriving stock (Eq, Ord, Show, Bounded, Enum)
  125 
  126 data Number = Sg | Pl
  127   deriving stock (Eq, Ord, Show, Bounded, Enum)
  128 
  129 nounCaseKey :: NounCase -> Number -> Text
  130 nounCaseKey c n = T.pack (show c <> show n)
  131 
  132 lookupNounForm :: RuntimeParadigms -> Text -> NounCase -> Number -> Maybe Text
  133 lookupNounForm rp lemma case_ number =
  134   lookupInMap (rpMap rp) <|> lookupInMap (rpExceptions rp)
  135   where
  136     lookupInMap mp = do
  137       entry <- M.lookup lemma mp
  138       guard (pePos entry == "Noun")
  139       case (case_, peGender entry, peAnimacy entry) of
  140         (Acc, Just "masc", Just "anim") ->
  141           M.lookup (nounCaseKey Gen number) (peForms entry)
  142         (Acc, Just "masc", Just "inan") ->
  143           M.lookup (nounCaseKey Nom number) (peForms entry)
  144         _ ->
  145           M.lookup (nounCaseKey case_ number) (peForms entry)
  146 
  147 lookupNounFormOr :: RuntimeParadigms -> Text -> NounCase -> Number -> Text -> Text
  148 lookupNounFormOr rp lemma case_ number fallback =
  149   fromMaybe fallback (lookupNounForm rp lemma case_ number)
  150 
  151 lemmaGender :: RuntimeParadigms -> Text -> Maybe Text
  152 lemmaGender rp lemma = lookupInMap (rpMap rp) <|> lookupInMap (rpExceptions rp)
  153   where
  154     lookupInMap mp = do
  155       entry <- M.lookup lemma mp
  156       peGender entry
  157 
  158 lemmaPos :: RuntimeParadigms -> Text -> Maybe Text
  159 lemmaPos rp lemma = lookupInMap (rpMap rp) <|> lookupInMap (rpExceptions rp)
  160   where
  161     lookupInMap mp = pePos <$> M.lookup lemma mp

  194 data AdjectiveForm
  195   = AdjFull NounCase Number (Maybe Gender)   -- full forms
  196   | AdjShort Gender                           -- short forms
  197   deriving stock (Eq, Ord, Show)
  198 
  199 data Gender = Masc | Femn | Neut
  200   deriving stock (Eq, Ord, Bounded, Enum)
  201 
  202 instance Show Gender where
  203   show Masc = "Masc"
  204   show Femn = "Fem"
  205   show Neut = "Neut"
  206 
  207 adjectiveFormKey :: AdjectiveForm -> Text
  208 adjectiveFormKey (AdjFull c Sg (Just g))  = T.pack (show c <> "Sg" <> show g)
  209 adjectiveFormKey (AdjFull c Pl _)         = T.pack (show c <> "Pl")
  210 adjectiveFormKey (AdjFull c Sg Nothing)   = T.pack (show c <> "Sg")
  211 adjectiveFormKey (AdjShort g)             = T.pack ("Short" <> show g)
  212 
  213 lookupAdjectiveForm :: RuntimeParadigms -> Text -> AdjectiveForm -> Maybe Text
  214 lookupAdjectiveForm rp lemma form = lookupInMap (rpMap rp) <|> lookupInMap (rpExceptions rp)
  215   where
  216     lookupInMap mp = do
  217       entry <- M.lookup lemma mp
  218       guard (pePos entry == "Adjective")
  219       M.lookup (adjectiveFormKey form) (peForms entry)
  220 
  221 lookupAdjectiveFormOr :: RuntimeParadigms -> Text -> AdjectiveForm -> Text -> Text
  222 lookupAdjectiveFormOr rp lemma form fallback =
  223   fromMaybe fallback (lookupAdjectiveForm rp lemma form)

  237 lookupExpression :: RuntimeParadigms -> Text -> Maybe Text
  238 lookupExpression rp lemma = do
  239   entry <- M.lookup lemma (rpMap rp)
  240   guard (pePos entry == "Expression")
  241   M.lookup "Base" (peForms entry)

  243 lookupMetaphor :: RuntimeParadigms -> Text -> Maybe Text -> Maybe (Text, Text)
  244 lookupMetaphor rp targetKeyword _mbase =
  245   let matches = filter (\(_, entry) -> pePos entry == "Metaphor" &&
  246         M.lookup "Target" (peForms entry) == Just targetKeyword) (M.toList (rpMap rp))
  247   in case matches of
  248        (k, entry) : _ ->
  249          case (M.lookup "Base" (peForms entry), M.lookup "Source" (peForms entry)) of
  250            (Just b, Just s) -> Just (b, s)
  251            _                -> Just (k, "образ")
  252        [] -> Nothing

  258 hasParadigm :: RuntimeParadigms -> Text -> Bool

  264 -- | Guess grammatical gender using the runtime paradigms database,
  265 -- falling back to suffix heuristics when the lemma is absent.
  266 guessGender :: RuntimeParadigms -> Text -> Maybe Gender
  267 guessGender rp t =
  268   case lemmaGender rp t of
  269     Just "femn" -> Just Femn
  270     Just "neut" -> Just Neut
  271     Just _      -> Just Masc
  272     Nothing     ->
  273       let w = T.toLower t
  274       in if T.isSuffixOf "а" w || T.isSuffixOf "я" w
  275            then Just Femn
  276          else if T.isSuffixOf "о" w || T.isSuffixOf "е" w
  277            then Just Neut
  278          else Just Masc
  279 
  280 -- | Like 'guessGender' but returns a default instead of 'Nothing'.
  281 guessGenderOr :: RuntimeParadigms -> Gender -> Text -> Gender
  282 guessGenderOr rp def t =
  283   case lemmaGender rp t of
  284     Just "femn" -> Femn
  285     Just "neut" -> Neut
  286     Just _      -> Masc
  287     Nothing     ->
  288       let w = T.toLower t
  289       in if T.isSuffixOf "а" w || T.isSuffixOf "я" w
  290            then Femn
  291          else if T.isSuffixOf "о" w || T.isSuffixOf "е" w
  292            then Neut
  293          else def

  310 inferNounForm :: RuntimeParadigms -> Text -> NounCase -> Number -> Maybe Text
  311 inferNounForm rp lemma targetCase number =
  312   let w = T.unpack (T.toLower lemma)
  313       g = guessGender rp lemma
  314       animacy = peAnimacy <$> M.lookup lemma (rpMap rp)
  315   in case (g, w) of
  316        (_, [])   -> Nothing
  317        (_, [_])  -> Nothing
  318        (Just Masc, cs) -> inferMasc cs animacy
  319        (Just Femn, cs) -> inferFemn cs
  320        (Just Neut, cs) -> inferNeut cs
  321        (_, _)     -> Nothing
  322      where
  323        safeLast cs = if null cs then Nothing else Just (cs !! (length cs - 1))
  324        safeInit cs = if null cs then [] else take (length cs - 1) cs
  325        safeLastEq cs c = case safeLast cs of Just c' -> c' == c; Nothing -> False
  326        isConsStem cs = not (null cs) &&
  327                        case safeLast cs of
  328                          Just l  -> l `notElem` ("аеёиоуыэюяьй" :: String)
  329                          Nothing -> False
  330        afterVelarOrSibilant s = case safeLast s of
  331          Just c  -> c `elem` ("кгхжшчщ" :: String)
  332          Nothing -> False
  333        decl base (sgG,sgD,sgA_anim,sgA_inan,sgI,sgL) (plG,plD,plA_anim,plA_inan,plI,plL) =
  334          let animacyLocal = case M.lookup lemma (rpMap rp) of
  335                Just entry -> peAnimacy entry
  336                Nothing    -> Nothing
  337              (sgA, plA) = case animacyLocal of
  338                Just "anim" -> (sgA_anim, plA_anim)
  339                Just "inan" -> (sgA_inan, plA_inan)
  340                _           -> (sgA_inan, plA_inan)  -- default to inanimate
  341          in case (number, targetCase) of
  342            (Sg, Gen) -> Just $ T.pack base <> T.pack sgG
  343            (Sg, Dat) -> Just $ T.pack base <> T.pack sgD
  344            (Sg, Acc) -> Just $ T.pack base <> T.pack sgA
  345            (Sg, Ins) -> Just $ T.pack base <> T.pack sgI
  346            (Sg, Loc) -> Just $ T.pack base <> T.pack sgL
  347            (Pl, Gen) -> Just $ T.pack base <> T.pack plG
  348            (Pl, Dat) -> Just $ T.pack base <> T.pack plD
  349            (Pl, Acc) -> Just $ T.pack base <> T.pack plA
  350            (Pl, Ins) -> Just $ T.pack base <> T.pack plI
  351            (Pl, Loc) -> Just $ T.pack base <> T.pack plL
  352            _         -> Nothing
  353        inferMasc cs mAnim
  354          | isConsStem cs =
  355              let plNom = if afterVelarOrSibilant cs then cs <> "и" else cs <> "ы"
  356                  plGen = if afterVelarOrSibilant cs then cs <> "ей" else cs <> "ов"
  357              in decl cs ("а","у","а",cs,"ом","е") (plGen,"ам",plGen,plNom,"ами","ах")
  358          | safeLastEq cs 'й' =
  359              let b = safeInit cs
  360              in decl b ("я","ю","я",cs,"ем","е") ("ев","ям","ев",b<>"и","ями","ях")
  361          | safeLastEq cs 'ь' =
  362              let b = safeInit cs
  363              in decl b ("я","ю","я",cs,"ём","и") ("ей","ям","ей",b<>"и","ями","ях")
  364          | otherwise = Nothing
  365        inferFemn cs
  366          | safeLastEq cs 'а' =
  367              let b = safeInit cs
  368                  yOrI = if afterVelarOrSibilant b then "и" else "ы"
  369              in case (number, targetCase) of
  370                   (Sg, Gen) -> Just $ T.pack b <> T.pack yOrI
  371                   (Sg, Dat) -> Just $ T.pack b <> "е"
  372                   (Sg, Acc) -> Just $ T.pack b <> "у"
  373                   (Sg, Ins) -> Just $ T.pack b <> "ой"
  374                   (Sg, Loc) -> Just $ T.pack b <> "е"
  375                   _         -> Nothing
  376          | safeLastEq cs 'я' =
  377              let b = safeInit cs
  378              in case (number, targetCase) of
  379                   (Sg, Gen) -> Just $ T.pack b <> "и"
  380                   (Sg, Dat) -> Just $ T.pack b <> "е"
  381                   (Sg, Acc) -> Just $ T.pack b <> "ю"
  382                   (Sg, Ins) -> Just $ T.pack b <> "ей"
  383                   (Sg, Loc) -> Just $ T.pack b <> "е"
  384                   _         -> Nothing
  385          | safeLastEq cs 'ь' =
  386              let b = safeInit cs
  387              in case (number, targetCase) of
  388                   (Sg, Gen) -> Just $ T.pack b <> "и"
  389                   (Sg, Dat) -> Just $ T.pack b <> "и"
  390                   (Sg, Acc) -> Just $ T.pack b <> "ь"
  391                   (Sg, Ins) -> Just $ T.pack b <> "ью"
  392                   (Sg, Loc) -> Just $ T.pack b <> "и"
  393                   _         -> Nothing
  394          | otherwise = Nothing
  395        inferNeut cs
  396          | safeLastEq cs 'о' =
  397              let b = safeInit cs
  398              in case (number, targetCase) of
  399                   (Sg, Gen) -> Just $ T.pack b <> "а"
  400                   (Sg, Dat) -> Just $ T.pack b <> "у"
  401                   (Sg, Acc) -> Just $ T.pack cs
  402                   (Sg, Ins) -> Just $ T.pack b <> "ом"
  403                   (Sg, Loc) -> Just $ T.pack b <> "е"
  404                   _         -> Nothing
  405          | safeLastEq cs 'е' =
  406              let b = safeInit cs
  407                  isIye = length cs >= 3 && cs !! (length cs - 2) == 'и'
  408              in if isIye
  409                 then let ib = take (length cs - 2) cs
  410                      in case (number, targetCase) of
  411                           (Sg, Gen) -> Just $ T.pack ib <> "ия"
  412                           (Sg, Dat) -> Just $ T.pack ib <> "ию"
  413                           (Sg, Acc) -> Just $ T.pack cs
  414                           (Sg, Ins) -> Just $ T.pack ib <> "ием"
  415                           (Sg, Loc) -> Just $ T.pack ib <> "ии"
  416                           (Pl, Gen) -> Just $ T.pack ib <> "ий"
  417                           (Pl, Dat) -> Just $ T.pack ib <> "иям"
  418                           (Pl, Acc) -> Just $ T.pack ib <> "ия"
  419                           (Pl, Ins) -> Just $ T.pack ib <> "иями"
  420                           (Pl, Loc) -> Just $ T.pack ib <> "иях"
  421                           _         -> Nothing
  422                 else case (number, targetCase) of
  423                        (Sg, Gen) -> Just $ T.pack b <> "я"
  424                        (Sg, Dat) -> Just $ T.pack b <> "ю"
  425                        (Sg, Acc) -> Just $ T.pack cs
  426                        (Sg, Ins) -> Just $ T.pack b <> "ем"
  427                        (Sg, Loc) -> Just $ T.pack b <> "е"
  428                        _         -> Nothing
  429          | otherwise = Nothing
