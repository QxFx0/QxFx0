   15 data PersistenceStage
   16   = StageStateBlobUpsert
   17   | StageSessionTouch
   18   | StageTurnQualityUpsert
   19   | StageShadowDivergenceInsert
   20   | StageRollbackTurnQuality
   21   | StageRollbackShadowDivergence
   22   | StageTxBegin
   23   | StageTxCommit
   24   | StageTxRollback
   25   | StageUnknown
   26   deriving stock (Eq, Show)
   27 
   28 renderPersistenceStage :: PersistenceStage -> Text
   29 renderPersistenceStage StageStateBlobUpsert       = "state_blob.upsert"
   30 renderPersistenceStage StageSessionTouch          = "session_touch.upsert"
   31 renderPersistenceStage StageTurnQualityUpsert     = "state_projection.upsert"
   32 renderPersistenceStage StageShadowDivergenceInsert = "shadow_divergence.upsert"
   33 renderPersistenceStage StageRollbackTurnQuality    = "state_projection.rollback"
   34 renderPersistenceStage StageRollbackShadowDivergence = "shadow_divergence.rollback"
   35 renderPersistenceStage StageTxBegin               = "tx_begin"
   36 renderPersistenceStage StageTxCommit              = "tx_commit"
   37 renderPersistenceStage StageTxRollback            = "tx_rollback"
   38 renderPersistenceStage StageUnknown                = "unknown"

   40 data PersistenceDiagnostic
   41   = PdSchemaMissingFields ![Text]
   42   | PdCorruptDecode
   43   | PdTransactionBeginFailed
   44   | PdTransactionCommitFailed
   45   | PdTransactionRollbackFailed
   46   | PdSaveFailed !PersistenceStage !(Maybe Text) !(Maybe Text)
   47   | PdRollbackFailed !PersistenceStage !(Maybe Text) !(Maybe Text)
   48   deriving stock (Eq, Show)
   49 
   50 data LoadStateResult
   51   = LoadStateMissing
   52   | LoadStateRestored !SystemState
   53   | LoadStateCorrupt ![PersistenceDiagnostic]
   54   deriving stock (Eq, Show)
   55 
   56 renderPersistenceDiagnostics :: [PersistenceDiagnostic] -> Text
   57 renderPersistenceDiagnostics = T.intercalate "; " . map renderOne
   58   where
   59     renderOne (PdSchemaMissingFields fields) =
   60       "state_schema_defaulted_fields:" <> T.intercalate "," fields
   61     renderOne PdCorruptDecode = "corrupt_decode"
   62     renderOne PdTransactionBeginFailed = "tx_begin_failed"
   63     renderOne PdTransactionCommitFailed = "tx_commit_failed"
   64     renderOne PdTransactionRollbackFailed = "tx_rollback_failed"
   65     renderOne (PdSaveFailed stage mTable mSqlite) =
   66       "save_failed stage=" <> renderPersistenceStage stage
   67       <> maybe "" (\t -> " table=" <> t) mTable
   68       <> maybe "" (\e -> " sqlite=\"" <> e <> "\"") mSqlite
   69     renderOne (PdRollbackFailed stage mTable mSqlite) =
   70       "rollback_failed stage=" <> renderPersistenceStage stage
   71       <> maybe "" (\t -> " table=" <> t) mTable
   72       <> maybe "" (\e -> " sqlite=\"" <> e <> "\"") mSqlite
