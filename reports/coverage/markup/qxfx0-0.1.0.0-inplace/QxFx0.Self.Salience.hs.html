    7 {-|
    8 Module      : QxFx0.Self.Salience
    9 Description : Phase-5 salience controller (pure).
   10 
   11 A typed, pure realisation of the salience controller pinned by
   12 @docs\/adr\/0010-salience-controller.md@.
   13 
   14 == What this module is
   15 
   16 The salience controller is the runtime component that decides,
   17 per turn, which of the two adjoint modes — 'Holistic' (right-
   18 hemispheric, generative) or 'Formal' (left-hemispheric,
   19 constraint-respecting) — should /lead/. Concretely it is a pure
   20 morphism
   21 
   22 @
   23 'Field' \\(\\times\\) 'ConatusEnergy' -> 'Salience'
   24 @
   25 
   26 producing a structured verdict that carries:
   27 
   28 * a continuous bias on @[0, 1]@ ('salienceHolisticBias'),
   29 * a confidence score on @[0, 1]@ ('salienceConfidence'),
   30 * a closed driver tag ('SalienceDriver') for trace.
   31 
   32 The verdict is then dispatched through the Phase-3 adjunction
   33 ('chooseBranch') to select which of two caller-supplied morphisms
   34 ('Holistic a -> b' or 'a -> Formal b') is actually evaluated.
   35 
   36 == What this module is /not/
   37 
   38 This module is the algebraic /shape/ of salience-driven dispatch.
   39 It does /not/ wire into the turn pipeline, does /not/ touch
   40 @Core.Intuition@ \/ @Core.ConsciousnessLoop@ \/ @RouteEffects@ \/
   41 @RenderStyle@ \/ @Recovery@. Those re-shapings are Phase-5.5 work
   42 and live in their own future PRs.
   43 
   44 == Decision rule (overview)
   45 
   46 @
   47 raw =
   48     weightResonance       * fieldResonance      f
   49   + weightAtmosphere      * atmosphereArousal (fieldAtmosphere f)
   50   - weightConsolidation   * fieldConsolidation  f      -- inverse
   51   + weightCounterfactual  * fieldCounterfactual f
   52   - weightFieldConfidence * fieldConfidence    f      -- inverse
   53 @
   54 
   55 Conatus gate: if @ceScalar < conatusGateThreshold@, the controller
   56 short-circuits to a 'PreferFormal' verdict with confidence @1.0@
   57 and driver 'DrivenByConatusGate'. Otherwise the raw score is
   58 squashed to @[0, 1]@ by a logistic and the dominant driver is
   59 the input whose absolute contribution to @raw@ was largest.
   60 
   61 The exact functional forms (sigmoid temperature, dispersion-based
   62 confidence) are implementation choices; the algebraic shape and
   63 its laws (totality, range, monotonicity, Conatus-gate priority,
   64 determinism) are the contract.
   65 
   66 == Anti-correlation discipline
   67 
   68 ADR-0007 mandates that "the non-leading mode listens but does not
   69 emit." This module enforces that operationally through three
   70 invariants:
   71 
   72 * 'chooseBranch' returns a single morphism, not a pair. Whichever
   73   branch the controller selects is the one whose value reaches the
   74   caller; the other branch is well-typed and may be evaluated for
   75   diagnostics, but its value does not flow into the output.
   76 * 'salienceVerdict' produces 'Tied' inside a small dead band
   77   around @0.5@, dispatched deterministically to the documented
   78   default ('PreferFormal' fallback) so that near-tied fields do
   79   not cause the system to flap between modes under noise.
   80 * The Conatus gate has uncontested priority: when
   81   @ceScalar < conatusGateThreshold@, no Field-derived signal
   82   contributes to the verdict.
   83 
   84 == Honest scope
   85 
   86 The default weights ('defaultSalienceWeights') are pinned to make
   87 the property tests pass and the operational mapping plausible.
   88 We do not claim they are calibrated. Calibration is Phase-7
   89 (lifeness gates) work.
   90 -}

  165 -- | The controller's verdict for a single turn.
  166 --
  167 -- Invariants (verified by 'Test.Suite.SelfSalience'):
  168 --
  169 -- * @0 \\<= salienceHolisticBias \\<= 1@
  170 -- * @0 \\<= salienceConfidence   \\<= 1@
  171 -- * Identical @(SalienceWeights, ConatusEnergy, Field)@ inputs
  172 --   produce identical 'Salience' values, including the discrete
  173 --   'salienceDriver' tag.
  174 data Salience = Salience
  175   { salienceHolisticBias :: !Double
  176     -- ^ In @[0, 1]@: @0@ = pure formal, @1@ = pure holistic.
  177   , salienceConfidence   :: !Double
  178     -- ^ In @[0, 1]@: @1@ = one driver decisively dominates,
  179     --   @0@ = contributions cancel.
  180   , salienceDriver       :: !SalienceDriver
  181     -- ^ Which input dominated the decision. Closed enum so it
  182     --   fits a trace record without further serialisation.
  183   }
  184   deriving stock (Eq, Show)

  186 -- | The dispatched form of a 'Salience' value.
  187 --
  188 -- 'Tied' falls inside the dead band defined by
  189 -- 'verdictThreshold'. Downstream code dispatches @Tied@ to the

  192   = PreferHolistic !Double  -- ^ Magnitude in @(0, 1]@.
  193   | PreferFormal   !Double  -- ^ Magnitude in @(0, 1]@.
  194   | Tied
  195   deriving stock (Eq, Show)

  201 -- | Tunable coefficients of the decision rule. Phase 5 ships
  202 -- 'defaultSalienceWeights'; Phase 7 (lifeness gates) is the
  203 -- calibration step.
  204 data SalienceWeights = SalienceWeights
  205   { weightResonance       :: !Double
  206     -- ^ Multiplier on 'fieldResonance'. Direction: positive
  207     --   (more resonance ⇒ more bias toward Holistic).
  208   , weightAtmosphere      :: !Double
  209     -- ^ Multiplier on @atmosphereArousal (fieldAtmosphere f)@.
  210     --   Direction: positive.
  211   , weightConsolidation   :: !Double
  212     -- ^ Multiplier on 'fieldConsolidation'. Direction: inverse
  213     --   (more consolidation ⇒ more bias toward Formal).
  214   , weightCounterfactual  :: !Double
  215     -- ^ Multiplier on 'fieldCounterfactual'. Direction: positive.
  216   , weightFieldConfidence :: !Double
  217     -- ^ Multiplier on 'fieldConfidence'. Direction: inverse
  218     --   (lower field confidence ⇒ more bias toward Formal, in
  219     --   keeping with "the system can\'t trust holistic signals;
  220     --   fall back to formal contracts").
  221   , conatusGateThreshold  :: !Double
  222     -- ^ If @ceScalar < this@, force 'PreferFormal' with full
  223     --   confidence and 'DrivenByConatusGate' driver.
  224   , verdictThreshold      :: !Double
  225     -- ^ Half-width of the dead band around @0.5@ within which
  226     --   the verdict is 'Tied'. In @[0, 0.5]@.
  227   , sigmoidTemperature    :: !Double
  228     -- ^ Slope parameter for the @sigmoid (raw \/ temperature)@
  229     --   squash. @1.0@ is the textbook default.
  230   }
  231   deriving stock (Eq, Show, Generic)
  232   deriving anyclass (NFData, ToJSON, FromJSON)

  297 -- | Compute the salience verdict for a single turn.
  298 --
  299 -- Total, deterministic, pure. Identical inputs produce identical
  300 -- 'Salience' values, including the discrete 'salienceDriver' tag.
  301 --
  302 -- The Conatus gate fires unconditionally when
  303 -- @ceScalar c \\< conatusGateThreshold w@, returning a
  304 -- 'PreferFormal'-shaped verdict with full confidence and
  305 -- 'DrivenByConatusGate'. Otherwise the rule of ADR-0010 §3 is
  306 -- applied.
  307 computeSalience :: SalienceWeights -> ConatusEnergy -> Field -> Salience
  308 computeSalience w ce f
  309   | ceScalar ce < conatusGateThreshold w =
  310       Salience
  311         { salienceHolisticBias = 0.0
  312         , salienceConfidence   = 1.0
  313         , salienceDriver       = DrivenByConatusGate
  314         }
  315   | otherwise =
  316       let cs = contributions w f
  317           raw =
  318             contribResonance      cs
  319             + contribAtmosphere     cs
  320             - contribConsolidation  cs
  321             + contribCounterfactual cs
  322             - contribFieldConfidence cs
  323           bias = sigmoid (raw / sigmoidTemperature w)
  324        in Salience
  325             { salienceHolisticBias = bias
  326             , salienceConfidence   = computeConfidence cs
  327             , salienceDriver       = dominantDriver cs
  328             }

  330 -- | Dispatch a 'Salience' to a 'SalienceVerdict' using the
  331 -- @verdictThreshold@ from the weights as the dead-band half-width.
  332 salienceVerdict :: SalienceWeights -> Salience -> SalienceVerdict
  333 salienceVerdict w s
  334   | bias > 0.5 + t = PreferHolistic bias
  335   | bias < 0.5 - t = PreferFormal   (1.0 - bias)
  336   | otherwise      = Tied
  337   where
  338     bias = salienceHolisticBias s
  339     t    = verdictThreshold w

  345 -- | Dispatch a precomputed 'SalienceVerdict' to one of two
  346 -- caller-supplied branches through the Phase-3 adjunction.
  347 --
  348 -- Both branches are always provided by the caller (the adjunction
  349 -- guarantees they are inter-translatable, see ADR-0008). The
  350 -- controller picks which one to /use/; the other remains a
  351 -- well-typed value the caller may evaluate for tracing.
  352 --
  353 -- 'Tied' is dispatched to the formal-first branch (the
  354 -- contract-driven mode is the safe default). This is the
  355 -- single-output-channel half of the anti-correlation discipline
  356 -- (ADR-0010 §5).
  357 --
  358 -- __Status (as of Phase 5.5e):__ 'chooseBranch' is now consumed by
  359 -- 'QxFx0.Core.TurnRender.Strategy.applySalienceToStyle' (the
  360 -- render-style dispatch point). This is the first production
  361 -- call site that threads a real 'Field' into the adjunction.
  362 -- Other channels (intuition flash strength, narrative fragment)
  363 -- still use direct @case salienceVerdict ... of@ rewrites
  364 -- because they are @Salience -> a -> a@ post-processors, not
  365 -- Field-parameterized dispatches.
  366 --
  367 -- Property-test coverage in @Test.Suite.SelfSalience@ exercises
  368 -- the adjunction laws on the type level; the runtime call site
  369 -- exercises them on the value level.
  370 chooseBranch
  371   :: SalienceVerdict
  372   -> (Holistic a -> b)   -- ^ Holistic-first branch.
  373   -> (a -> Formal b)     -- ^ Formal-first branch.
  374   -> (Holistic a -> b)
  375 chooseBranch v holistic formal = case v of
  376   PreferHolistic _ -> holistic
  377   PreferFormal   _ -> rightAdjunct formal
  378   Tied             -> rightAdjunct formal

  480 -- ---------------------------------------------------------------------------
  481 -- Phase-B bounded post-commitment adaptation
  482 -- ---------------------------------------------------------------------------
  483 
  484 -- | Bounded adaptation of salience weights.
  485 --
  486 -- * @signal@ in @[-1, 1]@: positive reinforces Holistic-bias weights,
  487 --   negative reinforces Formal-bias weights.
  488 -- * Learning rate capped at @0.02@ per turn.
  489 -- * All adapted weights clamped to @[0.0, 2.0]@.
  490 -- * Anti-drift: total deviation from 'defaultSalienceWeights'
  491 --   per weight cannot exceed @1.0@.
  492 --
  493 -- Empirical signal generation is deferred to Phase 7; this
  494 -- function provides the bounded mechanics so that future
  495 -- calibration only needs to change the caller-supplied signal.
  496 adaptSalienceWeights :: Double -> SalienceWeights -> SalienceWeights
  497 adaptSalienceWeights rawSignal w =
  498   let lr        = 0.02
  499       signal    = max (-1.0) (min 1.0 rawSignal)
  500       delta     = signal * lr
  501       def       = defaultSalienceWeights
  502       clamped x = max 0.0 (min 2.0 x)
  503       -- Anti-drift only applies when we are actually adapting;
  504       -- signal=0 must be identity to respect the gating contract.
  505       bounded target x =
  506         let adapted = clamped (x + delta)
  507         in if abs delta < 1e-12
  508              then x
  509              else max (target - 1.0) (min (target + 1.0) adapted)
  510   in w
  511        { weightResonance       = bounded (weightResonance       def) (weightResonance       w)
  512        , weightAtmosphere      = bounded (weightAtmosphere      def) (weightAtmosphere      w)
  513        , weightConsolidation   = bounded (weightConsolidation   def) (weightConsolidation   w)
  514        , weightCounterfactual  = bounded (weightCounterfactual  def) (weightCounterfactual  w)
  515        , weightFieldConfidence = bounded (weightFieldConfidence def) (weightFieldConfidence w)
  516        }

  518 -- ---------------------------------------------------------------------------
  519 -- Internals
  520 -- ---------------------------------------------------------------------------
  521 
  522 -- | Per-driver signed contributions to the raw score.
  523 --
  524 -- The sign of each contribution matches the rule direction
  525 -- (ADR-0010 §3): positive contributions push toward Holistic,
  526 -- negative toward Formal. Confidence and driver attribution use
  527 -- the absolute values.
  528 data Contributions = Contributions
  529   { contribResonance       :: !Double
  530   , contribAtmosphere      :: !Double
  531   , contribConsolidation   :: !Double
  532   , contribCounterfactual  :: !Double
  533   , contribFieldConfidence :: !Double
  534   }
  535 
  536 contributions :: SalienceWeights -> Field -> Contributions
  537 contributions w f = Contributions
  538   { contribResonance       = weightResonance       w * unResonance       (fieldResonance      f)
  539   , contribAtmosphere      = weightAtmosphere      w * atmosphereArousal (fieldAtmosphere     f)
  540   , contribConsolidation   = weightConsolidation   w * unConsolidation   (fieldConsolidation  f)
  541   , contribCounterfactual  = weightCounterfactual  w * unCounterfactual  (fieldCounterfactual f)
  542   , contribFieldConfidence = weightFieldConfidence w * unFieldConfidence (fieldConfidence     f)
  543   }
  544 
  545 -- | Closed-form logistic squash.
  546 sigmoid :: Double -> Double
  547 sigmoid x = 1.0 / (1.0 + exp (negate x))
  548 
  549 -- | The driver with the largest absolute contribution.
  550 --
  551 -- All-zero contributions return 'DrivenByDefault' (no signal
  552 -- exceeded the threshold of any kind).
  553 dominantDriver :: Contributions -> SalienceDriver
  554 dominantDriver cs
  555   | totalMagnitude == 0.0 = DrivenByDefault
  556   | otherwise             = fst (foldr1 pickLarger cs')
  557   where
  558     cs' =
  559       [ (DrivenByResonance      , abs (contribResonance       cs))
  560       , (DrivenByAtmosphere     , abs (contribAtmosphere      cs))
  561       , (DrivenByConsolidation  , abs (contribConsolidation   cs))
  562       , (DrivenByCounterfactual , abs (contribCounterfactual  cs))
  563       , (DrivenByFieldConfidence, abs (contribFieldConfidence cs))
  564       ]
  565     totalMagnitude = sum (map snd cs')
  566     pickLarger a@(_, va) b@(_, vb) = if va >= vb then a else b
  567 
  568 -- | Confidence as 1 minus normalised dispersion of the
  569 -- contributions.
  570 --
  571 -- Concretely: let @dominant = max |c_i|@ and
  572 -- @other = sum |c_i| − dominant@; with @n = 5@ contributions,
  573 -- @confidence = max 0 (1 − other / ((n−1) × dominant))@. This
  574 -- yields @1.0@ when one contribution dominates and the others
  575 -- are zero, and @0.0@ when all five contributions are equal in
  576 -- magnitude. All-zero contributions return @1.0@ (no signal to
  577 -- disagree on).
  578 computeConfidence :: Contributions -> Double
  579 computeConfidence cs
  580   | dominant == 0.0 = 1.0
  581   | otherwise       = max 0.0 (1.0 - other / (4.0 * dominant))
  582   where
  583     magnitudes =
  584       [ abs (contribResonance       cs)
  585       , abs (contribAtmosphere      cs)
  586       , abs (contribConsolidation   cs)
  587       , abs (contribCounterfactual  cs)
  588       , abs (contribFieldConfidence cs)
  589       ]
  590     dominant = maximum magnitudes
  591     other    = sum magnitudes - dominant
