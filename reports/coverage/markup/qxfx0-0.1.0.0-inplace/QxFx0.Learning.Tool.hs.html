   38 -- | Domain of expertise an external tool can cover.
   39 data ToolDomain
   40   = DomainSalience
   41     -- ^ Tool can provide salience-weight calibration data.
   42   | DomainKeyword
   43     -- ^ Tool can supply keyword / ontology coverage.
   44   | DomainLexicon
   45     -- ^ Tool can extend morphology or lexicon entries.
   46   | DomainGeneral
   47     -- ^ General-purpose reasoning (fallback, lower priority).
   48   deriving stock (Eq, Show, Generic)
   49     deriving anyclass (NFData, FromJSON, ToJSON)
   50 
   51 -- | Typed profile of an external knowledge source.
   52 data ExternalTool = ExternalTool
   53   { etName        :: !Text
   54     -- ^ Human-readable identifier (e.g. "local-calibration-script").
   55   , etDomain      :: !ToolDomain
   56     -- ^ Primary domain this tool covers.
   57   , etReliability :: !Double
   58     -- ^ Historical reliability in [0, 1].  Higher = more trustworthy.
   59   , etValidatable :: !Bool
   60     -- ^ Whether outputs from this tool can be run through the
   61     --   verify→simulate gate (WP4).  Non-validatable tools are
   62     --   rejected unless no validatable alternative exists.
   63   }
   64   deriving stock (Eq, Show, Generic)
   65     deriving anyclass (NFData, FromJSON, ToJSON)
   66 
   67 -- | Map a 'LearningNeed' to the 'ToolDomain' it requires.
   68 needDomain :: LearningNeed -> Maybe ToolDomain
   69 needDomain NeedSalienceCalibration = Just DomainSalience
   70 needDomain NeedKeywordEnrichment   = Just DomainKeyword
   71 needDomain NeedLexiconExtension    = Just DomainLexicon
   72 needDomain NeedNone                = Nothing
   73 
   74 -- | Select the best external tool for a given 'LearningNeed'.
   75 --
   76 -- Rules:
   77 -- 1. If 'NeedNone', no tool is selected.
   78 -- 2. Filter tools whose domain matches the need.
   79 -- 3. Prefer 'etValidatable = True'; if none, fall back to the
   80 --    highest-reliability non-validatable tool.
   81 -- 4. Among the candidate set, pick the one with highest
   82 --    'etReliability'.  Ties are broken by declaration order
   83 --    (stable, deterministic).
   84 selectTool :: LearningNeed -> [ExternalTool] -> Maybe ExternalTool
   85 selectTool need tools = do
   86   domain <- needDomain need
   87   let matching = filter ((== domain) . etDomain) tools
   88   if null matching
   89     then do
   90       -- Fallback: if no domain-specific tool, try DomainGeneral
   91       let general = filter ((== DomainGeneral) . etDomain) tools
   92       pickBest general
   93     else pickBest matching
   94   where
   95     pickBest candidates
   96       | null candidates = Nothing
   97       | otherwise       =
   98           let validatable = filter etValidatable candidates
   99               pool = if null validatable then candidates else validatable
  100           in maximumByReliability pool
  101 
  102     -- | Total: returns 'Nothing' only for an empty list.
  103     maximumByReliability :: [ExternalTool] -> Maybe ExternalTool
  104     maximumByReliability [] = Nothing
  105     maximumByReliability (x:xs) = Just (go x xs)
  106       where
  107         go best [] = best
  108         go best (y:ys)
  109           | etReliability y > etReliability best = go y ys
  110           | otherwise                            = go best ys
  111 
  112 -- | WP5: variant of 'selectTool' that overlays dynamic reliability
  113 -- from a runtime-maintained map.  If a tool's name is present in the
  114 -- map, the map value overrides 'etReliability' for selection
  115 -- purposes only; the static profile is not mutated.
  116 selectToolWithReliability
  117   :: LearningNeed
  118   -> M.Map Text Double   -- ^ dynamic reliability overrides
  119   -> [ExternalTool]
  120   -> Maybe ExternalTool
  121 selectToolWithReliability need relMap tools =
  122   let tools' = map (\t -> case M.lookup (etName t) relMap of
  123                             Just r  -> t { etReliability = clamp01 r }
  124                             Nothing -> t)
  125                  tools
  126   in selectTool need tools'
  127   where
  128     clamp01 x = max 0.0 (min 1.0 x)
  129 
  130 -- | WP5: update tool reliability after an outcome.
  131 --
  132 -- * Acceptance → +0.05 (capped at 1.0)
  133 -- * Rejection  → -0.10 (floored at 0.0)
  134 --
  135 -- These deltas are chosen so that 3 consecutive rejections drop a
  136 -- perfectly reliable tool below 0.7, making it less attractive than
  137 -- a fresh alternative.
  138 updateToolReliability :: Text -> Bool -> M.Map Text Double -> M.Map Text Double
  139 updateToolReliability toolName accepted relMap =
  140   let current = M.findWithDefault 0.5 toolName relMap
  141       delta = if accepted then 0.05 else (-0.10)
  142       updated = clamp01 (current + delta)
  143   in M.insert toolName updated relMap
  144   where
  145     clamp01 x = max 0.0 (min 1.0 x)

  147 -- | Render an 'ExternalTool' to a short telemetry tag.
  148 renderTool :: ExternalTool -> Text
  149 renderTool t = T.concat
  150   [ etName t
  151   , "|domain="
  152   , renderDomain (etDomain t)
  153   , "|reliability="
  154   , T.pack (show (etReliability t))
  155   , "|validatable="
  156   , if etValidatable t then "1" else "0"
  157   ]
  158 
  159 renderDomain :: ToolDomain -> Text
  160 renderDomain DomainSalience = "salience"
  161 renderDomain DomainKeyword    = "keyword"
  162 renderDomain DomainLexicon    = "lexicon"
  163 renderDomain DomainGeneral    = "general"
