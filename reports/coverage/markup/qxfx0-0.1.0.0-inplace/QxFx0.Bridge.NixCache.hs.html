   19 data NixResult
   20   = NixSuccess !Text
   21   | NixError !Text
   22   deriving stock (Eq, Show)
   23 
   24 data CacheEntry = CacheEntry
   25   { ceResult :: !NixResult
   26   , ceTimestamp :: !UTCTime
   27   } deriving stock (Eq, Show)
   28 
   29 data NixCache = NixCache
   30   { ncRef :: !(IORef (Map Text CacheEntry))
   31   , ncTTL :: !Double
   32   , ncMaxEntries :: !Int
   33   }
   34 
   35 newNixCache :: Double -> Int -> IO NixCache
   36 newNixCache ttl maxEntries = do
   37   ref <- newIORef M.empty
   38   return NixCache { ncRef = ref, ncTTL = ttl, ncMaxEntries = maxEntries }
   39 
   40 nixCacheLookup :: NixCache -> Text -> IO (Maybe NixResult)
   41 nixCacheLookup cache key = do
   42   now <- getCurrentTime
   43   m <- readIORef (ncRef cache)
   44   case M.lookup key m of
   45     Nothing -> return Nothing
   46     Just entry ->
   47       let age = diffUTCTime now (ceTimestamp entry)
   48       in if realToFrac age > ncTTL cache
   49          then return Nothing
   50          else return (Just (ceResult entry))
   51 
   52 nixCacheInsert :: NixCache -> Text -> NixResult -> IO ()
   53 nixCacheInsert cache key result = do
   54   now <- getCurrentTime
   55   let entry = CacheEntry result now
   56   atomicModifyIORef' (ncRef cache) $ \m ->
   57     let m' = M.insert key entry m
   58     in if M.size m' > ncMaxEntries cache
   59        then (evictToHalf (ncMaxEntries cache) m', ())
   60        else (m', ())
   61 
   62 evictToHalf :: Int -> Map Text CacheEntry -> Map Text CacheEntry
   63 evictToHalf maxEntries m
   64   | M.size m <= 100 = m
   65   | otherwise =
   66       let entries = M.toList m
   67           sorted = sortBy (\(_,a) (_,b) -> compare (ceTimestamp a) (ceTimestamp b)) entries
   68           target = maxEntries `div` 2
   69           keep = drop (length sorted - target) sorted
   70       in M.fromList keep

   72 checkWithCache :: NixCache -> FilePath -> Text -> Double -> Double -> IO NixGuardStatus
   73 checkWithCache cache nixPath concept agency tension = do
   74   let key = concept <> "|" <> renderKeyPart agency <> "|" <> renderKeyPart tension
   75   mResult <- nixCacheLookup cache key
   76   case mResult of
   77     Just (NixSuccess "true")  -> return Allowed
   78     Just (NixSuccess "false") -> return (Blocked "constitution blocked (cached)")
   79     Just (NixSuccess _)       -> return (Unavailable "cache payload mismatch")
   80     Just (NixError _)         -> do
   81       status <- checkConstitution nixPath concept agency tension
   82       case status of
   83         Allowed      -> nixCacheInsert cache key (NixSuccess "true")
   84         Blocked _    -> nixCacheInsert cache key (NixSuccess "false")
   85         Unavailable r -> nixCacheInsert cache key (NixError r)
   86       return status
   87     Nothing -> do
   88       status <- checkConstitution nixPath concept agency tension
   89       case status of
   90         Allowed      -> nixCacheInsert cache key (NixSuccess "true")
   91         Blocked _    -> nixCacheInsert cache key (NixSuccess "false")
   92         Unavailable r -> nixCacheInsert cache key (NixError r)
   93       return status
   94   where
   95     renderKeyPart :: Double -> Text
   96     renderKeyPart = T.pack . show
   97 
   98 cachedNixEval :: NixCache -> FilePath -> Text -> Double -> Double -> IO NixGuardStatus
   99 cachedNixEval = checkWithCache
