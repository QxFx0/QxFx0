   16 finalizeForce :: IllocutionaryForce -> Text -> Text
   17 finalizeForce IFAsk = ensureQuestion
   18 finalizeForce IFAssert = ensureSentence
   19 finalizeForce IFOffer = ensureSentence
   20 finalizeForce IFConfront = ensureSentence

   23 ensureQuestion :: Text -> Text
   24 ensureQuestion t =
   25   let trimmed = T.strip t
   26   in if T.null trimmed
   27        then "?"
   28        else case T.unsnoc trimmed of
   29          Just (_, '?') -> trimmed
   30          Just (rest, '.') -> rest <> "?"
   31          _ -> trimmed <> "?"
   32 
   33 ensureSentence :: Text -> Text
   34 ensureSentence t =
   35   let trimmed = T.strip t
   36   in if T.null trimmed
   37        then "."
   38        else case T.unsnoc trimmed of
   39          Just (_, '.') -> trimmed
   40          Just (rest, '?') -> rest <> "."
   41          Just (_, '!') -> trimmed
   42          _ -> trimmed <> "."
   43 
   44 fmtPct :: Double -> Text
   45 fmtPct x = T.pack (show (fromIntegral (round (x * 100) :: Int) :: Int)) <> "%"
   46 
   47 textShow :: Show a => a -> Text
   48 textShow = T.pack . show
