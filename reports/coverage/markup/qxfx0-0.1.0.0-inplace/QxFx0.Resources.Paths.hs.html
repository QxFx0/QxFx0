   43 resolveResourcePaths :: IO ResourcePaths
   44 resolveResourcePaths = do
   45   mResourceRootEnv <- lookupEnv "QXFX0_RESOURCE_ROOT"
   46   mRootEnv <- lookupEnv "QXFX0_ROOT"
   47   let explicitRootConfigured = maybe False (const True) mResourceRootEnv || maybe False (const True) mRootEnv
   48   rootResolution <- resolveOptionalResourceRoot
   49   mResourceRoot <-
   50     case rootResolution of
   51       Left err -> throwResourceError (T.pack err)
   52       Right mRoot -> pure mRoot
   53   migrationSql <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["migrations/001_initial_schema.sql"]
   54   morphologyPrepositional <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["resources/morphology/prepositional.json"]
   55   _morphologyGenitive <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["resources/morphology/genitive.json"]
   56   _morphologyNominative <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["resources/morphology/nominative.json"]
   57   _morphologyQuality <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["resources/morphology/lexicon_quality.json"]
   58   nixGuard <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["semantics/concepts.nix"]
   59   agdaSpec <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/R5Core.agda"]
   60   agdaSnapshot <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/r5-snapshot.tsv"]
   61   datalogPath <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/datalog/semantic_rules.dl", "semantic_rules.dl"]
   62   schemaSql <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/sql/schema.sql"]
   63   seedClusters <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/sql/seed_clusters.sql"]
   64   seedTemplates <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/sql/seed_templates.sql"]
   65   seedIdentity <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/sql/seed_identity.sql"]
   66   let migrationDir = takeDirectory migrationSql
   67       morphologyDir = takeDirectory morphologyPrepositional
   68       resourceDir = maybe (deriveResourceRoot schemaSql migrationDir morphologyDir) id mResourceRoot
   69   pure ResourcePaths
   70     { rpResourceDir = resourceDir
   71     , rpNixGuard = nixGuard
   72     , rpMigrationDir = migrationDir
   73     , rpMorphologyDir = morphologyDir
   74     , rpAgdaSpec = agdaSpec
   75     , rpAgdaSnapshot = agdaSnapshot
   76     , rpDatalogRules = datalogPath
   77     , rpSchemaSql = schemaSql
   78     , rpSeedClusters = seedClusters
   79     , rpSeedTemplates = seedTemplates
   80     , rpSeedIdentity = seedIdentity
   81     }

   83 findResourceDir :: IO FilePath
   84 findResourceDir = do
   85   mResourceRoot <- lookupEnv "QXFX0_RESOURCE_ROOT"
   86   mRoot <- lookupEnv "QXFX0_ROOT"
   87   let mExplicitRoot =
   88         case fmap normalise mResourceRoot of
   89           Just root -> Just root
   90           Nothing -> normalise <$> mRoot
   91   case mExplicitRoot of
   92     Just root -> do
   93       ok <- isResourceRoot root
   94       if ok
   95         then pure root
   96         else throwResourceError ("QXFX0 resource root is invalid: " <> T.pack root)
   97     Nothing -> discoverResourceRoot
   98 
   99 discoverInstalledDataDir :: IO (Maybe FilePath)
  100 discoverInstalledDataDir = do
  101   result <- tryIO getDataDir
  102   pure $
  103     case result of
  104       Left _ -> Nothing
  105       Right path -> Just (normalise path)
  106 
  107 discoverResourceRoot :: IO FilePath
  108 discoverResourceRoot = do
  109   cwd <- getCurrentDirectory
  110   exePath <- getExecutablePath
  111   mDataDir <- discoverInstalledDataDir
  112   let roots = nub (maybeToList mDataDir ++ ancestors 12 cwd ++ ancestors 16 (takeDirectory exePath))
  113   firstOk <- pickFirstRoot roots
  114   case firstOk of
  115     Just root -> pure root
  116     Nothing ->
  117       throwResourceError
  118         ( "Could not locate QxFx0 resource root."
  119         <> " cwd=" <> T.pack cwd
  120         <> ", executable=" <> T.pack exePath
  121         <> ", searched=" <> T.pack (renderSearchRoots roots)
  122         )
  123 
  124 pickFirstRoot :: [FilePath] -> IO (Maybe FilePath)
  125 pickFirstRoot [] = pure Nothing
  126 pickFirstRoot (root:rest) = do
  127   ok <- isResourceRoot root
  128   if ok then pure (Just root) else pickFirstRoot rest
  129 
  130 isResourceRoot :: FilePath -> IO Bool
  131 isResourceRoot root = do
  132   hasMigrations <- doesDirectoryExist (root </> "migrations")
  133   hasMorphology <- doesDirectoryExist (root </> "resources" </> "morphology")
  134   hasConcepts <- doesFileExist (root </> "semantics" </> "concepts.nix")
  135   hasSchema <- doesFileExist (root </> "spec" </> "sql" </> "schema.sql")
  136   pure (hasMigrations && hasMorphology && (hasConcepts || hasSchema))
  137 
  138 ancestors :: Int -> FilePath -> [FilePath]
  139 ancestors maxDepth start = go 0 (normalise start)
  140   where
  141     go n p
  142       | n > maxDepth = []
  143       | otherwise =
  144           let parent = takeDirectory p
  145           in if parent == p then [p] else p : go (n + 1) parent

  156 renderSearchRoots :: [FilePath] -> String
  157 renderSearchRoots roots =
  158   let preview = take 8 roots
  159       suffix = if length roots > length preview then ", ..." else ""
  160   in "[" ++ intercalate ", " preview ++ suffix ++ "]"

  162 resolveOptionalResourceRoot :: IO (Either String (Maybe FilePath))
  163 resolveOptionalResourceRoot = do
  164   mResourceRoot <- lookupEnv "QXFX0_RESOURCE_ROOT"
  165   mRoot <- lookupEnv "QXFX0_ROOT"
  166   let mExplicitRoot =
  167         case fmap normalise mResourceRoot of
  168           Just root -> Just root
  169           Nothing -> normalise <$> mRoot
  170   case mExplicitRoot of
  171     Just root -> do
  172       ok <- isResourceRoot root
  173       pure $
  174         if ok
  175           then Right (Just root)
  176           else Left ("QXFX0 resource root is invalid: " ++ root)
  177     Nothing -> do
  178       discovered <- tryQxFx0 discoverResourceRoot
  179       pure $
  180         case discovered of
  181           Right root -> Right (Just root)
  182           Left _ -> Right Nothing

  184 resolveResourceFile :: Bool -> Maybe FilePath -> [FilePath] -> IO FilePath
  185 resolveResourceFile allowDataFallback mRoot candidates = do
  186   let rootedCandidates = case mRoot of
  187         Just root -> map (normalise . (root </>)) candidates
  188         Nothing -> []
  189   rootExisting <- pickExistingPath rootedCandidates
  190   case rootExisting of
  191     Just path -> pure path
  192     Nothing | allowDataFallback -> do
  193       dataCandidates <- mapM resolveDataFileCandidate candidates
  194       case mapMaybe id dataCandidates of
  195         (path:_) -> pure path
  196         [] ->
  197           case rootedCandidates of
  198             (path:_) -> pure path
  199             [] ->
  200               case candidates of
  201                 (path:_) -> pure (normalise path)
  202                 [] -> throwResourceError "resolveResourceFile: empty candidate list"
  203     Nothing ->
  204       case rootedCandidates of
  205         (path:_) -> pure path
  206         [] ->
  207           case candidates of
  208             (path:_) -> pure (normalise path)
  209             [] -> throwResourceError "resolveResourceFile: empty candidate list"
  210 
  211 resolveDataFileCandidate :: FilePath -> IO (Maybe FilePath)
  212 resolveDataFileCandidate relativePath = do
  213   dataResult <- tryIO (getDataFileName relativePath)
  214   case dataResult of
  215     Left _ -> pure Nothing
  216     Right path -> do
  217       exists <- doesFileExist path
  218       pure $ if exists then Just (normalise path) else Nothing
  219 
  220 pickExistingPath :: [FilePath] -> IO (Maybe FilePath)
  221 pickExistingPath [] = pure Nothing
  222 pickExistingPath (path:rest) = do
  223   exists <- doesFileExist path
  224   if exists then pure (Just path) else pickExistingPath rest
  225 
  226 deriveResourceRoot :: FilePath -> FilePath -> FilePath -> FilePath
  227 deriveResourceRoot schemaSql migrationDir morphologyDir =
  228   let fromSchema = takeDirectory (takeDirectory (takeDirectory schemaSql))
  229       fromMigrations = takeDirectory migrationDir
  230       fromMorphology = takeDirectory (takeDirectory morphologyDir)
  231   in normalise $
  232       case () of
  233         _ | fromSchema /= "." -> fromSchema
  234           | fromMigrations /= "." -> fromMigrations
  235           | fromMorphology /= "." -> fromMorphology
  236           | otherwise -> "."
  237 
  238 throwResourceError :: Text -> IO a
  239 throwResourceError = throwQxFx0 . RuntimeInitError
