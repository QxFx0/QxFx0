   48 resolveResourcePaths :: IO ResourcePaths
   49 resolveResourcePaths = do
   50   mResourceRootEnv <- lookupEnv "QXFX0_RESOURCE_ROOT"
   51   mRootEnv <- lookupEnv "QXFX0_ROOT"
   52   mConceptsOverride <- lookupEnv "QXFX0_CONCEPTS_PATH"
   53   let explicitRootConfigured = maybe False (const True) mResourceRootEnv || maybe False (const True) mRootEnv
   54   rootResolution <- resolveOptionalResourceRoot
   55   mResourceRoot <-
   56     case rootResolution of
   57       Left err -> throwResourceError (T.pack err)
   58       Right mRoot -> pure mRoot
   59   migrationSql <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["migrations/001_initial_schema.sql"]
   60   morphologyPrepositional <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["resources/morphology/prepositional.json"]
   61   _morphologyGenitive <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["resources/morphology/genitive.json"]
   62   _morphologyNominative <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["resources/morphology/nominative.json"]
   63   _morphologyQuality <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["resources/morphology/lexicon_quality.json"]
   64   nixGuard <- resolveConceptsPath (not explicitRootConfigured) mResourceRoot mConceptsOverride
   65   agdaSpec <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/R5Core.agda"]
   66   agdaSnapshot <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/r5-snapshot.tsv"]
   67   datalogPath <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/datalog/semantic_rules.dl", "semantic_rules.dl"]
   68   schemaSql <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/sql/schema.sql"]
   69   seedClusters <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/sql/seed_clusters.sql"]
   70   seedTemplates <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/sql/seed_templates.sql"]
   71   seedIdentity <- resolveResourceFile (not explicitRootConfigured) mResourceRoot ["spec/sql/seed_identity.sql"]
   72   let migrationDir = takeDirectory migrationSql
   73       morphologyDir = takeDirectory morphologyPrepositional
   74       resourceDir = maybe (deriveResourceRoot schemaSql migrationDir morphologyDir) id mResourceRoot
   75   pure ResourcePaths
   76     { rpResourceDir = resourceDir
   77     , rpNixGuard = nixGuard
   78     , rpMigrationDir = migrationDir
   79     , rpMorphologyDir = morphologyDir
   80     , rpAgdaSpec = agdaSpec

   82     , rpDatalogRules = datalogPath
   83     , rpSchemaSql = schemaSql
   84     , rpSeedClusters = seedClusters

   87     }

   95         case fmap normalise mResourceRootEnv of

   99     case mExplicitRoot of
  100       Just root -> do

  106         rootResolution <- resolveOptionalResourceRoot
  107         case rootResolution of
  108           Left err -> throwResourceError (T.pack err)

  115   agdaSpec <- resolveReadinessFile mResourceRoot ["spec/R5Core.agda"]
  116   agdaSnapshot <- resolveReadinessFile mResourceRoot ["spec/r5-snapshot.tsv"]

  119   seedTemplates <- resolveReadinessFile mResourceRoot ["spec/sql/seed_templates.sql"]
  120   seedIdentity <- resolveReadinessFile mResourceRoot ["spec/sql/seed_identity.sql"]
  121   let migrationDir = takeDirectory migrationSql
  122       resourceDir = maybe (deriveResourceRoot schemaSql migrationDir morphologyDir) id mResourceRoot

  124     { rpResourceDir = resourceDir

  137 findResourceDir :: IO FilePath
  138 findResourceDir = do
  139   mResourceRoot <- lookupEnv "QXFX0_RESOURCE_ROOT"
  140   mRoot <- lookupEnv "QXFX0_ROOT"
  141   let mExplicitRoot =
  142         case fmap normalise mResourceRoot of
  143           Just root -> Just root
  144           Nothing -> normalise <$> mRoot
  145   case mExplicitRoot of
  146     Just root -> do
  147       ok <- isResourceRoot root
  148       if ok
  149         then pure root
  150         else throwResourceError ("QXFX0 resource root is invalid: " <> T.pack root)
  151     Nothing -> discoverResourceRoot

  153 discoverInstalledDataDir :: IO (Maybe FilePath)
  154 discoverInstalledDataDir = do
  155   result <- tryIO getDataDir
  156   pure $
  157     case result of
  158       Left _ -> Nothing
  159       Right path -> Just (normalise path)
  160 
  161 discoverResourceRoot :: IO FilePath
  162 discoverResourceRoot = do
  163   cwd <- getCurrentDirectory
  164   exePath <- getExecutablePath
  165   mDataDir <- discoverInstalledDataDir
  166   let roots = nub (maybeToList mDataDir ++ ancestors 12 cwd ++ ancestors 16 (takeDirectory exePath))
  167   firstOk <- pickFirstRoot roots
  168   case firstOk of
  169     Just root -> pure root
  170     Nothing ->
  171       throwResourceError
  172         ( "Could not locate QxFx0 resource root."
  173         <> " cwd=" <> T.pack cwd
  174         <> ", executable=" <> T.pack exePath
  175         <> ", searched=" <> T.pack (renderSearchRoots roots)
  176         )

  178 pickFirstRoot :: [FilePath] -> IO (Maybe FilePath)
  179 pickFirstRoot [] = pure Nothing
  180 pickFirstRoot (root:rest) = do
  181   ok <- isResourceRoot root
  182   if ok then pure (Just root) else pickFirstRoot rest
  183 
  184 isResourceRoot :: FilePath -> IO Bool
  185 isResourceRoot root = do
  186   hasMigrations <- doesDirectoryExist (root </> "migrations")
  187   hasMorphology <- doesDirectoryExist (root </> "resources" </> "morphology")
  188   hasConcepts <- doesFileExist (root </> "semantics" </> "concepts.nix")
  189   hasSchema <- doesFileExist (root </> "spec" </> "sql" </> "schema.sql")
  190   pure (hasMigrations && hasMorphology && (hasConcepts || hasSchema))
  191 
  192 ancestors :: Int -> FilePath -> [FilePath]
  193 ancestors maxDepth start = go 0 (normalise start)
  194   where
  195     go n p
  196       | n > maxDepth = []
  197       | otherwise =
  198           let parent = takeDirectory p
  199           in if parent == p then [p] else p : go (n + 1) parent

  210 resolveDatalogRuleCandidates :: IO [FilePath]
  211 resolveDatalogRuleCandidates = do
  212   mResourceRootEnv <- lookupEnv "QXFX0_RESOURCE_ROOT"
  213   mRootEnv <- lookupEnv "QXFX0_ROOT"
  214   let explicitRootConfigured = maybe False (const True) mResourceRootEnv || maybe False (const True) mRootEnv
  215       relCandidates = ["spec/datalog/semantic_rules.dl", "semantic_rules.dl"]
  216       mExplicitRoot =
  217         case fmap normalise mResourceRootEnv of
  218           Just root -> Just root
  219           Nothing -> normalise <$> mRootEnv
  220   mRoot <-
  221     case mExplicitRoot of
  222       Just root -> pure (Just root)
  223       Nothing -> do
  224         rootResolution <- resolveOptionalResourceRoot
  225         pure $ case rootResolution of
  226           Right resolved -> resolved
  227           Left _ -> Nothing
  228   let rootedCandidates = maybe [] (\root -> map (normalise . (root </>)) relCandidates) mRoot
  229   dataCandidates <-
  230     if explicitRootConfigured
  231       then pure []
  232       else mapMaybe id <$> mapM resolveDataFileCandidate relCandidates
  233   pure (nub (rootedCandidates <> dataCandidates))

  235 resolveHttpRuntimeScriptPath :: IO FilePath
  236 resolveHttpRuntimeScriptPath = do
  237   mExplicit <- lookupEnv "QXFX0_HTTP_RUNTIME"
  238   case mExplicit of
  239     Just path -> do
  240       let normalized = normalise path
  241       exists <- doesFileExist normalized
  242       if exists
  243         then validateExplicitHttpRuntimeScript normalized
  244         else throwResourceError ("QXFX0_HTTP_RUNTIME points to missing file: " <> T.pack normalized)
  245     Nothing -> do
  246       exePath <- getExecutablePath
  247       dataFileResult <- tryIO (getDataFileName "scripts/http_runtime.py")
  248       resourcePathsResult <- tryQxFx0 resolveResourcePaths
  249       let candidates =
  250             [ eitherToMaybe (normalise <$> dataFileResult)
  251             , eitherToMaybe (normalise . (</> "scripts" </> "http_runtime.py") . rpResourceDir <$> resourcePathsResult)
  252             , Just (normalise (takeDirectory exePath </> "scripts" </> "http_runtime.py"))
  253             ]
  254       resolveExistingHttpRuntimeScript candidates

  256 resolveConceptsPath :: Bool -> Maybe FilePath -> Maybe FilePath -> IO FilePath
  257 resolveConceptsPath allowDataFallback mRoot mOverride =
  258   case fmap normalise mOverride of
  259     Just overridePath -> do
  260       exists <- doesFileExist overridePath
  261       if exists
  262         then pure overridePath
  263         else throwResourceError ("QXFX0_CONCEPTS_PATH points to missing file: " <> T.pack overridePath)
  264     Nothing -> resolveResourceFile allowDataFallback mRoot ["semantics/concepts.nix"]

  267 resolveReadinessConceptsPath mRoot mOverride =
  268   case fmap normalise mOverride of
  269     Just overridePath -> pure overridePath
  270     Nothing -> resolveReadinessFile mRoot ["semantics/concepts.nix"]

  272 renderSearchRoots :: [FilePath] -> String
  273 renderSearchRoots roots =
  274   let preview = take 8 roots
  275       suffix = if length roots > length preview then ", ..." else ""
  276   in "[" ++ intercalate ", " preview ++ suffix ++ "]"

  278 resolveOptionalResourceRoot :: IO (Either String (Maybe FilePath))
  279 resolveOptionalResourceRoot = do
  280   mResourceRoot <- lookupEnv "QXFX0_RESOURCE_ROOT"
  281   mRoot <- lookupEnv "QXFX0_ROOT"
  282   let mExplicitRoot =
  283         case fmap normalise mResourceRoot of
  284           Just root -> Just root
  285           Nothing -> normalise <$> mRoot
  286   case mExplicitRoot of
  287     Just root -> do
  288       ok <- isResourceRoot root
  289       pure $
  290         if ok
  291           then Right (Just root)
  292           else Left ("QXFX0 resource root is invalid: " ++ root)
  293     Nothing -> do
  294       discovered <- tryQxFx0 discoverResourceRoot
  295       pure $
  296         case discovered of
  297           Right root -> Right (Just root)
  298           Left _ -> Right Nothing

  300 resolveResourceFile :: Bool -> Maybe FilePath -> [FilePath] -> IO FilePath
  301 resolveResourceFile allowDataFallback mRoot candidates = do
  302   let rootedCandidates = case mRoot of
  303         Just root -> map (normalise . (root </>)) candidates
  304         Nothing -> []
  305   rootExisting <- pickExistingPath rootedCandidates
  306   case rootExisting of
  307     Just path -> pure path
  308     Nothing | allowDataFallback -> do
  309       dataCandidates <- mapM resolveDataFileCandidate candidates
  310       case mapMaybe id dataCandidates of
  311         (path:_) -> pure path
  312         [] -> throwMissingResourceError candidates mRoot
  313     Nothing ->
  314       throwMissingResourceError candidates mRoot
  315 
  316 resolveReadinessFile :: Maybe FilePath -> [FilePath] -> IO FilePath
  317 resolveReadinessFile mRoot candidates =
  318   case mRoot of
  319     Just root ->
  320       case candidates of
  321         candidate:_ -> pure (normalise (root </> candidate))
  322         [] -> throwMissingResourceError [] mRoot
  323     Nothing -> resolveResourceFile True mRoot candidates
  324 
  325 resolveDataFileCandidate :: FilePath -> IO (Maybe FilePath)
  326 resolveDataFileCandidate relativePath = do
  327   dataResult <- tryIO (getDataFileName relativePath)
  328   case dataResult of
  329     Left _ -> pure Nothing
  330     Right path -> do
  331       exists <- doesFileExist path
  332       pure $ if exists then Just (normalise path) else Nothing
  333 
  334 pickExistingPath :: [FilePath] -> IO (Maybe FilePath)
  335 pickExistingPath [] = pure Nothing
  336 pickExistingPath (path:rest) = do
  337   exists <- doesFileExist path
  338   if exists then pure (Just path) else pickExistingPath rest
  339 
  340 resolveExistingHttpRuntimeScript :: [Maybe FilePath] -> IO FilePath
  341 resolveExistingHttpRuntimeScript candidates =
  342   pick [path | Just path <- candidates]
  343   where
  344     pick [] =
  345       throwResourceError
  346         ( "Could not locate http_runtime.py; checked: "
  347         <> T.pack (intercalate ", " [path | Just path <- candidates])
  348         <> ". Set QXFX0_HTTP_RUNTIME to an explicit script path."
  349         )
  350     pick (path:rest) = do
  351       exists <- doesFileExist path
  352       if exists then validateAutoResolvedHttpRuntimeScript path else pick rest
  353 
  354 validateExplicitHttpRuntimeScript :: FilePath -> IO FilePath
  355 validateExplicitHttpRuntimeScript path = do
  356   exePath <- getExecutablePath
  357   canonicalPath <- canonicalizePath path
  358   let scriptName = takeFileName canonicalPath
  359   if scriptName /= "http_runtime.py"
  360     then throwResourceError ("QXFX0_HTTP_RUNTIME must point to http_runtime.py, got: " <> T.pack canonicalPath)
  361     else do
  362       trustedRoots <- resolveTrustedHttpRuntimeScriptRoots exePath
  363       trustedMatches <- mapM (`isPathWithin` canonicalPath) trustedRoots
  364       if or trustedMatches
  365         then pure canonicalPath
  366         else throwResourceError ("QXFX0_HTTP_RUNTIME points outside trusted roots: " <> T.pack canonicalPath)
  367 
  368 validateAutoResolvedHttpRuntimeScript :: FilePath -> IO FilePath
  369 validateAutoResolvedHttpRuntimeScript path = do
  370   exePath <- getExecutablePath
  371   canonicalPath <- canonicalizePath path
  372   trustedRoots <- resolveTrustedHttpRuntimeScriptRoots exePath
  373   trustedMatches <- mapM (`isPathWithin` canonicalPath) trustedRoots
  374   if or trustedMatches
  375     then pure canonicalPath
  376     else
  377       throwResourceError
  378         ( "auto-resolved http_runtime.py outside trusted roots: " <> T.pack canonicalPath
  379         <> "; trusted roots: " <> T.pack (intercalate ", " trustedRoots)
  380         <> ". Set QXFX0_HTTP_RUNTIME to an explicit script path."
  381         )
  382 
  383 resolveTrustedHttpRuntimeScriptRoots :: FilePath -> IO [FilePath]
  384 resolveTrustedHttpRuntimeScriptRoots exePath = do
  385   dataScriptResult <- tryIO (getDataFileName "scripts/http_runtime.py")
  386   mResourceRoot <- lookupEnv "QXFX0_RESOURCE_ROOT"
  387   let roots =
  388         [ takeDirectory exePath
  389         , takeDirectory exePath </> "scripts"
  390         ]
  391         ++ maybe [] (\root -> [normalise root, normalise root </> "scripts"]) mResourceRoot
  392         ++ either (const []) (\path -> [takeDirectory path]) dataScriptResult
  393   existing <- filterM doesDirectoryExist roots
  394   canonical <- mapM canonicalizePath existing
  395   pure (nub canonical)

  397 eitherToMaybe :: Either a b -> Maybe b
  398 eitherToMaybe (Left _) = Nothing
  399 eitherToMaybe (Right value) = Just value

  401 throwMissingResourceError :: [FilePath] -> Maybe FilePath -> IO a
  402 throwMissingResourceError candidates mRoot =
  403   throwResourceError
  404     ( "resolveResourceFile: no existing candidate found"
  405     <> maybe "" (\root -> " for resource root=" <> T.pack root) mRoot
  406     <> " candidates=" <> T.pack (show candidates)
  407     )

  409 deriveResourceRoot :: FilePath -> FilePath -> FilePath -> FilePath
  410 deriveResourceRoot schemaSql migrationDir morphologyDir =
  411   let fromSchema = takeDirectory (takeDirectory (takeDirectory schemaSql))
  412       fromMigrations = takeDirectory migrationDir
  413       fromMorphology = takeDirectory (takeDirectory morphologyDir)
  414   in normalise $
  415       case () of
  416         _ | fromSchema /= "." -> fromSchema
  417           | fromMigrations /= "." -> fromMigrations
  418           | fromMorphology /= "." -> fromMorphology
  419           | otherwise -> "."
  420 
  421 throwResourceError :: Text -> IO a
  422 throwResourceError = throwQxFx0 . RuntimeInitError
