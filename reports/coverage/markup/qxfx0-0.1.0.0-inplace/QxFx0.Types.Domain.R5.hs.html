   99 data R5Verdict = R5Verdict
  100   { r5Family :: !CanonicalMoveFamily
  101   , r5Force :: !IllocutionaryForce
  102   , r5Clause :: !ClauseForm
  103   , r5Layer :: !SemanticLayer
  104   , r5Warranted :: !WarrantedMoveMode
  105   } deriving stock (Eq, Ord, Show, Read, Generic)
  106     deriving anyclass (NFData)
  107 
  108 instance ToJSON R5Verdict where
  109   toJSON r5 =
  110     object
  111       [ "family" .= r5Family r5
  112       , "force" .= r5Force r5
  113       , "clause" .= r5Clause r5
  114       , "layer" .= r5Layer r5
  115       , "warranted" .= r5Warranted r5
  116       ]
  117 
  118 instance FromJSON R5Verdict where
  119   parseJSON = withObject "R5Verdict" $ \o ->
  120     R5Verdict
  121       <$> o .: "family"
  122       <*> o .: "force"
  123       <*> o .: "clause"
  124       <*> o .: "layer"
  125       <*> o .: "warranted"

  181 forceForFamily :: CanonicalMoveFamily -> IllocutionaryForce
  182 forceForFamily CMGround = IFAssert
  183 forceForFamily CMDefine = IFAssert
  184 forceForFamily CMDistinguish = IFAssert

  186 forceForFamily CMDescribe = IFAssert
  187 forceForFamily CMPurpose = IFAssert
  188 forceForFamily CMHypothesis = IFAsk
  189 forceForFamily CMRepair = IFOffer
  190 forceForFamily CMContact = IFContact
  191 forceForFamily CMAnchor = IFAssert
  192 forceForFamily CMClarify = IFAsk
  193 forceForFamily CMDeepen = IFAsk
  194 forceForFamily CMConfront = IFConfront
  195 forceForFamily CMNextStep = IFOffer

  197 clauseFormForIF :: IllocutionaryForce -> ClauseForm
  198 clauseFormForIF IFAsk = Interrogative
  199 clauseFormForIF IFAssert = Declarative
  200 clauseFormForIF IFOffer = Hortative
  201 clauseFormForIF IFConfront = Imperative

  236 mkVerdict :: CanonicalMoveFamily -> R5Verdict
  237 mkVerdict fam =
  238   R5Verdict
  239     { r5Family = fam
  240     , r5Force = forceForFamily fam
  241     , r5Clause = clauseFormForIF (forceForFamily fam)
