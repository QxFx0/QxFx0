   93 data R5Verdict = R5Verdict
   94   { r5Family :: !CanonicalMoveFamily
   95   , r5Force :: !IllocutionaryForce
   96   , r5Clause :: !ClauseForm
   97   , r5Layer :: !SemanticLayer
   98   , r5Warranted :: !WarrantedMoveMode
   99   } deriving stock (Eq, Ord, Show, Read, Generic)
  100     deriving anyclass (NFData)
  101 
  102 instance ToJSON R5Verdict where
  103   toJSON r5 =
  104     object
  105       [ "family" .= r5Family r5
  106       , "force" .= r5Force r5
  107       , "clause" .= r5Clause r5
  108       , "layer" .= r5Layer r5
  109       , "warranted" .= r5Warranted r5
  110       ]
  111 
  112 instance FromJSON R5Verdict where
  113   parseJSON = withObject "R5Verdict" $ \o ->
  114     R5Verdict
  115       <$> o .: "family"
  116       <*> o .: "force"
  117       <*> o .: "clause"
  118       <*> o .: "layer"
  119       <*> o .: "warranted"

  121 forceForFamily :: CanonicalMoveFamily -> IllocutionaryForce
  122 forceForFamily CMGround = IFAssert
  123 forceForFamily CMDefine = IFAssert
  124 forceForFamily CMDistinguish = IFAssert
  125 forceForFamily CMReflect = IFAssert
  126 forceForFamily CMDescribe = IFAssert
  127 forceForFamily CMPurpose = IFAssert
  128 forceForFamily CMHypothesis = IFAsk
  129 forceForFamily CMRepair = IFOffer
  130 forceForFamily CMContact = IFContact
  131 forceForFamily CMAnchor = IFAssert
  132 forceForFamily CMClarify = IFAsk
  133 forceForFamily CMDeepen = IFAsk
  134 forceForFamily CMConfront = IFConfront
  135 forceForFamily CMNextStep = IFOffer
  136 
  137 clauseFormForIF :: IllocutionaryForce -> ClauseForm
  138 clauseFormForIF IFAsk = Interrogative
  139 clauseFormForIF IFAssert = Declarative
  140 clauseFormForIF IFOffer = Hortative
  141 clauseFormForIF IFConfront = Imperative
  142 clauseFormForIF IFContact = Declarative
  143 
  144 layerForFamily :: CanonicalMoveFamily -> SemanticLayer
  145 layerForFamily CMGround = ContentLayer
  146 layerForFamily CMDefine = ContentLayer
  147 layerForFamily CMDistinguish = ContentLayer
  148 layerForFamily CMReflect = MetaLayer
  149 layerForFamily CMDescribe = ContentLayer
  150 layerForFamily CMPurpose = ContentLayer
  151 layerForFamily CMHypothesis = MetaLayer
  152 layerForFamily CMRepair = MetaLayer
  153 layerForFamily CMContact = ContactLayer
  154 layerForFamily CMAnchor = ContentLayer
  155 layerForFamily CMClarify = MetaLayer
  156 layerForFamily CMDeepen = MetaLayer
  157 layerForFamily CMConfront = MetaLayer
  158 layerForFamily CMNextStep = MetaLayer

  160 warrantedForFamily :: CanonicalMoveFamily -> WarrantedMoveMode
  161 warrantedForFamily CMGround = AlwaysWarranted
  162 warrantedForFamily CMDefine = AlwaysWarranted
  163 warrantedForFamily CMDistinguish = ConditionallyWarranted
  164 warrantedForFamily CMReflect = AlwaysWarranted
  165 warrantedForFamily CMDescribe = AlwaysWarranted
  166 warrantedForFamily CMPurpose = ConditionallyWarranted
  167 warrantedForFamily CMHypothesis = ConditionallyWarranted
  168 warrantedForFamily CMRepair = AlwaysWarranted
  169 warrantedForFamily CMContact = AlwaysWarranted
  170 warrantedForFamily CMAnchor = AlwaysWarranted
  171 warrantedForFamily CMClarify = ConditionallyWarranted
  172 warrantedForFamily CMDeepen = ConditionallyWarranted

  176 mkVerdict :: CanonicalMoveFamily -> R5Verdict
  177 mkVerdict fam =
  178   R5Verdict
  179     { r5Family = fam
  180     , r5Force = forceForFamily fam
  181     , r5Clause = clauseFormForIF (forceForFamily fam)
  182     , r5Layer = layerForFamily fam
  183     , r5Warranted = warrantedForFamily fam
  184     }
