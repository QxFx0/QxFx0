    3 {-|
    4 Module      : QxFx0.Self.Invariants
    5 Description : Pure invariant checks over 'SelfBlanket' snapshots.
    6 
    7 Two check operators are provided:
    8 
    9 * 'checkInitialBlanket' for the post-bootstrap state (no previous
   10   blanket exists yet); and
   11 * 'checkBlanketTransition' for any commit-time transition between
   12   a previous and a current blanket.
   13 
   14 Both functions are pure and return the (possibly empty) list of
   15 'BlanketViolation' values produced. An empty list means the
   16 invariants hold; a non-empty list is a structural self-identity
   17 failure ('QxFx0.ExceptionPolicy.IdentityRupture').
   18 
   19 See @docs\/THEORY.md@ §4.1.
   20 -}
   21 module QxFx0.Self.Invariants
   22   ( checkInitialBlanket
   23   , checkBlanketTransition
   24   , renderBlanketViolations
   25   ) where
   26 
   27 import Data.Text (Text)
   28 import qualified Data.Text as T
   29 
   30 import QxFx0.Self.Types
   31   ( BlanketViolation (..)
   32   , SelfBlanket (..)
   33   , renderBlanketViolation
   34   )
   35 
   36 -- | Check the invariants that must hold for a freshly bootstrapped
   37 -- system, with no prior 'SelfBlanket' to compare against.
   38 --
   39 -- Currently enforces:
   40 --
   41 -- * 'BlanketEmptySession' — the session identifier must be non-empty
   42 -- * 'BlanketEmptyMorphology' — at least one morphology entry must
   43 --   be loaded
   44 --
   45 -- Returns @[]@ on success.
   46 checkInitialBlanket :: SelfBlanket -> [BlanketViolation]
   47 checkInitialBlanket sb =
   48   [ BlanketEmptySession    | T.null (sbSessionId sb)              ]
   49   <> [ BlanketEmptyMorphology | sbMorphologyTotalSize sb <= 0       ]
   50 
   51 -- | Check the invariants that must hold across a commit-time
   52 -- transition from a previous 'SelfBlanket' to a current one.
   53 --
   54 -- Currently enforces (in addition to all checks from
   55 -- 'checkInitialBlanket' applied to the current blanket):
   56 --
   57 -- * 'BlanketSessionChanged' — the session identifier must not change
   58 -- * 'BlanketTurnRegressed' — turn count must be non-decreasing
   59 -- * 'BlanketIdentityErased' — identity-claim count must be
   60 --   non-decreasing
   61 --
   62 -- Returns @[]@ on success.
   63 checkBlanketTransition :: SelfBlanket -> SelfBlanket -> [BlanketViolation]
   64 checkBlanketTransition prev cur =
   65   checkInitialBlanket cur
   66     <> [ BlanketSessionChanged (sbSessionId prev) (sbSessionId cur)
   67        | sbSessionId prev /= sbSessionId cur
   68        ]
   69     <> [ BlanketTurnRegressed (sbTurnCount prev) (sbTurnCount cur)
   70        | sbTurnCount cur < sbTurnCount prev
   71        ]
   72     <> [ BlanketIdentityErased (sbIdentityClaimsCount prev) (sbIdentityClaimsCount cur)
   73        | sbIdentityClaimsCount cur < sbIdentityClaimsCount prev
   74        ]
   75 
   76 -- | Render a list of violations to a single semicolon-separated
   77 -- diagnostic line, suitable for an
   78 -- 'QxFx0.ExceptionPolicy.IdentityRupture' payload.
   79 renderBlanketViolations :: [BlanketViolation] -> Text
   80 renderBlanketViolations = T.intercalate "; " . map renderBlanketViolation
