   11 Validates a parsed 'KnowledgeFruitPayload' before it is allowed to
   12 enter the knowledge tree.  Fail-closed: any ambiguity or conflict

   44 -- | Parsed payload from an external tool before validation.

   53   }
   54   deriving stock (Eq, Show, Generic)

   57 -- | Simplified morphology description from an LLM.
   58 -- We do NOT require full 'LexemeForm' fidelity at ingest; the
   59 -- sandbox gate (WP5) will exercise the real paradigm later.
   60 data MorphologyPayload = MorphologyPayload
   61   { mpGender     :: !(Maybe Text)
   62   , mpDeclension :: !(Maybe Text)
   63   , mpCases      :: !(Maybe (M.Map Text Text))
   64     -- ^ e.g. {"nom":"книга","gen":"книги",...}
   65   }
   66   deriving stock (Eq, Show, Generic)
   67     deriving anyclass (NFData, FromJSON, ToJSON)

   69 -- | Typed validation-failure taxonomy.
   70 -- Each constructor carries enough detail for telemetry and operator
   71 -- diagnosis without exposing the raw LLM response.
   72 data ValidationError
   73   = VeEmptyWord
   74     -- ^ Word field is missing or empty after stripping.
   75   | VeEmptyDefinition
   76     -- ^ Definition field is missing or empty.
   77   | VeDefinitionTooShort !Int !Int
   78     -- ^ actual word count, required minimum
   79   | VeSemanticallyEmpty
   80     -- ^ Definition consists only of stop-words / filler (e.g. "a thing").
   81   | VeMorphologyParseFailure !Text
   82     -- ^ Morphology JSON or map structure is malformed.
   83   | VeLexiconConflict !Text
   84     -- ^ Word+definition pair duplicates an existing lexicon entry.
   85   | VeInvalidSchema !Text
   86     -- ^ Top-level JSON schema mismatch; carries the expected field name.
   87   | VeInvalidField !Text !Text
   88     -- ^ field name, reason
   89   deriving stock (Eq, Show, Generic)
   90     deriving anyclass (NFData, FromJSON, ToJSON)
   91 
   92 -- | Minimum word count for a definition to be considered substantive.
   93 minDefinitionWords :: Int
   94 minDefinitionWords = 3
   95 
   96 -- | Validate a payload against the current morphology/lexicon baseline.
   97 -- Fail-closed: any ambiguity or conflict results in Left.
   98 validateFruitPayload
   99   :: KnowledgeFruitPayload
  100   -> MorphologyData          -- ^ current runtime morphology (for conflict detection)
  101   -> Either ValidationError KnowledgeFruitPayload
  102 validateFruitPayload payload morphData = do
  103   -- 1. Word presence
  104   when (T.null (T.strip (kfpWord payload))) (Left VeEmptyWord)
  105 
  106   -- 2. Definition substance
  107   let defStripped = T.strip (kfpDefinition payload)
  108   when (T.null defStripped) (Left VeEmptyDefinition)
  109   let wordCount = length (T.words defStripped)
  110   when (wordCount < minDefinitionWords)
  111     (Left (VeDefinitionTooShort wordCount minDefinitionWords))
  112 
  113   -- 2a. Semantic emptiness check: definition must not be only stop-words.
  114   let meaningfulWords = filter (not . isStopWord) (T.words defStripped)
  115   when (null meaningfulWords) (Left VeSemanticallyEmpty)
  116 
  117   -- 3. Morphology structural sanity (not deep paradigm fidelity)
  118   case kfpMorphology payload of
  119     Nothing -> pure ()
  120     Just mp -> do
  121       -- If cases are provided, every value must be non-empty.
  122       case mpCases mp of
  123         Nothing -> pure ()
  124         Just cMap ->
  125           forM_ (M.toList cMap) (\(k,v) ->
  126             when (T.null (T.strip v))
  127               (Left (VeInvalidField (T.concat ["morphology.case.", k]) "empty surface form")))
  128 
  129   -- 4. Lexicon/Morphology conflict check
  130   let word = T.toLower (T.strip (kfpWord payload))
  131   when (M.member word (mdNominative morphData))
  132     (Left (VeLexiconConflict (T.concat ["nominative collision: ", word])))
  133   when (M.member word (mdFormsBySurface morphData))
  134     (Left (VeLexiconConflict (T.concat ["surface collision: ", word])))
  135   case kfpMorphology payload >>= mpCases of
  136     Nothing -> pure ()
  137     Just cMap ->
  138       forM_ (M.toList cMap) (\(_caseKey, surfaceRaw) -> do
  139         let surface = T.toLower (T.strip surfaceRaw)
  140         when (not (T.null surface) && M.member surface (mdFormsBySurface morphData))
  141           (Left (VeLexiconConflict (T.concat ["surface collision: ", surface])))
  142         )
  143 
  144   pure payload
  145 
  146 -- | Minimal English stop-word list for semantic-emptiness detection.
  147 -- Expandable; kept small to avoid false positives in multilingual text.
  148 isStopWord :: Text -> Bool
  149 isStopWord w = T.toLower w `elem`
  150   [ "a", "an", "the", "is", "are", "was", "were", "be", "been"
  151   , "being", "have", "has", "had", "do", "does", "did", "will"
  152   , "would", "could", "should", "may", "might", "must", "shall"
  153   , "can", "need", "dare", "ought", "used", "to", "of", "in"
  154   , "for", "on", "with", "at", "by", "from", "as", "into"
  155   , "through", "during", "before", "after", "above", "below"
  156   , "between", "under", "and", "but", "or", "yet", "so"
  157   , "if", "because", "although", "though", "while", "where"
  158   , "when", "that", "which", "who", "whom", "whose", "what"
  159   , "this", "these", "those", "i", "you", "he", "she", "it"
  160   , "we", "they", "me", "him", "her", "us", "them", "my"
  161   , "your", "his", "its", "our", "their", "thing", "something"
  162   ]

  164 renderValidationError :: ValidationError -> Text
  165 renderValidationError err =
  166   case err of
  167     VeEmptyWord -> "empty_word"
  168     VeEmptyDefinition -> "empty_definition"
  169     VeDefinitionTooShort actual req ->
  170       T.concat ["definition_too_short: got ", T.pack (show actual), " words, need ", T.pack (show req)]
  171     VeSemanticallyEmpty -> "semantically_empty"
  172     VeMorphologyParseFailure t -> T.concat ["morphology_parse: ", t]
  173     VeLexiconConflict t -> T.concat ["lexicon_conflict: ", t]
  174     VeInvalidSchema t -> T.concat ["invalid_schema: ", t]
  175     VeInvalidField f r -> T.concat ["invalid_field: ", f, " -> ", r]
