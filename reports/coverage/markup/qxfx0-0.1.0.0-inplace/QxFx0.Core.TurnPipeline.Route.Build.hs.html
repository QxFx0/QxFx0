   64 buildRouteTurnPlan :: ShadowPolicy -> SystemState -> TurnInput -> TurnSignals -> RouteEffectPlan -> RouteEffectResults -> TurnPlan
   65 buildRouteTurnPlan shadowPolicy ss ti ts effectPlan effectResults =
   66   let atomSet = tiAtomSet ti
   67       intuitPosterior = tsIntuitPosterior ts
   68       rd = rsRoutingDecision (repStatic effectPlan)
   69       sc =
   70         computeShadowContext
   71           (rerShadowResult effectResults)
   72           (tiFrame ti)
   73           (tiNewTrace ti)
   74           intuitPosterior
   75           (tiEmbeddingQuality ti)
   76           (tiEmbSimilarity ti)
   77           (tsApiHealthy ts)
   78       shadowResolution0 = resolveShadowFamily shadowPolicy (rdFamily rd) sc
   79       -- WP2 (GAP2): bounded shadow-veto anti-loop.
   80       -- If the shadow gate has triggered more than 'maxVetosPerWindow'
   81       -- times within 'vetoWindowTurns', bypass the veto to prevent
   82       -- infinite oscillation and emit exhaustion telemetry.
   83       maxVetosPerWindow = 3
   84       vetoWindowTurns = 10
   85       currentTurn = ssTurnCount ss
   86       oldVetoState = ssShadowVetoState ss
   87       windowExpired = currentTurn - svsWindowStart oldVetoState > vetoWindowTurns
   88       (vetoCount, windowStart) =
   89         if windowExpired
   90           then (0, currentTurn)
   91           else (svsCount oldVetoState, svsWindowStart oldVetoState)
   92       (shadowResolution, vetoExhausted, newVetoCount, newWindowStart) =
   93         if srGateTriggered shadowResolution0
   94           then if vetoCount >= maxVetosPerWindow
   95                  then (shadowResolution0 { srGateTriggered = False }, True, vetoCount, windowStart)
   96                  else (shadowResolution0, False, vetoCount + 1, windowStart)
   97           else (shadowResolution0, False, vetoCount, windowStart)
   98       family0 = srEffectiveFamily shadowResolution
   99       (family, _, _) = familySenseBundle family0 (tiDialogueCommitmentLedger ti) (tiDialoguePhase ti) (tiDialogueThread ti) (tiSenseVector ti)
  100       recoveryBonus =
  101         legitimacyRecoveryBonus
  102           (scShadowStatus sc == ShadowMatch && not (scShadowHasDivergence sc))
  103           (rdStrategyFamily rd == Just family)
  104       newEgo = rdNewEgo rd
  105       renderStrategy = rdRenderStrategy rd
  106       renderStyle = adjustRenderStyleForSpeechPolicy (ssSpeechPolicyState ss) (rdRenderStyle rd)
  107       preRenderTruthStatus = derivePreRenderTruthContractStatus ti sc shadowResolution family
  108       rmpBase = buildRMPWithTruthContract preRenderTruthStatus family (tiDialogueCommitmentLedger ti) (tiDialoguePhase ti) (tiDialogueThread ti) (tiFrame ti) (tiSenseVector ti) (tiBestTopic ti) newEgo (tiNewTrace ti) (tiNixAvailable ti)
  109       rmp0 = applyRenderStrategyWithTruthContract preRenderTruthStatus family renderStrategy rmpBase
  110       rcp0 = (buildRCP family rmp0) {rcpStyle = renderStyle}
  111       rmp1 = modulateRMPWithNarrative (tsNarrativeFragment ts) rmp0
  112       rcp1 =
  113         case tsFlash ts of
  114           Just flash -> modulateRCPWithFlash (ifOverridesAll flash) rcp0
  115           Nothing -> rcp0
  116       agdaStatus = rerAgdaStatus effectResults
  117       agdaOk = agdaVerificationReady agdaStatus
  118       legitPreAgda = min 1.0 (scAdjustedBaseLegit sc + recoveryBonus)
  119       legitInput =
  120         if agdaOk
  121           then legitPreAgda
  122           else max 0.0 (legitPreAgda - agdaVerificationPenalty)
  123       (legitScore, rmpAfterLegit0, rcpFinal0, finalFamily0, _finalForce0) =
  124         applyLegitimacyToPlans legitInput family rmp1 rcp1 renderStyle
  125       lockedDiagnosticFamily =
  126         case diagnosticPropositionFamily (ipfPropositionType (tiFrame ti)) of
  127           Just diagnosticFamily
  128             | not (srGateTriggered shadowResolution)
  129                 && finalFamily0 /= CMRepair ->
  130                 Just diagnosticFamily
  131           _ ->
  132             Nothing
  133       finalFamily = maybe finalFamily0 id lockedDiagnosticFamily
  134       (finalFamily', finalSensePlan, finalMicroPlan) = familySenseBundle finalFamily (tiDialogueCommitmentLedger ti) (tiDialoguePhase ti) (tiDialogueThread ti) (tiSenseVector ti)
  135       finalForce' = forceForFamily finalFamily'
  136       postLegitTruthStatus = derivePostLegitTruthContractStatus preRenderTruthStatus ti sc shadowResolution legitScore finalFamily'
  137       rmpAfterLegit =
  138         rmpAfterLegit0
  139           { rmpFamily = finalFamily'
  140           , rmpForce = finalForce'
  141           , rmpSpeechAct = familyToSpeechAct finalFamily'
  142           , rmpRelation = familyToRelation finalFamily'
  143           , rmpEpistemic = capByTruthContract postLegitTruthStatus (rmpEpistemic rmpAfterLegit0)
  144           , rmpTruthContractStatus = postLegitTruthStatus
  145           , rmpCommitmentStrength = capCommitmentStrengthByTruthContract postLegitTruthStatus (rmpCommitmentStrength rmpAfterLegit0)
  146           , rmpSensePlan = finalSensePlan
  147           , rmpMicroPlan = finalMicroPlan
  148           }
  149       rcpFinal =
  150         if finalFamily' == finalFamily0
  151           then rcpFinal0
  152           else (buildRCP finalFamily' rmpAfterLegit) {rcpStyle = rcpStyle rcpFinal0}
  153       activeScene = inferActiveScene (tiNewTrace ti) (map maTag (asAtoms atomSet)) (ssActiveScene ss) defaultScenes
  154       metricsWithThresholds =
  155         recordThresholdProbe "shadow_gate" 1.0 (srGateTriggered shadowResolution)
  156           . recordThresholdProbe "legitimacy_pass" legitimacyPassThreshold
  157               (legitScore >= legitimacyPassThreshold)
  158           . recordThresholdProbe "parser_low_confidence" parserLowConfidenceThreshold
  159               (ipfConfidence (tiFrame ti) < parserLowConfidenceThreshold)
  160           . recordThresholdProbe "intuition_flash" flashThreshold
  161               (intuitPosterior >= flashThreshold)
  162           $ tiMetrics ti
  163    in TurnPlan
  164          { tpRouting = rd
  165          , tpFamily = family
  166          , tpRenderStyle = renderStyleText renderStyle
  167          , tpRmpAfterLegit = rmpAfterLegit
  168          , tpRcpFinal = rcpFinal
  169          , tpFinalFamily = finalFamily'
  170          , tpFinalForce = finalForce'
  171          , tpLegitScore = legitScore
  172          , tpActiveScene = activeScene
  173          , tpShadowStatus = scShadowStatus sc
  174          , tpShadowDivergence = scShadowHasDivergence sc
  175          , tpShadowDivergenceKind = scShadowDivergenceKind sc
  176          , tpShadowDivergenceSeverity = scShadowDivergenceSeverity sc
  177           , tpShadowGateTriggered = srGateTriggered shadowResolution
  178           , tpShadowSnapshotId = scShadowSnapshotId sc
  179           , tpShadowFamily = scShadowFamily sc
  180           , tpShadowForce = scShadowForce sc
  181           , tpShadowMessage =
  182               if vetoExhausted
  183                 then scShadowMessage sc <> "|shadow_veto_exhausted"
  184                 else scShadowMessage sc
  185           , tpShadowVetoState = ShadowVetoState newVetoCount newWindowStart
  186           , tpMetrics = metricsWithThresholds
  187            , tpDeliberation = rdDeliberation rd
  188            }

  190 derivePreRenderTruthContractStatus :: TurnInput -> ShadowContext -> ShadowResolution -> CanonicalMoveFamily -> TruthContractStatus
  191 derivePreRenderTruthContractStatus ti sc shadowResolution family
  192   | tiConatusGateFired ti = NonExpansiveRecoverySurface
  193   | srGateTriggered shadowResolution = NonExpansiveRecoverySurface
  194   | family == CMRepair = NonExpansiveRecoverySurface
  195   | scShadowStatus sc `elem` [ShadowDiverged, ShadowUnavailable] = ExplicitFallbackSurface
  196   | scShadowHasDivergence sc = ExplicitFallbackSurface
  197   | ipfConfidence (tiFrame ti) < parserLowConfidenceThreshold = ExplicitFallbackSurface
  198   | not (tiNixAvailable ti) = ExplicitFallbackSurface
  199   | otherwise = AssembledSurfacePreserved
  200 
  201 derivePostLegitTruthContractStatus :: TruthContractStatus -> TurnInput -> ShadowContext -> ShadowResolution -> Double -> CanonicalMoveFamily -> TruthContractStatus
  202 derivePostLegitTruthContractStatus preTruthStatus ti sc shadowResolution legitScore family
  203   | family == CMRepair = NonExpansiveRecoverySurface
  204   | srGateTriggered shadowResolution = NonExpansiveRecoverySurface
  205   | scShadowStatus sc `elem` [ShadowDiverged, ShadowUnavailable] = ExplicitFallbackSurface
  206   | scShadowHasDivergence sc = ExplicitFallbackSurface
  207   | ipfConfidence (tiFrame ti) < parserLowConfidenceThreshold = ExplicitFallbackSurface
  208   | legitScore < legitimacyRecoveryThreshold = NonExpansiveRecoverySurface
  209   | legitScore < legitimacyPassThreshold = ExplicitFallbackSurface
  210   | otherwise = preTruthStatus

  212 routeTurnPlan :: PipelineIO -> SystemState -> TurnInput -> TurnSignals -> IO TurnPlan
  213 routeTurnPlan pio ss ti ts = do
  214   let effectPlan = planRouteEffects ss ti ts
  215   effectResults <- resolveRouteEffects pio effectPlan
  216   pure (buildRouteTurnPlan (pipelineShadowPolicy pio) ss ti ts effectPlan effectResults)
  217 
  218 renderTurnOutput :: PipelineIO -> SystemState -> TurnInput -> TurnSignals -> TurnPlan -> IO TurnArtifacts
  219 renderTurnOutput pio ss ti ts tp = do
  220   let effectPlan = planRenderEffectsForRuntime (pipelineRuntimeMode pio) (pipelineLocalRecoveryPolicy pio) ss ti ts tp
  221   effectResults <- resolveRenderEffects pio effectPlan
  222   pure (buildTurnArtifacts ss ti ts tp effectPlan effectResults)
