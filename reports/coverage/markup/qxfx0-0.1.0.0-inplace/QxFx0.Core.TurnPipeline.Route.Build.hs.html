   53 buildRouteTurnPlan :: ShadowPolicy -> SystemState -> TurnInput -> TurnSignals -> RouteEffectPlan -> RouteEffectResults -> TurnPlan
   54 buildRouteTurnPlan shadowPolicy ss ti ts effectPlan effectResults =
   55   let atomSet = tiAtomSet ti
   56       intuitPosterior = tsIntuitPosterior ts
   57       rd = rsRoutingDecision (repStatic effectPlan)
   58       sc =
   59         computeShadowContext
   60           (rerShadowResult effectResults)
   61           (tiFrame ti)
   62           (tiNewTrace ti)
   63           intuitPosterior
   64           (tiEmbeddingQuality ti)
   65           (tiEmbSimilarity ti)
   66           (tsApiHealthy ts)
   67       shadowResolution = resolveShadowFamily shadowPolicy (rdFamily rd) sc
   68       family = srEffectiveFamily shadowResolution
   69       recoveryBonus =
   70         legitimacyRecoveryBonus
   71           (scShadowStatus sc == ShadowMatch && not (scShadowHasDivergence sc))
   72           (rdStrategyFamily rd == Just family)
   73       newEgo = rdNewEgo rd
   74       renderStrategy = rdRenderStrategy rd
   75       renderStyle = rdRenderStyle rd
   76       rmpBase = buildRMP family (tiFrame ti) (tiBestTopic ti) newEgo (tiNewTrace ti) (tiNixAvailable ti)
   77       rmp0 = applyRenderStrategy family renderStrategy rmpBase
   78       rcp0 = (buildRCP family rmp0) {rcpStyle = renderStyle}
   79       rmp1 = modulateRMPWithNarrative (tsNarrativeFragment ts) rmp0
   80       rcp1 =
   81         case tsFlash ts of
   82           Just flash -> modulateRCPWithFlash (ifOverridesAll flash) rcp0
   83           Nothing -> rcp0
   84       agdaStatus = rerAgdaStatus effectResults
   85       agdaOk = agdaVerificationReady agdaStatus
   86       legitPreAgda = min 1.0 (scAdjustedBaseLegit sc + recoveryBonus)
   87       legitInput =
   88         if agdaOk
   89           then legitPreAgda
   90           else max 0.0 (legitPreAgda - agdaVerificationPenalty)
   91       (legitScore, rmpAfterLegit0, rcpFinal0, finalFamily0, _finalForce0) =
   92         applyLegitimacyToPlans legitInput family rmp1 rcp1 renderStyle
   93       lockedDiagnosticFamily =
   94         case diagnosticPropositionFamily (ipfPropositionType (tiFrame ti)) of
   95           Just diagnosticFamily
   96             | not (srGateTriggered shadowResolution)
   97                 && finalFamily0 /= CMRepair ->
   98                 Just diagnosticFamily
   99           _ ->
  100             Nothing
  101       finalFamily = maybe finalFamily0 id lockedDiagnosticFamily
  102       finalForce = forceForFamily finalFamily
  103       rmpAfterLegit =
  104         rmpAfterLegit0
  105           { rmpFamily = finalFamily
  106           , rmpForce = finalForce
  107           }
  108       rcpFinal =
  109         if finalFamily == finalFamily0
  110           then rcpFinal0
  111           else (buildRCP finalFamily rmpAfterLegit) {rcpStyle = rcpStyle rcpFinal0}
  112       activeScene = inferActiveScene (tiNewTrace ti) (map maTag (asAtoms atomSet)) (ssActiveScene ss) defaultScenes
  113    in TurnPlan
  114         { tpFamily = family
  115         , tpNewEgo = newEgo
  116         , tpIdentitySignal = rdIdentitySignal rd
  117         , tpGuardReport = rdGuardReport rd
  118         , tpSemanticAnchor = rdSemanticAnchor rd
  119         , tpRenderStrategy = renderStrategy
  120         , tpRenderStyle = renderStyleText renderStyle
  121         , tpPrincipledMode =
  122             case (rdPressure rd, rdPrincipledMode rd) of
  123               (Just p, Just pmr) -> Just (p, pmr)
  124               _ -> Nothing
  125         , tpUpdatedOrbital = rdUpdatedOrbital rd
  126         , tpFromMs = rdFromMs rd
  127         , tpToMs = rdToMs rd
  128         , tpStrategyFamily = rdStrategyFamily rd
  129         , tpPreShadowFamily = rdFamily rd
  130         , tpRmpAfterLegit = rmpAfterLegit
  131         , tpRcpFinal = rcpFinal
  132         , tpFinalFamily = finalFamily
  133         , tpFinalForce = finalForce
  134         , tpLegitScore = legitScore
  135         , tpActiveScene = activeScene
  136         , tpShadowStatus = scShadowStatus sc
  137         , tpShadowDivergence = scShadowHasDivergence sc
  138         , tpShadowDivergenceKind = scShadowDivergenceKind sc
  139         , tpShadowDivergenceSeverity = scShadowDivergenceSeverity sc
  140         , tpShadowGateTriggered = srGateTriggered shadowResolution
  141         , tpShadowSnapshotId = scShadowSnapshotId sc
  142         , tpShadowFamily = scShadowFamily sc
  143         , tpShadowForce = scShadowForce sc
  144         , tpShadowMessage = scShadowMessage sc
  145         , tpMetrics = tiMetrics ti
  146         }

  148 routeTurnPlan :: PipelineIO -> SystemState -> TurnInput -> TurnSignals -> IO TurnPlan
  149 routeTurnPlan pio ss ti ts = do
  150   let effectPlan = planRouteEffects ss ti ts
  151   effectResults <- resolveRouteEffects pio effectPlan
  152   pure (buildRouteTurnPlan (pipelineShadowPolicy pio) ss ti ts effectPlan effectResults)
  153 
  154 renderTurnOutput :: PipelineIO -> SystemState -> TurnInput -> TurnSignals -> TurnPlan -> IO TurnArtifacts
  155 renderTurnOutput pio ss ti ts tp = do
  156   let effectPlan = planRenderEffectsForRuntime (pipelineRuntimeMode pio) (pipelineLocalRecoveryPolicy pio) ss ti ts tp
  157   effectResults <- resolveRenderEffects pio effectPlan
  158   pure (buildTurnArtifacts ss ti ts tp effectPlan effectResults)
