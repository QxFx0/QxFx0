   22 classifyWordUnits :: NormalizedInput -> [WordMeaningUnit]
   23 classifyWordUnits normalizedInput =
   24   assignSyntacticRoles (map classifyToken (niTokens normalizedInput))
   25 
   26 classifyToken :: Text -> WordMeaningUnit
   27 classifyToken token =
   28   let pos = guessPartOfSpeech token
   29       morphFeatures = guessMorphFeatures token
   30       semanticClasses = semanticClassesForToken token
   31       discourseFunctions = discourseFunctionsForToken token
   32       confidence
   33         | pos == PosUnknown = 0.35
   34         | isFunctionWord token = 0.7
   35         | otherwise = 0.82
   36       ambiguityCandidates =
   37         case pos of
   38           PosUnknown -> [token]
   39           PosAdverb -> [token, lemmaForToken token]
   40           PosAdjective -> [token, lemmaForToken token]
   41           _ -> [lemmaForToken token]
   42   in WordMeaningUnit
   43       { wmuSurfaceForm = token
   44       , wmuLemma = lemmaForToken token
   45       , wmuPartOfSpeech = pos
   46       , wmuMorphFeatures = morphFeatures
   47       , wmuSyntacticRole = SynUnknown
   48       , wmuSemanticClasses = semanticClasses
   49       , wmuDiscourseFunctions = discourseFunctions
   50       , wmuAmbiguityCandidates = ambiguityCandidates
   51       , wmuConfidence = confidence
   52       }
   53 
   54 assignSyntacticRoles :: [WordMeaningUnit] -> [WordMeaningUnit]
   55 assignSyntacticRoles units =
   56   zipWith assign [0 ..] units
   57   where
   58     rootIndex = findRootIndex units
   59     assign idx unit =
   60       unit {wmuSyntacticRole = inferRole rootIndex idx unit}
   61 
   62 inferRole :: Int -> Int -> WordMeaningUnit -> InputSyntacticRole
   63 inferRole rootIndex idx unit
   64   | idx == rootIndex = SynRoot
   65   | wmuPartOfSpeech unit == PosVerb = SynPredicate
   66   | idx < rootIndex && wmuPartOfSpeech unit `elem` [PosNoun, PosPronoun, PosNumeral] = SynSubject
   67   | idx > rootIndex && wmuPartOfSpeech unit `elem` [PosNoun, PosPronoun, PosNumeral] = SynObject
   68   | wmuPartOfSpeech unit == PosAdjective = SynAttribute
   69   | wmuPartOfSpeech unit `elem` [PosAdverb, PosPreposition] = SynCircumstance
   70   | wmuPartOfSpeech unit `elem` [PosConjunction, PosParticle, PosInterjection] = SynMarker
   71   | otherwise = SynUnknown
   72 
   73 findRootIndex :: [WordMeaningUnit] -> Int
   74 findRootIndex units =
   75   case findIndex ((== PosVerb) . wmuPartOfSpeech) units of
   76     Just idx -> idx
   77     Nothing ->
   78       case findIndex (\u -> wmuPartOfSpeech u `elem` [PosNoun, PosPronoun]) units of
   79         Just idx -> idx
   80         Nothing -> 0
