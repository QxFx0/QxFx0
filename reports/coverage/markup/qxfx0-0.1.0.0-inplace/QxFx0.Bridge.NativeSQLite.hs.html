   89 sqliteTransient :: Ptr ()
   90 sqliteTransient = nullPtr `plusPtr` (-1)
   91 
   92 sqlOk, sqlRow, sqlDone :: CInt
   93 sqlOk   = 0
   94 sqlRow  = 100
   95 sqlDone = 101
   96 
   97 sqlNullType :: CInt
   98 sqlNullType = 5

  100 open :: FilePath -> IO (Either Text Database)
  101 open path = alloca $ \ppDb -> do
  102   rc <- withCString path $ \cPath -> c_sqlite3_open cPath ppDb
  103   if rc == sqlOk
  104     then do db <- peek ppDb
  105             return (Right db)
  106     else return (Left $ "sqlite3_open failed: " <> T.pack (show rc))
  107 
  108 close :: Database -> IO ()
  109 close db = do
  110   rc <- c_sqlite3_close db
  111   if rc /= sqlOk
  112     then throwQxFx0 (SQLiteError $ T.pack $ "sqlite3_close failed: " ++ show rc)
  113     else return ()

  115 prepare :: Database -> Text -> IO (Either Text Statement)
  116 prepare db sql = alloca $ \ppStmt -> alloca $ \ppTail -> do
  117   let sqlBs = TE.encodeUtf8 sql
  118   rc <- BS.useAsCString sqlBs $ \cSql ->
  119     c_sqlite3_prepare_v2 db cSql (-1) ppStmt ppTail
  120   if rc == sqlOk
  121     then do stmt <- peek ppStmt
  122             if stmt == nullPtr
  123               then return (Left "sqlite3_prepare_v2 returned null statement")
  124               else return (Right stmt)
  125     else do errMsg <- getErrMsg db
  126             return (Left $ "prepare failed: " <> errMsg)
  127 
  128 getErrMsg :: Database -> IO Text
  129 getErrMsg db = do
  130   cStr <- c_sqlite3_errmsg db
  131   if cStr == nullPtr
  132     then return "<no error message>"
  133     else T.pack <$> peekCString cStr
  134 
  135 step :: Statement -> IO (Either Text CInt)
  136 step stmt = do
  137   rc <- c_sqlite3_step stmt
  138   if rc == sqlRow
  139     then return (Right sqlRow)
  140     else if rc == sqlDone
  141          then return (Right sqlDone)
  142          else return (Left $ "step failed: " <> T.pack (show rc))
  143 
  144 stepRow :: Statement -> IO Bool
  145 stepRow stmt = do
  146   rc <- c_sqlite3_step stmt
  147   return (rc == sqlRow)

  149 bindText :: Statement -> CInt -> Text -> IO (Either Text ())
  150 bindText stmt idx val = do
  151   let bs = TE.encodeUtf8 val
  152   rc <- BS.useAsCString bs $ \cStr ->
  153     c_sqlite3_bind_text stmt idx cStr (fromIntegral (BS.length bs)) sqliteTransient
  154   if rc == sqlOk
  155     then return (Right ())
  156     else return (Left $ "bindText failed: " <> T.pack (show rc))
  157 
  158 bindInt :: Statement -> CInt -> Int -> IO (Either Text ())
  159 bindInt stmt idx val = do
  160   rc <- c_sqlite3_bind_int stmt idx (fromIntegral val)
  161   if rc == sqlOk
  162     then return (Right ())
  163     else return (Left $ "bindInt failed: " <> T.pack (show rc))
  164 
  165 bindDouble :: Statement -> CInt -> Double -> IO (Either Text ())
  166 bindDouble stmt idx val = do
  167   rc <- c_sqlite3_bind_double stmt idx (realToFrac val)
  168   if rc == sqlOk
  169     then return (Right ())
  170     else return (Left $ "bindDouble failed: " <> T.pack (show rc))

  172 columnText :: Statement -> CInt -> IO Text
  173 columnText stmt idx = do
  174   cStr <- c_sqlite3_column_text stmt idx
  175   if cStr == nullPtr
  176     then return ""
  177     else do
  178       cLen <- c_sqlite3_column_bytes stmt idx
  179       bs <- BS.packCStringLen (cStr, fromIntegral cLen)
  180       return (TE.decodeUtf8With lenientDecode bs)
  181 
  182 columnInt :: Statement -> CInt -> IO Int
  183 columnInt stmt idx = fromIntegral <$> c_sqlite3_column_int stmt idx
  184 
  185 columnDouble :: Statement -> CInt -> IO Double
  186 columnDouble stmt idx = realToFrac <$> c_sqlite3_column_double stmt idx
  187 
  188 columnIsNull :: Statement -> CInt -> IO Bool
  189 columnIsNull stmt idx = do
  190   ty <- c_sqlite3_column_type stmt idx
  191   return (ty == sqlNullType)

  200 execSql :: Database -> Text -> IO (Either Text ())
  201 execSql db sql = do
  202   let sqlBs = TE.encodeUtf8 sql
  203   rc <- BS.useAsCString sqlBs $ \cSql ->
  204     c_sqlite3_exec db cSql nullPtr nullPtr nullPtr
  205   if rc == sqlOk
  206     then return (Right ())
  207     else do errMsg <- getErrMsg db
  208             return (Left $ "exec failed: " <> errMsg)

  210 withStatement :: Database -> Text -> (Statement -> IO a) -> IO (Either Text a)
  211 withStatement db sql action = do
  212   mStmt <- prepare db sql
  213   case mStmt of
  214     Left err -> return (Left err)
  215     Right stmt -> do
  216       result <- action stmt `finally` finalize stmt
  217       return (Right result)

  219 withDatabase :: FilePath -> (Database -> IO a) -> IO (Either Text a)
  220 withDatabase path action = do
  221   mDb <- open path
  222   case mDb of
  223     Left err -> return (Left err)
  224     Right db -> do
  225       result <- action db `finally` close db
  226       return (Right result)
