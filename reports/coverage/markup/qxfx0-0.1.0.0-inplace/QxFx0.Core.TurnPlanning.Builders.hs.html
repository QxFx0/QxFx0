   26 buildRMP :: CanonicalMoveFamily -> DialogueCommitmentLedger -> DialoguePhase -> DialogueThread -> InputPropositionFrame -> SenseVector -> Text -> EgoState -> AtomTrace -> Bool -> ResponseMeaningPlan
   27 buildRMP family ledger phase thread frame senseVector topic ego trace nixAvailable =

   30 buildRMPWithTruthContract :: TruthContractStatus -> CanonicalMoveFamily -> DialogueCommitmentLedger -> DialoguePhase -> DialogueThread -> InputPropositionFrame -> SenseVector -> Text -> EgoState -> AtomTrace -> Bool -> ResponseMeaningPlan
   31 buildRMPWithTruthContract truthStatus family ledger phase thread frame senseVector topic ego trace nixAvailable =
   32   let (family', sensePlan, microPlan) = familySenseBundle family ledger phase thread senseVector
   33       baseStance = familyToStance family'
   34       baseEpistemic = familyToEpistemic family'
   35       (feralStance, feralEpistemic) = feralDegradation nixAvailable baseStance baseEpistemic
   36       plannedTruthStatus = provisionalTruthContractStatus truthStatus family' phase thread ledger nixAvailable
   37       (finalStance, finalEpistemic) = threeStageModulationWithTruthContract plannedTruthStatus ego trace feralStance feralEpistemic
   38       plannedTopic =
   39         let topic0 = topicFromFrame thread ledger phase frame topic
   40             anchorTxt = unSemanticNodeId (svAnchor senseVector)
   41         in if T.null (T.strip topic0) then anchorTxt else topic0
   42       plannedScope = dialogueStructuralScope phase thread ledger plannedTopic
   43       primaryClaim = primaryClaimFromFrame plannedTruthStatus frame plannedTopic
   44       baseAst = claimAstFromFrame plannedTruthStatus frame plannedTopic ego
   45       primaryClaimAst = fmap (applyStanceToAst finalStance) baseAst
   46       contrastAxis =
   47         let axis0 = contrastAxisFromFrame frame
   48         in if T.null axis0 then senseAxisSummary (rspPreservedAxes sensePlan) else axis0
   49     in ResponseMeaningPlan
   50         { rmpFamily = family'
   51         , rmpForce = forceForFamily family'
   52         , rmpSpeechAct = familyToSpeechAct family'
   53         , rmpRelation = familyToRelation family'
   54         , rmpStrategy = familyToStrategy family'
   55         , rmpStance = finalStance
   56         , rmpEpistemic = finalEpistemic
   57         , rmpTopic = plannedTopic
   58         , rmpPrimaryClaim = primaryClaim
   59         , rmpPrimaryClaimAst = primaryClaimAst
   60         , rmpScope = plannedScope
   61         , rmpContrastAxis = contrastAxis
   62         , rmpImplicationDirection = implicationDirectionFromScope plannedScope plannedTruthStatus family'
   63         , rmpProvenance = BuiltClaim
   64         , rmpTruthContractStatus = plannedTruthStatus
   65         , rmpCommitmentStrength = capCommitmentStrengthByTruthContract plannedTruthStatus (epistemicConfidence finalEpistemic)
   66         , rmpDepthMode = familyDefaultDepthMode family'
   67         , rmpSensePlan = sensePlan
   68         , rmpMicroPlan = microPlan
   69         }

   71 provisionalTruthContractStatus :: TruthContractStatus -> CanonicalMoveFamily -> DialoguePhase -> DialogueThread -> DialogueCommitmentLedger -> Bool -> TruthContractStatus
   72 provisionalTruthContractStatus incoming family phase thread ledger nixAvailable
   73   | family == CMRepair = NonExpansiveRecoverySurface
   74   | not nixAvailable = ExplicitFallbackSurface
   75   | scopeConstrained = AssembledSurfacePreserved
   76   | otherwise =
   77       case incoming of
   78         CanonicalSurfacePreserved -> AssembledSurfacePreserved
   79         LegacyIncompleteSurface -> AssembledSurfacePreserved
   80         other -> other
   81   where
   82     scopeConstrained =
   83       phase `elem` [Clarifying, Repairing, Grounding, Contesting]
   84         || any ((`elem` [CsUnresolved, CsContested, CsSuspended]) . dcStatus) (dclItems ledger)
   85         || not (T.null (T.strip (dtPhaseScope thread)))

   87 topicFromFrame :: DialogueThread -> DialogueCommitmentLedger -> DialoguePhase -> InputPropositionFrame -> Text -> Text
   88 topicFromFrame thread ledger phase frame fallback =
   89   clampTopicToDialogueScope phase thread ledger rawTopic
   90   where
   91     rawTopic =
   92       case propositionTypeFromText (ipfPropositionType frame) of
   93         Just SelfKnowledgeQ
   94           | ipfSemanticTarget frame == "user" -> "твой контекст"
   95           | ipfSemanticTarget frame == "user_help" -> nonEmptyOr (ipfSemanticSubject frame) "помощь"
   96           | ipfSemanticTarget frame == "self_capability" -> nonEmptyOr (ipfSemanticSubject frame) "способность"
   97           | otherwise -> "моя роль"
   98         Just DialogueInvitationQ ->
   99           nonEmptyOr (ipfSemanticSubject frame) fallback
  100         Just ConceptKnowledgeQ ->
  101           nonEmptyOr (ipfSemanticSubject frame) fallback
  102         Just WorldCauseQ ->
  103           nonEmptyOr (ipfSemanticSubject frame) fallback
  104         Just PurposeQ ->
  105           nonEmptyOr (ipfSemanticSubject frame) fallback
  106         Just LocationFormationQ ->
  107           nonEmptyOr (ipfSemanticSubject frame) fallback
  108         Just SelfStateQ ->
  109           "мой текущий ход"
  110         Just ComparisonPlausibilityQ ->
  111           case ipfSemanticCandidates frame of
  112             [] -> nonEmptyOr (ipfFocusEntity frame) fallback
  113             xs -> T.intercalate " / " xs
  114         Just MisunderstandingReport ->
  115           "взаимопонимание"
  116         Just GenerativePrompt ->
  117           "мысль"
  118         Just ContemplativeTopic ->
  119           nonEmptyOr (ipfSemanticSubject frame) fallback
  120         Just OperationalStatusQ ->
  121           "работа"
  122         Just OperationalCauseQ ->
  123           "разбор смысла"
  124         Just SystemLogicQ ->
  125           "логика"
  126         _ ->
  127           nonEmptyOr (ipfFocusEntity frame) fallback

  129 clampTopicToDialogueScope :: DialoguePhase -> DialogueThread -> DialogueCommitmentLedger -> Text -> Text
  130 clampTopicToDialogueScope phase thread ledger proposedTopic =
  131   let threadFocus = T.strip (dtCurrentFocus thread)
  132       scopeFocus = T.strip (dtPhaseScope thread)
  133       canonicalFocus = structuralScopeText thread (nonEmptyOr scopeFocus threadFocus)
  134       normalizedProposed = T.strip proposedTopic
  135       inLockedPhase = phase `elem` [Clarifying, Repairing, Grounding, Contesting]
  136       hasOpenLedger = any ((`elem` [CsUnresolved, CsContested, CsSuspended]) . dcStatus) (dclItems ledger)
  137       topicEscapesScope =
  138         not (T.null canonicalFocus)
  139           && not (T.null normalizedProposed)
  140           && canonicalFocus /= normalizedProposed
  141           && normalizeScopeKey canonicalFocus /= normalizeScopeKey normalizedProposed
  142   in if T.null canonicalFocus
  143        then normalizedProposed
  144        else if T.null normalizedProposed
  145          then canonicalFocus
  146          else if (inLockedPhase || hasOpenLedger) && topicEscapesScope
  147             then canonicalFocus
  148             else normalizedProposed

  150 dialogueStructuralScope :: DialoguePhase -> DialogueThread -> DialogueCommitmentLedger -> Text -> Maybe PerspectiveScope
  151 dialogueStructuralScope phase thread ledger topicText
  152   | T.null normalizedTopic = dtStructuralScope thread
  153   | lockedPhase || hasOpenLedger = Just (ScopeTopic scopedTopic)
  154   | otherwise = Just (ScopeTopic normalizedTopic)
  155   where
  156     normalizedTopic = T.strip topicText
  157     scopedTopic = structuralScopeText thread normalizedTopic
  158     lockedPhase = phase `elem` [Clarifying, Repairing, Grounding, Contesting]
  159     hasOpenLedger = any ((`elem` [CsUnresolved, CsContested, CsSuspended]) . dcStatus) (dclItems ledger)

  161 structuralScopeText :: DialogueThread -> Text -> Text
  162 structuralScopeText thread fallback =
  163   case dtStructuralScope thread of
  164     Just (ScopeTopic topic) -> nonEmptyOr topic fallback
  165     Just (ScopeTheme theme) -> nonEmptyOr theme fallback
  166     Just (ScopeCluster cluster) -> nonEmptyOr cluster fallback
  167     Nothing -> fallback

  169 normalizeScopeKey :: Text -> Text
  170 normalizeScopeKey = T.unwords . T.words . T.toLower . T.strip

  172 implicationDirectionFromScope :: Maybe PerspectiveScope -> TruthContractStatus -> CanonicalMoveFamily -> Text
  173 implicationDirectionFromScope mScope truthStatus family
  174   | truthStatus `notElem` [CanonicalSurfacePreserved, AssembledSurfacePreserved] = "bounded"
  175   | family `elem` [CMNextStep, CMDeepen, CMPurpose] = maybe "bounded" (const "forward") mScope
  176   | otherwise = "bounded"

  223 claimAstFromFrame :: TruthContractStatus -> InputPropositionFrame -> Text -> EgoState -> Maybe ClaimAst
  224 claimAstFromFrame truthStatus frame fallback ego =
  225   let subject0 = nonEmptyOr (ipfSemanticSubject frame) fallback
  226       scopedSubject = if truthStatus `elem` [CanonicalSurfacePreserved, AssembledSurfacePreserved] then subject0 else fallback
  227       topicNP = mkTopicNP scopedSubject
  228       familyFallback = markDefaultLexemeFallback scopedSubject (fallbackAstForFamily (ipfCanonicalFamily frame) topicNP)
  229       withTopicFallback = markDefaultLexemeFallback scopedSubject
  230   in case propositionTypeFromText (ipfPropositionType frame) of
  231       Just DialogueInvitationQ ->
  232         let gfMod = if egoTension ego > 0.5 then ModStrictly else ModFirst
  233             gfNum = extractNumber frame
  234             gfAction = if egoAgency ego > 0.6
  235                        then ActDefine "granitsa_N"
  236                        else ActMaintain gfNum "ramka_N"
  237         in Just (withTopicFallback (MoveInvite topicNP gfMod gfAction))
  238       Just ConceptKnowledgeQ ->
  239         Just (withTopicFallback (MoveDefine topicNP RelIdentity (MkNP "ponyatie_N")))
  240       Just WorldCauseQ ->
  241         Just (withTopicFallback (MoveCause topicNP MechParse))
  242       Just PurposeQ ->
  243         Just (withTopicFallback (MovePurpose topicNP))
  244       Just SelfStateQ ->
  245         Just MoveSelfState
  246       Just ComparisonPlausibilityQ ->
  247         Just (withTopicFallback (buildComparisonAst frame topicNP))
  248       Just OperationalStatusQ ->
  249         Just MoveOperationalStatus
  250       Just OperationalCauseQ ->
  251         Just MoveOperationalCause
  252       Just SystemLogicQ ->
  253         Just MoveSystemLogic
  254       Just MisunderstandingReport ->
  255         Just MoveMisunderstanding
  256       Just GenerativePrompt ->
  257         Just MoveGenerativeThought
  258       Just ContemplativeTopic ->
  259         Just (withTopicFallback (MoveContemplative topicNP))
  260       Just ContactSignal ->
  261         Just (withTopicFallback (MoveContact topicNP))
  262       Just ExploratoryPrompt ->
  263         Just familyFallback
  264       Just AffectiveQ ->
  265         Just (withTopicFallback (MoveContact topicNP))
  266       Just ReflectiveQ ->
  267         Just (withTopicFallback (MoveReflect topicNP))
  268       Just DefinitionalQ ->
  269         Just (withTopicFallback (MoveDefine topicNP RelIdentity (MkNP "ponyatie_N")))
  270       Just DistinctionQ ->
  271         Just (withTopicFallback (buildComparisonAst frame topicNP))
  272       Just GroundQ ->
  273         Just (withTopicFallback (MoveGround topicNP))
  274       Just SelfDescQ ->
  275         Just (withTopicFallback (MoveDescribe topicNP))
  276       Just HypotheticalQ ->
  277         Just (withTopicFallback (MoveHypothesis topicNP))
  278       Just RepairSignal ->
  279         Just MoveMisunderstanding
  280       Just AnchorSignal ->
  281         Just (withTopicFallback (MoveAnchor topicNP))
  282       Just ClarifyQ ->
  283         Just (withTopicFallback (MoveClarify topicNP))
  284       Just DeepenQ ->
  285         Just (withTopicFallback (MoveDeepen topicNP))
  286       Just ConfrontQ ->
  287         Just (withTopicFallback (MoveConfront topicNP))
  288       Just NextStepQ ->
  289         Just (withTopicFallback (MoveNextStepLocal topicNP))
  290       Just PlainAssert ->
  291         Just (withTopicFallback (MoveGround topicNP))
  292       Just EpistemicQ ->
  293         Just (withTopicFallback (MoveClarify topicNP))
  294       Just RequestQ ->
  295         Just (withTopicFallback (MoveClarify topicNP))
  296       Just EvaluationQ ->
  297         Just (withTopicFallback (buildComparisonAst frame topicNP))
  298       Just NarrativeQ ->
  299         Just (withTopicFallback (MoveDescribe topicNP))
  300       Just SelfKnowledgeQ ->
  301         Just (withTopicFallback (MoveDescribe topicNP))
  302       Just LocationFormationQ ->
  303         Just (withTopicFallback (MoveGround topicNP))
  304       Nothing ->
  305         Just familyFallback

  307 mkTopicNP :: Text -> GfNP
  308 mkTopicNP topic =
  309   MkNP (fst (topicToGfLexemeDecision topic))

  311 markDefaultLexemeFallback :: Text -> ClaimAst -> ClaimAst
  312 markDefaultLexemeFallback topic ast =
  313   case topicToGfLexemeDecision topic of
  314     (_, Just "gf_default_lexeme") -> StanceWrapped "GfDefaultLexeme" ast
  315     _ -> ast

  317 buildComparisonAst :: InputPropositionFrame -> GfNP -> ClaimAst
  318 buildComparisonAst frame fallbackTopic =
  319   case ipfSemanticCandidates frame of
  320     left : right : _ ->
  321       let ast = MoveDistinguish (mkTopicNP left) (mkTopicNP right)
  322       in case topicToGfLexemeDecision left of
  323            (_, Just "gf_default_lexeme") -> StanceWrapped "GfDefaultLexeme" ast
  324            _ -> case topicToGfLexemeDecision right of
  325                   (_, Just "gf_default_lexeme") -> StanceWrapped "GfDefaultLexeme" ast
  326                   _ -> ast
  327     _ ->
  328       MoveCompare fallbackTopic (MkNP "ponyatie_N")

  330 fallbackAstForFamily :: CanonicalMoveFamily -> GfNP -> ClaimAst

  332   case family of
  333     CMGround -> MoveGround topicNP

  340     CMContact -> MoveContact topicNP

  343     CMDeepen -> MoveDeepen topicNP

  353 extractNumber :: InputPropositionFrame -> GfNumber
  354 extractNumber frame =
  355   let txt = T.toLower (ipfRawText frame)
  356   in if "мы" `T.isInfixOf` txt || "вы" `T.isInfixOf` txt || "нас" `T.isInfixOf` txt || "вас" `T.isInfixOf` txt
  357      then NumPl
  358      else NumSg

  378 nonEmptyOr :: Text -> Text -> Text

  380   | T.null (T.strip preferred) = fallback

  383 buildRCP :: CanonicalMoveFamily -> ResponseMeaningPlan -> ResponseContentPlan
  384 buildRCP family meaningPlan =
  385   ResponseContentPlan
  386     { rcpFamily = family
  387     , rcpOpening = familyToOpeningMove family
  388     , rcpCore = familyToCoreMove family
  389     , rcpLimit =
  390         if family == CMRepair
  391           then MoveAcknowledgeRupture
  392           else familyToCoreMove family
  393     , rcpContinuation = continuationMoveForScope meaningPlan
  394     , rcpStyle = styleForStance (rmpStance meaningPlan)
  395     }
  396 
  397 continuationMoveForScope :: ResponseMeaningPlan -> ContentMove
  398 continuationMoveForScope meaningPlan
  399   | rmpImplicationDirection meaningPlan /= "forward" = MoveStateBoundary
  400   | otherwise = senseContinuationMove (rspChosenOperator (rmpSensePlan meaningPlan))

  402 senseContinuationMove :: SenseOperator -> ContentMove
  403 senseContinuationMove op = case op of
  404   OpDefine -> MoveDefineFrame
  405   OpGround -> MoveGroundBasis
  406   OpDistinguish -> MoveShowContrast
  407   OpExplainCause -> MoveStateBoundary
  408   OpExplainPurpose -> MovePurposeTeleology
  409   OpReflect -> MoveReflectMirror
  410   OpConstrain -> MoveClarifyDisambiguate
  411   OpRepair -> MoveRepairBridge
  412   OpClarify -> MoveClarifyDisambiguate
  413   OpDeepen -> MoveDeepenProbe
  414   OpNextStep -> MoveNextStep

  416 styleForStance :: StanceMarker -> RenderStyle
  417 styleForStance stance
  418   | stance `elem` [Firm, Commit] = StyleFormal
  419   | stance `elem` [Honest, Explore] = StyleStandard
  420   | otherwise = StyleWarm

  422 familyDefaultDepthMode :: CanonicalMoveFamily -> DepthMode
  423 familyDefaultDepthMode family
  424   | family `elem` [CMDeepen, CMHypothesis, CMPurpose] = DeepDepth
  425   | otherwise = SurfaceDepth

  427 senseAxisSummary :: [SenseAxis] -> Text
  428 senseAxisSummary axes =
  429   case axes of
  430     [] -> ""
  431     xs -> T.intercalate "+" (map renderSenseAxis xs)
