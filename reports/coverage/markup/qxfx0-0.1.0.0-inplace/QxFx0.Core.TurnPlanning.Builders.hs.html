   21 buildRMP :: CanonicalMoveFamily -> InputPropositionFrame -> Text -> EgoState -> AtomTrace -> Bool -> ResponseMeaningPlan
   22 buildRMP family frame topic ego trace nixAvailable =
   23   let baseStance = familyToStance family
   24       baseEpistemic = familyToEpistemic family
   25       (feralStance, feralEpistemic) = feralDegradation nixAvailable baseStance baseEpistemic
   26       (finalStance, finalEpistemic) = threeStageModulation ego trace feralStance feralEpistemic
   27       plannedTopic = topicFromFrame frame topic
   28       primaryClaim = primaryClaimFromFrame frame plannedTopic
   29       baseAst = claimAstFromFrame frame plannedTopic ego
   30       primaryClaimAst = fmap (applyStanceToAst feralStance) baseAst
   31       contrastAxis = contrastAxisFromFrame frame
   32    in ResponseMeaningPlan
   33         { rmpFamily = family
   34         , rmpForce = forceForFamily family
   35         , rmpSpeechAct = familyToSpeechAct family
   36         , rmpRelation = familyToRelation family
   37         , rmpStrategy = familyToStrategy family
   38         , rmpStance = finalStance
   39         , rmpEpistemic = finalEpistemic
   40         , rmpTopic = plannedTopic
   41         , rmpPrimaryClaim = primaryClaim
   42         , rmpPrimaryClaimAst = primaryClaimAst
   43         , rmpContrastAxis = contrastAxis
   44         , rmpImplicationDirection = "forward"
   45         , rmpProvenance = BuiltClaim
   46         , rmpCommitmentStrength = epistemicConfidence finalEpistemic
   47         , rmpDepthMode = familyDefaultDepthMode family
   48         }

   50 topicFromFrame :: InputPropositionFrame -> Text -> Text
   51 topicFromFrame frame fallback =
   52   case propositionTypeFromText (ipfPropositionType frame) of
   53     Just SelfKnowledgeQ
   54       | ipfSemanticTarget frame == "user" -> "твой контекст"
   55       | ipfSemanticTarget frame == "user_help" -> nonEmptyOr (ipfSemanticSubject frame) "помощь"
   56       | ipfSemanticTarget frame == "self_capability" -> nonEmptyOr (ipfSemanticSubject frame) "способность"
   57       | otherwise -> "моя роль"
   58     Just DialogueInvitationQ ->
   59       nonEmptyOr (ipfSemanticSubject frame) fallback
   60     Just ConceptKnowledgeQ ->
   61       nonEmptyOr (ipfSemanticSubject frame) fallback
   62     Just WorldCauseQ ->
   63       nonEmptyOr (ipfSemanticSubject frame) fallback
   64     Just PurposeQ ->
   65       nonEmptyOr (ipfSemanticSubject frame) fallback
   66     Just LocationFormationQ ->
   67       nonEmptyOr (ipfSemanticSubject frame) fallback
   68     Just SelfStateQ ->
   69       "мой текущий ход"
   70     Just ComparisonPlausibilityQ ->
   71       case ipfSemanticCandidates frame of
   72         [] -> nonEmptyOr (ipfFocusEntity frame) fallback
   73         xs -> T.intercalate " / " xs
   74     Just MisunderstandingReport ->
   75       "взаимопонимание"
   76     Just GenerativePrompt ->
   77       "мысль"
   78     Just ContemplativeTopic ->
   79       nonEmptyOr (ipfSemanticSubject frame) fallback
   80     Just OperationalStatusQ ->
   81       "работа"
   82     Just OperationalCauseQ ->
   83       "разбор смысла"
   84     Just SystemLogicQ ->
   85       "логика"
   86     _ ->
   87       nonEmptyOr (ipfFocusEntity frame) fallback

  130 claimAstFromFrame :: InputPropositionFrame -> Text -> EgoState -> Maybe ClaimAst
  131 claimAstFromFrame frame fallback ego =
  132   let topicNP = mkTopicNP (nonEmptyOr (ipfSemanticSubject frame) fallback)
  133       familyFallback = fallbackAstForFamily (ipfCanonicalFamily frame) topicNP
  134   in case propositionTypeFromText (ipfPropositionType frame) of
  135       Just DialogueInvitationQ ->
  136         let gfMod = if egoTension ego > 0.5 then ModStrictly else ModFirst
  137             gfNum = extractNumber frame
  138             gfAction = if egoAgency ego > 0.6
  139                        then ActDefine "granitsa_N"
  140                        else ActMaintain gfNum "ramka_N"
  141         in Just (MoveInvite topicNP gfMod gfAction)
  142       Just ConceptKnowledgeQ ->
  143         Just (MoveDefine topicNP RelIdentity (MkNP "ponyatie_N"))
  144       Just WorldCauseQ ->
  145         Just (MoveCause topicNP MechParse)
  146       Just PurposeQ ->
  147         Just (MovePurpose topicNP)
  148       Just SelfStateQ ->
  149         Just MoveSelfState
  150       Just ComparisonPlausibilityQ ->
  151         Just (buildComparisonAst frame topicNP)
  152       Just OperationalStatusQ ->
  153         Just MoveOperationalStatus
  154       Just OperationalCauseQ ->
  155         Just MoveOperationalCause
  156       Just SystemLogicQ ->
  157         Just MoveSystemLogic
  158       Just MisunderstandingReport ->
  159         Just MoveMisunderstanding
  160       Just GenerativePrompt ->
  161         Just MoveGenerativeThought
  162       Just ContemplativeTopic ->
  163         Just (MoveContemplative topicNP)
  164       Just ContactSignal ->
  165         Just (MoveContact topicNP)
  166       Just AffectiveQ ->
  167         Just (MoveContact topicNP)
  168       Just ReflectiveQ ->
  169         Just (MoveReflect topicNP)
  170       Just DefinitionalQ ->
  171         Just (MoveDefine topicNP RelIdentity (MkNP "ponyatie_N"))
  172       Just DistinctionQ ->
  173         Just (buildComparisonAst frame topicNP)
  174       Just GroundQ ->
  175         Just (MoveGround topicNP)
  176       Just SelfDescQ ->
  177         Just (MoveDescribe topicNP)
  178       Just HypotheticalQ ->
  179         Just (MoveHypothesis topicNP)
  180       Just RepairSignal ->
  181         Just MoveMisunderstanding
  182       Just AnchorSignal ->
  183         Just (MoveAnchor topicNP)
  184       Just ClarifyQ ->
  185         Just (MoveClarify topicNP)
  186       Just DeepenQ ->
  187         Just (MoveDeepen topicNP)
  188       Just ConfrontQ ->
  189         Just (MoveConfront topicNP)
  190       Just NextStepQ ->
  191         Just (MoveNextStepLocal topicNP)
  192       Just PlainAssert ->
  193         Just (MoveGround topicNP)
  194       Just EpistemicQ ->
  195         Just (MoveClarify topicNP)
  196       Just RequestQ ->
  197         Just (MoveClarify topicNP)
  198       Just EvaluationQ ->
  199         Just (buildComparisonAst frame topicNP)
  200       Just NarrativeQ ->
  201         Just (MoveDescribe topicNP)
  202       Just SelfKnowledgeQ ->
  203         Just (MoveDescribe topicNP)
  204       Just LocationFormationQ ->
  205         Just (MoveGround topicNP)
  206       Nothing ->
  207         Just familyFallback

  209 mkTopicNP :: Text -> GfNP
  210 mkTopicNP = MkNP . topicToGfLexemeId

  212 buildComparisonAst :: InputPropositionFrame -> GfNP -> ClaimAst
  213 buildComparisonAst frame fallbackTopic =
  214   case ipfSemanticCandidates frame of
  215     left : right : _ ->
  216       MoveDistinguish (mkTopicNP left) (mkTopicNP right)
  217     _ ->
  218       MoveCompare fallbackTopic (MkNP "smysl_N")

  220 fallbackAstForFamily :: CanonicalMoveFamily -> GfNP -> ClaimAst
  221 fallbackAstForFamily family topicNP =
  222   case family of
  223     CMGround -> MoveGround topicNP
  224     CMDefine -> MoveDefine topicNP RelIdentity (MkNP "ponyatie_N")
  225     CMReflect -> MoveReflect topicNP
  226     CMDescribe -> MoveDescribe topicNP
  227     CMPurpose -> MovePurpose topicNP
  228     CMHypothesis -> MoveHypothesis topicNP
  229     CMRepair -> MoveMisunderstanding
  230     CMContact -> MoveContact topicNP
  231     CMAnchor -> MoveAnchor topicNP
  232     CMClarify -> MoveClarify topicNP
  233     CMDeepen -> MoveDeepen topicNP
  234     CMConfront -> MoveConfront topicNP
  235     CMNextStep -> MoveNextStepLocal topicNP
  236     CMDistinguish -> MoveDistinguish topicNP (MkNP "smysl_N")

  238 applyStanceToAst :: StanceMarker -> ClaimAst -> ClaimAst
  239 applyStanceToAst Tentative ast = StanceWrapped "ApplyStanceTentative" ast
  240 applyStanceToAst Firm ast      = StanceWrapped "ApplyStanceFirm" ast
  241 applyStanceToAst _ ast               = ast

  243 extractNumber :: InputPropositionFrame -> GfNumber
  244 extractNumber frame =
  245   let txt = T.toLower (ipfRawText frame)
  246   in if "мы" `T.isInfixOf` txt || "вы" `T.isInfixOf` txt || "нас" `T.isInfixOf` txt || "вас" `T.isInfixOf` txt
  247      then NumPl
  248      else NumSg

  250 contrastAxisFromFrame :: InputPropositionFrame -> Text
  251 contrastAxisFromFrame frame =
  252   case propositionTypeFromText (ipfPropositionType frame) of
  253     Just ComparisonPlausibilityQ -> "логичность"
  254     Just SelfKnowledgeQ
  255       | ipfSemanticTarget frame == "user" -> "границы знания"
  256       | ipfSemanticTarget frame == "user_help" -> "рамка помощи"
  257       | ipfSemanticTarget frame == "self_capability" -> "границы способности"
  258       | otherwise -> "самоописание"
  259     Just DialogueInvitationQ -> "рамка разговора"
  260     Just ConceptKnowledgeQ -> "границы знания"
  261     Just PurposeQ -> "назначение"
  262     Just SelfStateQ -> "внутренний ход"
  263     Just MisunderstandingReport -> "точка разрыва"
  264     Just GenerativePrompt -> "направление мысли"
  265     Just ContemplativeTopic -> "смысловой резонанс"
  266     _ -> ""

  268 nonEmptyOr :: Text -> Text -> Text
  269 nonEmptyOr preferred fallback
  270   | T.null (T.strip preferred) = fallback
  271   | otherwise = preferred

  273 buildRCP :: CanonicalMoveFamily -> ResponseMeaningPlan -> ResponseContentPlan
  274 buildRCP family meaningPlan =
  275   ResponseContentPlan
  276     { rcpFamily = family
  277     , rcpOpening = familyToOpeningMove family
  278     , rcpCore = familyToCoreMove family
  279     , rcpLimit =
  280         if family == CMRepair
  281           then MoveAcknowledgeRupture
  282           else familyToCoreMove family
  283     , rcpContinuation = MoveNextStep
  284     , rcpStyle = styleForStance (rmpStance meaningPlan)
  285     }
  286 
  287 styleForStance :: StanceMarker -> RenderStyle
  288 styleForStance stance
  289   | stance `elem` [Firm, Commit] = StyleFormal
  290   | stance `elem` [Honest, Explore] = StyleStandard
  291   | otherwise = StyleWarm

  293 familyDefaultDepthMode :: CanonicalMoveFamily -> DepthMode
  294 familyDefaultDepthMode family
  295   | family `elem` [CMDeepen, CMHypothesis, CMPurpose] = DeepDepth
  296   | otherwise = SurfaceDepth
