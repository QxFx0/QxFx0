   35 applyDreamDynamics :: UTCTime -> SystemState -> TurnInput -> TurnSignals -> TurnPlan -> TurnArtifacts -> MeaningGraph -> (Dream.DreamState, MeaningGraph, Int)
   36 applyDreamDynamics now ss ti ts tp ta baseGraph =
   37   let currentDreamState = semDreamState (ssSemantic ss)
   38       evidence = buildDreamThemeEvidence ti ts tp ta
   39       (dreamState', cycleLogs) =
   40         Dream.runDreamCatchup Dream.defaultDreamConfig evidence now currentDreamState
   41       edgeDeltas =
   42         if null cycleLogs
   43           then []
   44           else
   45             [ (edge, dreamEdgeDelta attractor edge)
   46             | edge <- mgEdges baseGraph
   47             , meCount edge >= dreamRewireMinEdgeCount
   48             ]
   49       (rewired, rewireEvents) =
   50         if null edgeDeltas
   51           then (baseGraph, [])
   52           else rewireMeaningGraphForDreamCycle now edgeDeltas baseGraph
   53       attractor = Dream.dsBiasAttractor dreamState'
   54   in (dreamState', rewired, length cycleLogs + length rewireEvents)

   56 buildDreamThemeEvidence :: TurnInput -> TurnSignals -> TurnPlan -> TurnArtifacts -> [Dream.DreamThemeEvidence]
   57 buildDreamThemeEvidence ti ts tp ta =
   58   [ Dream.DreamThemeEvidence
   59       { Dream.dteTheme = dreamThemeLabel
   60       , Dream.dteBias = dreamFamilyBiasProfile (tdFamily (taDecision ta))
   61       , Dream.dteExperienceWeight =
   62           min 1.0
   63             ( dreamExperienceWeightBase
   64             + tsIntuitPosterior ts * dreamExperienceWeightIntuitionScale
   65             + asLoad (tiAtomSet ti) * dreamExperienceWeightLoadScale
   66             )
   67       , Dream.dteQualityWeight = qualityWeight
   68       , Dream.dteBiographyPermission = not (tpShadowGateTriggered tp)
   69       }
   70   ]
   71   where
   72     dreamThemeLabel =
   73       let topic = tiBestTopic ti
   74       in if T.null topic then T.pack (show (tdFamily (taDecision ta))) else topic
   75     qualityWeight
   76       | tpShadowStatus tp == ShadowUnavailable =
   77           max dreamQualityWeightShadowUnavailableFloor
   78             (tpLegitScore tp * dreamQualityWeightShadowUnavailableScale)
   79       | tpShadowStatus tp == ShadowDiverged =
   80           max dreamQualityWeightShadowDivergedFloor
   81             (tpLegitScore tp * dreamQualityWeightShadowDivergedScale)
   82       | otherwise = min 1.0 (tpLegitScore tp + dreamQualityWeightStableBonus)
   83 
   84 dreamEdgeDelta :: CoreVec -> MeaningEdge -> Double
   85 dreamEdgeDelta attractor edge =
   86   let normWeight = min 1.0 (vecNorm attractor / dreamAttractorNormScale)
   87       attractorWeight =
   88         cvPresence attractor
   89           + cvDepth attractor
   90           + cvSteadiness attractor
   91           - cvDirectiveness attractor * dreamAttractorDirectivenessPenalty
   92       direction =
   93         if successRate edge > dreamRewireSuccessRateThreshold then 1.0 else -1.0
   94   in direction * dreamRewireBiasDelta * max dreamRewireWeightFloor (normWeight * max 0.0 attractorWeight)
