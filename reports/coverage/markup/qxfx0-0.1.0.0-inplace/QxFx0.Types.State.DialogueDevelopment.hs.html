  159     , "adrBoundedDelta" .= adrBoundedDelta record

  165 instance FromJSON AdaptiveDecisionRecord where
  166   parseJSON = withObject "AdaptiveDecisionRecord" $ \o -> do
  167     turn <- o .: "adrTurn"
  168     cause <- o .: "adrCause"
  169     evidence <- o .: "adrEvidence"
  170     confidence <- o .: "adrConfidence"
  171     boundedDelta <- o .:? "adrBoundedDelta" .!= []
  172     decision <- o .: "adrDecision"
  173     targets <- o .: "adrTargets"
  174     mutationRecords <- o .:? "adrMutationRecords" .!= legacyMutationRecords turn cause evidence confidence boundedDelta decision targets
  175     pure AdaptiveDecisionRecord
  176       { adrTurn = turn
  177       , adrCause = cause
  178       , adrEvidence = evidence
  179       , adrConfidence = confidence
  180       , adrBoundedDelta = boundedDelta
  181       , adrDecision = decision
  182       , adrTargets = targets
  183       , adrMutationRecords = mutationRecords
  184       }

  323 legacyEvidenceStrength :: Bool -> EvidenceStrength
  324 legacyEvidenceStrength strong =
  325   if strong then EvidenceStrong else EvidenceWeak
  326 
  327 legacyAdaptiveDecisionRecord
  328   :: Int
  329   -> DialogueOutcomeKind
  330   -> [Text]
  331   -> Bool
  332   -> EvidenceStrength
  333   -> AdaptiveDecisionRecord
  334 legacyAdaptiveDecisionRecord turn kind signals strong strength = AdaptiveDecisionRecord
  335   { adrTurn = turn
  336   , adrCause = "legacy_dialogue_outcome:" <> T.pack (show kind)
  337   , adrEvidence = signals
  338   , adrConfidence = legacyConfidence strength
  339   , adrBoundedDelta = ["recent_outcomes<=12"]
  340   , adrDecision = if strong then AdaptiveAccepted else AdaptiveObserved
  341   , adrTargets = legacyTargets kind strong
  342   , adrMutationRecords = legacyMutationRecords
  343       turn
  344       ("legacy_dialogue_outcome:" <> T.pack (show kind))
  345       signals
  346       (legacyConfidence strength)
  347       ["recent_outcomes<=12"]
  348       (if strong then AdaptiveAccepted else AdaptiveObserved)
  349       (legacyTargets kind strong)
  350   }
  351 
  352 legacyConfidence :: EvidenceStrength -> Double
  353 legacyConfidence strength =
  354   case strength of
  355     EvidenceObservational -> 0.10
  356     EvidenceWeak -> 0.25
  357     EvidenceModerate -> 0.50
  358     EvidenceStrong -> 0.75
  359 
  360 legacyTargets :: DialogueOutcomeKind -> Bool -> [AdaptiveMutationKind]
  361 legacyTargets kind strong =
  362   MutDialogueOutcome :
  363     if not strong
  364       then []
  365       else case kind of
  366         DialogueOutcomeSuccess -> [MutSpeechPolicy, MutClaimStance]
  367         DialogueOutcomeConflict -> [MutSpeechPolicy, MutClaimStance]
  368         DialogueOutcomeRepairRequested -> [MutSpeechPolicy]
  369         DialogueOutcomeRepeatedQuestion -> [MutSpeechPolicy]
  370         DialogueOutcomeDegraded -> [MutSpeechPolicy]
  371         DialogueOutcomePartialSuccess -> []
  372         DialogueOutcomeUncertain -> []
  373 
  374 legacyMutationRecords
  375   :: Int
  376   -> Text
  377   -> [Text]
  378   -> Double
  379   -> [Text]
  380   -> AdaptiveDecision
  381   -> [AdaptiveMutationKind]
  382   -> [AdaptiveMutationRecord]
  383 legacyMutationRecords turn cause evidence confidence boundedDelta decision targets =
  384   map targetRecord targets
  385   where
  386     targetRecord target = AdaptiveMutationRecord
  387       { amrTurnId = turn
  388       , amrKind = target
  389       , amrCause = cause
  390       , amrEvidence = evidence
  391       , amrEvidenceStrength = evidenceStrengthForDecision decision
  392       , amrConfidence = confidence
  393       , amrBoundedDelta = parseBoundedDelta boundedDelta
  394       , amrDecision = decision
  395       }
  396 
  397 evidenceStrengthForDecision :: AdaptiveDecision -> EvidenceStrength
  398 evidenceStrengthForDecision decision =
  399   case decision of
  400     AdaptiveAccepted -> EvidenceStrong
  401     AdaptivePromoted -> EvidenceStrong
  402     AdaptiveRolledBack -> EvidenceStrong
  403     AdaptiveObserved -> EvidenceWeak
  404     AdaptiveRejected -> EvidenceWeak
  405     AdaptiveQuarantined -> EvidenceWeak
  406 
  407 parseBoundedDelta :: [Text] -> Maybe Double
  408 parseBoundedDelta deltas =
  409   case deltas of
  410     [] -> Nothing
  411     _ -> Just 1.0
