   21 queryIdentityClaimsByFocus :: NSQL.Database -> [Text] -> IO [IdentityClaimRef]
   22 queryIdentityClaimsByFocus db keywords =
   23   case keywords of
   24     [] -> queryTopClaims db
   25     keyword : rest -> do
   26       claims <- queryExactTrigger db keyword
   27       if not (null claims)
   28         then pure claims
   29         else do
   30           ftsClaims <- queryFTS5 db keyword
   31           if not (null ftsClaims)
   32             then pure ftsClaims
   33             else do
   34               topicClaims <- queryTopicFallback db keyword
   35               if not (null topicClaims)
   36                 then pure topicClaims
   37                 else queryIdentityClaimsByFocus db rest

   39 queryExactTrigger :: NSQL.Database -> Text -> IO [IdentityClaimRef]
   40 queryExactTrigger db keyword = do
   41   let sql = "SELECT concept, text, confidence, source, topic FROM identity_claims WHERE concept = ? LIMIT 10"
   42   stmt <- prepareQuery db "queryExactTrigger" sql
   43   _ <- NSQL.bindText stmt 1 keyword
   44   claims <- collectClaims stmt
   45   NSQL.finalize stmt
   46   pure claims
   47 
   48 queryFTS5 :: NSQL.Database -> Text -> IO [IdentityClaimRef]
   49 queryFTS5 db keyword = do
   50   let sql = "SELECT concept, text, confidence, source, topic FROM identity_claims WHERE id IN (SELECT rowid FROM identity_claims_fts WHERE text MATCH ? LIMIT 10)"
   51   stmt <- prepareQuery db "queryFTS5" sql
   52   _ <- NSQL.bindText stmt 1 keyword
   53   claims <- collectClaims stmt
   54   NSQL.finalize stmt
   55   pure claims
   56 
   57 queryTopicFallback :: NSQL.Database -> Text -> IO [IdentityClaimRef]
   58 queryTopicFallback db topic = do
   59   let sql = "SELECT concept, text, confidence, source, topic FROM identity_claims WHERE topic = ? LIMIT 10"
   60   stmt <- prepareQuery db "queryTopicFallback" sql
   61   _ <- NSQL.bindText stmt 1 topic
   62   claims <- collectClaims stmt
   63   NSQL.finalize stmt
   64   pure claims

   66 queryTopClaims :: NSQL.Database -> IO [IdentityClaimRef]
   67 queryTopClaims db = do
   68   let sql = "SELECT concept, text, confidence, source, topic FROM identity_claims ORDER BY confidence DESC, id ASC LIMIT 10"
   69   stmt <- prepareQuery db "queryTopClaims" sql
   70   claims <- collectClaims stmt
   71   NSQL.finalize stmt
   72   pure claims
   73 
   74 collectClaims :: NSQL.Statement -> IO [IdentityClaimRef]
   75 collectClaims stmt = go []
   76   where
   77     go acc = do
   78       hasRow <- NSQL.stepRow stmt
   79       if hasRow
   80         then do
   81           concept <- NSQL.columnText stmt 0
   82           text_ <- NSQL.columnText stmt 1
   83           conf <- NSQL.columnDouble stmt 2
   84           source <- NSQL.columnText stmt 3
   85           topic <- NSQL.columnText stmt 4
   86           go (IdentityClaimRef concept text_ conf source topic : acc)
   87         else pure (reverse acc)

   89 maybeCheckpoint :: NSQL.Database -> Int -> IO ()
   90 maybeCheckpoint db turnCount =
   91   when (turnCount > 0 && turnCount `mod` 100 == 0) $ do
   92     _ <- NSQL.execSql db "PRAGMA wal_checkpoint(TRUNCATE);"
   93     pure ()
   94   where
   95     when True action = action
   96     when False _ = pure ()

   98 loadScenes :: NSQL.Database -> IO [SemanticScene]
   99 loadScenes db = do
  100   let sql = "SELECT MIN(confidence), MAX(confidence), topic, topic FROM identity_claims GROUP BY topic"
  101   stmt <- prepareQuery db "loadScenes" sql
  102   scenes <- collectScenes stmt
  103   NSQL.finalize stmt
  104   pure scenes
  105   where
  106     collectScenes stmt = go []
  107       where
  108         go acc = do
  109           hasRow <- NSQL.stepRow stmt
  110           if hasRow
  111             then do
  112               minC <- NSQL.columnDouble stmt 0
  113               maxC <- NSQL.columnDouble stmt 1
  114               desc <- NSQL.columnText stmt 2
  115               _ <- NSQL.columnText stmt 3
  116               go (SemanticScene minC maxC [] desc : acc)
  117             else pure (reverse acc)

  119 loadClusters :: NSQL.Database -> IO [ClusterDef]
  120 loadClusters db = do
  121   let sql = "SELECT name, keywords, priority FROM semantic_clusters ORDER BY priority DESC"
  122   stmt <- prepareQuery db "loadClusters" sql
  123   clusters <- collectClusters stmt
  124   NSQL.finalize stmt
  125   pure clusters
  126   where
  127     collectClusters stmt = go []
  128       where
  129         go acc = do
  130           hasRow <- NSQL.stepRow stmt
  131           if hasRow
  132             then do
  133               name <- NSQL.columnText stmt 0
  134               kwText <- NSQL.columnText stmt 1
  135               prio <- NSQL.columnDouble stmt 2
  136               let kws = T.splitOn "," kwText
  137               go (ClusterDef name kws prio : acc)
  138             else pure (reverse acc)

  140 prepareQuery :: NSQL.Database -> Text -> Text -> IO NSQL.Statement
  141 prepareQuery db ctx sql = do
  142   mStmt <- NSQL.prepare db sql
  143   case mStmt of
  144     Left err -> throwQxFx0 (SQLiteError ("prepare " <> ctx <> " failed: " <> err))
  145     Right stmt -> pure stmt
