   18 applyPerspectiveDecision :: Int -> PerspectiveRegistry -> PerspectiveInputBundle -> PerspectiveCandidate -> PerspectivePromotionDecision -> PerspectiveRegistry
   19 applyPerspectiveDecision turn registry bundle candidate decision =
   20   case decision of
   21     PpdObserveOnly -> registry { prLastUpdatedTurn = turn }
   22     PpdQuarantine -> upsertThread turn registry bundle candidate decision PerspectiveContested Nothing
   23     PpdAcceptBounded -> upsertThread turn registry bundle candidate decision PerspectiveContested Nothing
   24     PpdPromoteEndorsed -> upsertThread turn registry bundle candidate decision PerspectiveActive (Just turn)
   25     PpdReviseActive -> upsertThread turn registry bundle candidate decision PerspectiveActive (Just turn)
   26     PpdSuspendActive -> suspendActivePerspective turn registry candidate decision
   27     PpdRollbackPrior -> rollbackPriorPerspective turn registry candidate decision

   29 buildPerspectiveProjection :: PerspectiveRegistry -> PerspectiveScope -> Maybe PerspectiveProjection
   30 buildPerspectiveProjection registry scope = do
   31   thread <- M.lookup scope (prThreads registry)
   32   endorsed <- activeEndorsedPerspective thread
   33   pure PerspectiveProjection
   34     { ppScope = epScope endorsed
   35     , ppSummary = T.take 180 (epThesis endorsed)
   36     , ppOrientation = epOrientation endorsed
   37     , ppConfidenceBand = confidenceBand (epConfidence endorsed)
   38     , ppCautionLevel = cautionLevel endorsed
   39     , ppContested = epStatus endorsed `elem` [PerspectiveContested, PerspectiveSuspended]
   40     , ppPerspectiveVersion = epVersion endorsed
   41     , ppNormativeProfileId = epNormativeProfileId endorsed
   42     , ppNormativeProfileVersion = epNormativeProfileVersion endorsed
   43     , ppEvidenceCount = length (epSupportingClaims endorsed)
   44     , ppCounterargumentCount = length (epCounterarguments endorsed)
   45     , ppExplanationHandle = explanationHandle endorsed
   46     }
   47 
   48 buildActivePerspectiveProjections :: PerspectiveRegistry -> [PerspectiveProjection]
   49 buildActivePerspectiveProjections registry =
   50   mapMaybe (buildPerspectiveProjection registry) (activePerspectiveProjectionScopes registry)

   52 upsertThread :: Int -> PerspectiveRegistry -> PerspectiveInputBundle -> PerspectiveCandidate -> PerspectivePromotionDecision -> PerspectiveStatus -> Maybe Int -> PerspectiveRegistry
   53 upsertThread turn registry bundle candidate decision status endorsedTurn =
   54   let scope = pcScope candidate
   55       oldThread = M.lookup scope (prThreads registry)
   56       perspectiveId = maybe (PerspectiveId ("perspective-" <> T.pack (show (prNextPerspectiveOrdinal registry)))) ptPerspectiveId oldThread
   57       version = PerspectiveVersionId (prNextVersionOrdinal registry)
   58       priorActive = oldThread >>= ptActiveVersion
   59       oldConfidence = maybe 0.0 epConfidence (oldThread >>= activeEndorsedPerspective)
   60       oldNormative = maybe (npVersionId (pibNormativeProfile bundle)) epNormativeProfileVersion (oldThread >>= activeEndorsedPerspective)
   61       endorsed = EndorsedPerspective
   62         { epId = perspectiveId
   63         , epVersion = version
   64         , epScope = scope
   65         , epThesis = pcThesis candidate
   66         , epOrientation = pcOrientation candidate
   67         , epConfidence = pcConfidence candidate
   68         , epNormativeProfileId = npId (pibNormativeProfile bundle)
   69         , epNormativeProfileVersion = npVersionId (pibNormativeProfile bundle)
   70         , epStatus = status
   71         , epCreatedTurn = turn
   72         , epEndorsedTurn = endorsedTurn
   73         , epSupportingClaims = take 8 (pcSupportingClaims candidate)
   74         , epCounterarguments = take 8 (map renderCounterargumentRef (pibCounterarguments bundle))
   75         }
   76       revision = PerspectiveRevisionRecord
   77         { prrFromVersion = priorActive
   78         , prrToVersion = version
   79         , prrTrigger = promotionTrigger decision candidate
   80         , prrEvidenceDelta = fromIntegral (length (pibEvidence bundle))
   81         , prrCounterargumentDelta = pcCounterargumentPressure candidate
   82         , prrConfidenceDelta = pcConfidence candidate - oldConfidence
   83         , prrNormativeDelta = fromIntegral (npVersionId (pibNormativeProfile bundle) - oldNormative)
   84         , prrNormativeProfileVersion = npVersionId (pibNormativeProfile bundle)
   85         , prrDecision = decision
   86         , prrRollbackOf = Nothing
   87         }
   88       oldVersions = maybe [] ptVersions oldThread
   89       oldRevisions = maybe [] ptRevisionHistory oldThread
   90       activeVersion = if status == PerspectiveActive then Just version else priorActive
   91       thread = PerspectiveThread
   92         { ptPerspectiveId = perspectiveId
   93         , ptScope = scope
   94         , ptActiveVersion = activeVersion
   95         , ptVersions = boundVersions registry (endorsed : markPriorRevised status oldVersions)
   96         , ptRevisionHistory = take (prMaxRevisionsPerScope registry) (revision : oldRevisions)
   97         , ptStatus = status
   98         , ptLastUpdatedTurn = turn
   99         }
  100       threads = enforceRegistryBounds registry (M.insert scope thread (prThreads registry))
  101       nextPerspectiveOrdinal = case oldThread of
  102         Nothing -> prNextPerspectiveOrdinal registry + 1
  103         Just _ -> prNextPerspectiveOrdinal registry
  104   in registry
  105       { prThreads = threads
  106       , prNextPerspectiveOrdinal = nextPerspectiveOrdinal
  107       , prNextVersionOrdinal = prNextVersionOrdinal registry + 1
  108       , prLastUpdatedTurn = turn
  109       }

  111 suspendActivePerspective :: Int -> PerspectiveRegistry -> PerspectiveCandidate -> PerspectivePromotionDecision -> PerspectiveRegistry
  112 suspendActivePerspective turn registry candidate decision =
  113   case M.lookup (pcScope candidate) (prThreads registry) of
  114     Nothing -> registry { prLastUpdatedTurn = turn }
  115     Just thread ->
  116       let revision = PerspectiveRevisionRecord
  117             { prrFromVersion = ptActiveVersion thread
  118             , prrToVersion = fromMaybe (PerspectiveVersionId (prNextVersionOrdinal registry)) (ptActiveVersion thread)
  119             , prrTrigger = "counterargument_pressure_suspend"
  120             , prrEvidenceDelta = 0.0
  121             , prrCounterargumentDelta = pcCounterargumentPressure candidate
  122             , prrConfidenceDelta = 0.0
  123             , prrNormativeDelta = 0.0
  124             , prrNormativeProfileVersion = pcNormativeProfileVersion candidate
  125             , prrDecision = decision
  126             , prrRollbackOf = Nothing
  127             }
  128           thread' = thread
  129             { ptActiveVersion = Nothing
  130             , ptVersions = map (setStatusForActive (ptActiveVersion thread) PerspectiveSuspended) (ptVersions thread)
  131             , ptRevisionHistory = take (prMaxRevisionsPerScope registry) (revision : ptRevisionHistory thread)
  132             , ptStatus = PerspectiveSuspended
  133             , ptLastUpdatedTurn = turn
  134             }
  135       in registry
  136           { prThreads = enforceRegistryBounds registry (M.insert (pcScope candidate) thread' (prThreads registry))
  137           , prLastUpdatedTurn = turn
  138           }

  140 rollbackPriorPerspective :: Int -> PerspectiveRegistry -> PerspectiveCandidate -> PerspectivePromotionDecision -> PerspectiveRegistry
  141 rollbackPriorPerspective turn registry candidate decision =
  142   case M.lookup (pcScope candidate) (prThreads registry) of
  143     Nothing -> registry { prLastUpdatedTurn = turn }
  144     Just thread ->
  145       let active = ptActiveVersion thread
  146           remaining = filter (isViableRollbackVersion active) (ptVersions thread)
  147           replacement = case remaining of
  148             x:_ -> Just (epVersion x)
  149             [] -> Nothing
  150           revision = PerspectiveRevisionRecord
  151             { prrFromVersion = active
  152             , prrToVersion = fromMaybe (PerspectiveVersionId (prNextVersionOrdinal registry)) replacement
  153             , prrTrigger = "rollback_prior_promotion"
  154             , prrEvidenceDelta = 0.0
  155             , prrCounterargumentDelta = pcCounterargumentPressure candidate
  156             , prrConfidenceDelta = 0.0
  157             , prrNormativeDelta = 0.0
  158             , prrNormativeProfileVersion = pcNormativeProfileVersion candidate
  159             , prrDecision = decision
  160             , prrRollbackOf = active
  161             }
  162           thread' = thread
  163             { ptActiveVersion = replacement
  164             , ptVersions = map (statusAfterRollback active replacement) (ptVersions thread)
  165             , ptRevisionHistory = take (prMaxRevisionsPerScope registry) (revision : ptRevisionHistory thread)
  166             , ptStatus = maybe PerspectiveWithdrawn (const PerspectiveActive) replacement
  167             , ptLastUpdatedTurn = turn
  168             }
  169       in registry
  170           { prThreads = enforceRegistryBounds registry (M.insert (pcScope candidate) thread' (prThreads registry))
  171           , prLastUpdatedTurn = turn
  172           }

  174 promotionTrigger :: PerspectivePromotionDecision -> PerspectiveCandidate -> Text
  175 promotionTrigger decision candidate =
  176   promotionDecisionText decision <> ":confidence=" <> T.pack (show (pcConfidence candidate))
  177 
  178 promotionDecisionText :: PerspectivePromotionDecision -> Text
  179 promotionDecisionText decision =
  180   case decision of
  181     PpdObserveOnly -> "observe_only"
  182     PpdQuarantine -> "quarantine"
  183     PpdAcceptBounded -> "accept_bounded"
  184     PpdPromoteEndorsed -> "promote_endorsed"
  185     PpdReviseActive -> "revise_active"
  186     PpdSuspendActive -> "suspend_active"
  187     PpdRollbackPrior -> "rollback_prior"

  189 boundVersions :: PerspectiveRegistry -> [EndorsedPerspective] -> [EndorsedPerspective]
  190 boundVersions registry versions =
  191   let active = filter ((== PerspectiveActive) . epStatus) versions
  192       inactive = filter ((/= PerspectiveActive) . epStatus) versions
  193   in take (prMaxActivePerScope registry) active <> take (prMaxInactiveVersions registry) inactive

  195 enforceRegistryBounds :: PerspectiveRegistry -> M.Map PerspectiveScope PerspectiveThread -> M.Map PerspectiveScope PerspectiveThread
  196 enforceRegistryBounds registry threads =
  197   let activeThreads = filter (hasActiveThread . snd) (M.toList threads)
  198       allowedActiveScopes = map fst . take (prMaxActivePerspectives registry) . sortOn (Down . ptLastUpdatedTurn . snd) $ activeThreads
  199   in M.mapWithKey (enforceActiveScope allowedActiveScopes) threads
  200 
  201 hasActiveThread :: PerspectiveThread -> Bool
  202 hasActiveThread thread = activeEndorsedPerspective thread /= Nothing
  203 
  204 enforceActiveScope :: [PerspectiveScope] -> PerspectiveScope -> PerspectiveThread -> PerspectiveThread
  205 enforceActiveScope allowedScopes scope thread
  206   | scope `elem` allowedScopes = thread
  207   | hasActiveThread thread = suspendThreadForActiveCap thread
  208   | otherwise = thread
  209 
  210 suspendThreadForActiveCap :: PerspectiveThread -> PerspectiveThread
  211 suspendThreadForActiveCap thread = thread
  212   { ptActiveVersion = Nothing
  213   , ptVersions = map (setStatusForActive (ptActiveVersion thread) PerspectiveSuspended) (ptVersions thread)
  214   , ptStatus = PerspectiveSuspended
  215   }
  216 
  217 markPriorRevised :: PerspectiveStatus -> [EndorsedPerspective] -> [EndorsedPerspective]
  218 markPriorRevised status versions
  219   | status == PerspectiveActive = map reviseActive versions
  220   | otherwise = versions
  221   where
  222     reviseActive ep
  223       | epStatus ep == PerspectiveActive = ep { epStatus = PerspectiveRevised }
  224       | otherwise = ep
  225 
  226 setStatusForActive :: Maybe PerspectiveVersionId -> PerspectiveStatus -> EndorsedPerspective -> EndorsedPerspective
  227 setStatusForActive activeVersion status ep
  228   | Just (epVersion ep) == activeVersion = ep { epStatus = status }
  229   | otherwise = ep
  230 
  231 setStatusForVersion :: Maybe PerspectiveVersionId -> PerspectiveStatus -> EndorsedPerspective -> EndorsedPerspective
  232 setStatusForVersion version status ep
  233   | Just (epVersion ep) == version = ep { epStatus = status }
  234   | otherwise = ep
  235 
  236 statusAfterRollback :: Maybe PerspectiveVersionId -> Maybe PerspectiveVersionId -> EndorsedPerspective -> EndorsedPerspective
  237 statusAfterRollback active replacement =
  238   setStatusForVersion replacement PerspectiveActive . setStatusForActive active PerspectiveWithdrawn
  239 
  240 isViableRollbackVersion :: Maybe PerspectiveVersionId -> EndorsedPerspective -> Bool
  241 isViableRollbackVersion active ep =
  242   Just (epVersion ep) /= active
  243     && epStatus ep `elem` [PerspectiveRevised, PerspectiveContested, PerspectiveActive]
  244 
  245 confidenceBand :: Double -> Text
  246 confidenceBand value
  247   | value >= 0.80 = "high"
  248   | value >= 0.60 = "medium"
  249   | value >= 0.40 = "low"
  250   | otherwise = "minimal"
  251 
  252 cautionLevel :: EndorsedPerspective -> Text
  253 cautionLevel ep
  254   | epStatus ep `elem` [PerspectiveContested, PerspectiveSuspended] = "high"
  255   | epConfidence ep < 0.60 = "medium"
  256   | null (epCounterarguments ep) = "low"
  257   | otherwise = "medium"

  259 explanationHandle :: EndorsedPerspective -> Text
  260 explanationHandle ep =
  261   renderPerspectiveId (epId ep)
  262     <> ":v"
  263     <> T.pack (show (unPerspectiveVersionId (epVersion ep)))
  264     <> ":np"
  265     <> T.pack (show (epNormativeProfileVersion ep))

  267 renderCounterargumentRef :: CounterargumentRef -> Text
  268 renderCounterargumentRef (CounterargumentRef value) = value
