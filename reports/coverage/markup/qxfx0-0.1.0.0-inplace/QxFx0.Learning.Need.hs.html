    7 {-|
    8 Module      : QxFx0.Learning.Need
    9 Description : WP1 — Endogenous learning diagnostic drive.
   10 
   11 Computes a typed 'LearningNeed' from persistent runtime signal patterns.
   12 A need is raised only when a pattern is stable across a minimum window
   13 of turns; single-turn noise does not create a need.
   14 
   15 The three need classes mirror the three sustainable growth axes:
   16 
   17 * 'NeedSalienceCalibration' — weights are drifting but no empirical
   18   signal is available to correct them.
   19 * 'NeedKeywordEnrichment' — atom-trace shows repeated Searching hits
   20   with low consolidation (high counterfactual entropy).
   21 * 'NeedLexiconExtension' — morphology gaps are causing repeated
   22   fallback / unknown-topic recovery.
   23 -}

   49 -- | A typed deficit that the system can report to itself (and
   50 -- optionally to an external tool) as a reason to acquire new
   51 -- knowledge.
   52 data LearningNeed
   53   = NeedSalienceCalibration
   54     -- ^ Salience weights may be mis-calibrated; calibration data is
   55     --   needed to adjust 'SalienceWeights' empirically.
   56   | NeedKeywordEnrichment
   57     -- ^ Atom trace shows unresolved Searching / Doubt patterns with
   58     --   low consolidation; keyword coverage is insufficient.
   59   | NeedLexiconExtension
   60     -- ^ Morphology gaps cause repeated unknown-topic recovery or
   61     --   template fallback.
   62   | NeedNone
   63     -- ^ No persistent deficit detected in the current window.
   64   deriving stock (Eq, Show, Generic)
   65     deriving anyclass (NFData, FromJSON, ToJSON)
   66 
   67 -- | Trend of a need across the observation window.
   68 data NeedTrend
   69   = TrendRising
   70     -- ^ Deficit metric is increasing (getting worse).
   71   | TrendStable
   72     -- ^ Deficit metric is within a dead band.
   73   | TrendFalling
   74     -- ^ Deficit metric is decreasing (improving).
   75   deriving stock (Eq, Show, Generic)
   76     deriving anyclass (NFData, FromJSON, ToJSON)

   78 -- | State of the diagnostic drive.  Kept in 'SystemState' and
   79 -- updated once per turn in 'buildNextSystemState'.
   80 data LearningNeedState = LearningNeedState
   81   { lnsCurrentNeed :: !LearningNeed
   82     -- ^ The currently active need (or 'NeedNone').
   83     --   This is the threshold-gated need exposed to downstream.
   84   , lnsCandidateNeed :: !LearningNeed
   85     -- ^ The raw dominant need before persistence threshold is applied.
   86     --   Used internally so that persistence can accumulate across turns
   87     --   even when the threshold has not yet been met.
   88   , lnsLevel :: !Double
   89     -- ^ Normalised deficit severity in [0, 1].  0 = no deficit.
   90   , lnsTrend :: !NeedTrend
   91     -- ^ Direction of change over the observation window.
   92   , lnsPersistence :: !Int
   93     -- ^ How many consecutive turns this need has been observed.
   94     --   Resets to 0 when the need class changes or drops to None.
   95   , lnsLastSeenTurn :: !Int
   96     -- ^ Turn count of the most recent observation.
   97   , lnsHistory :: ![(Int, Double)]
   98     -- ^ (turn, level) pairs for the last N turns (capped at 20).
   99   , lnsUnknownWindowCount :: !Int
  100     -- ^ WP6.1: count of unknown-topic mentions in the current window.
  101   , lnsWindowStartTurn :: !Int
  102     -- ^ WP6.1: turn when the current observation window started.
  103   , lnsWindowGraftBaseline :: !Int
  104     -- ^ WP6.1: grafted count at window start (for stagnation detection).
  105   }
  106   deriving stock (Eq, Show, Generic)
  107     deriving anyclass (NFData)
  108

  176 -- | Backward-compatible wrapper: delegates to 'detectLearningNeedWithPressure'
  177 -- with empty pressure signals (keeps old conatus-based lexicon heuristic).
  178 detectLearningNeed
  179   :: ConatusEnergy
  180   -> Field
  181   -> Int
  182      -- ^ Number of repair-loop recoveries in the last 10 turns.
  183   -> Int
  184      -- ^ Number of unknown-topic recoveries in the last 10 turns.
  185   -> Int
  186      -- ^ Current turn count.
  187   -> LearningNeedState
  188   -> LearningNeedState
  189 detectLearningNeed conatusEnergy field repairCount unknownTopicCount turnCount oldState =
  190   detectLearningNeedWithPressure
  191     defaultLearningPressureConfig
  192     conatusEnergy
  193     field
  194     repairCount
  195     unknownTopicCount
  196     turnCount
  197     oldState
  198     False   -- isTopicUnknown
  199     0       -- currentGraftedCount

  201 -- | WP6.1: Detect a learning need with separate learning-pressure signals.
  202 --
  203 -- 'NeedLexiconExtension' is now driven by learning-pressure signals
  204 -- (unknown-topic window count, graft stagnation) instead of the Conatus
  205 -- scalar floor.  This decouples substrate-learning from structural
  206 -- health so that a large, healthy morphology does not permanently
  207 -- suppress lexicon extension.
  208 --
  209 -- Rules (priority order):
  210 -- 1. 'NeedLexiconExtension' — learning-pressure signals indicate
  211 --    substrate gap (unknown mentions in window + graft stagnation).
  212 -- 2. 'NeedSalienceCalibration' — unchanged (Conatus + field signals).
  213 -- 3. 'NeedKeywordEnrichment' — unchanged (field signals).
  214 -- 4. Otherwise 'NeedNone'.
  215 detectLearningNeedWithPressure
  216   :: LearningPressureConfig
  217   -> ConatusEnergy
  218   -> Field
  219   -> Int
  220   -> Int
  221   -> Int
  222   -> LearningNeedState
  223   -> Bool        -- ^ Is the current topic unknown (not in morphology/tree)?
  224   -> Int         -- ^ Current ktGraftedCount (for stagnation detection).
  225   -> LearningNeedState
  226 detectLearningNeedWithPressure cfg conatusEnergy field _repairCount unknownTopicCount turnCount oldState isTopicUnknown currentGraftedCount =
  227   let -- Window management
  228       windowSize = lpcWindowSize cfg
  229       windowExpired = turnCount - lnsWindowStartTurn oldState > windowSize
  230       -- Reset window if expired or on first call (windowStartTurn == 0)
  231       newWindowStart = if windowExpired || lnsWindowStartTurn oldState == 0
  232                          then turnCount
  233                          else lnsWindowStartTurn oldState
  234       newUnknownCount = if windowExpired
  235                           then (if isTopicUnknown then 1 else 0)
  236                           else lnsUnknownWindowCount oldState + (if isTopicUnknown then 1 else 0)
  237       newGraftBaseline = if windowExpired
  238                            then currentGraftedCount
  239                            else lnsWindowGraftBaseline oldState
  240       graftsInWindow = max 0 (currentGraftedCount - newGraftBaseline)
  241       stagnation = (turnCount - newWindowStart) >= lpcStagnationTurns cfg
  242                      && graftsInWindow == 0
  243                      && newUnknownCount >= lpcMinUnknownCount cfg
  244 
  245       -- WP6.1: Lexicon-extension heuristic — learning pressure, NOT conatus floor
  246       lexiconLevel =
  247         if newUnknownCount >= lpcMinUnknownCount cfg && stagnation
  248            then 0.7 + 0.05 * fromIntegral (min 5 newUnknownCount)
  249            else 0.0
  250 
  251       -- Salience-calibration heuristic: unchanged (Conatus + field)
  252       salienceLevel =
  253         let conf = unFieldConfidence (fieldConfidence field)
  254             cf   = unCounterfactual (fieldCounterfactual field)
  255         in if conf < 0.4 && cf > 0.5
  256               then 0.5 + 0.2 * (0.5 - conf) + 0.2 * cf
  257               else 0.0
  258 
  259       -- Keyword-enrichment heuristic: unchanged
  260       keywordLevel =
  261         let cons = unConsolidation (fieldConsolidation field)
  262             cf   = unCounterfactual (fieldCounterfactual field)
  263         in if cf > 0.6 && cons < 0.3
  264               then 0.6 + 0.2 * cf - 0.2 * cons
  265               else 0.0
  266 
  267       (candidateNeed, candidateLevel) =
  268         if lexiconLevel > 0.0
  269            then (NeedLexiconExtension, min 1.0 lexiconLevel)
  270            else if salienceLevel > 0.0
  271                 then (NeedSalienceCalibration, min 1.0 salienceLevel)
  272                 else if keywordLevel > 0.0
  273                      then (NeedKeywordEnrichment, min 1.0 keywordLevel)
  274                      else (NeedNone, 0.0)
  275 
  276       newPersistence =
  277         if candidateNeed == lnsCandidateNeed oldState && candidateNeed /= NeedNone
  278            then lnsPersistence oldState + 1
  279            else if candidateNeed == NeedNone
  280                 then 0
  281                 else 1
  282 
  283       newHistory = take maxHistoryLength ((turnCount, candidateLevel) : lnsHistory oldState)
  284       newTrend = computeTrend newHistory
  285 
  286       finalNeed =
  287         if candidateNeed /= NeedNone && newPersistence >= minPersistenceThreshold
  288            then candidateNeed
  289            else NeedNone
  290 
  291       finalLevel = if finalNeed == NeedNone then 0.0 else candidateLevel
  292 
  293   in LearningNeedState
  294        { lnsCurrentNeed = finalNeed
  295        , lnsCandidateNeed = candidateNeed
  296        , lnsLevel = finalLevel
  297        , lnsTrend = newTrend
  298        , lnsPersistence = newPersistence
  299        , lnsLastSeenTurn = turnCount
  300        , lnsHistory = newHistory
  301        , lnsUnknownWindowCount = newUnknownCount
  302        , lnsWindowStartTurn = newWindowStart
  303        , lnsWindowGraftBaseline = newGraftBaseline
  304        }
  305 
  306 computeTrend :: [(Int, Double)] -> NeedTrend
  307 computeTrend history =
  308   case history of
  309     [] -> TrendStable
  310     [_] -> TrendStable
  311     -- Compare last two points
  312     (_, y1) : (_, y0) : _
  313       | abs (y1 - y0) < 0.05 -> TrendStable
  314       | y1 > y0              -> TrendRising
  315       | otherwise            -> TrendFalling
