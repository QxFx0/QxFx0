   16 renderSemanticIntrospection :: SystemState -> Maybe TurnReplayTrace -> Text
   17 renderSemanticIntrospection ss mTrace =
   18   let trace = ssTrace ss
   19       ego = ssEgo ss
   20       mgEdgeCount = length (mgEdges (ssMeaningGraph ss))
   21       truthTag = maybe (textShow (ssTruthContractStatus ss)) (textShow . trcTruthContractStatus) mTrace
   22       authorityTag = maybe "none" (maybe "none" textShow . trcAuthorityClass) mTrace
   23       assemblyPathTag = maybe "none" (maybe "none" textShow . trcAssemblyPath) mTrace
   24       replayProvenanceTag = maybe "ReplayProvenanceLegacyIncomplete" (textShow . trcReplayProvenanceStatus) mTrace
   25       surfaceProvenanceTag = maybe "none" (maybe "none" textShow . trcSurfaceProvenance) mTrace
   26       contractProvenanceTag = maybe "none" (maybe "none" textShow . trcContractProvenance) mTrace
   27   in T.unlines
   28      [ "SEMANTIC_INTROSPECTION_BEGIN"
   29      , "turn: " <> textShow (ssTurnCount ss)
   30       , "ema: " <> textShow (atCurrentLoad trace)
   31       , "register: " <> inferRegisterFromTrace trace
   32       , "family: " <> textShow (ssLastFamily ss)
   33       , "focus: " <> ssLastTopic ss
   34       , "truth_contract: " <> truthTag
   35       , "authority_class: " <> authorityTag
   36       , "assembly_path: " <> assemblyPathTag
   37       , "replay_provenance_status: " <> replayProvenanceTag
   38       , "surface_provenance: " <> surfaceProvenanceTag
   39       , "contract_provenance: " <> contractProvenanceTag
   40       , "governance_runtime_fault: " <> maybe "none" (T.pack . show) (ssGovernanceRuntimeFault ss)
   41       , "nix: " <> textShow (obsNixCache (ssObservability ss))
   42       , "ego_agency: " <> textShow (egoAgency ego)
   43       , "meaning_graph_entries: " <> textShow mgEdgeCount
   44       , "SEMANTIC_INTROSPECTION_END"
   45       ]
   46   where
   47     inferRegisterFromTrace t
   48       | atCurrentLoad t > semanticIntrospectionHighLoadThreshold = "\1048\1089\1090\1086\1097\1077\1085\1080\1077"
   49       | atCurrentLoad t > semanticIntrospectionMediumLoadThreshold = "\1055\1086\1080\1089\1082"
   50       | otherwise              = "\1053\1077\1081\1090\1088\1072\1083\1100\1085\1086"
