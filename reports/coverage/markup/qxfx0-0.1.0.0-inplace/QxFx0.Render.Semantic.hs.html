   15 renderSemanticIntrospection :: SystemState -> Text
   16 renderSemanticIntrospection ss =
   17   let trace = ssTrace ss
   18       ego = ssEgo ss
   19       mgEdgeCount = length (mgEdges (ssMeaningGraph ss))
   20   in T.unlines
   21      [ "SEMANTIC_INTROSPECTION_BEGIN"
   22      , "turn: " <> textShow (ssTurnCount ss)
   23      , "ema: " <> textShow (atCurrentLoad trace)
   24      , "register: " <> inferRegisterFromTrace trace
   25      , "family: " <> textShow (ssLastFamily ss)
   26      , "focus: " <> ssLastTopic ss
   27       , "nix: " <> textShow (obsNixCache (ssObservability ss))
   28      , "ego_agency: " <> textShow (egoAgency ego)
   29      , "meaning_graph_entries: " <> textShow mgEdgeCount
   30      , "SEMANTIC_INTROSPECTION_END"
   31      ]
   32   where
   33     inferRegisterFromTrace t
   34       | atCurrentLoad t > semanticIntrospectionHighLoadThreshold = "\1048\1089\1090\1086\1097\1077\1085\1080\1077"
   35       | atCurrentLoad t > semanticIntrospectionMediumLoadThreshold = "\1055\1086\1080\1089\1082"
   36       | otherwise              = "\1053\1077\1081\1090\1088\1072\1083\1100\1085\1086"
