   78 flashThreshold :: Double
   79 flashThreshold = intuitionFlashThreshold

   81 likelihoodGivenFlash :: Double -> Double -> Double
   82 likelihoodGivenFlash resonance tension
   83   | resonance > intuitionHighResonanceThreshold && tension > intuitionHighTensionThreshold = intuitionFlashLikelihoodConvergent
   84   | resonance > intuitionElevatedResonanceThreshold || tension > intuitionElevatedTensionThreshold = intuitionFlashLikelihoodElevated
   85   | resonance > intuitionDeepResonanceThreshold = intuitionFlashLikelihoodDeep
   86   | otherwise                          = intuitionFlashLikelihoodBaseline
   87 
   88 likelihoodGivenNoFlash :: Double -> Double -> Double
   89 likelihoodGivenNoFlash resonance tension
   90   | resonance > intuitionHighResonanceThreshold && tension > intuitionHighTensionThreshold = intuitionNoFlashLikelihoodConvergent
   91   | resonance > intuitionElevatedResonanceThreshold || tension > intuitionElevatedTensionThreshold = intuitionNoFlashLikelihoodElevated
   92   | resonance > intuitionDeepResonanceThreshold = intuitionNoFlashLikelihoodDeep
   93   | otherwise                          = intuitionNoFlashBaselineLikelihood
   94 
   95 updatePosterior :: Double -> Double -> Double -> Double
   96 updatePosterior resonance tension prior =
   97   let pEH  = likelihoodGivenFlash  resonance tension
   98       pEnH = likelihoodGivenNoFlash resonance tension
   99       pE   = pEH * prior + pEnH * (1.0 - prior)
  100   in clamp01 ((pEH * prior) / max 1e-9 pE)

  137 checkIntuition :: Double -> Double -> Int -> IntuitiveState -> (Maybe IntuitiveFlash, IntuitiveState)
  138 checkIntuition resonance tension turnNumber state =
  139   let newPost = updatePosterior resonance tension (isPosterior state)
  140       newLongPost = updateLongPosterior resonance tension (isLongPosterior state)
  141       newCooldown = max 0 (isCooldown state - 1)
  142       state' = state
  143         { isPosterior = newPost
  144         , isLongPosterior = newLongPost
  145         , isCooldown = newCooldown
  146         }
  147       currentPosterior = effectivePosterior state'
  148   in if newCooldown > 0
  149      then (Nothing, state')
  150      else if currentPosterior >= flashThreshold
  151            then let flash = buildFlash resonance tension currentPosterior
  152                     state'' = state'
  153                       { isPosterior  = posteriorAfterFlash newPost
  154                       , isLongPosterior = longPosteriorAfterFlash newLongPost
  155                       , isCooldown   = intuitionCooldownTurns
  156                       , isFlashCount = isFlashCount state + 1
  157                       , isLastTurn   = turnNumber
  158                       }
  159                in (Just flash, state'')
  160           else (Nothing, state')

  162 posteriorAfterFlash :: Double -> Double
  163 posteriorAfterFlash posterior =
  164   max basePrior (posterior * intuitionPosteriorAfterFlashDecayFactor)
  165 
  166 longPosteriorAfterFlash :: Double -> Double
  167 longPosteriorAfterFlash posterior =
  168   max basePrior (posterior * intuitionLongPosteriorAfterFlashDecayFactor)
  169 
  170 updateLongPosterior :: Double -> Double -> Double -> Double
  171 updateLongPosterior resonance tension prior =
  172   clamp01
  173     ( prior * intuitionLongPosteriorPriorWeight
  174       + updatePosterior resonance tension prior * intuitionLongPosteriorCurrentWeight
  175     )

  177 buildFlash :: Double -> Double -> Double -> IntuitiveFlash
  178 buildFlash resonance tension posterior =
  179   let trigger = selectTrigger resonance tension
  180       strength = clamp01 ((posterior - flashThreshold) / (1.0 - flashThreshold))
  181       kernelSig = buildKernelSignal trigger
  182       directive = buildDirective trigger strength
  183   in IntuitiveFlash strength trigger kernelSig directive (strength > intuitionFlashOverrideStrengthThreshold)
  184 
  185 selectTrigger :: Double -> Double -> FlashTrigger
  186 selectTrigger resonance tension
  187   | resonance > intuitionHighResonanceThreshold && tension > intuitionHighTensionThreshold = ConvergenceTrigger
  188   | tension > intuitionCrisisTensionThreshold = CrisisMomentTrigger
  189   | resonance > intuitionHighResonanceThreshold = DeepResonanceTrigger
  190   | otherwise                         = PureKernelTrigger

  192 buildKernelSignal :: FlashTrigger -> Text
  193 buildKernelSignal trigger = T.intercalate ". "
  194   [ kernelSignalPresence
  195   , kernelSignalDepth
  196   , triggerContext trigger
  197   ]

  199 triggerContext :: FlashTrigger -> Text
  200 triggerContext DeepResonanceTrigger = triggerContextDeepResonance
  201 triggerContext CrisisMomentTrigger  = triggerContextCrisisMoment
  202 triggerContext PureKernelTrigger    = triggerContextPureKernel
  203 triggerContext ConvergenceTrigger   = triggerContextConvergence

  205 buildDirective :: FlashTrigger -> Double -> Text
  206 buildDirective trigger strength =
  207   let header = intuitionHeaderPrefix <> fmtPctText strength <> "]"
  208       trig   = intuitionTriggerLabel <> renderTrigger trigger
  209       ovrd   = if strength > intuitionFlashOverrideStrengthThreshold
  210                then intuitionOverrideDirective
  211                else intuitionSupplementaryDirective
  212   in T.unlines [header, trig, ovrd, intuitionFormDirective]

  214 fmtPctText :: Double -> Text
  215 fmtPctText value = T.pack (printf "%.0f%%" (value * 100.0) :: String)

  217 intuitionSignalStrength :: Double
  218 intuitionSignalStrength =
  219   let v = qxfx0CoreVec
  220       base = cvPresence v * cvDepth v * (1.0 - cvDirectiveness v) * cvAutonomy v
  221       steadyBonus = 1.0 + (cvSteadiness v - intuitionSteadinessBaseline) * intuitionSignalSteadyBonusScale
  222   in base * steadyBonus
