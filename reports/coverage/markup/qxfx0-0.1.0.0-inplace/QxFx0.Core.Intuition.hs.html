   92 likelihoodGivenFlash :: Double -> Double -> Double
   93 likelihoodGivenFlash resonance tension
   94   | resonance > intuitionHighResonanceThreshold && tension > intuitionHighTensionThreshold = intuitionFlashLikelihoodConvergent
   95   | resonance > intuitionElevatedResonanceThreshold || tension > intuitionElevatedTensionThreshold = intuitionFlashLikelihoodElevated
   96   | resonance > intuitionDeepResonanceThreshold = intuitionFlashLikelihoodDeep
   97   | otherwise                          = intuitionFlashLikelihoodBaseline
   98 
   99 likelihoodGivenNoFlash :: Double -> Double -> Double
  100 likelihoodGivenNoFlash resonance tension
  101   | resonance > intuitionHighResonanceThreshold && tension > intuitionHighTensionThreshold = intuitionNoFlashLikelihoodConvergent
  102   | resonance > intuitionElevatedResonanceThreshold || tension > intuitionElevatedTensionThreshold = intuitionNoFlashLikelihoodElevated
  103   | resonance > intuitionDeepResonanceThreshold = intuitionNoFlashLikelihoodDeep
  104   | otherwise                          = intuitionNoFlashBaselineLikelihood

  106 updatePosterior :: Double -> Double -> Double -> Double
  107 updatePosterior resonance tension prior =
  108   let pEH  = likelihoodGivenFlash  resonance tension
  109       pEnH = likelihoodGivenNoFlash resonance tension
  110       pE   = pEH * prior + pEnH * (1.0 - prior)
  111   in clamp01 ((pEH * prior) / max 1e-9 pE)

  151 checkIntuition :: Double -> Double -> Int -> IntuitiveState -> (Maybe IntuitiveFlash, IntuitiveState)
  152 checkIntuition resonance tension turnNumber state =
  153   let newPost = updatePosterior resonance tension (isPosterior state)
  154       newLongPost = updateLongPosterior resonance tension (isLongPosterior state)
  155       newCooldown = max 0 (isCooldown state - 1)
  156       state' = state
  157         { isPosterior = newPost
  158         , isLongPosterior = newLongPost
  159         , isCooldown = newCooldown
  160         }
  161       currentPosterior = effectivePosterior state'
  162   in if newCooldown > 0
  163      then (Nothing, state')
  164      else if currentPosterior >= flashThreshold
  165            then let flash = buildFlash resonance tension currentPosterior
  166                     state'' = state'
  167                       { isPosterior  = posteriorAfterFlash newPost
  168                       , isLongPosterior = longPosteriorAfterFlash newLongPost
  169                       , isCooldown   = intuitionCooldownTurns
  170                       , isFlashCount = isFlashCount state + 1
  171                       , isLastTurn   = turnNumber
  172                       }
  173                in (Just flash, state'')
  174           else (Nothing, state')

  176 posteriorAfterFlash :: Double -> Double
  177 posteriorAfterFlash posterior =
  178   max basePrior (posterior * intuitionPosteriorAfterFlashDecayFactor)
  179 
  180 longPosteriorAfterFlash :: Double -> Double
  181 longPosteriorAfterFlash posterior =
  182   max basePrior (posterior * intuitionLongPosteriorAfterFlashDecayFactor)
  183 
  184 updateLongPosterior :: Double -> Double -> Double -> Double
  185 updateLongPosterior resonance tension prior =
  186   clamp01
  187     ( prior * intuitionLongPosteriorPriorWeight
  188       + updatePosterior resonance tension prior * intuitionLongPosteriorCurrentWeight
  189     )

  191 buildFlash :: Double -> Double -> Double -> IntuitiveFlash
  192 buildFlash resonance tension posterior =
  193   let trigger = selectTrigger resonance tension
  194       strength = clamp01 ((posterior - flashThreshold) / (1.0 - flashThreshold))
  195       kernelSig = buildKernelSignal trigger
  196       directive = buildDirective trigger strength
  197   in IntuitiveFlash strength trigger kernelSig directive (strength > intuitionFlashOverrideStrengthThreshold)
  198 
  199 selectTrigger :: Double -> Double -> FlashTrigger
  200 selectTrigger resonance tension
  201   | resonance > intuitionHighResonanceThreshold && tension > intuitionHighTensionThreshold = ConvergenceTrigger
  202   | tension > intuitionCrisisTensionThreshold = CrisisMomentTrigger
  203   | resonance > intuitionHighResonanceThreshold = DeepResonanceTrigger
  204   | otherwise                         = PureKernelTrigger

  219 buildDirective :: FlashTrigger -> Double -> Text
  220 buildDirective trigger strength =
  221   let header = intuitionHeaderPrefix <> fmtPctText strength <> "]"
  222       trig   = intuitionTriggerLabel <> renderTrigger trigger
  223       ovrd   = if strength > intuitionFlashOverrideStrengthThreshold
  224                then intuitionOverrideDirective
  225                else intuitionSupplementaryDirective
  226   in T.unlines [header, trig, ovrd, intuitionFormDirective]

  228 fmtPctText :: Double -> Text
  229 fmtPctText value = T.pack (printf "%.0f%%" (value * 100.0) :: String)

  231 intuitionSignalStrength :: Double
  232 intuitionSignalStrength =
  233   let v = qxfx0CoreVec
  234       base = cvPresence v * cvDepth v * (1.0 - cvDirectiveness v) * cvAutonomy v
  235       steadyBonus = 1.0 + (cvSteadiness v - intuitionSteadinessBaseline) * intuitionSignalSteadyBonusScale
  236   in base * steadyBonus

  238 -- | Multiplier applied to flash strength under a 'Salience' verdict.
  239 --
  240 -- Capped at @1.0@ so neutral or holistic-leaning salience never
  241 -- /boosts/ flashes above their base strength; only formal-leaning
  242 -- salience attenuates. At @salienceHolisticBias = 0.5@ (the
  243 -- neutral / dead-band point) the multiplier is exactly @1.0@,
  244 -- so wiring the salience controller into a call site that does
  245 -- not yet supply rich Field signals is a strict no-op.
  246 salienceFlashMultiplier :: Salience -> Double
  247 salienceFlashMultiplier salience = min 1.0 (2.0 * salienceHolisticBias salience)
  248 
  249 -- | Re-shape an 'IntuitiveFlash' under a salience verdict. The
  250 -- flash strength is scaled by 'salienceFlashMultiplier'; the
  251 -- override flag and directive text are recomputed so the trace
  252 -- stays internally consistent.
  253 applySalienceToFlash :: Salience -> IntuitiveFlash -> IntuitiveFlash
  254 applySalienceToFlash salience flash =
  255   let multiplier   = salienceFlashMultiplier salience
  256       newStrength  = clamp01 (ifStrength flash * multiplier)
  257       newOverride  = newStrength > intuitionFlashOverrideStrengthThreshold
  258       newDirective = buildDirective (ifTrigger flash) newStrength
  259   in flash
  260        { ifStrength     = newStrength
  261        , ifDirective    = newDirective
  262        , ifOverridesAll = newOverride
  263        }

  265 -- | Salience-aware variant of 'checkIntuition'. The flash-detection
  266 -- decision is unchanged; the produced flash, if any, is
  267 -- post-processed via 'applySalienceToFlash'.
  268 checkIntuitionWithSalience
  269   :: Salience
  270   -> Double -> Double -> Int -> IntuitiveState
  271   -> (Maybe IntuitiveFlash, IntuitiveState)
  272 checkIntuitionWithSalience salience resonance tension turnNumber state =
  273   let (mFlash, state') = checkIntuition resonance tension turnNumber state
  274   in (fmap (applySalienceToFlash salience) mFlash, state')
