    9 Description : Phase 8 — End-to-end learning loop: transport -> validator -> parser -> sandbox -> graft -> telemetry.
   10 
   11 Pure function that takes the current 'SystemState', an external query
   12 result, and returns the updated state plus telemetry.  This is the
   13 vertical-slice integration point: the runtime executes the
   14 'TurnReqExternalQuery' effect elsewhere and feeds the result into
   15 this function at finalize time.
   16 
   17 The loop is fail-closed:
   18 - transport error   -> record telemetry, update tool reliability down
   19 - parse failure    -> record telemetry, reliability down
   20 - validation fail  -> quarantine fruit (not graft), reliability down
   21 - sandbox reject   -> record telemetry, reliability down
   22 - sandbox accept   -> graft fruit, update lexicon/morph if provided,
   23                       reliability up
   24 -}

  110 -- | Run one learning-step iteration.
  111 --
  112 -- Arguments:
  113 --   * current system state
  114 --   * selected tool (for reliability tracking)
  115 --   * active learning need (for strategy selection)
  116 --   * user query text (for telemetry)
  117 --   * maybe external query result (Nothing = no external call attempted)
  118 --
  119 -- Returns: (updated state, telemetry)
  120 runLearningStep
  121   :: SystemState
  122   -> ExternalTool
  123   -> LearningNeed
  124   -> Text
  125   -> Maybe (Either ExternalQueryError ExternalQueryResponse)
  126   -> (SystemState, LearningTelemetry)
  127 runLearningStep ss tool need query mResult =
  128   let turn = ssTurnCount ss
  129       tree0 = ssKnowledgeTree ss
  130       rel0  = ssToolReliability ss
  131       morph = ssMorphology ss
  132       needState = ssLearningNeedState ss
  133 
  134       baseTelemetry = emptyLearningTelemetry
  135         { ltQueryType    = Just (renderNeedTag need)
  136         , ltExternalTool = Just (etName tool)
  137         }
  138 
  139   in case mResult of
  140     Nothing ->
  141       -- No external call attempted.
  142       (ss, baseTelemetry)
  143 
  144     Just (Left err) ->
  145       -- Transport / API failure: fail-closed, penalise tool.
  146       let rel1 = updateToolReliability (etName tool) False rel0
  147           records =
  148             [ toolReliabilityMutation turn (etName tool) False (renderQueryError err)
  149             ]
  150           tel  = baseTelemetry
  151             { ltValidationStatus = case err of
  152                 EqeFallback _ -> "fallback_non_authoritative"
  153                 _ -> "transport_error"
  154             , ltRejectReason     = Just (renderQueryError err)
  155             }
  156       in (appendAdaptiveMutationRecords records (ss { ssToolReliability = rel1 }), tel)
  157 
  158     Just (Right resp) ->
  159       -- Got a response: parse -> validate -> sandbox -> graft.
  160       let preProxy = conatusProxyFromState ss
  161       in case parseLLMResponseToFruit resp of
  162         Nothing ->
  163           let rel1 = updateToolReliability (etName tool) False rel0
  164               records =
  165                 [ toolReliabilityMutation turn (etName tool) False "parser_rejected_schema_or_text"
  166                 ]
  167               tel  = baseTelemetry
  168                 { ltValidationStatus = "invalid_response"
  169                 , ltRejectReason     = Just "parser_rejected_schema_or_text"
  170                 }
  171           in (appendAdaptiveMutationRecords records (ss { ssToolReliability = rel1 }), tel)
  172 
  173         Just payload ->
  174           case validateFruitPayload payload morph of
  175             Left valErr ->
  176               let rel1 = updateToolReliability (etName tool) False rel0
  177                   ss1  = ss { ssKnowledgeTree = tree0, ssToolReliability = rel1 }
  178                   -- ss1 does not yet contain the quarantined fruit; add 0.01 proxy for tree growth
  179                   postProxy = conatusProxyFromState ss1 + 0.01
  180                   fruit = payloadToFruit payload turn False (postProxy - preProxy)
  181                   tree1 = quarantineFruit fruit tree0
  182                   ss2   = ss1 { ssKnowledgeTree = tree1 }
  183                   reason = renderValidationError valErr
  184                   records =
  185                     [ toolReliabilityMutation turn (etName tool) False reason
  186                     , knowledgeTreeMutation turn MutKnowledgeTree AdaptiveQuarantined EvidenceStrong "external_learning:validation_reject" reason
  187                     ]
  188                   tel   = baseTelemetry
  189                     { ltValidationStatus = "validation_reject"
  190                     , ltRejectReason     = Just reason
  191                     }
  192               in (appendAdaptiveMutationRecords records ss2, tel)
  193 
  194             Right validatedPayload ->
  195               case runSandboxGate ss validatedPayload of
  196                 SandboxReject metrics reason ->
  197                   let rel1 = updateToolReliability (etName tool) False rel0
  198                       ss1  = ss { ssKnowledgeTree = tree0, ssToolReliability = rel1 }
  199                       postProxy = conatusProxyFromState ss1
  200                       fruit = payloadToFruit validatedPayload turn True (postProxy - preProxy)
  201                       tree1 = quarantineFruit fruit tree0
  202                       ss2   = ss1 { ssKnowledgeTree = tree1 }
  203                       reasonText = renderSandboxRejectReason reason
  204                       records =
  205                         [ toolReliabilityMutation turn (etName tool) False reasonText
  206                         , knowledgeTreeMutation turn MutKnowledgeTree AdaptiveQuarantined EvidenceStrong "external_learning:sandbox_reject" reasonText
  207                         ]
  208                       tel   = baseTelemetry
  209                         { ltValidationStatus = "sandbox_reject"
  210                         , ltSandboxResult  = Just (renderSandboxMetrics metrics)
  211                         , ltRejectReason   = Just reasonText
  212                         }
  213                   in (appendAdaptiveMutationRecords records ss2, tel)
  214 
  215                 SandboxAccept metrics ->
  216                   let rel1 = updateToolReliability (etName tool) True rel0
  217                       morph1 = mergeMorphologyPayload morph validatedPayload
  218                       ss1  = ss { ssKnowledgeTree = tree0, ssToolReliability = rel1, ssMorphology = morph1 }
  219                       postProxy = conatusProxyFromState ss1 + 0.01
  220                       fruit = payloadToFruit validatedPayload turn True (postProxy - preProxy)
  221                       tree1 = graftFruit (renderNeedTag need) fruit tree0
  222                       ss2   = ss1 { ssKnowledgeTree = tree1 }
  223                       records =
  224                         [ toolReliabilityMutation turn (etName tool) True "sandbox_accept"
  225                         , knowledgeTreeMutation turn MutKnowledgeTree AdaptiveAccepted EvidenceStrong "external_learning:graft" (renderNeedTag need)
  226                         ]
  227                       tel   = baseTelemetry
  228                         { ltValidationStatus = "accept"
  229                         , ltSandboxResult    = Just (renderSandboxMetrics metrics)
  230                         , ltGraftTurn        = Just turn
  231                         }
  232                   in (appendAdaptiveMutationRecords records ss2, tel)

  234 -- | Compute a proxy Conatus scalar from the current SystemState.
  235 -- Includes the actual ConatusEnergy from SelfBlanket plus a small
  236 -- knowledge-tree-size term so that graft/quarantine changes are visible.
  237 conatusProxyFromState :: SystemState -> Double
  238 conatusProxyFromState s =
  239   let blanket = computeSelfBlanket s
  240       violations = checkInitialBlanket blanket
  241       energy = ceScalar (computeConatusEnergy blanket violations)
  242       tree = ssKnowledgeTree s
  243       treeSize = sum (map length (M.elems (ktBranches tree))) + length (ktQuarantine tree)
  244   in energy + 0.01 * fromIntegral treeSize

  246 -- | Convert a validated payload into a 'KnowledgeFruit'.
  247 -- The conatus delta is derived from actual state change (pre vs post)
  248 -- rather than trusting the external payload.
  249 payloadToFruit :: KnowledgeFruitPayload -> Int -> Bool -> Double -> KnowledgeFruit
  250 payloadToFruit payload turn validated conatusDelta =
  251   KnowledgeFruit
  252     { kfProposition     = kfpProposition payload
  253     , kfWord            = kfpWord payload
  254     , kfSource          = kfpSource payload
  255     , kfValidated       = validated
  256     , kfConatusDelta    = conatusDelta
  257     , kfPredictiveDelta = kfpPredictiveDelta payload
  258     , kfGraftedTurn     = if validated then Just turn else Nothing
  259     , kfObservedTurn    = turn
  260     }

  262 -- | Merge morphology payload into runtime MorphologyData.
  263 -- This is intentionally shallow: only adds surface forms to
  264 -- prepositional/genitive/nominative maps, never overwrites.
  265 -- 'mdFormsBySurface' is left untouched because 'LexemeForm' construction
  266 -- requires case/number/tier metadata that the LLM payload does not
  267 -- yet provide.
  268 mergeMorphologyPayload :: MorphologyData -> KnowledgeFruitPayload -> MorphologyData
  269 mergeMorphologyPayload morph payload =
  270   case kfpMorphology payload of
  271     Nothing -> morph
  272     Just mp ->
  273       case mpCases mp of
  274         Nothing -> morph
  275         Just cMap ->
  276           let normalizedWord = T.toLower (T.strip (kfpWord payload))
  277               normalizedCases = M.mapKeys T.toLower cMap
  278               applyCase key selector currentMap =
  279                 case M.lookup key normalizedCases of
  280                   Just surfaceValue | not (T.null (T.strip surfaceValue)) -> M.insert normalizedWord (T.toLower (T.strip surfaceValue)) currentMap
  281                   _ -> currentMap
  282               newSurfaceForms = learnedSurfaceForms normalizedWord normalizedCases
  283               mergedSurfaceForms = M.unionWith (++) newSurfaceForms (mdFormsBySurface morph)
  284           in morph
  285                { mdNominative     = applyCase "nom" id (mdNominative morph)
  286                , mdGenitive       = applyCase "gen" id (mdGenitive morph)
  287                , mdPrepositional  = applyCase "prep" id (mdPrepositional morph)
  288                 , mdFormsBySurface = mergedSurfaceForms
  289                 }
  290 
  291 learnedSurfaceForms :: Text -> M.Map Text Text -> M.Map Text [LexemeForm]
  292 learnedSurfaceForms lemma casesMap =
  293   M.fromListWith (++)
  294     [ (surfaceValue, [mkForm lexCase surfaceValue])
  295     | (caseKey, surfaceValueRaw) <- M.toList casesMap
  296     , let surfaceValue = T.toLower (T.strip surfaceValueRaw)
  297     , not (T.null surfaceValue)
  298     , Just lexCase <- [learnedLexemeCase caseKey]
  299     ]
  300   where
  301     mkForm lexCase surfaceValue =
  302       LexemeForm
  303         { lfSurface = surfaceValue
  304         , lfLemma = lemma
  305         , lfPOS = "noun"
  306         , lfCase = lexCase
  307         , lfNumber = SingularNumber
  308         , lfTier = AutoCoverageTier
  309         , lfQuality = 0.55
  310         }
  311 
  312 learnedLexemeCase :: Text -> Maybe LexemeCase
  313 learnedLexemeCase rawCase =
  314   case T.toLower (T.strip rawCase) of
  315     "nom" -> Just NominativeCase
  316     "nominative" -> Just NominativeCase
  317     "gen" -> Just GenitiveCase
  318     "genitive" -> Just GenitiveCase
  319     "dat" -> Just DativeCase
  320     "dative" -> Just DativeCase
  321     "acc" -> Just AccusativeCase
  322     "accusative" -> Just AccusativeCase
  323     "ins" -> Just InstrumentalCase
  324     "instrumental" -> Just InstrumentalCase
  325     "prep" -> Just PrepositionalCase
  326     "prepositional" -> Just PrepositionalCase
  327     _ -> Nothing

  335 renderQueryError :: ExternalQueryError -> Text
  336 renderQueryError (EqeNetworkUnavailable t) = T.concat ["network:", t]
  337 renderQueryError (EqeAuthFailure t)        = T.concat ["auth:", t]
  338 renderQueryError (EqeRateLimited t)        = T.concat ["rate_limit:", t]
  339 renderQueryError (EqeServerError t)        = T.concat ["server:", t]
  340 renderQueryError (EqeTimeout t)            = T.concat ["timeout:", t]
  341 renderQueryError (EqeInvalidResponse t)    = T.concat ["invalid:", t]
  342 renderQueryError (EqeFallback reason)      = T.concat ["fallback:", renderFallbackReason reason]
  343 renderQueryError EqeEmptyResponse          = "empty"

  371 toolReliabilityMutation :: Int -> Text -> Bool -> Text -> AdaptiveMutationRecord
  372 toolReliabilityMutation turn toolName accepted reason = AdaptiveMutationRecord
  373   { amrTurnId = turn
  374   , amrKind = MutToolReliability
  375   , amrCause = if accepted then "tool_reliability:accepted" else "tool_reliability:rejected"
  376   , amrEvidence = ["tool=" <> toolName, "reason=" <> reason]
  377   , amrEvidenceStrength = EvidenceStrong
  378   , amrConfidence = 0.80
  379   , amrBoundedDelta = Just (if accepted then 0.05 else 0.10)
  380   , amrDecision = if accepted then AdaptiveAccepted else AdaptiveRejected
  381   }

  402 -- | Convenience wrapper: apply the learning loop to a system state
  403 -- when an external query result is present.
  404 -- If no result was carried, returns the state unchanged.
  405 applyExternalLearning :: SystemState -> Maybe (Either ExternalQueryError ExternalQueryResponse) -> SystemState
  406 applyExternalLearning ss mResult =
  407   case mResult of
  408     Nothing -> ss
  409     Just result ->
  410       let needState = ssLearningNeedState ss
  411           need = lnsCurrentNeed needState
  412           mTool = selectToolWithReliability need (ssToolReliability ss) defaultAvailableTools
  413           tool = fromMaybe (ExternalTool "unknown" DomainGeneral 0.5 False) mTool
  414           turn = ssTurnCount ss
  415           proposalId = CalibrationId turn
  416           submittedGuard = recordProposalSubmission (ssGuardrailState ss) turn proposalId
  417           ssSubmitted = ss { ssGuardrailState = submittedGuard }
  418           queryText = case result of
  419             Right resp -> eqrToolName resp <> " " <> T.pack (show (eqrLatencyMs resp))
  420             Left _     -> ""
  421           (updated, telemetry) = runLearningStep ssSubmitted tool need queryText (Just result)
  422           finalizedGuard =
  423             case ltValidationStatus telemetry of
  424               "accept" -> recordAcceptance (ssGuardrailState updated)
  425               "not_attempted" -> ssGuardrailState updated
  426               _ -> recordRejection (ssGuardrailState updated) turn
  427       in updated { ssGuardrailState = finalizedGuard }
