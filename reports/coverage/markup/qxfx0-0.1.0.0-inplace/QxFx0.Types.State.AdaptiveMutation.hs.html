   83 renderEvidenceStrength :: EvidenceStrength -> Text
   84 renderEvidenceStrength strength =
   85   case strength of
   86     EvidenceObservational -> "EvidenceObservational"
   87     EvidenceWeak -> "EvidenceWeak"
   88     EvidenceModerate -> "EvidenceModerate"
   89     EvidenceStrong -> "EvidenceStrong"
   90 
   91 parseEvidenceStrengthText :: Text -> Parser EvidenceStrength
   92 parseEvidenceStrengthText text =
   93   case text of
   94     "EvidenceObservational" -> pure EvidenceObservational
   95     "EvidenceWeak" -> pure EvidenceWeak
   96     "EvidenceModerate" -> pure EvidenceModerate
   97     "EvidenceStrong" -> pure EvidenceStrong
   98     "AdaptiveEvidenceWeak" -> pure EvidenceWeak
   99     "AdaptiveEvidenceModerate" -> pure EvidenceModerate
  100     "AdaptiveEvidenceStrong" -> pure EvidenceStrong
  101     _ -> failUnknown "EvidenceStrength" text
