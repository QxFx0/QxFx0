  180 prepareTurn :: PipelineIO -> SystemState -> Text -> Text -> Text -> IO PreparedTurn
  181 prepareTurn pio ss input sessionId requestId = do
  182   -- Phase C: resolve time up-front so buildPrepareEffectPlan is deterministic
  183   currentTime <- resolvePrepareCurrentTime pio
  184   let prepareEffects = buildPrepareEffectPlan ss input currentTime
  185   prepareResults <- Prepare.resolvePrepareEffects pio prepareEffects
  186   let ti' = Prepare.buildTurnInput ss requestId sessionId prepareEffects prepareResults
  187       ts = Prepare.buildTurnSignals prepareResults
  188   pure (PreparedTurn ti' ts)
  189 
  190 -- | Resolve current time through the runtime effect boundary.
  191 -- Phase C abstraction: uses the same 'TimeSource' that backs
  192 -- 'TurnReqCurrentTime' so tests can override it via env var.
  193 resolvePrepareCurrentTime :: PipelineIO -> IO UTCTime
  194 resolvePrepareCurrentTime pio = do
  195   result <- resolveTurnEffect pio TurnReqCurrentTime
  196   case result of
  197     TurnResCurrentTime t -> pure t
  198     _ -> throwQxFx0 (PersistenceError (T.pack "prepare current time effect returned unexpected result"))

  200 planTurn :: PipelineIO -> SystemState -> PreparedTurn -> IO PlannedTurn
  201 planTurn pio ss (PreparedTurn ti ts) = do
  202   let routeEffects = Route.planRouteEffects ss ti ts
  203   routeResults <- Route.resolveRouteEffects pio routeEffects
  204   let tp = Route.buildRouteTurnPlan (pipelineShadowPolicy pio) ss ti ts routeEffects routeResults
  205   pure (PlannedTurn ti ts tp)
  206 
  207 renderTurn :: PipelineIO -> SystemState -> PlannedTurn -> IO RenderedTurn
  208 renderTurn pio ss (PlannedTurn ti ts tp) = do
  209   let renderEffects = Route.planRenderEffectsForRuntime (pipelineRuntimeMode pio) (pipelineLocalRecoveryPolicy pio) ss ti ts tp
  210   renderResults <- Route.resolveRenderEffects pio renderEffects
  211   let ta = Route.buildTurnArtifacts ss ti ts tp renderEffects renderResults
  212   pure (RenderedTurn ti ts tp ta)

  214 finalizeTurn :: PipelineIO -> SystemState -> Text -> Text -> RenderedTurn -> IO TurnResult
  215 finalizeTurn pio ss sessionId _requestId (RenderedTurn ti ts tp ta) = do
  216   let precommitPlan = Finalize.planFinalizePrecommit ss ti ts tp ta
  217   precommitResults <- Finalize.resolveFinalizePrecommit pio precommitPlan
  218   let precommitBundle = Finalize.buildFinalizePrecommit (pipelineUpdateHistory pio) ss ti ts tp ta precommitPlan precommitResults
  219       commitPlan = Finalize.planFinalizeCommit sessionId ss ti ts ta precommitBundle
  220   commitResults <- Finalize.resolveFinalizeCommit pio commitPlan
  221   let turnResult = Finalize.buildFinalizeTurnResult (RenderedTurn ti ts tp ta) precommitBundle commitResults
  222   Finalize.resolveFinalizePostCommit (trMetrics turnResult)
  223   pure turnResult
