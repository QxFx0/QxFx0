   40 computeRoutingPhase :: CanonicalMoveFamily -> InputPropositionFrame -> AtomSet -> UserState
   41                     -> SystemState -> [Text] -> Text -> RoutingPhase
   42 computeRoutingPhase recommendedFamily frame atomSet nextUserState systemState history input =
   43   let propositionFamily = ipfCanonicalFamily frame
   44       familyByParserConfidence =
   45         if ipfConfidence frame >= parserHighConfidenceThreshold
   46           then propositionFamily
   47           else recommendedFamily
   48       semanticInput0 =
   49         buildSemanticInputSimple input atomSet frame recommendedFamily (asRegister atomSet) (usNeedLayer nextUserState)
   50       semanticFamily = semanticInputFamilyHint semanticInput0
   51       familyMerged = mergeFamilySignalsWithFrame recommendedFamily frame familyByParserConfidence semanticFamily
   52       parserLocked =
   53         ipfConfidence frame >= parserHighConfidenceThreshold
   54           && ipfPropositionType frame /= T.pack "PlainAssert"
   55       pressure = detectPressure input history
   56       principledModeResult = fmap principledMode pressure
   57       pressureBand = pressureBandFromState (classifyPressure pressure history)
   58       fromMeaningState = MeaningState ResonanceMed pressureBand (toDepthBand (ssTurnCount systemState))
   59       toMeaningState = MeaningState ResonanceMed pressureBand (toDepthBand (ssTurnCount systemState + 1))
   60       predictedStrategy = predictStrategy fromMeaningState toMeaningState (ssMeaningGraph systemState)
   61       chosenStrategy = fromMaybe (defaultStrategy fromMeaningState) predictedStrategy
   62       strategyFamily = strategyFamilyHint chosenStrategy
   63       familyAfterStrategy
   64         | parserLocked = familyMerged
   65         | otherwise = maybe familyMerged (`preferFamily` familyMerged) strategyFamily
   66       preEgo = updateEgoFromTurn (ssEgo systemState) familyAfterStrategy (computeTensionDelta input systemState)
   67       prevDirective = steerDirectiveWithOrbitalSimple (ssOrbitalMemory systemState) defaultCoreDirective
   68       collapseRisk = egoTension preEgo
   69       freezeRisk = 1.0 - egoAgency preEgo
   70       orbitalPhase = classifyOrbitalPhaseSimple collapseRisk freezeRisk
   71       encounterMode = classifyEncounterModeSimple orbitalPhase (egoTension preEgo)
   72       updatedOrbital = updateOrbitalMemorySimple (ssOrbitalMemory systemState) orbitalPhase encounterMode prevDirective
   73       identitySignal0 =
   74         buildIdentitySignalSimple orbitalPhase encounterMode prevDirective
   75           (asRegister atomSet) (usNeedLayer nextUserState) familyAfterStrategy (forceForFamily familyAfterStrategy)
   76    in RoutingPhase
   77         { rpFamilyMerged = familyMerged
   78         , rpMPressure = pressure
   79         , rpPrincipledModeResult = principledModeResult
   80         , rpPressureBand = pressureBand
   81         , rpFromMs = fromMeaningState
   82         , rpToMs = toMeaningState
   83         , rpChosenStrategy = chosenStrategy
   84         , rpStrategyFamily = strategyFamily
   85         , rpFamilyAfterStrategy = familyAfterStrategy
   86         , rpPreEgo = preEgo
   87         , rpPrevDirective = prevDirective
   88         , rpOrbitalPhase = orbitalPhase
   89         , rpEncounterMode = encounterMode
   90         , rpUpdatedOrbital = updatedOrbital
   91         , rpIdentitySignal0 = identitySignal0
   92         }

   94 mergeFamilySignals :: CanonicalMoveFamily -> CanonicalMoveFamily -> CanonicalMoveFamily -> CanonicalMoveFamily
   95 mergeFamilySignals recommended parser semantic
   96   | parser /= recommended && parser /= CMGround = parser
   97   | semantic /= recommended && semantic /= CMGround = semantic
   98   | otherwise = recommended

  100 mergeFamilySignalsWithFrame :: CanonicalMoveFamily -> InputPropositionFrame -> CanonicalMoveFamily -> CanonicalMoveFamily -> CanonicalMoveFamily
  101 mergeFamilySignalsWithFrame recommended frame parser semantic
  102   | ipfConfidence frame >= parserHighConfidenceThreshold
  103       && ipfPropositionType frame /= T.pack "PlainAssert"
  104       && parser /= recommended = parser
  105   | otherwise = mergeFamilySignals recommended parser semantic

  110 semanticInputFamilyHint :: SemanticInput -> CanonicalMoveFamily
  111 semanticInputFamilyHint semanticInput =
  112   let atoms = asAtoms (siAtomSet semanticInput)
  113       frame = siPropositionFrame semanticInput
  114       recommended = siRecommendedFamily semanticInput
  115       rawLower = T.toLower (ipfRawText frame)
  116       hasConfrontLexeme = T.pack "противореч" `T.isInfixOf` rawLower
  117       hasAtom p = any (\a -> p (maTag a)) atoms
  118       isContact (NeedContact _) = True; isContact _ = False
  119       isRepair (Exhaustion _) = True; isRepair (AgencyLost _) = True; isRepair _ = False
  120       isConfront (Contradiction _ _) = True; isConfront _ = False
  121       isDeepen (NeedMeaning _) = True; isDeepen (Searching _) = True; isDeepen _ = False
  122       isClarify (Verification _) = True; isClarify (Doubt _) = True; isClarify _ = False
  123       isAnchor (Anchoring _) = True; isAnchor (AgencyFound _) = True; isAnchor _ = False
  124       keepRecommendedOnClarify =
  125         not (ipfIsQuestion frame)
  126           && recommended `elem` [CMGround, CMDefine, CMDistinguish, CMDescribe, CMPurpose, CMContact, CMDeepen]
  127           && ipfPropositionType frame `notElem` map T.pack ["ClarifyQ", "EpistemicQ", "RequestQ"]
  128       keepRecommendedOnAnchorQuestion =
  129         recommended `elem` [CMGround, CMDefine, CMDistinguish, CMDescribe, CMPurpose]
  130           && ipfPropositionType frame `elem` map T.pack
  131             [ "SelfStateQ"
  132             , "SelfKnowledgeQ"
  133             , "SystemLogicQ"
  134             , "WorldCauseQ"
  135             , "OperationalCauseQ"
  136             , "ConceptKnowledgeQ"
  137             , "PurposeQ"
  138             , "DistinctionQ"
  139             , "ComparisonPlausibilityQ"
  140             ]
  141   in if hasConfrontLexeme then CMConfront
  142      else if hasAtom isContact then CMContact
  143      else if hasAtom isRepair then CMRepair
  144      else if hasAtom isConfront then CMConfront
  145      else if hasAtom isDeepen then CMDeepen
  146      else if hasAtom isClarify
  147        then if keepRecommendedOnClarify then recommended else CMClarify
  148      else if hasAtom isAnchor
  149        then if keepRecommendedOnAnchorQuestion then recommended else CMAnchor
  150      else case (siNeedLayer semanticInput, siRegister semanticInput) of
  151        (ContactLayer, _) -> CMContact
  152        (_, Contact) -> CMContact
  153        (_, Exhaust) -> CMRepair
  154        (_, Anchor) -> CMAnchor
  155        (_, Search) -> CMDeepen
  156        _ -> recommended

  158 strategyFamilyHint :: ResponseStrategy -> Maybe CanonicalMoveFamily
  159 strategyFamilyHint strategy =
  160   case rsMove strategy of
  161     CounterMove -> Just CMConfront
  162     ReframeMove -> Just CMClarify
  163     QuestionMove -> Just CMDeepen
  164     ValidateMove -> Just CMContact
  165     SilenceMove -> Just CMAnchor
  166 
  167 identityFamilyHint :: IdentitySignal -> Maybe CanonicalMoveFamily
  168 identityFamilyHint signal =
  169   case isEncounterMode signal of
  170     EncounterRecovery -> Just CMContact
  171     EncounterHolding -> Just CMAnchor
  172     EncounterPressure -> Just CMRepair
  173     EncounterCounterweight -> Just CMClarify
  174     EncounterMirroring -> Just CMReflect
  175     EncounterExploration -> Nothing
