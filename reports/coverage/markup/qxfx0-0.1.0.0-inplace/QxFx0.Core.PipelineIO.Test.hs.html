   74 mkTestPipelineIO :: TestPipelineConfig -> PipelineIO
   75 mkTestPipelineIO cfg =
   76   let consciousState = unsafePerformIO (newMVar initialLoop)
   77       intuitionState = unsafePerformIO (newMVar defaultIntuitiveState)
   78   in PipelineIO
   79       { pioRuntimeMode = tpcRuntimeMode cfg
   80       , pioShadowPolicy = tpcShadowPolicy cfg
   81       , pioLocalRecoveryPolicy = tpcLocalRecoveryPolicy cfg
   82       , pioInterpreter = tpcInterpreter cfg
   83       , pioUpdateHistory = tpcUpdateHistory cfg
   84       , pioModifyConsciousLoop = \f -> modifyMVar consciousState $ \st -> do
   85           (st', result) <- f st
   86           pure (st', result)
   87       , pioModifyIntuition = \f -> modifyMVar intuitionState $ \st -> do
   88           (st', result) <- f st
   89           pure (st', result)
   90       , pioConatusPrior = defaultConatusPrior
   91       }

   93 defaultTestInterpreter :: TurnEffectInterpreter
   94 defaultTestInterpreter request =
   95   case request of
   96     TurnReqEmbedding inputText ->
   97       TurnResEmbedding <$> textToEmbeddingResult inputText
   98     TurnReqNixGuard _ _ _ ->
   99       pure (TurnResNixGuard (Unavailable "nix_unavailable_default_test_pipeline"))
  100     TurnReqConsciousness semanticInput humanTheta resonance conatusEnergy salienceWeights -> do
  101       let salience = computeSalience salienceWeights conatusEnergy (emptyField { fieldResonance = mkResonance resonance })
  102           (nextLoop, fragment) =
  103             runConsciousnessLoopWithSalience salience initialLoop semanticInput humanTheta resonance
  104           nextNarrative = clLastNarrative nextLoop
  105           nextFragment = if T.null fragment then Nothing else Just fragment
  106       pure (TurnResConsciousness nextLoop nextNarrative nextFragment)
  107     TurnReqIntuition _inputText resonance tension turnNumber _conatusEnergy _salienceWeights _semanticConfig -> do
  108       let (mFlash, intuitive') = checkIntuition resonance tension turnNumber defaultIntuitiveState
  109       pure (TurnResIntuition mFlash (effectivePosterior intuitive') intuitive')
  110     TurnReqApiHealth ->
  111       pure (TurnResApiHealth False)
  112     TurnReqShadow _ _ _ ->
  113       pure
  114         (TurnResShadow
  115           Nothing
  116           ShadowUnavailable
  117           emptyShadowDivergence
  118           (ShadowSnapshotId "shadow:test_default")
  119           ["shadow_unavailable_default_test_pipeline"])
  120     TurnReqAgdaVerify ->
  121       pure (TurnResAgdaVerify AgdaMissingWitness)
  122     TurnReqCurrentTime ->
  123       pure (TurnResCurrentTime fixedTestTime)
  124     TurnReqRequestId ->
  125       pure (TurnResRequestId "request-id-test")
  126     TurnReqReadEnv _ ->
  127       pure (TurnResReadEnv Nothing)
  128     TurnReqTestMarkOnceFile _ ->
  129       pure (TurnResTestMarkOnceFile False)
  130     TurnReqSemanticIntrospectionEnv ->
  131       pure (TurnResSemanticIntrospectionEnv False)
  132     TurnReqCommitRuntimeState _ _ _ ->
  133       pure TurnResCommitRuntimeState
  134     TurnReqSaveState _ _ _ ->
  135       pure (TurnResSaveState (Left (PdSaveFailed StageUnknown Nothing (Just "persistence_unavailable_default_test_pipeline"))))
  136     TurnReqRollbackTurnProjections _ _ ->
  137       pure (TurnResRollbackTurnProjections (Left (PdRollbackFailed StageUnknown Nothing (Just "persistence_unavailable_default_test_pipeline"))))
  138     TurnReqCheckpoint _ ->
  139       pure TurnResCheckpointCompleted
  140     TurnReqLinearizeClaimAst _ _ _ ->
  141       pure (TurnResLinearizeClaimAst (Left "pgf_unavailable_default_test_pipeline"))
  142     TurnReqLinearizeDialogAtoms _ _ _ ->
  143       pure (TurnResLinearizeDialogAtoms (Left "pgf_unavailable_default_test_pipeline"))
  144     TurnReqExternalQuery _ _ _ ->
  145       pure (TurnResExternalQuery (Left (EqeInvalidResponse "external_query_unavailable_default_test_pipeline")))
  146
