    1 {-# LANGUAGE DerivingStrategies #-}
    2 {-# LANGUAGE OverloadedStrings #-}
    3 {-| Per-session critical section registry for serializing concurrent turn execution. -}
    4 module QxFx0.Core.SessionLock
    5   ( SessionLockManager
    6   , SessionLockStats(..)
    7   , newSessionLockManager
    8   , withSessionLock
    9   , sessionLockStats
   10   , sessionLockOverflowShardIndex
   11   , sessionLockOverflowShardCount
   12   ) where
   13 
   14 import Control.Concurrent.MVar (MVar, newMVar, withMVar, tryReadMVar)
   15 import Control.Concurrent.STM (TVar, atomically, newTVarIO, readTVar, modifyTVar')
   16 import Control.Monad (replicateM)
   17 import Data.Map.Strict (Map)
   18 import qualified Data.Map.Strict as Map
   19 import Data.List (sortOn)
   20 import Data.Ord (comparing)
   21 import Data.Text (Text)
   22 import qualified Data.Text as T
   23 import System.Environment (lookupEnv)

   26 data SessionLockEntry = SessionLockEntry
   27   { sleLock :: !(MVar ())
   28   , sleLastUsed :: !Int
   29   }
   30 
   31 data SessionLockManager = SessionLockManager
   32   { slmLocks :: !(TVar (Map Text SessionLockEntry))
   33   , slmOverflowShards :: ![MVar ()]
   34   , slmMaxTrackedLocks :: !Int
   35   , slmAccessCounter :: !(TVar Int)
   36   }
   37 
   38 data SessionLockStats = SessionLockStats
   39   { slsTrackedLocks :: !Int
   40   , slsMaxTrackedLocks :: !Int
   41   , slsOverflowActive :: !Bool
   42   } deriving stock (Eq, Show)
   43 
   44 newSessionLockManager :: IO SessionLockManager
   45 newSessionLockManager = do
   46   locks <- newTVarIO Map.empty
   47   overflowShards <- replicateM sessionLockOverflowShardCount (newMVar ())
   48   accessCounter <- newTVarIO 0
   49   cap <- resolveMaxTrackedLocks
   50   pure SessionLockManager
   51     { slmLocks = locks
   52     , slmOverflowShards = overflowShards
   53     , slmMaxTrackedLocks = cap
   54     , slmAccessCounter = accessCounter
   55     }
   56 
   57 resolveMaxTrackedLocks :: IO Int
   58 resolveMaxTrackedLocks = do
   59   mEnv <- lookupEnv "QXFX0_MAX_SESSION_LOCKS"
   60   case mEnv of
   61     Nothing -> pure defaultMaxTrackedLocks
   62     Just raw ->
   63       case reads raw of
   64         [(n, "")] | n > 0 -> pure n
   65         _ -> do
   66           hPutStrLn stderr $ "QXFX0_MAX_SESSION_LOCKS invalid (" <> raw <> "), using default " <> show defaultMaxTrackedLocks
   67           pure defaultMaxTrackedLocks
   68 
   69 defaultMaxTrackedLocks :: Int
   70 defaultMaxTrackedLocks = 4096
   71 
   72 withSessionLock :: SessionLockManager -> Text -> IO a -> IO a
   73 withSessionLock mgr sessionId action = do
   74   lock <- getOrCreateLock mgr sessionId
   75   withMVar lock (\_ -> action)
   76 
   77 sessionLockStats :: SessionLockManager -> IO SessionLockStats
   78 sessionLockStats mgr = atomically $ do
   79   locks <- readTVar (slmLocks mgr)
   80   let tracked = Map.size locks
   81       maxTracked = slmMaxTrackedLocks mgr
   82   pure
   83     SessionLockStats
   84       { slsTrackedLocks = tracked
   85       , slsMaxTrackedLocks = maxTracked
   86       , slsOverflowActive = tracked >= maxTracked
   87       }
   88 
   89 getOrCreateLock :: SessionLockManager -> Text -> IO (MVar ())
   90 getOrCreateLock mgr sessionId = do
   91   stamp <- nextAccessStamp mgr
   92   mExisting <- atomically $ do
   93     locks <- readTVar (slmLocks mgr)
   94     case Map.lookup sessionId locks of
   95       Just entry -> do
   96         modifyTVar' (slmLocks mgr) (Map.adjust (\entry' -> entry' { sleLastUsed = stamp }) sessionId)
   97         pure (Just (sleLock entry))
   98       Nothing -> pure Nothing
   99   case mExisting of
  100     Just lock -> pure lock
  101     Nothing -> do
  102       newLock <- newMVar ()
  103       result <- atomically $ do
  104         locks <- readTVar (slmLocks mgr)
  105         case Map.lookup sessionId locks of
  106           Just existing -> pure (Just (sleLock existing))
  107           Nothing ->
  108             if Map.size locks >= slmMaxTrackedLocks mgr
  109               then pure Nothing
  110               else do
  111                 modifyTVar' (slmLocks mgr) (Map.insert sessionId (SessionLockEntry newLock stamp))
  112                 pure (Just newLock)
  113       case result of
  114         Just lock -> pure lock
  115         Nothing -> do
  116           evicted <- evictFreeTrackedLock mgr sessionId newLock stamp
  117           case evicted of
  118             Just lock -> pure lock
  119             Nothing -> do
  120               hPutStrLn stderr $ "session_lock_overflow: session=" <> T.unpack sessionId <> " tracked=" <> show (slmMaxTrackedLocks mgr) <> " cap=" <> show (slmMaxTrackedLocks mgr)
  121               pure (pickOverflowShard mgr sessionId)
  122 
  123 nextAccessStamp :: SessionLockManager -> IO Int
  124 nextAccessStamp mgr = atomically $ do
  125   current <- readTVar (slmAccessCounter mgr)
  126   let next = current + 1
  127   modifyTVar' (slmAccessCounter mgr) (const next)
  128   pure next
  129 
  130 evictFreeTrackedLock :: SessionLockManager -> Text -> MVar () -> Int -> IO (Maybe (MVar ()))
  131 evictFreeTrackedLock mgr sessionId newLock stamp = do
  132   snapshot <- atomically $ readTVar (slmLocks mgr)
  133   let candidates = reverse (sortOn (sleLastUsed . snd) (Map.toList snapshot))
  134   tryCandidates candidates
  135   where
  136     tryCandidates [] = pure Nothing
  137     tryCandidates ((victimId, victimEntry) : rest) = do
  138       free <- tryReadMVar (sleLock victimEntry)
  139       case free of
  140         Nothing -> tryCandidates rest
  141         Just _ -> do
  142           result <- atomically $ do
  143             locks <- readTVar (slmLocks mgr)
  144             case (Map.lookup sessionId locks, Map.lookup victimId locks) of
  145               (Just existing, _) -> pure (Just (sleLock existing))
  146               (Nothing, Just currentVictim)
  147                 | sleLastUsed currentVictim == sleLastUsed victimEntry && Map.size locks >= slmMaxTrackedLocks mgr -> do
  148                     modifyTVar' (slmLocks mgr) (Map.insert sessionId (SessionLockEntry newLock stamp) . Map.delete victimId)
  149                     pure (Just newLock)
  150               _ -> pure Nothing
  151           case result of
  152             Just lock -> pure (Just lock)
  153             Nothing -> tryCandidates rest
  154 
  155 pickOverflowShard :: SessionLockManager -> Text -> MVar ()
  156 pickOverflowShard mgr sessionId =
  157   let shards = slmOverflowShards mgr
  158       idx = sessionLockOverflowShardIndex sessionId `mod` max 1 (length shards)
  159   in shards !! idx
  160 
  161 sessionLockOverflowShardCount :: Int
  162 sessionLockOverflowShardCount = 8
  163 
  164 sessionLockOverflowShardIndex :: Text -> Int
  165 sessionLockOverflowShardIndex = T.foldl' step 5381
  166   where
  167     step acc ch = acc * 33 + fromEnum ch
