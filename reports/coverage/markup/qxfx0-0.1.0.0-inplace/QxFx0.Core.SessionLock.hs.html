    1 {-# LANGUAGE DerivingStrategies #-}
    2 {-# LANGUAGE OverloadedStrings #-}
    3 {-| Per-session critical section registry for serializing concurrent turn execution. -}
    4 module QxFx0.Core.SessionLock
    5   ( SessionLockManager
    6   , SessionLockStats(..)
    7   , newSessionLockManager
    8   , withSessionLock
    9   , sessionLockStats
   10   ) where
   11 
   12 import Control.Concurrent.MVar (MVar, newMVar, withMVar)
   13 import Control.Concurrent.STM (TVar, atomically, newTVarIO, readTVar, modifyTVar')
   14 import Data.Map.Strict (Map)
   15 import qualified Data.Map.Strict as Map
   16 import Data.Text (Text)
   17 import qualified Data.Text as T
   18 import System.Environment (lookupEnv)
   19 import System.IO (hPutStrLn, stderr)
   20 
   21 data SessionLockManager = SessionLockManager
   22   { slmLocks :: !(TVar (Map Text (MVar ())))
   23   , slmOverflowLock :: !(MVar ())
   24   , slmMaxTrackedLocks :: !Int
   25   }
   26 
   27 data SessionLockStats = SessionLockStats
   28   { slsTrackedLocks :: !Int
   29   , slsMaxTrackedLocks :: !Int
   30   , slsOverflowActive :: !Bool
   31   } deriving stock (Eq, Show)
   32 
   33 newSessionLockManager :: IO SessionLockManager
   34 newSessionLockManager = do
   35   locks <- newTVarIO Map.empty
   36   overflowLock <- newMVar ()
   37   cap <- resolveMaxTrackedLocks
   38   pure SessionLockManager
   39     { slmLocks = locks
   40     , slmOverflowLock = overflowLock
   41     , slmMaxTrackedLocks = cap
   42     }
   43 
   44 resolveMaxTrackedLocks :: IO Int
   45 resolveMaxTrackedLocks = do
   46   mEnv <- lookupEnv "QXFX0_MAX_SESSION_LOCKS"
   47   case mEnv of
   48     Nothing -> pure defaultMaxTrackedLocks
   49     Just raw ->
   50       case reads raw of
   51         [(n, "")] | n > 0 -> pure n
   52         _ -> do
   53           hPutStrLn stderr $ "QXFX0_MAX_SESSION_LOCKS invalid (" <> raw <> "), using default " <> show defaultMaxTrackedLocks
   54           pure defaultMaxTrackedLocks
   55 
   56 defaultMaxTrackedLocks :: Int
   57 defaultMaxTrackedLocks = 4096
   58 
   59 withSessionLock :: SessionLockManager -> Text -> IO a -> IO a
   60 withSessionLock mgr sessionId action = do
   61   lock <- getOrCreateLock mgr sessionId
   62   withMVar lock (\_ -> action)
   63 
   64 sessionLockStats :: SessionLockManager -> IO SessionLockStats
   65 sessionLockStats mgr = atomically $ do
   66   locks <- readTVar (slmLocks mgr)
   67   let tracked = Map.size locks
   68       maxTracked = slmMaxTrackedLocks mgr
   69   pure
   70     SessionLockStats
   71       { slsTrackedLocks = tracked
   72       , slsMaxTrackedLocks = maxTracked
   73       , slsOverflowActive = tracked >= maxTracked
   74       }
   75 
   76 getOrCreateLock :: SessionLockManager -> Text -> IO (MVar ())
   77 getOrCreateLock mgr sessionId = do
   78   mExisting <- atomically $ do
   79     locks <- readTVar (slmLocks mgr)
   80     pure $ Map.lookup sessionId locks
   81   case mExisting of
   82     Just lock -> pure lock
   83     Nothing -> do
   84       newLock <- newMVar ()
   85       result <- atomically $ do
   86         locks <- readTVar (slmLocks mgr)
   87         case Map.lookup sessionId locks of
   88           Just existing -> pure (Right existing)
   89           Nothing ->
   90             if Map.size locks >= slmMaxTrackedLocks mgr
   91               then pure (Left (Map.size locks))
   92               else do
   93                 modifyTVar' (slmLocks mgr) (Map.insert sessionId newLock)
   94                 pure (Right newLock)
   95       case result of
   96         Right lock -> pure lock
   97         Left tracked -> do
   98           hPutStrLn stderr $ "session_lock_overflow: session=" <> T.unpack sessionId <> " tracked=" <> show tracked <> " cap=" <> show (slmMaxTrackedLocks mgr)
   99           pure (slmOverflowLock mgr)
