    7 {-|
    8 Module      : QxFx0.Learning.Parser
    9 Description : WP4 — Parse external tool response into KnowledgeFruitPayload.
   10 
   11 Supports strict JSON schema first; falls back to constrained text
   12 parsing only when the response body is not valid JSON.
   13 
   14 The JSON schema expected from an LLM (or script) is:
   15 
   16 ```json
   17 {
   18   "proposition": "string",
   19   "word": "string",
   20   "definition": "string",
   21   "morphology": {
   22     "gender": "masculine|feminine|neuter",
   23     "declension": "first|second|...",
   24     "cases": {
   25       "nom": "...", "gen": "...", "dat": "...",
   26       "acc": "...", "ins": "...", "pre": "..."
   27     }
   28   },
   29   "source": "llm|human|script|internal",
   30   "conatusDelta": 0.0,
   31   "predictiveDelta": 0.0
   32 }
   33 ```
   34 
   35 All fields except @morphology@ are required.  Missing optional fields
   36 yield 'Nothing' / 'False' / '0.0' defaults.
   37 -}

   63 -- | Top-level entry point: parse an 'ExternalQueryResponse' into a
   64 -- 'KnowledgeFruitPayload'.
   65 --
   66 -- 1. Try the structured field (already extracted JSON from Mistral
   67 --    chat-completion envelope).
   68 -- 2. Try the raw body as JSON.
   69 -- 3. Fallback to constrained text parsing.
   70 parseLLMResponseToFruit :: ExternalQueryResponse -> Maybe KnowledgeFruitPayload
   71 parseLLMResponseToFruit resp =
   72   -- First: try structured field (JSON inside the LLM content field)
   73   parseJsonPayload (eqrStructured resp)
   74   -- Second: try raw body (in case structured extraction failed)
   75   <|> parseJsonPayload (eqrRawBody resp)
   76   -- Third: fallback text parser
   77   <|> parseStructuredTextToFruit (eqrRawBody resp)
   78 
   79 -- | Parse a strict JSON payload.
   80 parseJsonPayload :: Text -> Maybe KnowledgeFruitPayload
   81 parseJsonPayload txt =
   82   case decodeStrict (TE.encodeUtf8 txt) of
   83     Nothing -> Nothing
   84     Just val ->
   85       case parseEither parseFruitPayload val of
   86         Left _  -> Nothing
   87         Right p -> Just p
   88 
   89 parseFruitPayload :: Value -> Parser KnowledgeFruitPayload
   90 parseFruitPayload = withObject "KnowledgeFruitPayload" $ \o -> do
   91   prop      <- o .:  "proposition"
   92   word      <- o .:  "word"
   93   def       <- o .:  "definition"
   94   morphObj  <- o .:? "morphology"
   95   srcTag    <- o .:? "source" .!= "llm"
   96   cDelta    <- o .:? "conatusDelta" .!= 0.0
   97   pDelta    <- o .:? "predictiveDelta" .!= 0.0
   98   morph <- case morphObj of
   99     Nothing -> pure Nothing
  100     Just m  -> Just <$> parseMorphologyPayload m
  101   pure KnowledgeFruitPayload
  102     { kfpProposition     = prop
  103     , kfpWord            = word
  104     , kfpDefinition      = def
  105     , kfpMorphology      = morph
  106     , kfpSource          = parseSourceTag srcTag
  107     , kfpConatusDelta    = cDelta
  108     , kfpPredictiveDelta = pDelta
  109     }
  110 
  111 parseMorphologyPayload :: Value -> Parser MorphologyPayload
  112 parseMorphologyPayload = withObject "MorphologyPayload" $ \o ->
  113   MorphologyPayload
  114     <$> o .:? "gender"
  115     <*> o .:? "declension"
  116     <*> o .:? "cases"
  117 
  118 parseSourceTag :: Text -> KnowledgeSource
  119 parseSourceTag "llm"      = SourceLLM
  120 parseSourceTag "human"    = SourceHuman
  121 parseSourceTag "script"   = SourceScript
  122 parseSourceTag "internal" = SourceInternal
  123 parseSourceTag _          = SourceLLM
  124 
  125 -- | Fallback constrained text parser.
  126 --
  127 -- Expects a very rigid format (one per line):
  128 --   Word: <word>
  129 --   Definition: <definition>
  130 --   Proposition: <proposition>
  131 --   ConatusDelta: <float>
  132 --   PredictiveDelta: <float>
  133 --
  134 -- If any required line is missing or malformed, returns 'Nothing'.
  135 parseStructuredTextToFruit :: Text -> Maybe KnowledgeFruitPayload
  136 parseStructuredTextToFruit txt = do
  137   let lines' = map T.strip (T.lines txt)
  138       dict   = M.fromList
  139                  [ (T.toLower (T.takeWhile (/= ':') line), T.strip (T.drop 1 (T.dropWhile (/= ':') line)))
  140                  | line <- lines'
  141                  , T.isInfixOf ":" line
  142                  ]
  143   word        <- M.lookup (T.pack "word") dict
  144   definition  <- M.lookup (T.pack "definition") dict
  145   proposition <- Just (fromMaybe word (M.lookup (T.pack "proposition") dict))
  146   let cDelta  = readDouble (fromMaybe "0.0" (M.lookup (T.pack "conatusdelta") dict))
  147       pDelta  = readDouble (fromMaybe "0.0" (M.lookup (T.pack "predictivedelta") dict))
  148   pure KnowledgeFruitPayload
  149     { kfpProposition     = proposition
  150     , kfpWord            = word
  151     , kfpDefinition      = definition
  152     , kfpMorphology      = Nothing
  153     , kfpSource          = SourceLLM
  154     , kfpConatusDelta    = cDelta
  155     , kfpPredictiveDelta = pDelta
  156     }
  157 
  158 readDouble :: Text -> Double
  159 readDouble t =
  160   case reads (T.unpack (T.strip t)) of
  161     [(d, "")] -> d
  162     _         -> 0.0
