   20 deriveDialogueThread :: DialogueThread -> DialogueCommitmentLedger -> DialogueState -> UtteranceSemanticFrame -> DialogueThread
   21 deriveDialogueThread previous ledger dialogue frame =
   22   let focus = firstNonEmpty [usfTopic frame, usfFocus frame, dsLastTopic dialogue, dtCurrentFocus previous]
   23       structuralScope = deriveStructuralScope previous ledger frame focus
   24       activeQuestion = if usfSpeechAct frame == ActAsk then Just (usfRawText frame) else dtActiveQuestion previous
   25       userGoal = firstNonEmptyMaybe (maybeToList (usfTarget frame) ++ [usfTopic frame] ++ maybeToList (dtUserGoal previous))
   26       intentHypothesis = Just (renderSpeechAct (usfSpeechAct frame))
   27       acceptedTerms = take 8 . nub $ dtAcceptedTerms previous ++ map wmuLemma (usfWordUnits frame)
   28       openLoops = take 8 . nub $ unresolvedClaims ledger ++ if usfSpeechAct frame == ActAsk then [usfRawText frame] else []
   29       clarifiedItems = if usfSpeechAct frame == ActAsk then dtClarifiedItems previous else take 8 . nub $ usfTopic frame : dtClarifiedItems previous
   30       unclarifiedItems = take 8 . nub $ openLoops ++ filter (not . T.null) [usfFocus frame]
   31       resistance = clamp01 (if usfPolarity frame == PolarityNegative then 0.65 else 0.25)
   32       topicConfidence = clamp01 (max (usfConfidence frame) (dtTopicConfidence previous * 0.8))
   33   in DialogueThread
   34       { dtCurrentFocus = focus
   35       , dtActiveQuestion = activeQuestion
   36       , dtUserGoal = userGoal
   37       , dtIntentHypothesis = intentHypothesis
   38       , dtOpenLoops = openLoops
   39       , dtAcceptedTerms = acceptedTerms
   40       , dtTopicConfidence = topicConfidence
   41        , dtResistance = resistance
   42        , dtClarifiedItems = clarifiedItems
   43        , dtUnclarifiedItems = unclarifiedItems
   44        , dtStructuralScope = structuralScope
   45        , dtPhaseScope = focus
   46        }

   48 deriveStructuralScope :: DialogueThread -> DialogueCommitmentLedger -> UtteranceSemanticFrame -> Text -> Maybe PerspectiveScope
   49 deriveStructuralScope previous ledger frame focus =
   50   case firstNonEmptyMaybe [usfTopic frame, usfFocus frame, focus] of
   51     Nothing -> dtStructuralScope previous
   52     Just scopeText ->
   53       case dtStructuralScope previous of
   54         Just priorScope
   55           | lockedScope && normalizeScopeKey (scopeTextOf priorScope) /= normalizeScopeKey scopeText ->
   56               Just priorScope
   57         _ -> Just (ScopeTopic scopeText)
   58   where
   59     lockedScope =
   60       any ((`elem` [CsUnresolved, CsContested, CsSuspended]) . dcStatus) (dclItems ledger)
   61         || usfSpeechAct frame == ActAsk
   62 
   63     scopeTextOf :: PerspectiveScope -> Text
   64     scopeTextOf scope =
   65       case scope of
   66         ScopeTopic topic -> topic
   67         ScopeTheme theme -> theme
   68         ScopeCluster cluster -> cluster
   69 
   70     normalizeScopeKey :: Text -> Text
   71     normalizeScopeKey = T.unwords . T.words . T.toLower . T.strip

   73 deriveDialogueCommitmentLedger :: DialogueCommitmentLedger -> UtteranceSemanticFrame -> DialogueCommitmentLedger
   74 deriveDialogueCommitmentLedger (DialogueCommitmentLedger existing) frame =
   75   let currentClaim = firstNonEmpty [usfTopic frame, usfFocus frame, usfRawText frame]
   76       nextItem
   77         | T.null currentClaim = Nothing
   78         | usfSpeechAct frame == ActAsk = Just (DialogueCommitment currentClaim CsUnresolved 0)
   79         | usfPolarity frame == PolarityNegative = Just (DialogueCommitment currentClaim CsContested 0)
   80         | otherwise = Just (DialogueCommitment currentClaim CsAccepted 0)
   81       merged = take 16 (existing ++ maybe [] pure nextItem)
   82   in DialogueCommitmentLedger merged

   84 deriveDialoguePhase :: DialogueThread -> DialogueCommitmentLedger -> UtteranceSemanticFrame -> DialoguePhase
   85 deriveDialoguePhase thread ledger frame
   86   | hasContested ledger = Contesting
   87   | hasRepairNeed thread frame = Repairing
   88   | hasUnresolved ledger = Clarifying
   89   | usfSpeechAct frame == ActAsk = Exploring
   90   | commitmentAllowsAdvance ledger = Advancing
   91   | otherwise = Grounding
   92 
   93 commitmentAllowsAdvance :: DialogueCommitmentLedger -> Bool
   94 commitmentAllowsAdvance (DialogueCommitmentLedger items) =
   95   not (any ((`elem` [CsUnresolved, CsContested, CsSuspended]) . dcStatus) items)
   96 
   97 hasUnresolved :: DialogueCommitmentLedger -> Bool
   98 hasUnresolved (DialogueCommitmentLedger items) = any ((== CsUnresolved) . dcStatus) items
   99 
  100 hasContested :: DialogueCommitmentLedger -> Bool
  101 hasContested (DialogueCommitmentLedger items) = any ((== CsContested) . dcStatus) items
  102 
  103 hasRepairNeed :: DialogueThread -> UtteranceSemanticFrame -> Bool
  104 hasRepairNeed thread frame =
  105   dtResistance thread > 0.6 || any isRepairClass (concatMap wmuSemanticClasses (usfWordUnits frame))
  106   where
  107     isRepairClass SemDialogueRepair = True
  108     isRepairClass _ = False
  109 
  110 unresolvedClaims :: DialogueCommitmentLedger -> [Text]
  111 unresolvedClaims (DialogueCommitmentLedger items) = [dcClaim item | item <- items, dcStatus item == CsUnresolved]

  113 renderSpeechAct :: InputSpeechAct -> Text
  114 renderSpeechAct act = case act of
  115   ActAssert -> "assert"
  116   ActAsk -> "ask"
  117   ActRequest -> "request"
  118   ActInvite -> "invite"
  119   ActReport -> "report"
  120   ActUnknown -> "unknown"

  122 firstNonEmpty :: [Text] -> Text
  123 firstNonEmpty = fromMaybe "" . firstNonEmptyMaybe
  124 
  125 firstNonEmptyMaybe :: [Text] -> Maybe Text
  126 firstNonEmptyMaybe = go
  127   where
  128     go [] = Nothing
  129     go (x:xs)
  130       | T.null (T.strip x) = go xs
  131       | otherwise = Just x
  132 
  133 clamp01 :: Double -> Double
  134 clamp01 = max 0.0 . min 1.0

  136 maybeToList :: Maybe Text -> [Text]
  137 maybeToList (Just value) = [value]
