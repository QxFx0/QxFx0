   11 
   12 import Control.DeepSeq (NFData)
   13 import Data.Aeson
   14   ( FromJSON(..)
   15   , ToJSON(..)
   16   , object
   17   , withObject
   18   , (.:)
   19   , (.:?)
   20   , (.!=)
   21   , (.=)
   22   )
   23 import Data.Text (Text)

   34 import QxFx0.Types.Observability

   39   )
   40 import QxFx0.Types.SemanticConfig (SemanticConfig, defaultSemanticConfig)

   43 data SemanticState = SemanticState
   44   { semTrace :: !AtomTrace
   45   , semMeaningGraph :: !MeaningGraph
   46   , semKernelPulse :: !KernelPulse
   47   , semBlockedConcepts :: ![Text]
   48   , semClusters :: ![ClusterDef]
   49   , semDreamState :: !DreamState
   50   , semIntuitionState :: !(Maybe IntuitiveState)
   51   , semSemanticAnchor :: !(Maybe SemanticAnchor)
   52   , semLastTurnDecision :: !(Maybe TurnDecision)
   53   , semIntuitConfidence :: !Double
   54   , semConfig :: !SemanticConfig
   55   } deriving stock (Eq, Show, Generic)
   56     deriving anyclass (NFData)
   57 
   58 instance ToJSON SemanticState where
   59   toJSON sem = object
   60     [ "trace" .= semTrace sem
   61     , "meaningGraph" .= semMeaningGraph sem
   62     , "kernelPulse" .= semKernelPulse sem
   63     , "blockedConcepts" .= semBlockedConcepts sem
   64     , "clusters" .= semClusters sem
   65     , "dreamState" .= semDreamState sem
   66     , "intuitionState" .= semIntuitionState sem
   67     , "semanticAnchor" .= semSemanticAnchor sem
   68     , "lastTurnDecision" .= semLastTurnDecision sem
   69     , "intuitConfidence" .= semIntuitConfidence sem
   70     , "semanticConfig" .= semConfig sem
   71     ]
   72 
   73 instance FromJSON SemanticState where
   74   parseJSON = withObject "SemanticState" $ \o -> SemanticState
   75     <$> o .: "trace"
   76     <*> o .: "meaningGraph"
   77     <*> o .: "kernelPulse"
   78     <*> o .: "blockedConcepts"
   79     <*> o .: "clusters"
   80     <*> o .:? "dreamState" .!= emptyDreamState zeroVec
   81     <*> o .:? "intuitionState" .!= Just defaultIntuitiveState
   82     <*> o .:? "semanticAnchor" .!= Nothing
   83     <*> o .:? "lastTurnDecision" .!= Nothing
   84     <*> o .: "intuitConfidence"
   85     <*> o .:? "semanticConfig" .!= defaultSemanticConfig
   86 
   87 emptySemanticState :: SemanticState
   88 emptySemanticState = SemanticState
   89   { semTrace = emptyAtomTrace
   90   , semMeaningGraph = emptyMeaningGraph
   91   , semKernelPulse = emptyKernelPulse
   92   , semBlockedConcepts = []
   93   , semClusters = []
   94   , semDreamState = emptyDreamState zeroVec
   95   , semIntuitionState = Just defaultIntuitiveState
   96   , semSemanticAnchor = Nothing
   97   , semLastTurnDecision = Nothing
   98   , semIntuitConfidence = 0.0
   99   , semConfig = defaultSemanticConfig
  100   }
