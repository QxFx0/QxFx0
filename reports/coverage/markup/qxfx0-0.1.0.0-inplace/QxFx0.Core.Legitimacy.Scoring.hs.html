    4 {-| Quantitative legitimacy scoring over parser confidence, shadow divergence, and load. -}
    5 module QxFx0.Core.Legitimacy.Scoring
    6   ( legitimacyScore
    7   , legitimacyRecoveryBonus
    8   ) where
    9 
   10 import QxFx0.Types.Thresholds
   11   ( legitimacyApiPenalty
   12   , legitimacyConfidencePenaltyWeight
   13   , legitimacyHighLoadPenalty
   14   , legitimacyHighLoadThreshold
   15   , legitimacyShadowAgreementBonus
   16   , legitimacyStableRouteBonus
   17   )
   18 import QxFx0.Types.ShadowDivergence
   19   ( ShadowDivergence
   20   , computeShadowLegitimacyPenalty
   21   )
   22 
   23 legitimacyScore :: Double -> ShadowDivergence -> Double -> Bool -> Double
   24 legitimacyScore parserConfidence shadowDivergence emaLoad apiHealthy =
   25   let base = 1.0
   26       confidencePenalty = max 0.0 (1.0 - parserConfidence) * legitimacyConfidencePenaltyWeight
   27       shadowPenalty = computeShadowLegitimacyPenalty shadowDivergence
   28       emaPenalty =
   29         if emaLoad > legitimacyHighLoadThreshold
   30           then legitimacyHighLoadPenalty
   31           else 0.0
   32       apiPenalty =
   33         if apiHealthy
   34           then 0.0
   35           else legitimacyApiPenalty
   36    in max 0.0 (min 1.0 (base - confidencePenalty - shadowPenalty - emaPenalty - apiPenalty))
   37 
   38 legitimacyRecoveryBonus :: Bool -> Bool -> Double
   39 legitimacyRecoveryBonus shadowMatched stableRoute =
   40   shadowBonus + stableBonus
   41   where
   42     shadowBonus =
   43       if shadowMatched
   44         then legitimacyShadowAgreementBonus
   45         else 0.0
   46     stableBonus =
   47       if stableRoute
   48         then legitimacyStableRouteBonus
   49         else 0.0
