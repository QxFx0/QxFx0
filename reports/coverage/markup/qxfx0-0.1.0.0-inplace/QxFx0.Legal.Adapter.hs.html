   19 -- | A legal fact retrieved from an external or embedded source.
   20 data LegalFact = LegalFact
   21   { lfSourceId :: !Text    -- ^ Canonical source identifier (e.g. "ru.civil.code.art1")
   22   , lfSection :: !Text     -- ^ Section or article reference
   23   , lfText :: !Text        -- ^ Fact text in Russian
   24   , lfConfidence :: !Double -- ^ Confidence score in [0,1]
   25   , lfUncertainty :: !Bool  -- ^ True when the fact has known uncertainty / caveats
   26   } deriving stock (Eq, Show)
   27 
   28 -- | Mandatory safety disclaimer appended to every legal response.
   29 legalDisclaimer :: Text

   73 -- | Retrieve a legal fact by subject keyword.
   74 --   Returns 'Nothing' for non-legal topics or unknown keywords.
   75 retrieveLegalFact :: Text -> IO (Maybe LegalFact)
   76 retrieveLegalFact key = pure $ M.lookup (T.toLower (T.strip key)) legalFactDb
   77 
   78 -- | Render a 'LegalFact' into a single knowledge fragment.
   79 --   Includes the fact text, source trace, confidence annotation, and mandatory disclaimer.
   80 legalFactToKnowledgeFragment :: LegalFact -> Text
   81 legalFactToKnowledgeFragment lf =
   82   let sourceTrace = "[источник: " <> lfSourceId lf <> " | раздел: " <> lfSection lf <> "]"
   83       confidenceLabel
   84         | lfConfidence lf >= 0.95 = " (высокая достоверность)"
   85         | lfUncertainty lf        = " (есть неопределенность)"
   86         | otherwise               = ""
   87   in T.intercalate " "
   88        [ lfText lf
   89        , sourceTrace
   90        , confidenceLabel
   91        , legalDisclaimer
   92        ]
