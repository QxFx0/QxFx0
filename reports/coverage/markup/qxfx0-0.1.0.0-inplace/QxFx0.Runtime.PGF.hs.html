   20 defaultPgfPath :: FilePath
   21 defaultPgfPath = "spec/gf/QxFx0Syntax.pgf"
   22 
   23 linearizeClaimAstGf :: Maybe FilePath -> ClaimAst -> IO (Either Text Text)
   24 linearizeClaimAstGf mPgfPath ast =
   25   case astToGfExpr ast of
   26     Left err -> pure (Left err)
   27     Right expr -> do
   28       let pgfPath = fromMaybe defaultPgfPath mPgfPath
   29       exists <- doesFileExist pgfPath
   30       if not exists
   31         then pure (Left ("pgf_missing:" <> T.pack pgfPath))
   32         else do
   33           (exitCode, stdoutText, stderrText) <-
   34             readProcessWithExitCode "gf" ["-run", pgfPath] (T.unpack expr <> "\n")
   35           case exitCode of
   36             ExitSuccess ->
   37               case extractLinearization (T.pack stdoutText) of
   38                 Just rendered -> pure (Right rendered)
   39                 Nothing -> pure (Left "gf_empty_output")
   40             ExitFailure _ ->
   41               let err = summarizeGfFailure (T.pack stderrText) (T.pack stdoutText)
   42               in pure (Left err)
   43 
   44 astToGfExpr :: ClaimAst -> Either Text Text
   45 astToGfExpr ast =
   46   case ast of
   47     MoveInvite (MkNP topic) modifier action ->
   48       do
   49         actionExpr <- gfActionExpr action
   50         Right ("MoveInvite (MkNP " <> topic <> ") " <> gfModifierExpr modifier <> " " <> actionExpr)
   51     MoveDefine (MkNP subj) rel (MkNP obj) ->
   52       Right ("MoveDefine (MkNP " <> subj <> ") " <> gfRelationExpr rel <> " (MkNP " <> obj <> ")")
   53     MoveCause (MkNP subj) mech ->
   54       Right ("MoveCause (MkNP " <> subj <> ") " <> gfMechanismExpr mech)
   55     MovePurpose (MkNP topic) ->
   56       Right ("MovePurpose (MkNP " <> topic <> ")")
   57     MoveSelfState ->
   58       Right "MoveSelfState"
   59     MoveCompare (MkNP left) (MkNP right) ->
   60       Right ("MoveCompare (MkNP " <> left <> ") (MkNP " <> right <> ")")
   61     MoveOperationalStatus ->
   62       Right "MoveOperationalStatus"
   63     MoveOperationalCause ->
   64       Right "MoveOperationalCause"
   65     MoveSystemLogic ->
   66       Right "MoveSystemLogic"
   67     MoveMisunderstanding ->
   68       Right "MoveMisunderstanding"
   69     MoveGenerativeThought ->
   70       Right "MoveGenerativeThought"
   71     MoveContemplative (MkNP topic) ->
   72       Right ("MoveContemplative (MkNP " <> topic <> ")")
   73     MoveGround (MkNP topic) ->
   74       Right ("MoveGround (MkNP " <> topic <> ")")
   75     MoveContact (MkNP topic) ->
   76       Right ("MoveContact (MkNP " <> topic <> ")")
   77     MoveReflect (MkNP topic) ->
   78       Right ("MoveReflect (MkNP " <> topic <> ")")
   79     MoveDescribe (MkNP topic) ->
   80       Right ("MoveDescribe (MkNP " <> topic <> ")")
   81     MoveDeepen (MkNP topic) ->
   82       Right ("MoveDeepen (MkNP " <> topic <> ")")
   83     MoveConfront (MkNP topic) ->
   84       Right ("MoveConfront (MkNP " <> topic <> ")")
   85     MoveAnchor (MkNP topic) ->
   86       Right ("MoveAnchor (MkNP " <> topic <> ")")
   87     MoveClarify (MkNP topic) ->
   88       Right ("MoveClarify (MkNP " <> topic <> ")")
   89     MoveNextStepLocal (MkNP topic) ->
   90       Right ("MoveNextStepLocal (MkNP " <> topic <> ")")
   91     MoveHypothesis (MkNP topic) ->
   92       Right ("MoveHypothesis (MkNP " <> topic <> ")")
   93     MoveDistinguish (MkNP left) (MkNP right) ->
   94       Right ("MoveDistinguish (MkNP " <> left <> ") (MkNP " <> right <> ")")
   95     StanceWrapped "ApplyStanceTentative" inner ->
   96       ("ApplyStanceTentative (" <>) . (<> ")") <$> astToGfExpr inner
   97     StanceWrapped "ApplyStanceFirm" inner ->
   98       ("ApplyStanceFirm (" <>) . (<> ")") <$> astToGfExpr inner
   99     StanceWrapped _ inner ->
  100       astToGfExpr inner
  101     ClaimPurpose subject ->
  102       Right ("MovePurpose (MkNP " <> sanitizeLegacyLexemeId subject <> ")")
  103     ClaimSelfState ->
  104       Right "MoveSelfState"
  105     ClaimComparison left right ->
  106       Right
  107         ( "MoveCompare (MkNP "
  108             <> sanitizeLegacyLexemeId left
  109             <> ") (MkNP "
  110             <> sanitizeLegacyLexemeId right
  111             <> ")"
  112         )

  114 gfModifierExpr :: GfModifier -> Text
  115 gfModifierExpr ModFirst = "ModFirst"
  116 gfModifierExpr ModStrictly = "ModStrictly"
  117 
  118 gfRelationExpr :: GfRelation -> Text
  119 gfRelationExpr RelIdentity = "RelIdentity"
  120 
  121 gfMechanismExpr :: GfMechanism -> Text
  122 gfMechanismExpr MechParse = "MechParse"
  123 
  124 gfNumberExpr :: GfNumber -> Text
  125 gfNumberExpr NumSg = "NumSg"
  126 gfNumberExpr NumPl = "NumPl"
  127 
  128 gfActionExpr :: GfVP -> Either Text Text
  129 gfActionExpr action =
  130   case action of
  131     ActMaintain number obj -> Right ("ActMaintain " <> gfNumberExpr number <> " " <> obj)
  132     ActDefine obj -> Right ("ActDefine " <> obj)

  134 extractLinearization :: Text -> Maybe Text
  135 extractLinearization rawOutput =
  136   let linesNoEmpty = filter (not . T.null) (map T.strip (T.lines rawOutput))
  137       pick =
  138         listToMaybe
  139           ( reverse
  140               [ normalizeQuoted line
  141               | line <- linesNoEmpty
  142               , not (">" `T.isPrefixOf` line)
  143               ]
  144           )
  145   in pick
  146 
  147 normalizeQuoted :: Text -> Text
  148 normalizeQuoted line =
  149   case T.uncons line of
  150     Just ('"', rest) ->
  151       case T.unsnoc rest of
  152         Just (middle, '"') -> middle
  153         _ -> line
  154     _ -> line
  155 
  156 summarizeGfFailure :: Text -> Text -> Text
  157 summarizeGfFailure stderrText stdoutText =
  158   let stderrLine = firstUsefulLine stderrText
  159       stdoutLine = firstUsefulLine stdoutText
  160       merged =
  161         case (stderrLine, stdoutLine) of
  162           (Just e, _) -> e
  163           (Nothing, Just o) -> o
  164           _ -> "gf_failed"
  165   in "gf_failed:" <> merged
  166 
  167 firstUsefulLine :: Text -> Maybe Text
  168 firstUsefulLine =
  169   listToMaybe
  170     . filter (not . T.null)
  171     . map (T.take 160 . T.dropAround isSpace)
  172     . T.lines
  173 
  174 sanitizeLegacyLexemeId :: Text -> Text
  175 sanitizeLegacyLexemeId = topicToGfLexemeId
