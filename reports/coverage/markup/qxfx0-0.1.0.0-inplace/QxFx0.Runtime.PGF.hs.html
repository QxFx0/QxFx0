   44 linearizeClaimAstGf :: Maybe FilePath -> ClaimAst -> IO (Either Text GfLinearizationResult)

   47 linearizeClaimAstGfLang :: Maybe FilePath -> Text -> ClaimAst -> IO (Either Text GfLinearizationResult)
   48 linearizeClaimAstGfLang mPgfPath lang ast =
   49   case astToGfExpr ast of
   50     Left err -> pure (Left err)
   51     Right expr -> do
   52       result <- linearizeExpr mPgfPath lang expr
   53       pure $ case result of
   54         Left err -> Left err
   55         Right raw
   56           | lang == "QxFx0SyntaxRus" ->
   57               Right (mkGfLinearizationResult mPgfPath lang RussianCompatShimRoute AuthorityShim (Just "russian_compatibility_shim") (fallbackSurfaceText ast raw) (artifactManifestFor mPgfPath lang RussianCompatShimRoute AuthorityShim (fallbackSurfaceText ast raw)))
   58           | otherwise ->
   59               Right (mkGfLinearizationResult mPgfPath lang PgfClaimRoute AuthorityCanonical Nothing raw (artifactManifestFor mPgfPath lang PgfClaimRoute AuthorityCanonical raw))

   78 linearizeExpr :: Maybe FilePath -> Text -> Text -> IO (Either Text Text)
   79 linearizeExpr mPgfPath lang expr = do
   80   let pgfPath = fromMaybe defaultPgfPath mPgfPath
   81   exists <- doesFileExist pgfPath
   82   if not exists
   83     then pure (Left ("pgf_missing:" <> T.pack pgfPath))
   84     else do
   85       result <- try $ do
   86         pgf <- PGF.readPGF pgfPath
   87         let langs = PGF.languages pgf
   88         case Map.lookup (T.unpack lang) langs of
   89           Nothing -> pure (Left ("pgf_lang_not_found:" <> lang <> ":available=" <> T.pack (show (Map.keys langs))))
   90           Just concr -> case PGF.readExpr (T.unpack expr) of
   91             Nothing -> pure (Left ("pgf_parse_expr_failed:" <> expr))
   92             Just pgfExpr ->
   93               let raw = T.pack (PGF.linearize concr pgfExpr)
   94               in pure (Right raw)
   95       case result of
   96         Left (e :: IOException) -> pure (Left ("pgf_exception:" <> T.pack (show e)))
   97         Right r -> pure r

   99 fallbackSurface :: ClaimAst -> Either Text Text
  100 fallbackSurface ast =
  101   maybe (Left "pgf_russian_fallback_failed:shim_route") Right (linearizeClaimAstRus ast StyleStandard emptyMorphologyData)

  103 fallbackSurfaceText :: ClaimAst -> Text -> Text
  104 fallbackSurfaceText ast fallbackText =
  105   maybe fallbackText id (linearizeClaimAstRus ast StyleStandard emptyMorphologyData)

  107 mkGfLinearizationResult :: Maybe FilePath -> Text -> AssemblyPath -> AuthorityClass -> Maybe Text -> Text -> ArtifactManifest -> GfLinearizationResult
  108 mkGfLinearizationResult mPgfPath lang assemblyPath authorityClass fallbackReason text manifest =
  109   GfLinearizationResult
  110     { glrText = text
  111     , glrLanguage = lang
  112     , glrAuthorityClass = authorityClass
  113     , glrAssemblyPath = assemblyPath
  114     , glrArtifactManifest = manifest { amPgfPath = mPgfPath }
  115     , glrFallbackReason = fallbackReason
  116     }

  118 artifactManifestFor :: Maybe FilePath -> Text -> AssemblyPath -> AuthorityClass -> Text -> ArtifactManifest
  119 artifactManifestFor mPgfPath lang assemblyPath authorityClass text =
  120   ArtifactManifest
  121     { amPgfPath = mPgfPath
  122     , amPgfHash = Just (fingerprintText (T.pack (show mPgfPath) <> "|" <> lang <> "|" <> T.pack (show assemblyPath) <> "|" <> T.pack (show authorityClass) <> "|" <> text <> "|toolchain=pgf2"))
  123     , amPgfSourceManifestHash = Just "gf_source_manifest:gf2"
  124     , amGeneratedInputLexiconHash = Just (inputGeneratedLexiconProvenanceTag <> ":required")
  125     , amGfMapHash = Just (gfMapProvenanceTag <> ":required")
  126     , amToolchainIdentity = Just "gf-compiler:pgf2"
  127     , amToolchainMarker = "runtime_pgf;lang=" <> lang <> ";route=" <> T.pack (show assemblyPath) <> ";authority=" <> T.pack (show authorityClass) <> ";compiler=pgf2"
  128     }

  130 fingerprintText :: Text -> Text
  131 fingerprintText = T.pack . ("fnv1a64:" ++) . (`showHex` "") . T.foldl' fnv1a64 14695981039346656037
  132   where
  133     fnv1a64 :: Word64 -> Char -> Word64
  134     fnv1a64 h ch = (h `xor` fromIntegral (fromEnum ch)) * 1099511628211

  139 astToGfExpr :: ClaimAst -> Either Text Text
  140 astToGfExpr ast =
  141   case ast of
  142     MoveInvite (MkNP topic) modifier action ->
  143       do
  144         actionExpr <- gfActionExpr action
  145         Right ("MoveInvite (MkNP " <> topic <> ") " <> gfModifierExpr modifier <> " " <> actionExpr)
  146     MoveDefine (MkNP subj) rel (MkNP obj) ->
  147       Right ("MoveDefine (MkNP " <> subj <> ") " <> gfRelationExpr rel <> " (MkNP " <> obj <> ")")
  148     MoveCause (MkNP subj) mech ->
  149       Right ("MoveCause (MkNP " <> subj <> ") " <> gfMechanismExpr mech)
  150     MovePurpose (MkNP topic) ->
  151       Right ("MovePurpose (MkNP " <> topic <> ")")
  152     MoveSelfState ->
  153       Right "MoveSelfState"
  154     MoveCompare (MkNP left) (MkNP right) ->
  155       Right ("MoveCompare (MkNP " <> left <> ") (MkNP " <> right <> ")")
  156     MoveOperationalStatus ->
  157       Right "MoveOperationalStatus"
  158     MoveOperationalCause ->
  159       Right "MoveOperationalCause"
  160     MoveSystemLogic ->
  161       Right "MoveSystemLogic"
  162     MoveMisunderstanding ->
  163       Right "MoveMisunderstanding"
  164     MoveGenerativeThought ->
  165       Right "MoveGenerativeThought"
  166     MoveContemplative (MkNP topic) ->
  167       Right ("MoveContemplative (MkNP " <> topic <> ")")
  168     MoveGround (MkNP topic) ->
  169       Right ("MoveGround (MkNP " <> topic <> ")")
  170     MoveContact (MkNP topic) ->
  171       Right ("MoveContact (MkNP " <> topic <> ")")
  172     MoveReflect (MkNP topic) ->
  173       Right ("MoveReflect (MkNP " <> topic <> ")")
  174     MoveDescribe (MkNP topic) ->
  175       Right ("MoveDescribe (MkNP " <> topic <> ")")
  176     MoveDeepen (MkNP topic) ->
  177       Right ("MoveDeepen (MkNP " <> topic <> ")")
  178     MoveConfront (MkNP topic) ->
  179       Right ("MoveConfront (MkNP " <> topic <> ")")
  180     MoveAnchor (MkNP topic) ->
  181       Right ("MoveAnchor (MkNP " <> topic <> ")")
  182     MoveClarify (MkNP topic) ->
  183       Right ("MoveClarify (MkNP " <> topic <> ")")
  184     MoveNextStepLocal (MkNP topic) ->
  185       Right ("MoveNextStepLocal (MkNP " <> topic <> ")")
  186     MoveHypothesis (MkNP topic) ->
  187       Right ("MoveHypothesis (MkNP " <> topic <> ")")
  188     MoveDistinguish (MkNP left) (MkNP right) ->
  189       Right ("MoveDistinguish (MkNP " <> left <> ") (MkNP " <> right <> ")")
  190     StanceWrapped "ApplyStanceTentative" inner ->
  191       ("ApplyStanceTentative (" <>) . (<> ")") <$> astToGfExpr inner
  192     StanceWrapped "ApplyStanceFirm" inner ->
  193       ("ApplyStanceFirm (" <>) . (<> ")") <$> astToGfExpr inner
  194     StanceWrapped _ inner ->
  195       astToGfExpr inner
  196     ClaimPurpose subject ->
  197       Right ("MovePurpose (MkNP " <> fst (sanitizeLegacyLexemeDecision subject) <> ")")
  198     ClaimSelfState ->
  199       Right "MoveSelfState"
  200     ClaimComparison left right ->
  201       Right
  202         ( "MoveCompare (MkNP "
  203             <> sanitizeLegacyLexemeId left
  204             <> legacyShimSuffix left
  205             <> ") (MkNP "
  206             <> sanitizeLegacyLexemeId right
  207             <> legacyShimSuffix right
  208             <> ")"
  209         )

  225 gfActionExpr :: GfVP -> Either Text Text
  226 gfActionExpr action =
  227   case action of
  228     ActMaintain number obj -> Right ("ActMaintain " <> gfNumberExpr number <> " " <> obj)
  229     ActDefine obj -> Right ("ActDefine " <> obj)

  231 dialogAtomsToGfExpr :: DialogAtoms -> Either Text Text
  232 dialogAtomsToGfExpr da =
  233   let topicStr = daTopicNominative da
  234       intent = if hasTag TUserIntent da then headAtomValue TUserIntent da else ""
  235   in case lookupTopicGfLexemeId buildGfLexemeMap topicStr of
  236        Nothing -> Left ("unresolved_topic_lexeme:" <> topicStr)
  237        Just gfTopic ->
  238          if intent == "define"
  239            then Right ("MoveDefine (MkNP " <> gfTopic <> ") (MkNP " <> gfTopic <> ")")
  240            else if intent == "ground"
  241              then Right ("MoveGround (MkNP " <> gfTopic <> ")")
  242              else Right ("MoveGround (MkNP " <> gfTopic <> ")")
  243 
  244 sanitizeLegacyLexemeId :: Text -> Text
  245 sanitizeLegacyLexemeId = LegacyGfMap.topicToGfLexemeId

  247 sanitizeLegacyLexemeDecision :: Text -> (Text, Maybe Text)

  250 legacyShimSuffix :: Text -> Text
  251 legacyShimSuffix raw =
  252   case sanitizeLegacyLexemeDecision raw of
  253     (_, Just _) -> ""
  254     _ -> ""
