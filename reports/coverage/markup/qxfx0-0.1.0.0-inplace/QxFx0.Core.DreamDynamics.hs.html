   22 -- | Run one dream update step for a bounded time delta.
   23 -- Applies drift decay, evidence quality gates, and bias relaxation.
   24 runDreamCycle :: DreamConfig -> [DreamThemeEvidence] -> UTCTime -> NominalDiffTime -> DreamState -> (DreamState, DreamCycleLog)
   25 runDreamCycle cfg evidence now delta dreamState =
   26   let hours = max 0.0 (realToFrac delta / 3600.0)
   27       r5 = dsR5State dreamState
   28       driftBefore = r5KernelDrift r5
   29       driftAfter = decayKernelDrift cfg hours driftBefore
   30       attractor = computeBiasAttractor cfg evidence
   31       biasBefore = r5ReflectionBias r5
   32       biasAfter = relaxReflectionBias cfg hours attractor biasBefore
   33       r5After = r5 { r5KernelDrift = driftAfter, r5ReflectionBias = biasAfter }
   34       nextState = dreamState
   35         { dsR5State = r5After
   36         , dsBiasAttractor = attractor
   37         , dsLastDreamTime = now
   38         , dsDreamCycleCount = dsDreamCycleCount dreamState + 1
   39         }
   40       (acceptedCount, rejectedCount) = evidenceGateCounts cfg evidence
   41       events =
   42         [ DriftDecayApplied hours (vecNorm driftBefore) (vecNorm driftAfter)
   43         , QualityGateApplied acceptedCount rejectedCount
   44         , AttractorComputed (vecNorm attractor)
   45         , BiasRelaxationApplied (vecNorm biasBefore) (vecNorm biasAfter) (vecNorm (vecSub biasAfter biasBefore))
   46         ]
   47       cycleLog = DreamCycleLog now hours r5 r5After attractor events
   48   in (nextState, cycleLog)

   50 -- | Catch up missed dream cycles by splitting elapsed time into fixed-size steps.
   51 runDreamCatchup :: DreamConfig -> [DreamThemeEvidence] -> UTCTime -> DreamState -> (DreamState, [DreamCycleLog])
   52 runDreamCatchup cfg evidence now dreamState
   53   | hoursSinceLastDream < dcDreamThresholdHours cfg = (dreamState, [])
   54   | otherwise = (finalState, reverse reverseLogs)
   55   where
   56     cappedHours = min (dcMaxCatchupHours cfg) hoursSinceLastDream
   57     cycleHours = max dreamMinCycleDurationHours (dcCycleDurationHours cfg)
   58     fullSteps = floor (cappedHours / cycleHours)
   59     remainder = cappedHours - fromIntegral fullSteps * cycleHours
   60     stepHours = replicate fullSteps cycleHours ++ [remainder | remainder > 1e-9]
   61     startTime = dsLastDreamTime dreamState
   62     hoursSinceLastDream = max 0.0 (realToFrac (diffUTCTime now startTime) / 3600.0)
   63     step (stateAcc, logsAcc) hours =
   64       let stepNow = addUTCTime (realToFrac (hours * 3600.0) :: NominalDiffTime) (dsLastDreamTime stateAcc)
   65           delta = realToFrac (hours * 3600.0) :: NominalDiffTime
   66           (state', log') = runDreamCycle cfg evidence stepNow delta stateAcc
   67       in (state', log' : logsAcc)
   68 
   69     (finalState, reverseLogs) = foldl' step (dreamState, []) stepHours

   71 decayKernelDrift :: DreamConfig -> Double -> CoreVec -> CoreVec
   72 decayKernelDrift cfg hours drift = vecScale (exp (negate (dcDriftLambdaPerHour cfg) * hours)) drift
   73 
   74 relaxReflectionBias :: DreamConfig -> Double -> CoreVec -> CoreVec -> CoreVec
   75 relaxReflectionBias cfg hours attractor currentBias =
   76   clampVecNorm (dcMaxReflectionBiasNorm cfg) (vecAdd currentBias cappedDelta)
   77   where
   78     alpha = dcBiasRelaxAlphaPerHour cfg * max 0.0 hours
   79     targetDelta = vecScale alpha (vecSub attractor currentBias)
   80     cappedDelta = clampVecNorm (dcBiasDeltaCapPerCycle cfg) targetDelta

   82 -- | Compute a bounded weighted attractor from accepted evidence only.
   83 computeBiasAttractor :: DreamConfig -> [DreamThemeEvidence] -> CoreVec
   84 computeBiasAttractor cfg evidence =
   85   clampVecNorm (dcMaxAttractorNorm cfg) weightedMean
   86   where
   87     accepted = [ (deaAcceptedWeight audit, dteBias item)
   88                | (item, audit) <- zip evidence (auditDreamEvidence cfg evidence)
   89                , deaAcceptedWeight audit > 0.0
   90                ]
   91     totalWeight = sum [ weight | (weight, _) <- accepted ]
   92     weightedMean
   93       | totalWeight < 1e-9 = zeroVec
   94       | otherwise = vecScale (1.0 / totalWeight) (foldl' vecAdd zeroVec [ vecScale weight bias | (weight, bias) <- accepted ])
   95 
   96 evidenceGateCounts :: DreamConfig -> [DreamThemeEvidence] -> (Int, Int)
   97 evidenceGateCounts cfg evidence = foldl' step (0, 0) (auditDreamEvidence cfg evidence)
   98   where
   99     step (accepted, rejected) audit
  100       | deaAcceptedWeight audit > 0.0 = (accepted + 1, rejected)
  101       | otherwise = (accepted, rejected + 1)
  102 
  103 auditDreamEvidence :: DreamConfig -> [DreamThemeEvidence] -> [DreamEvidenceAudit]
  104 auditDreamEvidence cfg = map auditOne
  105   where
  106     auditOne item =
  107       let quality = clamp01 (dteQualityWeight item)
  108           experience = clamp01 (dteExperienceWeight item)
  109           decision
  110             | not (dteBiographyPermission item) = Left RejectedByBiographyPermission
  111             | quality < dcMinQualityWeight cfg = Left (RejectedByLowQuality quality)
  112             | experience <= 0.0 = Left (RejectedByZeroExperienceWeight experience)
  113             | otherwise = Right (experience * quality)
  114       in DreamEvidenceAudit
  115            (dteTheme item)
  116            (either (const 0.0) id decision)
  117            quality
  118            experience
  119             (either Just (const Nothing) decision)

  121 buildTransitionMatrix :: MeaningGraph -> M.Map MeaningStateId (M.Map MeaningStateId Double)
  122 buildTransitionMatrix mg =
  123   let outgoing = foldl' addEdge M.empty (mgEdges mg)
  124       normalize m = M.mapWithKey (\_ neighbors ->
  125         let total = sum (M.elems neighbors)
  126         in if total > 1e-9 then M.map (/ total) neighbors else neighbors) m
  127       -- COMPAT GLUE: Source v2 used 3-field MeaningEdge; target has 9 fields.
  128       -- We only need the IDs and dream bias (weight) for the transition matrix.
  129       addEdge acc (MeaningEdge { meFromId = from, meToId = to, meDreamBias = w }) =
  130         M.insertWith (M.unionWith (+)) from (M.singleton to w) acc
  131   in normalize outgoing
