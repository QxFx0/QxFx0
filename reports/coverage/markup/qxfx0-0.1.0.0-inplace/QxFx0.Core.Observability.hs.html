    1 {-# LANGUAGE OverloadedStrings, RecordWildCards #-}
    2 {-| Turn-level observability primitives: timings, structured metric logs, and warnings. -}
    3 module QxFx0.Core.Observability
    4   ( RequestId
    5   , PhaseTiming(..)
    6   , TurnMetrics(..)
    7   , emptyTurnMetrics
    8   , recordPhase
    9   , addPhase
   10   , renderMetricsLog
   11   , logMetrics
   12   , hPutStrLnWarning
   13   ) where
   14 
   15 import Data.Text (Text)
   16 import qualified Data.Text as T
   17 import qualified Data.Text.IO as T
   18 import Data.Time.Clock (UTCTime, diffUTCTime)
   19 import System.IO (hPutStrLn, stderr)
   20 import QxFx0.Types.Text (textShow)
   21 
   22 type RequestId = Text
   23 
   24 data PhaseTiming = PhaseTiming
   25   { ptPhase :: !Text
   26   , ptStart :: !UTCTime
   27   , ptEnd   :: !UTCTime
   28   }
   29 
   30 data TurnMetrics = TurnMetrics
   31   { tmRequestId    :: !RequestId
   32   , tmSessionId    :: !Text
   33   , tmPhases       :: ![PhaseTiming]
   34   , tmTurnCount    :: !Int
   35   , tmFamily       :: !Text
   36   , tmEmbeddingSource :: !Text
   37   , tmNixStatus    :: !Text
   38   , tmSafetyStatus :: !Text
   39   , tmApiHealthy   :: !Bool
   40   , tmError        :: !(Maybe Text)
   41   }
   42 
   43 emptyTurnMetrics :: RequestId -> Text -> TurnMetrics
   44 emptyTurnMetrics rid sid = TurnMetrics
   45   { tmRequestId    = rid
   46   , tmSessionId    = sid
   47   , tmPhases       = []
   48   , tmTurnCount    = 0
   49   , tmFamily       = ""
   50   , tmEmbeddingSource = ""
   51   , tmNixStatus    = ""
   52   , tmSafetyStatus = ""
   53   , tmApiHealthy   = True
   54   , tmError        = Nothing
   55   }
   56 
   57 recordPhase :: Text -> UTCTime -> UTCTime -> PhaseTiming
   58 recordPhase = PhaseTiming
   59 
   60 addPhase :: PhaseTiming -> TurnMetrics -> TurnMetrics
   61 addPhase pt tm = tm { tmPhases = pt : tmPhases tm }
   62 
   63 renderMetricsLog :: TurnMetrics -> Text
   64 renderMetricsLog TurnMetrics{..} = T.intercalate " "
   65   [ "qxfx0_turn"
   66   , "request_id=" <> tmRequestId
   67   , "session_id=" <> tmSessionId
   68   , "turn=" <> textShow tmTurnCount
   69   , "family=" <> tmFamily
   70   , "embedding=" <> tmEmbeddingSource
   71   , "nix=" <> tmNixStatus
   72   , "safety=" <> tmSafetyStatus
   73   , "api_healthy=" <> (if tmApiHealthy then "1" else "0")
   74   , "phases=" <> T.intercalate "," (map renderPhaseTiming tmPhases)
   75   , "total_ms=" <> textShow (totalDurationMs tmPhases)
   76   , case tmError of
   77       Nothing -> ""
   78       Just e  -> "error=" <> e
   79   ]
   80 
   81 renderPhaseTiming :: PhaseTiming -> Text
   82 renderPhaseTiming PhaseTiming{..} = ptPhase <> ":" <> textShow (phaseDurationMs ptStart ptEnd) <> "ms"
   83 
   84 phaseDurationMs :: UTCTime -> UTCTime -> Double
   85 phaseDurationMs start end = realToFrac (diffUTCTime end start) * 1000.0
   86 
   87 totalDurationMs :: [PhaseTiming] -> Double
   88 totalDurationMs [] = 0.0
   89 totalDurationMs phases =
   90   let starts = map ptStart phases
   91       ends   = map ptEnd phases
   92   in case (starts, ends) of
   93        (_:_, _:_) -> phaseDurationMs (minimum starts) (maximum ends)
   94        _ -> 0.0
   95 
   96 logMetrics :: TurnMetrics -> IO ()
   97 -- NOTE: this module is intentionally used from runtime boundaries; direct stderr output stays here
   98 -- to avoid threading an IO logger through pure planning code paths.
   99 logMetrics = T.hPutStrLn stderr . renderMetricsLog
  100 
  101 hPutStrLnWarning :: String -> IO ()
  102 hPutStrLnWarning = hPutStrLn stderr
