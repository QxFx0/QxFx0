   82 recordPhase :: Text -> UTCTime -> UTCTime -> PhaseTiming
   83 recordPhase = PhaseTiming

   85 addPhase :: PhaseTiming -> TurnMetrics -> TurnMetrics
   86 addPhase pt tm = tm { tmPhases = pt : tmPhases tm }

   88 recordThresholdProbe :: Text -> Double -> Bool -> TurnMetrics -> TurnMetrics
   89 recordThresholdProbe signal threshold fired tm =
   90   tm
   91     { tmThresholds =
   92         ThresholdProbe signal threshold fired : tmThresholds tm
   93     }

   95 renderMetricsLog :: TurnMetrics -> Text
   96 renderMetricsLog TurnMetrics{..} = T.intercalate " "
   97   [ "qxfx0_turn"
   98   , "request_id=" <> sanitizeLogField tmRequestId
   99   , "session_id=" <> sanitizeLogField tmSessionId
  100   , "turn=" <> textShow tmTurnCount
  101   , "family=" <> sanitizeLogField tmFamily
  102   , "embedding=" <> sanitizeLogField tmEmbeddingSource
  103   , "nix=" <> sanitizeLogField tmNixStatus
  104   , "safety=" <> sanitizeLogField tmSafetyStatus
  105   , "api_healthy=" <> (if tmApiHealthy then "1" else "0")
  106   , "phases=" <> T.intercalate "," (map renderPhaseTiming tmPhases)
  107   , "thresholds=" <> T.intercalate "," (map renderThresholdProbe tmThresholds)
  108   , "total_ms=" <> textShow (totalDurationMs tmPhases)
  109   , case tmError of
  110       Nothing -> ""
  111       Just e  -> "error=" <> sanitizeLogField e
  112   ]
  113 
  114 renderThresholdProbe :: ThresholdProbe -> Text
  115 renderThresholdProbe ThresholdProbe{..} =
  116   thrSignal
  117     <> ":"
  118     <> textShow thrThreshold
  119     <> ":"
  120     <> (if thrFired then "1" else "0")
  121 
  122 renderPhaseTiming :: PhaseTiming -> Text
  123 renderPhaseTiming PhaseTiming{..} = ptPhase <> ":" <> textShow (phaseDurationMs ptStart ptEnd) <> "ms"
  124 
  125 phaseDurationMs :: UTCTime -> UTCTime -> Double
  126 phaseDurationMs start end = realToFrac (diffUTCTime end start) * 1000.0
  127 
  128 totalDurationMs :: [PhaseTiming] -> Double
  129 totalDurationMs [] = 0.0
  130 totalDurationMs phases =
  131   let starts = map ptStart phases
  132       ends   = map ptEnd phases
  133   in case (starts, ends) of
  134        (_:_, _:_) -> phaseDurationMs (minimum starts) (maximum ends)
  135        _ -> 0.0

  137 logMetrics :: TurnMetrics -> IO ()
  138 -- NOTE: this module is intentionally used from runtime boundaries; direct stderr output stays here
  139 -- to avoid threading an IO logger through pure planning code paths.
  140 logMetrics = T.hPutStrLn stderr . renderMetricsLog
  141 
  142 hPutStrLnWarning :: Text -> IO ()
  143 hPutStrLnWarning = T.hPutStrLn stderr

  145 sanitizeLogField :: Text -> Text
  146 sanitizeLogField = T.filter (\c -> c >= ' ' && c <= '~' && c /= '\\' && c /= '\n' && c /= '\r')
