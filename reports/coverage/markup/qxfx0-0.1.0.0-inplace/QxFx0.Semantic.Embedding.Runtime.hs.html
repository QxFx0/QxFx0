   30 data EmbeddingSelection = EmbeddingSelection !EmbeddingBackend !Bool !(Maybe String)
   31 
   32 checkApiHealth :: APIHealthCache -> IO Bool
   33 checkApiHealth cache = ehOperational <$> checkEmbeddingHealth cache
   34 
   35 checkEmbeddingHealth :: APIHealthCache -> IO EmbeddingHealth
   36 checkEmbeddingHealth cache = do
   37   selection <- resolveEmbeddingSelection
   38   case selection of
   39     EmbeddingSelection EmbeddingBackendLocalDeterministic explicit _ ->
   40       pure EmbeddingHealth
   41         { ehBackend = EmbeddingBackendLocalDeterministic
   42         , ehQuality = EmbeddingQualityHeuristic
   43         , ehExplicit = explicit
   44         , ehOperational = True
   45         , ehStrictReady = True
   46         }
   47     EmbeddingSelection EmbeddingBackendRemoteHTTP explicit mUrl ->
   48       case mUrl of
   49         Nothing ->
   50           pure EmbeddingHealth
   51             { ehBackend = EmbeddingBackendRemoteHTTP
   52             , ehQuality = EmbeddingQualityModeled
   53             , ehExplicit = explicit
   54             , ehOperational = False
   55             , ehStrictReady = False
   56             }
   57         Just url -> do
   58           operational <- cachedRemoteHealth cache url
   59           pure EmbeddingHealth
   60             { ehBackend = EmbeddingBackendRemoteHTTP
   61             , ehQuality = EmbeddingQualityModeled
   62             , ehExplicit = explicit
   63             , ehOperational = operational
   64             , ehStrictReady = explicit && operational
   65             }

   67 textToEmbedding :: String -> IO Embedding
   68 textToEmbedding input = erEmbedding <$> textToEmbeddingResult input
   69 
   70 textToEmbeddingResult :: String -> IO EmbeddingResult
   71 textToEmbeddingResult input = do
   72   selection <- resolveEmbeddingSelection
   73   case selection of
   74     EmbeddingSelection EmbeddingBackendLocalDeterministic explicit _ ->
   75       pure EmbeddingResult
   76         { erEmbedding = fallbackEmbedding input
   77         , erSource = if explicit then EmbeddingLocalDeterministic else EmbeddingLocalImplicit
   78         }
   79     EmbeddingSelection EmbeddingBackendRemoteHTTP _ (Just url) ->
   80       tryRemote url input
   81     EmbeddingSelection EmbeddingBackendRemoteHTTP _ Nothing ->
   82       pure EmbeddingResult
   83         { erEmbedding = fallbackEmbedding input
   84         , erSource = EmbeddingRemoteFailureLocalFallback
   85         }
   86   where
   87     tryRemote url txt = do
   88       res <- fetchRemoteEmbedding url txt
   89       case res of
   90         Left _ ->
   91           pure EmbeddingResult
   92             { erEmbedding = fallbackEmbedding txt
   93             , erSource = EmbeddingRemoteFailureLocalFallback
   94             }
   95         Right embedding ->
   96           pure EmbeddingResult
   97             { erEmbedding = embedding
   98             , erSource = EmbeddingRemote
   99             }

  101 resolveEmbeddingSelection :: IO EmbeddingSelection
  102 resolveEmbeddingSelection = do
  103   mBackend <- fmap normalizeBackend <$> lookupEnv "QXFX0_EMBEDDING_BACKEND"
  104   mUrl <- lookupEnv "EMBEDDING_API_URL"
  105   pure $
  106     case mBackend of
  107       Just "local" -> EmbeddingSelection EmbeddingBackendLocalDeterministic True Nothing
  108       Just "local-deterministic" -> EmbeddingSelection EmbeddingBackendLocalDeterministic True Nothing
  109       Just "remote" -> EmbeddingSelection EmbeddingBackendRemoteHTTP True mUrl
  110       Just "remote-http" -> EmbeddingSelection EmbeddingBackendRemoteHTTP True mUrl
  111       _ -> EmbeddingSelection EmbeddingBackendLocalDeterministic False Nothing
  112   where
  113     normalizeBackend = map toLower

  115 cachedRemoteHealth :: APIHealthCache -> String -> IO Bool
  116 cachedRemoteHealth cache url = do
  117   now <- getCurrentTime
  118   modifyMVar cache $ \mCache ->
  119     case mCache of
  120       Just (ts, status) | diffUTCTime now ts < 60 -> pure (mCache, status)
  121       _ -> do
  122         status <- performRealHealthCheck url
  123         pure (Just (now, status), status)
  124 
  125 performRealHealthCheck :: String -> IO Bool
  126 performRealHealthCheck url =
  127   either (const False) (const True)
  128     <$> fetchRemoteEmbeddingWithTimeout (HTTPClient.responseTimeoutMicro 2000000) url "healthcheck"
  129 
  130 fetchRemoteEmbedding :: String -> String -> IO (Either Text Embedding)
  131 fetchRemoteEmbedding =
  132   fetchRemoteEmbeddingWithTimeout (HTTPClient.responseTimeoutMicro 5000000)
  133 
  134 fetchRemoteEmbeddingWithTimeout :: ResponseTimeout -> String -> String -> IO (Either Text Embedding)
  135 fetchRemoteEmbeddingWithTimeout timeout url txt = do
  136   requestResult <- try @HttpException (HTTP.parseRequest url)
  137   case requestResult of
  138     Left err -> pure (Left ("invalid embedding url: " <> showText err))
  139     Right baseRequest -> do
  140       let request =
  141             HTTP.setRequestIgnoreStatus $
  142               HTTP.setRequestResponseTimeout timeout $
  143                 HTTP.setRequestBodyJSON (Aeson.object ["model" Aeson..= ("all-minilm" :: String), "prompt" Aeson..= txt]) $
  144                   HTTP.setRequestMethod "POST" baseRequest
  145       responseResult <- try @HttpException (HTTP.httpBS request)
  146       case responseResult of
  147         Left err -> pure (Left ("embedding request failed: " <> showText err))
  148         Right response
  149           | not (successfulStatus response) ->
  150               pure (Left ("unexpected embedding status: " <> showText (HTTP.getResponseStatusCode response)))
  151           | not (jsonContentType response) ->
  152               pure (Left ("unexpected embedding content-type: " <> renderContentTypes response))
  153           | otherwise ->
  154               pure (parseEmbeddingResponse (HTTP.getResponseBody response))
  155 
  156 successfulStatus :: HTTP.Response a -> Bool
  157 successfulStatus response =
  158   let code = HTTP.getResponseStatusCode response
  159    in code >= 200 && code < 300
  160 
  161 jsonContentType :: HTTP.Response a -> Bool
  162 jsonContentType response =
  163   any isJsonHeader (HTTP.getResponseHeader "Content-Type" response)
  164   where
  165     isJsonHeader raw =
  166       let lowered = B8.map toLower raw
  167        in "application/json" `B8.isPrefixOf` B8.dropWhile (== ' ') lowered
  168 
  169 renderContentTypes :: HTTP.Response a -> Text
  170 renderContentTypes response =
  171   case HTTP.getResponseHeader "Content-Type" response of
  172     [] -> "<missing>"
  173     headers -> T.intercalate ", " (map (T.pack . B8.unpack) headers)
  174 
  175 parseEmbeddingResponse :: B8.ByteString -> Either Text Embedding
  176 parseEmbeddingResponse body = do
  177   value <- case Aeson.eitherDecodeStrict' body of
  178     Left err -> Left ("invalid embedding json: " <> showText err)
  179     Right parsed -> Right parsed
  180   case Aeson.parseMaybe (Aeson.withObject "resp" (Aeson..: "embedding")) value of
  181     Just list ->
  182       let vec = V.fromList list
  183        in if V.length vec == embeddingDim
  184             then Right vec
  185             else Left ("unexpected embedding dimension: expected " <> showText embeddingDim <> ", got " <> showText (V.length vec))
  186     Nothing -> Left "invalid embedding format"
  187 
  188 showText :: Show a => a -> Text
  189 showText = T.pack . show
