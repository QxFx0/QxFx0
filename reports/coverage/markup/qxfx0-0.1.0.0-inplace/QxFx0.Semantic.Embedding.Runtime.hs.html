   47 checkApiHealth :: APIHealthCache -> IO Bool
   48 checkApiHealth cache = withFreshEmbeddingManager (`checkApiHealthWithManager` cache)

   50 checkApiHealthWithManager :: Manager -> APIHealthCache -> IO Bool
   51 checkApiHealthWithManager manager cache = ehOperational <$> checkEmbeddingHealthWithManager manager cache

   53 checkEmbeddingHealth :: APIHealthCache -> IO EmbeddingHealth
   54 checkEmbeddingHealth cache = withFreshEmbeddingManager (`checkEmbeddingHealthWithManager` cache)
   55 
   56 checkEmbeddingHealthWithManager :: Manager -> APIHealthCache -> IO EmbeddingHealth
   57 checkEmbeddingHealthWithManager manager cache = do
   58   selection <- resolveEmbeddingSelection
   59   case selection of
   60     EmbeddingSelection
   61       { esBackend = EmbeddingBackendLocalDeterministic
   62       , esExplicit = explicit
   63       , esIssues = issues
   64       } ->
   65       pure EmbeddingHealth
   66         { ehBackend = EmbeddingBackendLocalDeterministic
   67         , ehQuality = EmbeddingQualityHeuristic
   68         , ehExplicit = explicit
   69         , ehOperational = True
   70         , ehStrictReady = True
   71         , ehIssues = issues
   72         }
   73     EmbeddingSelection
   74       { esBackend = EmbeddingBackendRemoteHTTP
   75       , esExplicit = explicit
   76       , esRemoteUrl = mUrl
   77       , esIssues = selectionIssues
   78       } ->
   79         case mUrl of
   80           Nothing ->
   81             pure EmbeddingHealth
   82               { ehBackend = EmbeddingBackendRemoteHTTP
   83               , ehQuality = EmbeddingQualityModeled
   84               , ehExplicit = explicit
   85               , ehOperational = False
   86               , ehStrictReady = False
   87               , ehIssues = selectionIssues
   88               }
   89           Just url -> do
   90             operational <- cachedRemoteHealth manager cache url
   91             let healthIssues = if operational then selectionIssues else selectionIssues <> ["remote_embedding_unhealthy"]
   92             pure EmbeddingHealth
   93               { ehBackend = EmbeddingBackendRemoteHTTP
   94               , ehQuality = EmbeddingQualityModeled
   95               , ehExplicit = explicit
   96               , ehOperational = operational
   97               , ehStrictReady = explicit && operational
   98               , ehIssues = healthIssues
   99               }

  107 textToEmbeddingResult :: Text -> IO EmbeddingResult
  108 textToEmbeddingResult input = withFreshEmbeddingManager (`textToEmbeddingResultWithManager` input)
  109 
  110 textToEmbeddingResultWithManager :: Manager -> Text -> IO EmbeddingResult
  111 textToEmbeddingResultWithManager manager input = do
  112   selection <- resolveEmbeddingSelection
  113   case selection of
  114     EmbeddingSelection
  115       { esBackend = EmbeddingBackendLocalDeterministic
  116       , esExplicit = explicit
  117       } ->
  118       pure EmbeddingResult
  119         { erEmbedding = fallbackEmbedding input
  120         , erSource = if explicit then EmbeddingLocalDeterministic else EmbeddingLocalImplicit
  121         }
  122     EmbeddingSelection
  123       { esBackend = EmbeddingBackendRemoteHTTP
  124       , esExplicit = explicit
  125       , esRemoteUrl = mUrl
  126       , esIssues = issues
  127       } ->
  128         case mUrl of
  129           Just url -> tryRemote explicit url issues input
  130           Nothing -> failRemote explicit (if null issues then ["remote_embedding_url_missing"] else issues) input
  131   where
  132     tryRemote explicit url issues txt = do
  133       res <- fetchRemoteEmbeddingWithManager manager url txt
  134       case res of
  135         Left _ -> failRemote explicit (issues <> ["remote_embedding_unhealthy"]) txt
  136         Right embedding ->
  137           pure EmbeddingResult
  138             { erEmbedding = embedding
  139             , erSource = EmbeddingRemote
  140             }
  141 
  142     failRemote explicit issues txt
  143       | explicit = throwQxFx0 (EmbeddingError (T.intercalate ";" issues))
  144       | otherwise =
  145           pure EmbeddingResult
  146             { erEmbedding = fallbackEmbedding txt
  147             , erSource = EmbeddingRemoteFailureLocalFallback
  148             }

  150 resolveEmbeddingSelection :: IO EmbeddingSelection
  151 resolveEmbeddingSelection = do
  152   mBackend <- fmap normalizeBackend <$> lookupEnv "QXFX0_EMBEDDING_BACKEND"
  153   mUrl <- lookupEnv "EMBEDDING_API_URL"
  154   pure $ case mBackend of
  155     Just "local" ->
  156       EmbeddingSelection
  157         { esBackend = EmbeddingBackendLocalDeterministic
  158         , esExplicit = True
  159         , esRemoteUrl = Nothing
  160         , esIssues = []
  161         }
  162     Just "local-deterministic" ->
  163       EmbeddingSelection
  164         { esBackend = EmbeddingBackendLocalDeterministic
  165         , esExplicit = True
  166         , esRemoteUrl = Nothing
  167         , esIssues = []
  168         }
  169     Just "remote" -> remoteSelection mUrl
  170     Just "remote-http" -> remoteSelection mUrl
  171     _ ->
  172       EmbeddingSelection
  173         { esBackend = EmbeddingBackendLocalDeterministic
  174         , esExplicit = False
  175         , esRemoteUrl = Nothing
  176         , esIssues = []
  177         }
  178   where
  179     normalizeBackend = map toLower
  180     remoteSelection mRemoteUrl =
  181       case mRemoteUrl of
  182         Nothing ->
  183           EmbeddingSelection
  184             { esBackend = EmbeddingBackendRemoteHTTP
  185             , esExplicit = True
  186             , esRemoteUrl = Nothing
  187             , esIssues = ["remote_embedding_url_missing"]
  188             }
  189         Just url ->
  190           case validateEmbeddingUrl url of
  191             Just validated ->
  192               EmbeddingSelection
  193                 { esBackend = EmbeddingBackendRemoteHTTP
  194                 , esExplicit = True
  195                 , esRemoteUrl = Just validated
  196                 , esIssues = []
  197                 }
  198             Nothing ->
  199               EmbeddingSelection
  200                 { esBackend = EmbeddingBackendRemoteHTTP
  201                 , esExplicit = True
  202                 , esRemoteUrl = Nothing
  203                 , esIssues = ["invalid_remote_embedding_url"]
  204                 }
  205 
  206 validateEmbeddingUrl :: String -> Maybe String
  207 validateEmbeddingUrl rawUrl
  208   | null rawUrl = Nothing
  209   | otherwise =
  210       case parseURI rawUrl of
  211         Nothing -> Nothing
  212         Just uri -> do
  213           auth <- uriAuthority uri
  214           let scheme = map toLower (uriScheme uri)
  215               host = map toLower (uriRegName auth)
  216               loopbackHosts = ["127.0.0.1", "localhost", "::1"]
  217           if null host || not (null (uriUserInfo auth))
  218             then Nothing
  219             else case scheme of
  220               "https:" -> Just rawUrl
  221               "http:" | host `elem` loopbackHosts && allowedLoopbackPort (uriPort auth) -> Just rawUrl
  222               _ -> Nothing
  223 
  224 allowedLoopbackPort :: String -> Bool
  225 allowedLoopbackPort port =
  226   case port of
  227     "" -> True
  228     ":8080" -> True
  229     ":11434" -> True
  230     _ -> False

  232 cachedRemoteHealth :: Manager -> APIHealthCache -> String -> IO Bool
  233 cachedRemoteHealth manager cache url = do
  234   now <- getCurrentTime
  235   mCache <- readMVar cache
  236   case mCache of
  237     Just (ts, status) | diffUTCTime now ts < 60 -> pure status
  238     _ -> do
  239       status <- performRealHealthCheck manager url
  240       modifyMVar_ cache $ \current ->
  241         case current of
  242           Just (ts, cachedStatus) | ts > now -> pure current
  243           _ -> pure (Just (now, status))
  244       pure status

  246 performRealHealthCheck :: Manager -> String -> IO Bool
  247 performRealHealthCheck manager url =
  248   either (const False) (const True)
  249     <$> fetchRemoteEmbeddingWithTimeout manager (HTTPClient.responseTimeoutMicro 2000000) url "healthcheck"

  251 fetchRemoteEmbedding :: String -> Text -> IO (Either Text Embedding)
  252 fetchRemoteEmbedding url txt = withFreshEmbeddingManager (\manager -> fetchRemoteEmbeddingWithManager manager url txt)

  254 withFreshEmbeddingManager :: (Manager -> IO a) -> IO a
  255 withFreshEmbeddingManager action = do
  256   manager <- newTlsManager
  257   action manager

  259 fetchRemoteEmbeddingWithManager :: Manager -> String -> Text -> IO (Either Text Embedding)
  260 fetchRemoteEmbeddingWithManager manager =
  261   fetchRemoteEmbeddingWithTimeout manager (HTTPClient.responseTimeoutMicro 5000000)
  262 
  263 fetchRemoteEmbeddingWithTimeout :: Manager -> ResponseTimeout -> String -> Text -> IO (Either Text Embedding)
  264 fetchRemoteEmbeddingWithTimeout manager timeout url txt = do
  265   requestResult <- try @HttpException (HTTP.parseRequest url)
  266   case requestResult of
  267     Left err -> pure (Left ("invalid embedding url: " <> showText err))
  268     Right baseRequest -> do
  269       let request =
  270             HTTP.setRequestIgnoreStatus $
  271               HTTP.setRequestResponseTimeout timeout $
  272                 HTTP.setRequestBodyJSON (Aeson.object ["model" Aeson..= ("all-minilm" :: String), "prompt" Aeson..= txt]) $
  273                   HTTP.setRequestMethod "POST" baseRequest
  274       responseResult <- try @HttpException $ do
  275         withResponse request manager $ \response -> do
  276           if not (successfulStatusCode (statusCode (responseStatus response)))
  277             then pure (Left ("unexpected embedding status: " <> showText (statusCode (responseStatus response))))
  278             else if not (jsonContentTypeSimple (responseHeaders response))
  279               then pure (Left ("unexpected embedding type: " <> renderContentTypesSimple (responseHeaders response)))
  280               else do
  281                 bodyResult <- readResponseBodyLimited embeddingMaxResponseBytes (responseBody response)
  282                 pure (bodyResult >>= parseEmbeddingResponse)
  283       case responseResult of
  284         Left err -> pure (Left ("embedding request failed: " <> showText err))
  285         Right result -> pure result
  286 
  287 embeddingMaxResponseBytes :: Int
  288 embeddingMaxResponseBytes = 1048576
  289 
  290 successfulStatusCode :: Int -> Bool
  291 successfulStatusCode code = code >= 200 && code < 300
  292 
  293 jsonContentTypeSimple :: HTTP.ResponseHeaders -> Bool
  294 jsonContentTypeSimple headers =
  295   any isJsonHeader [ value | (name, value) <- headers, CI.foldedCase name == "content-type" ]
  296   where
  297     isJsonHeader raw =
  298       let lowered = B8.map toLower raw
  299       in "application/json" `B8.isPrefixOf` B8.dropWhile (== ' ') lowered
  300 
  301 renderContentTypesSimple :: HTTP.ResponseHeaders -> Text
  302 renderContentTypesSimple headers =
  303   let values = [ T.pack (B8.unpack value) | (name, value) <- headers, CI.foldedCase name == "content-type" ]
  304   in case values of
  305        [] -> "<missing>"
  306        xs -> T.intercalate ", " xs
  307 
  308 parseEmbeddingResponse :: B8.ByteString -> Either Text Embedding
  309 parseEmbeddingResponse body = do
  310   value <- case Aeson.eitherDecodeStrict' body of
  311     Left err -> Left ("invalid embedding json: " <> showText err)
  312     Right parsed -> Right parsed
  313   case Aeson.parseMaybe (Aeson.withObject "resp" (Aeson..: "embedding")) value of
  314     Just list ->
  315       let vec = V.fromList list
  316        in if V.length vec == embeddingDim
  317             then Right vec
  318             else Left ("unexpected embedding dimension: expected " <> showText embeddingDim <> ", got " <> showText (V.length vec))
  319     Nothing -> Left "invalid embedding format"
  320 
  321 showText :: Show a => a -> Text
  322 showText = T.pack . show
  323 
  324 readResponseBodyLimited :: Int -> BodyReader -> IO (Either Text B8.ByteString)
  325 readResponseBodyLimited maxBytes reader = go 0 []
  326   where
  327     go total chunks = do
  328       chunk <- brRead reader
  329       if BS.null chunk
  330         then pure (Right (B8.concat (reverse chunks)))
  331         else do
  332           let total' = total + BS.length chunk
  333           if total' > maxBytes
  334             then pure (Left ("embedding response too large: " <> showText total'))
  335             else go total' (chunk : chunks)
