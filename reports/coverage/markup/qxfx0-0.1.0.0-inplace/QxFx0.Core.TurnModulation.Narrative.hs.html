   36 modulateRMPWithNarrative :: Maybe Text -> ResponseMeaningPlan -> ResponseMeaningPlan
   37 modulateRMPWithNarrative narrativeFragment meaningPlan =
   38   case narrativeFragment of
   39     Just fragment | not (T.null fragment) ->
   40       meaningPlan
   41         { rmpDepthMode =
   42             if T.length fragment > narrativeDeepCharsThreshold
   43               then DeepDepth
   44               else rmpDepthMode meaningPlan
   45         , rmpTopic =
   46             if T.null (rmpTopic meaningPlan)
   47               then T.take 40 fragment
   48               else rmpTopic meaningPlan
   49         }
   50     _ -> meaningPlan

   53 modulateRCPWithFlash overridesAll contentPlan =

   58 narrativeFamilyHint :: ConsciousnessNarrative -> Maybe CanonicalMoveFamily
   59 narrativeFamilyHint narrative =
   60   let skill = cnSkillInPlay narrative
   61       skillTokens = tokenizeKeywordText skill
   62       conflict = cnConflict narrative
   63    in if containsKeywordPhrase skillTokens narrativeSkillSilenceName
   64         then Just CMAnchor
   65         else
   66           if not (T.null conflict)
   67             then Just CMReflect
   68             else
   69               if containsKeywordPhrase skillTokens narrativeSkillListenName
   70                 then Just CMContact
   71                 else
   72                   if containsKeywordPhrase skillTokens narrativeSkillPresenceName
   73                     then Just CMContact
   74                     else
   75                       if containsKeywordPhrase skillTokens narrativeSkillResonateName
   76                         then Just CMContact
   77                         else
   78                           if containsKeywordPhrase skillTokens narrativeSkillTeachName
   79                             then Just CMClarify
   80                             else
   81                               if containsKeywordPhrase skillTokens narrativeSkillAnalyzeName
   82                                 then Just CMDeepen
   83                                 else
   84                                   if containsKeywordPhrase skillTokens narrativeSkillResistName
   85                                     then Just CMGround
   86                                     else
   87                                       if containsKeywordPhrase skillTokens narrativeSkillHoldName
   88                                         then Just CMAnchor
   89                                         else Nothing

   91 intuitionFamilyHint :: Double -> Maybe CanonicalMoveFamily
   92 intuitionFamilyHint posterior
   93   | posterior > narrativeIntuitionFamilyHintThreshold = Just CMDeepen
   94   | otherwise = Nothing
