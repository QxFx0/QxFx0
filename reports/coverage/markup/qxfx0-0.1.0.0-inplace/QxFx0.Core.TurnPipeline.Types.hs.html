   68 data TurnInput = TurnInput
   69   { tiStartTime :: !UTCTime
   70   , tiEmbedding :: !Embedding
   71   , tiEmbeddingSource :: !EmbeddingSource
   72   , tiEmbeddingQuality :: !EmbeddingQuality
   73   , tiEmbSimilarity :: !Double
   74   , tiAtomSet :: !AtomSet
   75   , tiNewTrace :: !AtomTrace
   76   , tiNextUserState :: !UserState
   77   , tiRecommendedFamily :: !CanonicalMoveFamily
   78   , tiFrame :: !InputPropositionFrame
   79   , tiNixStatus :: !NixGuardStatus
   80   , tiNixAvailable :: !Bool
   81   , tiIsNixBlocked :: !Bool
   82   , tiConceptToCheck :: !Text
   83   , tiBestTopic :: !Text
   84   , tiMetrics :: !TurnMetrics
   85   , tiConatusEnergy :: !ConatusEnergy
   86     -- ^ Phase 6 (M6): runtime Conatus energy carried from
   87     --   'QxFx0.Core.TurnPipeline.Effects.psConatusEnergy'.
   88     --   Single source of truth across the turn: route-stage
   89     --   ('buildLocalRecoveryPlan') and finalize-stage trace
   90     --   ('buildTurnProjection') previously each recomputed
   91     --   'computeConatusEnergy' independently; now both reads
   92     --   funnel through this field for the pre-turn state.
   93   , tiBlanketViolationCount :: !Int
   94     -- ^ Phase 6 (M6): violation count from
   95     --   'QxFx0.Core.TurnPipeline.Effects.psBlanketViolationCount',
   96     --   used by 'buildLocalRecoveryPlan' to construct the
   97     --   @"blanket_violations=N"@ evidence line.
   98   , tiConatusGateFired :: !Bool
   99     -- ^ Phase 6 addendum (M6.1): precomputed Conatus gate flag
  100     --   from 'psConatusGateFired'.  Single source of truth for
  101     --   both recovery and salience decisions.
  102   , tiField :: !Field
  103     -- ^ Per-turn 'QxFx0.Self.Field.Field' carried from
  104     --   'QxFx0.Core.TurnPipeline.Effects.psField'.  All five
  105     --   components are populated from runtime signals
  106     --   (Resonance from atom-trace, Atmosphere from Ego +
  107     --   legitimacy, Consolidation from narrative-success window,
  108     --   Counterfactual from candidate-family entropy, Confidence
  109     --   derived from the other four).  Single source of truth
  110     --   across the turn so routing ('routeFamily'), salience
  111     --   computation, and finalize-stage trace share the same
  112     --   Field instead of each constructing 'emptyField'
  113     --   independently.
  114   , tiFieldHeuristics :: !FieldHeuristics
  115     -- ^ Phase 6.7: the heuristic parameters used to build
  116     --   'tiField'.  Mirrors 'psFieldHeuristics' from
  117     --   'PrepareStatic' so downstream stages can read the
  118     --   same record without reconstructing defaults.
  119   , tiSenseVector :: !SenseVector
  120     -- ^ Canonical sense-layer bridge extracted from the same semantic
  121     --   interpretation that produced 'tiFrame'. Used to constrain
  122     --   response planning toward bounded adjacent meaning-continuation.
  123   , tiDialogueThread :: !DialogueThread
  124   , tiDialogueCommitmentLedger :: !DialogueCommitmentLedger
  125   , tiDialoguePhase :: !DialoguePhase
  126   , tiTruthContractStatus :: !TruthContractStatus
  127   , tiEssence :: !Essence
  128     -- ^ Phase 9: the pre-turn essence carrier, populated by
  129     --   'buildPrepareEffectPlan' from 'ssEssence'.  Single source
  130     --   of truth for the turn's essence-layer reads.
  131   }

  201 tpPrincipledModePair :: TurnPlan -> Maybe (PressureSignal, PrincipledMode)
  202 tpPrincipledModePair tp =
  203   case (rdPressure (tpRouting tp), rdPrincipledMode (tpRouting tp)) of
  204     (Just p, Just pm) -> Just (p, pm)
  205     _ -> Nothing
