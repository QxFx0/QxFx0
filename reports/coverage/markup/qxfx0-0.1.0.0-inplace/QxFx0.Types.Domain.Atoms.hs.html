   53 data AtomTag
   54   = Searching !Text
   55   | Exhaustion !Text
   56   | Verification !Text
   57   | Doubt !Text
   58   | NeedContact !Text
   59   | NeedMeaning !Text
   60   | AgencyLost !Double
   61   | AgencyFound !Double
   62   | Anchoring !Text
   63   | Contradiction !Text !Text
   64   | CustomAtom !Text !Text
   65   | AffectiveAtom !Text !Double
   66   deriving stock (Eq, Ord, Show, Read, Generic)
   67   deriving anyclass (NFData)
   68 
   69 instance ToJSON AtomTag where
   70   toJSON (Searching t) = object ["tag" .= ("Searching" :: Text), "payload" .= t]
   71   toJSON (Exhaustion t) = object ["tag" .= ("Exhaustion" :: Text), "payload" .= t]
   72   toJSON (Verification t) = object ["tag" .= ("Verification" :: Text), "payload" .= t]
   73   toJSON (Doubt t) = object ["tag" .= ("Doubt" :: Text), "payload" .= t]
   74   toJSON (NeedContact t) = object ["tag" .= ("NeedContact" :: Text), "payload" .= t]
   75   toJSON (NeedMeaning t) = object ["tag" .= ("NeedMeaning" :: Text), "payload" .= t]
   76   toJSON (AgencyLost d) = object ["tag" .= ("AgencyLost" :: Text), "payload" .= d]
   77   toJSON (AgencyFound d) = object ["tag" .= ("AgencyFound" :: Text), "payload" .= d]
   78   toJSON (Anchoring t) = object ["tag" .= ("Anchoring" :: Text), "payload" .= t]
   79   toJSON (Contradiction a b) = object ["tag" .= ("Contradiction" :: Text), "left" .= a, "right" .= b]
   80   toJSON (CustomAtom t v) = object ["tag" .= ("CustomAtom" :: Text), "label" .= t, "payload" .= v]
   81   toJSON (AffectiveAtom t d) = object ["tag" .= ("AffectiveAtom" :: Text), "label" .= t, "valence" .= d]
   82 
   83 instance FromJSON AtomTag where
   84   parseJSON = withObject "AtomTag" $ \o -> do
   85     tag <- o .: "tag"
   86     case tag :: Text of
   87       "Searching" -> Searching <$> o .: "payload"
   88       "Exhaustion" -> Exhaustion <$> o .: "payload"
   89       "Verification" -> Verification <$> o .: "payload"
   90       "Doubt" -> Doubt <$> o .: "payload"
   91       "NeedContact" -> NeedContact <$> o .: "payload"
   92       "NeedMeaning" -> NeedMeaning <$> o .: "payload"
   93       "AgencyLost" -> AgencyLost <$> o .: "payload"
   94       "AgencyFound" -> AgencyFound <$> o .: "payload"
   95       "Anchoring" -> Anchoring <$> o .: "payload"
   96       "Contradiction" -> Contradiction <$> o .: "left" <*> o .: "right"
   97       "CustomAtom" -> CustomAtom <$> o .: "label" <*> o .: "payload"
   98       "AffectiveAtom" -> AffectiveAtom <$> o .: "label" <*> o .: "valence"
   99       _ -> fail $ "unknown AtomTag: " ++ T.unpack tag

  252 data LexemeForm = LexemeForm
  253   { lfSurface :: !Text
  254   , lfLemma :: !Text
  255   , lfPOS :: !Text
  256   , lfCase :: !LexemeCase
  257   , lfNumber :: !LexemeNumber
  258   , lfTier :: !SourceTier
  259   , lfQuality :: !Double
  260   } deriving stock (Eq, Show, Generic)
  261     deriving anyclass (NFData)
  262 
  263 instance ToJSON LexemeForm where
  264   toJSON (LexemeForm s l p c n t q) = object
  265     [ "surface"  .= s
  266     , "lemma"    .= l
  267     , "pos"      .= p
  268     , "case"     .= c
  269     , "number"   .= n
  270     , "tier"     .= t
  271     , "quality"  .= q
  272     ]
  273 
  274 instance FromJSON LexemeForm where
  275   parseJSON v = parseObject v <|> parseArray v
  276     where
  277       parseObject = withObject "LexemeForm" $ \o -> LexemeForm
  278         <$> o .: "surface"
  279         <*> o .: "lemma"
  280         <*> o .: "pos"
  281         <*> o .: "case"
  282         <*> o .: "number"
  283         <*> o .: "tier"
  284         <*> o .: "quality"
  285       parseArray = withArray "LexemeForm" $ \arr -> do
  286         let a = V.toList arr
  287         case a of
  288           [s, l, p, c, n, t, q] -> LexemeForm
  289             <$> parseJSON s
  290             <*> parseJSON l
  291             <*> parseJSON p
  292             <*> parseJSON c
  293             <*> parseJSON n
  294             <*> parseJSON t
  295             <*> parseJSON q
  296           _ -> fail $ "Parser: LexemeForm compact array must have exactly 7 elements: [surface, lemma, pos, case, number, tier, quality]"
