  261 classifyLegitimacyOutcome :: LegitimacyStatus -> LegitimacyReason -> WarrantedMoveMode -> ShadowStatus -> ShadowDivergenceSeverity -> LegitimacyOutcome
  262 classifyLegitimacyOutcome status reason warrantedMode shadowStatus shadowSeverity =
  263   LegitimacyOutcome
  264     { loDisposition = disposition
  265     , loStatus = status
  266     , loReason = reason
  267     , loWarrantedMode = warrantedMode
  268     }
  269   where
  270     disposition
  271       | warrantedMode == NeverWarranted = DispositionDeny
  272       | shadowStatus == ShadowDiverged && shadowSeverity /= ShadowSeverityAdvisory = DispositionRepair
  273       | status == LegitimacyRecovery = DispositionRepair
  274       | reason == ReasonShadowDivergence = DispositionRepair
  275       | reason == ReasonShadowUnavailable = DispositionAdvisory
  276       | reason == ReasonLowParserConfidence = DispositionAdvisory
  277       | otherwise = DispositionPermit
