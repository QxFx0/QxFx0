   25 data AgdaVerificationResult
   26   = AgdaPass
   27   | AgdaTypeCheckFailed !Text
   28   | AgdaNotAvailable !Text
   29   | AgdaSnapshotMismatch ![(Text, Text)]
   30   deriving stock (Eq, Show)
   31 
   32 agdaTypeCheck :: IO AgdaVerificationResult
   33 agdaTypeCheck = do
   34   paths <- resolveResourcePaths
   35   let agdaFile = rpAgdaSpec paths
   36       agdaDir = takeDirectory agdaFile
   37       agdaModule = takeFileName agdaFile
   38   timeoutMicros <- resolveAgdaTimeoutMicros
   39   result <- catchIO
   40     (do mProcessResult <- timeout timeoutMicros
   41           (readCreateProcessWithExitCode (proc "agda" [agdaModule]) { cwd = Just agdaDir } "")
   42         case mProcessResult of
   43           Nothing ->
   44             return
   45               (AgdaTypeCheckFailed
   46                 ( "agda typecheck timed out after "
   47                     <> T.pack (show (timeoutMicros `div` 1000))
   48                     <> "ms"
   49                 ))
   50           Just (exitCode, _stdout, stderr) ->
   51             case exitCode of
   52               ExitSuccess -> return AgdaPass
   53               ExitFailure code
   54                 | code == 127 -> return (AgdaNotAvailable "agda not found in PATH")
   55                 | otherwise -> return (AgdaTypeCheckFailed $ T.pack stderr))
   56     (\e -> return (AgdaNotAvailable $ "agda exception: " <> T.pack (show e)))
   57   return result
   58 
   59 resolveAgdaTimeoutMicros :: IO Int
   60 resolveAgdaTimeoutMicros = do
   61   mConfigured <- lookupEnv "QXFX0_AGDA_TIMEOUT_MS"
   62   let timeoutMs =
   63         case mConfigured >>= readMaybe of
   64           Just ms | ms > 0 -> ms
   65           _ -> agdaTypecheckTimeoutMsDefault
   66   pure (timeoutMs * 1000)

   68 verifyAgainstSnapshot :: FilePath -> IO AgdaVerificationResult
   69 verifyAgainstSnapshot snapshotPath = do
   70   contents <- catchIO (TIO.readFile snapshotPath)
   71     (\e -> return $ "ERROR: " <> T.pack (show e))
   72   if T.isPrefixOf "ERROR:" contents
   73     then return $ AgdaNotAvailable $ "cannot read snapshot: " <> contents
   74     else do
   75       let rows = filter (not . T.null) $ T.lines contents
   76           mismatches = catMaybes $ map checkRow rows
   77       if null mismatches
   78         then return AgdaPass
   79         else return $ AgdaSnapshotMismatch mismatches
   80   where
   81     checkRow :: Text -> Maybe (Text, Text)
   82     checkRow line =
   83       let fields = T.splitOn "\t" line
   84       in case fields of
   85         ("family":_) -> Nothing
   86         (famStr:forceStr:thirdStr:fourthStr:rest) ->
   87           (case parseFamily famStr of
   88             Left err -> Just (famStr, err)
   89             Right fam ->
   90               let expectedForce = T.pack (show (forceForFamily fam))
   91                   expectedClause = T.pack (show (clauseFormForIF (forceForFamily fam)))
   92                   expectedLayer = T.pack (show (layerForFamily fam))
   93                   expectedWarranted = T.pack (show (warrantedForFamily fam))
   94                   layerNames = ["ContentLayer", "MetaLayer", "ContactLayer"]
   95               in case (thirdStr, fourthStr, rest) of
   96                    (clauseStr, layerStr, warrantedStr:_)
   97                      | clauseStr `elem` ["Declarative", "Interrogative", "Hortative", "Imperative"] ->
   98                          if forceStr == expectedForce
   99                               && clauseStr == expectedClause
  100                               && layerStr == expectedLayer
  101                               && warrantedStr == expectedWarranted
  102                            then Nothing
  103                            else Just
  104                              ( famStr
  105                              , "expected(" <> T.intercalate "/" [expectedForce, expectedClause, expectedLayer, expectedWarranted]
  106                                  <> ") got(" <> T.intercalate "/" [forceStr, clauseStr, layerStr, warrantedStr] <> ")"
  107                              )
  108                    (layerStr, warrantedStr, _)
  109                      | layerStr `elem` layerNames ->
  110                          if forceStr == expectedForce
  111                               && layerStr == expectedLayer
  112                               && warrantedStr == expectedWarranted
  113                            then Nothing
  114                            else Just
  115                              ( famStr
  116                              , "expected(" <> T.intercalate "/" [expectedForce, expectedLayer, expectedWarranted]
  117                                  <> ") got(" <> T.intercalate "/" [forceStr, layerStr, warrantedStr] <> ")"
  118                              )
  119                    _ -> Just (famStr, "unrecognized snapshot row format: " <> line))
  120         [famStr] -> Just (famStr, "malformed snapshot row (too few fields): " <> line)
  121         (_:_)    -> Just ("<unknown>", "malformed snapshot row (too few fields): " <> line)
  122         [] -> Nothing
  123 
  124     parseFamily :: Text -> Either Text CanonicalMoveFamily
  125     parseFamily txt = case readMaybe (T.unpack txt) of
  126       Just fam -> Right fam
  127       Nothing  -> Left ("unrecognized_family: " <> txt)
  128 
  129 verifyR5WithAgda :: IO AgdaVerificationResult
  130 verifyR5WithAgda = do
  131   paths <- resolveResourcePaths
  132   let snapshotPath = rpAgdaSnapshot paths
  133   typeResult <- agdaTypeCheck
  134   case typeResult of
  135     AgdaPass -> verifyAgainstSnapshot snapshotPath
  136     other -> return other
