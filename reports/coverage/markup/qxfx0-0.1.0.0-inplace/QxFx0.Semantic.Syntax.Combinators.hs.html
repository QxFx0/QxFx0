  209 -- | Build NP from raw text (already inflected) with explicit agreement metadata.
  210 -- Use only when the form is pre-computed and no paradigm lookup is possible.
  211 mkNPRaw :: Text -> NounCase -> Number -> Gender -> NP
  212 mkNPRaw form case_ number gender =
  213   NP { npLemma = form, npCase = case_, npNumber = number, npGender = gender
  214      , npAPs = [], npRelClause = Nothing }

  216 -- | Build VP from raw text (already inflected) with explicit agreement metadata.
  217 mkVPRaw :: Text -> Int -> Number -> Maybe Gender -> Tense -> VP

  222 -- | Frozen multivord VP — no paradigm lookup possible. Grammar fields are dummies.

  225   VP { vpLemma = form, vpPerson = 3, vpNumber = Sg, vpGender = Nothing
  226      , vpTense = Present, vpAdv = [], vpPP = [], vpSubord = Nothing }

  228 -- | Build AP from raw text (already inflected) with explicit agreement metadata.
  229 mkAPRaw :: Text -> Gender -> Number -> NounCase -> AP
  230 mkAPRaw form gender number case_ =
  231   AP { apLemma = form, apGender = gender, apNumber = number, apCase = case_ }

  233 -- | Minimal Russian adjective inflection for nominative singular and plural.
  234 -- Used as a safety net when the dictionary lookup fails.
  235 inflectAdjectiveLemma :: Text -> Gender -> Number -> NounCase -> Text
  236 inflectAdjectiveLemma lemma gender number case_
  237   | case_ == Nom && number == Sg =
  238       case gender of
  239         Masc -> lemma
  240         Femn -> adjustEnding lemma [("ый", "ая"), ("ий", "яя"), ("ой", "ая")]
  241         Neut -> adjustEnding lemma [("ый", "ое"), ("ий", "ее"), ("ой", "ое")]
  242   | case_ == Nom && number == Pl =
  243       adjustEnding lemma [("ый", "ые"), ("ий", "ие"), ("ой", "ые")]
  244   | otherwise = lemma
  245   where
  246     adjustEnding w replacements =
  247       case filter (\(suf, _) -> T.isSuffixOf suf w) replacements of
  248         ((suf, repl):_) -> T.dropEnd (T.length suf) w <> repl
  249         []             -> w

  251 mkNP :: RuntimeParadigms -> Text -> NounCase -> Number -> Either MorphError NP
  252 mkNP rp lemma case_ number = do
  253   _ <- validatePOS rp lemma "Noun"
  254   form <- lookupFormOrError rp lemma case_ number
  255   g <- genderOrError rp lemma
  256   pure NP { npLemma = form, npCase = case_, npNumber = number, npGender = g
  257           , npAPs = [], npRelClause = Nothing }

  259 mkVP :: RuntimeParadigms -> Text -> Int -> Number -> Maybe Gender -> Tense
  260      -> Either MorphError VP
  261 mkVP rp lemma person number gender tense = do
  262   _ <- validatePOS rp lemma "Verb"
  263   form <- verbFormOrError rp lemma person number gender tense
  264   pure VP
  265     { vpLemma = form
  266     , vpPerson = person
  267     , vpNumber = number
  268     , vpGender = gender
  269     , vpTense = tense
  270     , vpAdv = []
  271     , vpPP = []
  272     , vpSubord = Nothing
  273     }

  275 mkS :: NP -> VP -> Either MorphError S
  276 mkS subj pred = do
  277   -- Subject-verb number agreement
  278   when (npNumber subj /= vpNumber pred)
  279     (Left (NumberMismatch (npLemma subj) (vpLemma pred)))
  280   -- For past tense, gender must agree too
  281   case vpTense pred of
  282     Past | npNumber subj == Sg ->
  283       case vpGender pred of
  284         Just g | g /= npGender subj ->
  285           Left (GenderMismatch (npLemma subj) (vpLemma pred))
  286         _ -> Right ()
  287     _ -> Right ()
  288   pure SimpleS { sSubject = subj, sPredicate = pred, sTopicFront = Nothing, sQuestion = False, sModal = Nothing, sFill = Nothing, sQuant = Nothing }
  289 
  290 mkAP :: RuntimeParadigms -> Text -> Gender -> Number -> NounCase
  291      -> Either MorphError AP
  292 mkAP rp lemma gender number case_ = do
  293   _ <- validatePOS rp lemma "Adjective"
  294   form <- adjFormOrError rp lemma gender number case_
  295   pure AP { apLemma = form, apGender = gender, apNumber = number, apCase = case_ }

  297 addAP :: NP -> AP -> Either MorphError NP
  298 addAP np ap = do
  299   when (npGender np /= apGender ap)
  300     (Left (GenderMismatch (npLemma np) (apLemma ap)))
  301   when (npNumber np /= apNumber ap)
  302     (Left (NumberMismatch (npLemma np) (apLemma ap)))
  303   when (npCase np /= apCase ap)
  304     (Left (CaseMismatch (npLemma np) (apLemma ap)))
  305   pure np { npAPs = ap : npAPs np }

  307 addAdv :: VP -> Adv -> Either MorphError VP
  308 addAdv vp adv = pure vp { vpAdv = adv : vpAdv vp }

  310 mkAdv :: RuntimeParadigms -> Text -> Either MorphError Adv
  311 mkAdv rp lemma =
  312   case lookupAdverbForm rp lemma of
  313     Just form -> Right (Adv form)
  314     Nothing   -> Left (MissingForm lemma "adverb")
  315 
  316 mkExpression :: RuntimeParadigms -> Text -> Either MorphError Text
  317 mkExpression rp lemma =
  318   case lookupExpression rp lemma of
  319     Just form -> Right form
  320     Nothing   -> Left (MissingForm lemma "expression")

  322 addAdvToS :: S -> Adv -> S
  323 addAdvToS s adv = case s of
  324   SimpleS subj pred mFront isQ mModal mFill mQuant ->
  325     SimpleS subj (pred { vpAdv = adv : vpAdv pred }) mFront isQ mModal mFill mQuant
  326   EllipticalS pred mFront isQ mModal mFill mQuant ->
  327     EllipticalS (pred { vpAdv = adv : vpAdv pred }) mFront isQ mModal mFill mQuant
  328   CoordS' (CoordS s1 coord s2) ->
  329     CoordS' (CoordS (addAdvToS s1 adv) coord (addAdvToS s2 adv))
  330 
  331 -- | Build Adv from raw text (no paradigm lookup).
  332 mkAdvRaw :: Text -> Adv
  333 mkAdvRaw t = Adv t

  335 addModalParticle :: Text -> S -> S
  336 addModalParticle particle s = case s of
  337   SimpleS subj pred mFront isQ _ mFill mQuant ->
  338     SimpleS subj pred mFront isQ (Just particle) mFill mQuant
  339   EllipticalS pred mFront isQ _ mFill mQuant ->
  340     EllipticalS pred mFront isQ (Just particle) mFill mQuant
  341   CoordS' _ -> s
  342 
  343 addFiller :: Text -> S -> S
  344 addFiller filler s = case s of
  345   SimpleS subj pred mFront isQ mModal _ mQuant ->
  346     SimpleS subj pred mFront isQ mModal (Just filler) mQuant
  347   EllipticalS pred mFront isQ mModal _ mQuant ->
  348     EllipticalS pred mFront isQ mModal (Just filler) mQuant
  349   CoordS' _ -> s
  350 
  351 addQuantifierToS :: Text -> S -> S
  352 addQuantifierToS quant s = case s of
  353   SimpleS subj pred mFront isQ mModal mFill _ ->
  354     SimpleS subj pred mFront isQ mModal mFill (Just quant)
  355   EllipticalS pred mFront isQ mModal mFill _ ->
  356     EllipticalS pred mFront isQ mModal mFill (Just quant)
  357   CoordS' _ -> s

  497 -- | Turn a sentence into a question.

  499 mkQuestion s IFAsk = case s of
  500   SimpleS {} -> s { sQuestion = True }
  501   EllipticalS {} -> s { eQuestion = True }
  502   CoordS' _ -> s

  505 mkComparative :: RuntimeParadigms -> Text -> Text -> Text -> CmpDir -> Either MorphError S
  506 mkComparative rp subj obj axis dir = do
  507   subjNP <- mkNP rp subj Nom Sg
  508   objNP <- case dir of
  509     CmpMore  -> mkNP rp obj Gen Sg
  510     CmpLess  -> mkNP rp obj Gen Sg
  511     CmpEqual -> mkNP rp obj Gen Sg
  512   let compText = case dir of
  513         CmpMore  -> "важнее"
  514         CmpLess  -> "менее важно"
  515         CmpEqual -> "так же важно как"
  516   let subjNP' = subjNP { npAPs = [AP compText (npGender subjNP) (npNumber subjNP) (npCase subjNP)] }
  517   objNP' <- addAP objNP (AP compText (npGender objNP) (npNumber objNP) (npCase objNP))
  518   vp <- mkVP rp "быть" 3 Sg Nothing Present
  519   s <- mkS subjNP vp
  520   mkTopicFront objNP' s

  526 linearizeS :: S -> Text
  527 linearizeS s = case s of
  528   SimpleS { sSubject = subj, sPredicate = pred, sTopicFront = mFront, sQuestion = isQ, sModal = mModal, sFill = mFill, sQuant = mQuant } ->
  529     let core = T.unwords [linearizeNP subj, linearizeVPWith (npGender subj) (npNumber subj) pred]
  530         punct = if isQ then "?" else "."
  531         withModal = case mModal of
  532                       Nothing -> core
  533                       Just p  -> insertAfterFirstWord p core
  534         withFiller = case mFill of
  535                        Nothing -> withModal
  536                        Just f  -> f <> ", " <> withModal
  537         withQuant = case mQuant of
  538                        Nothing -> withFiller
  539                        Just q  -> q <> " " <> withFiller
  540     in case mFront of
  541          Nothing  -> withQuant <> punct
  542          Just np  -> linearizeNP np <> " — " <> withQuant <> punct
  543   EllipticalS { ePredicate = pred, eTopicFront = mFront, eQuestion = isQ, eModal = mModal, eFill = mFill, eQuant = mQuant } ->
  544     let core = linearizeVP pred
  545         punct = if isQ then "?" else "."
  546         withModal = case mModal of
  547                       Nothing -> core
  548                       Just p  -> insertAfterFirstWord p core
  549         withFiller = case mFill of
  550                        Nothing -> withModal
  551                        Just f  -> f <> ", " <> withModal
  552         withQuant = case mQuant of
  553                        Nothing -> withFiller
  554                        Just q  -> q <> " " <> withFiller
  555     in case mFront of
  556          Nothing  -> withQuant <> punct
  557          Just np  -> linearizeNP np <> " — " <> withQuant <> punct
  558   CoordS' (CoordS s1 coord s2) ->
  559     let s1text = fromMaybe (linearizeS s1) (T.stripSuffix "." (linearizeS s1))
  560     in s1text <> ", " <> coordWord coord <> " " <> linearizeS s2
  561 
  562 insertAfterFirstWord :: Text -> Text -> Text
  563 insertAfterFirstWord w txt =
  564   case T.words txt of
  565     []     -> w
  566     (x:xs) -> T.unwords (x : w : xs)
  567 
  568 coordWord :: Coord -> Text
  569 coordWord = \case
  570   CoordI  -> "и"
  571   CoordA  -> "а"
  572   CoordNo -> "но"
  573   CoordIli -> "или"
  574   CoordDash -> "—"
  575 
  576 subordWord :: Subord -> Text
  577 subordWord = \case
  578   SubChto       -> "что"
  579   SubKogda      -> "когда"
  580   SubPotomuChto -> "потому что"
  581   SubEsli       -> "если"
  582   SubKotory     -> "который"
  583   SubGde        -> "где"
  584 
  585 kotoryForm :: Gender -> Number -> Text
  586 kotoryForm Masc Sg  = "который"
  587 kotoryForm Femn Sg  = "которая"
  588 kotoryForm Neut Sg  = "которое"
  589 kotoryForm _    Pl  = "которые"
  590 
  591 linearizeNP :: NP -> Text
  592 linearizeNP np =
  593   let adjs = T.unwords (map linearizeAP (reverse (npAPs np)))
  594       headN = npLemma np
  595       rel = case npRelClause np of
  596               Nothing -> ""
  597               Just clause -> ", " <> kotoryForm (npGender np) (npNumber np) <> " " <> linearizeS clause
  598       base = if T.null adjs then headN else adjs <> " " <> headN
  599   in if T.null rel then base else base <> rel
  600 
  601 linearizeVP :: VP -> Text
  602 linearizeVP vp = linearizeVPWith Masc Sg vp
  603 
  604 linearizeVPWith :: Gender -> Number -> VP -> Text
  605 linearizeVPWith g n vp =
  606   let verb = vpLemma vp
  607       adverbs = T.unwords (map linearizeAdv (reverse (vpAdv vp)))
  608       pps = T.unwords (map linearizePP (reverse (vpPP vp)))
  609       subord = case vpSubord vp of
  610                  Nothing -> ""
  611                  Just (SubordClause conj clause) ->
  612                    let pron = case conj of
  613                                 SubKotory -> kotoryForm g n
  614                                 _         -> subordWord conj
  615                    in ", " <> pron <> " " <> linearizeSClause clause
  616       parts = filter (not . T.null) [adverbs, verb, pps]
  617       core = T.unwords parts
  618   in if T.null subord then core else core <> subord
  619 
  620 linearizeSClause :: S -> Text
  621 linearizeSClause s = fromMaybe (linearizeS s) (T.stripSuffix "." (linearizeS s))
  622 
  623 linearizeAP :: AP -> Text
  624 linearizeAP = apLemma
  625 
  626 linearizeAdv :: Adv -> Text
  627 linearizeAdv (Adv t) = t
  628 
  629 linearizePP :: PP -> Text
  630 linearizePP pp =
  631   let prep = ppPreposition pp
  632       npText = linearizeNP (ppNP pp)
  633   in if T.null prep then npText else prep <> " " <> npText

  639 validatePOS :: RuntimeParadigms -> Text -> Text -> Either MorphError ()
  640 validatePOS rp lemma expected =
  641   case lemmaPos rp lemma of
  642     Nothing -> Left (MissingParadigm lemma)
  643     Just actual | actual == expected -> Right ()
  644     Just actual -> Left (InvalidPOS lemma actual)
  645 
  646 lookupFormOrError :: RuntimeParadigms -> Text -> NounCase -> Number
  647                   -> Either MorphError Text
  648 lookupFormOrError rp lemma case_ number =
  649   case lookupNounForm rp lemma case_ number of
  650     Nothing -> Left (MissingForm lemma (T.pack (show case_ <> show number)))
  651     Just form -> Right form
  652 
  653 genderOrError :: RuntimeParadigms -> Text -> Either MorphError Gender
  654 genderOrError rp lemma =
  655   case lemmaGender rp lemma of
  656     Nothing -> Left (MissingParadigm lemma)
  657     Just "masc"  -> Right Masc
  658     Just "femn"  -> Right Femn
  659     Just "neut"  -> Right Neut
  660     Just g       -> Left (InvalidPOS lemma g)
  661 
  662 verbFormOrError :: RuntimeParadigms -> Text -> Int -> Number -> Maybe Gender -> Tense
  663                 -> Either MorphError Text
  664 verbFormOrError rp lemma person number gender tense =
  665   let form = case tense of
  666         Past -> case number of
  667           Sg -> case gender of
  668             Just Masc -> PastSgMasc
  669             Just Femn  -> PastSgFem
  670             Just Neut -> PastSgNeut
  671             Nothing   -> PastSgMasc  -- default
  672           Pl -> PastPl
  673         Present -> case (person, number) of
  674           (1, Sg) -> Pres1Sg; (2, Sg) -> Pres2Sg; (3, Sg) -> Pres3Sg
  675           (1, Pl) -> Pres1Pl; (2, Pl) -> Pres2Pl; (3, Pl) -> Pres3Pl
  676           _       -> Pres3Sg
  677         Future -> case (person, number) of
  678           (1, Sg) -> Fut1Sg; (2, Sg) -> Fut2Sg; (3, Sg) -> Fut3Sg
  679           (1, Pl) -> Fut1Pl; (2, Pl) -> Fut2Pl; (3, Pl) -> Fut3Pl
  680           _       -> Fut3Sg
  681   in case lookupVerbForm rp lemma form of
  682        Nothing -> Left (MissingForm lemma (T.pack (show form)))
  683        Just w  -> Right w
  684 
  685 adjFormOrError :: RuntimeParadigms -> Text -> Gender -> Number -> NounCase
  686                -> Either MorphError Text
  687 adjFormOrError rp lemma gender number case_ =
  688   let form = AdjFull case_ number (Just gender)
  689   in case lookupAdjectiveForm rp lemma form of
  690        Nothing -> Left (MissingForm lemma (T.pack (show form)))
  691        Just w  -> Right w
