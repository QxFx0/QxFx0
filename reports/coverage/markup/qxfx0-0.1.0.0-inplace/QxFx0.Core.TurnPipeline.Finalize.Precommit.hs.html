   46 planFinalizePrecommit :: SystemState -> TurnInput -> TurnSignals -> TurnPlan -> TurnArtifacts -> FinalizePrecommitPlan
   47 planFinalizePrecommit systemState _turnInput _turnSignals turnPlan turnArtifacts =
   48   let decision = taDecision turnArtifacts
   49       outcomeFamily = tdFamily decision
   50       outcomeVerdict = mkVerdict outcomeFamily
   51       consecutiveReflect =
   52         if outcomeFamily == CMReflect
   53           then ssConsecutiveReflect systemState + 1
   54           else 0
   55       transitionWon =
   56         case taSurfaceProv turnArtifacts of
   57           FromRecovery -> False
   58           _ -> tpStrategyFamily turnPlan == Just (rmpFamily (tpRmpAfterLegit turnPlan))
   59       meaningGraphBase =
   60         recordTransition
   61           (tpFromMs turnPlan)
   62           (tpToMs turnPlan)
   63           (tpRenderStrategy turnPlan)
   64           transitionWon
   65           (ssMeaningGraph systemState)
   66       static =
   67         FinalizeStatic
   68           { fsOutcomeFamily = outcomeFamily
   69           , fsOutcomeVerdict = outcomeVerdict
   70           , fsConsecReflect = consecutiveReflect
   71           , fsTransitionWon = transitionWon
   72           , fsMeaningGraphBase = meaningGraphBase
   73           }
   74    in FinalizePrecommitPlan
   75         { fppStatic = static
   76         , fppCurrentTimeRequest = FinalizeReqCurrentTime
   77         , fppIntrospectionRequest = FinalizeReqSemanticIntrospectionEnv
   78         }

   80 resolveFinalizePrecommit :: PipelineIO -> FinalizePrecommitPlan -> IO FinalizePrecommitResults
   81 resolveFinalizePrecommit pipelineIO plan = do
   82   (currentTime, (semanticIntrospectionEnabled, warnMorphologyFallbackEnabled)) <-
   83     concurrently
   84       (resolveCurrentTime pipelineIO plan)
   85       ( concurrently
   86           (resolveIntrospectionEnv pipelineIO plan)
   87           (resolveWarnMorphologyFallbackEnv pipelineIO plan)
   88       )
   89   pure
   90     FinalizePrecommitResults
   91       { fprCurrentTime = currentTime
   92       , fprRuntimeMode = pipelineRuntimeModeText (pipelineRuntimeMode pipelineIO)
   93       , fprShadowPolicy = shadowPolicyText (pipelineShadowPolicy pipelineIO)
   94       , fprLocalRecoveryPolicy = localRecoveryPolicyText (pipelineLocalRecoveryPolicy pipelineIO)
   95       , fprSemanticIntrospectionEnabled = semanticIntrospectionEnabled
   96       , fprWarnMorphologyFallbackEnabled = warnMorphologyFallbackEnabled
   97       }

   99 buildFinalizePrecommit :: (Text -> Seq Text -> Seq Text) -> SystemState -> TurnInput -> TurnSignals -> TurnPlan -> TurnArtifacts -> FinalizePrecommitPlan -> FinalizePrecommitResults -> FinalizePrecommitBundle
  100 buildFinalizePrecommit updateHistory systemState turnInput turnSignals turnPlan turnArtifacts precommitPlan precommitResults =
  101   let static = fppStatic precommitPlan
  102       (newDreamState, newMeaningGraph, rewireEventsCount) =
  103         applyDreamDynamics
  104           (fprCurrentTime precommitResults)
  105           systemState
  106           turnInput
  107           turnSignals
  108           turnPlan
  109           turnArtifacts
  110           (fsMeaningGraphBase static)
  111       nextSystemState =
  112         buildNextSystemState
  113           updateHistory
  114           systemState
  115           turnInput
  116           turnSignals
  117           turnPlan
  118           turnArtifacts
  119           newDreamState
  120           newMeaningGraph
  121           (fsOutcomeFamily static)
  122           (fsOutcomeVerdict static)
  123           (fsConsecReflect static)
  124       projection =
  125         buildTurnProjection
  126           (fprRuntimeMode precommitResults)
  127           (fprShadowPolicy precommitResults)
  128           (fprLocalRecoveryPolicy precommitResults)
  129           (fprSemanticIntrospectionEnabled precommitResults)
  130           (fprWarnMorphologyFallbackEnabled precommitResults)
  131           nextSystemState
  132           turnInput
  133           turnSignals
  134           turnPlan
  135           turnArtifacts
  136       wantIntrospection =
  137         fprSemanticIntrospectionEnabled precommitResults
  138           || ssOutputMode systemState == SemanticIntrospectionOutput
  139       (outputWithIntrospection, finalSafetyStatus) =
  140         buildFinalOutput wantIntrospection systemState (taGuardSurface turnArtifacts) nextSystemState
  141    in FinalizePrecommitBundle
  142         { fpbNextSs = nextSystemState
  143         , fpbProjection = projection
  144         , fpbOutput = outputWithIntrospection
  145         , fpbFinalSafetyStatus = finalSafetyStatus
  146         , fpbOutcomeFamily = fsOutcomeFamily static
  147         , fpbDecision = taDecision turnArtifacts
  148         , fpbRewireEventsCount = rewireEventsCount
  149         }

  151 resolveCurrentTime :: PipelineIO -> FinalizePrecommitPlan -> IO UTCTime
  152 resolveCurrentTime pipelineIO plan =
  153   case finalizePrecommitRequestToTurnEffect (fppCurrentTimeRequest plan) of
  154     Just _ -> do
  155       result <- resolveTurnEffect pipelineIO TurnReqCurrentTime
  156       case result of
  157         TurnResCurrentTime currentTime -> pure currentTime
  158         _ -> throwQxFx0 (PersistenceError "current time effect returned unexpected result")
  159     Nothing ->
  160       throwQxFx0 (PersistenceError "missing current time request")
  161 
  162 resolveIntrospectionEnv :: PipelineIO -> FinalizePrecommitPlan -> IO Bool
  163 resolveIntrospectionEnv pipelineIO plan =
  164   case finalizePrecommitRequestToTurnEffect (fppIntrospectionRequest plan) of
  165     Just _ -> do
  166       result <- resolveTurnEffect pipelineIO TurnReqSemanticIntrospectionEnv
  167       case result of
  168         TurnResSemanticIntrospectionEnv hasIntrospectionEnv -> pure hasIntrospectionEnv
  169         _ -> pure False
  170     Nothing ->
  171       pure False
  172 
  173 resolveWarnMorphologyFallbackEnv :: PipelineIO -> FinalizePrecommitPlan -> IO Bool
  174 resolveWarnMorphologyFallbackEnv pipelineIO plan =
  175   case finalizePrecommitRequestToTurnEffect (fppIntrospectionRequest plan) of
  176     Just _ -> do
  177       result <- resolveTurnEffect pipelineIO (TurnReqReadEnv "QXFX0_WARN_MORPHOLOGY_FALLBACK")
  178       case result of
  179         TurnResReadEnv (Just "1") -> pure True
  180         TurnResReadEnv _ -> pure False
  181         _ -> pure False
  182     Nothing ->
  183       pure False
  184 
  185 finalizePrecommitRequestToTurnEffect :: FinalizePrecommitRequest -> Maybe TurnEffectRequest
  186 finalizePrecommitRequestToTurnEffect request =
  187   case request of
  188     FinalizeReqCurrentTime -> Just TurnReqCurrentTime
  189     FinalizeReqSemanticIntrospectionEnv -> Just TurnReqSemanticIntrospectionEnv
