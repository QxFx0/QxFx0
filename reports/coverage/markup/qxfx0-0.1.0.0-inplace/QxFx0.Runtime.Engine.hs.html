   39 runTurn :: RuntimeContext -> SystemState -> Text -> Text -> IO (SystemState, Text)
   40 runTurn ctx ss input sessionId
   41   | T.length input > maxInputLength = return (ss, "\1054\1096\1080\1073\1082\1072: \1090\1077\1082\1089\1090 \1089\1083\1080\1096\1082\1086\1084 \1076\1083\1080\1085\1085\1099\1081.")
   42   | otherwise = withRuntimeSession ctx sessionId $ do
   43       turnResult <- try (runTurnBody ctx ss input sessionId) :: IO (Either QxFx0Exception (SystemState, Text))
   44       case turnResult of
   45         Left (EssenceRupture detail) -> pure (ss, "Turn blocked: essence rupture [" <> detail <> "]")
   46         Left (EmbeddingError detail) -> pure (ss, "Turn blocked: embedding backend [" <> detail <> "]")
   47         Left err -> throwQxFx0 err
   48         Right result -> pure result

   50 runTurnBody :: RuntimeContext -> SystemState -> Text -> Text -> IO (SystemState, Text)
   51 runTurnBody ctx ss input sessionId = do
   52   let pio = toPipelineIO ctx
   53   reqId <- resolveRequestId pio
   54   prepared <- prepareTurn pio ss input sessionId reqId
   55   planned <- planTurn pio ss prepared
   56   rendered <- renderTurn pio ss planned
   57   tr <- finalizeTurn pio ss sessionId reqId rendered
   58   pure (trNextSs tr, trOutput tr)
   59 
   60 resolveRequestId :: PipelineIO -> IO Text
   61 resolveRequestId pio = do
   62   requestIdResult <- resolveTurnEffect pio TurnReqRequestId
   63   case requestIdResult of
   64     TurnResRequestId rid -> pure rid
   65     _ -> throwQxFx0 (RuntimeInitError "request id effect returned unexpected result")

   67 runTurnInSession :: Session -> Text -> IO (Session, Text)
   68 runTurnInSession session text = do
   69   readiness <- checkSessionReadiness session
   70   runtimeMode <- resolveRuntimeMode
   71   let strictMode = isStrictRuntimeMode runtimeMode
   72   case evaluateBootstrapReadiness runtimeMode readiness of
   73     Left failure ->
   74       pure
   75         ( session { sessReadinessMode = readiness }
   76         , renderTurnGateFailure failure
   77         )
   78     Right _ ->
   79       case readiness of
   80         Degraded failed -> do
   81           hPutStrLn stderr $ "[degraded] optional components unavailable: " ++ show failed
   82           continueWithHealthCheck runtimeMode strictMode session readiness
   83         _ ->
   84           continueWithHealthCheck runtimeMode strictMode session readiness
   85   where
   86     continueWithHealthCheck runtimeMode strictMode s readiness = do
   87       let ss = sessSystemState s
   88           runtime = sessRuntime s
   89           sid = sessSessionId s
   90       if strictMode
   91         then do
   92           health <- checkHealth runtime
   93           case evaluateStrictHealth runtimeMode health of
   94             Left failure ->
   95               pure
   96                 ( s { sessReadinessMode = readiness }
   97                 , renderTurnGateFailure failure
   98                 )
   99             Right _ ->
  100               continueTurn s readiness ss runtime sid
  101         else continueTurn s readiness ss runtime sid
  102 
  103     continueTurn s readiness ss runtime sid = do
  104       turnResult <- try (runTurn runtime ss text sid) :: IO (Either QxFx0Exception (SystemState, Text))
  105       case turnResult of
  106         Left (AgdaGateError detail) ->
  107           pure
  108             ( s { sessReadinessMode = readiness }
  109             , "Turn blocked: strict runtime requires Agda verification [" <> detail <> "]"
  110             )
  111         Left (EmbeddingError detail) ->
  112           pure
  113             ( s { sessReadinessMode = readiness }
  114             , "Turn blocked: embedding backend [" <> detail <> "]"
  115             )
  116         Left (EssenceRupture detail) ->
  117           pure
  118             ( s { sessReadinessMode = readiness }
  119             , "Turn blocked: essence rupture [" <> detail <> "]"
  120             )
  121         Left err ->
  122           throwQxFx0 err
  123         Right (nextSs, response) -> do
  124           let !session' = s { sessSystemState = nextSs, sessReadinessMode = readiness }
  125           pure (session', response)
  126 
  127 loop :: Session -> IO ()
  128 loop session = do
  129   T.putStr $ "\n[" <> promptMode (sessOutputMode session) <> "] [QxFx0] > "
  130   eof <- hIsEOF stdin
  131   if eof
  132     then pure ()
  133     else do
  134       inputResult <- try T.getLine :: IO (Either IOException Text)
  135       case inputResult of
  136         Left _ -> pure ()
  137         Right input ->
  138           let boundedInput = T.take maxInputLength input
  139           in case T.strip boundedInput of
  140             ":quit" -> do
  141               T.putStrLn "State saved. Bye."
  142               exitSuccess
  143             ":help" -> printHelp >> loop session
  144             ":dialogue" ->
  145               T.putStrLn "Output mode: DIALOGUE" >> loop (setOutputMode DialogueMode session)
  146             ":semantic" ->
  147               T.putStrLn "Output mode: SEMANTIC" >> loop (setOutputMode SemanticIntrospectionMode session)
  148             ":state" -> printStateSummary session >> loop session
  149             text
  150               | not (T.null text) -> do
  151                   (session', safeResponse) <- runTurnInSession session text
  152                   T.putStrLn safeResponse
  153                   loop session'
  154             _ -> loop session
  155   where
  156     promptMode DialogueMode = "DIALOGUE"
  157     promptMode SemanticIntrospectionMode = "SEMANTIC"
  158     setOutputMode mode s = s
  159       { sessOutputMode = mode
  160       , sessSystemState = (sessSystemState s) { ssOutputMode = runtimeToDialogueMode mode }
  161       }
