    1 {-# LANGUAGE OverloadedStrings #-}

    3 module QxFx0.Runtime.Paths
    4   ( resolveDbPath
    5   , resolveSessionId
    6   ) where
    7 
    8 import Data.Maybe (fromMaybe)
    9 import Data.Text (Text)
   10 import qualified Data.Text as T
   11 import Data.Time.Clock (getCurrentTime)
   12 import Data.Time.Format (defaultTimeLocale, formatTime)
   13 import System.Environment (lookupEnv)
   14 import System.FilePath ((</>))
   15 
   16 resolveSessionId :: IO Text
   17 resolveSessionId = do
   18   mSessionId <- lookupEnv "QXFX0_SESSION_ID"
   19   case fmap T.strip (T.pack <$> mSessionId) of
   20     Just sid | not (T.null sid) -> pure sid
   21     _ -> do
   22       now <- getCurrentTime
   23       pure ("session-" <> T.pack (formatTime defaultTimeLocale "%Y%m%d%H%M%S%q" now))
   24 
   25 resolveDbPath :: IO FilePath
   26 resolveDbPath = do
   27   mDbPath <- lookupEnv "QXFX0_DB"
   28   case mDbPath of
   29     Just dbPath -> pure dbPath
   30     Nothing -> do
   31       mStateDir <- lookupEnv "QXFX0_STATE_DIR"
   32       stateDir <- case mStateDir of
   33         Just dir -> pure dir
   34         Nothing -> do
   35           mXdgStateHome <- lookupEnv "XDG_STATE_HOME"
   36           mHome <- lookupEnv "HOME"
   37           pure $ case mXdgStateHome of
   38             Just xdgStateHome -> xdgStateHome </> "qxfx0"
   39             Nothing -> fromMaybe "." ((</> ".local/state/qxfx0") <$> mHome)
   40       pure (stateDir </> "qxfx0.db")
