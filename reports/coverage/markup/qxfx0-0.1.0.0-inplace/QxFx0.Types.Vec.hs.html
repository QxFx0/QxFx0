   20 data CoreVec = CoreVec
   21   { cvPresence      :: !Double
   22   , cvDirectiveness :: !Double
   23   , cvSteadiness    :: !Double
   24   , cvDepth         :: !Double
   25   , cvAutonomy      :: !Double
   26   } deriving stock (Show, Read, Eq, Generic)
   27     deriving anyclass (NFData)
   28 
   29 instance ToJSON CoreVec where
   30   toJSON v = object
   31     [ "presence"      .= cvPresence v
   32     , "directiveness" .= cvDirectiveness v
   33     , "steadiness"    .= cvSteadiness v
   34     , "depth"         .= cvDepth v
   35     , "autonomy"      .= cvAutonomy v
   36     ]
   37 
   38 instance FromJSON CoreVec where
   39   parseJSON = withObject "CoreVec" $ \v ->
   40     CoreVec <$> v .: "presence"
   41             <*> v .: "directiveness"
   42             <*> v .: "steadiness"
   43             <*> v .: "depth"
   44             <*> v .: "autonomy"
   45 
   46 zeroVec :: CoreVec
   47 zeroVec = CoreVec 0 0 0 0 0
   48 
   49 vecAdd :: CoreVec -> CoreVec -> CoreVec
   50 vecAdd a b = CoreVec
   51   { cvPresence = cvPresence a + cvPresence b
   52   , cvDirectiveness = cvDirectiveness a + cvDirectiveness b
   53   , cvSteadiness = cvSteadiness a + cvSteadiness b
   54   , cvDepth = cvDepth a + cvDepth b
   55   , cvAutonomy = cvAutonomy a + cvAutonomy b
   56   }
   57 
   58 vecSub :: CoreVec -> CoreVec -> CoreVec
   59 vecSub a b = CoreVec
   60   { cvPresence = cvPresence a - cvPresence b
   61   , cvDirectiveness = cvDirectiveness a - cvDirectiveness b
   62   , cvSteadiness = cvSteadiness a - cvSteadiness b
   63   , cvDepth = cvDepth a - cvDepth b
   64   , cvAutonomy = cvAutonomy a - cvAutonomy b
   65   }
   66 
   67 vecScale :: Double -> CoreVec -> CoreVec
   68 vecScale s v = CoreVec
   69   { cvPresence = s * cvPresence v
   70   , cvDirectiveness = s * cvDirectiveness v
   71   , cvSteadiness = s * cvSteadiness v
   72   , cvDepth = s * cvDepth v
   73   , cvAutonomy = s * cvAutonomy v
   74   }
   75 
   76 vecNorm :: CoreVec -> Double
   77 vecNorm v = sqrt
   78   ( cvPresence v * cvPresence v
   79   + cvDirectiveness v * cvDirectiveness v
   80   + cvSteadiness v * cvSteadiness v
   81   + cvDepth v * cvDepth v
   82   + cvAutonomy v * cvAutonomy v
   83   )
   84 
   85 clampVecNorm :: Double -> CoreVec -> CoreVec
   86 clampVecNorm maxNorm v
   87   | normV < 1e-9 = zeroVec
   88   | normV <= maxNorm = v
   89   | otherwise = vecScale (maxNorm / normV) v
   90   where
   91     normV = vecNorm v
