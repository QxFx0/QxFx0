   23 data DepthMode
   24   = SurfaceDepth
   25   | MediumDepth
   26   | DeepDepth
   27   deriving stock (Eq, Ord, Show, Read, Generic)
   28   deriving anyclass (NFData)
   29 
   30 depthModeText :: DepthMode -> Text
   31 depthModeText SurfaceDepth = "surface"
   32 depthModeText MediumDepth = "medium"
   33 depthModeText DeepDepth = "deep"
   34 
   35 parseDepthModeText :: Text -> Maybe DepthMode
   36 parseDepthModeText txt = case T.toLower (T.strip txt) of
   37   "surface" -> Just SurfaceDepth
   38   "medium" -> Just MediumDepth
   39   "deep" -> Just DeepDepth
   40   _ -> Nothing
   41 
   42 instance ToJSON DepthMode where
   43   toJSON = String . depthModeText
   44 
   45 instance FromJSON DepthMode where
   46   parseJSON = withText "DepthMode" $ \txt ->
   47     case parseDepthModeText txt of
   48       Just mode -> pure mode
   49       Nothing -> fail ("unknown DepthMode: " ++ T.unpack txt)
