  142 -- | Local helper: derive the canonical pre-turn Salience verdict
  143 -- from a 'TurnInput'.  Used by both 'buildNextSystemState' (for
  144 -- 'dsLastSalienceBias' on the persisted state) and
  145 -- 'buildTurnProjection' (for the @trcSalience*@ trace fields).
  146 -- Computed identically in both call sites; centralised here so
  147 -- they cannot drift.
  148 turnInputSalience :: TurnInput -> Salience
  149 turnInputSalience ti = salienceFromConatusEnergy (tiConatusEnergy ti) (tiField ti)

  151 -- | WP1 (contour closure): validate the reconciled 'Plan' against a
  152 -- pre-turn committed essence.  Only runs when the pre-turn state
  153 -- is already 'EssenceCommitted' — the commitment itself is law-driven
  154 -- in 'buildNextSystemState' and does not depend on a feature flag.
  155 computeEssenceValidation
  156   :: TurnInput
  157   -> TurnPlan
  158   -> Essence          -- ^ post-'buildNextSystemState' essence
  159   -> Maybe CommitmentTrigger
  160   -> Either EssenceViolation ()
  161 computeEssenceValidation ti tp nextEssence mTrigger =
  162   case (mTrigger, nextEssence) of
  163     (Just trigger, EssenceUncommitted _) ->
  164       Left (ViolationRefusedCommitment trigger)
  165     (_, EssenceCommitted _ commitment)
  166       | Just delib <- tpDeliberation tp
  167       -> case validatePlan commitment (delibReconciled delib) of
  168            Right _ -> Right ()
  169            Left v  -> Left v
  170     _ -> Right ()

  172 selfTuningMutationRecords
  173   :: Int
  174   -> Double
  175   -> SignalComponents
  176   -> SalienceWeights
  177   -> FieldHeuristics
  178   -> SalienceWeights
  179   -> FieldHeuristics
  180   -> [AdaptiveMutationRecord]
  181 selfTuningMutationRecords turn signal comps newWeights newHeuristics oldWeights oldHeuristics =
  182   salienceRecord <> fieldRecord
  183   where
  184     signalEvidence =
  185       [ "signal=" <> textShow signal
  186       , "conatus=" <> textShow (scConatusTrend comps)
  187       , "uncertainty=" <> textShow (scUncertaintyTrend comps)
  188       , "loop=" <> textShow (scLoopRisk comps)
  189       , "branch=" <> textShow (scBranchHealthTrend comps)
  190       ]
  191     strength = if abs signal >= 0.15 then EvidenceStrong else EvidenceWeak
  192     decision = if abs signal >= 0.15 then AdaptiveAccepted else AdaptiveObserved
  193     salienceDelta = salienceWeightsMaxDelta oldWeights newWeights
  194     fieldDelta = fieldHeuristicsMaxDelta oldHeuristics newHeuristics
  195     salienceRecord =
  196       if oldWeights == newWeights
  197         then []
  198         else
  199           [ AdaptiveMutationRecord
  200               { amrTurnId = turn
  201               , amrKind = MutSalienceWeights
  202               , amrCause = "finalize:bounded_salience_adaptation"
  203               , amrEvidence = signalEvidence
  204               , amrEvidenceStrength = strength
  205               , amrConfidence = min 1.0 (abs signal)
  206               , amrBoundedDelta = Just salienceDelta
  207               , amrDecision = decision
  208               }
  209           ]
  210     fieldRecord =
  211       if oldHeuristics == newHeuristics
  212         then []
  213         else
  214           [ AdaptiveMutationRecord
  215               { amrTurnId = turn
  216               , amrKind = MutFieldHeuristics
  217               , amrCause = "finalize:bounded_field_heuristics_adaptation"
  218               , amrEvidence = signalEvidence
  219               , amrEvidenceStrength = strength
  220               , amrConfidence = min 1.0 (abs signal)
  221               , amrBoundedDelta = Just fieldDelta
  222               , amrDecision = decision
  223               }
  224           ]

  262 appendCalibrationEntries :: Int -> CalibrationLog -> [Maybe CalibrationProposal] -> CalibrationLog
  263 appendCalibrationEntries turn (CalibrationLog existing) proposals =
  264   let concrete = [ proposal | Just proposal <- proposals ]
  265       nextBase = maybe 1 ((+ 1) . unCalibrationId . ceId) (safeLastEntry existing)
  266       prevId = currentCalibrationVersion (CalibrationLog existing)
  267       step (entries, nextId, previous) proposal =
  268         let (entry, CalibrationId nextRaw) = acceptProposal (CalibrationId nextId) proposal turn previous
  269         in (entries ++ [entry], nextRaw, Just (ceId entry))
  270       (newEntries, _, _) = foldl step ([], nextBase, prevId) concrete
  271   in CalibrationLog (existing ++ newEntries)
  272   where
  273     safeLastEntry [] = Nothing
  274     safeLastEntry xs = foldl (\_ entry -> Just entry) Nothing xs

  276 calibrationMutationRecord :: Int -> CalibrationDecision -> Double -> SignalComponents -> AdaptiveMutationRecord
  277 calibrationMutationRecord turn decision signal comps = AdaptiveMutationRecord
  278   { amrTurnId = turn
  279   , amrKind = MutCalibration
  280   , amrCause = "finalize:calibration_signal"
  281   , amrEvidence =
  282       [ "decision=" <> renderCalibrationDecision decision
  283       , "signal=" <> textShow signal
  284       , "conatus=" <> textShow (scConatusTrend comps)
  285       , "uncertainty=" <> textShow (scUncertaintyTrend comps)
  286       , "loop=" <> textShow (scLoopRisk comps)
  287       , "branch=" <> textShow (scBranchHealthTrend comps)
  288       ]
  289   , amrEvidenceStrength = if abs signal >= 0.15 then EvidenceStrong else EvidenceWeak
  290   , amrConfidence = min 1.0 (abs signal)
  291   , amrBoundedDelta = Just 1.0
  292   , amrDecision = adaptiveDecisionForCalibration decision
  293   }

  295 adaptiveDecisionForCalibration :: CalibrationDecision -> AdaptiveDecision

  297   case decision of
  298     CdApplySignal -> AdaptiveAccepted

  311 salienceWeightsMaxDelta :: SalienceWeights -> SalienceWeights -> Double
  312 salienceWeightsMaxDelta old new = maximumOrZero
  313   [ abs (weightResonance new - weightResonance old)
  314   , abs (weightAtmosphere new - weightAtmosphere old)
  315   , abs (weightConsolidation new - weightConsolidation old)
  316   , abs (weightCounterfactual new - weightCounterfactual old)
  317   , abs (weightFieldConfidence new - weightFieldConfidence old)
  318   ]
  319 
  320 fieldHeuristicsMaxDelta :: FieldHeuristics -> FieldHeuristics -> Double
  321 fieldHeuristicsMaxDelta old new = maximumOrZero
  322   [ abs (fhDefaultNarrativeRate new - fhDefaultNarrativeRate old)
  323   , abs (fhTopicStabilityBoost new - fhTopicStabilityBoost old)
  324   , abs (fhHolisticStreakBoostRate new - fhHolisticStreakBoostRate old)
  325   , abs (fhHolisticStreakBoostCap new - fhHolisticStreakBoostCap old)
  326   , abs (fhLegitimacyMidpoint new - fhLegitimacyMidpoint old)
  327   , abs (fhLegitimacyBonusScale new - fhLegitimacyBonusScale old)
  328   ]
  329 
  330 maximumOrZero :: [Double] -> Double
  331 maximumOrZero = foldr max 0.0

  333 buildNextSystemState :: (Text -> Seq Text -> Seq Text) -> SystemState -> TurnInput -> TurnSignals -> TurnPlan -> TurnArtifacts -> DreamState -> MeaningGraph -> CanonicalMoveFamily -> R5Verdict -> Int -> (SystemState, Maybe CommitmentTrigger)
  334 buildNextSystemState updateHistory ss ti ts tp ta newDreamState newMeaningGraph outcomeFamily outcomeVerdict consecReflect =
  335   let !newHumanHistory = updateHistory (ipfRawText (tiFrame ti)) (ssHistory ss)
  336       updatedNixCache = updateStateNixCache (tiConceptToCheck ti) (tiNixStatus ti) (obsNixCache (ssObservability ss))
  337       turnSalience = turnInputSalience ti
  338       newHolisticStreak = if isHolisticFamily outcomeFamily then ssHolisticStreak ss + 1 else 0
  339       narrativeSuccess = maybe False (not . T.null) (tsNarrativeFragment ts)
  340       newNarrativeSuccess = take 5 (narrativeSuccess : ssRecentNarrativeSuccess ss)
  341       -- WP1 (contour closure): law-driven commitment.
  342       -- 'shouldCommit' is always evaluated; no feature flag.
  343       -- The trigger (if any) is exposed for downstream validation.
  344       (nextEssence, commitmentTrigger) =
  345         case tiEssence ti of
  346           EssenceUncommitted trajectory ->
  347             let trajectory' =
  348                   witness
  349                     defaultEssenceModulation
  350                     (ssTurnCount ss + 1)
  351                     (tiConatusEnergy ti)
  352                     (tiField ti)
  353                     (fromMaybe defaultDeliberation (tpDeliberation tp))
  354                     trajectory
  355             in case shouldCommit defaultEssenceModulation trajectory' of
  356                  Nothing      -> (EssenceUncommitted trajectory', Nothing)
  357                  Just trigger ->
  358                    ( EssenceCommitted
  359                        trajectory'
  360                        (commit (ssTurnCount ss + 1) trigger trajectory')
  361                    , Just trigger
  362                    )
  363           EssenceCommitted trajectory commitment ->
  364              -- Sticky: committed essences are never reverted. We still
  365              -- ingest a witness so etAngstLevel/etConatusFloor track
  366              -- post-commit deliberation for diagnostics.
  367              let trajectory' =
  368                    witness
  369                      defaultEssenceModulation
  370                      (ssTurnCount ss + 1)
  371                      (tiConatusEnergy ti)
  372                      (tiField ti)
  373                      (fromMaybe defaultDeliberation (tpDeliberation tp))
  374                      trajectory
  375               in (EssenceCommitted trajectory' commitment, Nothing)
  376       -- Phase 7: bounded calibration signal + rooted tree maintenance.
  377       -- Compute signal from conatus trend, uncertainty, loop risk,
  378       -- and branch health.  Only adapt when committed.
  379       (adaptedWeights, adaptedHeuristics, nextTree, calibrationSnapshots, nextCalibrationLog, adaptiveRecords) =
  380         case nextEssence of
  381           EssenceCommitted _ commitment ->
  382             let -- Extract counterfactual from the pre-turn Field
  383                 counterfactual =
  384                   case tiField ti of
  385                     field -> unCounterfactual (fieldCounterfactual field)
  386                 -- Proxy loop risk: blocked-concept count / 10 (window)
  387                 loopCount = length (ssBlockedConcepts ss)
  388                 windowSize = 10
  389                 (calSignal, sigComps) =
  390                   computeCalibrationSignal
  391                     (ssLearningNeedState ss)
  392                     counterfactual
  393                     loopCount
  394                     windowSize
  395                     (ssKnowledgeTree ss)
  396                 signal = unCalibrationSignal calSignal
  397                 (shouldApplySignal, calibrationDecision) =
  398                   applyCalibrationGated defaultSignalPipelineConfig calSignal (ssCalibrationSnapshots ss)
  399                 calibrationSnapshot = CalibrationSnapshot
  400                   { csTimestamp = tiStartTime ti
  401                   , csRunId = tmSessionId (tiMetrics ti) <> ":" <> tmRequestId (tiMetrics ti)
  402                   , csComponents = sigComps
  403                   , csSignal = signal
  404                   , csDecision = calibrationDecision
  405                   }
  406                 -- Update knowledge tree root from commitment
  407                 treeWithRoot =
  408                   (ssKnowledgeTree ss)
  409                     { ktRootMode = renderEssenceMode (ecMode commitment)
  410                     , ktRootTrigger = renderCommitmentTrigger (ecTrigger commitment)
  411                     }
  412                 -- Structural maintenance: prune stale fruits and branches
  413                 currentTurn = ssTurnCount ss + 1
  414                 (treePruned, _prunedBranches) =
  415                   pruneBranches currentTurn (-0.5) 3 treeWithRoot
  416                 (treeClean, _prunedFruits) =
  417                   pruneFruits currentTurn treePruned
  418                 promotionRule = textShow outcomeFamily
  419                 (treePromoted, promotedFromQuarantine, _rejectedFromQuarantine) =
  420                   promoteFromQuarantine currentTurn 2 promotionRule treeClean
  421                 weights1 = if shouldApplySignal then adaptSalienceWeights signal (ssSalienceWeights ss) else ssSalienceWeights ss
  422                 heuristics1 = if shouldApplySignal then adaptFieldHeuristics signal (ssFieldHeuristics ss) else ssFieldHeuristics ss
  423                 calibrationLog1 =
  424                   if shouldApplySignal
  425                     then appendCalibrationEntries currentTurn (ssCalibrationLog ss)
  426                       [ if weights1 /= ssSalienceWeights ss then Just (ProposalSalienceWeights weights1) else Nothing
  427                       , if heuristics1 /= ssFieldHeuristics ss then Just (ProposalFieldHeuristics heuristics1) else Nothing
  428                       ]
  429                     else ssCalibrationLog ss
  430                 tuningRecords = calibrationMutationRecord currentTurn calibrationDecision signal sigComps :
  431                   selfTuningMutationRecords currentTurn signal sigComps weights1 heuristics1 (ssSalienceWeights ss) (ssFieldHeuristics ss)
  432                 treeRecords = knowledgeMaintenanceMutationRecords currentTurn _prunedBranches _prunedFruits
  433                   <> quarantinePromotionMutationRecords currentTurn promotedFromQuarantine
  434             in ( weights1
  435                , heuristics1
  436                , treePromoted
  437                , take 100 (calibrationSnapshot : ssCalibrationSnapshots ss)
  438                , calibrationLog1
  439                , tuningRecords <> treeRecords
  440                )
  441           _ -> (ssSalienceWeights ss, ssFieldHeuristics ss, ssKnowledgeTree ss, ssCalibrationSnapshots ss, ssCalibrationLog ss, [])
  442       -- WP6.1: endogenous learning diagnostic drive with separate learning pressure.
  443       bestTopic = tiBestTopic ti
  444       isTopicUnknown = not (T.null bestTopic)
  445                          && not (hasKnownMorphologyForm (ssMorphology ss) bestTopic)
  446                          && not (isTermKnownInKnowledgeTree bestTopic nextTree)
  447       currentGraftedCount = ktGraftedCount nextTree
  448       newLearningNeedState =
  449           detectLearningNeedWithPressure
  450            defaultLearningPressureConfig
  451            (tiConatusEnergy ti)
  452            (tiField ti)
  453            0 -- repairCount: historical recovery tracking deferred to Phase 7
  454            (length (ssBlockedConcepts ss)) -- proxy: blocked concepts indicate substrate gaps
  455            (ssTurnCount ss + 1)
  456             (ssLearningNeedState ss)
  457             isTopicUnknown
  458             currentGraftedCount
  459       executedOutcome = taExecutedOutcome ta
  460       baseNext = ss
  461         { ssDialogue = (ssDialogue ss)
  462           { dsHistory = newHumanHistory
  463           , dsActiveScene = tpActiveScene tp
  464           , dsLastFamily = outcomeFamily
  465           , dsLastTopic = tiBestTopic ti
  466           , dsUserState = tiNextUserState ti
  467           , dsLastForce = r5Force outcomeVerdict
  468           , dsLastLayer = r5Layer outcomeVerdict
  469           , dsRecentFamilies = take recentFamiliesLimit (outcomeFamily : ssRecentFamilies ss)
  470           , dsRawInputHistory = appendHistoryBounded rawInputHistoryLimit (ssRawInputHistory ss) (ipfRawText (tiFrame ti))
  471           , dsTurnCount = ssTurnCount ss + 1
  472           , dsConsecutiveReflect = consecReflect
  473           , dsLastEmbedding = Just (tiEmbedding ti)
  474           , dsLastSalienceBias = salienceHolisticBias turnSalience
  475           , dsHolisticStreak = newHolisticStreak
  476           , dsRecentNarrativeSuccess = newNarrativeSuccess
  477           }
  478       , ssIdentity = (ssIdentity ss)
  479           { idsEgo = tpNewEgo tp
  480           , idsOrbitalMemory = tpUpdatedOrbital tp
  481           , idsLastGuardReport = Just (tpGuardReport tp)
  482           }
  483       , ssSemantic = (ssSemantic ss)
  484           { semTrace = tiNewTrace ti
  485           , semMeaningGraph = newMeaningGraph
  486           , semDreamState = newDreamState
  487           , semIntuitionState = Just (tsIntuitionState ts)
  488           , semKernelPulse = (ssKernelPulse ss) { kpActive = True, kpLastUpdate = ssTurnCount ss + 1 }
  489           , semBlockedConcepts =
  490               case tiNixStatus ti of
  491                 Blocked reason -> retainBlockedConcepts reason (ssBlockedConcepts ss)
  492                 _ -> ssBlockedConcepts ss
  493           , semIntuitConfidence = tsIntuitPosterior ts
  494           , semSemanticAnchor = tpSemanticAnchor tp
  495           , semLastTurnDecision = Just (taDecision ta)
  496           }
  497       , ssObservability = (ssObservability ss)
  498           { obsNixCache = updatedNixCache
  499           , obsTelemetry = (obsTelemetry (ssObservability ss))
  500               { lrtFamily = outcomeFamily
  501               , lrtTopic = tiBestTopic ti
  502               , lrtApiHealthy = tsApiHealthy ts
  503               , lrtGuardStatus = tdGuardStatus (taDecision ta)
  504               , lrtSurfaceRoute = taSurfaceProv ta
  505               }
  506           , obsEmbeddingApiHealthy = tsApiHealthy ts
  507           , obsLastLegitimacyScore = tpLegitScore tp
  508           }
  509       , ssEssence = nextEssence
  510       , ssSalienceWeights = adaptedWeights
  511       , ssFieldHeuristics = adaptedHeuristics
  512       , ssShadowVetoState = tpShadowVetoState tp
  513       , ssProvisionalAtoms =
  514           let currentTurn = ssTurnCount ss
  515               canonicalSet = tiAtomSet ti
  516               observed = foldr
  517                 (\atom acc -> observeNovelAtom (maTag atom) currentTurn acc)
  518                 (ssProvisionalAtoms ss)
  519                 (asAtoms canonicalSet)
  520               decayed = decayProvisionalAtoms currentTurn observed
  521               (remaining, _promoted) = promoteProvisionalAtoms currentTurn decayed
  522           in take maxProvisionalAtoms (resolveCollisions canonicalSet remaining)
  523       , ssLearningNeedState = newLearningNeedState
  524        , ssDialogueThread = advanceDialogueThreadAfterTurn (tiDialogueThread ti) (tpRmpAfterLegit tp) executedOutcome
  525       , ssDialogueCommitmentLedger = advanceDialogueLedgerAfterTurn (tiDialogueCommitmentLedger ti) executedOutcome
  526       , ssDialoguePhase = advanceDialoguePhaseAfterTurn (tiDialoguePhase ti) executedOutcome
  527       , ssTruthContractStatus = etoTruthContractStatus executedOutcome
  528       , ssGuardrailState = ssGuardrailState ss
  529         -- ^ WP5: guardrails currently pass through unchanged;
  530         --   proposal-submission tracking will be wired when the
  531         --   external-tool bridge is integrated.
  532       , ssCalibrationLog = nextCalibrationLog
  533       , ssKnowledgeTree = nextTree
  534         -- ^ Phase 7: rooted knowledge tree.  Root updated from
  535         --   committed essence; structural pruning applied each turn.
  536       , ssCalibrationSnapshots = calibrationSnapshots
  537       }
  538       nextWithLog = appendAdaptiveMutationRecords adaptiveRecords baseNext
  539       in ( nextWithLog
  540     , commitmentTrigger
  541     )

  797 advanceDialogueThreadAfterTurn :: DialogueThread -> ResponseMeaningPlan -> ExecutedTurnOutcome -> DialogueThread
  798 advanceDialogueThreadAfterTurn thread meaningPlan outcome =
  799   let family = etoFamily outcome
  800       currentFocus = dtCurrentFocus thread
  801       chooseNonEmpty primary fallback = if T.null primary then fallback else primary
  802       clarified = case family of
  803         CMClarify -> currentFocus : dtClarifiedItems thread
  804         CMGround -> currentFocus : dtClarifiedItems thread
  805         _ -> dtClarifiedItems thread
  806       unresolved = case family of
  807         CMRepair -> dtOpenLoops thread
  808         CMClarify -> dtOpenLoops thread
  809         _ -> filter (/= currentFocus) (dtOpenLoops thread)
  810       nextFocus =
  811         case family of
  812           CMClarify -> currentFocus
  813           CMGround -> currentFocus
  814           CMRepair -> currentFocus
  815           _ -> if T.null currentFocus then dtPhaseScope thread else currentFocus
  816       nextScope =
  817         case family of
  818           CMClarify -> nextFocus
  819           CMGround -> nextFocus
  820           CMRepair -> chooseNonEmpty (dtPhaseScope thread) nextFocus
  821           CMConfront -> chooseNonEmpty (dtPhaseScope thread) nextFocus
  822           _ -> chooseNonEmpty nextFocus (dtPhaseScope thread)
  823       nextStructuralScope =
  824         case etoAuthorityClass outcome of
  825           AuthorityCanonical -> rmpScope meaningPlan <|> dtStructuralScope thread
  826           AuthorityAssembled -> rmpScope meaningPlan <|> dtStructuralScope thread
  827           _ -> dtStructuralScope thread
  828   in thread
  829       { dtCurrentFocus = nextFocus
  830       , dtPhaseScope = nextScope
  831       , dtClarifiedItems = take 8 clarified
  832       , dtOpenLoops = take 8 unresolved
  833       , dtResistance = case family of
  834           CMRepair -> min 1.0 (dtResistance thread + 0.15)
  835           CMGround -> max 0.0 (dtResistance thread - 0.10)
  836           _ -> dtResistance thread
  837       , dtStructuralScope = nextStructuralScope
  838       }

  840 advanceDialogueLedgerAfterTurn :: DialogueCommitmentLedger -> ExecutedTurnOutcome -> DialogueCommitmentLedger
  841 advanceDialogueLedgerAfterTurn (DialogueCommitmentLedger items) outcome =
  842   let family = etoFamily outcome
  843       nonAuthoritative = etoAuthorityClass outcome `notElem` [AuthorityCanonical, AuthorityAssembled]
  844       rewrite item =
  845         if T.null (dcClaim item)
  846           then item
  847           else if nonAuthoritative
  848             then case family of
  849               CMRepair -> item { dcStatus = CsSuspended }
  850               _ -> item { dcStatus = downgradeCommitmentStatus (dcStatus item) }
  851             else case family of
  852               CMClarify -> item { dcStatus = CsUnresolved }
  853               CMRepair -> item { dcStatus = CsSuspended }
  854               CMGround -> item { dcStatus = CsAccepted }
  855               CMConfront -> item { dcStatus = CsContested }
  856               _ -> item
  857   in DialogueCommitmentLedger (map rewrite items)

  860 advanceDialoguePhaseAfterTurn prior outcome

  868         CMClarify -> Clarifying
  869         CMGround -> Grounding

  888 deriveLearningReplayVerdict :: SystemState -> TurnArtifacts -> LearningReplayVerdict
  889 deriveLearningReplayVerdict nextSs ta =
  890   case firstAttempt of
  891     Nothing -> LearningReplayVerdict "not_attempted" Nothing Nothing Nothing
  892     Just (Left err) ->
  893       LearningReplayVerdict
  894         { lrvStatus = case err of
  895             EqeFallback _ -> "fallback_non_authoritative"
  896             _ -> "transport_error"
  897         , lrvSandboxResult = Nothing
  898         , lrvGraftTurn = Nothing
  899         , lrvRejectReason = Just (renderExternalQueryError err)
  900         }
  901     Just (Right _) ->
  902       case () of
  903         _ | not authoritativeTurn -> LearningReplayVerdict "observed_non_authoritative" (firstCauseEvidence "sandbox_accept") Nothing (Just "truth_contract_ceiling_non_authoritative")
  904           | any isGraft currentTurnRecords -> LearningReplayVerdict "accept" (firstCauseEvidence "sandbox_accept") (Just currentTurn) Nothing
  905           | any isSandboxReject currentTurnRecords -> LearningReplayVerdict "sandbox_reject" (firstMutationEvidence "external_learning:sandbox_reject") Nothing (firstMutationEvidence "external_learning:sandbox_reject")
  906           | any isValidationReject currentTurnRecords -> LearningReplayVerdict "validation_reject" Nothing Nothing (firstMutationEvidence "external_learning:validation_reject")
  907           | any isParserReject currentTurnRecords -> LearningReplayVerdict "invalid_response" Nothing Nothing (Just "parser_rejected_schema_or_text")
  908           | otherwise -> LearningReplayVerdict "observed_non_authoritative" Nothing Nothing (Just "learning_outcome_unresolved")
  909   where
  910     authoritativeTurn = truthContractIsAuthoritative (taTruthContractStatus ta)
  911     currentTurn = ssTurnCount nextSs
  912     currentTurnRecords = filter ((== currentTurn) . amrTurnId) (ssAdaptiveMutationLog nextSs)
  913     firstAttempt = taExternalQueryResult ta <|> taExploratoryQueryResult ta
  914     isGraft record = amrCause record == "external_learning:graft" && amrDecision record == AdaptiveAccepted
  915     isSandboxReject record = amrCause record == "external_learning:sandbox_reject"
  916     isValidationReject record = amrCause record == "external_learning:validation_reject"
  917     isParserReject record = amrCause record == "tool_reliability:rejected" && any (== "reason=parser_rejected_schema_or_text") (amrEvidence record)
  918     firstMutationEvidence cause =
  919       case [ evidence | record <- currentTurnRecords, amrCause record == cause, evidence:_ <- [amrEvidence record] ] of
  920         value:_ -> Just value
  921         [] -> Nothing
  922     firstCauseEvidence reason =
  923       case [ evidence | record <- currentTurnRecords, any (== ("reason=" <> reason)) (amrEvidence record), evidence:_ <- [amrEvidence record] ] of
  924         value:_ -> Just value
  925         [] -> Nothing

  931 dedupePreservingOrder :: [Text] -> [Text]
  932 dedupePreservingOrder = go Set.empty
  933   where
  934     go _ [] = []
  935     go seen (value : rest)
  936       | value `Set.member` seen = go seen rest
  937       | otherwise = value : go (Set.insert value seen) rest
