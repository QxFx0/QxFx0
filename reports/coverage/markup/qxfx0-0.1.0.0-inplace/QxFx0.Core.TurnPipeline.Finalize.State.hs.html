   42 buildNextSystemState :: (Text -> Seq Text -> Seq Text) -> SystemState -> TurnInput -> TurnSignals -> TurnPlan -> TurnArtifacts -> DreamState -> MeaningGraph -> CanonicalMoveFamily -> R5Verdict -> Int -> SystemState
   43 buildNextSystemState updateHistory ss ti ts tp ta newDreamState newMeaningGraph outcomeFamily outcomeVerdict consecReflect =
   44   let !newHumanHistory = updateHistory (ipfRawText (tiFrame ti)) (ssHistory ss)
   45       updatedNixCache = updateStateNixCache (tiConceptToCheck ti) (tiNixStatus ti) (obsNixCache (ssObservability ss))
   46   in ss
   47       { ssDialogue = (ssDialogue ss)
   48           { dsHistory = newHumanHistory
   49           , dsActiveScene = tpActiveScene tp
   50           , dsLastFamily = outcomeFamily
   51           , dsLastTopic = tiBestTopic ti
   52           , dsUserState = tiNextUserState ti
   53           , dsLastForce = r5Force outcomeVerdict
   54           , dsLastLayer = r5Layer outcomeVerdict
   55           , dsRecentFamilies = take recentFamiliesLimit (outcomeFamily : ssRecentFamilies ss)
   56           , dsRawInputHistory = appendHistoryBounded rawInputHistoryLimit (ssRawInputHistory ss) (ipfRawText (tiFrame ti))
   57           , dsTurnCount = ssTurnCount ss + 1
   58           , dsConsecutiveReflect = consecReflect
   59           , dsLastEmbedding = Just (tiEmbedding ti)
   60           }
   61       , ssIdentity = (ssIdentity ss)
   62           { idsEgo = tpNewEgo tp
   63           , idsOrbitalMemory = tpUpdatedOrbital tp
   64           , idsLastGuardReport = Just (tpGuardReport tp)
   65           }
   66       , ssSemantic = (ssSemantic ss)
   67           { semTrace = tiNewTrace ti
   68           , semMeaningGraph = newMeaningGraph
   69           , semDreamState = newDreamState
   70           , semIntuitionState = Just (tsIntuitionState ts)
   71           , semKernelPulse = (ssKernelPulse ss) { kpActive = True, kpLastUpdate = ssTurnCount ss + 1 }
   72           , semBlockedConcepts =
   73               case tiNixStatus ti of
   74                 Blocked reason -> retainBlockedConcepts reason (ssBlockedConcepts ss)
   75                 _ -> ssBlockedConcepts ss
   76           , semIntuitConfidence = tsIntuitPosterior ts
   77           , semSemanticAnchor = tpSemanticAnchor tp
   78           , semLastTurnDecision = Just (taDecision ta)
   79           }
   80       , ssObservability = (ssObservability ss)
   81           { obsNixCache = updatedNixCache
   82           , obsTelemetry = (obsTelemetry (ssObservability ss))
   83               { lrtFamily = outcomeFamily
   84               , lrtTopic = tiBestTopic ti
   85               , lrtApiHealthy = tsApiHealthy ts
   86               , lrtGuardStatus = tdGuardStatus (taDecision ta)
   87               , lrtSurfaceRoute = taSurfaceProv ta
   88               }
   89           , obsEmbeddingApiHealthy = tsApiHealthy ts
   90           , obsLastLegitimacyScore = tpLegitScore tp
   91           }
   92       }

   94 buildTurnProjection
   95   :: Text
   96   -> Text
   97   -> Text
   98   -> Bool
   99   -> Bool
  100   -> SystemState
  101   -> TurnInput
  102   -> TurnSignals
  103   -> TurnPlan
  104   -> TurnArtifacts
  105   -> TurnProjection
  106 buildTurnProjection runtimeMode shadowPolicy localRecoveryPolicy semanticIntrospectionEnabled warnMorphologyFallbackEnabled nextSs ti ts tp ta =
  107   let decision = taDecision ta
  108       parserConfidence = ipfConfidence (tiFrame ti)
  109       parserErrors = if parserConfidence < parserLowConfidenceThreshold then ["low_confidence"] else []
  110       scenePressure
  111         | asLoad (tiAtomSet ti) <= scenePressureLowThreshold = PressureLow
  112         | asLoad (tiAtomSet ti) <= scenePressureMediumThreshold = PressureMedium
  113         | otherwise = PressureHigh
  114       legitScore = tpLegitScore tp
  115       legitimacyStatus
  116         | legitScore >= legitimacyPassThreshold = LegitimacyPass
  117         | legitScore >= legitimacyRecoveryThreshold = LegitimacyDegraded
  118         | otherwise = LegitimacyRecovery
  119       legitimacyReason
  120         | tpShadowGateTriggered tp = ReasonShadowDivergence
  121         | tpShadowStatus tp == ShadowUnavailable = ReasonShadowUnavailable
  122         | parserConfidence < parserLowConfidenceThreshold = ReasonLowParserConfidence
  123         | otherwise = ReasonOk
  124       ownerFamily = tdFamily decision
  125       ownerForce = tdForce decision
  126       warrantedMode = warrantedForFamily ownerFamily
  127       legitimacyOutcome = classifyLegitimacyOutcome legitimacyStatus legitimacyReason warrantedMode (tpShadowStatus tp) (tpShadowDivergenceSeverity tp)
  128       requestId = tmRequestId (tiMetrics ti)
  129       sessionId = tmSessionId (tiMetrics ti)
  130       intuitionHint = ifDirective <$> tsFlash ts
  131       (recoveryCause, recoveryStrategy, recoveryEvidence) =
  132         case taLocalRecoveryCause ta of
  133           Just cause ->
  134             (Just cause, taLocalRecoveryStrategy ta, taLocalRecoveryEvidence ta)
  135           Nothing
  136             | runtimeMode == "degraded" ->
  137                 (Just RecoveryRuntimeDegraded, Just StrategyNarrowScope, ["runtime_mode=degraded"])
  138           Nothing ->
  139             (Nothing, Nothing, [])
  140       replayTrace =
  141         TurnReplayTrace
  142           { trcRequestId = requestId
  143           , trcSessionId = sessionId
  144           , trcRuntimeMode = runtimeMode
  145           , trcShadowPolicy = shadowPolicy
  146           , trcLocalRecoveryPolicy = localRecoveryPolicy
  147           , trcRecoveryCause = recoveryCause
  148           , trcRecoveryStrategy = recoveryStrategy
  149           , trcRecoveryEvidence = recoveryEvidence
  150           , trcSemanticIntrospectionEnabled = semanticIntrospectionEnabled
  151           , trcWarnMorphologyFallbackEnabled = warnMorphologyFallbackEnabled
  152           , trcRequestedFamily = tiRecommendedFamily ti
  153           , trcStrategyFamily = tpStrategyFamily tp
  154           , trcNarrativeHint = tsNarrativeFragment ts
  155           , trcIntuitionHint = intuitionHint
  156           , trcPreShadowFamily = tpPreShadowFamily tp
  157           , trcShadowSnapshotId = tpShadowSnapshotId tp
  158           , trcShadowStatus = tpShadowStatus tp
  159           , trcShadowDivergenceKind = tpShadowDivergenceKind tp
  160           , trcShadowDivergenceSeverity = tpShadowDivergenceSeverity tp
  161           , trcShadowResolvedFamily = tpFamily tp
  162           , trcFinalFamily = tdFamily decision
  163           , trcFinalForce = tdForce decision
  164           , trcDecisionDisposition = loDisposition legitimacyOutcome
  165           , trcLegitimacyReason = legitimacyReason
  166           , trcParserConfidence = parserConfidence
  167           , trcEmbeddingQuality = embeddingQualityText (tiEmbeddingQuality ti)
  168           , trcClaimAst = taClaimAst ta
  169           , trcLinearizationLang = taLinearizationLang ta
  170           , trcLinearizationOk = taLinearizationOk ta
  171           , trcFallbackReason = taLinearizationFallbackReason ta
  172           }
  173   in TurnProjection
  174       { tqpTurn = ssTurnCount nextSs
  175       , tqpParserMode = ParserFrameV1
  176       , tqpParserConfidence = parserConfidence
  177       , tqpParserErrors = parserErrors
  178       , tqpPlannerMode = case tpPrincipledMode tp of Just _ -> PrincipledPlanner; Nothing -> DefaultPlanner
  179       , tqpPlannerDecision = tpFamily tp
  180       , tqpAtomRegister = asRegister (tiAtomSet ti)
  181       , tqpAtomLoad = asLoad (tiAtomSet ti)
  182       , tqpScenePressure = scenePressure
  183       , tqpSceneRequest = tiBestTopic ti
  184       , tqpSceneStance = usNeedLayer (tiNextUserState ti)
  185       , tqpRenderLane = rsMove (tpRenderStrategy tp)
  186       , tqpRenderStyle = tdRenderStyle decision
  187       , tqpLegitimacyStatus = legitimacyStatus
  188       , tqpLegitimacyReason = legitimacyReason
  189       , tqpWarrantedMode = warrantedMode
  190       , tqpDecisionDisposition = loDisposition legitimacyOutcome
  191       , tqpOwnerFamily = ownerFamily
  192       , tqpOwnerForce = ownerForce
  193       , tqpShadowStatus = tpShadowStatus tp
  194       , tqpShadowSnapshotId = tpShadowSnapshotId tp
  195       , tqpShadowDivergenceKind = tpShadowDivergenceKind tp
  196       , tqpShadowFamily = tpShadowFamily tp
  197       , tqpShadowForce = tpShadowForce tp
  198       , tqpShadowMessage = tpShadowMessage tp
  199       , tqpReplayTrace = replayTrace
  200       , tqpDivergence = tpShadowDivergence tp
  201       }

  203 buildFinalOutput :: Bool -> SystemState -> Guard.GuardSurface -> SystemState -> (Text, Guard.SafetyStatus)
  204 buildFinalOutput wantIntrospection ss baseSurface nextSs =
  205   let preIntrospectionSurface =
  206         if wantIntrospection
  207           then
  208             let introspectionText = renderSemanticIntrospection nextSs
  209             in baseSurface
  210                 { Guard.gsRenderedText = Guard.gsRenderedText baseSurface <> "\n" <> introspectionText
  211                 , Guard.gsSegments = Guard.gsSegments baseSurface <> [Guard.RenderSegment Guard.SegmentIntrospection introspectionText]
  212                 }
  213           else baseSurface
  214       finalSafetyStatus = Guard.postRenderSafetyCheckSurface preIntrospectionSurface (F.toList (ssHistory ss))
  215       outputText = safeOutputText preIntrospectionSurface baseSurface finalSafetyStatus
  216   in (outputText, finalSafetyStatus)

  218 finalizeMetrics :: TurnInput -> TurnArtifacts -> CanonicalMoveFamily -> TurnDecision -> SystemState -> Bool -> Guard.SafetyStatus -> UTCTime -> UTCTime -> TurnMetrics
  219 finalizeMetrics ti ta outcomeFamily decision savedSs apiHealthy finalSafetyStatus tSave0 tSave1 =
  220   let !metrics5 =
  221         addPhase (recordPhase "save_state" tSave0 tSave1)
  222           $ (taMetrics ta)
  223               { tmTurnCount = ssTurnCount savedSs
  224               , tmFamily = textShow outcomeFamily
  225               , tmNixStatus = textShow (tdGuardStatus decision)
  226               , tmSafetyStatus = textShow finalSafetyStatus
  227               , tmApiHealthy = apiHealthy
  228               }
  229       !metricsFinal = addPhase (recordPhase "total" (tiStartTime ti) tSave1) metrics5
  230   in metricsFinal

  232 retainBlockedConcepts :: Text -> [Text] -> [Text]
  233 retainBlockedConcepts latestReason existing =
  234   take blockedConceptsRetentionLimit (dedupePreservingOrder (latestReason : existing))
  235 
  236 dedupePreservingOrder :: [Text] -> [Text]
  237 dedupePreservingOrder = go Set.empty
  238   where
  239     go _ [] = []
  240     go seen (value : rest)
  241       | value `Set.member` seen = go seen rest
  242       | otherwise = value : go (Set.insert value seen) rest
