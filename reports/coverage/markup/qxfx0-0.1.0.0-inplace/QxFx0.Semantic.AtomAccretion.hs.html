    5 {-|
    6 Module      : QxFx0.Semantic.AtomAccretion
    7 Description : WP3 (GAP3) provisional-atom ontology quarantine and accretion.
    8 
    9 When the runtime encounters semantic atoms that do not yet belong to
   10 any canonical cluster, it places them in a provisional quarantine.  A
   11 provisional atom must be observed at least 'defaultProvisionalAtomMinOccurrences'
   12 times across at least 'defaultProvisionalAtomMinTurnSpan' turns before it is
   13 promoted.  Provisional atoms expire after 'defaultProvisionalAtomTTL' turns
   14 without re-observation.  If a provisional atom later collides with a
   15 canonical atom (produced by the existing cluster ontology), it is
   16 removed in favour of the canonical entry.
   17 
   18 This module is pure and total; all side effects (state mutation) are
   19 delegated to the caller ('buildNextSystemState').
   20 -}
   21 module QxFx0.Semantic.AtomAccretion
   22   ( observeNovelAtom
   23   , promoteProvisionalAtoms
   24   , decayProvisionalAtoms
   25   , resolveCollisions
   26   ) where
   27 
   28 import Data.List (find)
   29 
   30 import QxFx0.Types.Domain.Atoms
   31   ( AtomSet(..)
   32   , AtomTag(..)
   33   , MeaningAtom(..)
   34   , ProvisionalAtom(..)
   35   , defaultProvisionalAtomMinOccurrences
   36   , defaultProvisionalAtomMinTurnSpan
   37   , defaultProvisionalAtomTTL
   38   )
   39 
   40 -- | Observe an 'AtomTag' at the given turn.  If the tag already
   41 -- exists in the provisional list (and is not yet promoted), bump its
   42 -- occurrence count and refresh 'paLastSeenTurn'.  If it is promoted,
   43 -- leave it untouched.  If it does not exist, append a new
   44 -- 'ProvisionalAtom' with count 1.
   45 observeNovelAtom
   46   :: AtomTag
   47   -> Int
   48   -> [ProvisionalAtom]
   49   -> [ProvisionalAtom]
   50 observeNovelAtom tag currentTurn atoms =
   51   case find (\a -> paTag a == tag) atoms of
   52     Just existing
   53       | paPromoted existing -> atoms
   54       | otherwise ->
   55           let updated = existing
   56                 { paOccurrences = paOccurrences existing + 1
   57                 , paLastSeenTurn = currentTurn
   58                 }
   59           in map (\a -> if paTag a == tag then updated else a) atoms
   60     Nothing ->
   61       ProvisionalAtom
   62         { paTag = tag
   63         , paOccurrences = 1
   64         , paFirstSeenTurn = currentTurn
   65         , paLastSeenTurn = currentTurn
   66         , paPromoted = False
   67         }
   68       : atoms
   69 
   70 -- | Evaluate provisional atoms for promotion at the given turn.
   71 -- Returns @(remaining provisional atoms, promoted tags)@.
   72 -- Promotion requires:
   73 --
   74 --   * occurrences >= 'defaultProvisionalAtomMinOccurrences'
   75 --   * turn span (last - first) >= 'defaultProvisionalAtomMinTurnSpan'
   76 --
   77 -- Promoted atoms remain in the provisional list with 'paPromoted'
   78 -- set to 'True' so they are not re-evaluated or decayed.
   79 promoteProvisionalAtoms
   80   :: Int
   81   -> [ProvisionalAtom]
   82   -> ([ProvisionalAtom], [AtomTag])
   83 promoteProvisionalAtoms currentTurn atoms =
   84   let (promoted, remaining) =
   85         foldr
   86           (\a (ps, rs) ->
   87              if not (paPromoted a)
   88                    && paOccurrences a >= defaultProvisionalAtomMinOccurrences
   89                    && paLastSeenTurn a - paFirstSeenTurn a >= defaultProvisionalAtomMinTurnSpan
   90                then (paTag a : ps, a { paPromoted = True, paLastSeenTurn = currentTurn } : rs)
   91                else (ps, a : rs))
   92           ([], [])
   93           atoms
   94   in (remaining, promoted)
   95 
   96 -- | Remove un-promoted provisional atoms whose TTL has expired.
   97 -- TTL is measured from 'paLastSeenTurn'.
   98 decayProvisionalAtoms
   99   :: Int
  100   -> [ProvisionalAtom]
  101   -> [ProvisionalAtom]
  102 decayProvisionalAtoms currentTurn =
  103   filter
  104     (\a -> paPromoted a || currentTurn - paLastSeenTurn a <= defaultProvisionalAtomTTL)
  105 
  106 -- | Remove provisional atoms whose tag already appears in the
  107 -- canonical 'AtomSet' (collision detection).  The canonical copy
  108 -- wins.  Promoted atoms are retained in the provisional record
  109 -- unless they also collide, in which case they are redundant.
  110 resolveCollisions
  111   :: AtomSet
  112   -> [ProvisionalAtom]
  113   -> [ProvisionalAtom]
  114 resolveCollisions canonical atoms =
  115   let canonicalTags = map maTag (asAtoms canonical)
  116   in filter (\a -> paPromoted a || paTag a `notElem` canonicalTags) atoms
