   18 data QxFx0Exception
   19   = PersistenceError Text
   20   | PersistenceTxError !PersistenceStage !Text
   21   | SQLiteError Text
   22   | RuntimeInitError Text
   23   | EmbeddingError Text
   24   | ThresholdParseError Text
   25   | AgdaGateError Text
   26   | IdentityRupture !Text
   27     -- ^ Structural self-identity ('QxFx0.Self.Types.SelfBlanket')
   28     -- violated. Categorical failure: the running system is no longer
   29     -- /this system/. Not recoverable in the @PersistenceError@ /
   30     -- @RuntimeInitError@ sense; the session must be re-bootstrapped
   31     -- (or, in production, the process restarted). Payload is the
   32     -- semicolon-joined rendering of one or more
   33     -- 'QxFx0.Self.Types.BlanketViolation' values.
   34   | EssenceRupture !Text
   35     -- ^ Post-commitment 'Plan' violates committed 'EssenceMode'.
   36     -- Categorical failure: the system has acted contrary to what
   37     -- it has chosen to be. Not recoverable; the session must abort
   38     -- the turn (no persistence). Co-located with 'IdentityRupture'
   39     -- in 'Finalize.Commit'; the two failures are orthogonal.
   40   deriving stock (Eq)

   50 renderQxFx0ExceptionForLog :: QxFx0Exception -> Text
   51 renderQxFx0ExceptionForLog ex =
   52   case ex of
   53     PersistenceError _ -> T.pack "PersistenceError(<redacted>)"
   54     PersistenceTxError stage _ -> T.pack "PersistenceTxError(stage=" <> T.pack (show stage) <> T.pack ", detail=<redacted>)"
   55     SQLiteError _ -> T.pack "SQLiteError(<redacted>)"
   56     RuntimeInitError _ -> T.pack "RuntimeInitError(<redacted>)"
   57     EmbeddingError _ -> T.pack "EmbeddingError(<redacted>)"
   58     ThresholdParseError _ -> T.pack "ThresholdParseError(<redacted>)"
   59     AgdaGateError _ -> T.pack "AgdaGateError(<redacted>)"
   60     IdentityRupture _ -> T.pack "IdentityRupture(<redacted>)"
   61     EssenceRupture _ -> T.pack "EssenceRupture(<redacted>)"

   63 tryAsync :: IO a -> IO (Either SomeException a)
   64 tryAsync action = do
   65   result <- try action
   66   case result of
   67     Left se -> case fromException se of
   68       Just (_ :: AsyncException) -> throwIO se
   69       Nothing -> pure (Left se)
   70     Right v -> pure (Right v)
