   88 -- | Detect whether input text is English (pure Latin, no Cyrillic).
   89 isEnglishInput :: Text -> Bool
   90 isEnglishInput input =
   91   let letters = T.filter isAlpha input
   92       hasLatin = T.any (\c -> ('a' <= c && c <= 'z') || ('A' <= c && c <= 'Z')) letters
   93       hasCyrillic = T.any (\c -> ('а' <= c && c <= 'я') || ('А' <= c && c <= 'Я') || c == 'ё' || c == 'Ё') letters
   94   in hasLatin && not hasCyrillic

   97 -- This is a minimal EN counterpart to linearizeClaimAstRus.

  101 linearizeClaimAstEn :: ClaimAst -> Maybe Text
  102 linearizeClaimAstEn ast =
  103   case ast of
  104     StanceWrapped _ innerAst -> linearizeClaimAstEn innerAst
  105     ClaimPurpose topic ->
  106       let t = T.toLower (T.strip topic)
  107       in if T.null t then Nothing else Just ("The purpose of " <> t <> " reveals itself through repeatable roles in action.")
  108     ClaimSelfState ->
  109       Just "I maintain local self-state through typed parsing and move routing."
  110     ClaimComparison left right ->
  111       let l = T.toLower (T.strip left)
  112           r = T.toLower (T.strip right)
  113       in if T.null l || T.null r then Nothing else Just ("Comparison of " <> l <> " and " <> r <> " is stable only within an explicit frame.")
  114     MoveInvite (MkNP gfTopic) gfMod gfAction ->
  115       let topic = maybe "" glfPrep (lookupGfLexemeForms gfTopic)
  116           modStr = case gfMod of
  117             ModFirst -> "first "
  118             ModStrictly -> "strictly "
  119           vpStr = case gfAction of
  120             ActMaintain _ "ramka_N" -> "maintain the frame"
  121             ActDefine "granitsa_N"  -> "define the boundary"
  122             ActMaintain _ obj       -> "maintain " <> maybe obj glfAcc (lookupGfLexemeForms obj)
  123             ActDefine obj           -> "define " <> maybe obj glfAcc (lookupGfLexemeForms obj)
  124       in if T.null topic then Nothing else Just ("Yes, let us talk about " <> topic <> ". I will " <> modStr <> vpStr <> " to keep focus.")
  125     MoveDefine (MkNP gfSubj) _ (MkNP gfObj) ->
  126       let subj = maybe "" glfNom (lookupGfLexemeForms gfSubj)
  127           obj  = maybe "" glfNom (lookupGfLexemeForms gfObj)
  128       in if T.null subj || T.null obj then Nothing else Just (capitalize subj <> " is a form of " <> obj <> ".")
  129     MoveGround (MkNP gfSubj) ->
  130       let subj = maybe "" glfNom (lookupGfLexemeForms gfSubj)

  132     MoveDistinguish (MkNP gfA) (MkNP gfB) ->
  133       let a = maybe "" glfNom (lookupGfLexemeForms gfA)
  134           b = maybe "" glfNom (lookupGfLexemeForms gfB)

  136     MoveContact (MkNP gfSubj) ->

  151     MoveOperationalCause ->
  152       Just "The operational issue is in parsing and routing."
  153     MoveSelfState ->
  154       Just "I maintain local self-state through typed parsing and move routing."
  155     MoveCompare (MkNP gfLeft) (MkNP gfRight) ->
  156       let left  = maybe "" glfNom (lookupGfLexemeForms gfLeft)
  157           right = maybe "" glfNom (lookupGfLexemeForms gfRight)
  158       in if T.null left || T.null right then Nothing else Just ("Comparison of " <> left <> " and " <> right <> " is stable only within an explicit frame.")

  195 renderDialogueArtifact :: InputPropositionFrame -> ResponseMeaningPlan -> ResponseContentPlan -> Text -> [IdentityClaimRef] -> MorphologyData -> DialogueRenderArtifact
  196 renderDialogueArtifact frame rmp rcp topic claims morph =
  197   case renderStructuredDialogueArtifact frame rmp rcp (rcpStyle rcp) morph of
  198     Just artifact -> artifact
  199     Nothing ->
  200       let fallbackReason =
  201             case propositionTypeFromText (ipfPropositionType frame) of
  202               Nothing -> "unknown_proposition_type"
  203               Just ptype ->
  204                 if structuredDialogueType ptype
  205                   then "structured_body_returned_plain"
  206                   else "proposition_type_not_structured"
  207       in if isEnglishInput (ipfRawText frame)
  208       then
  209         let topicText = cleanTopic topic
  210             enBody = "I am here to continue the dialogue in English."
  211                      <> if T.null topicText then "" else " I received your message about " <> topicText <> "."
  212             rendered = finalizeForce (rmpForce rmp) (T.strip enBody)
  213         in DialogueRenderArtifact
  214             { draRenderedText = rendered
  215             , draQuestionLike = rmpForce rmp == IFAsk
  216             , draStylePrefixText = ""
  217             , draTemplateBodyText = enBody
  218             , draClaimText = ""
  219             , draClaimAst = Nothing
  220             , draLinearizationLang = Just "en_fallback"
  221             , draLinearizationOk = False
  222             , draFallbackReason = Just ("en_unstructured_fallback:" <> fallbackReason)
  223             , draContractProvenance = FallbackRoute
  224             , draSurfaceProvenance = FromFallback
  225             , draDerivationTags = ["surface=en_unstructured", "fallback=" <> fallbackReason]
  226             , draDialogAtoms = emptyDialogAtoms
  227             }
  228       else
  229         let cleanedTopic = cleanTopic topic
  230             openingText = moveToText (rcpOpening rcp) cleanedTopic morph
  231             coreText = moveToText (rcpCore rcp) cleanedTopic morph
  232             limitText = moveToText (rcpLimit rcp) cleanedTopic morph
  233             contText = moveToText (rcpContinuation rcp) cleanedTopic morph
  234             stylePrefixText = stylePrefix (rcpStyle rcp)
  235             microPlan = rmpMicroPlan rmp
  236             microPrefaceText = microPlanPrefix rmp cleanedTopic
  237             claimText = case claims of
  238               (c:_) ->
  239                 case sanitizeIdentityClaimText (icrText c) of
  240                   Just txt -> " " <> txt
  241                   Nothing -> ""
  242               []    -> ""
  243             parts = take (max 1 (mpStructureBudget microPlan + 1)) . dedupeText $ filter (not . T.null) [microPrefaceText, openingText, coreText, limitText]
  244             body = T.intercalate (microPlanDelimiter (rcpStyle rcp) microPlan) parts
  245             fullBody = appendContinuation microPlan body contText
  246             withClaims = if T.null claimText then fullBody else fullBody <> claimText
  247             withStyle = if T.null stylePrefixText then withClaims else stylePrefixText <> " " <> withClaims
  248             rendered = finalizeForce (rmpForce rmp) (T.strip withStyle)
  249         in DialogueRenderArtifact
  250             { draRenderedText = rendered
  251             , draQuestionLike = rmpForce rmp == IFAsk
  252             , draStylePrefixText = stylePrefixText
  253             , draTemplateBodyText = fullBody
  254             , draClaimText = claimText
  255             , draClaimAst = Nothing
  256             , draLinearizationLang = Nothing
  257             , draLinearizationOk = False
  258             , draFallbackReason = Just ("ru_unstructured_fallback:" <> fallbackReason)
  259             , draContractProvenance = FallbackRoute
  260             , draSurfaceProvenance = FromFallback
  261             , draDerivationTags = ["surface=ru_unstructured", "fallback=" <> fallbackReason]
  262             , draDialogAtoms = emptyDialogAtoms
  263             }

  269 renderStructuredDialogueArtifact :: InputPropositionFrame -> ResponseMeaningPlan -> ResponseContentPlan -> RenderStyle -> MorphologyData -> Maybe DialogueRenderArtifact
  270 renderStructuredDialogueArtifact frame rmp rcp renderStyle morph = do
  271   propositionType <- propositionTypeFromText (ipfPropositionType frame)
  272   if not (structuredDialogueType propositionType)
  273     then Nothing
  274     else
  275       let (body0, claimAst, mLang, linearizationOk, fallbackReason) =
  276             structuredBody propositionType frame rmp renderStyle morph
  277           body1 = applyMicroPlanToStructuredBody rmp renderStyle body0
  278           continuationText = structuredContinuationText frame rmp rcp morph
  279           body = appendContinuation (rmpMicroPlan rmp) body1 continuationText
  280           rendered = finalizeForce IFAssert (T.strip body)
  281           contractProv = contractProvenanceForArtifact fallbackReason claimAst
  282           surfaceProv = surfaceProvenanceForArtifact fallbackReason claimAst
  283       in Just
  284            DialogueRenderArtifact
  285              { draRenderedText = rendered
  286              , draQuestionLike = False
  287              , draStylePrefixText = ""
  288              , draTemplateBodyText = body
  289              , draClaimText = ""
  290               , draClaimAst = claimAst
  291               , draLinearizationLang = mLang
  292               , draLinearizationOk = linearizationOk
  293               , draFallbackReason = fallbackReason
  294               , draContractProvenance = contractProv
  295               , draSurfaceProvenance = surfaceProv
  296                , draDerivationTags = artifactDerivationTags propositionType linearizationOk fallbackReason claimAst
  297               , draDialogAtoms = emptyDialogAtoms
  298               }

  963 isLikelyBrokenGenitive :: Text -> Text -> Bool
  964 isLikelyBrokenGenitive raw gen =
  965   let rawN = T.toLower (T.strip raw)
  966       genN = T.toLower (T.strip gen)
  967       endsWithVowel t =
  968         case T.unsnoc t of
  969           Just (_, ch) -> Char.toLower ch `elem` ("аеёиоуыэюя" :: String)
  970           Nothing -> False
  971       infinitiveGenitiveArtifact =
  972         ("ть" `T.isSuffixOf` rawN || "ти" `T.isSuffixOf` rawN)
  973           && "ти" `T.isSuffixOf` genN
  974       vowelPlusAArtifact = endsWithVowel rawN && genN == rawN <> "а"
  975   in T.null genN || infinitiveGenitiveArtifact || vowelPlusAArtifact

 1548 claimAstLexemes :: ClaimAst -> [Text]
 1549 claimAstLexemes ast =
 1550   case ast of
 1551     MoveInvite (MkNP topic) _ action -> topic : gfActionLexemes action
 1552     MoveDefine (MkNP subj) _ (MkNP obj) -> [subj, obj]
 1553     MoveCause (MkNP subj) _ -> [subj]
 1554     MovePurpose (MkNP topic) -> [topic]
 1555     MoveCompare (MkNP left) (MkNP right) -> [left, right]
 1556     MoveContemplative (MkNP topic) -> [topic]
 1557     MoveGround (MkNP topic) -> [topic]
 1558     MoveContact (MkNP topic) -> [topic]
 1559     MoveReflect (MkNP topic) -> [topic]
 1560     MoveDescribe (MkNP topic) -> [topic]
 1561     MoveDeepen (MkNP topic) -> [topic]
 1562     MoveConfront (MkNP topic) -> [topic]
 1563     MoveAnchor (MkNP topic) -> [topic]
 1564     MoveClarify (MkNP topic) -> [topic]
 1565     MoveNextStepLocal (MkNP topic) -> [topic]
 1566     MoveHypothesis (MkNP topic) -> [topic]
 1567     MoveDistinguish (MkNP left) (MkNP right) -> [left, right]
 1568     StanceWrapped _ inner -> claimAstLexemes inner
 1569     _ -> []
