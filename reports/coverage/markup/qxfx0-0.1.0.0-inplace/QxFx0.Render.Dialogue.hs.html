   70 renderDialogueArtifact :: InputPropositionFrame -> ResponseMeaningPlan -> ResponseContentPlan -> Text -> [IdentityClaimRef] -> MorphologyData -> DialogueRenderArtifact
   71 renderDialogueArtifact frame rmp rcp topic claims morph =
   72   case renderStructuredDialogueArtifact frame rmp (rcpStyle rcp) morph of
   73     Just artifact -> artifact
   74     Nothing ->
   75       let cleanedTopic = cleanTopic topic
   76           openingText = moveToText (rcpOpening rcp) cleanedTopic morph
   77           coreText = moveToText (rcpCore rcp) cleanedTopic morph
   78           limitText = moveToText (rcpLimit rcp) cleanedTopic morph
   79           contText = moveToText (rcpContinuation rcp) cleanedTopic morph
   80           stylePrefixText = stylePrefix (rcpStyle rcp)
   81           claimText = case claims of
   82             (c:_) ->
   83               case sanitizeIdentityClaimText (icrText c) of
   84                 Just txt -> " " <> txt
   85                 Nothing -> ""
   86             []    -> ""
   87           parts = dedupeText (filter (not . T.null) [openingText, coreText, limitText])
   88           body = T.intercalate (styleDelimiter (rcpStyle rcp)) (take 3 parts)
   89           fullBody = if T.null contText then body else body <> arrowSeparator <> contText
   90           withClaims = if T.null claimText then fullBody else fullBody <> claimText
   91           withStyle = if T.null stylePrefixText then withClaims else stylePrefixText <> " " <> withClaims
   92           rendered = finalizeForce (rmpForce rmp) (T.strip withStyle)
   93       in DialogueRenderArtifact
   94           { draRenderedText = rendered
   95           , draQuestionLike = rmpForce rmp == IFAsk
   96           , draStylePrefixText = stylePrefixText
   97           , draTemplateBodyText = fullBody
   98           , draClaimText = claimText
   99           , draClaimAst = Nothing
  100           , draLinearizationLang = Nothing
  101           , draLinearizationOk = False
  102           , draFallbackReason = Nothing
  103           }

  109 renderStructuredDialogueArtifact :: InputPropositionFrame -> ResponseMeaningPlan -> RenderStyle -> MorphologyData -> Maybe DialogueRenderArtifact
  110 renderStructuredDialogueArtifact frame rmp renderStyle morph = do
  111   propositionType <- propositionTypeFromText (ipfPropositionType frame)
  112   if not (structuredDialogueType propositionType)
  113     then Nothing
  114     else
  115       let (body, claimAst, mLang, linearizationOk, fallbackReason) =
  116             structuredBody propositionType frame rmp renderStyle morph
  117           rendered = finalizeForce IFAssert (T.strip body)
  118       in Just
  119           DialogueRenderArtifact
  120             { draRenderedText = rendered
  121             , draQuestionLike = False
  122             , draStylePrefixText = ""
  123             , draTemplateBodyText = body
  124             , draClaimText = ""
  125             , draClaimAst = claimAst
  126             , draLinearizationLang = mLang
  127             , draLinearizationOk = linearizationOk
  128             , draFallbackReason = fallbackReason
  129             }

  349 linearizeOrFallback :: ClaimAst -> RenderStyle -> MorphologyData -> Text -> ClaimLinearization

  365 linearizeClaimAstRus :: ClaimAst -> RenderStyle -> MorphologyData -> Maybe Text
  366 linearizeClaimAstRus ast renderStyle morph =
  367   case ast of
  368     StanceWrapped "ApplyStanceTentative" innerAst ->
  369       case linearizeClaimAstRus innerAst renderStyle morph of
  370         Just inner -> Just ("Возможно, нам стоит сказать, что " <> inner)
  371         Nothing -> Nothing
  372     StanceWrapped "ApplyStanceFirm" innerAst ->
  373       case linearizeClaimAstRus innerAst renderStyle morph of
  374         Just inner -> Just ("Зафиксируем строго: " <> inner)
  375         Nothing -> Nothing
  376     StanceWrapped _ innerAst ->
  377       linearizeClaimAstRus innerAst renderStyle morph
  378     MoveDefine (MkNP gfSubj) RelIdentity (MkNP gfObj) ->
  379       let subjNom = maybe "смысл" glfNom (lookupGfLexemeForms gfSubj)
  380           objIns = maybe "смыслом" glfIns (lookupGfLexemeForms gfObj)
  381       in Just (subjNom <> " является " <> objIns <> ".")
  382     MoveCause (MkNP gfSubj) MechParse ->
  383       let subjGen = maybe "смысла" glfGen (lookupGfLexemeForms gfSubj)
  384           seed = "cause|" <> subjGen
  385       in Just (pickDeterministic seed
  386           [ "Причиной " <> subjGen <> " служит механизм локального разбора."
  387           , "В моём локальном контуре причина " <> subjGen <> " объясняется механизмом локального разбора."
  388           , "Для " <> subjGen <> " я беру причинную схему через механизм локального разбора."
  389           , "Причинное объяснение " <> subjGen <> " у меня строится через механизм локального разбора."
  390           ])
  391     MoveInvite (MkNP gfTopic) gfMod gfAction ->
  392       -- Runtime fallback mirrors the GF surface when PGF is unavailable.
  393       let prepForm = maybe "смысле" glfPrep (lookupGfLexemeForms gfTopic)
  394           modStr = case gfMod of

  579 isLikelyBrokenGenitive :: Text -> Text -> Bool
  580 isLikelyBrokenGenitive raw gen =
  581   let rawN = T.toLower (T.strip raw)
  582       genN = T.toLower (T.strip gen)
  583       endsWithVowel t =
  584         case T.unsnoc t of
  585           Just (_, ch) -> Char.toLower ch `elem` ("аеёиоуыэюя" :: String)
  586           Nothing -> False
  587       infinitiveGenitiveArtifact =
  588         ("ть" `T.isSuffixOf` rawN || "ти" `T.isSuffixOf` rawN)
  589           && "ти" `T.isSuffixOf` genN
  590       vowelPlusAArtifact = endsWithVowel rawN && genN == rawN <> "а"
  591   in T.null genN || infinitiveGenitiveArtifact || vowelPlusAArtifact

  901 heuristicGenitive :: Text -> Text
  902 heuristicGenitive word
  903   | T.null word = word
  904   | isVerbLikeTopic word = "действия"
  905   | "ия" `T.isSuffixOf` word = T.dropEnd 2 word <> "ии"
  906   | "ие" `T.isSuffixOf` word = T.dropEnd 2 word <> "ия"
  907   | "и" `T.isSuffixOf` word = word
  908   | "ы" `T.isSuffixOf` word = word
  909   | "у" `T.isSuffixOf` word = word
  910   | "ю" `T.isSuffixOf` word = word
  911   | "а" `T.isSuffixOf` word =
  912       let stem = T.dropEnd 1 word
  913       in stem <> (if hardConsonantStem stem then "и" else "ы")
  914   | "я" `T.isSuffixOf` word = T.dropEnd 1 word <> "и"
  915   | "ь" `T.isSuffixOf` word = T.dropEnd 1 word <> "и"
  916   | "й" `T.isSuffixOf` word = T.dropEnd 1 word <> "я"
  917   | "о" `T.isSuffixOf` word = T.dropEnd 1 word <> "а"
  918   | "е" `T.isSuffixOf` word = T.dropEnd 1 word <> "я"
  919   | isLikelyRussianWord word = word <> "а"
  920   | otherwise = word
