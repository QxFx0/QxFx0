   98 assemblePerspectiveInput :: SystemState -> ConatusEnergy -> Bool -> Field -> PerspectiveInputBundle
   99 assemblePerspectiveInput ss conatusEnergy conatusGateFired field =
  100   let scope = selectPerspectiveScope ss
  101       registry = ssPerspectiveRegistry ss
  102       normativeProfile = activeNormativeProfile scope registry
  103       lineage = lineageForScope scope registry
  104       beliefStore = ssBeliefStore ss
  105       counterarguments = counterargumentsForScope scope beliefStore (ssDialogueOutcomeLearning ss) lineage
  106   in PerspectiveInputBundle
  107       { pibScope = scope
  108       , pibEvidence = evidenceForScope scope (ssKnowledgeTree ss) (ssDialogueOutcomeLearning ss)
  109       , pibStanceSlice = stanceSliceForScope scope beliefStore
  110       , pibIdentitySlice = identitySlice ss
  111       , pibConatusSlice = conatusSlice conatusEnergy conatusGateFired field
  112       , pibNormativeProfile = normativeProfile
  113       , pibCounterarguments = counterarguments
  114       , pibRevisionLineage = lineage
  115       }

  117 opinionCore :: PerspectiveInputBundle -> PerspectiveCandidate
  118 opinionCore bundle =
  119   let evidenceCount = length (pibEvidence bundle)
  120       stanceCount = length (pibStanceSlice bundle)
  121       counterCount = length (pibCounterarguments bundle)
  122       lineage = pibRevisionLineage bundle
  123       np = pibNormativeProfile bundle
  124       counterPressure = clampUnit (fromIntegral counterCount / 4.0)
  125       normativeAlignment = computeNormativeAlignment np counterPressure (pibConatusSlice bundle)
  126       internalTension = clampUnit (0.55 * counterPressure + 0.45 * (1.0 - normativeAlignment))
  127       supportScore = clampUnit ((fromIntegral evidenceCount * 0.28) + (fromIntegral stanceCount * 0.14))
  128       confidence = clampUnit (0.18 + supportScore + (0.25 * normativeAlignment) - (0.35 * counterPressure) - (0.20 * internalTension))
  129       orientation = orientationFromScores confidence counterPressure normativeAlignment internalTension
  130       supportingClaims = take 8 (map renderEvidenceRef (pibEvidence bundle) <> map renderClaimStanceRef (pibStanceSlice bundle))
  131       baseVersion = case lineage of
  132         recent:_ -> Just (prrToVersion recent)
  133         [] -> Nothing
  134   in PerspectiveCandidate
  135       { pcScope = pibScope bundle
  136       , pcThesis = thesisForScope (pibScope bundle) orientation supportingClaims
  137       , pcOrientation = orientation
  138       , pcConfidence = confidence
  139       , pcSupportingClaims = supportingClaims
  140       , pcCounterargumentPressure = counterPressure
  141       , pcNormativeAlignment = normativeAlignment
  142       , pcInternalTension = internalTension
  143       , pcBaseVersion = baseVersion
  144       , pcNormativeProfileVersion = npVersionId np
  145       }

  147 evaluatePerspectiveAdmissibility :: PerspectiveInputBundle -> PerspectiveCandidate -> PerspectiveAdmissibility
  148 evaluatePerspectiveAdmissibility bundle candidate
  149   | null (pibEvidence bundle) && null (pibStanceSlice bundle) =
  150       PerspectiveInadmissible "insufficient_evidence"
  151   | T.null (isSessionId (pibIdentitySlice bundle)) =
  152       PerspectiveInadmissible "identity_violation:empty_session"
  153   | csGateFired (pibConatusSlice bundle) || csEnergy (pibConatusSlice bundle) < 0.0 =
  154       PerspectiveInadmissible "conatus_violation"
  155   | pcCounterargumentPressure candidate > 0.74 =
  156       PerspectiveInadmissible "excessive_counterargument_pressure"
  157   | npConflictPolicy (pibNormativeProfile bundle) == "invalid" =
  158       PerspectiveInadmissible "invalid_normative_conflict"
  159   | pcNormativeAlignment candidate < 0.25 =
  160       PerspectiveInadmissible "invalid_normative_conflict"
  161   | pcConfidence candidate < 0.30 =
  162       PerspectiveInadmissible "invalid_confidence_geometry"
  163   | length (pcSupportingClaims candidate) > 8 =
  164       PerspectiveInadmissible "boundedness_violation:supporting_claims"
  165   | invalidRevisionStep bundle candidate =
  166       PerspectiveInadmissible "invalid_revision_step"
  167   | pcCounterargumentPressure candidate > 0.45 =
  168       PerspectiveAdmissibleQuarantined
  169   | pcConfidence candidate >= 0.62 && pcNormativeAlignment candidate >= 0.55 =
  170       PerspectiveAdmissibleAccepted
  171   | otherwise =
  172       PerspectiveAdmissibleObserved

  174 decidePerspectivePromotion :: PerspectiveRegistry -> PerspectiveInputBundle -> PerspectiveCandidate -> PerspectiveAdmissibility -> PerspectivePromotionDecision
  175 decidePerspectivePromotion registry bundle candidate admissibility =
  176   case admissibility of
  177     PerspectiveInadmissible _ -> PpdObserveOnly
  178     PerspectiveAdmissibleObserved -> PpdObserveOnly
  179     PerspectiveAdmissibleQuarantined
  180       | rollbackPressure registry candidate -> PpdRollbackPrior
  181       | contestedActivePressure registry candidate -> PpdSuspendActive
  182       | otherwise -> PpdQuarantine
  183     PerspectiveAdmissibleAccepted
  184       | weakEvidenceOnly bundle -> PpdObserveOnly
  185       | rollbackPressure registry candidate -> PpdRollbackPrior
  186       | contestedActivePressure registry candidate -> PpdSuspendActive
  187       | hasActivePerspective (pcScope candidate) registry
  188           && stableCorroboration bundle candidate -> PpdReviseActive
  189       | stableCorroboration bundle candidate -> PpdPromoteEndorsed
  190       | otherwise -> PpdAcceptBounded

  192 applyPerspectiveOperator :: SystemState -> ConatusEnergy -> Bool -> Field -> SystemState
  193 applyPerspectiveOperator ss conatusEnergy conatusGateFired field =
  194   let bundle = assemblePerspectiveInput ss conatusEnergy conatusGateFired field
  195       candidate = opinionCore bundle
  196       admissibility = evaluatePerspectiveAdmissibility bundle candidate
  197       registry0 = ssPerspectiveRegistry ss
  198       decision = decidePerspectivePromotion registry0 bundle candidate admissibility
  199       prospectiveRegistry = applyPerspectiveDecision (ssTurnCount ss) registry0 bundle candidate decision
  200       prospectiveAdaptiveRecord = perspectiveMutationRecord (ssTurnCount ss) prospectiveRegistry bundle candidate admissibility decision
  201       previousEvent = listToMaybe (reverse (ssGovernanceHistory ss))
  202       sequenceNo = length (ssGovernanceHistory ss) + 1
  203       prospectiveEvent = buildPerspectiveGovernanceEvent
  204         ActorRuntime
  205         sequenceNo
  206         (ssSessionId ss)
  207         previousEvent
  208         registry0
  209         prospectiveRegistry
  210         bundle
  211         candidate
  212         admissibility
  213         (Just decision)
  214         decision
  215         (buildPerspectiveProjection prospectiveRegistry (pcScope candidate))
  216    in case appendGovernanceEventRecord prospectiveEvent ss of
  217         Left err -> ss { ssGovernanceRuntimeFault = Just (GrfAppendRejected err) }
  218         Right ssWithGovernance ->
  219            case verifyPerspectiveRegistryRebuild (ssGovernanceHistory ssWithGovernance) prospectiveRegistry of
  220               Right _ ->
  221                 case rebuildGovernedViews ssWithGovernance of
  222                   Right (_, rebuiltProjection) ->
  223                     (commitGovernedPerspectiveProjection rebuiltProjection prospectiveAdaptiveRecord ssWithGovernance)
  224                       { ssGovernanceRuntimeFault = Nothing }
  225                   Left err ->
  226                     ss { ssGovernanceRuntimeFault = Just (GrfRebuildUnavailable err) }
  227               Left "governance perspective registry rebuild mismatch" ->
  228                 ss { ssGovernanceRuntimeFault = Just GrfRebuildMismatch }
  229               Left err ->
  230                 ss { ssGovernanceRuntimeFault = Just (GrfRebuildUnavailable err) }

  232 selectPerspectiveScope :: SystemState -> PerspectiveScope
  233 selectPerspectiveScope ss =
  234   let topic = firstNonEmpty
  235         [ lastOutcomeTopic (ssDialogueOutcomeLearning ss)
  236         , strongestClaimTopic (ssBeliefStore ss)
  237         , strongestKnowledgeTopic (ssKnowledgeTree ss)
  238         ]
  239       scopedTopic = clampPerspectiveScopeTopic (ssDialoguePhase ss) (ssDialogueThread ss) (fromMaybe "general" topic)
  240       threadScope = dtStructuralScope (ssDialogueThread ss)
  241   in case threadScope of
  242        Just scope
  243          | ssDialoguePhase ss `elem` [Clarifying, Grounding, Repairing, Contesting, Closing] -> scope
  244          | normalizeDialogueText (scopeText scope) == normalizeDialogueText scopedTopic -> scope
  245        _ -> ScopeTopic scopedTopic

  247 clampPerspectiveScopeTopic :: DialoguePhase -> DialogueThread -> Text -> Text
  248 clampPerspectiveScopeTopic phase thread proposedTopic =
  249   let proposed = normalizeDialogueText proposedTopic
  250       currentFocus = normalizeDialogueText (dtCurrentFocus thread)
  251       phaseScope = normalizeDialogueText (dtPhaseScope thread)
  252       activeScope = firstNonEmpty [structuralScopeTextMaybe (dtStructuralScope thread), nonEmptyMaybe phaseScope, nonEmptyMaybe currentFocus]
  253       lockedPhase = phase `elem` [Clarifying, Grounding, Repairing, Contesting, Closing]
  254   in case activeScope of
  255        Nothing -> if T.null proposed then "general" else proposed
  256        Just scope
  257          | T.null proposed -> scope
  258          | lockedPhase && normalizeDialogueText scope /= normalizeDialogueText proposed -> scope
  259          | otherwise -> proposed

  261 nonEmptyMaybe :: Text -> Maybe Text
  262 nonEmptyMaybe txt
  263   | T.null txt = Nothing
  264   | otherwise = Just txt

  266 activeNormativeProfile :: PerspectiveScope -> PerspectiveRegistry -> NormativeProfile
  267 activeNormativeProfile scope registry =
  268   case M.lookup (prActiveNormativeProfileId registry) (prNormativeProfiles registry) of
  269     Just profile | normativeProfileApplies scope profile -> profile
  270     _ -> case filter (normativeProfileApplies scope) (M.elems (prNormativeProfiles registry)) of
  271       profile:_ -> profile
  272       [] -> defaultNormativeProfile

  274 lineageForScope :: PerspectiveScope -> PerspectiveRegistry -> [PerspectiveRevisionRecord]
  275 lineageForScope scope registry =
  276   case M.lookup scope (prThreads registry) of
  277     Nothing -> []
  278     Just thread -> take (prMaxLineageSlice registry) (ptRevisionHistory thread)

  280 evidenceForScope :: PerspectiveScope -> KnowledgeTree -> DialogueOutcomeLearningState -> [EvidenceRef]
  281 evidenceForScope scope tree outcomes =
  282   take 12 (knowledgeEvidence scope tree <> dialogueEvidence scope outcomes)
  283 
  284 knowledgeEvidence :: PerspectiveScope -> KnowledgeTree -> [EvidenceRef]
  285 knowledgeEvidence scope tree =
  286   let topic = scopeText scope
  287       fruits = concatMap brFruits (concat (M.elems (ktBranches tree)))
  288       relevant fruit =
  289         kfValidated fruit
  290           && (T.null topic || topic `T.isInfixOf` normalizeDialogueText (kfWord fruit <> " " <> kfProposition fruit))
  291   in map (EvidenceRef . ("knowledge:" <>) . T.take 120 . kfProposition) (filter relevant fruits)
  292 
  293 dialogueEvidence :: PerspectiveScope -> DialogueOutcomeLearningState -> [EvidenceRef]
  294 dialogueEvidence scope outcomes =
  295   let topic = scopeText scope
  296       relevant sample =
  297           dosStrongUpdate sample
  298             && dosKind sample `elem` [DialogueOutcomeSuccess, DialogueOutcomeConflict, DialogueOutcomeRepeatedQuestion]
  299             && (T.null topic || normalizeDialogueText (dosTopic sample) == topic)
  300   in map (EvidenceRef . dialogueEvidenceText) (take 6 (filter relevant (dolRecentOutcomes outcomes)))
  301 
  302 stanceSliceForScope :: PerspectiveScope -> BeliefStore -> [ClaimStanceRef]
  303 stanceSliceForScope scope store =
  304   let topic = scopeText scope
  305       relevant record =
  306           brConfidence record >= 0.45
  307             && brPolarity record `elem` [BeliefAffirmed, BeliefTentative, BeliefContested]
  308             && (T.null topic || topic `T.isInfixOf` normalizeDialogueText (brClaim record))
  309       scored record = (brConfidence record, record)
  310   in map (ClaimStanceRef . stanceRefText) . take 8 . map snd . sortOn (Down . fst) . map scored . filter relevant . M.elems $ bsClaims store
  311 
  312 counterargumentsForScope :: PerspectiveScope -> BeliefStore -> DialogueOutcomeLearningState -> [PerspectiveRevisionRecord] -> [CounterargumentRef]
  313 counterargumentsForScope scope store outcomes lineage =
  314   take 8 (stanceCounterarguments scope store <> dialogueCounterarguments scope outcomes <> lineageCounterarguments lineage)
  315 
  316 stanceCounterarguments :: PerspectiveScope -> BeliefStore -> [CounterargumentRef]
  317 stanceCounterarguments scope store =
  318   let topic = scopeText scope
  319       relevant record =
  320         brPolarity record `elem` [BeliefContested, BeliefRejected, BeliefAmbivalent]
  321           && (T.null topic || topic `T.isInfixOf` normalizeDialogueText (brClaim record))
  322   in map (CounterargumentRef . ("stance_counter:" <>) . brClaim) (filter relevant (M.elems (bsClaims store)))
  323 
  324 dialogueCounterarguments :: PerspectiveScope -> DialogueOutcomeLearningState -> [CounterargumentRef]
  325 dialogueCounterarguments scope outcomes =
  326   let topic = scopeText scope
  327       relevant sample =
  328         dosKind sample == DialogueOutcomeConflict
  329           && (T.null topic || normalizeDialogueText (dosTopic sample) == topic)
  330   in map (CounterargumentRef . ("dialogue_conflict:" <>) . dosTopic) (take 4 (filter relevant (dolRecentOutcomes outcomes)))
  331 
  332 lineageCounterarguments :: [PerspectiveRevisionRecord] -> [CounterargumentRef]
  333 lineageCounterarguments = mapMaybe lineageCounterargument
  334   where
  335     lineageCounterargument record
  336       | prrCounterargumentDelta record > 0.25 = Just (CounterargumentRef ("revision_pressure:" <> prrTrigger record))
  337       | otherwise = Nothing

  349 conatusSlice :: ConatusEnergy -> Bool -> Field -> ConatusSlice
  350 conatusSlice energy gateFired field =
  351   let fieldConf = unFieldConfidence (fieldConfidence field)
  352   in ConatusSlice
  353       { csEnergy = ceScalar energy
  354       , csGateFired = gateFired
  355       , csFieldConfidence = fieldConf
  356       , csStability = clampUnit (fieldConf + min 0.25 (max 0.0 (ceScalar energy / 80.0)))
  357       }

  359 invalidRevisionStep :: PerspectiveInputBundle -> PerspectiveCandidate -> Bool
  360 invalidRevisionStep bundle candidate =
  361   case (pibRevisionLineage bundle, pcBaseVersion candidate) of
  362     ([], Nothing) -> False
  363     ([], Just _) -> True
  364     (recent:_, Just baseVersion) -> prrToVersion recent /= baseVersion
  365     (_:_, Nothing) -> True

  367 weakEvidenceOnly :: PerspectiveInputBundle -> Bool
  368 weakEvidenceOnly bundle =
  369   null (pibEvidence bundle) || all weakRef (pibEvidence bundle)
  370   where
  371     weakRef = isWeakAcknowledgementText . renderEvidenceRef

  373 contestedActivePressure :: PerspectiveRegistry -> PerspectiveCandidate -> Bool
  374 contestedActivePressure registry candidate =
  375   hasActivePerspective (pcScope candidate) registry
  376     && pcCounterargumentPressure candidate >= 0.65
  377 
  378 rollbackPressure :: PerspectiveRegistry -> PerspectiveCandidate -> Bool
  379 rollbackPressure registry candidate =
  380   hasActivePerspective (pcScope candidate) registry
  381     && hasRollbackCandidate (pcScope candidate) registry
  382     && pcCounterargumentPressure candidate >= 0.50
  383 
  384 hasRollbackCandidate :: PerspectiveScope -> PerspectiveRegistry -> Bool
  385 hasRollbackCandidate scope registry =
  386   case M.lookup scope (prThreads registry) of
  387     Nothing -> False
  388     Just thread -> any (isRollbackCandidate (ptActiveVersion thread)) (ptVersions thread)
  389   where
  390     isRollbackCandidate activeVersion ep =
  391       Just (epVersion ep) /= activeVersion
  392         && epStatus ep `elem` [PerspectiveRevised, PerspectiveContested, PerspectiveActive]
  393 
  394 hasActivePerspective :: PerspectiveScope -> PerspectiveRegistry -> Bool
  395 hasActivePerspective scope registry =
  396   case M.lookup scope (prThreads registry) of
  397     Nothing -> False
  398     Just thread -> case activeEndorsedPerspective thread of
  399       Just ep -> epStatus ep == PerspectiveActive
  400       Nothing -> False

  402 stableCorroboration :: PerspectiveInputBundle -> PerspectiveCandidate -> Bool
  403 stableCorroboration bundle candidate =
  404   length (pibEvidence bundle) >= 2
  405     && pcConfidence candidate >= 0.66
  406     && pcNormativeAlignment candidate >= 0.55
  407     && pcInternalTension candidate <= 0.50

  409 upsertThread :: Int -> PerspectiveRegistry -> PerspectiveInputBundle -> PerspectiveCandidate -> PerspectivePromotionDecision -> PerspectiveStatus -> Maybe Int -> PerspectiveRegistry
  410 upsertThread turn registry bundle candidate decision status endorsedTurn =
  411   let scope = pcScope candidate
  412       oldThread = M.lookup scope (prThreads registry)
  413       perspectiveId = maybe (PerspectiveId ("perspective-" <> T.pack (show (prNextPerspectiveOrdinal registry)))) ptPerspectiveId oldThread
  414       version = PerspectiveVersionId (prNextVersionOrdinal registry)
  415       priorActive = oldThread >>= ptActiveVersion
  416       oldConfidence = maybe 0.0 epConfidence (oldThread >>= activeEndorsedPerspective)
  417       oldNormative = maybe (npVersionId (pibNormativeProfile bundle)) epNormativeProfileVersion (oldThread >>= activeEndorsedPerspective)
  418       endorsed = EndorsedPerspective
  419         { epId = perspectiveId
  420         , epVersion = version
  421         , epScope = scope
  422         , epThesis = pcThesis candidate
  423         , epOrientation = pcOrientation candidate
  424         , epConfidence = pcConfidence candidate
  425         , epNormativeProfileId = npId (pibNormativeProfile bundle)
  426         , epNormativeProfileVersion = npVersionId (pibNormativeProfile bundle)
  427         , epStatus = status
  428         , epCreatedTurn = turn
  429         , epEndorsedTurn = endorsedTurn
  430         , epSupportingClaims = take 8 (pcSupportingClaims candidate)
  431         , epCounterarguments = take 8 (map renderCounterargumentRef (pibCounterarguments bundle))
  432         }
  433       revision = PerspectiveRevisionRecord
  434         { prrFromVersion = priorActive
  435         , prrToVersion = version
  436         , prrTrigger = promotionTrigger decision candidate
  437         , prrEvidenceDelta = fromIntegral (length (pibEvidence bundle))
  438         , prrCounterargumentDelta = pcCounterargumentPressure candidate
  439         , prrConfidenceDelta = pcConfidence candidate - oldConfidence
  440         , prrNormativeDelta = fromIntegral (npVersionId (pibNormativeProfile bundle) - oldNormative)
  441         , prrNormativeProfileVersion = npVersionId (pibNormativeProfile bundle)
  442         , prrDecision = decision
  443         , prrRollbackOf = Nothing
  444         }
  445       oldVersions = maybe [] ptVersions oldThread
  446       oldRevisions = maybe [] ptRevisionHistory oldThread
  447       activeVersion = if status == PerspectiveActive then Just version else priorActive
  448       thread = PerspectiveThread
  449         { ptPerspectiveId = perspectiveId
  450         , ptScope = scope
  451         , ptActiveVersion = activeVersion
  452         , ptVersions = boundVersions registry (endorsed : markPriorRevised status oldVersions)
  453         , ptRevisionHistory = take (prMaxRevisionsPerScope registry) (revision : oldRevisions)
  454         , ptStatus = status
  455         , ptLastUpdatedTurn = turn
  456        }
  457       threads = enforceRegistryBounds registry (M.insert scope thread (prThreads registry))
  458       nextPerspectiveOrdinal = case oldThread of
  459         Nothing -> prNextPerspectiveOrdinal registry + 1
  460         Just _ -> prNextPerspectiveOrdinal registry
  461   in registry
  462       { prThreads = threads
  463       , prNextPerspectiveOrdinal = nextPerspectiveOrdinal
  464       , prNextVersionOrdinal = prNextVersionOrdinal registry + 1
  465       , prLastUpdatedTurn = turn
  466       }

  468 suspendActivePerspective :: Int -> PerspectiveRegistry -> PerspectiveCandidate -> PerspectivePromotionDecision -> PerspectiveRegistry
  469 suspendActivePerspective turn registry candidate decision =
  470   case M.lookup (pcScope candidate) (prThreads registry) of
  471     Nothing -> registry { prLastUpdatedTurn = turn }
  472     Just thread ->
  473       let revision = PerspectiveRevisionRecord
  474             { prrFromVersion = ptActiveVersion thread
  475             , prrToVersion = fromMaybe (PerspectiveVersionId (prNextVersionOrdinal registry)) (ptActiveVersion thread)
  476             , prrTrigger = "counterargument_pressure_suspend"
  477             , prrEvidenceDelta = 0.0
  478             , prrCounterargumentDelta = pcCounterargumentPressure candidate
  479             , prrConfidenceDelta = 0.0
  480             , prrNormativeDelta = 0.0
  481             , prrNormativeProfileVersion = pcNormativeProfileVersion candidate
  482             , prrDecision = decision
  483             , prrRollbackOf = Nothing
  484             }
  485           thread' = thread
  486             { ptActiveVersion = Nothing
  487             , ptVersions = map (setStatusForActive (ptActiveVersion thread) PerspectiveSuspended) (ptVersions thread)
  488             , ptRevisionHistory = take (prMaxRevisionsPerScope registry) (revision : ptRevisionHistory thread)
  489             , ptStatus = PerspectiveSuspended
  490             , ptLastUpdatedTurn = turn
  491             }
  492         in registry
  493             { prThreads = enforceRegistryBounds registry (M.insert (pcScope candidate) thread' (prThreads registry))
  494             , prLastUpdatedTurn = turn
  495             }

  497 rollbackPriorPerspective :: Int -> PerspectiveRegistry -> PerspectiveCandidate -> PerspectivePromotionDecision -> PerspectiveRegistry
  498 rollbackPriorPerspective turn registry candidate decision =
  499   case M.lookup (pcScope candidate) (prThreads registry) of
  500     Nothing -> registry { prLastUpdatedTurn = turn }
  501     Just thread ->
  502       let active = ptActiveVersion thread
  503           remaining = filter (isViableRollbackVersion active) (ptVersions thread)
  504           replacement = case remaining of
  505             x:_ -> Just (epVersion x)
  506             [] -> Nothing
  507           revision = PerspectiveRevisionRecord
  508             { prrFromVersion = active
  509             , prrToVersion = fromMaybe (PerspectiveVersionId (prNextVersionOrdinal registry)) replacement
  510             , prrTrigger = "rollback_prior_promotion"
  511             , prrEvidenceDelta = 0.0
  512             , prrCounterargumentDelta = pcCounterargumentPressure candidate
  513             , prrConfidenceDelta = 0.0
  514             , prrNormativeDelta = 0.0
  515             , prrNormativeProfileVersion = pcNormativeProfileVersion candidate
  516             , prrDecision = decision
  517             , prrRollbackOf = active
  518             }
  519           thread' = thread
  520             { ptActiveVersion = replacement
  521             , ptVersions = map (statusAfterRollback active replacement) (ptVersions thread)
  522             , ptRevisionHistory = take (prMaxRevisionsPerScope registry) (revision : ptRevisionHistory thread)
  523             , ptStatus = maybe PerspectiveWithdrawn (const PerspectiveActive) replacement
  524             , ptLastUpdatedTurn = turn
  525             }
  526       in registry
  527           { prThreads = enforceRegistryBounds registry (M.insert (pcScope candidate) thread' (prThreads registry))
  528           , prLastUpdatedTurn = turn
  529           }

  531 perspectiveMutationRecord :: Int -> PerspectiveRegistry -> PerspectiveInputBundle -> PerspectiveCandidate -> PerspectiveAdmissibility -> PerspectivePromotionDecision -> AdaptiveMutationRecord
  532 perspectiveMutationRecord turn registry bundle candidate admissibility decision = AdaptiveMutationRecord
  533   { amrTurnId = turn
  534   , amrKind = MutPerspective
  535   , amrCause = "perspective:" <> promotionDecisionText decision <> ":" <> admissibilityText admissibility
  536   , amrEvidence = take 8 (map renderEvidenceRef (pibEvidence bundle) <> map renderClaimStanceRef (pibStanceSlice bundle) <> map renderCounterargumentRef (pibCounterarguments bundle))
  537   , amrEvidenceStrength = evidenceStrengthForPerspective bundle decision
  538   , amrConfidence = pcConfidence candidate
  539   , amrBoundedDelta = Just (fromIntegral (prMaxRevisionsPerScope registry))
  540   , amrDecision = adaptiveDecisionForPromotion decision

  546   | decision `elem` [PpdPromoteEndorsed, PpdReviseActive, PpdRollbackPrior] = EvidenceStrong

  561 computeNormativeAlignment :: NormativeProfile -> Double -> ConatusSlice -> Double
  562 computeNormativeAlignment np counterPressure cs =
  563   let safety = M.findWithDefault 1.0 "safety" (npPriorities np)
  564       stability = M.findWithDefault 0.9 "stability" (npPriorities np)
  565       counterWeight = M.findWithDefault 0.7 "counterargument" (npPriorities np)
  566       conflictPenalty = if npConflictPolicy np == "permissive" then 0.08 else 0.18
  567       base = (0.45 * safety) + (0.35 * stability) + (0.20 * csStability cs)
  568       penalty = (counterPressure * counterWeight * 0.30) + conflictPenalty
  569   in clampUnit (base - penalty)

  571 orientationFromScores :: Double -> Double -> Double -> Double -> Text
  572 orientationFromScores confidence counterPressure normativeAlignment internalTension
  573   | counterPressure >= 0.45 || internalTension >= 0.55 = "contested"
  574   | confidence >= 0.72 && normativeAlignment >= 0.60 = "endorsed"
  575   | confidence >= 0.55 = "provisional"

  608 boundVersions :: PerspectiveRegistry -> [EndorsedPerspective] -> [EndorsedPerspective]
  609 boundVersions registry versions =
  610   let active = filter ((== PerspectiveActive) . epStatus) versions
  611       inactive = filter ((/= PerspectiveActive) . epStatus) versions

  614 enforceRegistryBounds :: PerspectiveRegistry -> M.Map PerspectiveScope PerspectiveThread -> M.Map PerspectiveScope PerspectiveThread
  615 enforceRegistryBounds registry threads =
  616   let activeThreads = filter (hasActiveThread . snd) (M.toList threads)
  617       allowedActiveScopes = map fst . take (prMaxActivePerspectives registry) . sortOn (Down . ptLastUpdatedTurn . snd) $ activeThreads
  618   in M.mapWithKey (enforceActiveScope allowedActiveScopes) threads

  620 hasActiveThread :: PerspectiveThread -> Bool

  623 enforceActiveScope :: [PerspectiveScope] -> PerspectiveScope -> PerspectiveThread -> PerspectiveThread
  624 enforceActiveScope allowedScopes scope thread
  625   | scope `elem` allowedScopes = thread
  626   | hasActiveThread thread = suspendThreadForActiveCap thread

  630 suspendThreadForActiveCap thread = thread
