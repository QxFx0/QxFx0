  160 data InputRouteHint = InputRouteHint
  161   { irhType :: !InputRouteType
  162   , irhTag :: !Text
  163   , irhReason :: !Text
  164   , irhRuleScore :: !Double
  165   , irhSemanticScore :: !Double
  166   , irhSyntacticScore :: !Double
  167   , irhEmbeddingScore :: !Double
  168   , irhEvidence :: ![Text]
  169   , irhConfidence :: !Double
  170   } deriving stock (Eq, Show, Generic)
  171     deriving anyclass (NFData)
  172 
  173 instance ToJSON InputRouteHint where
  174   toJSON hint =
  175     object
  176       [ "irhType" .= irhType hint
  177       , "irhTag" .= irhTag hint
  178       , "irhReason" .= irhReason hint
  179       , "irhRuleScore" .= irhRuleScore hint
  180       , "irhSemanticScore" .= irhSemanticScore hint
  181       , "irhSyntacticScore" .= irhSyntacticScore hint
  182       , "irhEmbeddingScore" .= irhEmbeddingScore hint
  183       , "irhEvidence" .= irhEvidence hint
  184       , "irhConfidence" .= irhConfidence hint
  185       ]
  186 
  187 instance FromJSON InputRouteHint where
  188   parseJSON = withObject "InputRouteHint" $ \o -> do
  189     hintType <- o .: "irhType"
  190     hintTag <- o .: "irhTag"
  191     hintReason <- o .: "irhReason"
  192     hintConfidence <- o .: "irhConfidence"
  193     hintRule <- o .:? "irhRuleScore" .!= hintConfidence
  194     hintSemantic <- o .:? "irhSemanticScore" .!= 0.0
  195     hintSyntactic <- o .:? "irhSyntacticScore" .!= 0.0
  196     hintEmbedding <- o .:? "irhEmbeddingScore" .!= 0.0
  197     hintEvidence <- o .:? "irhEvidence" .!= []
  198     pure InputRouteHint
  199       { irhType = hintType
  200       , irhTag = hintTag
  201       , irhReason = hintReason
  202       , irhRuleScore = hintRule
  203       , irhSemanticScore = hintSemantic
  204       , irhSyntacticScore = hintSyntactic
  205       , irhEmbeddingScore = hintEmbedding
  206       , irhEvidence = hintEvidence
  207       , irhConfidence = hintConfidence
  208       }
