   47 ensureSchemaMigrations :: NSQL.Database -> IO ()
   48 ensureSchemaMigrations db = do
   49   sqlSet <- loadBootstrapSqlSet
   50   -- 1. Run canonical schema (IF NOT EXISTS — safe for old and new DBs)
   51   execBootstrapSql db ("schema.sql", bootstrapSchema sqlSet)
   52   -- 2. Determine current version state and act accordingly
   53   versionStatus <- readSchemaVersion db
   54   case versionStatus of
   55     SchemaVersionMissingTable -> do
   56       isV1 <- isV1Shape db
   57       if isV1
   58         then do
   59           runPendingMigrations db 1
   60           seedAfterMigration sqlSet db
   61         else do
   62           isV2 <- isV2Shape db
   63           if isV2
   64             then do
   65               insertSchemaVersion db currentSchemaVersion currentSchemaDescription
   66               seedAfterMigration sqlSet db
   67             else do
   68               -- Fresh DB: no tables and no schema_version
   69               insertSchemaVersion db currentSchemaVersion currentSchemaDescription
   70               seedAfterMigration sqlSet db
   71     SchemaVersionEmpty -> do
   72       isV1 <- isV1Shape db
   73       if isV1
   74         then do
   75           runPendingMigrations db 1
   76           seedAfterMigration sqlSet db
   77         else do
   78           isV2 <- isV2Shape db
   79           if isV2
   80             then do
   81               insertSchemaVersion db currentSchemaVersion currentSchemaDescription
   82               seedAfterMigration sqlSet db
   83             else
   84               throwQxFx0
   85                 (SQLiteError "schema_version table exists but is empty, and database shape is not v1 or v2")
   86     SchemaVersionPresent 1 -> do
   87       runPendingMigrations db 1
   88       seedAfterMigration sqlSet db
   89     SchemaVersionPresent v
   90       | v == currentSchemaVersion ->
   91           pure ()
   92       | v < currentSchemaVersion -> do
   93           runPendingMigrations db v
   94           seedAfterMigration sqlSet db
   95       | otherwise ->
   96           throwQxFx0
   97             (SQLiteError $
   98               "schema_version mismatch: expected "
   99                 <> T.pack (show currentSchemaVersion)
  100                 <> ", found "
  101                 <> T.pack (show v)
  102                 <> ". Downgrade is not supported.")
  103   -- 3. Post-bootstrap validation
  104   contractResult <- checkSchemaContract db
  105   case contractResult of
  106     SchemaContractOk _ -> pure ()
  107     SchemaContractFreshBootstrapable -> pure ()
  108     SchemaContractVersionBehind expected actual ->
  109       throwQxFx0 (SQLiteError $
  110         "schema version behind: expected " <> T.pack (show expected)
  111         <> ", found " <> T.pack (show actual))
  112     SchemaContractMissingTable t ->
  113       throwQxFx0 (SQLiteError $ "schema contract failed: missing table " <> t)
  114     SchemaContractMissingColumns t cols ->
  115       throwQxFx0 (SQLiteError $
  116         "schema contract failed: table " <> t <> " missing columns: "
  117         <> T.intercalate ", " cols)
  118     SchemaContractMissingIndex ix ->
  119       throwQxFx0 (SQLiteError $ "schema contract failed: missing index " <> ix)
  120     SchemaContractMissingTrigger tr ->
  121       throwQxFx0 (SQLiteError $ "schema contract failed: missing trigger " <> tr)
  122     SchemaContractMissingFTS fts ->
  123       throwQxFx0 (SQLiteError $ "schema contract failed: missing fts table " <> fts)
  124     SchemaContractInconsistent reason ->
  125       throwQxFx0 (SQLiteError $ "schema contract inconsistent: " <> reason)
  126     SchemaContractQueryFailed msg ->
  127       throwQxFx0 (SQLiteError $ "schema contract query failed: " <> msg)

  129 seedAfterMigration :: BootstrapSqlSet -> NSQL.Database -> IO ()
  130 seedAfterMigration sqlSet db = do
  131   seedBootstrapSql db "semantic_clusters" ("seed_clusters.sql", bootstrapClusters sqlSet)
  132   seedBootstrapSql db "realization_templates" ("seed_templates.sql", bootstrapTemplates sqlSet)
  133   seedBootstrapSql db "identity_claims" ("seed_identity.sql", bootstrapIdentity sqlSet)
  134 
  135 data BootstrapSqlSet = BootstrapSqlSet
  136   { bootstrapSchema :: !Text
  137   , bootstrapClusters :: !Text
  138   , bootstrapTemplates :: !Text
  139   , bootstrapIdentity :: !Text
  140   }
  141 
  142 loadBootstrapSqlSet :: IO BootstrapSqlSet
  143 loadBootstrapSqlSet = do
  144   pathsResult <- tryQxFx0 resolveResourcePaths
  145   case pathsResult of
  146     Left err ->
  147       allowEmbeddedSqlFallback
  148         ("resolveResourcePaths failed: " <> T.pack (show err))
  149         fallbackBootstrapSqlSet
  150     Right paths ->
  151       BootstrapSqlSet
  152         <$> loadSqlFileOrFallback (rpSchemaSql paths) schemaSQL
  153         <*> loadSqlFileOrFallback (rpSeedClusters paths) seedClustersSQL
  154         <*> loadSqlFileOrFallback (rpSeedTemplates paths) seedTemplatesSQL
  155         <*> loadSqlFileOrFallback (rpSeedIdentity paths) seedIdentitySQL
  156 
  157 fallbackBootstrapSqlSet :: BootstrapSqlSet
  158 fallbackBootstrapSqlSet =
  159   BootstrapSqlSet
  160     { bootstrapSchema = schemaSQL
  161     , bootstrapClusters = seedClustersSQL
  162     , bootstrapTemplates = seedTemplatesSQL
  163     , bootstrapIdentity = seedIdentitySQL
  164     }
  165 
  166 loadSqlFileOrFallback :: FilePath -> Text -> IO Text
  167 loadSqlFileOrFallback path fallback = do
  168   exists <- doesFileExist path
  169   if exists
  170     then TIO.readFile path `catchIO` onFailure
  171     else allowEmbeddedSqlFallback ("SQL file not found: " <> T.pack path) fallback
  172   where
  173     onFailure :: IOException -> IO Text
  174     onFailure err =
  175       allowEmbeddedSqlFallback
  176         ("SQL file read failed: " <> T.pack path <> " (" <> T.pack (show err) <> ")")
  177         fallback
  178 
  179 allowEmbeddedSqlFallback :: Text -> a -> IO a
  180 allowEmbeddedSqlFallback reason fallbackValue = do
  181   mFallback <- lookupEnv "QXFX0_ALLOW_EMBEDDED_SQL_FALLBACK"
  182   case mFallback of
  183     Just "1" -> pure fallbackValue
  184     _ ->
  185       throwQxFx0
  186         (SQLiteError $
  187           "embedded SQL fallback disabled; canonical spec/sql is required: " <> reason
  188         )

  190 insertSchemaVersion :: NSQL.Database -> Int -> Text -> IO ()
  191 insertSchemaVersion db version description = do
  192   mStmt <- NSQL.prepare db "INSERT OR REPLACE INTO schema_version(version, description) VALUES(?, ?)"
  193   case mStmt of
  194     Left err -> throwQxFx0 (SQLiteError $ "schema version insert prepare failed: " <> err)
  195     Right stmt -> do
  196       _ <- NSQL.bindInt stmt 1 version
  197       _ <- NSQL.bindText stmt 2 description
  198       _ <- NSQL.step stmt
  199       _ <- tryQxFx0 (NSQL.finalize stmt)
  200       pure ()

  202 -- | Explicit, idempotent migration step.
  203 data MigrationStep
  204   = EnsureColumn !Text !Text !Text       -- ^ table, column, definition
  205   deriving stock (Eq, Show)
  206 
  207 -- | Run all pending migrations from (currentVersion + 1) up to currentSchemaVersion.
  208 --   Wrapped in a transaction: version marker is only committed after validation succeeds.
  209 runPendingMigrations :: NSQL.Database -> Int -> IO ()
  210 runPendingMigrations db currentVersion = do
  211   let steps = resolveMigrationSteps currentVersion
  212   beginResult <- NSQL.execSql db "BEGIN IMMEDIATE;"
  213   case beginResult of
  214     Left err ->
  215       throwQxFx0 (SQLiteError $ "migration tx begin failed: " <> err)
  216     Right _ ->
  217       pure ()
  218   -- Run migration steps
  219   flip finally (rollbackOnException db) $ do
  220     mapM_ (runMigrationStep db) steps
  221     -- Validate before committing
  222     contractResult <- validateSchemaContractTablesAndColumns db
  223     case contractResult of
  224       SchemaContractOk _ -> do
  225         insertSchemaVersion db currentSchemaVersion currentSchemaDescription
  226         commitResult <- NSQL.execSql db "COMMIT;"
  227         case commitResult of
  228           Left err ->
  229             throwQxFx0 (SQLiteError $ "migration tx commit failed: " <> err)
  230           Right _ ->
  231             pure ()
  232       other -> do
  233         rollbackResult <- NSQL.execSql db "ROLLBACK;"
  234         case rollbackResult of
  235           Left rbErr ->
  236             throwQxFx0 (SQLiteError $
  237               "migration validation failed: " <> renderSchemaContractResult other
  238               <> "; rollback also failed: " <> rbErr)
  239           Right _ ->
  240             throwQxFx0 (SQLiteError $
  241               "migration validation failed: " <> renderSchemaContractResult other)
  242 
  243 rollbackOnException :: NSQL.Database -> IO ()
  244 rollbackOnException db = do
  245   _ <- NSQL.execSql db "ROLLBACK;"
  246   pure ()

  248 resolveMigrationSteps :: Int -> [MigrationStep]
  249 resolveMigrationSteps fromVersion
  250   = concat
  251       [ if fromVersion < 2
  252           then
  253             [ EnsureColumn "turn_quality" "warranted_mode" "TEXT NOT NULL DEFAULT 'ConditionallyWarranted'"
  254             , EnsureColumn "turn_quality" "decision_disposition" "TEXT NOT NULL DEFAULT 'advisory'"
  255             , EnsureColumn "turn_quality" "shadow_snapshot_id" "TEXT NOT NULL DEFAULT ''"
  256             , EnsureColumn "turn_quality" "shadow_divergence_kind" "TEXT NOT NULL DEFAULT 'none'"
  257             , EnsureColumn "turn_quality" "replay_trace_json" "TEXT NOT NULL DEFAULT '{}'"
  258             ]
  259           else []
  260       , if fromVersion < 3
  261           then
  262             [ EnsureColumn "shadow_divergence_log" "shadow_snapshot_id" "TEXT NOT NULL DEFAULT ''"
  263             , EnsureColumn "shadow_divergence_log" "shadow_divergence_kind" "TEXT NOT NULL DEFAULT 'none'"
  264             ]
  265           else []
  266       ]

  268 runMigrationStep :: NSQL.Database -> MigrationStep -> IO ()
  269 runMigrationStep db step = case step of
  270   EnsureColumn table col def -> do
  271     missing <- columnMissing db table col
  272     if missing
  273       then do
  274         let sql = "ALTER TABLE " <> table <> " ADD COLUMN " <> col <> " " <> def
  275         result <- NSQL.execSql db sql
  276         case result of
  277           Left err ->
  278             throwQxFx0
  279               (SQLiteError $
  280                 "migration_failed stage=ensure_column table=" <> table
  281                 <> " column=" <> col <> " sqlite=" <> err)
  282           Right _ -> pure ()
  283       else pure ()

  285 execBootstrapSql :: NSQL.Database -> (FilePath, Text) -> IO ()
  286 execBootstrapSql db (label, sqlText) = do
  287   result <- NSQL.execSql db sqlText
  288   case result of
  289     Left err -> throwQxFx0 (SQLiteError $ T.pack ("bootstrap SQL failed for " <> label <> ": ") <> err)
  290     Right _ -> pure ()

  292 seedBootstrapSql :: NSQL.Database -> Text -> (FilePath, Text) -> IO ()
  293 seedBootstrapSql db tableName payload = do
  294   query <- tableHasRowsQuery tableName
  295   hasRows <- tableHasRows db query
  296   if hasRows
  297     then pure ()
  298     else execBootstrapSql db payload
  299 
  300 tableHasRowsQuery :: Text -> IO Text
  301 tableHasRowsQuery tableName =
  302   case tableName of
  303     "semantic_clusters" -> pure "SELECT 1 FROM semantic_clusters LIMIT 1"
  304     "realization_templates" -> pure "SELECT 1 FROM realization_templates LIMIT 1"
  305     "identity_claims" -> pure "SELECT 1 FROM identity_claims LIMIT 1"
  306     _ ->
  307       throwQxFx0
  308         (SQLiteError
  309           ("unsupported bootstrap seed table name: " <> tableName))
  310 
  311 tableHasRows :: NSQL.Database -> Text -> IO Bool
  312 tableHasRows db sql = do
  313   withPreparedStatement db sql "tableHasRows query" NSQL.stepRow
  314 
  315 withPreparedStatement :: NSQL.Database -> Text -> Text -> (NSQL.Statement -> IO a) -> IO a
  316 withPreparedStatement db sql context action = do
  317   mStmt <- NSQL.prepare db sql
  318   case mStmt of
  319     Left err ->
  320       throwQxFx0 (SQLiteError (context <> " prepare failed: " <> err))
  321     Right stmt ->
  322       action stmt `finally` finalizeBestEffort stmt
  323 
  324 finalizeBestEffort :: NSQL.Statement -> IO ()
  325 finalizeBestEffort stmt = do
  326   _ <- tryQxFx0 (NSQL.finalize stmt)
  327   pure ()
