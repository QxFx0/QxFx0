   24 type Embedding = V.Vector Float

   27 data EmbeddingBackend
   28   = EmbeddingBackendLocalDeterministic
   29   | EmbeddingBackendRemoteHTTP
   30   deriving stock (Eq, Show)
   31 
   32 data EmbeddingQuality
   33   = EmbeddingQualityHeuristic
   34   | EmbeddingQualityModeled
   35   deriving stock (Eq, Show)
   36 
   37 data EmbeddingHealth = EmbeddingHealth
   38   { ehBackend :: !EmbeddingBackend
   39   , ehQuality :: !EmbeddingQuality
   40   , ehExplicit :: !Bool
   41   , ehOperational :: !Bool
   42   , ehStrictReady :: !Bool
   43   }
   44   deriving stock (Eq, Show)
   45 
   46 data EmbeddingSource
   47   = EmbeddingRemote
   48   | EmbeddingLocalDeterministic
   49   | EmbeddingLocalImplicit

   51   deriving stock (Eq, Show)
   52 
   53 data EmbeddingResult = EmbeddingResult
   54   { erEmbedding :: !Embedding
   55   , erSource :: !EmbeddingSource
   56   }
   57   deriving stock (Eq, Show)

   63 embeddingQualityText :: EmbeddingQuality -> Text

   67 embeddingSourceText :: EmbeddingSource -> Text
   68 embeddingSourceText EmbeddingRemote = "remote_http"
   69 embeddingSourceText EmbeddingLocalDeterministic = "local_deterministic"
   70 embeddingSourceText EmbeddingLocalImplicit = "local_implicit"
   71 embeddingSourceText EmbeddingRemoteFailureLocalFallback = "remote_failure_local_fallback"
   72 
   73 embeddingSourceQuality :: EmbeddingSource -> EmbeddingQuality
   74 embeddingSourceQuality source = case source of
   75   EmbeddingRemote -> EmbeddingQualityModeled
   76   EmbeddingLocalDeterministic -> EmbeddingQualityHeuristic
   77   EmbeddingLocalImplicit -> EmbeddingQualityHeuristic
   78   EmbeddingRemoteFailureLocalFallback -> EmbeddingQualityHeuristic
