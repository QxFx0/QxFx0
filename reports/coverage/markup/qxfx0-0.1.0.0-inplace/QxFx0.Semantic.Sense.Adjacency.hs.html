   10 adjacentOperators :: SenseOperator -> [SenseOperator]
   11 adjacentOperators op = case op of
   12   OpDefine -> [OpDistinguish, OpGround, OpClarify]
   13   OpGround -> [OpDefine, OpExplainCause, OpClarify]
   14   OpDistinguish -> [OpExplainCause, OpConstrain, OpClarify]
   15   OpExplainCause -> [OpConstrain, OpExplainPurpose, OpGround]
   16   OpExplainPurpose -> [OpConstrain, OpNextStep]
   17   OpReflect -> [OpClarify, OpConstrain]
   18   OpConstrain -> [OpClarify, OpNextStep, OpGround]
   19   OpRepair -> [OpClarify, OpGround]
   20   OpClarify -> [OpGround, OpDefine, OpDeepen]
   21   OpDeepen -> [OpReflect, OpNextStep]
   22   OpNextStep -> [OpGround]
   23 
   24 allowedTransition :: SenseOperator -> SenseOperator -> Bool
   25 allowedTransition from to = to == from || to `elem` adjacentOperators from
