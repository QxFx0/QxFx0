   79 normalizeClaimText :: Text -> Text
   80 normalizeClaimText =
   81   T.intercalate " " . T.words . T.toLower . stripPunct
   82   where
   83     stripPunct t =
   84       let noMd = T.replace "**" "" . T.replace "*" "" . T.replace "#" "" . T.replace "`" "" $ t
   85       in T.filter (\c -> not (isPunctuation c) || c == '-' || c == '\'') noMd
   86 
   87 inferUserState :: [ClusterDef] -> Text -> UserState
   88 inferUserState clusters rawText =
   89   let tokens = tokenizeDomainText rawText
   90       matching = filter (\c -> any (containsDomainPhrase tokens) (cdKeywords c)) clusters
   91       best =
   92         case matching of
   93           [] -> Nothing
   94           (c:cs) -> Just (foldl' (\a b -> if cdPriority b > cdPriority a then b else a) c cs)
   95       tone =
   96         case best of
   97           Nothing -> defaultUserTone
   98           Just c -> cdName c
   99       (needLayer, reg) = inferLayerReg tokens
  100       readiness =
  101         case best of
  102           Nothing -> defaultUserReadiness
  103           Just c -> min 1.0 (cdPriority c)
  104   in UserState tone needLayer readiness reg
  105   where
  106     inferLayerReg tokens
  107       | containsAnyDomainPhrase tokens ["почему", "зачем", "как", "что такое"] =
  108           (ContentLayer, Search)
  109       | containsAnyDomainPhrase tokens ["помоги", "подожди", "одиноко"] =
  110           (ContactLayer, Contact)
  111       | containsAnyDomainPhrase tokens ["точно", "конечно", "безусловно"] =
  112           (ContentLayer, Anchor)
  113       | otherwise =
  114           (ContentLayer, Neutral)
  115 
  116 tokenizeDomainText :: Text -> [Text]
  117 tokenizeDomainText =
  118   filter (not . T.null) . T.words . T.map normalizeChar . T.toLower
  119   where
  120     normalizeChar ch
  121       | isAlphaNum ch = ch
  122       | otherwise = ' '
  123 
  124 containsDomainPhrase :: [Text] -> Text -> Bool
  125 containsDomainPhrase haystack phrase =
  126   let needle = tokenizeDomainText phrase
  127   in not (null needle) && any (isPrefixOf needle) (tails haystack)
  128 
  129 containsAnyDomainPhrase :: [Text] -> [Text] -> Bool
  130 containsAnyDomainPhrase haystack = any (containsDomainPhrase haystack)
