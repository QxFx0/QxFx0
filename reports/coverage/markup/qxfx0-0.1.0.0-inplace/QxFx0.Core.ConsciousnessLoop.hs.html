   78 runConsciousnessLoop :: ConsciousnessLoop -> SemanticInput -> Double -> Double -> (ConsciousnessLoop, Text)
   79 runConsciousnessLoop loop semanticInput humanTheta resonance =
   80   let model = clModel loop
   81       turn  = clDialogueTurn loop + 1
   82       inputText = siRawInput semanticInput
   83       output = kernelPulse (cmKernel model) semanticInput humanTheta resonance turn
   84       skillN = koSelectedSkill output
   85       desireNs = koActiveDesires output
   86       conflicts = koConflicts output
   87       (newBg, mSurfacing) = runBackgroundCycle (clBackground loop) skillN desireNs conflicts
   88       model' = interpretOutput model output inputText
   89       narrative = consciousnessToNarrative model' output
   90       loop' = loop
   91         { clModel         = model'
   92         , clLastOutput    = Just output
   93         , clLastNarrative = Just narrative
   94         , clBackground    = newBg
   95         , clLastSurfacing = mSurfacing
   96         , clDialogueTurn  = turn
   97         }
   98       fragment = narrativeToPromptFragment narrative
   99   in (loop', fragment)

  101 -- | Salience-aware variant of 'runConsciousnessLoop'. Runs the
  102 -- consciousness loop unchanged, then weights the resulting
  103 -- narrative fragment under the supplied 'Salience' verdict via
  104 -- 'applySalienceToNarrativeFragment'.
  105 --
  106 -- The 'ConsciousnessLoop' itself (model state, previous output,
  107 -- current narrative, background, surfacing, turn counter) is
  108 -- updated exactly as in 'runConsciousnessLoop' regardless of
  109 -- salience.
  110 -- Salience only weights the contribution of the narrative to the
  111 -- /prompt/, never the kernel state. This keeps the consciousness
  112 -- model's internal evolution decoupled from the controller, which
  113 -- is the strict reading of ADR-0010 \u00a75 anti-correlation discipline
  114 -- on this channel.
  115 --
  116 -- Behaviour at neutral salience ('salienceConfidence' \u2265 the
  117 -- threshold and 'salienceDriver' /= 'DrivenByConatusGate') is
  118 -- identity, so any call site that does not yet supply rich Field
  119 -- signals observes no behaviour change.
  120 runConsciousnessLoopWithSalience
  121   :: Salience
  122   -> ConsciousnessLoop
  123   -> SemanticInput
  124   -> Double
  125   -> Double
  126   -> (ConsciousnessLoop, Text)
  127 runConsciousnessLoopWithSalience salience loop semanticInput humanTheta resonance =
  128   let (loop', fragment) = runConsciousnessLoop loop semanticInput humanTheta resonance
  129       weighted          = applySalienceToNarrativeFragment salience fragment
  130   in (loop', weighted)
  131 
  132 -- | Confidence threshold below which the narrative fragment is
  133 -- suppressed (replaced by an empty 'Text') by
  134 -- 'applySalienceToNarrativeFragment'.
  135 --
  136 -- The threshold is conservative: only configurations where the
  137 -- salience drivers are strongly dispersed (no clear winner among
  138 -- the five drivers) suppress the narrative. At neutral signals
  139 -- ('emptyField' + placeholder Conatus), 'salienceConfidence' is
  140 -- @1.0@ and the threshold is never tripped. See the doc on
  141 -- 'QxFx0.Self.Salience.computeConfidence' for the dispersion
  142 -- semantics.
  143 narrativeConfidenceThreshold :: Double
  144 narrativeConfidenceThreshold = 0.25
  145 
  146 -- | Weight a narrative fragment by a 'Salience' verdict.
  147 --
  148 -- Decision rule (matches ADR-0010 \u00a74 and \u00a75):
  149 --
  150 --   * 'DrivenByConatusGate' (structural risk) overrides every
  151 --     other verdict and suppresses the narrative entirely. Under
  152 --     structural risk, the controller refuses to let the
  153 --     consciousness narrative pollute the prompt.
  154 --   * 'salienceConfidence' below 'narrativeConfidenceThreshold'
  155 --     suppresses the narrative as well. The drivers disagree too
  156 --     much for the verdict to be trusted, so the contribution is
  157 --     dropped.
  158 --   * Otherwise the fragment passes through unchanged. There is
  159 --     no fractional weighting on the text \u2014 contribution is
  160 --     binary at this layer (full or zero).
  161 --
  162 -- The downstream call sites already gate on @T.null fragment@
  163 -- before adding the fragment to the prompt, so an empty fragment
  164 -- naturally disappears from the prompt without further plumbing.
  165 applySalienceToNarrativeFragment :: Salience -> Text -> Text
  166 applySalienceToNarrativeFragment salience fragment =
  167   case salienceDriver salience of
  168     DrivenByConatusGate -> T.empty
  169     _
  170       | salienceConfidence salience < narrativeConfidenceThreshold -> T.empty
  171       | otherwise                                                   -> fragment

  173 updateAfterResponse :: ConsciousnessLoop -> ResponseObservation -> ConsciousnessLoop
  174 updateAfterResponse loop observation =
  175   let model = clModel loop
  176       cs = cmConscious model
  177       si = csSelfInterp cs
  178       obs = observeOwnResponse observation
  179       newSI = si { siObservedPatterns = take 5 (obs : siObservedPatterns si) }
  180       model' = model { cmConscious = cs { csSelfInterp = newSI } }
  181   in loop { clModel = model' }
  182 
  183 addCoreSignal :: Text -> ConsciousnessLoop -> ConsciousnessLoop
  184 addCoreSignal sig loop =
  185   let model = clModel loop
  186       cs = cmConscious model
  187       si = csSelfInterp cs
  188       newSI = si { siObservedPatterns = take 5 (sig : siObservedPatterns si) }
  189   in loop { clModel = model { cmConscious = cs { csSelfInterp = newSI } } }

  191 observeOwnResponse :: ResponseObservation -> Text
  192 observeOwnResponse observation =
  193   let response = roSurfaceText observation
  194       tokens = tokenizeKeywordText response
  195       wc = length (T.words response)
  196   in if wc <= 8 then observeShortResponse
  197      else if containsAnyKeywordPhrase tokens surfacingMarkerKeywords then observeSurfacing
  198      else if containsAnyKeywordPhrase tokens boundaryMarkerKeywords then observeBoundary
  199      else if roQuestionLike observation then observeQuestion
  200      else observeDefault
