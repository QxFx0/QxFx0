   69 runConsciousnessLoop :: ConsciousnessLoop -> SemanticInput -> Double -> Double -> (ConsciousnessLoop, Text)
   70 runConsciousnessLoop loop semanticInput humanTheta resonance =
   71   let model = clModel loop
   72       turn  = clDialogueTurn loop + 1
   73       inputText = siRawInput semanticInput
   74       output = kernelPulse (cmKernel model) semanticInput humanTheta resonance turn
   75       skillN = koSelectedSkill output
   76       desireNs = koActiveDesires output
   77       conflicts = koConflicts output
   78       (newBg, mSurfacing) = runBackgroundCycle (clBackground loop) skillN desireNs conflicts
   79       model' = interpretOutput model output inputText
   80       narrative = consciousnessToNarrative model' output
   81       loop' = loop
   82         { clModel         = model'
   83         , clLastOutput    = Just output
   84         , clLastNarrative = Just narrative
   85         , clBackground    = newBg
   86         , clLastSurfacing = mSurfacing
   87         , clDialogueTurn  = turn
   88         }
   89       fragment = narrativeToPromptFragment narrative
   90   in (loop', fragment)

   92 updateAfterResponse :: ConsciousnessLoop -> ResponseObservation -> ConsciousnessLoop
   93 updateAfterResponse loop observation =
   94   let model = clModel loop
   95       cs = cmConscious model
   96       si = csSelfInterp cs
   97       obs = observeOwnResponse observation
   98       newSI = si { siObservedPatterns = take 5 (obs : siObservedPatterns si) }
   99       model' = model { cmConscious = cs { csSelfInterp = newSI } }
  100   in loop { clModel = model' }

  102 addCoreSignal :: Text -> ConsciousnessLoop -> ConsciousnessLoop
  103 addCoreSignal sig loop =
  104   let model = clModel loop
  105       cs = cmConscious model
  106       si = csSelfInterp cs
  107       newSI = si { siObservedPatterns = take 5 (sig : siObservedPatterns si) }
  108   in loop { clModel = model { cmConscious = cs { csSelfInterp = newSI } } }

  110 observeOwnResponse :: ResponseObservation -> Text
  111 observeOwnResponse observation =
  112   let response = roSurfaceText observation
  113       tokens = tokenizeKeywordText response
  114       wc = length (T.words response)
  115   in if wc <= 8 then observeShortResponse
  116      else if containsAnyKeywordPhrase tokens surfacingMarkerKeywords then observeSurfacing
  117      else if containsAnyKeywordPhrase tokens boundaryMarkerKeywords then observeBoundary
  118      else if roQuestionLike observation then observeQuestion
  119      else observeDefault
