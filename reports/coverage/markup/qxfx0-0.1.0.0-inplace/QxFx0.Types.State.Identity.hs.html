   93 data IdentityState = IdentityState
   94   { idsEgo :: !EgoState
   95   , idsIdentityClaims :: ![IdentityClaimRef]
   96   , idsOrbitalMemory :: !OrbitalMemory
   97   , idsLastGuardReport :: !(Maybe IdentityGuardReport)
   98   } deriving stock (Eq, Show, Generic)
   99     deriving anyclass (NFData)
  100 
  101 instance ToJSON IdentityState where
  102   toJSON ids = object
  103     [ "ego" .= idsEgo ids
  104     , "identityClaims" .= idsIdentityClaims ids
  105     , "orbitalMemory" .= idsOrbitalMemory ids
  106     , "lastGuardReport" .= idsLastGuardReport ids
  107     ]
  108 
  109 instance FromJSON IdentityState where
  110   parseJSON = withObject "IdentityState" $ \o -> IdentityState
  111     <$> o .: "ego"
  112     <*> o .: "identityClaims"
  113     <*> o .: "orbitalMemory"
  114     <*> o .:? "lastGuardReport" .!= Nothing
  115 
  116 emptyIdentityState :: IdentityState
  117 emptyIdentityState = IdentityState
  118   { idsEgo = emptyEgoState
  119   , idsIdentityClaims = []
  120   , idsOrbitalMemory = emptyOrbitalMemory
  121   , idsLastGuardReport = Nothing
  122   }
