   37 data SenseAxis

   77 data SenseVector = SenseVector
   78   { svAnchor :: !SemanticNodeId
   79   , svLexicalFamily :: !(Maybe FamilyId)
   80   , svEvidenceNodes :: ![SemanticNodeId]
   81   , svAxes :: !(Map SenseAxis Double)
   82   , svOperators :: ![SenseOperator]
   83   , svPolarity :: !SensePolarity
   84   , svAgent :: !(Maybe SemanticNodeId)
   85   , svTarget :: !(Maybe SemanticNodeId)
   86   , svConfidence :: !Double
   87   } deriving stock (Eq, Show, Generic)
   88     deriving anyclass (NFData, FromJSON, ToJSON)

  137 renderSenseAxis :: SenseAxis -> Text
  138 renderSenseAxis axis = case axis of
  139   AxIdentity -> "identity"
  140   AxCause -> "cause"
  141   AxPurpose -> "purpose"
  142   AxContrast -> "contrast"
  143   AxBoundary -> "boundary"
  144   AxAction -> "action"
  145   AxState -> "state"
  146   AxKnowledge -> "knowledge"
  147   AxNorm -> "norm"
  148   AxTime -> "time"
  149   AxPossibility -> "possibility"
  150   AxSelf -> "self"
  151   AxOther -> "other"
  152   AxRepair -> "repair"
  153 
  154 renderSenseOperator :: SenseOperator -> Text
  155 renderSenseOperator op = case op of
  156   OpDefine -> "define"
  157   OpGround -> "ground"
  158   OpDistinguish -> "distinguish"
  159   OpExplainCause -> "explain_cause"
  160   OpExplainPurpose -> "explain_purpose"
  161   OpReflect -> "reflect"
  162   OpConstrain -> "constrain"
  163   OpRepair -> "repair"
  164   OpClarify -> "clarify"
  165   OpDeepen -> "deepen"
  166   OpNextStep -> "next_step"
