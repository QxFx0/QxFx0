  100 replayTraceLoadStatus :: ReplayTraceLoad -> Text
  101 replayTraceLoadStatus traceLoad =
  102   case traceLoad of
  103     ReplayTraceMissing -> "missing"
  104     ReplayTraceDbError reason -> "db_error:" <> reason
  105     ReplayTraceDecodeError reason -> "decode_error:" <> reason
  106     ReplayTraceSchemaError reason -> "schema_error:" <> reason
  107     ReplayTraceLoaded summary -> rtsLoadStatus summary
  108 
  109 replayTraceSummaryMaybe :: ReplayTraceLoad -> Maybe ReplayTraceSummary
  110 replayTraceSummaryMaybe traceLoad =
  111   case traceLoad of
  112     ReplayTraceLoaded summary -> Just summary
  113     _ -> Nothing

  136 governanceSummaryLines :: SystemState -> [Text]
  137 governanceSummaryLines ss =
  138   let history = ssGovernanceHistory ss
  139       liveProjection = ssGovernanceProjection ss
  140       liveRegistry = ssPerspectiveRegistry ss
  141       rebuildResult = rebuildGovernedViews ss
  142       latestProvenance = latestGovernanceProvenance history
  143       (rebuildStatus, summaryProjection, staleCount) =
  144         case rebuildResult of
  145           Right (rebuiltRegistry, rebuiltProjection)
  146             | rebuiltProjection == liveProjection -> ("ok", rebuiltProjection, 0)
  147             | otherwise ->
  148                 ( "mismatch"
  149                 , rebuiltProjection
  150                 , registryDifferenceCount liveRegistry rebuiltRegistry
  151                 )
  152           Left _ -> ("unavailable", liveProjection, M.size (prThreads liveRegistry))
  153   in
  154     [ "governance_events_count: " <> renderValue (length history)
  155     , "governance_fingerprint: " <> governanceHistoryFingerprint history
  156     , "governance_rebuild_status: " <> rebuildStatus
  157     , "active_perspectives_count: " <> renderValue (length (gpActivePerspectiveProjections summaryProjection))
  158     , "governed_refs_count: " <> renderValue (length (gpGovernedRefs summaryProjection))
  159     , "governance_denied_count: " <> renderValue (countGovernanceLifecycle GlsDenied history)
  160      , "governance_rollback_count: " <> renderValue (countGovernanceLifecycle GlsRolledBack history)
  161     , "governance_stale_count: " <> renderValue staleCount
  162     , "governance_freeze_status: " <> governanceFreezeStatus history
  163     , "governance_authority_status: " <> renderEpistemicStatus (governanceAuthorityStatus ss rebuildStatus)
  164     , "governance_runtime_fault: " <> maybe "none" (T.pack . show) (ssGovernanceRuntimeFault ss)
  165     , "latest_governed_subject: " <> maybe "n/a" snd latestProvenance
  166     , "latest_governance_reason_tag: " <> maybe "n/a" fst latestProvenance
  167     , "salience_field_contract_status: " <> salienceFieldContractStatus ss
  168     , "plan_narrative_tone_contract_status: " <> planNarrativeToneContractStatus ss
  169     , "bayesian_contract_status: " <> bayesianContractStatus ss
  170     ]

  172 stateSummaryLines :: Session -> IO [Text]
  173 stateSummaryLines session = do
  174   traceLoad <- loadLatestReplayTrace session
  175   let ss = sessSystemState session
  176       latestTrace = replayTraceSummaryMaybe traceLoad
  177       renderValue :: Show a => a -> Text
  178       renderValue = T.pack . show
  179       essenceModeTag =
  180         case ssEssence ss of
  181           EssenceUncommitted _ -> "uncommitted"
  182           EssenceCommitted _ c -> "committed:" <> renderEssenceMode (ecMode c)
  183       recoveryCauseTag = maybe "n/a" id (rtsRecoveryCause =<< latestTrace)
  184       recoveryStrategyTag = maybe "n/a" id (rtsRecoveryStrategy =<< latestTrace)
  185       shadowSeverityTag =
  186         case rtsShadowSeverity =<< latestTrace of
  187           Just raw -> raw
  188           Nothing -> "n/a"
  189       gradientTag = maybe "n/a" renderGradientTag (latestTrace >>= (parseGradientFromEvidence . rtsRecoveryEvidence))
  190       truthContractTag = maybe "n/a" id (latestTrace >>= rtsTruthContractStatus)
  191       authorityClassTag = maybe "n/a" id (latestTrace >>= rtsAuthorityClass)
  192       assemblyPathTag = maybe "n/a" id (latestTrace >>= rtsAssemblyPath)
  193       replayProvenanceTag = maybe "n/a" id (latestTrace >>= rtsReplayProvenanceStatus)
  194       surfaceProvenanceTag = maybe "n/a" id (latestTrace >>= rtsSurfaceProvenance)
  195       contractProvenanceTag = maybe "n/a" id (latestTrace >>= rtsContractProvenance)
  196       replayTraceLoadTag = replayTraceLoadStatus traceLoad
  197   pure
  198     $ [ "STATE_BEGIN"
  199       , "session_id: " <> sessSessionId session
  200       , "state_origin: " <> renderStateOrigin (sessStateOrigin session)
  201       , "turns: " <> renderValue (ssTurnCount ss)
  202       , "output_mode: " <> renderRuntimeOutputMode (sessOutputMode session)
  203       , "atom_trace_ema: " <> renderValue (atCurrentLoad (ssTrace ss))
  204       , "last_family: " <> renderValue (ssLastFamily ss)
  205       , "last_topic: " <> ssLastTopic ss
  206       , "ego_agency: " <> renderValue (egoAgency (ssEgo ss))
  207       , "ego_tension: " <> renderValue (egoTension (ssEgo ss))
  208       , "meaning_graph: " <> graphStats (ssMeaningGraph ss)
  209       , "kernel_pulse: " <> renderValue (kpActive (ssKernelPulse ss))
  210       , "essence_mode: " <> essenceModeTag
  211       , "recovery_cause: " <> recoveryCauseTag
  212       , "shadow_severity: " <> shadowSeverityTag
  213       , "gradient(m,c,t): " <> gradientTag
  214       , "strategy: " <> recoveryStrategyTag
  215       , "learning_authority_status: " <> renderEpistemicStatus (learningAuthorityStatus latestTrace)
  216       , "gf_lexical_authority_status: " <> renderEpistemicStatus (gfLexicalAuthorityStatus latestTrace)
  217       , "formal_contour_status: " <> renderEpistemicStatus (formalContourStatus latestTrace)
  218       , "truth_contract: " <> truthContractTag
  219       , "authority_class: " <> authorityClassTag
  220       , "assembly_path: " <> assemblyPathTag
  221       , "replay_provenance_status: " <> replayProvenanceTag
  222       , "surface_provenance: " <> surfaceProvenanceTag
  223       , "contract_provenance: " <> contractProvenanceTag
  224       , "replay_trace_load_status: " <> replayTraceLoadTag
  225       , "r5_core_version: " <> renderValue (r5cVersionId defaultR5CoreProfile)
  226       , "r5_policy_version: " <> renderValue (r5pVersionId defaultR5PolicyProfile)
  227       , "r5_policy_authority_status: " <> renderEpistemicStatus r5PolicyAuthorityStatus
  228       ] ++ governanceSummaryLines ss ++ ["STATE_END"]

  250 countGovernanceLifecycle :: GovernanceLifecycleStatus -> [GovernanceEvent] -> Int

  253 registryDifferenceCount :: PerspectiveRegistry -> PerspectiveRegistry -> Int
  254 registryDifferenceCount live rebuilt =
  255   let liveThreads = prThreads live
  256       rebuiltThreads = prThreads rebuilt
  257       scopes = S.union (M.keysSet liveThreads) (M.keysSet rebuiltThreads)
  258   in length [ () | scope <- S.toList scopes, M.lookup scope liveThreads /= M.lookup scope rebuiltThreads ]

  260 governanceFreezeStatus :: [GovernanceEvent] -> Text
  261 governanceFreezeStatus history =
  262   case [ event | event <- reverse history, isFreezeEvent event ] of
  263     [] -> "none"
  264     event:_ ->
  265       let envelope = geEnvelope event
  266       in case (geeLifecycleStatus envelope, geeDecision envelope) of
  267            (GlsDenied, _) -> "denied"
  268            (GlsRolledBack, _) -> "released"
  269            (_, GovFreeze) -> "frozen"
  270            (_, GovSuspend) -> "released"
  271            _ -> "active"

  284 governanceAuthorityStatus :: SystemState -> Text -> EpistemicStatus
  285 governanceAuthorityStatus ss rebuildStatus =
  286   case (truthContractIsAuthoritative (ssTruthContractStatus ss), ssGovernanceRuntimeFault ss, rebuildStatus) of
  287     (False, _, _) -> EpstNonAuthoritative
  288     (True, Just _, _) -> EpstNonAuthoritative

  290     (True, Nothing, "mismatch") -> EpstDegraded
  291     (True, Nothing, _) -> EpstAdvisory

  306       Just "CanonicalSurfacePreserved" -> EpstAuthoritative
  307       Just "AssembledSurfacePreserved" -> EpstAdvisory
  308       Just "CompatibilityShimSurface" -> EpstNonAuthoritative
  309       Just "DefaultedSurface" -> EpstNonAuthoritative

  348 loadLatestReplayTrace :: Session -> IO ReplayTraceLoad
  349 loadLatestReplayTrace session = do
  350   result <- try $ withRuntimeDb (sessRuntime session) $ \db -> do
  351     let sql = "SELECT replay_trace_json FROM turn_quality WHERE session_id = ? ORDER BY turn DESC LIMIT 1"
  352     mStmt <- NSQL.prepare db sql
  353     case mStmt of
  354       Left err -> pure (ReplayTraceDbError ("prepare_failed:" <> err))
  355       Right stmt -> do
  356         result' <- (
  357           do
  358             _ <- NSQL.bindText stmt 1 (sessSessionId session)
  359             hasRow <- NSQL.stepRow stmt
  360             if not hasRow
  361               then pure ReplayTraceMissing
  362               else do
  363                 payload <- NSQL.columnText stmt 0
  364                 pure (decodeReplayTraceSummary payload)
  365           ) `finally` finalizeQuietly stmt
  366         pure result'
  367   pure (either (ReplayTraceDbError . T.pack . show) id (result :: Either SomeException ReplayTraceLoad))
  368   where
  369     decodeReplayTraceSummary :: Text -> ReplayTraceLoad
  370     decodeReplayTraceSummary payload =
  371       case Aeson.decode (BL.fromStrict (TE.encodeUtf8 payload)) :: Maybe Aeson.Value of
  372         Just (Aeson.Object obj) ->
  373           let recoveryCause = decodeScalarField "trcRecoveryCause" obj
  374               recoveryStrategy = decodeScalarField "trcRecoveryStrategy" obj
  375               shadowSeverity = decodeScalarField "trcShadowDivergenceSeverity" obj
  376               evidence = decodeStringArrayField "trcRecoveryEvidence" obj
  377               learningValidationStatus = decodeScalarField "trcLearningValidationStatus" obj
  378               fallbackReason = decodeScalarField "trcFallbackReason" obj
  379               authorityClass = decodeScalarField "trcAuthorityClass" obj
  380               truthContractStatus = decodeScalarField "trcTruthContractStatus" obj
  381               assemblyPath = decodeScalarField "trcAssemblyPath" obj
  382               replayProvenanceStatus =
  383                 case decodeScalarField "trcReplayProvenanceStatus" obj of
  384                   Just value -> Just value
  385                   Nothing -> Just "ReplayProvenanceLegacyIncomplete"
  386               surfaceProvenance = decodeScalarField "trcSurfaceProvenance" obj
  387               contractProvenance = decodeScalarField "trcContractProvenance" obj
  388               derivationTags = decodeStringArrayField "trcDerivationTags" obj
  389             in ReplayTraceLoaded ReplayTraceSummary
  390                  { rtsRecoveryCause = recoveryCause
  391                  , rtsRecoveryStrategy = recoveryStrategy
  392                  , rtsRecoveryEvidence = evidence
  393                  , rtsShadowSeverity = shadowSeverity
  394                  , rtsLearningValidationStatus = learningValidationStatus
  395                  , rtsFallbackReason = fallbackReason
  396                  , rtsAuthorityClass = authorityClass
  397                  , rtsTruthContractStatus = truthContractStatus
  398                  , rtsAssemblyPath = assemblyPath
  399                  , rtsReplayProvenanceStatus = replayProvenanceStatus
  400                   , rtsSurfaceProvenance = surfaceProvenance
  401                   , rtsContractProvenance = contractProvenance
  402                   , rtsDerivationTags = derivationTags
  403                   , rtsLoadStatus = "loaded"
  404                  }
  405         Just _ -> ReplayTraceSchemaError "replay_trace_not_object"
  406         Nothing -> ReplayTraceDecodeError "replay_trace_decode_failed"
  407 
  408     decodeScalarField :: Text -> Aeson.Object -> Maybe Text
  409     decodeScalarField key obj =
  410       case KM.lookup (K.fromText key) obj of
  411         Just (Aeson.String t) -> Just t
  412         Just (Aeson.Number n) -> Just (T.pack (show n))
  413         Just (Aeson.Bool True) -> Just "true"
  414         Just (Aeson.Bool False) -> Just "false"
  415         _ -> Nothing
  416 
  417     decodeStringArrayField :: Text -> Aeson.Object -> [Text]
  418     decodeStringArrayField key obj =
  419       case KM.lookup (K.fromText key) obj of
  420         Just (Aeson.Array arr) ->
  421           [ t | Aeson.String t <- V.toList arr ]
  422         _ -> []

  424 finalizeQuietly :: NSQL.Statement -> IO ()
  425 finalizeQuietly stmt = do
  426   _ <- try (NSQL.finalize stmt) :: IO (Either SomeException ())
  427   pure ()

  429 parseGradientFromEvidence :: [Text] -> Maybe (Double, Double, Double)
  430 parseGradientFromEvidence evidence =
  431   let findMarker prefix =
  432         case [ rest | entry <- evidence, Just rest <- [T.stripPrefix prefix entry] ] of
  433           rest : _ -> case reads (T.unpack (T.takeWhile (/= ' ') rest)) of
  434             [(n, _)] -> Just n
  435             _ -> Nothing
  436           [] -> Nothing
  437       m = findMarker "conatus_gradient_m="
  438       c = findMarker "conatus_gradient_c="
  439       t = findMarker "conatus_gradient_t="
  440   in case (m, c, t) of
  441        (Just m', Just c', Just t') -> Just (m', c', t')
  442        _ -> Nothing

  444 renderGradientTag :: (Double, Double, Double) -> Text
  445 renderGradientTag (m, c, t) =
  446   T.concat [ "m=", T.pack (show m), ",c=", T.pack (show c), ",t=", T.pack (show t) ]
