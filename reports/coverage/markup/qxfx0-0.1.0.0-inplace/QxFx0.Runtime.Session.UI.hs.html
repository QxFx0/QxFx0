   30 printHelp :: IO ()
   31 printHelp = do
   32   T.putStrLn "Interactive commands:"
   33   T.putStrLn "  :help      show commands"
   34   T.putStrLn "  :state     show compact runtime state"
   35   T.putStrLn "  :dialogue  natural dialogue output"
   36   T.putStrLn "  :semantic  semantic introspection output"
   37   T.putStrLn "  :quit      save state and exit"
   38   T.putStrLn ""
   39   T.putStrLn "Environment:"
   40   T.putStrLn "  QXFX0_SESSION_ID    session identifier"
   41   T.putStrLn "  QXFX0_DB            database path"
   42   T.putStrLn "  QXFX0_ROOT          project root"
   43   T.putStrLn "  QXFX0_RUNTIME_MODE  strict(default)|degraded(test harness only)"
   44   T.putStrLn "  QXFX0_EMBEDDING_BACKEND  local-deterministic|remote-http"
   45   T.putStrLn "  QXFX0_SESSION_LOCK  enable session locking"
   46 
   47 printStateSummary :: Session -> IO ()
   48 printStateSummary session = mapM_ T.putStrLn (stateSummaryLines session)
   49 
   50 stateSummaryLines :: Session -> [Text]
   51 stateSummaryLines session =
   52   let ss = sessSystemState session
   53       renderValue :: Show a => a -> Text
   54       renderValue = T.pack . show
   55    in [ "STATE_BEGIN"
   56       , "session_id: " <> sessSessionId session
   57       , "turns: " <> renderValue (ssTurnCount ss)
   58       , "output_mode: " <> renderRuntimeOutputMode (sessOutputMode session)
   59       , "atom_trace_ema: " <> renderValue (atCurrentLoad (ssTrace ss))
   60       , "last_family: " <> renderValue (ssLastFamily ss)
   61       , "last_topic: " <> ssLastTopic ss
   62       , "ego_agency: " <> renderValue (egoAgency (ssEgo ss))
   63       , "ego_tension: " <> renderValue (egoTension (ssEgo ss))
   64       , "meaning_graph: " <> graphStats (ssMeaningGraph ss)
   65       , "kernel_pulse: " <> renderValue (kpActive (ssKernelPulse ss))
   66       , "STATE_END"
   67       ]
