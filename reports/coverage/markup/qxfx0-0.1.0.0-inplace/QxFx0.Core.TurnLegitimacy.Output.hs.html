   21 finalizeOutput :: GuardSurface -> [Text] -> (GuardSurface, SurfaceProvenance)
   22 finalizeOutput preSafetySurface history =
   23   let safetyStatus = postRenderSafetyCheckSurface preSafetySurface history
   24       renderedSurface = fallbackSurfaceOnBlock preSafetySurface recoverySurface safetyStatus
   25       surfaceProvenance =
   26         case safetyStatus of
   27           InvariantBlock _ -> FromRecovery
   28           _ -> FromDB
   29    in (renderedSurface, surfaceProvenance)
   30 
   31 safeOutputText :: GuardSurface -> GuardSurface -> SafetyStatus -> Text
   32 safeOutputText okSurface blockedSurface safetyStatus =
   33   gsRenderedText (fallbackSurfaceOnBlock okSurface blockedSurface safetyStatus)
