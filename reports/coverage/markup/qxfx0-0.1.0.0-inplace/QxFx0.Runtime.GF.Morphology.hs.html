   36 toNominative :: MorphologyData -> Text -> Text
   37 toNominative md w =
   38   case M.lookup w (mdNominative md) of
   39     Just f -> f
   40     Nothing -> case M.lookup (T.toLower w) (mdNominative md) of
   41       Just f -> f
   42       Nothing -> resolveCandidateNominative md w
   43 
   44 resolveCandidateNominative :: MorphologyData -> Text -> Text
   45 resolveCandidateNominative md surface =
   46   case resolveLexemeForm md surface (Just NominativeCase) Nothing of
   47     Just form -> lfLemma form
   48     Nothing -> surface
   49 
   50 genitiveForm :: MorphologyData -> Text -> Text
   51 genitiveForm md w = case M.lookup w (mdGenitive md) of
   52   Just f -> f
   53   Nothing -> case M.lookup (T.toLower w) (mdGenitive md) of
   54     Just f -> f
   55     Nothing -> resolveCandidateGenitive md w
   56 
   57 resolveCandidateGenitive :: MorphologyData -> Text -> Text
   58 resolveCandidateGenitive md surface =
   59   case resolveLexemeForm md surface (Just GenitiveCase) Nothing of
   60     Just form -> lfSurface form
   61     Nothing -> surface
   62 
   63 accusativeForm :: MorphologyData -> Text -> Text
   64 accusativeForm md w =
   65   case resolveCandidateAccusative md w of
   66     Just form -> form
   67     Nothing ->
   68       let gender = guessGender w
   69           animacy = guessAnimacy w
   70       in case (gender, animacy) of
   71             (Masculine, Inanimate) -> w
   72             (Neuter, _) -> w
   73             (Feminine, Inanimate) -> feminineAccusativeByRule w
   74             _ -> genitiveForm md w
   75 
   76 resolveCandidateAccusative :: MorphologyData -> Text -> Maybe Text
   77 resolveCandidateAccusative md surface =
   78   lfSurface <$> resolveLexemeForm md surface (Just AccusativeCase) Nothing
   79 
   80 data Animacy = Animate | Inanimate
   81   deriving stock (Eq, Show)
   82 
   83 data Gender = Masculine | Feminine | Neuter
   84   deriving stock (Eq, Show)
   85 
   86 guessAnimacy :: Text -> Animacy
   87 guessAnimacy _ = Inanimate
   88 
   89 guessGender :: Text -> Gender
   90 guessGender w
   91   | T.toLower w `elem` feminineSoftSignExceptions = Feminine
   92   | T.isSuffixOf morphFemSuffixA lower = Feminine
   93   | T.isSuffixOf morphFemSuffixYa lower = Feminine
   94   | T.isSuffixOf morphNounSuffixA lower = Feminine
   95   | T.isSuffixOf morphNounSuffixIya lower = Feminine
   96   | T.isSuffixOf morphNeutSuffixO lower = Neuter
   97   | T.isSuffixOf morphNeutSuffixE lower = Neuter
   98   | otherwise = Masculine
   99   where
  100     lower = T.toLower w

  103 feminineSoftSignExceptions =

  105   , "кость", "лестница", "постель", "тень", "цель", "чь", "цепь"
  106   , "боль", "роль", "сталь", "соль", "мать", "дочь"
  107   , "любовь", "морковь", "пыль", "медь", "гроз", "кровь"
  108   , "область", "гавань", "мелочь"
  109   ]

  111 feminineAccusativeByRule :: Text -> Text
  112 feminineAccusativeByRule w
  113   | T.isSuffixOf morphFemSuffixA lower = T.dropEnd 1 lower <> "у"
  114   | T.isSuffixOf morphFemSuffixYa lower = T.dropEnd 1 lower <> "ю"
  115   | T.isSuffixOf morphNounSuffixA lower = T.dropEnd 1 lower <> "у"
  116   | T.isSuffixOf morphNounSuffixIya lower = T.dropEnd 2 lower <> "ию"
  117   | otherwise = w
  118   where
  119     lower = T.toLower w

  121 prepositionalForm :: MorphologyData -> Text -> Text
  122 prepositionalForm md w = case M.lookup w (mdPrepositional md) of
  123   Just f -> f
  124   Nothing -> case M.lookup (T.toLower w) (mdPrepositional md) of
  125     Just f -> f
  126     Nothing -> resolveCandidatePrepositional md w

  128 resolveCandidatePrepositional :: MorphologyData -> Text -> Text
  129 resolveCandidatePrepositional md surface =
  130   case resolveLexemeForm md surface (Just PrepositionalCase) Nothing of
  131     Just form -> lfLemma form
  132     Nothing -> surface

  134 data VerbPerson
  135   = Person1Sg
  136   | Person2Sg
  137   | Person3Sg
  138   | Person1Pl
  139   | Person2Pl
  140   | Person3Pl
  141   deriving stock (Eq, Show)
  142 
  143 conjugateVerb :: MorphologyData -> Text -> VerbPerson -> Text
  144 conjugateVerb md infinitive person =
  145   case resolveLexemeForm md infinitive (Just NominativeCase) (case person of
  146     Person1Pl -> Just PluralNumber
  147     Person2Pl -> Just PluralNumber
  148     Person3Pl -> Just PluralNumber
  149     _ -> Just SingularNumber) of
  150     Just form -> lfSurface form
  151     Nothing -> conjugateByRule infinitive person
  152 
  153 conjugateByRule :: Text -> VerbPerson -> Text
  154 conjugateByRule infinitive person =
  155   let stem = infinitiveStem infinitive
  156       conjClass = classifyVerb infinitive
  157   in case conjClass of
  158        FirstConj ->
  159          case person of
  160            Person1Sg -> stem <> first1SgEnding infinitive
  161            Person2Sg -> stem <> "ешь"
  162            Person3Sg -> stem <> "ет"
  163            Person1Pl -> stem <> "ем"
  164            Person2Pl -> stem <> "ете"
  165            Person3Pl -> stem <> "ют"
  166        SecondConj ->
  167          case person of
  168            Person1Sg -> stem <> second1SgEnding infinitive
  169            Person2Sg -> stem <> "ишь"
  170            Person3Sg -> stem <> "ит"
  171            Person1Pl -> stem <> "им"
  172            Person2Pl -> stem <> "ите"
  173            Person3Pl -> stem <> "ат"
  174 
  175 infinitiveStem :: Text -> Text
  176 infinitiveStem infinitive =
  177   let lower = T.toLower infinitive
  178   in case () of
  179        _
  180          | Just s <- T.stripSuffix "ивать" lower -> s
  181          | Just s <- T.stripSuffix "евать" lower -> s
  182          | Just s <- T.stripSuffix "овать" lower -> s
  183          | Just s <- T.stripSuffix "ывать" lower -> s
  184          | Just s <- T.stripSuffix "ать" lower -> s
  185          | Just s <- T.stripSuffix "ять" lower -> s
  186          | Just s <- T.stripSuffix "еть" lower -> s
  187          | Just s <- T.stripSuffix "ить" lower -> s
  188          | Just s <- T.stripSuffix "чь" lower -> s <> "г"
  189          | Just s <- T.stripSuffix "ти" lower -> s
  190          | otherwise -> lower
  191 
  192 classifyVerb :: Text -> ConjugationClass
  193 classifyVerb infinitive =
  194   let lower = T.toLower infinitive
  195   in case () of
  196        _
  197          | lower `elem` secondConjExceptions -> SecondConj
  198          | T.isSuffixOf "ить" lower -> SecondConj
  199          | T.isSuffixOf "еть" lower -> FirstConj
  200          | T.isSuffixOf "ать" lower -> FirstConj
  201          | T.isSuffixOf "ять" lower -> FirstConj
  202          | T.isSuffixOf "оть" lower -> FirstConj
  203          | T.isSuffixOf "уть" lower -> FirstConj
  204          | T.isSuffixOf "ыть" lower -> FirstConj
  205          | T.isSuffixOf "чь" lower -> FirstConj
  206          | T.isSuffixOf "ти" lower -> FirstConj
  207          | otherwise -> FirstConj
  208 
  209 secondConjExceptions :: [Text]
  210 secondConjExceptions =
  211   [ "смотреть", "видеть", "ненавидеть", "обидеть", "зависеть", "терпеть"
  212   , "вертеть", "дышать", "держать", "слышать", "гнать"
  213   ]
  214 
  215 data ConjugationClass = FirstConj | SecondConj
  216   deriving stock (Eq, Show)

  218 first1SgEnding :: Text -> Text
  219 first1SgEnding infinitive =
  220   let lower = T.toLower infinitive
  221   in case () of
  222        _
  223          | T.isSuffixOf "овать" lower -> "ую"
  224          | T.isSuffixOf "евать" lower -> "ую"
  225          | T.isSuffixOf "ивать" lower -> "иваю"
  226          | T.isSuffixOf "ывать" lower -> "ываю"
  227          | T.isSuffixOf "ать" lower -> "аю"
  228          | T.isSuffixOf "ять" lower -> "яю"
  229          | otherwise -> "у"
  230 
  231 second1SgEnding :: Text -> Text
  232 second1SgEnding infinitive =
  233   let lower = T.toLower infinitive
  234   in case () of
  235        _
  236          | T.isSuffixOf "ить" lower -> "ю"
  237          | T.isSuffixOf "еть" lower -> "ю"
  238          | otherwise -> "ю"

  240 pluralNominative :: MorphologyData -> Text -> Text
  241 pluralNominative md w =
  242   case resolveLexemeForm md w (Just NominativeCase) (Just PluralNumber) of
  243     Just form -> lfSurface form
  244     Nothing -> pluralizeByRule w

  250     Nothing -> pluralizeGenitiveByRule w

  258 pluralPrepositional :: MorphologyData -> Text -> Text
  259 pluralPrepositional md w =
  260   case resolveLexemeForm md w (Just PrepositionalCase) (Just PluralNumber) of
  261     Just form -> lfLemma form
  262     Nothing -> pluralPrepositionalByRule w

  264 pluralizeByRule :: Text -> Text
  265 pluralizeByRule w =
  266   let lower = T.toLower w
  267   in case () of
  268        _
  269          | T.isSuffixOf "а" lower -> T.dropEnd 1 lower <> "ы"
  270          | T.isSuffixOf "я" lower -> T.dropEnd 1 lower <> "и"
  271          | T.isSuffixOf "о" lower -> T.dropEnd 1 lower <> "а"
  272          | T.isSuffixOf "е" lower -> T.dropEnd 1 lower <> "и"
  273          | T.isSuffixOf "ия" lower -> T.dropEnd 2 lower <> "ии"
  274          | T.isSuffixOf "ье" lower -> T.dropEnd 2 lower <> "ья"
  275          | T.isSuffixOf "ь" lower -> lower <> "и"
  276          | otherwise -> lower <> "ы"
  277 
  278 pluralizeGenitiveByRule :: Text -> Text
  279 pluralizeGenitiveByRule w =
  280   let lower = T.toLower w
  281   in case () of
  282        _
  283          | T.isSuffixOf "а" lower -> T.dropEnd 1 lower <> "ы"
  284          | T.isSuffixOf "я" lower -> T.dropEnd 1 lower <> "и"
  285          | T.isSuffixOf "о" lower -> T.dropEnd 1 lower <> "ов"
  286          | T.isSuffixOf "е" lower -> T.dropEnd 1 lower <> "ей"
  287          | T.isSuffixOf "ия" lower -> T.dropEnd 2 lower <> "ий"
  288          | T.isSuffixOf "ь" lower -> T.dropEnd 1 lower <> "ей"
  289          | otherwise -> lower <> "ов"
  290 
  291 pluralPrepositionalByRule :: Text -> Text
  292 pluralPrepositionalByRule w =
  293   let lower = T.toLower w
  294   in case () of
  295        _
  296          | T.isSuffixOf "а" lower -> T.dropEnd 1 lower <> "ах"
  297          | T.isSuffixOf "я" lower -> T.dropEnd 1 lower <> "ях"
  298          | T.isSuffixOf "о" lower -> T.dropEnd 1 lower <> "ах"
  299          | T.isSuffixOf "е" lower -> T.dropEnd 1 lower <> "ах"
  300          | T.isSuffixOf "ия" lower -> T.dropEnd 2 lower <> "иях"
  301          | T.isSuffixOf "ь" lower -> T.dropEnd 1 lower <> "ах"
  302          | otherwise -> lower <> "ах"
