   31 data GfLexemeForms = GfLexemeForms
   32   { glfNom :: !Text
   33   , glfGen :: !Text
   34   , glfPrep :: !Text
   35   , glfAcc :: !Text
   36   , glfIns :: !Text
   37   } deriving stock (Eq, Show)
   38 
   39 data GfMapData = GfMapData
   40   { gmdFormToFun :: !(M.Map Text Text)
   41   , gmdFunToForms :: !(M.Map Text GfLexemeForms)
   42   }

   44 -- | Structured status of the GF lexicon map load at startup.
   45 -- Consumers must check this if they need to distinguish a healthy
   46 -- lexicon from a degraded/failed load.
   47 data GfMapLoadStatus
   48   = GfMapLoaded !Int
   49     -- ^ Successfully loaded with the given number of form→fun entries.
   50   | GfMapLoadFailed !Text
   51     -- ^ Failed to load; the 'Text' carries a machine-readable reason tag.
   52   deriving stock (Eq, Show)
   53 
   54 defaultGfLexemeId :: Text
   55 defaultGfLexemeId = "ponyatie_N"
   56 
   57 lookupTopicGfLexemeId :: Text -> Maybe Text
   58 lookupTopicGfLexemeId rawTopic =
   59   let normalized = normalizeText rawTopic
   60       candidates = normalized : maybeToList (stripTopicMarker normalized)
   61       lookupFirst [] = Nothing
   62       lookupFirst (x:xs) = M.lookup x (gmdFormToFun gfMapData) <|> lookupFirst xs
   63   in lookupFirst candidates
   64   where
   65     maybeToList = maybe [] pure
   66 
   67 topicToGfLexemeId :: Text -> Text
   68 topicToGfLexemeId = fst . topicToGfLexemeDecision
   69 
   70 topicToGfLexemeDecision :: Text -> (Text, Maybe Text)
   71 topicToGfLexemeDecision topic =
   72   case gfMapFallbackReason gfMapLoadStatus of
   73     Just reason -> (defaultGfLexemeId, Just reason)
   74     Nothing ->
   75       case lookupTopicGfLexemeId topic of
   76         Just lexemeId -> (lexemeId, Nothing)
   77         Nothing -> (defaultGfLexemeId, Just "gf_default_lexeme")
   78 
   79 lookupGfLexemeForms :: Text -> Maybe GfLexemeForms
   80 lookupGfLexemeForms funId = M.lookup funId (gmdFunToForms gfMapData)
   81 
   82 stripTopicMarker :: Text -> Maybe Text
   83 stripTopicMarker txt =
   84   listToMaybe
   85     [ rest
   86     | marker <- ["о ", "об ", "обо ", "про "]
   87     , Just restRaw <- [T.stripPrefix marker txt]
   88     , let rest = T.strip restRaw
   89     , not (T.null rest)
   90     ]
   91 
   92 normalizeText :: Text -> Text
   93 normalizeText = T.toLower . T.replace "ё" "е" . T.strip
   94 
   95 -- | Read-only immutable GF lexicon map loaded once from the canonical
   96 -- GF lexicon funmap. The runtime surface remains pure while the load
   97 -- status stays explicit and operator-visible.
   98 gfMapLoadResult :: (GfMapData, GfMapLoadStatus)
   99 gfMapLoadResult = unsafePerformIO loadCanonicalGfMap
  100 {-# NOINLINE gfMapLoadResult #-}
  101 
  102 gfMapData :: GfMapData
  103 gfMapData = fst gfMapLoadResult
  104 
  105 gfMapLoadStatus :: GfMapLoadStatus
  106 gfMapLoadStatus = snd gfMapLoadResult
  107 
  108 gfMapProvenanceTag :: Text
  109 gfMapProvenanceTag =
  110   case gfMapLoadStatus of
  111     GfMapLoaded count -> "gf_map_loaded:" <> T.pack (show count)
  112     GfMapLoadFailed reason -> "gf_map_failed:" <> reason
  113 
  114 gfMapFallbackReason :: GfMapLoadStatus -> Maybe Text
  115 gfMapFallbackReason status =
  116   case status of
  117     GfMapLoaded _ -> Nothing
  118     GfMapLoadFailed reason -> Just ("gf_map_unavailable:" <> reason)
  119 
  120 -- | Pure, total loader from optional file content.
  121 -- Separated from IO so the failure paths are unit-testable.
  122 loadGfMapFromContent :: Maybe Text -> (GfMapData, GfMapLoadStatus)
  123 loadGfMapFromContent Nothing =
  124   (emptyGfMapData, GfMapLoadFailed "resource_missing_or_unreadable")
  125 loadGfMapFromContent (Just content) =
  126   let parsed = parseGfMapData content
  127       entryCount = M.size (gmdFormToFun parsed)
  128   in if entryCount == 0
  129        then (parsed, GfMapLoadFailed "resource_empty_or_unparseable")
  130        else (parsed, GfMapLoaded entryCount)
  131 
  132 loadCanonicalGfMap :: IO (GfMapData, GfMapLoadStatus)
  133 loadCanonicalGfMap = do
  134   mContent <- readCanonicalFunmap
  135   pure (loadGfMapFromContent mContent)
  136 
  137 loadGfMapStatusFromPath :: FilePath -> IO GfMapLoadStatus
  138 loadGfMapStatusFromPath path = do
  139   mContent <- catchIO (Just . T.pack <$> readFile path) (const (pure Nothing))
  140   pure (snd (loadGfMapFromContent mContent))
  141 
  142 readCanonicalFunmap :: IO (Maybe Text)
  143 readCanonicalFunmap = do
  144   mResourceRoot <- lookupEnv "QXFX0_RESOURCE_ROOT"
  145   mRoot <- lookupEnv "QXFX0_ROOT"
  146   let mExplicitRoot = mResourceRoot <|> mRoot
  147   explicitContent <- case mExplicitRoot of
  148     Just root ->
  149       catchIO
  150         (Just . T.pack <$> readFile (root </> "spec" </> "gf" </> "lexicon_funmap.tsv"))
  151         (const (pure Nothing))
  152     Nothing -> pure Nothing
  153   case explicitContent of
  154     Just content -> pure (Just content)
  155     Nothing -> do
  156       dataPath <- catchIO (Just <$> getDataFileName "spec/gf/lexicon_funmap.tsv") (const (pure Nothing))
  157       case dataPath of
  158         Just path ->
  159           catchIO
  160             (Just . T.pack <$> readFile path)
  161             (const readRepoFallback)
  162         Nothing ->
  163           readRepoFallback
  164   where
  165     readRepoFallback =
  166       catchIO
  167         (Just . T.pack <$> readFile ("spec" </> "gf" </> "lexicon_funmap.tsv"))
  168         (const (pure Nothing))

  170 parseGfMapData :: Text -> GfMapData
  171 parseGfMapData content =
  172   foldl insertRow emptyGfMapData parsedRows
  173   where
  174     parsedRows = mapMaybeRow (T.lines content)
  175     mapMaybeRow = foldr (\line acc -> maybe acc (:acc) (parseRow line)) []
  176 
  177 insertRow :: GfMapData -> (Text, Text, GfLexemeForms) -> GfMapData
  178 insertRow acc (funId, lemma, forms) =
  179   let formKeys =
  180         [ lemma
  181         , glfNom forms
  182         , glfGen forms
  183         , glfPrep forms
  184         , glfAcc forms
  185         , glfIns forms
  186         ]
  187       formToFun' =
  188         foldr
  189           (\k m -> M.insertWith (\_ old -> old) (normalizeText k) funId m)
  190           (gmdFormToFun acc)
  191           formKeys
  192       funToForms' = M.insertWith (\_ old -> old) funId forms (gmdFunToForms acc)
  193   in GfMapData formToFun' funToForms'
  194 
  195 parseRow :: Text -> Maybe (Text, Text, GfLexemeForms)
  196 parseRow row =
  197   case T.splitOn "\t" row of
  198     [funId, lemma, _pos, nominative, genitive, prepositional] ->
  199       let forms =
  200             GfLexemeForms
  201               { glfNom = normalizeText nominative
  202               , glfGen = normalizeText genitive
  203               , glfPrep = normalizeText prepositional
  204               , glfAcc = normalizeText nominative
  205               , glfIns = normalizeText nominative
  206               }
  207       in Just (T.strip funId, normalizeText lemma, forms)
  208     [funId, lemma, _pos, nominative, genitive, prepositional, accusative, instrumental] ->
  209       let forms =
  210             GfLexemeForms
  211               { glfNom = normalizeText nominative
  212               , glfGen = normalizeText genitive
  213               , glfPrep = normalizeText prepositional
  214               , glfAcc = normalizeText accusative
  215               , glfIns = normalizeText instrumental
  216               }
  217       in Just (T.strip funId, normalizeText lemma, forms)
  218     _ ->
  219       Nothing
  220 
  221 emptyGfMapData :: GfMapData
  222 emptyGfMapData =
  223   GfMapData
  224     { gmdFormToFun = M.empty
  225     , gmdFunToForms = M.empty
  226     }
