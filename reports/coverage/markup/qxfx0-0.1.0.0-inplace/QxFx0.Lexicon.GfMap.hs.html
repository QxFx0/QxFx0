   24 data GfLexemeForms = GfLexemeForms
   25   { glfNom :: !Text
   26   , glfGen :: !Text
   27   , glfPrep :: !Text
   28   , glfAcc :: !Text
   29   , glfIns :: !Text
   30   } deriving stock (Eq, Show)
   31 
   32 data GfMapData = GfMapData
   33   { gmdFormToFun :: !(M.Map Text Text)
   34   , gmdFunToForms :: !(M.Map Text GfLexemeForms)
   35   }
   36 
   37 defaultGfLexemeId :: Text
   38 defaultGfLexemeId = "smysl_N"
   39 
   40 topicToGfLexemeId :: Text -> Text
   41 topicToGfLexemeId rawTopic =
   42   let normalized = normalizeText rawTopic
   43       candidates = normalized : maybeToList (stripTopicMarker normalized)
   44       lookupFirst [] = Nothing
   45       lookupFirst (x:xs) = M.lookup x (gmdFormToFun gfMapData) <|> lookupFirst xs
   46   in fromMaybe defaultGfLexemeId (lookupFirst candidates)
   47   where
   48     maybeToList = maybe [] pure
   49 
   50 lookupGfLexemeForms :: Text -> Maybe GfLexemeForms
   51 lookupGfLexemeForms funId = M.lookup funId (gmdFunToForms gfMapData)
   52 
   53 stripTopicMarker :: Text -> Maybe Text
   54 stripTopicMarker txt =
   55   listToMaybe
   56     [ rest
   57     | marker <- ["о ", "об ", "обо ", "про "]
   58     , Just restRaw <- [T.stripPrefix marker txt]
   59     , let rest = T.strip restRaw
   60     , not (T.null rest)
   61     ]
   62 
   63 normalizeText :: Text -> Text
   64 normalizeText = T.toLower . T.replace "ё" "е" . T.strip
   65 
   66 {-# NOINLINE gfMapData #-}
   67 gfMapData :: GfMapData
   68 gfMapData = unsafePerformIO loadGfMapData
   69 
   70 loadGfMapData :: IO GfMapData
   71 loadGfMapData = do
   72   dataPathResult <- try (getDataFileName "spec/gf/lexicon_funmap.tsv") :: IO (Either SomeException FilePath)
   73   let bundledPath = either (const "") id dataPathResult
   74       fallbackPath = "spec/gf/lexicon_funmap.tsv"
   75   mContent <- tryReadPath bundledPath >>= \case
   76     Just content -> pure (Just content)
   77     Nothing -> tryReadPath fallbackPath
   78   pure (maybe emptyGfMapData parseGfMapData mContent)
   79 
   80 tryReadPath :: FilePath -> IO (Maybe Text)
   81 tryReadPath path
   82   | null path = pure Nothing
   83   | otherwise = do
   84       exists <- doesFileExist path
   85       if not exists
   86         then pure Nothing
   87         else do
   88           readResult <- try (TIO.readFile path) :: IO (Either SomeException Text)
   89           pure (either (const Nothing) Just readResult)
   90 
   91 parseGfMapData :: Text -> GfMapData
   92 parseGfMapData content =
   93   foldl insertRow emptyGfMapData parsedRows
   94   where
   95     parsedRows = mapMaybeRow (drop 1 (T.lines content))
   96     mapMaybeRow = foldr (\line acc -> maybe acc (:acc) (parseRow line)) []
   97 
   98 insertRow :: GfMapData -> (Text, Text, GfLexemeForms) -> GfMapData
   99 insertRow acc (funId, lemma, forms) =
  100   let formKeys =
  101         [ lemma
  102         , glfNom forms
  103         , glfGen forms
  104         , glfPrep forms
  105         , glfAcc forms
  106         , glfIns forms
  107         ]
  108       formToFun' =
  109         foldr
  110           (\k m -> M.insertWith (\_ old -> old) (normalizeText k) funId m)
  111           (gmdFormToFun acc)
  112           formKeys
  113       funToForms' = M.insertWith (\_ old -> old) funId forms (gmdFunToForms acc)
  114   in GfMapData formToFun' funToForms'
  115 
  116 parseRow :: Text -> Maybe (Text, Text, GfLexemeForms)
  117 parseRow row =
  118   case T.splitOn "\t" row of
  119     [funId, lemma, _pos, nominative, genitive, prepositional] ->
  120       let forms =
  121             GfLexemeForms
  122               { glfNom = normalizeText nominative
  123               , glfGen = normalizeText genitive
  124               , glfPrep = normalizeText prepositional
  125               , glfAcc = normalizeText nominative
  126               , glfIns = normalizeText nominative
  127               }
  128       in Just (T.strip funId, normalizeText lemma, forms)
  129     [funId, lemma, _pos, nominative, genitive, prepositional, accusative, instrumental] ->
  130       let forms =
  131             GfLexemeForms
  132               { glfNom = normalizeText nominative
  133               , glfGen = normalizeText genitive
  134               , glfPrep = normalizeText prepositional
  135               , glfAcc = normalizeText accusative
  136               , glfIns = normalizeText instrumental
  137               }
  138       in Just (T.strip funId, normalizeText lemma, forms)
  139     _ ->
  140       Nothing
  141 
  142 emptyGfMapData :: GfMapData
  143 emptyGfMapData =
  144   GfMapData
  145     { gmdFormToFun = M.empty
  146     , gmdFunToForms = M.empty
  147     }
