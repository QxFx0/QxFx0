   26 egoModulateStance :: EgoState -> StanceMarker -> StanceMarker
   27 egoModulateStance ego base =
   28   let tension = egoTension ego
   29       agency = egoAgency ego
   30    in if tension > criticalTensionThreshold && agency < lowAgencyThreshold
   31         then HoldBack
   32         else
   33           if tension > elevatedTensionThreshold
   34             then
   35               case base of
   36                 Commit -> Honest
   37                 Firm -> Observe
   38                 stance -> stance
   39             else
   40               if agency > highAgencyThreshold
   41                 then
   42                   case base of
   43                     Tentative -> Explore
   44                     HoldBack -> Observe
   45                     stance -> stance
   46                 else base
   47 
   48 egoModulateEpistemic :: EgoState -> EpistemicStatus -> EpistemicStatus
   49 egoModulateEpistemic ego base =
   50   let tension = egoTension ego
   51       agency = egoAgency ego
   52    in if tension > elevatedTensionThreshold
   53         then degradeEpistemic base
   54         else if agency > highAgencyThreshold
   55           then strengthenEpistemic base
   56           else base
   57 
   58 egoModulateEpistemicWithTruthContract :: TruthContractStatus -> EgoState -> EpistemicStatus -> EpistemicStatus
   59 egoModulateEpistemicWithTruthContract truthStatus ego base =
   60   capByTruthContract truthStatus (egoModulateEpistemic ego base)

   62 traceModulateStance :: AtomTrace -> StanceMarker -> StanceMarker
   63 traceModulateStance trace base =
   64   let load = atCurrentLoad trace
   65    in if load > highTraceLoadThreshold
   66         then
   67           case base of
   68             Commit -> Honest
   69             Firm -> Explore
   70             stance -> stance
   71         else base

   73 threeStageModulation :: EgoState -> AtomTrace -> StanceMarker -> EpistemicStatus -> (StanceMarker, EpistemicStatus)
   74 threeStageModulation ego trace baseStance baseEpistemic =
   75   let stage2Stance = egoModulateStance ego baseStance
   76       stage2Epistemic = egoModulateEpistemic ego baseEpistemic
   77    in (traceModulateStance trace stage2Stance, stage2Epistemic)
   78 
   79 threeStageModulationWithTruthContract :: TruthContractStatus -> EgoState -> AtomTrace -> StanceMarker -> EpistemicStatus -> (StanceMarker, EpistemicStatus)
   80 threeStageModulationWithTruthContract truthStatus ego trace baseStance baseEpistemic =
   81   let stage2Stance = egoModulateStance ego baseStance
   82       stage2Epistemic = egoModulateEpistemicWithTruthContract truthStatus ego baseEpistemic
   83    in (traceModulateStance trace stage2Stance, stage2Epistemic)

   85 feralDegradation :: Bool -> StanceMarker -> EpistemicStatus -> (StanceMarker, EpistemicStatus)
   86 feralDegradation nixAvailable baseStance baseEpistemic =
   87   if nixAvailable
   88     then (baseStance, baseEpistemic)
   89     else (escalateStance baseStance, degradeEpistemic baseEpistemic)

   91 antiStuck :: Int -> EgoState -> CanonicalMoveFamily -> Maybe CanonicalMoveFamily
   92 antiStuck consecutiveReflect ego currentFamily =
   93   if consecutiveReflect >= 3 && currentFamily == CMReflect
   94     then Just CMDeepen
   95     else
   96       if egoAgency ego < veryLowAgencyThreshold && egoTension ego > elevatedTensionThreshold
   97         then Just CMRepair
   98         else Nothing

  100 degradeEpistemic :: EpistemicStatus -> EpistemicStatus
  101 degradeEpistemic (Known confidence) = Probable confidence
  102 degradeEpistemic (Probable confidence) = Uncertain confidence
  103 degradeEpistemic (Uncertain confidence) = Speculative confidence
  104 degradeEpistemic status = status
  105 
  106 strengthenEpistemic :: EpistemicStatus -> EpistemicStatus
  107 strengthenEpistemic (Speculative confidence) = Uncertain confidence
  108 strengthenEpistemic (Uncertain confidence) = Probable confidence
  109 strengthenEpistemic (Probable confidence) = Known confidence
  110 strengthenEpistemic status = status
  111 
  112 escalateStance :: StanceMarker -> StanceMarker
  113 escalateStance Commit = Firm
  114 escalateStance Firm = Honest
  115 escalateStance Honest = Observe
  116 escalateStance Explore = Tentative
  117 escalateStance Tentative = HoldBack
  118 escalateStance stance = stance
