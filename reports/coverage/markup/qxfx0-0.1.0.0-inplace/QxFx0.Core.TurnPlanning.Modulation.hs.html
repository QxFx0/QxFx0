   23 egoModulateStance :: EgoState -> StanceMarker -> StanceMarker
   24 egoModulateStance ego base =
   25   let tension = egoTension ego
   26       agency = egoAgency ego
   27    in if tension > criticalTensionThreshold && agency < lowAgencyThreshold
   28         then HoldBack
   29         else
   30           if tension > elevatedTensionThreshold
   31             then
   32               case base of
   33                 Commit -> Honest
   34                 Firm -> Observe
   35                 stance -> stance
   36             else
   37               if agency > highAgencyThreshold
   38                 then
   39                   case base of
   40                     Tentative -> Explore
   41                     HoldBack -> Observe
   42                     stance -> stance
   43                 else base
   44 
   45 egoModulateEpistemic :: EgoState -> EpistemicStatus -> EpistemicStatus
   46 egoModulateEpistemic ego base =
   47   let tension = egoTension ego
   48       agency = egoAgency ego
   49    in if tension > elevatedTensionThreshold
   50         then degradeEpistemic base
   51         else if agency > highAgencyThreshold
   52           then strengthenEpistemic base
   53           else base
   54 
   55 traceModulateStance :: AtomTrace -> StanceMarker -> StanceMarker
   56 traceModulateStance trace base =
   57   let load = atCurrentLoad trace
   58    in if load > highTraceLoadThreshold
   59         then
   60           case base of
   61             Commit -> Honest
   62             Firm -> Explore
   63             stance -> stance
   64         else base
   65 
   66 threeStageModulation :: EgoState -> AtomTrace -> StanceMarker -> EpistemicStatus -> (StanceMarker, EpistemicStatus)
   67 threeStageModulation ego trace baseStance baseEpistemic =
   68   let stage2Stance = egoModulateStance ego baseStance
   69       stage2Epistemic = egoModulateEpistemic ego baseEpistemic
   70    in (traceModulateStance trace stage2Stance, stage2Epistemic)

   72 feralDegradation :: Bool -> StanceMarker -> EpistemicStatus -> (StanceMarker, EpistemicStatus)
   73 feralDegradation nixAvailable baseStance baseEpistemic =
   74   if nixAvailable
   75     then (baseStance, baseEpistemic)
   76     else (escalateStance baseStance, degradeEpistemic baseEpistemic)

   78 antiStuck :: Int -> EgoState -> CanonicalMoveFamily -> Maybe CanonicalMoveFamily
   79 antiStuck consecutiveReflect ego currentFamily =
   80   if consecutiveReflect >= 3 && currentFamily == CMReflect
   81     then Just CMDeepen
   82     else
   83       if egoAgency ego < veryLowAgencyThreshold && egoTension ego > elevatedTensionThreshold
   84         then Just CMRepair
   85         else Nothing
   86 
   87 degradeEpistemic :: EpistemicStatus -> EpistemicStatus
   88 degradeEpistemic (Known confidence) = Probable confidence
   89 degradeEpistemic (Probable confidence) = Uncertain confidence
   90 degradeEpistemic (Uncertain confidence) = Speculative confidence
   91 degradeEpistemic status = status
   92 
   93 strengthenEpistemic :: EpistemicStatus -> EpistemicStatus
   94 strengthenEpistemic (Speculative confidence) = Uncertain confidence
   95 strengthenEpistemic (Uncertain confidence) = Probable confidence
   96 strengthenEpistemic (Probable confidence) = Known confidence
   97 strengthenEpistemic status = status

   99 escalateStance :: StanceMarker -> StanceMarker
  100 escalateStance Commit = Firm
  101 escalateStance Firm = Honest
  102 escalateStance Honest = Observe
  103 escalateStance Explore = Tentative
  104 escalateStance Tentative = HoldBack
  105 escalateStance stance = stance
