  176 mkExecutedTurnOutcome
  177   :: CanonicalMoveFamily
  178   -> IllocutionaryForce
  179   -> AuthorityClass
  180   -> ContractProvenance
  181   -> SurfaceProvenance
  182   -> AssemblyPath
  183   -> ArtifactManifest
  184   -> Either Text ExecutedTurnOutcome
  185 mkExecutedTurnOutcome family force authority contractProv surfaceProv assemblyPath manifest = do
  186   let bad condition msg = if condition then Left msg else Right ()
  187       transitionWon = authority `elem` [AuthorityCanonical, AuthorityAssembled]
  188       truthStatus = truthContractStatusForAuthority authority assemblyPath
  189   bad (assemblyPath == RussianCompatShimRoute && authority /= AuthorityShim)
  190       "executed_outcome_invalid:russian_compat_must_be_shim"
  191   bad (assemblyPath == GuardRecoveryRoute && transitionWon)
  192       "executed_outcome_invalid:recovery_route_cannot_win"
  193   bad (authority `elem` [AuthorityFallback, AuthorityShim, AuthorityDefault, AuthorityGeneratedArtifact, AuthorityLegacyIncomplete, AuthorityRecovery] && transitionWon)
  194       "executed_outcome_invalid:non_authoritative_route_cannot_win"
  195   pure ExecutedTurnOutcome
  196     { etoFamily = family
  197     , etoForce = force
  198     , etoAuthorityClass = authority
  199     , etoTruthContractStatus = truthStatus
  200     , etoContractProvenance = contractProv
  201     , etoSurfaceProvenance = surfaceProv
  202     , etoAssemblyPath = assemblyPath
  203     , etoArtifactManifest = manifest
  204     , etoTransitionWon = transitionWon
  205     }

  207 truthContractStatusForAuthority :: AuthorityClass -> AssemblyPath -> TruthContractStatus
  208 truthContractStatusForAuthority authority assemblyPath =
  209   case (authority, assemblyPath) of
  210     (AuthorityCanonical, _) -> CanonicalSurfacePreserved
  211     (AuthorityAssembled, _) -> AssembledSurfacePreserved
  212     (AuthorityRecovery, _) -> NonExpansiveRecoverySurface
  213     (AuthorityFallback, _) -> ExplicitFallbackSurface
  214     (AuthorityShim, _) -> CompatibilityShimSurface
  215     (AuthorityDefault, _) -> DefaultedSurface
  216     (AuthorityGeneratedArtifact, _) -> GeneratedArtifactSurface
  217     (AuthorityLegacyIncomplete, _) -> LegacyIncompleteSurface

  369 data ConstitutionalThresholds = ConstitutionalThresholds
  370   { ctAgencyMin :: !Double
  371   , ctTensionMax :: !Double
  372   , ctAmbiguityPenalty :: !Double
  373   , ctShadowDivergencePenalty :: !Double
  374   , ctLocalRecoveryThreshold :: !Double
  375   } deriving stock (Eq, Show, Generic)
  376   deriving anyclass (NFData)
  377 instance ToJSON ConstitutionalThresholds where
  378   toJSON ct =
  379     object
  380       [ "ctAgencyMin" .= ctAgencyMin ct
  381       , "ctTensionMax" .= ctTensionMax ct
  382       , "ctAmbiguityPenalty" .= ctAmbiguityPenalty ct
  383       , "ctShadowDivergencePenalty" .= ctShadowDivergencePenalty ct
  384       , "ctLocalRecoveryThreshold" .= ctLocalRecoveryThreshold ct
  385       ]
  386 
  387 instance FromJSON ConstitutionalThresholds where
  388   parseJSON = withObject "ConstitutionalThresholds" $ \obj -> do
  389     localRecoveryThreshold <-
  390       obj .:? "ctLocalRecoveryThreshold"
  391         >>= maybe
  392           (obj .:? "ctLlmFallbackThreshold")
  393           (pure . Just)
  394     ConstitutionalThresholds
  395       <$> obj .: "ctAgencyMin"
  396       <*> obj .: "ctTensionMax"
  397       <*> obj .: "ctAmbiguityPenalty"
  398       <*> obj .: "ctShadowDivergencePenalty"
  399       <*> pure (maybe constitutionalLocalRecoveryThresholdDefault id localRecoveryThreshold)
