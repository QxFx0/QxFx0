  218 data ConstitutionalThresholds = ConstitutionalThresholds
  219   { ctAgencyMin :: !Double
  220   , ctTensionMax :: !Double
  221   , ctAmbiguityPenalty :: !Double
  222   , ctShadowDivergencePenalty :: !Double
  223   , ctLocalRecoveryThreshold :: !Double
  224   } deriving stock (Eq, Show, Generic)
  225   deriving anyclass (NFData)
  226 instance ToJSON ConstitutionalThresholds where
  227   toJSON ct =
  228     object
  229       [ "ctAgencyMin" .= ctAgencyMin ct
  230       , "ctTensionMax" .= ctTensionMax ct
  231       , "ctAmbiguityPenalty" .= ctAmbiguityPenalty ct
  232       , "ctShadowDivergencePenalty" .= ctShadowDivergencePenalty ct
  233       , "ctLocalRecoveryThreshold" .= ctLocalRecoveryThreshold ct
  234       ]
  235 
  236 instance FromJSON ConstitutionalThresholds where
  237   parseJSON = withObject "ConstitutionalThresholds" $ \obj -> do
  238     localRecoveryThreshold <-
  239       obj .:? "ctLocalRecoveryThreshold"
  240         >>= maybe
  241           (obj .:? "ctLlmFallbackThreshold")
  242           (pure . Just)
  243     ConstitutionalThresholds
  244       <$> obj .: "ctAgencyMin"
  245       <*> obj .: "ctTensionMax"
  246       <*> obj .: "ctAmbiguityPenalty"
  247       <*> obj .: "ctShadowDivergencePenalty"
  248       <*> pure (maybe constitutionalLocalRecoveryThresholdDefault id localRecoveryThreshold)
  249
