   19 buildResponseSensePlan :: CanonicalMoveFamily -> DialogueCommitmentLedger -> DialoguePhase -> SenseVector -> ResponseSensePlan
   20 buildResponseSensePlan family ledger phase inputVec =
   21   let inputOp = case svOperators inputVec of
   22         op:_ -> op
   23         [] -> OpGround
   24       familyOp = familyToSenseOperator family
   25       chosenOp0 = if allowedTransition inputOp familyOp then familyOp else fallbackContinuation inputOp
   26       chosenOp = legalizeOperator phase ledger chosenOp0
   27       preservedAxes = map fst (take 2 (sortOn (Down . snd) (toList (svAxes inputVec))))
   28       reason
   29         | chosenOp == familyOp = "family_aligned_continuation"
   30         | otherwise = "bounded_adjacent_shift"
   31       distance = if chosenOp == inputOp then 0 else 1
   32   in ResponseSensePlan
   33       { rspInputVector = inputVec
   34       , rspChosenOperator = chosenOp
   35       , rspPreservedAxes = preservedAxes
   36       , rspShiftReason = reason
   37       , rspDistance = distance
   38       }

   40 familySenseBundle :: CanonicalMoveFamily -> DialogueCommitmentLedger -> DialoguePhase -> DialogueThread -> SenseVector -> (CanonicalMoveFamily, ResponseSensePlan, MicroPlan)
   41 familySenseBundle family ledger phase thread inputVec =
   42   let constrainedFamily = constrainFamilyBySense family ledger phase inputVec
   43       sensePlan = buildResponseSensePlan constrainedFamily ledger phase inputVec
   44       microPlan = buildMicroPlan thread ledger phase sensePlan
   45   in (constrainedFamily, sensePlan, microPlan)
   46 
   47 constrainFamilyBySense :: CanonicalMoveFamily -> DialogueCommitmentLedger -> DialoguePhase -> SenseVector -> CanonicalMoveFamily
   48 constrainFamilyBySense family ledger phase inputVec =
   49   let plan = buildResponseSensePlan family ledger phase inputVec
   50   in case (family, rspChosenOperator plan) of
   51        (CMGround, OpDefine) -> CMDefine
   52        (CMGround, OpExplainCause) -> CMClarify
   53        (CMDescribe, OpRepair) -> CMRepair
   54        (CMDescribe, OpNextStep) -> CMNextStep
   55        (CMDescribe, OpDistinguish) -> CMDistinguish
   56        (CMGround, OpRepair) -> CMRepair
   57        (CMDeepen, OpDeepen) -> CMDeepen
   58        (CMPurpose, OpExplainPurpose) -> CMPurpose
   59        (CMConfront, OpDistinguish) -> CMConfront
   60        (other, _) -> other

   62 buildMicroPlan :: DialogueThread -> DialogueCommitmentLedger -> DialoguePhase -> ResponseSensePlan -> MicroPlan
   63 buildMicroPlan thread ledger phase sensePlan =
   64   let operator = rspChosenOperator sensePlan
   65       phaseBudget = case phase of
   66         Repairing -> 1
   67         Clarifying -> 1
   68         Grounding -> 1
   69         Exploring -> 2
   70         Advancing -> 2
   71         Contesting -> 1
   72         Closing -> 1
   73       explicitness = clamp01 (0.35 + fromIntegral (length (dtAcceptedTerms thread)) * 0.02 + if commitmentAllowsAdvance ledger then 0.15 else (-0.10))
   74       moves = case operator of
   75         OpRepair -> ["repair"]
   76         OpClarify -> ["clarify"]
   77         OpGround -> ["ground"]
   78         OpDefine -> ["define"]
   79         OpExplainCause -> ["explain_cause"]
   80         OpDistinguish -> ["distinguish"]
   81         OpDeepen -> ["deepen"]
   82         OpReflect -> ["reflect"]
   83         OpExplainPurpose -> ["explain_purpose"]
   84         OpNextStep -> ["next_step"]
   85         OpConstrain -> ["constrain"]
   86       fallbackPolicy = case phase of
   87         Repairing -> "repair_first"
   88         Clarifying -> "clarify_first"
   89         Grounding -> "ground_first"
   90         Contesting -> "contest_bound"
   91         Closing -> "close_bound"
   92         _ -> "safe_degrade"
   93   in MicroPlan
   94       { mpRhetoricalMoves = take 2 moves
   95       , mpExplicitness = explicitness
   96       , mpStructureBudget = phaseBudget
   97       , mpFallbackPolicy = fallbackPolicy
   98       }

  100 commitmentAllowsAdvance :: DialogueCommitmentLedger -> Bool
  101 commitmentAllowsAdvance (DialogueCommitmentLedger items) =
  102   not (any ((`elem` [CsUnresolved, CsContested, CsSuspended]) . dcStatus) items)
  103 
  104 clamp01 :: Double -> Double
  105 clamp01 = max 0.0 . min 1.0

  107 familyToSenseOperator :: CanonicalMoveFamily -> SenseOperator
  108 familyToSenseOperator family = case family of
  109   CMDefine -> OpDefine
  110   CMGround -> OpGround
  111   CMClarify -> OpClarify
  112   CMRepair -> OpRepair
  113   CMNextStep -> OpNextStep
  114   CMDeepen -> OpDeepen
  115   CMConfront -> OpDistinguish
  116   CMPurpose -> OpExplainPurpose
  117   CMDistinguish -> OpDistinguish
  118   CMReflect -> OpReflect
  119   CMDescribe -> OpGround
  120   CMAnchor -> OpGround
  121   CMContact -> OpGround
  122   CMHypothesis -> OpReflect
  123   _ -> OpGround
  124 
  125 fallbackContinuation :: SenseOperator -> SenseOperator
  126 fallbackContinuation op = case adjacentOperators op of
  127   next:_ -> next
  128   [] -> op
  129 
  130 legalizeOperator :: DialoguePhase -> DialogueCommitmentLedger -> SenseOperator -> SenseOperator
  131 legalizeOperator phase ledger op
  132   | not (commitmentAllowsOperator ledger op) = OpClarify
  133   | not (phaseAllows phase op) = phaseFallback phase
  134   | otherwise = op
  135 
  136 commitmentAllowsOperator :: DialogueCommitmentLedger -> SenseOperator -> Bool
  137 commitmentAllowsOperator (DialogueCommitmentLedger items) op =
  138   case op of
  139     OpNextStep -> clear
  140     OpDeepen -> clear
  141     OpExplainPurpose -> clear
  142     _ -> True
  143   where
  144     clear = not (any ((`elem` [CsUnresolved, CsContested, CsSuspended]) . dcStatus) items)
  145 
  146 phaseAllows :: DialoguePhase -> SenseOperator -> Bool
  147 phaseAllows phase op = op `elem` case phase of
  148   Exploring -> [OpGround, OpDefine, OpDistinguish, OpClarify, OpReflect]
  149   Clarifying -> [OpClarify, OpGround, OpDefine, OpDistinguish, OpRepair]
  150   Grounding -> [OpGround, OpDefine, OpExplainCause, OpClarify]
  151   Repairing -> [OpRepair, OpClarify, OpGround]
  152   Advancing -> [OpNextStep, OpExplainPurpose, OpDeepen, OpGround]
  153   Contesting -> [OpDistinguish, OpConstrain, OpClarify, OpGround]
  154   Closing -> [OpGround, OpReflect, OpNextStep]
  155 
  156 phaseFallback :: DialoguePhase -> SenseOperator
  157 phaseFallback phase = case phase of
  158   Repairing -> OpRepair
  159   Clarifying -> OpClarify
  160   Grounding -> OpGround
  161   Contesting -> OpConstrain
  162   Closing -> OpReflect
  163   Exploring -> OpGround
  164   Advancing -> OpGround
