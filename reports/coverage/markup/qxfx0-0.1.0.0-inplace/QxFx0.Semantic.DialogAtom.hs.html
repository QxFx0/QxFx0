   47 --------------------------------------------------------------------------------
   48 -- Atom tags — unified across facts AND dialog
   49 --------------------------------------------------------------------------------
   50 
   51 {-| Semantic tag for any piece of information in the turn context.
   52 
   53   Split into three families:
   54   - Factual: subject, properties, quantities, mechanisms — from knowledge base
   55   - Dialog: user intent, contact state, phase, stance — from turn routing
   56   - Meta: identity, constraints, goals — from system state
   57 -}
   58 data AtomTag
   59   -- Fact atoms (from knowledge decomposition)
   60   = TSubject          -- ^ Entity under discussion
   61   | TProperty         -- ^ Attribute/quality
   62   | TQuantity         -- ^ Numeric value
   63   | TUnit             -- ^ Unit of measurement
   64   | TPeriod           -- ^ Time frame
   65   | TEffect           -- ^ Result/consequence
   66   | TMechanism        -- ^ How it works
   67   | TContext          -- ^ Broader system/framework
   68   | TRelation         -- ^ Connection to other concepts
   69   | TScale            -- ^ Comparison reference
   70   | TLocation         -- ^ Where
   71   | TDiscoverer       -- ^ Who/first described
   72   | TYear             -- ^ When (historical)
   73   | TReason           -- ^ Why it matters
   74     -- Dialog atoms (from routing + proposition classification)
   75   | TUserIntent       -- ^ What user wants: define, clarify, ground, reflect, complain...
   76   | TTopic            -- ^ Subject of conversation (word/phrase)
   77   | TStance           -- ^ System stance: firm, tentative, honest, exploratory
   78   | TEmotion          -- ^ Emotional tone: distress, curious, hopeful, confrontational
   79   | TKnowledge        -- ^ Epistemic status: known, probable, uncertain, unknown
   80   | TPhase            -- ^ Dialogue phase: opening, deepening, anchoring, repair, closing
   81   | TTurnGoal         -- ^ What this turn aims to achieve
   82     -- Meta atoms (from system state)
   83   | TIdentity         -- ^ System self-description
   84   | TConstraint       -- ^ Limitation / boundary
   85   | TCapability       -- ^ What the system can do
   86   | TGreeting         -- ^ Smalltalk / contact initiation
   87   | TFarewell         -- ^ Ending interaction
   88   | TConnector        -- ^ Discourse linker: "кстати", "впрочем", "более того"
   89   | THedge            -- ^ Hedging softener: "как бы", "в некотором смысле", "пожалуй"
   90   | TSoftener         -- ^ Tone softener: "если хочешь", "может быть", "давай попробуем"
   91   | TModifier         -- ^ Adverbial/adjectival modifier from input: "коротко", "подробно", "только", "ещё"
   92    | TAgreement      -- ^ Agreement signal: "верно", "совершенно верно", "именно так"
   93    | TDisagreement   -- ^ Disagreement signal: "не совсем", "я бы поспорил", "это не так"
   94    | TRelatedTopic   -- ^ Related topic for natural transitions: "кстати о сосудах..."
   95    | TModal          -- ^ Modal particle: "ведь", "же", "разве", "всё-таки", "вообще-то"
   96    | TFiller         -- ^ Conversational filler: "ну", "так", "значит", "слушай", "смотри", "понимаешь"
   97    | TNumber          -- ^ Cardinal/ordinal: "три", "первый", "оба", "несколько"
   98    | TQuantifier      -- ^ "все", "каждый", "любой", "некоторые", "многие", "ни один", "большинство"
   99    | TEmphatic        -- ^ "именно", "даже", "просто", "всего лишь", "хотя бы"
  100 
  101   deriving stock (Eq, Ord, Show, Generic, Bounded, Enum, Read)
  102   deriving anyclass (NFData, ToJSON, FromJSON)
  103 
  104 instance ToJSONKey AtomTag where
  105   toJSONKey = ToJSONKeyValue toJSON toEncoding
  106 instance FromJSONKey AtomTag where
  107   fromJSONKey = FromJSONKeyTextParser $ \t ->
  108     case readMaybe (T.unpack t) of
  109       Just tag -> pure tag
  110       Nothing  -> fail ("Unknown AtomTag: " ++ T.unpack t)

  112 --------------------------------------------------------------------------------
  113 -- AtomSlot — a tagged value with full morphological forms
  114 --------------------------------------------------------------------------------
  115 
  116 {-| One atom slot = tag + text value + 5 Russian case forms + plural. -}
  117 data AtomSlot = AtomSlot
  118   { asTag     :: !AtomTag
  119   , asValue   :: !Text          -- ^ Primary form (nominative or raw)
  120   , asGen     :: !Text          -- ^ Genitive
  121   , asPrep    :: !Text          -- ^ Prepositional
  122   , asAcc     :: !Text          -- ^ Accusative
  123   , asIns     :: !Text          -- ^ Instrumental
  124   , asPlural  :: !Text          -- ^ Plural nominative
  125   } deriving stock (Eq, Show, Generic)
  126   deriving anyclass (NFData, ToJSON, FromJSON)
  127 
  128 {-| Slot with no morphological variation (numbers, raw text). -}
  129 plainSlot :: AtomTag -> Text -> AtomSlot
  130 plainSlot tag val = AtomSlot tag val val val val val val
  131 
  132 {-| Slot with full 5-case Russian noun forms. -}
  133 nounSlot :: AtomTag -> Text -> Text -> Text -> Text -> Text -> Text -> AtomSlot
  134 nounSlot = AtomSlot

  136 --------------------------------------------------------------------------------
  137 -- DialogAtoms — the full state of one turn
  138 --------------------------------------------------------------------------------
  139 
  140 {-| All atoms available for a single turn assembly.
  141 
  142   Populated from:
  143   - InputPropositionFrame (user intent, topic, emotion)
  144   - CanonicalMoveFamily + StanceMarker (system stance)
  145   - SystemState (identity, phase, turn count)
  146   - Knowledge base lookup (factual atoms if a known topic)
  147 -}
  148 data DialogAtoms = DialogAtoms
  149   { daSlots :: !(Map AtomTag [AtomSlot])
  150     -- ^ All slots, possibly multiple per tag
  151   , daUserRaw :: !Text
  152     -- ^ Original user input (for seed/hash/pickDeterministic)
  153   , daTurn :: !Int
  154     -- ^ Turn index in this session
  155   } deriving stock (Eq, Show, Generic)
  156   deriving anyclass (NFData, ToJSON, FromJSON)
  157 
  158 emptyDialogAtoms :: DialogAtoms
  159 emptyDialogAtoms = DialogAtoms M.empty "" 0

  165 daTopicNominative :: DialogAtoms -> Text
  166 daTopicNominative  = headAtomValue TTopic
  167 
  168 daTopicGenitive :: DialogAtoms -> Text
  169 daTopicGenitive da = case M.lookup TTopic (daSlots da) of
  170   Just (s:_) -> asGen s; _ -> "темы"
  171 
  172 daTopicPrepositional :: DialogAtoms -> Text
  173 daTopicPrepositional da = case M.lookup TTopic (daSlots da) of
  174   Just (s:_) -> asPrep s; _ -> "теме"
  175 
  176 daTopicAccusative :: DialogAtoms -> Text
  177 daTopicAccusative da = case M.lookup TTopic (daSlots da) of
  178   Just (s:_) -> asAcc s; _ -> "тему"
  179 
  180 atomValue :: AtomTag -> DialogAtoms -> [AtomSlot]
  181 atomValue tag (DialogAtoms m _ _) = M.findWithDefault [] tag m
  182 
  183 atomValues :: AtomTag -> DialogAtoms -> [Text]
  184 atomValues tag = map asValue . atomValue tag
  185 
  186 hasTag :: AtomTag -> DialogAtoms -> Bool
  187 hasTag tag = not . null . atomValue tag
  188 
  189 headAtomValue :: AtomTag -> DialogAtoms -> Text
  190 headAtomValue tag da = case atomValue tag da of
  191   (s:_) -> asValue s; [] -> ""

  193 {-| Pick the correct Russian 3rd-person pronoun for a subject. -}
  194 pronoun :: Text -> Text
  195 pronoun w =
  196   let w' = T.toLower (T.strip w)
  197   in case M.lookup w' genderDict of
  198        Just p -> p
  199        Nothing ->
  200          if " " `T.isInfixOf` w'
  201            then "они"
  202            else case T.unsnoc w' of
  203                   Just (_, 'а') | not ("мя" `T.isSuffixOf` w') -> "она"
  204                   Just (_, 'я') -> "она"
  205                   Just (_, 'о') -> "оно"
  206                   Just (_, 'е') -> "оно"
  207                   Just (_, 'ь') -> "она"
  208                   _ -> "он"
  209
