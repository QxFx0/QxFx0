   27 data EngagementLevel = HighEngagement | MediumEngagement | LowEngagement
   28   deriving stock (Eq, Show, Generic)
   29   deriving anyclass (NFData, FromJSON, ToJSON)
   30 
   31 data DialogPhase = PhaseOpening | PhaseExploring | PhaseDeep | PhaseClosing
   32   deriving stock (Eq, Show, Generic)
   33   deriving anyclass (NFData, FromJSON, ToJSON)
   34 
   35 data TurnMemory = TurnMemory
   36   { tmrTurnIndex :: !Int
   37   , tmrTopic :: !Text
   38   , tmrFamily :: !CanonicalMoveFamily
   39   , tmrRendered :: !Text
   40   } deriving stock (Eq, Show, Generic)
   41     deriving anyclass (NFData, FromJSON, ToJSON)
   42 
   43 data DiscourseState = DiscourseState
   44   { dscTurnMemory :: !(Seq TurnMemory)
   45   , dscTopicChain :: ![Text]
   46   , dscAnaphora :: !(Map Text Text)
   47   , dscDiscourseMarkers :: ![Text]
   48   , dscEngagement :: !EngagementLevel
   49   , dscPhase :: !DialogPhase
   50   } deriving stock (Eq, Show, Generic)
   51     deriving anyclass (NFData, FromJSON, ToJSON)
   52 
   53 emptyDiscourseState :: DiscourseState
   54 emptyDiscourseState = DiscourseState
   55   { dscTurnMemory = Seq.empty
   56   , dscTopicChain = []
   57   , dscAnaphora = M.empty
   58   , dscDiscourseMarkers = []
   59   , dscEngagement = MediumEngagement
   60   , dscPhase = PhaseOpening
   61   }
   62 
   63 recomputeDiscourse :: DiscourseState -> DiscourseState
   64 recomputeDiscourse ds =
   65   let turnCount = Seq.length (dscTurnMemory ds)
   66       phase = case () of
   67         _ | turnCount <= 2  -> PhaseOpening
   68         _ | turnCount <= 7  -> PhaseExploring
   69         _                   -> PhaseDeep
   70   in ds { dscPhase = phase }
