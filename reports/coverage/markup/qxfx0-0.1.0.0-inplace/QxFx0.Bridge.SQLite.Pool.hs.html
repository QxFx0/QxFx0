    5 {-| Connection and pool lifecycle helpers for the runtime SQLite backend. -}
    6 module QxFx0.Bridge.SQLite.Pool
    7   ( QxFx0DB(..)
    8   , WorkerDBPool
    9   , newDBPool
   10   , closeDBPool
   11   , withDB
   12   , withPooledDB
   13   , execOrThrow
   14   ) where
   15 
   16 import Control.Concurrent.MVar (MVar, newMVar, putMVar, takeMVar)
   17 import Control.Exception (finally, mask, mask_, onException, throwIO)
   18 import Data.Text (Text)
   19 import qualified Data.Text as T
   20 import qualified QxFx0.Bridge.NativeSQLite as NSQL
   21 import QxFx0.ExceptionPolicy
   22   ( QxFx0Exception(SQLiteError)
   23   , tryAsync
   24   , throwQxFx0

   26 import System.Timeout (timeout)

   28 data QxFx0DB = QxFx0DB
   29   { qdbPath :: !FilePath
   30   , qdbConn :: !NSQL.Database
   31   }
   32 
   33 data WorkerDBPool = WorkerDBPool
   34   { poolMVar :: !(MVar [NSQL.Database])
   35   , poolPath :: !FilePath
   36   , poolSize :: !Int
   37   }
   38 
   39 newDBPool :: FilePath -> Int -> IO WorkerDBPool
   40 newDBPool path size = do
   41   conns <- buildPoolConnections size []
   42   mvar <- newMVar conns
   43   pure WorkerDBPool {poolMVar = mvar, poolPath = path, poolSize = size}
   44   where
   45     buildPoolConnections 0 acc = pure (reverse acc)
   46     buildPoolConnections n acc = do
   47       openResult <- tryAsync (openInitializedConnection path)
   48       case openResult of
   49         Right db ->
   50           buildPoolConnections (n - 1) (db : acc)
   51         Left ex -> do
   52           mapM_ safeClose acc
   53           throwIO ex
   54 
   55 closeDBPool :: WorkerDBPool -> IO ()
   56 closeDBPool pool = mask_ $ do
   57   conns <- takeMVar (poolMVar pool)
   58   finally
   59     (mapM_ NSQL.close conns)
   60     (putMVar (poolMVar pool) [])
   61 
   62 withDB :: FilePath -> (NSQL.Database -> IO a) -> IO (Either Text a)
   63 withDB path action = do
   64   mDb <- NSQL.open path
   65   case mDb of
   66     Left err -> pure (Left err)
   67     Right db ->
   68       finally
   69         (do
   70             result <- tryAsync $ do
   71               execOrThrow db "PRAGMA journal_mode=WAL;"
   72               execOrThrow db "PRAGMA foreign_keys=ON;"
   73               action db
   74             case result of
   75               Right value -> pure (Right value)
   76               Left ex -> pure (Left ("db action failed: " <> T.pack (show ex))))
   77         (safeClose db)
   78 
   79 withPooledDB :: WorkerDBPool -> (NSQL.Database -> IO a) -> IO a
   80 withPooledDB pool action = mask $ \restore -> do
   81   mConns <- timeout poolAcquireTimeoutMicros (takeMVar (poolMVar pool))
   82   conns <-
   83     case mConns of
   84       Nothing ->
   85         throwQxFx0
   86           (SQLiteError
   87             ("timed out waiting for SQLite pool connection (pool_size="
   88               <> T.pack (show (poolSize pool))
   89               <> ")"))
   90       Just available -> pure available
   91   withConnections restore conns
   92   where
   93     poolAcquireTimeoutMicros :: Int
   94     poolAcquireTimeoutMicros = 5000000
   95 
   96     withConnections restore [] = do
   97       db <- openInitializedConnection (poolPath pool)
   98       finally
   99         (restore (action db))
  100         (safeClose db `finally` putMVar (poolMVar pool) [])
  101     withConnections restore (db : dbs) =
  102       finally
  103         (restore (action db))
  104         (restorePooledConnection db dbs)
  105 
  106     restorePooledConnection db dbs = do
  107       mDb <- sanitizeForPool db
  108       case mDb of
  109         Just cleanDb ->
  110           putMVar (poolMVar pool) (cleanDb : dbs)
  111         Nothing ->
  112           putMVar (poolMVar pool) dbs
  113 
  114     sanitizeForPool db = do
  115       rollbackResult <- NSQL.execSql db "ROLLBACK;"
  116       case rollbackResult of
  117         Right _ ->
  118           pure (Just db)
  119         Left err
  120           | isNoActiveTransactionError err ->
  121               pure (Just db)
  122           | otherwise -> do
  123               safeClose db
  124               replacement <- tryAsync (openInitializedConnection (poolPath pool))
  125               case replacement of
  126                 Right freshDb -> pure (Just freshDb)
  127                 Left _ -> pure Nothing
  128 
  129 isNoActiveTransactionError :: Text -> Bool
  130 isNoActiveTransactionError err =
  131   "no transaction is active" `T.isInfixOf` T.toLower err
  132 
  133 openInitializedConnection :: FilePath -> IO NSQL.Database
  134 openInitializedConnection path = do
  135   mDb <- NSQL.open path
  136   case mDb of
  137     Left err ->
  138       throwQxFx0 (SQLiteError err)
  139     Right db ->
  140       (do
  141           execOrThrow db "PRAGMA journal_mode=WAL;"
  142           execOrThrow db "PRAGMA synchronous=NORMAL;"
  143           execOrThrow db "PRAGMA foreign_keys=ON;"
  144           pure db)
  145       `onException` safeClose db
  146 
  147 safeClose :: NSQL.Database -> IO ()
  148 safeClose db = do
  149   _ <- tryAsync (NSQL.close db)
  150   pure ()
  151 
  152 execOrThrow :: NSQL.Database -> Text -> IO ()
  153 execOrThrow db sql = do
  154   result <- NSQL.execSql db sql
  155   case result of
  156     Left err ->
  157       throwQxFx0
  158         (SQLiteError ("sqlite exec failed for `" <> sql <> "`: " <> err))
  159     Right _ -> pure ()
