   51 -- | Stable identifier for a meaning state used as graph node key.
   52 meaningStateId :: MeaningState -> MeaningStateId
   53 meaningStateId ms = concat
   54   [ show (msResonance ms), "_"
   55   , show (msPressure ms), "_"
   56   , show (msDepth ms)
   57   ]

   59 toDepthBand :: Int -> DepthBand
   60 toDepthBand d
   61   | d <= 1    = DepthShallow
   62   | d == 2    = DepthMech
   63   | d == 3    = DepthPattern
   64   | otherwise = DepthAxiom

   72 successRate :: MeaningEdge -> Double
   73 successRate e
   74   | meCount e == 0 = 0.0
   75   | otherwise       = fromIntegral (meWins e) / fromIntegral (meCount e)

   77 -- | Record one observed transition and update edge counters.
   78 recordTransition :: MeaningState -> MeaningState -> ResponseStrategy -> Bool -> MeaningGraph -> MeaningGraph
   79 recordTransition fromState toState strat win g =
   80   let fid = meaningStateId fromState
   81       tid = meaningStateId toState
   82       mEdge = find (\e -> meFromId e == fid && meToId e == tid) (mgEdges g)
   83   in case mEdge of
   84        Just edge ->
   85          let updated = edge
   86                { meCount = meCount edge + 1
   87                , meWins  = meWins edge + if win then 1 else 0
   88                }
   89          in g { mgEdges = replaceEdge (mgEdges g) updated }
   90        Nothing ->
   91          let newEdge = MeaningEdge
   92                { meFromId = fid
   93                , meToId = tid
   94                , meFrom = fromState
   95                , meTo = toState
   96                , meStrategy = strat
   97                , meCount = 1
   98                , meWins = if win then 1 else 0
   99                , meDreamBias = 0.0
  100                , meLastRewiredAt = Nothing
  101                }
  102          in g { mgEdges = take maxEdges (newEdge : mgEdges g)
  103               , mgTurnCount = mgTurnCount g + 1 }
  104 
  105 replaceEdge :: [MeaningEdge] -> MeaningEdge -> [MeaningEdge]
  106 replaceEdge edges newE = map (\e -> if meFromId e == meFromId newE && meToId e == meToId newE then newE else e) edges

  108 -- | Predict a reusable strategy when the edge routing score is above threshold.
  109 predictStrategy :: MeaningState -> MeaningState -> MeaningGraph -> Maybe ResponseStrategy
  110 predictStrategy fromState toState g =
  111   let fid = meaningStateId fromState
  112       tid = meaningStateId toState
  113   in case find (\e -> meFromId e == fid && meToId e == tid) (mgEdges g) of
  114      Just edge
  115         | edgeRoutingScore edge > meaningGraphRoutingThreshold -> Just (meStrategy edge)
  116         | otherwise -> Nothing
  117      Nothing -> Nothing

  119 -- | Apply dream-cycle deltas to edge biases and return emitted rewire events.
  120 rewireMeaningGraphForDreamCycle :: UTCTime -> [(MeaningEdge, Double)] -> MeaningGraph -> (MeaningGraph, [Text])
  121 rewireMeaningGraphForDreamCycle now edgeDeltas g =
  122   let (newEdges, events) = unzip (map (rewireMeaningEdge now) edgeDeltas)
  123       g' = g { mgEdges = mergeEdges (mgEdges g) newEdges }
  124   in (g', concat events)
  125 
  126 rewireMeaningEdge :: UTCTime -> (MeaningEdge, Double) -> (MeaningEdge, [Text])
  127 rewireMeaningEdge now (edge, delta) =
  128   let newBias = clampSymmetric meaningGraphDreamBiasLimit (meDreamBias edge + delta)
  129       actualDelta = newBias - meDreamBias edge
  130       edge' = edge
  131         { meDreamBias = newBias
  132         , meLastRewiredAt = Just now
  133         }
  134       event = if abs actualDelta > 1e-6
  135               then [T.pack (meFromId edge) <> "->" <> T.pack (meToId edge) <> " bias:" <> textShow newBias]
  136               else []
  137   in (edge', event)
  138 
  139 mergeEdges :: [MeaningEdge] -> [MeaningEdge] -> [MeaningEdge]
  140 mergeEdges existing newEdges = foldl (\es ne -> replaceEdge es ne) existing newEdges

  142 defaultStrategy :: MeaningState -> ResponseStrategy
  143 defaultStrategy ms = ResponseStrategy
  144   { rsDepth    = case msResonance ms of ResonanceHigh -> DeepResp; ResonanceMed -> ModerateResp; ResonanceLow -> ShallowResp
  145   , rsStance   = case msPressure ms of PressHeavy -> HoldStance; PressLight -> AcknowledgeStance; PressNone -> OpenStance
  146   , rsMove     = case msPressure ms of PressHeavy -> CounterMove; PressLight -> ReframeMove; PressNone -> ValidateMove
  147   , rsDensityT = case msDepth ms of DepthShallow -> DensityMed; DepthMech -> DensityMed; DepthPattern -> DensityHigh; DepthAxiom -> DensityHigh
  148   }

  150 graphStats :: MeaningGraph -> Text
  151 graphStats mg =
  152   let nEdges = length (mgEdges mg)
  153       avgRate = if nEdges == 0 then 0.0 else sum (map successRate (mgEdges mg)) / fromIntegral nEdges
  154   in textShow nEdges <> " edges, avg success " <> textShow avgRate
  155 
  156 edgeRoutingScore :: MeaningEdge -> Double
  157 edgeRoutingScore edge = successRate edge + meDreamBias edge

  159 clampSymmetric :: Double -> Double -> Double
  160 clampSymmetric limit v = max (-limit) (min limit v)
