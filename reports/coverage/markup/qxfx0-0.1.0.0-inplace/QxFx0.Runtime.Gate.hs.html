    1 {-# LANGUAGE DerivingStrategies #-}
    2 {-# LANGUAGE OverloadedStrings #-}

    4 module QxFx0.Runtime.Gate
    5   ( RuntimeGateFailure(..)
    6   , evaluateBootstrapReadiness
    7   , evaluateStrictHealth
    8   , renderBootstrapGateFailure
    9   , renderTurnGateFailure
   10   ) where
   11 
   12 import Data.Text (Text)
   13 import qualified Data.Text as T
   14 
   15 import QxFx0.Resources (ReadinessComponent, ReadinessMode(..))
   16 import QxFx0.Runtime.Health (SystemHealth(..))
   17 import QxFx0.Runtime.Mode (RuntimeMode, isStrictRuntimeMode)
   18 
   19 data RuntimeGateFailure
   20   = GateCriticalReadiness ![ReadinessComponent]
   21   | GateOptionalReadiness ![ReadinessComponent]
   22   | GateStrictHealth !SystemHealth
   23   deriving stock (Eq, Show)
   24 
   25 evaluateBootstrapReadiness :: RuntimeMode -> ReadinessMode -> Either RuntimeGateFailure ()
   26 evaluateBootstrapReadiness runtimeMode readinessMode =
   27   case readinessMode of
   28     NotReady failed ->
   29       Left (GateCriticalReadiness failed)
   30     Degraded failed | isStrictRuntimeMode runtimeMode ->
   31       Left (GateOptionalReadiness failed)
   32     _ ->
   33       Right ()
   34 
   35 evaluateStrictHealth :: RuntimeMode -> SystemHealth -> Either RuntimeGateFailure ()
   36 evaluateStrictHealth runtimeMode health
   37   | not (isStrictRuntimeMode runtimeMode) =
   38       Right ()
   39   | shReady health && shStatus health == "ok" =
   40       Right ()
   41   | otherwise =
   42       Left (GateStrictHealth health)
   43 
   44 renderBootstrapGateFailure :: RuntimeGateFailure -> Text
   45 renderBootstrapGateFailure failure =
   46   case failure of
   47     GateCriticalReadiness failed ->
   48       "System cannot start: critical components unavailable: " <> renderComponents failed
   49     GateOptionalReadiness failed ->
   50       "Strict runtime requires all optional resources: " <> renderComponents failed
   51     GateStrictHealth health ->
   52       "Strict runtime requires status=ok, got " <> shStatus health <> " [" <> shReadinessMode health <> "]; " <> renderStrictHealthDetail health
   53 
   54 renderTurnGateFailure :: RuntimeGateFailure -> Text
   55 renderTurnGateFailure failure =
   56   case failure of
   57     GateCriticalReadiness failed ->
   58       "Turn blocked: critical components unavailable: " <> renderComponents failed
   59     GateOptionalReadiness failed ->
   60       "Turn blocked: strict runtime requires all optional resources: " <> renderComponents failed
   61     GateStrictHealth health ->
   62       "Turn blocked: strict runtime requires status=ok, got " <> shStatus health <> " [" <> shReadinessMode health <> "]; " <> renderStrictHealthDetail health
   63 
   64 renderComponents :: [ReadinessComponent] -> Text
   65 renderComponents = T.pack . show
   66 
   67 renderStrictHealthDetail :: SystemHealth -> Text
   68 renderStrictHealthDetail health =
   69   let components =
   70         [ "db=" <> T.pack (show (shDbAlive health && shDbBootstrapable health))
   71         , "schema=" <> T.pack (show (shSchemaOk health)) <> "(v=" <> T.pack (show (shSchemaVersion health)) <> ")"
   72         , "agda=" <> T.pack (show (shAgdaReady health))
   73         , "datalog=" <> T.pack (show (shDatalogReady health))
   74         , "embed=" <> T.pack (show (shEmbeddingAlive health))
   75         , "decision_local_only=" <> T.pack (show (shDecisionPathLocalOnly health))
   76         , "network_optional_only=" <> T.pack (show (shNetworkOptionalOnly health))
   77         , "llm_decision_path=" <> T.pack (show (shLlmDecisionPath health))
   78         , "morpho=" <> T.pack (show (shMorphoReady health))
   79         , "morph_backend=" <> shMorphBackend health
   80         , "nix=" <> T.pack (show (shNixReady health))
   81             <> "(policy=" <> T.pack (show (shNixPolicyPresent health)) <> ")"
   82         ]
   83       issues = concat
   84         [ if null (shAgdaIssues health) then [] else ["agda_issues=" <> T.intercalate "," (shAgdaIssues health)]
   85         , if null (shDatalogIssues health) then [] else ["datalog_issues=" <> T.intercalate "," (shDatalogIssues health)]
   86         , if null (shNixIssues health) then [] else ["nix_issues=" <> T.intercalate "," (shNixIssues health)]
   87         , if T.null (shSchemaReason health) then [] else ["schema_reason=" <> shSchemaReason health]
   88         ]
   89   in T.intercalate "; " (components ++ issues)
