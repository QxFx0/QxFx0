   39 planFinalizeCommit :: Text -> SystemState -> TurnSignals -> TurnArtifacts -> FinalizePrecommitBundle -> FinalizeCommitPlan
   40 planFinalizeCommit sessionId previousState turnSignals turnArtifacts bundle =
   41   FinalizeCommitPlan
   42     { fcpResponseObservation =
   43         ResponseObservation
   44           { roSurfaceText = taFinalRendered turnArtifacts
   45           , roQuestionLike = Guard.gsQuestionLike (taGuardSurface turnArtifacts)
   46           }
   47     , fcpPreviewConsciousLoop = tsConsciousLoop' turnSignals
   48     , fcpPreviewIntuition = tsIntuitionState turnSignals
   49     , fcpPreviousState = previousState
   50     , fcpSaveState = fpbNextSs bundle
   51     , fcpSessionId = sessionId
   52     , fcpProjection = fpbProjection bundle
   53     , fcpRewireEventsCount = fpbRewireEventsCount bundle
   54     }

   56 resolveFinalizeCommit :: PipelineIO -> FinalizeCommitPlan -> IO FinalizeCommitResults
   57 resolveFinalizeCommit pipelineIO commitPlan = do
   58   unless (fcpRewireEventsCount commitPlan == 0) $
   59     hPutStrLnWarning ("Dream rewiring: " ++ show (fcpRewireEventsCount commitPlan) ++ " edges adjusted")
   60 
   61   saveStart <- resolveCommitCurrentTime pipelineIO
   62   saveResult <-
   63     resolveTurnEffect
   64       pipelineIO
   65       (TurnReqSaveState (fcpSaveState commitPlan) (fcpSessionId commitPlan) (Just (fcpProjection commitPlan)))
   66   savedState <-
   67     case saveResult of
   68       TurnResSaveState (Right savedSystemState) -> pure savedSystemState
   69       TurnResSaveState (Left err) -> throwQxFx0 (PersistenceError ("saveStateWithProjection failed: " <> renderPersistenceDiagnostics [err]))
   70       _ -> throwQxFx0 (PersistenceError "saveStateWithProjection returned unexpected result")
   71   commitAttempt <-
   72     tryAsync (attemptCommitRuntimeState pipelineIO commitPlan (fcpPreviewIntuition commitPlan))
   73   case commitAttempt of
   74     Right () ->
   75       pure ()
   76     Left commitErr -> do
   77       recoveryAttempt <- tryAsync (recoverRuntimeTurnState pipelineIO commitPlan savedState)
   78       case recoveryAttempt of
   79         Right () ->
   80           hPutStrLnWarning
   81             ("[warn] commit runtime state failed after save; state re-hydrated from persisted snapshot: "
   82               <> show commitErr)
   83         Left recoveryErr -> do
   84           projectionsRollbackSucceeded <- attemptRollbackPersistedProjections pipelineIO commitPlan
   85           unless projectionsRollbackSucceeded $
   86             hPutStrLnWarning
   87               "[warn] rollback of persisted turn projections failed after commit/recovery failure"
   88           stateRollbackSucceeded <- attemptRollbackPersistedState pipelineIO commitPlan
   89           unless stateRollbackSucceeded $
   90             hPutStrLnWarning
   91               "[warn] rollback to previous persisted state failed after commit/recovery failure"
   92           let projectionsRollbackText = if projectionsRollbackSucceeded then "ok" else "failed"
   93               stateRollbackText = if stateRollbackSucceeded then "ok" else "failed"
   94           throwQxFx0
   95             (PersistenceError
   96               ("commit runtime state failed after saveState: "
   97                 <> T.pack (show commitErr)
   98                 <> "; recovery failed: "
   99                 <> T.pack (show recoveryErr)
  100                 <> "; projections rollback="
  101                 <> projectionsRollbackText
  102                 <> "; state rollback="
  103                 <> stateRollbackText))
  104   runBestEffortPostCommit "housekeeping" $ do
  105     maybeInjectPostCommitTailException pipelineIO
  106     _ <- resolveTurnEffect pipelineIO (TurnReqCheckpoint (ssTurnCount savedState))
  107     pure ()
  108   saveEnd <- resolveCommitCurrentTime pipelineIO
  109   pure
  110     FinalizeCommitResults
  111       { fcrSavedSs = savedState
  112       , fcrSaveStart = saveStart
  113       , fcrSaveEnd = saveEnd
  114       }

  116 buildFinalizeTurnResult :: TurnInput -> TurnSignals -> TurnArtifacts -> FinalizePrecommitBundle -> FinalizeCommitResults -> TurnResult
  117 buildFinalizeTurnResult turnInput turnSignals turnArtifacts bundle commitResults =
  118   let savedState = fcrSavedSs commitResults
  119       metricsFinal =
  120         finalizeMetrics
  121           turnInput
  122           turnArtifacts
  123           (fpbOutcomeFamily bundle)
  124           (fpbDecision bundle)
  125           savedState
  126           (tsApiHealthy turnSignals)
  127           (fpbFinalSafetyStatus bundle)
  128           (fcrSaveStart commitResults)
  129           (fcrSaveEnd commitResults)
  130    in TurnResult
  131         { trNextSs = savedState
  132         , trOutput = fpbOutput bundle
  133         , trMetrics = metricsFinal
  134         }

  136 resolveFinalizePostCommit :: TurnMetrics -> IO ()
  137 resolveFinalizePostCommit metrics =
  138   runBestEffortPostCommit "metrics_log" $
  139     logMetrics metrics

  141 resolveCommitCurrentTime :: PipelineIO -> IO UTCTime
  142 resolveCommitCurrentTime pipelineIO = do
  143   result <- resolveTurnEffect pipelineIO TurnReqCurrentTime
  144   case result of
  145     TurnResCurrentTime currentTime -> pure currentTime
  146     _ -> throwQxFx0 (PersistenceError "current time effect returned unexpected result")
  147 
  148 runBestEffortPostCommit :: String -> IO a -> IO ()
  149 runBestEffortPostCommit label action = do
  150   result <- tryAsync action
  151   case result of
  152     Left err -> hPutStrLnWarning $ "[warn] post-commit " <> label <> " failed: " <> show err
  153     Right _ -> pure ()
  154 
  155 attemptCommitRuntimeState :: PipelineIO -> FinalizeCommitPlan -> IntuitiveState -> IO ()
  156 attemptCommitRuntimeState pipelineIO commitPlan previewIntuition = do
  157   commitResult <-
  158     resolveTurnEffect
  159       pipelineIO
  160       (TurnReqCommitRuntimeState
  161         (fcpPreviewConsciousLoop commitPlan)
  162         previewIntuition
  163         (fcpResponseObservation commitPlan))
  164   case commitResult of
  165     TurnResCommitRuntimeState -> pure ()
  166     _ -> throwQxFx0 (PersistenceError "commit runtime state effect returned unexpected result")
  167 
  168 recoverRuntimeTurnState :: PipelineIO -> FinalizeCommitPlan -> SystemState -> IO ()
  169 recoverRuntimeTurnState pipelineIO commitPlan savedState =
  170   attemptCommitRuntimeState
  171     pipelineIO
  172     commitPlan
  173     (maybe (fcpPreviewIntuition commitPlan) id (ssIntuitionState savedState))
  174 
  175 attemptRollbackPersistedProjections :: PipelineIO -> FinalizeCommitPlan -> IO Bool
  176 attemptRollbackPersistedProjections pipelineIO commitPlan = do
  177   rollbackAttempt <-
  178     tryAsync
  179       (resolveTurnEffect
  180         pipelineIO
  181         (TurnReqRollbackTurnProjections (fcpSessionId commitPlan) (ssTurnCount (fcpPreviousState commitPlan))))
  182   case rollbackAttempt of
  183     Left _ ->
  184       pure False
  185     Right (TurnResRollbackTurnProjections (Right ())) ->
  186       pure True
  187     Right _ ->
  188       pure False
  189 
  190 attemptRollbackPersistedState :: PipelineIO -> FinalizeCommitPlan -> IO Bool
  191 attemptRollbackPersistedState pipelineIO commitPlan = do
  192   rollbackAttempt <-
  193     tryAsync
  194       (resolveTurnEffect
  195         pipelineIO
  196         (TurnReqSaveState (fcpPreviousState commitPlan) (fcpSessionId commitPlan) Nothing))
  197   case rollbackAttempt of
  198     Left _ ->
  199       pure False
  200     Right (TurnResSaveState (Right _)) ->
  201       pure True
  202     Right _ ->
  203       pure False
  204 
  205 maybeInjectPostCommitTailException :: PipelineIO -> IO ()
  206 maybeInjectPostCommitTailException pipelineIO = do
  207   testModeResult <- resolveTurnEffect pipelineIO (TurnReqReadEnv "QXFX0_TEST_MODE")
  208   case testModeResult of
  209     TurnResReadEnv (Just "1") -> do
  210       markerPathResult <-
  211         resolveTurnEffect pipelineIO (TurnReqReadEnv "QXFX0_TEST_POST_COMMIT_TAIL_EXCEPTION_ONCE_FILE")
  212       case markerPathResult of
  213         TurnResReadEnv (Just pathTextRaw) ->
  214           let pathText = T.strip pathTextRaw
  215           in unless (T.null pathText) $ do
  216               markResult <- resolveTurnEffect pipelineIO (TurnReqTestMarkOnceFile pathText)
  217               case markResult of
  218                 TurnResTestMarkOnceFile True ->
  219                   throwQxFx0 (PersistenceError "test_post_commit_tail_exception")
  220                 _ ->
  221                   pure ()
  222         _ ->
  223           pure ()
  224     _ ->
  225       pure ()
