   21 compareShadowOutput :: R5Verdict -> R5Verdict -> ShadowDivergence
   22 compareShadowOutput haskellVerdict datalogVerdict =
   23   let familyMismatch = r5Family haskellVerdict /= r5Family datalogVerdict
   24       forceMismatch = r5Force haskellVerdict /= r5Force datalogVerdict
   25       clauseMismatch = r5Clause haskellVerdict /= r5Clause datalogVerdict
   26       layerMismatch = r5Layer haskellVerdict /= r5Layer datalogVerdict
   27       warrantedMismatch = r5Warranted haskellVerdict /= r5Warranted datalogVerdict
   28       hasMismatch =
   29         or
   30           [ familyMismatch
   31           , forceMismatch
   32           , clauseMismatch
   33           , layerMismatch
   34           , warrantedMismatch
   35           ]
   36    in ShadowDivergence
   37         { sdKind = if hasMismatch then ShadowVerdictMismatch else ShadowNoDivergence
   38         , sdFamilyMismatch = familyMismatch
   39         , sdForceMismatch = forceMismatch
   40         , sdClauseMismatch = clauseMismatch
   41         , sdLayerMismatch = layerMismatch
   42         , sdWarrantedMismatch = warrantedMismatch
   43         }
