   67 compileAndRunDatalog :: Text -> CanonicalMoveFamily -> IO (Either Text R5Verdict)
   68 compileAndRunDatalog dlSource fam = do
   69   programResult <-
   70     if T.null (T.strip dlSource)
   71       then loadDatalogProgram
   72       else pure (Right dlSource)
   73   case programResult of
   74     Left err -> pure (Left err)
   75     Right program -> do
   76       execResult <- resolveSouffleExecutable
   77       case execResult of
   78         Left err -> pure (Left err)
   79         Right executable ->
   80           fmap deVerdict
   81             <$> executeDatalogShadowWithExecutable
   82                   executable
   83                   program
   84                   (buildShadowSnapshot fam (forceForFamily fam) [])
   85 
   86 compileAndRunDatalogWithExecutable :: FilePath -> Text -> CanonicalMoveFamily -> IO (Either Text R5Verdict)
   87 compileAndRunDatalogWithExecutable executable dlSource fam = do
   88   programResult <-
   89     if T.null (T.strip dlSource)
   90       then loadDatalogProgram
   91       else pure (Right dlSource)
   92   case programResult of
   93     Left err -> pure (Left err)
   94     Right program ->
   95       fmap deVerdict
   96         <$> executeDatalogShadowWithExecutable
   97               executable
   98               program
   99               (buildShadowSnapshot fam (forceForFamily fam) [])

  101 runDatalogShadow :: CanonicalMoveFamily -> IllocutionaryForce -> [AtomTag] -> IO ShadowResult
  102 runDatalogShadow fam force atoms = do
  103   let snapshot = buildShadowSnapshot fam force atoms
  104   programResult <- loadDatalogProgram
  105   case programResult of
  106     Left err -> pure (shadowUnavailableResult snapshot ShadowExecutionError err)
  107     Right _ -> do
  108       execResult <- resolveSouffleExecutable
  109       case execResult of
  110         Left err -> pure (shadowUnavailableResult snapshot ShadowUnavailableDivergence err)
  111         Right executable -> runDatalogShadowWithExecutableSnapshot executable snapshot
  112 
  113 runDatalogShadowWithExecutable :: FilePath -> CanonicalMoveFamily -> IllocutionaryForce -> [AtomTag] -> IO ShadowResult
  114 runDatalogShadowWithExecutable executable fam force atoms =
  115   runDatalogShadowWithExecutableSnapshot executable (buildShadowSnapshot fam force atoms)

  117 resolveSouffleExecutable :: IO (Either Text FilePath)
  118 resolveSouffleExecutable = do
  119   envConfigured <- lookupEnv "QXFX0_SOUFFLE_BIN"
  120   case nonEmptyString (fmap stripWhitespace envConfigured) of
  121     Just candidate ->
  122       resolveConfiguredSouffleExecutable candidate
  123     Nothing -> do
  124       localPath <- findExecutable "souffle"
  125       case localPath of
  126         Just path -> pure (Right path)
  127         Nothing -> resolveSouffleExecutableFromFlake
  128 
  129 resolveConfiguredSouffleExecutable :: FilePath -> IO (Either Text FilePath)
  130 resolveConfiguredSouffleExecutable configuredPath
  131   | any isPathSeparator configuredPath =
  132       validateConfiguredSoufflePath configuredPath
  133   | otherwise = do
  134       found <- findExecutable configuredPath
  135       pure $
  136         case found of
  137           Just path -> Right path
  138           Nothing ->
  139             Left ("configured souffle executable not found in PATH: " <> T.pack configuredPath)
  140 
  141 validateConfiguredSoufflePath :: FilePath -> IO (Either Text FilePath)
  142 validateConfiguredSoufflePath configuredPath =
  143   catchIO
  144     (do canonicalPath <- canonicalizePath configuredPath
  145         exists <- doesFileExist canonicalPath
  146         if not exists
  147           then pure (Left ("configured souffle binary missing: " <> T.pack canonicalPath))
  148           else do
  149             perms <- getPermissions canonicalPath
  150             if not (Directory.executable perms)
  151               then pure (Left ("configured souffle binary is not executable: " <> T.pack canonicalPath))
  152               else do
  153                 trustedRoots <- resolveSouffleTrustedRoots
  154                 trustedMatches <- mapM (`isPathWithin` canonicalPath) trustedRoots
  155                 pure $
  156                   if or trustedMatches
  157                     then Right canonicalPath
  158                     else
  159                       Left
  160                         ( "configured souffle binary is outside trusted roots: "
  161                             <> T.pack canonicalPath
  162                         ))
  163     (\e -> pure (Left ("cannot resolve configured souffle binary: " <> compactDiagnostic (T.pack (show e)))))
  164 
  165 resolveSouffleTrustedRoots :: IO [FilePath]
  166 resolveSouffleTrustedRoots = do
  167   mEnvRoot <- lookupEnv "QXFX0_ROOT"
  168   mResourceRoot <- catchIO
  169     (Just . rpResourceDir <$> resolveResourcePaths)
  170     (\_ -> pure Nothing)
  171   let configuredRoots =
  172         [ "/nix/store"
  173         , "/usr/bin"
  174         , "/usr/local/bin"
  175         ]
  176       discoveredRoots =
  177         maybeToList (nonEmptyString (fmap stripWhitespace mEnvRoot))
  178           <> maybeToList mResourceRoot
  179       candidates = nub (configuredRoots <> discoveredRoots)
  180   existing <- filterM doesDirectoryExist candidates
  181   mapM canonicalizePath existing
  182 
  183 stripWhitespace :: String -> String
  184 stripWhitespace = reverse . dropWhile isSpace . reverse . dropWhile isSpace
  185 
  186 nonEmptyString :: Maybe String -> Maybe String
  187 nonEmptyString = (>>= (\value -> if null value then Nothing else Just value))

  189 runDatalogShadowWithExecutableSnapshot :: FilePath -> ShadowSnapshot -> IO ShadowResult
  190 runDatalogShadowWithExecutableSnapshot executable snapshot = do
  191   let haskellVerdict = mkVerdict (ssRequestedFamily snapshot)
  192   programResult <- loadDatalogProgram
  193   case programResult of
  194     Left err ->
  195       pure (shadowUnavailableResult snapshot ShadowExecutionError err)
  196     Right program -> do
  197       execResult <- executeDatalogShadowWithExecutable executable program snapshot
  198       case execResult of
  199         Left err ->
  200           pure (shadowUnavailableResult snapshot ShadowExecutionError err)
  201         Right execution -> do
  202           let datalogVerdict = deVerdict execution
  203               divergence = compareShadowOutput haskellVerdict datalogVerdict
  204               anyMismatch =
  205                 or
  206                   [ sdFamilyMismatch divergence
  207                   , sdForceMismatch divergence
  208                   , sdClauseMismatch divergence
  209                   , sdLayerMismatch divergence
  210                   , sdWarrantedMismatch divergence
  211                   ]
  212           pure ShadowResult
  213             { srStatus = if anyMismatch then ShadowDiverged else ShadowMatch
  214             , srDivergence = divergence
  215             , srDatalogVerdict = Just datalogVerdict
  216             , srSnapshotId = mkShadowSnapshotId snapshot
  217             , srDiagnostics = deDiagnostics execution
  218             }

  220 resolveSouffleExecutableFromFlake :: IO (Either Text FilePath)
  221 resolveSouffleExecutableFromFlake =
  222   catchIO
  223     (do paths <- resolveResourcePaths
  224         let repoRoot = rpResourceDir paths
  225         evalResult <- resolveSouffleExecutablePathFromFlake repoRoot
  226         case evalResult of
  227           Left err -> pure (Left err)
  228           Right path -> do
  229             validated <- validateResolvedSouffleExecutable "flake-resolved souffle" path
  230             case validated of
  231               Right executable -> pure (Right executable)
  232               Left missingErr ->
  233                 if "flake-resolved souffle missing:" `T.isPrefixOf` missingErr
  234                   then materializeSouffleExecutableFromFlake repoRoot
  235                   else pure (Left missingErr))
  236     (\e -> pure (Left ("cannot resolve souffle executable: " <> compactDiagnostic (T.pack (show e)))))
  237 
  238 resolveSouffleExecutablePathFromFlake :: FilePath -> IO (Either Text FilePath)
  239 resolveSouffleExecutablePathFromFlake repoRoot = do
  240   let command =
  241         flakeCommand
  242           repoRoot
  243           ["eval", "--raw", ".#apps.x86_64-linux.souffle-runtime.program"]
  244   (exitCode, stdout, stderr) <- readCreateProcessWithExitCode command ""
  245   pure $
  246     case exitCode of
  247       ExitSuccess -> Right (T.unpack (T.strip (T.pack stdout)))
  248       ExitFailure _ ->
  249         Left ("nix eval for souffle-runtime failed: " <> compactDiagnostic (T.pack stderr))
  250 
  251 materializeSouffleExecutableFromFlake :: FilePath -> IO (Either Text FilePath)
  252 materializeSouffleExecutableFromFlake repoRoot = do
  253   let command =
  254         flakeCommand
  255           repoRoot
  256           ["build", "--no-link", "--print-out-paths", ".#souffle-runtime"]
  257   (exitCode, stdout, stderr) <- readCreateProcessWithExitCode command ""
  258   case exitCode of
  259     ExitSuccess -> do
  260       let outPath = T.unpack (T.strip (T.pack stdout))
  261           executable = outPath </> "bin" </> "souffle"
  262       validateResolvedSouffleExecutable "flake-built souffle" executable
  263     ExitFailure _ ->
  264       pure (Left ("nix build for souffle-runtime failed: " <> compactDiagnostic (T.pack stderr)))
  265 
  266 flakeCommand :: FilePath -> [String] -> CreateProcess
  267 flakeCommand repoRoot args =
  268   (proc "nix" (["--option", "warn-dirty", "false", "--extra-experimental-features", "nix-command flakes"] <> args))
  269     { cwd = Just repoRoot }

  271 validateResolvedSouffleExecutable :: Text -> FilePath -> IO (Either Text FilePath)
  272 validateResolvedSouffleExecutable sourceLabel path = do
  273   exists <- doesFileExist path
  274   if not exists
  275     then pure (Left (sourceLabel <> " missing: " <> T.pack path))
  276     else do
  277       perms <- getPermissions path
  278       pure $
  279         if Directory.executable perms
  280           then Right path
  281           else Left (sourceLabel <> " is not executable: " <> T.pack path)

  283 loadDatalogProgram :: IO (Either Text Text)
  284 loadDatalogProgram =
  285   catchIO
  286     (do checkedPaths <- resolveDatalogRuleCandidates
  287         let readFirst [] = pure (Left ("datalog rules missing: checked=" <> T.intercalate "|" (map T.pack checkedPaths)))
  288             readFirst (rulesPath:rest) = do
  289               result <- try (TIO.readFile rulesPath) :: IO (Either IOException Text)
  290               case result of
  291                 Right program -> pure (Right program)
  292                 Left _ -> readFirst rest
  293         readFirst checkedPaths)
  294     (\e -> pure (Left ("cannot load datalog rules: " <> compactDiagnostic (T.pack (show e)))))

  296 executeDatalogShadowWithExecutable :: FilePath -> Text -> ShadowSnapshot -> IO (Either Text DatalogExecution)
  297 executeDatalogShadowWithExecutable executable dlSource snapshot = do
  298   souffleTimeoutMicros <- resolveSouffleTimeoutMicros
  299   result <- catchIO
  300     (do (dlFile, outDir, cleanup) <- createShadowTempFiles
  301         let program = dlSource <> "\n" <> renderRuntimeFacts snapshot
  302         finalResult <-
  303           (do TIO.writeFile dlFile program
  304               runResult <- timeout souffleTimeoutMicros (readProcessWithExitCode executable ["-D", outDir, dlFile] "")
  305               case runResult of
  306                 Nothing ->
  307                   pure
  308                     (Left
  309                       ("souffle timed out after " <> T.pack (show (souffleTimeoutMicros `div` 1000)) <> "ms"))
  310                 Just (exitCode, _stdout, stderr) ->
  311                   case exitCode of
  312                     ExitSuccess -> parseShadowOutput outDir
  313                     ExitFailure _ -> pure (Left ("souffle failed: " <> compactDiagnostic (T.pack stderr))))
  314           `finally` cleanup
  315         pure finalResult)
  316     (\e -> pure (Left ("datalog exception: " <> compactDiagnostic (T.pack (show e)))))
  317   pure result
  318 
  319 resolveSouffleTimeoutMicros :: IO Int
  320 resolveSouffleTimeoutMicros = do
  321   mRaw <- lookupEnv "QXFX0_SOUFFLE_TIMEOUT_MS"
  322   let timeoutMs =
  323         case mRaw >>= readMaybe of
  324           Just n | n > 0 -> n
  325           _ -> 5000
  326   pure (timeoutMs * 1000)
