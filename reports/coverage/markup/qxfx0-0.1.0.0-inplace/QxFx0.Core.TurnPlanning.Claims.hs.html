   14 integrateIdentityClaims :: [IdentityClaimRef] -> CanonicalMoveFamily -> Text -> [IdentityClaimRef]
   15 integrateIdentityClaims claims _family topic =
   16   let scored = scoreClaims claims topic
   17       topClaims = selectTopClaims 5 scored
   18       deduped = deduplicateClaims topClaims
   19    in filter (\claim -> icrConfidence claim >= minIdentityClaimConfidence) deduped
