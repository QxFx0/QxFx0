   18 inferActiveScene :: AtomTrace -> [AtomTag] -> SemanticScene -> [SemanticScene] -> SemanticScene
   19 inferActiveScene trace currentTags currentScene scenes =
   20   let load = atCurrentLoad trace
   21       scored = map (scoreScene load currentTags currentScene) scenes
   22       candidates = filter (\(s, _) -> load >= sceLoadMin s && load <= sceLoadMax s) scored
   23   in case candidates of
   24        (_:_) -> fst $ maximumBy (comparing snd) candidates
   25        [] -> case scenes of
   26                (s:_) -> s
   27                []    -> defaultScene
   28 
   29 scoreScene :: Double -> [AtomTag] -> SemanticScene -> SemanticScene -> (SemanticScene, Double)
   30 scoreScene load tags currentScene scene =
   31   let rangeCenter = (sceLoadMin scene + sceLoadMax scene) / 2
   32       rangeWidth = max semanticSceneRangeWidthFloor (sceLoadMax scene - sceLoadMin scene)
   33       proximity = 1.0 - (abs (load - rangeCenter) / rangeWidth)
   34       tagMatchCount = length $ filter (`elem` sceTags scene) tags
   35       tagBonus = fromIntegral tagMatchCount * semanticSceneTagBonusPerMatch
   36       hysteresisBonus =
   37         if sceDescription scene == sceDescription currentScene
   38           then semanticSceneHysteresisBonus
   39           else 0.0
   40       totalScore = proximity + tagBonus + hysteresisBonus
   41   in (scene, totalScore)
