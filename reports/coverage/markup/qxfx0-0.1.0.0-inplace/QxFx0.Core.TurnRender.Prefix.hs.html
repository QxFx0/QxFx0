   22 renderPrincipledPrefix :: Maybe PressureSignal -> PrincipledMode -> Text
   23 renderPrincipledPrefix pressure mode =
   24   case pressure of
   25     Nothing -> ""
   26     Just signal ->
   27       let firstLine = T.takeWhile (/= '\n') (principledToPromptSection signal mode)
   28        in if T.null firstLine then "" else firstLine
   29 
   30 renderStylePrefix :: RenderStyle -> Text

   40 renderAnchorPrefix :: SemanticAnchor -> Text
   41 renderAnchorPrefix anchor =
   42   if saStability anchor >= anchorStabilityThreshold
   43     then "Якорь: " <> dominantChannelText (saDominantChannel anchor)
   44     else ""
