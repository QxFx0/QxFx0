   25 deriveSemanticAnchor :: Maybe SemanticAnchor -> SemanticInput -> Text -> Int -> Maybe SemanticAnchor
   26 deriveSemanticAnchor previous semanticInput topic turnNo
   27   | T.null (T.strip topic) && asLoad (siAtomSet semanticInput) < anchorNoTopicLowLoadThreshold = previous
   28   | otherwise =
   29       let channel = anchorChannel semanticInput
   30           previousChannel = fmap saDominantChannel previous
   31           sameChannel = previousChannel == Just channel
   32           establishedAt = if sameChannel then maybe turnNo saEstablishedAtTurn previous else turnNo
   33           strength = clamp01 (asLoad (siAtomSet semanticInput))
   34           stability =
   35             if sameChannel
   36               then clamp01 (maybe anchorCarryStabilityBaseline saStability previous + anchorCarryStabilityStep)
   37               else anchorResetStabilityDefault
   38        in Just SemanticAnchor
   39             { saDominantChannel = channel
   40             , saSecondaryChannel =
   41                 if T.null topic
   42                   then Nothing
   43                   else Just (T.take anchorTopicPreviewChars topic)
   44             , saEstablishedAtTurn = establishedAt
   45             , saStrength = strength
   46             , saStability = stability
   47             }

   49 snapshotIdentitySignal :: IdentitySignal -> IdentitySignalSnapshot
   50 snapshotIdentitySignal signal =
   51   IdentitySignalSnapshot
   52     { issOrbitalPhase = isOrbitalPhase signal
   53     , issEncounterMode = isEncounterMode signal
   54     , issContactStrength = isContactStrength signal
   55     , issBoundaryStrength = isBoundaryStrength signal
   56     , issAbstractionBudget = isAbstractionBudget signal
   57     , issMoveBias = isMoveBias signal
   58     , issRegister = isRegister signal
   59     , issNeedLayer = isNeedLayer signal
   60     }

   62 anchorChannel :: SemanticInput -> DominantChannel
   63 anchorChannel semanticInput =
   64   case (siNeedLayer semanticInput, siRegister semanticInput, siRecommendedFamily semanticInput) of
   65     (ContactLayer, _, _) -> ChannelContact
   66     (_, Exhaust, _) -> ChannelRepair
   67     (_, Anchor, _) -> ChannelAnchor
   68     (_, Contact, _) -> ChannelContact
   69     (_, _, CMClarify) -> ChannelClarify
   70     (_, _, CMDefine) -> ChannelDefine
   71     (_, _, CMReflect) -> ChannelReflect
   72     _ -> ChannelGround
