   18 applyLegitimacyPenalty :: Double -> ResponseMeaningPlan -> (Double, ResponseMeaningPlan)
   19 applyLegitimacyPenalty score meaningPlan
   20   | score >= legitimacyPassThreshold = (score, meaningPlan)
   21   | score >= legitimacyRecoveryThreshold =
   22       ( score
   23       , meaningPlan
   24           { rmpStance = degradeStance (rmpStance meaningPlan)
   25           , rmpDepthMode =
   26               if score < legitimacyCautionThreshold
   27                 then SurfaceDepth
   28                 else rmpDepthMode meaningPlan
   29           }
   30       )
   31   | otherwise =
   32       (score, meaningPlan {rmpFamily = CMRepair, rmpStance = Honest})
   33 
   34 legitimacyStyleOverride :: Double -> Maybe RenderStyle
   35 legitimacyStyleOverride score
   36   | score >= legitimacyPassThreshold = Nothing
   37   | score >= legitimacyCautionThreshold = Nothing
   38   | score >= legitimacyRecoveryThreshold = Just StyleCautious
   39   | otherwise = Just StyleRecovery
   40 
   41 styleFromLegitimacy :: Double -> Maybe RenderStyle
   42 styleFromLegitimacy = legitimacyStyleOverride
   43 
   44 degradeStance :: StanceMarker -> StanceMarker
   45 degradeStance Commit = Honest
   46 degradeStance Firm = Observe
   47 degradeStance Honest = Tentative
   48 degradeStance Explore = Tentative
   49 degradeStance stance = stance
