    4 {-| Resource readiness checks for runtime bootstrapping and strict/degraded mode decisions. -}
    5 module QxFx0.Resources.Readiness
    6   ( assessResourceReadiness
    7   , checkResourceReadiness
    8   , ReadinessStatus(..)
    9   , ReadinessComponent(..)
   10   , ReadinessMode(..)
   11   , computeReadinessMode
   12   ) where
   13 
   14 import QxFx0.ExceptionPolicy (tryQxFx0)
   15 import QxFx0.Resources.Morphology (validateMorphologyResources)
   16 import QxFx0.Resources.Paths
   17   ( ResourcePaths(..)
   18   , resolveResourcePaths
   19   )
   20 import System.Directory (doesDirectoryExist, doesFileExist)
   21 import System.FilePath (takeDirectory)
   22 
   23 data ReadinessComponent
   24   = RcResourceRoot
   25   | RcDatabase
   26   | RcMorphology
   27   | RcNixPolicy
   28   | RcAgdaSpec
   29   | RcDatalogRules
   30   | RcSchema
   31   deriving stock (Show, Eq, Enum, Bounded)
   32 
   33 data ReadinessStatus = ReadinessStatus
   34   { rsComponents :: [(ReadinessComponent, Bool, String)]
   35   , rsIsReady    :: Bool
   36   , rsIsDegraded :: Bool
   37   } deriving stock (Show, Eq)
   38 
   39 data ReadinessMode
   40   = Ready
   41   | Degraded [ReadinessComponent]
   42   | NotReady [ReadinessComponent]
   43   deriving stock (Show, Eq)
   44 
   45 assessResourceReadiness :: FilePath -> IO ReadinessStatus
   46 assessResourceReadiness dbFile = do
   47   dbExists <- doesFileExist dbFile
   48   dbDirOk <- doesDirectoryExist (takeDirectory dbFile)
   49   let dbOk = dbExists || dbDirOk
   50   pathsResult <- tryQxFx0 resolveResourcePaths
   51   case pathsResult of
   52     Left err -> pure (resourceRootFailureStatus dbOk dbFile (show err))
   53     Right paths -> checkResourceReadiness paths dbFile
   54 
   55 checkResourceReadiness :: ResourcePaths -> FilePath -> IO ReadinessStatus
   56 checkResourceReadiness paths dbFile = do
   57   dbExists <- doesFileExist dbFile
   58   dbDirOk <- doesDirectoryExist (takeDirectory dbFile)
   59   let dbOk = dbExists || dbDirOk
   60   checks <- mapM (checkComponent paths dbOk) [minBound .. maxBound]
   61   let critical = [RcResourceRoot, RcDatabase, RcMorphology, RcSchema]
   62       allOk = all (\(_, ok, _) -> ok) checks
   63       criticalOk = all (\(c, ok, _) -> c `notElem` critical || ok) checks
   64       degraded = criticalOk && not allOk
   65   pure ReadinessStatus
   66     { rsComponents = checks
   67     , rsIsReady = criticalOk
   68     , rsIsDegraded = degraded
   69     }
   70   where
   71     checkComponent :: ResourcePaths -> Bool -> ReadinessComponent -> IO (ReadinessComponent, Bool, String)
   72     checkComponent rp dbOk rc = case rc of
   73       RcResourceRoot ->
   74         pure (rc, True, "Resource root resolved: " ++ rpResourceDir rp)
   75       RcDatabase ->
   76         pure (rc, dbOk, if dbOk then "DB path accessible" else "DB directory not accessible: " ++ dbFile)
   77       RcMorphology -> do
   78         (ok, detail) <- validateMorphologyResources (rpMorphologyDir rp)
   79         pure (rc, ok, detail)
   80       RcNixPolicy -> do
   81         ok <- doesFileExist (rpNixGuard rp)
   82         pure (rc, ok, if ok then "Nix policy found" else "Nix policy missing (optional): " ++ rpNixGuard rp)
   83       RcAgdaSpec -> do
   84         ok <- doesFileExist (rpAgdaSpec rp)
   85         pure (rc, ok, if ok then "Agda spec found" else "Agda spec missing (optional): " ++ rpAgdaSpec rp)
   86       RcDatalogRules -> do
   87         ok <- doesFileExist (rpDatalogRules rp)
   88         pure (rc, ok, if ok then "Datalog rules found" else "Datalog rules missing (optional): " ++ rpDatalogRules rp)
   89       RcSchema -> do
   90         ok <- doesFileExist (rpSchemaSql rp)
   91         pure (rc, ok, if ok then "Schema SQL found" else "Schema SQL missing: " ++ rpSchemaSql rp)
   92 
   93 computeReadinessMode :: ReadinessStatus -> ReadinessMode
   94 computeReadinessMode status =
   95   let critical = [RcResourceRoot, RcDatabase, RcMorphology, RcSchema]
   96       failedCritical = [c | (c, ok, _) <- rsComponents status, not ok, c `elem` critical]
   97       failedOptional = [c | (c, ok, _) <- rsComponents status, not ok, c `notElem` critical]
   98   in if not (null failedCritical)
   99        then NotReady failedCritical
  100        else if null failedOptional then Ready else Degraded failedOptional
  101 
  102 resourceRootFailureStatus :: Bool -> FilePath -> String -> ReadinessStatus
  103 resourceRootFailureStatus dbOk dbFile msg =
  104   ReadinessStatus
  105     { rsComponents =
  106         [ (RcResourceRoot, False, "Resource root unavailable: " ++ msg)
  107         , (RcDatabase, dbOk, if dbOk then "DB path accessible" else "DB directory not accessible: " ++ dbFile)
  108         , (RcMorphology, False, "Morphology unavailable because resource root could not be resolved")
  109         , (RcNixPolicy, False, "Nix policy unavailable because resource root could not be resolved")
  110         , (RcAgdaSpec, False, "Agda spec unavailable because resource root could not be resolved")
  111         , (RcDatalogRules, False, "Datalog rules unavailable because resource root could not be resolved")
  112         , (RcSchema, False, "Schema SQL unavailable because resource root could not be resolved")
  113         ]
  114     , rsIsReady = False
  115     , rsIsDegraded = False
  116     }
