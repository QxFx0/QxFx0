   28 scoreClaims :: [IdentityClaimRef] -> Text -> [(IdentityClaimRef, Double)]
   29 scoreClaims claims focusEntity =
   30   map (\c -> (c, claimScore c focusEntity)) claims
   31 
   32 claimScore :: IdentityClaimRef -> Text -> Double
   33 claimScore claim focus =
   34   let focusTokens = tokenizeKeywordText focus
   35       conceptMatch = if containsKeywordPhrase focusTokens (icrConcept claim) then claimConceptMatchWeight else 0.0
   36       topicMatch = if containsKeywordPhrase focusTokens (icrTopic claim) then claimTopicMatchWeight else 0.0
   37       textMatch = if containsKeywordPhrase (tokenizeKeywordText (icrText claim)) focus then claimTextMatchWeight else 0.0
   38       confWeight = icrConfidence claim * claimConfidenceWeight
   39   in conceptMatch + topicMatch + textMatch + confWeight
   40 
   41 selectTopClaims :: Int -> [(IdentityClaimRef, Double)] -> [IdentityClaimRef]
   42 selectTopClaims n scored =
   43   let sorted = sortBy (\(_, a) (_, b) -> compare b a) scored
   44   in take n (map fst sorted)
   45 
   46 deduplicateClaims :: [IdentityClaimRef] -> [IdentityClaimRef]
   47 deduplicateClaims = nubBy (\a b -> icrConcept a == icrConcept b && icrTopic a == icrTopic b)
   48 
   49 agencyValid :: SubjectState -> Bool
   50 agencyValid ss = ssAgency ss >= lowAgencyThreshold
   51 
   52 tensionValid :: SubjectState -> Bool
   53 tensionValid ss = ssTension ss <= criticalTensionThreshold
