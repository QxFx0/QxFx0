   53 applyRenderStrategy :: CanonicalMoveFamily -> ResponseStrategy -> ResponseMeaningPlan -> ResponseMeaningPlan
   54 applyRenderStrategy family strategy meaningPlan =
   55   meaningPlan
   56     { rmpStrategy = strategyToAnswerStrategy family strategy
   57     , rmpStance = responseStanceToMarker (rsStance strategy)
   58     , rmpEpistemic = strategyEpistemicFromDepth (rsDepth strategy) (rmpEpistemic meaningPlan)
   59     , rmpDepthMode = strategyDepthMode (rsDepth strategy)
   60     }

   71 renderStyleFromDecision :: ResponseStrategy -> Maybe PrincipledMode -> NarrativeTone -> IdentitySignal -> Maybe SemanticAnchor -> SemanticInput -> RenderStyle
   72 renderStyleFromDecision strategy mode tone identitySignal semanticAnchor semanticInput =
   73   let byStrategy = styleFromStrategy strategy
   74       byPrincipled = styleFromPrincipled mode byStrategy
   75       byTone = styleFromTone tone byPrincipled
   76       byIdentity = styleFromIdentity identitySignal byTone
   77       bySemantic = styleFromSemantic semanticInput byIdentity
   78    in maybe bySemantic (`styleFromAnchor` bySemantic) semanticAnchor

   80 -- | Salience-aware variant of 'renderStyleFromDecision'. The base
   81 -- style is picked by the existing strategy/principled/identity/
   82 -- semantic/anchor chain, then post-processed under the salience
   83 -- verdict via 'applySalienceToStyle'.
   84 --
   85 -- Behaviour at neutral salience ('Tied' verdict, 'DrivenByDefault'
   86 -- driver) is identity, so any call site that does not yet supply
   87 -- rich Field signals observes no behaviour change.
   88 renderStyleFromDecisionWithSalience
   89   :: SalienceWeights
   90   -> Salience
   91   -> Field
   92   -> ResponseStrategy
   93   -> Maybe PrincipledMode
   94   -> NarrativeTone
   95   -> IdentitySignal
   96   -> Maybe SemanticAnchor
   97   -> SemanticInput
   98   -> RenderStyle
   99 renderStyleFromDecisionWithSalience salienceWeights salience field strategy mode tone identitySignal semanticAnchor semanticInput =
  100   applySalienceToStyle
  101     salienceWeights
  102     salience
  103     field
  104     (renderStyleFromDecision strategy mode tone identitySignal semanticAnchor semanticInput)

  106 -- | Re-shape a 'RenderStyle' under a 'Salience' verdict via the
  107 -- 'Holistic ⊣ Formal' adjunction.  This is the first runtime call
  108 -- site of 'chooseBranch', making the adjunction types participate
  109 -- in execution rather than remaining pure algebra.
  110 --
  111 -- Decision rule (matches ADR-0010 §4 and §5):
  112 --
  113 --   * 'DrivenByConatusGate' (structural risk) overrides every
  114 --     other verdict and forces 'StyleRecovery'. This is the strict
  115 --     reading of the Conatus-gate priority discipline.
  116 --   * 'PreferFormal' verdict leans neutral and warm picks toward
  117 --     formal/contract-respecting variants.
  118 --   * 'PreferHolistic' verdict leans neutral and direct picks
  119 --     toward expressive/holistic variants.
  120 --   * 'Tied' verdict is identity — the dead band preserves the
  121 --     style picked by the existing strategy chain.
  122 --
  123 -- Already-specialised picks (Clinical, Cautious, Recovery, and the
  124 -- styles already aligned with the verdict) are preserved on every
  125 -- branch so the controller does not /override/ a more informative
  126 -- pick from the strategy chain.
  127 applySalienceToStyle :: SalienceWeights -> Salience -> Field -> RenderStyle -> RenderStyle
  128 applySalienceToStyle salienceWeights salience field baseStyle =
  129   case salienceDriver salience of
  130     DrivenByConatusGate -> StyleRecovery
  131     _ ->
  132       chooseBranch
  133         (salienceVerdict salienceWeights salience)
  134         (\(Holistic (base, _field)) -> holisticLean base)
  135         (\base -> Formal (\_field -> formalLean base))
  136         (Holistic (baseStyle, field))
  137   where
  138     formalLean StyleStandard = StyleFormal
  139     formalLean StyleWarm     = StyleFormal
  140     formalLean StyleDirect   = StyleFormal
  141     formalLean other         = other
  142 
  143     holisticLean StyleStandard = StylePoetic
  144     holisticLean StyleFormal   = StylePoetic
  145     holisticLean StyleDirect   = StyleWarm
  146     holisticLean other         = other

  153 strategyToAnswerStrategy :: CanonicalMoveFamily -> ResponseStrategy -> AnswerStrategy
  154 strategyToAnswerStrategy _family strategy =
  155   case rsMove strategy of
  156     CounterMove -> DeepenThenProbe
  157     ReframeMove -> ClarifyThenDisambiguate
  158     QuestionMove -> DeepenThenProbe
  159     ValidateMove -> ContactThenBridge
  160     SilenceMove -> AnchorThenStabilize

  162 responseStanceToMarker :: ResponseStance -> StanceMarker
  163 responseStanceToMarker HoldStance = Firm
  164 responseStanceToMarker OpenStance = Explore
  165 responseStanceToMarker RedirectStance = Observe
  166 responseStanceToMarker AcknowledgeStance = Honest

  168 strategyEpistemicFromDepth :: ResponseDepth -> EpistemicStatus -> EpistemicStatus
  169 strategyEpistemicFromDepth depth current =
  170   let modulated =
  171         case depth of
  172           DeepResp ->
  173             case current of
  174               Known confidence ->
  175                 Known
  176                   (min strategyDeepKnownConfidenceCap (confidence + strategyDeepKnownConfidenceBoost))
  177               Probable confidence ->
  178                 Known
  179                   (min strategyDeepProbableToKnownCap (confidence + strategyDeepProbableToKnownBoost))
  180               Uncertain confidence ->
  181                 Probable
  182                   (min strategyDeepUncertainToProbableCap (confidence + strategyDeepUncertainToProbableBoost))
  183               Unknown confidence ->
  184                 Uncertain
  185                   (min strategyDeepUnknownToUncertainCap (confidence + strategyDeepUncertainToProbableBoost))
  186               Speculative confidence ->
  187                 Uncertain
  188                   (min strategyDeepSpeculativeToUncertainCap (confidence + strategyDeepUncertainToProbableBoost))
  189           ModerateResp -> current
  190           ShallowResp ->
  191             case current of
  192               Known confidence ->
  193                 Probable
  194                   (max strategyShallowKnownToProbableFloor (confidence - strategyShallowKnownPenalty))
  195               Probable confidence ->
  196                 Uncertain
  197                   (max strategyShallowProbableToUncertainFloor (confidence - strategyShallowProbablePenalty))
  198               other -> other
  199   in modulated

  201 strategyEpistemicFromDepthWithTruthContract :: TruthContractStatus -> ResponseDepth -> EpistemicStatus -> EpistemicStatus
  202 strategyEpistemicFromDepthWithTruthContract truthStatus depth current =
  203   capByTruthContract truthStatus (strategyEpistemicFromDepth depth current)

  208 styleFromStrategy :: ResponseStrategy -> RenderStyle
  209 styleFromStrategy strategy =
  210   case (rsStance strategy, rsMove strategy, rsDepth strategy) of
  211     (HoldStance, _, _) -> StyleFormal
  212     (_, CounterMove, _) -> StyleDirect
  213     (_, _, DeepResp) -> StylePoetic
  214     (AcknowledgeStance, _, _) -> StyleWarm
  215     _ -> StyleStandard
  216 
  217 styleFromPrincipled :: Maybe PrincipledMode -> RenderStyle -> RenderStyle
  218 styleFromPrincipled mode fallback =
  219   case mode of
  220     Just HoldGround -> StyleFormal
  221     Just AcknowledgeAndHold -> StyleWarm
  222     _ -> fallback
  223 
  224 styleFromTone :: NarrativeTone -> RenderStyle -> RenderStyle
  225 styleFromTone tone fallback =
  226   case tone of
  227     NarrativeRecovery -> StyleRecovery
  228     NarrativeWarm -> StyleWarm
  229     NarrativeFormal -> StyleFormal
  230     NarrativeTerse -> StyleCautious
  231     NarrativeNeutral -> fallback
  232 
  233 styleFromIdentity :: IdentitySignal -> RenderStyle -> RenderStyle
  234 styleFromIdentity identitySignal fallback =
  235   case isEncounterMode identitySignal of
  236     EncounterRecovery -> StyleWarm
  237     EncounterHolding -> StyleFormal
  238     EncounterPressure -> StyleDirect
  239     _ -> fallback
  240 
  241 styleFromSemantic :: SemanticInput -> RenderStyle -> RenderStyle
  242 styleFromSemantic semanticInput fallback =
  243   case siNeedLayer semanticInput of
  244     ContactLayer -> StyleWarm
  245     MetaLayer -> StyleFormal
  246     ContentLayer -> fallback

  248 styleFromAnchor :: SemanticAnchor -> RenderStyle -> RenderStyle
  249 styleFromAnchor anchor fallback
  250   | saDominantChannel anchor == ChannelContact = StyleWarm
  251   | saDominantChannel anchor == ChannelAnchor = StyleFormal
  252   | otherwise = fallback
