   37 applyRenderStrategy :: CanonicalMoveFamily -> ResponseStrategy -> ResponseMeaningPlan -> ResponseMeaningPlan
   38 applyRenderStrategy family strategy meaningPlan =
   39   meaningPlan
   40     { rmpStrategy = strategyToAnswerStrategy family strategy
   41     , rmpStance = responseStanceToMarker (rsStance strategy)
   42     , rmpEpistemic = strategyEpistemicFromDepth (rsDepth strategy) (rmpEpistemic meaningPlan)
   43     , rmpDepthMode = strategyDepthMode (rsDepth strategy)
   44     }

   46 renderStyleFromDecision :: ResponseStrategy -> Maybe PrincipledMode -> IdentitySignal -> Maybe SemanticAnchor -> SemanticInput -> RenderStyle
   47 renderStyleFromDecision strategy mode identitySignal semanticAnchor semanticInput =
   48   let byStrategy = styleFromStrategy strategy
   49       byPrincipled = styleFromPrincipled mode byStrategy
   50       byIdentity = styleFromIdentity identitySignal byPrincipled
   51       bySemantic = styleFromSemantic semanticInput byIdentity
   52    in maybe bySemantic (`styleFromAnchor` bySemantic) semanticAnchor

   54 strategyDepthMode :: ResponseDepth -> DepthMode
   55 strategyDepthMode DeepResp = DeepDepth
   56 strategyDepthMode ModerateResp = MediumDepth
   57 strategyDepthMode ShallowResp = SurfaceDepth
   58 
   59 strategyToAnswerStrategy :: CanonicalMoveFamily -> ResponseStrategy -> AnswerStrategy
   60 strategyToAnswerStrategy _family strategy =
   61   case rsMove strategy of
   62     CounterMove -> DeepenThenProbe
   63     ReframeMove -> ClarifyThenDisambiguate
   64     QuestionMove -> DeepenThenProbe
   65     ValidateMove -> ContactThenBridge
   66     SilenceMove -> AnchorThenStabilize
   67 
   68 responseStanceToMarker :: ResponseStance -> StanceMarker
   69 responseStanceToMarker HoldStance = Firm
   70 responseStanceToMarker OpenStance = Explore
   71 responseStanceToMarker RedirectStance = Observe
   72 responseStanceToMarker AcknowledgeStance = Honest
   73 
   74 strategyEpistemicFromDepth :: ResponseDepth -> EpistemicStatus -> EpistemicStatus
   75 strategyEpistemicFromDepth depth current =
   76   case depth of
   77     DeepResp ->
   78       case current of
   79         Known confidence ->
   80           Known
   81             (min strategyDeepKnownConfidenceCap (confidence + strategyDeepKnownConfidenceBoost))
   82         Probable confidence ->
   83           Known
   84             (min strategyDeepProbableToKnownCap (confidence + strategyDeepProbableToKnownBoost))
   85         Uncertain confidence ->
   86           Probable
   87             (min strategyDeepUncertainToProbableCap (confidence + strategyDeepUncertainToProbableBoost))
   88         Unknown confidence ->
   89           Uncertain
   90             (min strategyDeepUnknownToUncertainCap (confidence + strategyDeepUncertainToProbableBoost))
   91         Speculative confidence ->
   92           Uncertain
   93             (min strategyDeepSpeculativeToUncertainCap (confidence + strategyDeepUncertainToProbableBoost))
   94     ModerateResp -> current
   95     ShallowResp ->
   96       case current of
   97         Known confidence ->
   98           Probable
   99             (max strategyShallowKnownToProbableFloor (confidence - strategyShallowKnownPenalty))
  100         Probable confidence ->
  101           Uncertain
  102             (max strategyShallowProbableToUncertainFloor (confidence - strategyShallowProbablePenalty))
  103         other -> other

  105 strategyDepthLabel :: ResponseDepth -> Text
  106 strategyDepthLabel = depthModeText . strategyDepthMode

  108 styleFromStrategy :: ResponseStrategy -> RenderStyle
  109 styleFromStrategy strategy =
  110   case (rsStance strategy, rsMove strategy, rsDepth strategy) of
  111     (HoldStance, _, _) -> StyleFormal
  112     (_, CounterMove, _) -> StyleDirect
  113     (_, _, DeepResp) -> StylePoetic
  114     (AcknowledgeStance, _, _) -> StyleWarm
  115     _ -> StyleStandard
  116 
  117 styleFromPrincipled :: Maybe PrincipledMode -> RenderStyle -> RenderStyle
  118 styleFromPrincipled mode fallback =
  119   case mode of
  120     Just HoldGround -> StyleFormal
  121     Just AcknowledgeAndHold -> StyleWarm
  122     _ -> fallback
  123 
  124 styleFromIdentity :: IdentitySignal -> RenderStyle -> RenderStyle
  125 styleFromIdentity identitySignal fallback =
  126   case isEncounterMode identitySignal of
  127     EncounterRecovery -> StyleWarm
  128     EncounterHolding -> StyleFormal
  129     EncounterPressure -> StyleDirect
  130     _ -> fallback
  131 
  132 styleFromSemantic :: SemanticInput -> RenderStyle -> RenderStyle
  133 styleFromSemantic semanticInput fallback =
  134   case siNeedLayer semanticInput of
  135     ContactLayer -> StyleWarm
  136     MetaLayer -> StyleFormal
  137     ContentLayer -> fallback
  138 
  139 styleFromAnchor :: SemanticAnchor -> RenderStyle -> RenderStyle
  140 styleFromAnchor anchor fallback
  141   | saDominantChannel anchor == ChannelContact = StyleWarm
  142   | saDominantChannel anchor == ChannelAnchor = StyleFormal
  143   | otherwise = fallback
