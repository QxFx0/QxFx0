   35 data ParsedToken = ParsedToken
   36   { ptIdx     :: !Int
   37   , ptWord    :: !Text
   38   , ptLemma   :: !Text
   39   , ptPos     :: !Text          -- ^ NOUN, VERB, ADJ, ADP, ADV, PART, PRON...

   56 parseInput :: SemanticConfig -> Text -> IO ParsedInput
   57 parseInput cfg raw = pure (ruleBasedParse cfg raw)

   59 emptyParsedInput :: Text -> ParsedInput
   60 emptyParsedInput raw = ParsedInput [] 0 raw

   62 --------------------------------------------------------------------------------
   63 -- Rule-based fallback parser
   64 --------------------------------------------------------------------------------
   65 
   66 {-| Fast, deterministic rule-based parser for Russian text. Keeps downstream
   67    semantic slots (subject, object, negation, verb, agreement) functional in
   68    the supported local-only production parser contour. -}
   69 ruleBasedParse :: SemanticConfig -> Text -> ParsedInput
   70 ruleBasedParse cfg raw =
   71   let tokens = tokenizeRussian raw
   72       indexed = zip [0..] tokens
   73   in case indexed of
   74        [] -> emptyParsedInput raw
   75        _ ->
   76          let rootIdx = fromMaybe 0 (findIndex (isVerbLike cfg . T.toLower . snd) indexed)
   77              hasVerbRoot = any (isVerbLike cfg . T.toLower . snd) indexed
   78              negWords = scParseParticles cfg
   79              standaloneParticles = ["да", "нет", "ок", "угу", "ага"]
   80              mkToken (idx, word) =
   81                let lemma = T.toLower word
   82                    pos   = if not hasVerbRoot && (lemma `elem` negWords || lemma `elem` standaloneParticles)
   83                              then "PART"
   84                              else guessPos cfg lemma
   85                    (dep, headIdx)
   86                      | idx == rootIdx && hasVerbRoot = ("ROOT", idx)
   87                      | idx == rootIdx = ("dep", idx)
   88                      | lemma `elem` negWords = ("advmod", rootIdx)
   89                      | isPronoun cfg lemma = ("nsubj", rootIdx)
   90                      | pos == "NOUN" = ("obj", rootIdx)
   91                      | pos == "ADJ" = ("amod", rootIdx)
   92                      | otherwise = ("dep", rootIdx)
   93                in ParsedToken idx word lemma pos dep headIdx
   94          in ParsedInput (map mkToken indexed) rootIdx raw
   95 
   96 tokenizeRussian :: Text -> [Text]
   97 tokenizeRussian raw =
   98   let words = T.words raw
   99       stripEdge t = T.dropWhile isPunctEdge (T.dropWhileEnd isPunctEdge t)
  100       isPunctEdge c = isPunctuation c || c `elem` ("«»—…–“”" :: String)
  101   in filter (not . T.null) (map stripEdge words)
  102 
  103 -- | Heuristic: word ends with a typical Russian verb suffix.
  104 isVerbLike :: SemanticConfig -> Text -> Bool
  105 isVerbLike cfg w =
  106   let endings = scParseVerbEndings cfg
  107   in any (`T.isSuffixOf` w) endings
  108 
  109 isPronoun :: SemanticConfig -> Text -> Bool
  110 isPronoun cfg w = w `elem` scParsePronouns cfg
  111 
  112 isAdjective :: SemanticConfig -> Text -> Bool
  113 isAdjective cfg w =
  114   let endings = scParseAdjEndings cfg
  115   in any (`T.isSuffixOf` w) endings
  116 
  117 guessPos :: SemanticConfig -> Text -> Text
  118 guessPos cfg w
  119   | isPronoun cfg w          = "PRON"
  120   | w `elem` scParsePrepositions cfg = "ADP"
  121   | w `elem` scParseParticles cfg    = "PART"
  122   | w `elem` scParseAdverbs cfg      = "ADV"
  123   | w `elem` scParseConjunctions cfg = "CCONJ"
  124   | isAdjective cfg w        = "ADJ"
  125   | isVerbLike cfg w         = "VERB"
  126   | otherwise                = "NOUN"
