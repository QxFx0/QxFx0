   40 data PressureChannel
   41   = DesireConflict | PatternNotice | SelfModelGap | TrajectoryInsight | ProductModeResistance
   42   deriving stock (Show, Read, Eq, Ord, Enum, Bounded)

   44 data BackgroundPressure = BackgroundPressure
   45   { channel  :: PressureChannel
   46   , pressure :: Double
   47   , content  :: Text
   48   , turns    :: Int
   49   } deriving stock (Show, Read)

   51 data BackgroundState = BackgroundState
   52   { bsPressures    :: [BackgroundPressure]
   53   , bsLastSurfaced :: Int
   54   , bsTurnCount    :: Int

   57 data SurfacingEvent = SurfacingEvent
   58   { seChannel  :: PressureChannel

   64 surfacingThreshold :: Double
   65 surfacingThreshold = backgroundSurfacingThreshold

   67 minSurfacingGap :: Int
   68 minSurfacingGap = 3

   70 initialBackground :: BackgroundState
   71 initialBackground = BackgroundState
   72   { bsPressures    = map emptyChannel [minBound .. maxBound]
   73   , bsLastSurfaced = -100
   74   , bsTurnCount    = 0
   75   }
   76 
   77 emptyChannel :: PressureChannel -> BackgroundPressure
   78 emptyChannel ch = BackgroundPressure ch 0.0 "" 0
   79 
   80 tickBackground :: BackgroundState -> BackgroundState
   81 tickBackground bs = bs
   82   { bsPressures = map tick (bsPressures bs)
   83   , bsTurnCount = bsTurnCount bs + 1
   84   }
   85   where
   86     tick bp = bp { pressure = pressure bp * backgroundPressureDecayFactor, turns = turns bp + 1 }
   87 
   88 recordKernelAction :: BackgroundState -> Text -> [Text] -> [Text] -> BackgroundState
   89 recordKernelAction bs skillUsed _activeDesires conflicts =
   90   let bs1 = if not (null conflicts)
   91             then addPressure bs DesireConflict backgroundDesireConflictDelta (desireConflictPrefix <> T.intercalate ", " conflicts)
   92             else bs
   93       bs2 = if skillUsed `elem` [productModeSkillAnalyzeName, productModeSkillComposeName, productModeSkillWriteName]
   94             then addPressure bs1 ProductModeResistance backgroundProductModeResistanceDelta (productModeSkillLabel <> skillUsed)
   95             else bs1
   96   in bs2
   97 
   98 recordDesireConflict :: BackgroundState -> Text -> BackgroundState
   99 recordDesireConflict bs conflictDesc =
  100   addPressure bs DesireConflict backgroundExplicitConflictDelta conflictDesc
  101 
  102 addPressure :: BackgroundState -> PressureChannel -> Double -> Text -> BackgroundState
  103 addPressure bs ch delta newContent =
  104   let newPressures = map update (bsPressures bs)
  105   in bs { bsPressures = newPressures }
  106   where
  107     update bp
  108       | channel bp == ch = bp
  109           { pressure = min 1.0 (pressure bp + delta)
  110           , content  = if T.null (content bp) then newContent else content bp <> " / " <> newContent
  111           , turns    = 0
  112           }
  113       | otherwise = bp
  114 
  115 checkSurfacing :: BackgroundState -> Maybe SurfacingEvent
  116 checkSurfacing bs
  117   | bsTurnCount bs - bsLastSurfaced bs < minSurfacingGap = Nothing
  118   | otherwise =
  119       let candidates = filter (\bp -> pressure bp >= surfacingThreshold) (bsPressures bs)
  120       in case sortBy (comparing (Down . pressure)) candidates of
  121            []     -> Nothing
  122            (bp:_) -> Just (buildSurfacingEvent bp)
  123 
  124 buildSurfacingEvent :: BackgroundPressure -> SurfacingEvent
  125 buildSurfacingEvent bp =
  126   let (what, fragment) = channelSurfacing (channel bp) (content bp) (pressure bp)
  127   in SurfacingEvent (channel bp) what fragment (pressure bp)
  128 
  129 channelSurfacing :: PressureChannel -> Text -> Double -> (Text, Text)
  130 channelSurfacing DesireConflict pc p =
  131   (surfacingDesireConflictPrefix <> pc, conflictFragment pc p)
  132 channelSurfacing PatternNotice pc _ =
  133   (surfacingPatternNoticePrefix <> pc, patternNoticeFragmentPrefix <> pc <> patternNoticeFragmentSuffix)
  134 channelSurfacing SelfModelGap pc _ =
  135   (surfacingSelfModelGapPrefix <> pc, selfModelGapFragmentPrefix <> pc <> patternNoticeFragmentSuffix)
  136 channelSurfacing TrajectoryInsight pc _ =
  137   (surfacingTrajectoryInsightPrefix <> pc, trajectoryInsightFragmentPrefix <> pc <> patternNoticeFragmentSuffix)
  138 channelSurfacing ProductModeResistance pc _ =
  139   (surfacingProductModePrefix <> pc, surfacingProductModeFragment)
  140 
  141 conflictFragment :: Text -> Double -> Text
  142 conflictFragment pc p
  143   | p > backgroundConflictStrongThreshold = conflictStrongFragment <> T.take 60 pc <> conflictStrongSuffix
  144   | p > backgroundConflictMediumThreshold = conflictMediumFragment <> T.take 60 pc <> patternNoticeFragmentSuffix
  145   | otherwise = conflictLightFragment <> T.take 60 pc <> patternNoticeFragmentSuffix
  146 
  147 surfacingToFragment :: SurfacingEvent -> Text
  148 surfacingToFragment = seFragment
  149 
  150 runBackgroundCycle :: BackgroundState -> Text -> [Text] -> [Text] -> (BackgroundState, Maybe SurfacingEvent)
  151 runBackgroundCycle bs skill desires conflicts =
  152   let bs1 = tickBackground bs
  153       bs2 = recordKernelAction bs1 skill desires conflicts
  154       mEv = checkSurfacing bs2
  155       bs3 = maybe bs2 (applySurfacing bs2) mEv
  156   in (bs3, mEv)
  157 
  158 applySurfacing :: BackgroundState -> SurfacingEvent -> BackgroundState
  159 applySurfacing bs ev = bs
  160   { bsPressures = map reset (bsPressures bs)
  161   , bsLastSurfaced = bsTurnCount bs
  162   }
  163   where
  164     reset bp
  165       | channel bp == seChannel ev = bp { pressure = 0.0, content = "" }
  166       | otherwise = bp
