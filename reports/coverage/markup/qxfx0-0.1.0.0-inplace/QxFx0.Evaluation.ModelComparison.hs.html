  206 -- Runner
  207 
  208 -- | Single turn: query external tool, run learning loop, capture telemetry.
  209 runTurn
  210   :: SystemState
  211   -> LLMTransport
  212   -> ExternalTool
  213   -> LearningNeed
  214   -> ModelId
  215   -> TurnPrompt
  216   -> IO (SystemState, TurnResult)
  217 runTurn ss transport tool need modelId prompt = do
  218   t0 <- getCurrentTime
  219   result <- queryExternalTool transport tool need (tpText prompt)
  220   t1 <- getCurrentTime
  221   let latency = round (diffUTCTime t1 t0 * 1000.0) :: Int
  222       (ss', tel) = runLearningStep ss tool need (tpText prompt) (Just result)
  223       d    = ssDialogue ss
  224       ss'' = ss' { ssDialogue = d { dsTurnCount = dsTurnCount d + 1 } }
  225       tr = TurnResult
  226         { trModelId     = modelId
  227         , trPromptIndex = tpIndex prompt
  228         , trOutcome     = result
  229         , trLatencyMs   = latency
  230         , trTelemetry   = tel
  231         }
  232   return (ss'', tr)

  234 -- | Run one model through all prompts sequentially.
  235 -- State is forked from the supplied baseline.
  236 runModelSession
  237   :: Int           -- ^ session id
  238   -> ModelId
  239   -> LLMTransport
  240   -> [TurnPrompt]
  241   -> SystemState   -- ^ baseline state
  242   -> IO (SystemState, [TurnResult])
  243 runModelSession _sessionId modelId transport prompts ss0 =
  244   let tool = ExternalTool "llm-augment" DomainLexicon 0.7 True
  245       need = NeedLexiconExtension
  246       go ss [] acc = return (ss, reverse acc)
  247       go ss (p:ps) acc = do
  248         (ss', tr) <- runTurn ss transport tool need modelId p
  249         go ss' ps (tr : acc)
  250   in go ss0 prompts []

  252 -- | Deterministic Fisher-Yates shuffle keyed by an integer seed.
  253 deterministicShuffle :: Int -> [a] -> [a]
  254 deterministicShuffle seed xs =
  255   let gen = mkStdGen seed
  256       keys = take (length xs) (randoms gen :: [Int])
  257   in map snd (sortBy (comparing fst) (zip keys xs))
  258 
  259 -- | Run an interleaved session: every prompt is issued to every model,
  260 -- but turn order is deterministically shuffled per session seed.
  261 -- Each model carries its own state fork.
  262 runInterleavedSession
  263   :: Int
  264   -> [(ModelId, LLMTransport)]
  265   -> [TurnPrompt]
  266   -> SystemState
  267   -> IO [(ModelId, [TurnResult])]
  268 runInterleavedSession sessionId models prompts ss0 = do
  269   let modelIds = map fst models
  270       transports = M.fromList models
  271       states0 = M.fromList [(mid, ss0) | mid <- modelIds]
  272       pairs = concatMap (\p -> [(mid, p) | mid <- modelIds]) prompts
  273       shuffled = deterministicShuffle sessionId pairs
  274   (states, acc) <- foldM (runPair transports) (states0, M.empty) shuffled
  275   let results = map (\mid -> (mid, reverse (M.findWithDefault [] mid acc))) modelIds
  276       -- Restore final states so caller can inspect them if desired
  277       _finalStates = states
  278   return results
  279   where
  280     tool = ExternalTool "llm-augment" DomainLexicon 0.7 True
  281     need = NeedLexiconExtension
  282 
  283     runPair trans (states, acc) (mid, prompt) =
  284       case M.lookup mid trans of
  285         Nothing -> return (states, acc)
  286         Just transport ->
  287           case M.lookup mid states of
  288             Nothing -> return (states, acc)
  289             Just ss -> do
  290               (ss', tr) <- runTurn ss transport tool need mid prompt
  291               let states' = M.insert mid ss' states
  292                   acc'    = M.insertWith (++) mid [tr] acc
  293               return (states', acc')

  295 -- | Full comparison run across all models and sessions.
  296 -- Each session starts from the same baseline state.
  297 runComparison
  298   :: Text                       -- ^ run identifier
  299   -> SessionMode
  300   -> [(ModelId, LLMTransport)]  -- ^ models under test
  301   -> [TurnPrompt]
  302   -> SystemState               -- ^ baseline state
  303   -> IO ComparisonRun
  304 runComparison runId mode models prompts ss0 = do
  305   t0 <- getCurrentTime
  306   outcomes <- mapM (runModelOutcomes mode ss0) models
  307   let globalIncidents = concatMap moTotalIncidentsList outcomes
  308   return ComparisonRun
  309     { crRunId         = runId
  310     , crTimestamp     = t0
  311     , crModelOutcomes = outcomes
  312     , crGlobalIncidents = globalIncidents
  313     }
  314   where
  315     sessionCount = 3 :: Int
  316 
  317     runModelOutcomes Sequential baseline (mid, transport) = do
  318       sessions <- mapM (\sid -> do
  319         (ss', results) <- runModelSession sid mid transport prompts baseline
  320         let incidents = detectIncidents mid results
  321         return (aggregateSession mid sid results) { soIncidents = incidents }
  322         ) [1..sessionCount]
  323       return (aggregateModel mid sessions)
  324 
  325     runModelOutcomes Interleaved baseline (mid, transport) = do
  326       sessions <- mapM (\sid -> do
  327         resultsMap <- runInterleavedSession sid [(mid, transport)] prompts baseline
  328         let results = case lookup mid resultsMap of
  329                         Just rs -> rs
  330                         Nothing -> []
  331             incidents = detectIncidents mid results
  332         return (aggregateSession mid sid results) { soIncidents = incidents }
  333         ) [1..sessionCount]
  334       return (aggregateModel mid sessions)
  335 
  336     moTotalIncidentsList mo = concatMap soIncidents (moSessions mo)

  341 -- | Aggregate a single session's turn results.
  342 aggregateSession :: ModelId -> Int -> [TurnResult] -> SessionOutcome
  343 aggregateSession modelId sessionId results =
  344   SessionOutcome
  345     { soModelId               = modelId
  346     , soSessionId             = sessionId
  347     , soTotalTurns            = length results
  348     , soSuccessCount          = countWhere isAccept
  349     , soTransportErrorCount   = countWhere isTransportError
  350     , soParseRejectCount      = countWhere isParseReject
  351     , soValidationRejectCount = countWhere isValidationReject
  352     , soSandboxRejectCount    = countWhere isSandboxReject
  353     , soSandboxAcceptCount    = countWhere isAccept
  354     , soTotalLatencyMs        = sum (map trLatencyMs results)
  355     , soIncidents             = []
  356     }
  357   where
  358     status r = ltValidationStatus (trTelemetry r)
  359     isAccept r       = status r == "accept"
  360     isTransportError r = case trOutcome r of
  361                            Left _  -> True
  362                            Right _ -> False
  363     isParseReject r      = status r == "invalid_response"
  364     isValidationReject r = status r == "validation_reject"
  365     isSandboxReject r    = status r == "sandbox_reject"
  366     countWhere p = length (filter p results)
  367 
  368 -- | Aggregate across sessions for one model.
  369 aggregateModel :: ModelId -> [SessionOutcome] -> ModelOutcome
  370 aggregateModel modelId sessions =
  371   let totalTurns    = sum (map soTotalTurns sessions)
  372       totalSuccess  = sum (map soSuccessCount sessions)
  373       totalLatency  = sum (map soTotalLatencyMs sessions)
  374       totalTransportErrors = sum (map soTransportErrorCount sessions)
  375       totalParseRejects    = sum (map soParseRejectCount sessions)
  376       totalValRejects    = sum (map soValidationRejectCount sessions)
  377       totalValidatorChecks = max 0 (totalTurns - totalTransportErrors)
  378       totalSandboxChecks   = max 0 (totalValidatorChecks - totalParseRejects - totalValRejects)
  379       totalIncidents = sum (map (length . soIncidents) sessions)
  380   in ModelOutcome
  381     { moModelId              = modelId
  382     , moSessions             = sessions
  383     , moAvgSuccessRate       = rate totalSuccess totalTurns
  384     , moAvgLatencyMs         = rate totalLatency totalTurns
  385     , moAvgValidatorAcceptRate = rate totalSuccess totalValidatorChecks
  386     , moAvgSandboxPassRate     = rate totalSuccess totalSandboxChecks
  387     , moTotalIncidents       = totalIncidents
  388     }
  389   where
  390     rate _ 0 = 0.0
  391     rate n d = fromIntegral n / fromIntegral d

  393 -- ---------------------------------------------------------------------------
  394 -- Incident detection
  395 
  396 detectIncidents :: ModelId -> [TurnResult] -> [ComparisonIncident]
  397 detectIncidents modelId results =
  398   concat
  399     [ detectConsecutiveTransportErrors modelId results
  400     , detectConsecutiveValidatorRejects modelId results
  401     , detectConsecutiveSandboxRejects modelId results
  402     , detectRequestRejectLoops modelId results
  403     ]
  404 
  405 -- | 3+ consecutive transport errors.
  406 detectConsecutiveTransportErrors :: ModelId -> [TurnResult] -> [ComparisonIncident]
  407 detectConsecutiveTransportErrors modelId results =
  408   map (\(i, len) -> IncidentConsecutiveTransportErrors modelId i len) $
  409   findConsecutive 3 (\r -> case trOutcome r of Left _ -> True; _ -> False) results
  410 
  411 -- | 5+ consecutive validator-level rejects (transport, parse, validation).
  412 detectConsecutiveValidatorRejects :: ModelId -> [TurnResult] -> [ComparisonIncident]
  413 detectConsecutiveValidatorRejects modelId results =
  414   map (\(i, len) -> IncidentConsecutiveValidatorRejects modelId i len) $
  415   findConsecutive 5 (\r ->
  416     let s = ltValidationStatus (trTelemetry r)
  417     in s `elem` ["transport_error", "invalid_response", "validation_reject"]
  418   ) results
  419 
  420 -- | 3+ consecutive sandbox rejects with the *same* degradation tag.
  421 detectConsecutiveSandboxRejects :: ModelId -> [TurnResult] -> [ComparisonIncident]
  422 detectConsecutiveSandboxRejects modelId results =
  423   concatMap (\(i, len, tag) -> [IncidentConsecutiveSandboxRejects modelId i tag len]) $
  424   findConsecutiveWithSameReason 3 isSandboxReject getSandboxReason results
  425   where
  426     isSandboxReject r = ltValidationStatus (trTelemetry r) == "sandbox_reject"
  427     getSandboxRejectReason r = ltRejectReason (trTelemetry r)
  428     getSandboxReason r =
  429       if isSandboxReject r
  430         then getSandboxRejectReason r
  431         else Nothing
  432 
  433 -- | Repeated request->reject loop without new grafts.
  434 -- Simplified: 5+ consecutive turns where graft is Nothing and status /= accept.
  435 detectRequestRejectLoops :: ModelId -> [TurnResult] -> [ComparisonIncident]
  436 detectRequestRejectLoops modelId results =
  437   map (\(_i, len) -> IncidentRequestRejectLoop modelId len) $
  438   findConsecutive 5 (\r ->
  439     ltGraftTurn (trTelemetry r) == Nothing && ltValidationStatus (trTelemetry r) /= "accept"
  440   ) results
  441 
  442 -- | Find runs of at least @minLen@ elements satisfying predicate @p@.
  443 -- Returns 1-based (startIndex, runLength).
  444 findConsecutive :: Int -> (a -> Bool) -> [a] -> [(Int, Int)]
  445 findConsecutive minLen p = go 1
  446   where
  447     go _ [] = []
  448     go i xs@(y:_) =
  449       if p y
  450         then let (run, rest) = span p xs
  451                  len = length run
  452              in if len >= minLen
  453                   then (i, len) : go (i + len) rest
  454                   else go (i + len) rest
  455         else go (i + 1) (drop 1 xs)
  456     go _ _ = []
  457 
  458 -- | Like 'findConsecutive' but also requires the same 'Just' reason.
  459 findConsecutiveWithSameReason :: Int -> (a -> Bool) -> (a -> Maybe Text) -> [a] -> [(Int, Int, Text)]
  460 findConsecutiveWithSameReason minLen p getR = go 1
  461   where
  462     go _ [] = []
  463     go i xs@(y:_) =
  464       if p y
  465         then case getR y of
  466                Just r0 -> let (run, rest) = span (\z -> p z && getR z == Just r0) xs
  467                               len = length run
  468                           in if len >= minLen
  469                                then (i, len, r0) : go (i + len) rest
  470                                else go (i + len) rest
  471                Nothing -> go (i + 1) (drop 1 xs)
  472         else go (i + 1) (drop 1 xs)
  473     go _ _ = []

  478 -- | Render a comparison run as markdown-like telemetry text.
  479 renderComparisonRun :: ComparisonRun -> Text
  480 renderComparisonRun cr =
  481   T.concat
  482     [ "# Model Comparison Run: ", crRunId cr, "\n"
