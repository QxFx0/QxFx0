  105 data PrepareStatic = PrepareStatic
  106   { psInputText :: !Text
  107   , psAtomSet :: !AtomSet
  108   , psNewTrace :: !AtomTrace
  109   , psNextUserState :: !UserState
  110   , psRecommendedFamily :: !CanonicalMoveFamily
  111   , psFrame :: !InputPropositionFrame
  112   , psConceptToCheck :: !Text
  113   , psBestTopic :: !Text
  114   , psResonance :: !Double
  115   , psAtomLoad :: !Double
  116   , psConatusEnergy :: !ConatusEnergy
  117     -- ^ Phase 2.5 (M2d): the runtime Conatus energy computed
  118     --   from the current 'SelfBlanket' and its 'BlanketViolation's.
  119     --   Stored once per turn so downstream recovery-decision call
  120     --   sites (e.g. 'buildLocalRecoveryPlan' in
  121     --   'QxFx0.Core.TurnPipeline.Route.Render') can priority-check
  122     --   the Conatus gate without recomputing from 'SystemState'.
  123   , psBlanketViolationCount :: !Int
  124     -- ^ Phase 6 (M6): the count of 'BlanketViolation's reported
  125     --   by 'checkInitialBlanket' on the current 'SelfBlanket'.
  126     --   Stored alongside 'psConatusEnergy' so the recovery-plan
  127     --   evidence line @"blanket_violations=N"@ at the
  128     --   render-stage call site can be reconstructed without a
  129     --   second 'computeSelfBlanket' + 'checkInitialBlanket' pass.
  130   , psConatusGateFired :: !Bool
  131     -- ^ Phase 6 addendum (M6.1): single-source-of-truth flag for
  132     --   the Conatus gate.  Computed once in 'buildPrepareEffectPlan'
  133     --   by 'conatusGateFires', then read by both the salience
  134     --   controller and the recovery-decision site in
  135     --   'buildLocalRecoveryPlan'.  Eliminates the previous
  136     --   duplicate call to 'conatusGateFires'.
  137   , psField :: !Field
  138     -- ^ Per-turn 'QxFx0.Self.Field.Field' snapshot.  All five
  139     --   components are populated from runtime signals:
  140     --
  141     --     * 'fieldResonance'      — atom-trace current load
  142     --       ('psResonance').
  143     --     * 'fieldAtmosphere'     — valence = (ego agency −
  144     --       ego tension) modulated by legitimacy; arousal =
  145     --       ego tension.
  146     --     * 'fieldConsolidation'  — sliding-window narrative
  147     --       success rate, optionally floored by topic stability.
  148     --     * 'fieldCounterfactual' — normalised entropy of
  149     --       candidate family weights, boosted by holistic streak.
  150     --     * 'fieldConfidence'     — derived by
  151     --       'deriveFieldConfidence' from the four above.
  152     --
  153     --   Threaded through 'tiField' so all routing and salience-
  154     --   decision call sites share one canonical pre-turn Field.
  155   , psFieldHeuristics :: !FieldHeuristics
  156     -- ^ Phase 6.7: heuristics used to build 'psField'.
  157     --   Threaded through 'TurnInput' so downstream stages
  158     --   (e.g. salience computation) can read the same record.
  159   , psCurrentTime :: !UTCTime
  160     -- ^ Phase C: deterministic time injection point.
  161     --   Captured at prepare-stage entry so 'buildTurnInput' can
  162     --   set 'tiStartTime' without relying on the resolved timeline.
  163     --   Enables deterministic unit tests with a fixed time source.
  164   , psSenseVector :: !SenseVector
  165     -- ^ Canonical sense bridge extracted from the same utterance-level
  166     --   semantic interpretation used to derive 'psFrame'.
  167   , psDialogueThread :: !DialogueThread
  168   , psDialogueCommitmentLedger :: !DialogueCommitmentLedger
  169   , psDialoguePhase :: !DialoguePhase
  170   , psTruthContractStatus :: !TruthContractStatus
  171   , psEssence :: !Essence
  172     -- ^ Phase 9: pre-turn essence carrier from 'ssEssence'.
  173     --   Threaded through 'tiEssence' so witness ingestion in
  174     --   'buildNextSystemState' sees the canonical trajectory.
  175   } deriving stock (Eq, Show)

  195 buildPrepareEffectPlan :: SystemState -> Text -> UTCTime -> PrepareEffectPlan
  196 buildPrepareEffectPlan ss input currentTime =
  197   let atomSet = collectAtoms input (ssClusters ss)
  198       newTrace = updateTrace (ssTrace ss) (ssTurnCount ss) atomSet
  199       nextUserState = inferUserState (ssClusters ss) input
  200       logicResults = runSemanticLogic atomSet
  201       sortedLogic = L.sortBy (\(_, w1) (_, w2) -> compare w2 w1) logicResults
  202       recommendedFamily = case sortedLogic of
  203         ((fam, _):_) -> fam
  204         [] -> CMGround
  205       semanticFrame = buildUtteranceSemanticFrame input
  206       frame = parseProposition input
  207       senseVector = extractSenseVector semanticFrame
  208       dialogueLedger = deriveDialogueCommitmentLedger (ssDialogueCommitmentLedger ss) semanticFrame
  209       dialogueThread = deriveDialogueThread (ssDialogueThread ss) dialogueLedger (ssDialogue ss) semanticFrame
  210       dialoguePhase = deriveDialoguePhase dialogueThread dialogueLedger semanticFrame
  211       atomFocus = case asAtoms atomSet of
  212         (a:_) -> extractObjectFromAtom a
  213         [] -> ""
  214       conceptToCheck =
  215         firstNonEmpty
  216           [ ipfFocusNominative frame
  217           , ipfFocusEntity frame
  218           , atomFocus
  219           , fromMaybe fallbackWord (listToMaybe (T.words input))
  220           ]
  221       focus = firstNonEmpty [ipfFocusNominative frame, ipfFocusEntity frame, atomFocus, ssLastTopic ss]
  222       bestTopic = if T.null focus then ssLastTopic ss else focus
  223       resonance = atCurrentLoad newTrace
  224       atomLoad = asLoad atomSet
  225       semanticInput =
  226         buildSemanticInputSimple
  227           input
  228           atomSet
  229           frame
  230           recommendedFamily
  231           (ipfRegisterHint frame)
  232           (ipfSemanticLayer frame)
  233       blanket = computeSelfBlanket ss
  234       violations = checkInitialBlanket blanket
  235       conatusEnergy = computeConatusEnergy blanket violations
  236       violationCount = length violations
  237       conatusGateFired = conatusGateFires conatusEnergy
  238       -- Phase 7: populate four of five Field components via
  239       -- the calibrated 'FieldHeuristics' compute functions.
  240       -- 'fieldConfidence' is derived below.
  241       fieldHeuristics = ssFieldHeuristics ss
  242       preparedField0 = emptyField
  243         { fieldResonance      = mkResonance resonance
  244         , fieldAtmosphere     = computeAtmosphere fieldHeuristics
  245                                   (egoAgency (ssEgo ss))
  246                                   (egoTension (ssEgo ss))
  247                                   (obsLastLegitimacyScore (ssObservability ss))
  248         , fieldConsolidation  = computeConsolidation fieldHeuristics
  249                                   (ssRecentNarrativeSuccess ss)
  250                                   (not (T.null bestTopic) && bestTopic == ssLastTopic ss)
  251         , fieldCounterfactual = computeCounterfactual fieldHeuristics
  252                                   (map snd sortedLogic)
  253                                   (ssHolisticStreak ss)
  254         }
  255       preparedField = preparedField0
  256         { fieldConfidence = deriveFieldConfidence preparedField0
  257         }
  258       static = PrepareStatic
  259         { psInputText = input
  260         , psAtomSet = atomSet
  261         , psNewTrace = newTrace
  262         , psNextUserState = nextUserState
  263         , psRecommendedFamily = recommendedFamily
  264         , psFrame = frame
  265         , psConceptToCheck = conceptToCheck
  266         , psBestTopic = bestTopic
  267         , psResonance = resonance
  268         , psAtomLoad = atomLoad
  269         , psConatusEnergy = conatusEnergy
  270         , psBlanketViolationCount = violationCount
  271       , psConatusGateFired = conatusGateFired
  272       , psField = preparedField
  273       , psFieldHeuristics = fieldHeuristics
  274         -- ^ Phase 6.7: heuristics used to build 'psField'.
  275         --   Threaded through 'TurnInput' so downstream stages
  276         --   (e.g. salience computation) can read the same record.
  277       , psCurrentTime = currentTime
  278       , psSenseVector = senseVector
  279       , psDialogueThread = dialogueThread
  280       , psDialogueCommitmentLedger = dialogueLedger
  281       , psDialoguePhase = dialoguePhase
  282       , psTruthContractStatus = ssTruthContractStatus ss
  283       , psEssence = ssEssence ss
  284       }
  285   in PrepareEffectPlan
  286       { pepStatic = static
  287       , pepCapturedCurrentTime = currentTime
  288       , pepEmbeddingRequest = PrepareReqEmbedding input
  289       , pepNixGuardRequest = PrepareReqNixGuard conceptToCheck resonance atomLoad
  290       , pepConsciousnessRequest =
  291           PrepareReqConsciousness semanticInput (egoAgency (ssEgo ss)) resonance conatusEnergy (ssSalienceWeights ss)
  292       , pepIntuitionRequest =
  293           PrepareReqIntuition input resonance (egoTension (ssEgo ss)) (ssTurnCount ss + 1) conatusEnergy (ssSalienceWeights ss) (ssSemanticConfig ss)
  294       , pepApiHealthRequest = PrepareReqApiHealth
  295       }
  296   where
  297     firstNonEmpty = fromMaybe "" . listToMaybe . filter (not . T.null)
