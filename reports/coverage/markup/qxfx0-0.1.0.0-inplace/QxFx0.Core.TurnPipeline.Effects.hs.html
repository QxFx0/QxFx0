  104 buildPrepareEffectPlan :: SystemState -> Text -> PrepareEffectPlan
  105 buildPrepareEffectPlan ss input =
  106   let atomSet = collectAtoms input (ssClusters ss)
  107       newTrace = updateTrace (ssTrace ss) (ssTurnCount ss) atomSet
  108       nextUserState = inferUserState (ssClusters ss) input
  109       logicResults = runSemanticLogic atomSet
  110       recommendedFamily = case L.sortBy (\(_, w1) (_, w2) -> compare w2 w1) logicResults of
  111         ((fam, _):_) -> fam
  112         [] -> CMGround
  113       frame = parseProposition input
  114       atomFocus = case asAtoms atomSet of
  115         (a:_) -> extractObjectFromAtom a
  116         [] -> ""
  117       conceptToCheck =
  118         firstNonEmpty
  119           [ ipfFocusNominative frame
  120           , ipfFocusEntity frame
  121           , atomFocus
  122           , fromMaybe fallbackWord (listToMaybe (T.words input))
  123           ]
  124       focus = firstNonEmpty [ipfFocusNominative frame, ipfFocusEntity frame, atomFocus, ssLastTopic ss]
  125       bestTopic = if T.null focus then ssLastTopic ss else focus
  126       resonance = atCurrentLoad newTrace
  127       atomLoad = asLoad atomSet
  128       semanticInput =
  129         buildSemanticInputSimple
  130           input
  131           atomSet
  132           frame
  133           recommendedFamily
  134           (ipfRegisterHint frame)
  135           (ipfSemanticLayer frame)
  136       static = PrepareStatic
  137         { psInputText = input
  138         , psAtomSet = atomSet
  139         , psNewTrace = newTrace
  140         , psNextUserState = nextUserState
  141         , psRecommendedFamily = recommendedFamily
  142         , psFrame = frame
  143         , psConceptToCheck = conceptToCheck
  144         , psBestTopic = bestTopic
  145         , psResonance = resonance
  146         , psAtomLoad = atomLoad
  147         }
  148   in PrepareEffectPlan
  149       { pepStatic = static
  150       , pepEmbeddingRequest = PrepareReqEmbedding input
  151       , pepNixGuardRequest = PrepareReqNixGuard conceptToCheck resonance atomLoad
  152       , pepConsciousnessRequest = PrepareReqConsciousness semanticInput (egoAgency (ssEgo ss)) resonance
  153       , pepIntuitionRequest = PrepareReqIntuition resonance (egoTension (ssEgo ss)) (ssTurnCount ss + 1)
  154       , pepApiHealthRequest = PrepareReqApiHealth
  155       }
  156   where
  157     firstNonEmpty = fromMaybe "" . listToMaybe . filter (not . T.null)
