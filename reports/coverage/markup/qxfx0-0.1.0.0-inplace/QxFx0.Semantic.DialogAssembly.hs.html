   50 assembleTurn :: RuntimeParadigms -> DialogAtoms -> RenderStyle -> DiscourseState -> Either MorphError Text
   51 assembleTurn rp da style ds = assembleTurnWithGraph rp da style ds Nothing

   53 assembleTurnWithGraph :: RuntimeParadigms -> DialogAtoms -> RenderStyle -> DiscourseState -> Maybe GeodesicPlan -> Either MorphError Text
   54 assembleTurnWithGraph rp da style ds mGeodesicPlan =
   55   let recomputed = recomputeDiscourse ds
   56       usePoliteness = dscEngagement recomputed == LowEngagement && topicHash da `mod` 10 < 5
   57       useEmphatic = hasTag TEmphatic da && topicHash da `mod` 10 < 4
   58       plan = planAssemblyWithTransition da ds mGeodesicPlan
   59   in case plan of
   60     []       -> Right ""
   61     [step]   -> do
   62       s <- runStep rp da style step
   63       let withEmph = if useEmphatic
   64                      then addFiller (headAtomValue TEmphatic da) s
   65                      else s
   66           withModal = if hasTag TModal da
   67                       then addModalParticle (headAtomValue TModal da) withEmph
   68                       else withEmph
   69           withFiller = if hasTag TFiller da
   70                        then addFiller (headAtomValue TFiller da) withModal
   71                        else withModal
   72           withPolite = if usePoliteness
   73                        then addFiller "если можно" withFiller
   74                        else withFiller
   75       Right (linearizeS withPolite)
   76     (firstStep:restSteps) -> do
   77       let planLen = length restSteps + 1
   78           useSelfCorrect = planLen >= 3 && topicHash da `mod` 10 < 3
   79       firstSent <- runStep rp da style firstStep
   80       let withEmph = if useEmphatic
   81                      then addFiller (headAtomValue TEmphatic da) firstSent
   82                      else firstSent
   83           withModal = if hasTag TModal da
   84                       then addModalParticle (headAtomValue TModal da) withEmph
   85                       else withEmph
   86           withFiller = if hasTag TFiller da
   87                        then addFiller (headAtomValue TFiller da) withModal
   88                        else withModal
   89           withPolite = if usePoliteness
   90                        then addFiller "если можно" withFiller
   91                        else withFiller
   92       if useSelfCorrect && planLen >= 2
   93         then do
   94           let (initSteps, [lastStep]) = splitAt (planLen - 1) (firstStep:restSteps)
   95           initSents <- mapM (runStep rp da style) initSteps
   96           lastSent <- runStep rp da style lastStep
   97           let corrected = addFiller "то есть" lastSent
   98           combined <- foldM (\acc s -> mkCoordS CoordA acc s) withPolite (initSents ++ [corrected])
   99           Right (linearizeS combined)
  100         else do
  101           restSents <- mapM (runStep rp da style) restSteps
  102           combined <- foldM (\acc s -> mkCoordS CoordA acc s) withPolite restSents
  103           Right (linearizeS combined)
  104 data StepId = SAcknowledge | SStance | SEngage | SReflect
  105   | SIdentity    | SConstraint | SRepair
  106   | SContact     | SGround | SClarify | SDeepen
  107   | SNext | SGreet | SFarewell
  108   | SHedge | SConnect
  109   | SDefine | SPurpose | SDistinguish
  110   | SAgree | SDisagree | SRelated
  111   | SModal | SQuantify | SSelfCorrect | SEmphatic | SNarrative | SMetaphor
  112   deriving (Eq, Ord, Show, Bounded, Enum)

  114 type AssemblyPlan = [StepId]

  116 planAssembly :: DialogAtoms -> DiscourseState -> AssemblyPlan
  117 planAssembly da ds =
  118   let recomputed = recomputeDiscourse ds
  119       intent = headAtomValue TUserIntent da
  120       topic = headAtomValue TTopic da
  121       isGreet = hasTag TGreeting da
  122       isFarewell = hasTag TFarewell da
  123       isDistress = emotionIs "distress" da
  124       topicRepeated = topic `elem` dscTopicChain ds
  125       turnDepth = length (dscTurnMemory ds)
  126       primary = case () of
  127         _ | isGreet                    -> SGreet
  128         _ | isFarewell                 -> SFarewell
  129         _ | intent == "define"         -> SDefine
  130         _ | intent == "clarify"        -> SClarify
  131         _ | intent == "deepen"         -> SDeepen
  132         _ | intent == "ask_purpose"    -> SPurpose
  133         _ | intent == "hypothesize"    -> SDeepen
  134         _ | intent == "next"           -> SNext
  135         _ | intent == "complain"       -> SRepair
  136         _ | intent == "distinguish"    -> SDistinguish
  137         _ | intent == "reflect"        -> SReflect
  138         _ | intent == "describe"       -> if hasTag TScale da then SNarrative else SAcknowledge
  139         _ | intent == "ground"         -> SGround
  140         _ | intent == "connect"        -> SEngage
  141         _ | intent == "anchor"         -> SGround
  142         _ | isDistress                 -> SReflect
  143         _ | otherwise                  -> SEngage
  144       hasTopic = hasTag TTopic da
  145       hasRelated = hasTag TRelatedTopic da
  146       planLen = case dscEngagement recomputed of
  147         HighEngagement   -> 4
  148         MediumEngagement -> 3
  149         LowEngagement    -> 2
  150       -- Suppress grounding in deep phase
  151       needGround = hasTopic && not topicRepeated && dscPhase recomputed /= PhaseDeep
  152       secondary = case primary of
  153         SGround -> Nothing
  154         _ | needGround                   -> Just SGround
  155         _ | hasTopic && turnDepth < 3    -> Just SGround
  156         _ -> Nothing
  157       tertiary = if topicRepeated && hasRelated then Just SRelated else Nothing
  158       hasProps = hasTag TProperty da
  159       qStep = if hasProps then Just SQuantify else Nothing
  160       allSteps = primary : maybeToList qStep ++ maybeToList secondary ++ maybeToList tertiary
  161   in take planLen allSteps

  163 planAssemblyWithTransition :: DialogAtoms -> DiscourseState -> Maybe GeodesicPlan -> AssemblyPlan
  164 planAssemblyWithTransition da ds mGeodesicPlan =
  165   let basePlan = planAssembly da ds
  166       topic = headAtomValue TTopic da
  167       prevTopic = case dscTopicChain ds of
  168                     (prev:_) | prev /= topic && not (T.null prev) && not (T.null topic) -> Just prev
  169                     _ -> Nothing
  170       transitionSteps = case (prevTopic, mGeodesicPlan) of
  171         (Just _, Just (BridgedJump _)) -> [SContact]
  172         _ -> []
  173   in if null transitionSteps then basePlan else transitionSteps ++ basePlan

  175 emotionIs :: Text -> DialogAtoms -> Bool
  176 emotionIs tag da = any ((tag ==) . asValue) (atomValue TEmotion da)

  182 runStep :: RuntimeParadigms -> DialogAtoms -> RenderStyle -> StepId -> Either MorphError S
  183 runStep rp da style = \case
  184   SAcknowledge -> ackStep rp da style
  185   SStance      -> stStep rp da style
  186   SEngage      -> engStep rp da style
  187   SReflect     -> refStep rp da style
  188   SIdentity    -> idStep rp da style
  189   SConstraint  -> conStep rp da style
  190   SRepair      -> repStep rp da style
  191   SContact     -> con2Step rp da style
  192   SGround      -> gndStep rp da style
  193   SClarify     -> clarStep rp da style
  194   SDeepen      -> dStep rp da style
  195   SNext        -> nStep rp da style
  196   SGreet       -> greStep rp da style
  197   SFarewell    -> farStep rp da style
  198   SHedge       -> hedgeStep rp da style
  199   SConnect     -> connectStep rp da style
  200   SDefine      -> defStep rp da style
  201   SPurpose     -> purpStep rp da style
  202   SDistinguish -> distStep rp da style
  203   SAgree       -> agreeStep rp da style
  204   SDisagree    -> disagreeStep rp da style
  205   SRelated     -> relatedStep rp da style
  206   SModal       -> modalStep rp da style
  207   SQuantify    -> quantifyStep rp da style
  208   SSelfCorrect  -> selfCorrectStep rp da style
  209   SEmphatic     -> emphaticStep rp da style
  210   SNarrative    -> narrativeStep rp da style
  211   SMetaphor     -> metaphorStep rp da style

  217 -- | ACKNOWLEDGE (3 variants)
  218 ackStep :: RuntimeParadigms -> DialogAtoms -> RenderStyle -> Either MorphError S
  219 ackStep rp da _style = do
  220   let intent = headAtomValue TUserIntent da
  221       fillers = atomValues TFiller da
  222       useFiller = not (null fillers) && (T.length (daUserRaw da) `mod` 10 < 3)
  223       v = topicHash da `mod` 3
  224   base <- case intent of
  225     "define" -> case v of
  226       0 -> do
  227         subj <- pure (mkPronounNP "я" Nom)
  228         vp <- mkVP_ rp "слышать" 1 Sg Nothing Present
  229         topic <- topicNP rp da Acc
  230         let pp = mkPP "о" topic
  231         vp2 <- addPP vp pp
  232         s <- mkS subj vp2
  233         pure (mkQuestion s IFAsk)
  234       1 -> do
  235         subj <- pure (mkPronounNP "я" Nom)
  236         vp <- mkVP_ rp "слышать" 1 Sg Nothing Present
  237         topic <- topicNP rp da Gen
  238         vp2 <- addPP vp (mkPP "про" topic)
  239         s <- mkS subj vp2
  240         pure (mkQuestion s IFAsk)
  241       _ -> do
  242         tf <- mkNP_ rp "вопрос" Nom Sg Masc
  243         topic <- topicNP rp da Gen
  244         vp <- mkVP_ rp "касаться" 3 Sg Nothing Present
  245         vp2 <- addObj vp topic
  246         s1 <- mkS tf vp2
  247         subj <- pure (mkPronounNP "я" Nom)
  248         vpI <- mkVP_ rp "слышать" 1 Sg Nothing Present
  249         s2 <- mkS subj vpI
  250         mkCoordS CoordDash s1 s2
  251     _ -> case v of
  252       0 -> do
  253         subj <- pure (mkPronounNP "я" Nom)
  254         vp <- mkVP_ rp "слышать" 1 Sg Nothing Present
  255         mkS subj vp
  256       1 -> do
  257         subj <- pure (mkPronounNP "я" Nom)
  258         vp <- mkVP_ rp "слышать" 1 Sg Nothing Present
  259         adv <- mkAdv rp "хорошо"
  260         pure (addAdvToS (mkElliptical vp) adv)
  261       _ -> do
  262         subj <- pure (mkPronounNP "я" Nom)
  263         vp <- mkVP_ rp "слышать" 1 Sg Nothing Present
  264         mkS subj vp
  265   pure (if useFiller then addFiller (fromMaybe "" (listToMaybe fillers)) base else base)

  393 -- | GROUND (2 variants)
  394 gndStep :: RuntimeParadigms -> DialogAtoms -> RenderStyle -> Either MorphError S
  395 gndStep rp da _style = do
  396   topic <- topicNP rp da Nom
  397   let v = topicHash da `mod` 2
  398       quants = atomValues TQuantifier da
  399       useQuant = T.length (daTopicNominative da) `mod` 10 < 4
  400   base <- case v of
  401     0 -> do
  402       linkNP <- mkNP_ rp "связь" Nom Sg Femn
  403       vp <- mkVP_ rp "быть" 3 Sg Nothing Present
  404       ap <- mkAP_ rp "опорный" (npGender topic) Sg Nom
  405       topic2 <- addAP topic ap
  406       s <- mkS topic2 vp
  407       mkTopicFront linkNP s
  408     _ -> do
  409       linkNP <- mkNP_ rp "связь" Nom Sg Femn
  410       ap <- mkAP_ rp "опорный" Femn Sg Nom
  411       link2 <- addAP linkNP ap
  412       ap2 <- mkAP_ rp "опорный" (npGender topic) Sg Nom
  413       topic2 <- addAP topic ap2
  414       vp <- mkVP_ rp "быть" 3 Sg Nothing Present
  415       s1 <- mkS link2 vp
  416       s2 <- mkS topic2 vp
  417       mkCoordS CoordDash s1 s2
  418   pure (if useQuant && not (null quants) then addQuantifierToS (fromMaybe "" (listToMaybe quants)) base else base)

  559   case lookupMetaphor rp topic Nothing of
  560     Just (_, source) -> do
  561       let subj = mkNPRaw topic Nom Sg (guessGenderOr rp Masc topic)
  562       let advUnsafe = mkAdvRaw "вдуматься"
  563       vp <- mkVP_ rp "быть" 3 Sg Nothing Present
  564       let pp = mkPP "как" (mkNPRaw source Nom Sg Neut)
  565       vp' <- addPP vp pp
  566       s <- mkS subj vp'
  567       pure (addAdvToS s advUnsafe)
  568     Nothing -> do
  569       subj <- pure (mkNPRaw topic Nom Sg Masc)
  570       vp <- mkVP_ rp "быть" 3 Sg Nothing Present
  571       mkS subj vp

  686 topicNP :: RuntimeParadigms -> DialogAtoms -> NounCase -> Either MorphError NP
  687 topicNP rp da case_ =
  688   let raw = headAtomValue TTopic da
  689       nom = if T.null raw then "тема" else raw
  690   in case mkNP rp nom case_ Sg of
  691        Right np -> Right np
  692        Left _ ->
  693          -- Graceful degradation: placeholder NP preserves surface text
  694          let g = guessGenderOr rp Masc nom
  695          in Right (mkNPRaw nom case_ Sg g)

  697 adjAsPredicateS :: RuntimeParadigms -> NP -> Text -> Gender -> Either MorphError S
  698 adjAsPredicateS rp subj lemma gender = do
  699   ap <- mkAP_ rp lemma gender (npNumber subj) (npCase subj)
  700   subj2 <- addAP subj ap
  701   vp <- mkVP_ rp "быть" (case npNumber subj of Sg -> 3; Pl -> 3) (npNumber subj) Nothing Present
  702   mkS subj2 vp

  704 -- | Build APs from TProperty atoms (up to 3)
  705 propertyAPs :: RuntimeParadigms -> DialogAtoms -> NP -> Either MorphError [AP]
  706 propertyAPs rp da _subj =
  707   let props = atomValues TProperty da
  708       tryMakeAP p = mkAP_ rp p Masc Sg Nom
  709   in Right (mapMaybe (eitherToMaybe . tryMakeAP) (take 3 props))
  710   where
  711     eitherToMaybe (Right x) = Just x
  712     eitherToMaybe (Left _)   = Nothing
