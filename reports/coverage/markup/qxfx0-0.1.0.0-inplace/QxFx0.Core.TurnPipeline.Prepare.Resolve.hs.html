   44 resolvePrepareEffects :: PipelineIO -> PrepareEffectPlan -> IO PrepareEffectResults
   45 resolvePrepareEffects pio effectPlan = do
   46   let t0 = pepCapturedCurrentTime effectPlan
   47   _ <- evaluate (pepStatic effectPlan)
   48   let tStatic1 = t0
   49       requestPairs :: [(Text, Maybe TurnEffectRequest)]
   50       requestPairs =
   51         [ ("embedding", prepareRequestToTurnEffect (pepEmbeddingRequest effectPlan))
   52         , ("nix", prepareRequestToTurnEffect (pepNixGuardRequest effectPlan))
   53         , ("consciousness", prepareRequestToTurnEffect (pepConsciousnessRequest effectPlan))
   54         , ("intuition", prepareRequestToTurnEffect (pepIntuitionRequest effectPlan))
   55         , ("api_health", prepareRequestToTurnEffect (pepApiHealthRequest effectPlan))
   56         ]
   57       scheduledPairs :: [(Text, TurnEffectRequest)]
   58       scheduledPairs =
   59         scheduleTurnEffects pio (psConatusEnergy (pepStatic effectPlan))
   60           [ (label, request)
   61           | (label, Just request) <- requestPairs
   62           ]
   63   resolved <- forConcurrently scheduledPairs $ \(label, turnRequest) -> do
   64     result <- resolveTurnEffect pio turnRequest
   65     pure (label, result)
   66   let embeddingResult =
   67         fromMaybe fallbackEmbeddingResult $ do
   68           (_, TurnResEmbedding value) <- firstMatch (\(label, result) -> label == "embedding" && isEmbeddingResult result) resolved
   69           Just value
   70       nixStatus =
   71         fromMaybe (Blocked "nix_guard_missing_request") $ do
   72           (_, TurnResNixGuard value) <- firstMatch (\(label, result) -> label == "nix" && isNixResult result) resolved
   73           Just value
   74       consciousnessResult =
   75         fromMaybe (initialLoop, Nothing, Nothing) $ do
   76           (_, TurnResConsciousness loop narrative fragment) <- firstMatch (\(label, result) -> label == "consciousness" && isConsciousnessResult result) resolved
   77           Just (loop, narrative, fragment)
   78       intuitionResult =
   79         fromMaybe (Nothing, defaultPosterior, defaultIntuitiveState) $ do
   80           (_, TurnResIntuition flash posterior state) <- firstMatch (\(label, result) -> label == "intuition" && isIntuitionResult result) resolved
   81           Just (flash, posterior, state)
   82       apiHealthy =
   83         fromMaybe False $ do
   84           (_, TurnResApiHealth value) <- firstMatch (\(label, result) -> label == "api_health" && isApiHealthResult result) resolved
   85           Just value
   86       timedEmb = TimedResult t0 t0 embeddingResult
   87       timedNix = TimedResult t0 t0 nixStatus
   88       timedConsciousness = TimedResult t0 t0 consciousnessResult
   89       timedIntuition = TimedResult t0 t0 intuitionResult
   90       timedApiHealth = TimedResult t0 t0 apiHealthy
   91       (consciousLoop', currentNarrative, narrativeFragment') = consciousnessResult
   92       (mFlash, intuitPosterior, intuitionState) = intuitionResult
   93   pure PrepareEffectResults
   94     { perTimeline = PrepareTimeline
   95         { ptlStartTime = t0
   96         , ptlPrepareStaticDone = tStatic1
   97         , ptlEmbeddingStart = trStarted timedEmb
   98         , ptlEmbeddingDone = trEnded timedEmb
   99         , ptlNixStart = trStarted timedNix
  100         , ptlNixDone = trEnded timedNix
  101         , ptlConsciousnessStart = trStarted timedConsciousness
  102         , ptlConsciousnessDone = trEnded timedConsciousness
  103         , ptlIntuitionStart = trStarted timedIntuition
  104         , ptlIntuitionDone = trEnded timedIntuition
  105         , ptlApiHealthStart = trStarted timedApiHealth
  106         , ptlApiHealthDone = trEnded timedApiHealth
  107         }
  108     , perTurnCurrentTime = t0
  109     , perEmbeddingResult = trValue timedEmb
  110     , perNixStatus = trValue timedNix
  111     , perConsciousLoop = consciousLoop'
  112     , perCurrentNarrative = currentNarrative
  113     , perNarrativeFragment = narrativeFragment'
  114     , perFlash = mFlash
  115     , perIntuitPosterior = intuitPosterior
  116     , perIntuitionState = intuitionState
  117     , perApiHealthy = trValue timedApiHealth
  118     }

  120 prepareRequestToTurnEffect :: PrepareEffectRequest -> Maybe TurnEffectRequest
  121 prepareRequestToTurnEffect request =
  122   case request of
  123     PrepareReqEmbedding inputText ->
  124       Just (TurnReqEmbedding inputText)
  125     PrepareReqNixGuard concept currentLoad atomLoad ->
  126       Just (TurnReqNixGuard concept currentLoad atomLoad)
  127     PrepareReqConsciousness semanticInput humanTheta resonance conatusEnergy salienceWeights ->
  128       Just (TurnReqConsciousness semanticInput humanTheta resonance conatusEnergy salienceWeights)
  129     PrepareReqIntuition inputText resonance tension turnNumber conatusEnergy salienceWeights semanticConfig ->
  130       Just (TurnReqIntuition inputText resonance tension turnNumber conatusEnergy salienceWeights semanticConfig)
  131     PrepareReqApiHealth ->
  132       Just TurnReqApiHealth
  133 
  134 defaultPosterior :: Double
  135 defaultPosterior = effectivePosterior defaultIntuitiveState

  137 fallbackEmbeddingResult :: EmbeddingResult
  138 fallbackEmbeddingResult = EmbeddingResult { erEmbedding = mempty, erSource = EmbeddingLocalImplicit }

  140 isEmbeddingResult :: TurnEffectResult -> Bool
  141 isEmbeddingResult result = case result of
  142   TurnResEmbedding _ -> True
  143   _ -> False
  144 
  145 isNixResult :: TurnEffectResult -> Bool
  146 isNixResult result = case result of
  147   TurnResNixGuard _ -> True
  148   _ -> False
  149 
  150 isConsciousnessResult :: TurnEffectResult -> Bool
  151 isConsciousnessResult result = case result of
  152   TurnResConsciousness _ _ _ -> True
  153   _ -> False
  154 
  155 isIntuitionResult :: TurnEffectResult -> Bool
  156 isIntuitionResult result = case result of
  157   TurnResIntuition _ _ _ -> True
  158   _ -> False
  159 
  160 isApiHealthResult :: TurnEffectResult -> Bool
  161 isApiHealthResult result = case result of
  162   TurnResApiHealth _ -> True
  163   _ -> False

  165 firstMatch :: (a -> Bool) -> [a] -> Maybe a
  166 firstMatch predicate = go
  167   where
  168     go [] = Nothing
  169     go (x:xs)
  170       | predicate x = Just x
  171       | otherwise = go xs
