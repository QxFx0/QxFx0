   48 resolvePrepareEffects :: PipelineIO -> PrepareEffectPlan -> IO PrepareEffectResults
   49 resolvePrepareEffects pio effectPlan = do
   50   t0 <- resolveEffectCurrentTime pio
   51   _ <- evaluate (pepStatic effectPlan)
   52   tStatic1 <- resolveEffectCurrentTime pio
   53   (timedEmb, timedNix, timedConsciousness, timedIntuition, timedApiHealth) <-
   54     runConcurrently $
   55       (,,,,)
   56         <$> Concurrently (timeAction pio (resolveEmbedding pio effectPlan))
   57         <*> Concurrently (timeAction pio (resolveNixGuard pio effectPlan))
   58         <*> Concurrently (timeAction pio (resolveConsciousness pio effectPlan))
   59         <*> Concurrently (timeAction pio (resolveIntuition pio effectPlan))
   60         <*> Concurrently (timeAction pio (resolveApiHealth pio effectPlan))
   61   let (consciousLoop', currentNarrative, narrativeFragment') = trValue timedConsciousness
   62       (mFlash, intuitPosterior, intuitionState) = trValue timedIntuition
   63   pure PrepareEffectResults
   64     { perTimeline = PrepareTimeline
   65         { ptlStartTime = t0
   66         , ptlPrepareStaticDone = tStatic1
   67         , ptlEmbeddingStart = trStarted timedEmb
   68         , ptlEmbeddingDone = trEnded timedEmb
   69         , ptlNixStart = trStarted timedNix
   70         , ptlNixDone = trEnded timedNix
   71         , ptlConsciousnessStart = trStarted timedConsciousness
   72         , ptlConsciousnessDone = trEnded timedConsciousness
   73         , ptlIntuitionStart = trStarted timedIntuition
   74         , ptlIntuitionDone = trEnded timedIntuition
   75         , ptlApiHealthStart = trStarted timedApiHealth
   76         , ptlApiHealthDone = trEnded timedApiHealth
   77         }
   78     , perEmbeddingResult = trValue timedEmb
   79     , perNixStatus = trValue timedNix
   80     , perConsciousLoop = consciousLoop'
   81     , perCurrentNarrative = currentNarrative
   82     , perNarrativeFragment = narrativeFragment'
   83     , perFlash = mFlash
   84     , perIntuitPosterior = intuitPosterior
   85     , perIntuitionState = intuitionState
   86     , perApiHealthy = trValue timedApiHealth
   87     }

   89 resolveEmbedding :: PipelineIO -> PrepareEffectPlan -> IO EmbeddingResult
   90 resolveEmbedding pio plan =
   91   case prepareRequestToTurnEffect (pepEmbeddingRequest plan) of
   92     Just request -> do
   93       result <- resolveTurnEffect pio request
   94       case result of
   95         TurnResEmbedding embeddingResult -> pure embeddingResult
   96         _ -> emptyEmbeddingResult
   97     Nothing ->
   98       emptyEmbeddingResult
   99 
  100 resolveNixGuard :: PipelineIO -> PrepareEffectPlan -> IO NixGuardStatus
  101 resolveNixGuard pio plan =
  102   case prepareRequestToTurnEffect (pepNixGuardRequest plan) of
  103     Just request -> do
  104       result <- resolveTurnEffect pio request
  105       case result of
  106         TurnResNixGuard nixStatus -> pure nixStatus
  107         _ -> pure (Unavailable "unexpected_prepare_nix_result")
  108     Nothing ->
  109       pure (Unavailable "unexpected_prepare_nix_request")
  110 
  111 resolveConsciousness :: PipelineIO -> PrepareEffectPlan -> IO (ConsciousnessLoop, Maybe ConsciousnessNarrative, Maybe Text)
  112 resolveConsciousness pio plan =
  113   case prepareRequestToTurnEffect (pepConsciousnessRequest plan) of
  114     Just request -> do
  115       result <- resolveTurnEffect pio request
  116       case result of
  117         TurnResConsciousness consciousLoop currentNarrative narrativeFragment ->
  118           pure (consciousLoop, currentNarrative, narrativeFragment)
  119         _ ->
  120           pure (initialLoop, Nothing, Nothing)
  121     Nothing ->
  122       pure (initialLoop, Nothing, Nothing)
  123 
  124 resolveIntuition :: PipelineIO -> PrepareEffectPlan -> IO (Maybe IntuitiveFlash, Double, IntuitiveState)
  125 resolveIntuition pio plan =
  126   case prepareRequestToTurnEffect (pepIntuitionRequest plan) of
  127     Just request -> do
  128       result <- resolveTurnEffect pio request
  129       case result of
  130         TurnResIntuition mFlash posterior intuitionState -> pure (mFlash, posterior, intuitionState)
  131         _ -> pure (Nothing, defaultPosterior, defaultIntuitiveState)
  132     Nothing ->
  133       pure (Nothing, defaultPosterior, defaultIntuitiveState)
  134 
  135 resolveApiHealth :: PipelineIO -> PrepareEffectPlan -> IO Bool
  136 resolveApiHealth pio plan =
  137   case prepareRequestToTurnEffect (pepApiHealthRequest plan) of
  138     Just request -> do
  139       result <- resolveTurnEffect pio request
  140       case result of
  141         TurnResApiHealth apiHealthy -> pure apiHealthy
  142         _ -> pure False
  143     Nothing ->
  144       pure False
  145 
  146 prepareRequestToTurnEffect :: PrepareEffectRequest -> Maybe TurnEffectRequest
  147 prepareRequestToTurnEffect request =
  148   case request of
  149     PrepareReqEmbedding inputText ->
  150       Just (TurnReqEmbedding inputText)
  151     PrepareReqNixGuard concept currentLoad atomLoad ->
  152       Just (TurnReqNixGuard concept currentLoad atomLoad)
  153     PrepareReqConsciousness semanticInput humanTheta resonance ->
  154       Just (TurnReqConsciousness semanticInput humanTheta resonance)
  155     PrepareReqIntuition resonance tension turnNumber ->
  156       Just (TurnReqIntuition resonance tension turnNumber)
  157     PrepareReqApiHealth ->
  158       Just TurnReqApiHealth
  159 
  160 emptyEmbeddingResult :: IO EmbeddingResult
  161 emptyEmbeddingResult =
  162   pure $
  163     EmbeddingResult
  164       { erEmbedding = mempty
  165       , erSource = EmbeddingLocalImplicit
  166       }
  167 
  168 defaultPosterior :: Double
  169 defaultPosterior = effectivePosterior defaultIntuitiveState
  170 
  171 timeAction :: PipelineIO -> IO a -> IO (TimedResult a)
  172 timeAction pio action = do
  173   started <- resolveEffectCurrentTime pio
  174   value <- action
  175   ended <- resolveEffectCurrentTime pio
  176   pure TimedResult
  177     { trStarted = started
  178     , trEnded = ended
  179     , trValue = value
  180     }
  181 
  182 resolveEffectCurrentTime :: PipelineIO -> IO UTCTime
  183 resolveEffectCurrentTime pio = do
  184   result <- resolveTurnEffect pio TurnReqCurrentTime
  185   case result of
  186     TurnResCurrentTime currentTime -> pure currentTime
  187     _ -> throwQxFx0 (PersistenceError "prepare timeline current time effect returned unexpected result")
