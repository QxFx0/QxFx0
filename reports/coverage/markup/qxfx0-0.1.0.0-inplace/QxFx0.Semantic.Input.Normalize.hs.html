    1 {-# LANGUAGE DerivingStrategies #-}

    5 module QxFx0.Semantic.Input.Normalize
    6   ( NormalizedInput(..)
    7   , normalizeInput
    8   ) where
    9 
   10 import Data.Char (isAlphaNum)
   11 import Data.Text (Text)
   12 import qualified Data.Text as T
   13 
   14 data NormalizedInput = NormalizedInput
   15   { niRawText :: !Text
   16   , niNormalizedText :: !Text
   17   , niTokens :: ![Text]
   18   , niIsQuestion :: !Bool
   19   } deriving stock (Eq, Show)
   20 
   21 normalizeInput :: Text -> NormalizedInput
   22 normalizeInput rawText =
   23   let stripped = T.strip rawText
   24       lower = T.toLower stripped
   25       normalized = normalizeOrthography lower
   26       tokens =
   27         filter (not . T.null)
   28           (map trimToken (T.words (T.map normalizeChar normalized)))
   29       isQuestion = T.isSuffixOf "?" stripped || isQuestionByCue tokens
   30   in NormalizedInput
   31        { niRawText = rawText
   32        , niNormalizedText = normalized
   33        , niTokens = tokens
   34        , niIsQuestion = isQuestion
   35        }
   36 
   37 isQuestionByCue :: [Text] -> Bool
   38 isQuestionByCue [] = False
   39 isQuestionByCue (firstToken:rest) =
   40   firstToken `elem`
   41     [ "что", "кто", "где", "когда", "почему", "зачем", "как"
   42     , "какой", "какая", "какое", "каков", "какова", "каковы"
   43     , "чем", "сколько", "ли", "чей", "чья", "чье", "чьё"
   44     ]
   45   || [firstToken, secondToken] `elem` [["для", "чего"], ["в", "чем"], ["в", "чём"]]
   46   where
   47     secondToken =
   48       case rest of
   49         [] -> ""
   50         (x:_) -> x
   51 
   52 normalizeOrthography :: Text -> Text
   53 normalizeOrthography = T.replace "ё" "е"
   54 
   55 normalizeChar :: Char -> Char
   56 normalizeChar c
   57   | isAlphaNum c = c
   58   | c == '-' = c
   59   | otherwise = ' '
   60 
   61 trimToken :: Text -> Text
   62 trimToken = T.dropAround (\c -> not (isAlphaNum c) && c /= '-')
