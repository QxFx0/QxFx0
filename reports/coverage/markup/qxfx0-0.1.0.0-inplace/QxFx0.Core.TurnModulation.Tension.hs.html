   26 computeTensionDelta :: Text -> SystemState -> Double
   27 computeTensionDelta rawText systemState =
   28   let tokens = tokenizeKeywordText rawText
   29       hasNegative = containsAnyKeywordPhrase tokens tensionNegativeKeywords
   30       hasDistress = containsAnyKeywordPhrase tokens tensionDistressKeywords
   31       baseDelta =
   32         if hasNegative
   33           then tensionNegativeDelta
   34           else if hasDistress then tensionDistressDelta else 0.0
   35    in baseDelta + egoTension (ssEgo systemState) * tensionEgoCarryFactor
