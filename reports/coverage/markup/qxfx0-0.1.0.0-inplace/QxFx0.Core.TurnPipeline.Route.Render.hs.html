  159 -- | WP6.1 anti-overblocking: do not dedup-block terms that are too
  160 -- short, contain digits, or look like noise / mis-parsed fragments.
  161 isTopicNoisyOrAmbiguous :: Text -> Bool
  162 isTopicNoisyOrAmbiguous t =
  163   T.length t < 3
  164     || T.any isDigit t
  165     || T.any isPunctuation t
  166   where
  167     isDigit c = c >= '0' && c <= '9'
  168     isPunctuation c = c `elem` (".,;:!?-–—…\"'()[]{}" :: String)

  170 planRenderEffectsForRuntimeImpl :: RuntimeParadigms -> PipelineRuntimeMode -> LocalRecoveryPolicy -> SystemState -> TurnInput -> TurnSignals -> TurnPlan -> RenderEffectPlan
  171 planRenderEffectsForRuntimeImpl rp runtimeMode localRecoveryPolicy ss ti ts tp =
  172   let bestTopic = tiBestTopic ti
  173       mFlash = tsFlash ts
  174       consciousLoop' = tsConsciousLoop' ts
  175       mNarrative = clLastNarrative consciousLoop'
  176       rmpAfterLegit = tpRmpAfterLegit tp
  177       rcpFinal = tpRcpFinal tp
  178       semanticAnchor = tpSemanticAnchor tp
  179       (mPressure, principledModeResult) =
  180         case tpPrincipledModePair tp of
  181           Just (p, pmr) -> (Just p, Just pmr)
  182           Nothing -> (Nothing, Nothing)
  183       identityClaims = integrateIdentityClaims (ssIdentityClaims ss) (tpFamily tp) bestTopic
  184       input = ipfRawText (tiFrame ti)
  185       structuredSurface = hasStructuredDialogueSurface (tiFrame ti)
  186       mGeodesicPlan =
  187         let topicChain = dscTopicChain (ssDiscourse ss)
  188         in case topicChain of
  189           (prev:_) | prev /= bestTopic && not (T.null prev) && not (T.null bestTopic) ->
  190             Just (geodesicRouter (ssMeaningGraph ss) prev bestTopic (take 3 topicChain))
  191           _ -> Nothing
  192       -- WP2: GF-first rendering. Assembly path is primary for all dialogue branches.
  193       -- Template fallback is only used when assembly produces empty text (no PGF/runtime).
  194       viaAssembly = renderArtifactViaAssembly rp ss (tiFrame ti) rmpAfterLegit rcpFinal
  195                         bestTopic identityClaims (ssMorphology ss) (rcpStyle rcpFinal) (emptyParsedInput input) mNarrative mGeodesicPlan
  196       assemblyFallbackReason = fromMaybe "assembly_empty_fallback" (draFallbackReason viaAssembly)
  197       dialogueArtifact
  198         | not (T.null (draRenderedText viaAssembly)) = viaAssembly
  199         | otherwise =
  200             (renderDialogueArtifact (tiFrame ti) rmpAfterLegit rcpFinal bestTopic identityClaims (ssMorphology ss))
  201               { draFallbackReason = Just assemblyFallbackReason }
  202       forceFinalized =
  203         if structuredSurface
  204           then draRenderedText dialogueArtifact
  205           else finalizeForce (rmpForce rmpAfterLegit) (draRenderedText dialogueArtifact)
  206       finalRender =
  207         case mFlash of
  208           Just flash ->
  209             if ifOverridesAll flash
  210               then forceFinalized <> "\n[" <> ifDirective flash <> "]"
  211               else forceFinalized
  212           Nothing -> forceFinalized
  213       renderWithContext = finalRender
  214       narrativeFragment = maybe "" id (tsNarrativeFragment ts)
  215       narrativeEnriched =
  216         if structuredSurface || T.null narrativeFragment
  217           then renderWithContext
  218           else renderWithContext <> "\n" <> T.take 40 narrativeFragment
  219       surfacingFragment =
  220         case structuredSurface of
  221           True -> ""
  222           False ->
  223             case clLastSurfacing consciousLoop' of
  224               Just se -> surfacingToFragment se
  225               Nothing -> ""
  226       renderWithBg =
  227         if T.null surfacingFragment
  228           then narrativeEnriched
  229           else narrativeEnriched <> "\n" <> surfacingFragment
  230       structuredQuestion =
  231         ipfConfidence (tiFrame ti) >= parserLowConfidenceThreshold
  232           && ipfPropositionType (tiFrame ti) /= "PlainAssert"
  233       morphologyWarning =
  234         if T.any isSpace input
  235              && not structuredQuestion
  236              && not (T.null bestTopic)
  237              && not (hasKnownMorphologyForm (ssMorphology ss) bestTopic)
  238           then Just bestTopic
  239           else Nothing
  240       localRecoveryPlan =
  241         buildLocalRecoveryPlan runtimeMode localRecoveryPolicy ss ti tp morphologyWarning
  242       -- WP3 dedup: skip external query if term is already known.
  243       -- WP6.1 anti-overblocking: do NOT block noisy/ambiguous/low-confidence topics.
  244       topicIsNoisy = isTopicNoisyOrAmbiguous bestTopic
  245       parserConfidenceLow =
  246         ipfConfidence (tiFrame ti) < parserLowConfidenceThreshold
  247       mDedupSkipReason =
  248         if topicIsNoisy || parserConfidenceLow
  249           then Nothing
  250           else if hasKnownMorphologyForm (ssMorphology ss) bestTopic
  251             then Just "already_known_morphology"
  252             else if isTermKnownInKnowledgeTree bestTopic (ssKnowledgeTree ss)
  253               then Just "already_known_tree"
  254               else Nothing
  255       mExternalQueryRequest =
  256         case mDedupSkipReason of
  257           Just _ -> Nothing
  258           Nothing ->
  259             case localRecoveryPlan of
  260               Just plan | isRequestStrategy (lrpStrategy plan) ->
  261                 let need = lnsCurrentNeed (ssLearningNeedState ss)
  262                     mTool = selectToolWithReliability need (ssToolReliability ss) defaultAvailableTools
  263                     queryText = buildExternalQueryText need (tiBestTopic ti)
  264                 in case mTool of
  265                      Just tool -> Just (tool, need, queryText)
  266                      Nothing   -> Nothing
  267               _ -> Nothing
  268       -- Phase 9: autonomous exploratory learning query.
  269       -- Triggered when:
  270       --   1. No request-driven external query already planned
  271       --   2. Learning need is active (not NeedNone)
  272       --   3. Guardrails allow (rate limit, circuit breaker)
  273       --   4. A suitable tool is available
  274       mExploratoryQueryRequest =
  275         case mDedupSkipReason of
  276           Just _ -> Nothing
  277           Nothing ->
  278             case mExternalQueryRequest of
  279               Just _ -> Nothing  -- request-driven query takes priority
  280               Nothing ->
  281                 let need = lnsCurrentNeed (ssLearningNeedState ss)
  282                     lns = ssLearningNeedState ss
  283                     toolRel = ssToolReliability ss
  284                     guard = ssGuardrailState ss
  285                     currentTurn = ssTurnCount ss
  286                 in if need == NeedNone
  287                       then Nothing
  288                       else if not (canSubmitProposal guard currentTurn)
  289                         then Nothing
  290                         else
  291                           let mTool = selectToolWithReliability need toolRel defaultAvailableTools
  292                               queryText = buildExploratoryQueryText need (tiBestTopic ti)
  293                           in case mTool of
  294                                Just tool | etReliability tool >= 0.3 -> Just (tool, need, queryText)
  295                                _ -> Nothing
  296     in RenderEffectPlan
  297         { repRenderStatic = RenderStatic
  298            { rsRenderWithBg = renderWithBg
  299            , rsTemplateArtifact = dialogueArtifact
  300            , rsPreferredGfLang = detectInputGfLang input
  301            , rsResolvedAssemblyPath = Nothing
  302            , rsResolvedAuthorityClass = Nothing
  303            , rsArtifactManifest = Nothing
  304            , rsModePrefixText =
  305                if structuredSurface
  306                  then ""
  307                 else
  308                   T.intercalate
  309                     "\n"
  310                     (filter
  311                       (not . T.null)
  312                       [ renderStylePrefix (rcpStyle rcpFinal)
  313                       , maybe "" (renderPrincipledPrefix mPressure) principledModeResult
  314                       ])
  315           , rsAnchorPrefixText =
  316               if not (not structuredSurface && tpFinalFamily tp == CMAnchor)
  317                 then ""
  318                 else maybe "" renderAnchorPrefix semanticAnchor
  319           , rsNarrativeFragmentText =
  320               if structuredSurface
  321                 then ""
  322                 else narrativeFragment
  323           , rsSurfacingFragmentText = surfacingFragment
  324           }
  325       , repTurnInput = ti
  326       , repLocalRecoveryPlan = localRecoveryPlan
  327       , repRenderMorphologyWarning = morphologyWarning
  328       , repKnowledgeTopic = bestTopic
  329       , repExternalQueryRequest = mExternalQueryRequest
  330       , repExploratoryQueryRequest = mExploratoryQueryRequest
  331       , repExternalQuerySkipReason = mDedupSkipReason
  332        }
  333   where
  334     isRequestStrategy StrategyRequestCalibration = True
  335     isRequestStrategy StrategyRequestRule        = True
  336     isRequestStrategy StrategyRequestConcept     = True
  337     isRequestStrategy _                          = False
  338 
  339     buildExternalQueryText :: LearningNeed -> Text -> Text
  340     buildExternalQueryText need topic =
  341       let topicText = if T.null topic then "concept" else topic
  342       in case need of
  343            -- Direct topic query for deterministic mock-table matching
  344            -- (first word is the topic itself).
  345            NeedLexiconExtension    -> topicText
  346            NeedKeywordEnrichment   -> T.concat ["Explain ", topicText]
  347            NeedSalienceCalibration -> T.concat ["Calibrate salience for ", topicText]
  348            NeedNone                -> ""
  349 
  350     buildExploratoryQueryText :: LearningNeed -> Text -> Text
  351     buildExploratoryQueryText need topic =
  352       let topicText = if T.null topic then "concept" else topic
  353       in case need of
  354            NeedLexiconExtension    -> T.concat ["Explore definition of ", topicText]
  355            NeedKeywordEnrichment   -> T.concat ["Explore keywords for ", topicText]
  356            NeedSalienceCalibration -> T.concat ["Explore salience calibration for ", topicText]
  357            NeedNone                -> ""

  359 resolveRenderEffects :: PipelineIO -> RenderEffectPlan -> IO RenderEffectResults
  360 resolveRenderEffects pio effectPlan = do
  361   let tRender0 = tiStartTime (repTurnInput effectPlan)
  362   warnMorphologyFallback <- shouldWarnMorphologyFallback pio
  363   case repRenderMorphologyWarning effectPlan of
  364     Just bestTopic | warnMorphologyFallback ->
  365       hPutStrLnWarning ("Morphology fallback: unknown topic lexeme: " <> bestTopic)
  366     _ ->
  367       pure ()
  368 
  369   resolvedRenderStatic <- resolveRuntimeGfLinearization pio (repRenderStatic effectPlan)
  370   -- WP3: Minimal legal DB adapter. Try legal lookup on the knowledge topic.
  371   -- For non-legal topics retrieveLegalFact returns Nothing; behavior is unchanged.
  372   mLegalFact <- retrieveLegalFact (repKnowledgeTopic effectPlan)
  373   let mKnowledgeFact = legalFactToKnowledgeFragment <$> mLegalFact
  374       mKnowledgeSource = lfSourceId <$> mLegalFact
  375   let scheduledRequests =
  376         scheduleTurnEffects pio (tiConatusEnergy (repTurnInput effectPlan))
  377           (  [ ("request", TurnReqExternalQuery tool need queryText)
  378              | Just (tool, need, queryText) <- [repExternalQueryRequest effectPlan]
  379              ]
  380           <> [ ("explore", TurnReqExternalQuery tool need queryText)
  381              | Just (tool, need, queryText) <- [repExploratoryQueryRequest effectPlan]
  382              ]
  383           )
  384   resolvedEffects <- forConcurrently scheduledRequests $ \(label, request) -> do
  385     result <- resolveTurnEffect pio request
  386     pure (label, result)
  387   let mExternalQueryResult =
  388         case firstMatch (\(label, _) -> label == "request") resolvedEffects of
  389           Nothing -> Nothing
  390           Just (_, TurnResExternalQuery res) -> Just res
  391           Just _ -> Just (Left (EqeInvalidResponse "unexpected_effect_result"))
  392       mExploratoryQueryResult =
  393         case firstMatch (\(label, _) -> label == "explore") resolvedEffects of
  394           Nothing -> Nothing
  395           Just (_, TurnResExternalQuery res) -> Just res
  396           Just _ -> Just (Left (EqeInvalidResponse "unexpected_effect_result"))
  397   let tRender1 = tiStartTime (repTurnInput effectPlan)
  398   pure RenderEffectResults
  399     { rerRenderTimeline = RenderTimeline
  400         { rtlRenderStart = tRender0
  401         , rtlRenderEnd = tRender1
  402         }
  403     , rerResolvedRenderStatic = resolvedRenderStatic
  404     , rerKnowledgeFact = mKnowledgeFact
  405     , rerKnowledgeFactSource = mKnowledgeSource
  406     , rerExternalQueryResult = mExternalQueryResult
  407     , rerExploratoryQueryResult = mExploratoryQueryResult
  408     , rerExternalQuerySkipReason = repExternalQuerySkipReason effectPlan
  409     }

  411 buildTurnArtifacts :: SystemState -> TurnInput -> TurnSignals -> TurnPlan -> RenderEffectPlan -> RenderEffectResults -> TurnArtifacts
  412 buildTurnArtifacts ss ti _ts tp effectPlan effectResults =
  413   let renderStatic = fromMaybe (repRenderStatic effectPlan) (rerResolvedRenderStatic effectResults)
  414       renderWithBg = rsRenderWithBg renderStatic
  415       localRecoveryPlan = repLocalRecoveryPlan effectPlan
  416       localRecoveryText = lrpSurface <$> localRecoveryPlan
  417       knowledgeFragment = maybe "" ("\n[знание] " <>) (rerKnowledgeFact effectResults)
  418       preSafetyRendered =
  419         case localRecoveryText of
  420           Just fb -> renderWithBg <> "\n" <> fb <> knowledgeFragment
  421           Nothing -> renderWithBg <> knowledgeFragment
  422       preSafetySurface =
  423         Guard.GuardSurface
  424           { Guard.gsRenderedText = preSafetyRendered
  425           , Guard.gsSegments =
  426               filter (not . T.null . Guard.rsText)
  427                 [ Guard.RenderSegment Guard.SegmentTemplate (draStylePrefixText (rsTemplateArtifact renderStatic))
  428                 , Guard.RenderSegment Guard.SegmentTemplate (rsModePrefixText renderStatic)
  429                 , Guard.RenderSegment Guard.SegmentTemplate (rsAnchorPrefixText renderStatic)
  430                 , Guard.RenderSegment Guard.SegmentTemplate (draTemplateBodyText (rsTemplateArtifact renderStatic))
  431                 , Guard.RenderSegment Guard.SegmentIdentityClaim (draClaimText (rsTemplateArtifact renderStatic))
  432                 , Guard.RenderSegment Guard.SegmentNarrative (rsNarrativeFragmentText renderStatic)
  433                 , Guard.RenderSegment Guard.SegmentSurfacing (rsSurfacingFragmentText renderStatic)
  434                 ]
  435                   ++ maybe [] (\fb -> [Guard.RenderSegment Guard.SegmentLocalRecovery fb]) localRecoveryText
  436           , Guard.gsQuestionLike = draQuestionLike (rsTemplateArtifact renderStatic)
  437           }
  438       timeline = rerRenderTimeline effectResults
  439       !metrics4 = addPhase (recordPhase "render" (rtlRenderStart timeline) (rtlRenderEnd timeline)) (tpMetrics tp)
  440       guardSafety = Guard.postRenderSafetyCheckSurface preSafetySurface (F.toList (ssHistory ss))
  441       templateArtifact = rsTemplateArtifact renderStatic
  442       (renderedSurface, finalizeSurfaceProv) = finalizeOutput preSafetySurface (F.toList (ssHistory ss))
  443       surfaceProv = case finalizeSurfaceProv of
  444         FromRecovery -> FromRecovery
  445         _ -> draSurfaceProvenance templateArtifact
  446       contractProv = case surfaceProv of
  447         FromRecovery -> RecoveryRoute
  448         _ -> draContractProvenance templateArtifact
  449       assemblyPath = deriveAssemblyPath renderStatic templateArtifact finalizeSurfaceProv
  450       authorityClass = deriveAuthorityClass renderStatic contractProv surfaceProv assemblyPath
  451       artifactManifest = deriveArtifactManifest renderStatic templateArtifact assemblyPath authorityClass
  452       derivationTags =
  453         draDerivationTags templateArtifact
  454           <> [ "surface_provenance=" <> T.pack (show surfaceProv)
  455              , "contract_provenance=" <> T.pack (show contractProv)
  456              , "assembly_path=" <> T.pack (show assemblyPath)
  457              , "authority_class=" <> T.pack (show authorityClass)
  458              ]
  459           <> if finalizeSurfaceProv == FromRecovery then ["finalize=guard_blocked_non_expansive"] else ["finalize=preserve_upstream_surface"]
  460       rendered = Guard.gsRenderedText renderedSurface
  461       (recoveryCause, recoveryStrategy, recoveryEvidence) =
  462         case surfaceProv of
  463           FromRecovery ->
  464             (Just RecoveryRenderBlocked, Just StrategySafeRecovery, ["render_guard=blocked"])
  465           _ ->
  466             case localRecoveryPlan of
  467               Just plan -> (Just (lrpCause plan), Just (lrpStrategy plan), lrpEvidence plan)
  468               Nothing -> (Nothing, Nothing, [])
  469       -- Phase 8 (M3): deliberation override of recovery cause.
  470       -- We merge by severity rather than simple Maybe-fallback so
  471       -- that future hemispheric proposals (Package D and beyond)
  472       -- cannot silently downgrade a more severe local cause.
  473       -- Today this is semantically equivalent because the only
  474       -- deliberation-produced cause is RecoveryConatusGate
  475       -- (severity 100, dominates everything).
  476       delibRecoveryCause = tpDeliberation tp >>= planRecoveryCause . delibReconciled
  477       finalRecoveryCause = pickHigherSeverity delibRecoveryCause recoveryCause
  478       decision = TurnDecision
  479         { tdFamily = case surfaceProv of FromRecovery -> CMRepair; _ -> tpFinalFamily tp
  480         , tdForce = case surfaceProv of FromRecovery -> IFOffer; _ -> tpFinalForce tp
  481         , tdRenderStrategy = tpRenderStrategy tp
  482         , tdRenderStyle = rcpStyle (tpRcpFinal tp)
  483         , tdGuardStatus =
  484             if tpShadowGateTriggered tp
  485               then Blocked (tpShadowMessage tp)
  486               else case guardSafety of
  487                 Guard.InvariantBlock w -> Blocked w
  488                 _ -> tiNixStatus ti
  489         , tdGuardReport = tpGuardReport tp
  490         , tdLegitimacy = tpLegitScore tp
  491         , tdIdentity = snapshotIdentitySignal (tpIdentitySignal tp)
  492         , tdSemanticAnchor = tpSemanticAnchor tp
  493         }
  494       executedOutcome = safeExecutedTurnOutcome decision authorityClass contractProv surfaceProv assemblyPath artifactManifest
  495       executedDecision =
  496         decision
  497           { tdFamily = etoFamily executedOutcome
  498           , tdForce = etoForce executedOutcome
  499           }
  500       finalRendered = truthContractRebindRenderedText (etoTruthContractStatus executedOutcome) authorityClass rendered
  501       finalGuardSurface = renderedSurface { Guard.gsRenderedText = finalRendered }
  502   in TurnArtifacts
  503        { taPreSafetyRendered = preSafetyRendered
  504        , taGuardSurface = finalGuardSurface
  505         , taRendered = finalRendered
  506        , taSurfaceProv = surfaceProv
  507        , taContractProv = contractProv
  508        , taAuthorityClass = authorityClass
  509        , taTruthContractStatus = etoTruthContractStatus executedOutcome
  510        , taAssemblyPath = assemblyPath
  511        , taArtifactManifest = artifactManifest
  512        , taExecutedOutcome = executedOutcome
  513        , taDerivationTags = derivationTags
  514         , taFinalRendered = finalRendered
  515        , taClaimAst = draClaimAst (rsTemplateArtifact renderStatic)
  516       , taLinearizationLang = draLinearizationLang (rsTemplateArtifact renderStatic)

  606 buildLocalRecoveryPlan :: PipelineRuntimeMode -> LocalRecoveryPolicy -> SystemState -> TurnInput -> TurnPlan -> Maybe Text -> Maybe LocalRecoveryPlan
  607 buildLocalRecoveryPlan _ LocalRecoveryDisabled _ _ _ _ = Nothing
  608 buildLocalRecoveryPlan runtimeMode LocalRecoveryEnabled ss ti tp morphologyWarning =
  609   let parserConfidence = ipfConfidence (tiFrame ti)
  610       legitScore = tpLegitScore tp
  611       lowLegitThreshold = ctLocalRecoveryThreshold (obsConstitutionalThresholds (ssObservability ss))
  612       candidateFamilies = localRecoveryCandidateFamilies ti tp
  613       hasCandidateSplit = length candidateFamilies >= 2
  614       hasDisambiguationCue = mentionsDisambiguationCue (ipfRawText (tiFrame ti))
  615       parserStrategy =
  616         if hasCandidateSplit && hasDisambiguationCue
  617           then StrategyDistinguishCandidates
  618           else StrategyAskClarification
  619       parserEvidence =
  620         ["parser_confidence=" <> T.pack (show parserConfidence)]
  621           <> if hasCandidateSplit
  622             then ["candidate_families=" <> renderCandidateFamilies candidateFamilies]
  623             else []
  624       -- Phase 2.5 (M2d) + Phase 6 (M6): the Conatus gate is the
  625       -- highest-priority recovery driver. When the runtime
  626       -- Conatus energy drops below 'conatusGateThreshold' (see
  627       -- 'QxFx0.Self.Salience.defaultSalienceWeights'), the system
  628       -- is in structural risk and 'StrategySafeRecovery' is forced
  629       -- regardless of shadow / parser / legitimacy / runtime-mode
  630       -- signals. The cause is tagged 'RecoveryConatusGate' — a
  631       -- dedicated 'LocalRecoveryCause' variant introduced together
  632       -- with this guard so the trace and the JSON schema can
  633       -- distinguish a structural-Conatus event from an
  634       -- environmental 'RecoveryRuntimeDegraded' event.
  635       --
  636       -- M6: the energy and the violation count are read straight
  637       -- from 'TurnInput', where the Prepare stage stored them
  638       -- (see 'QxFx0.Core.TurnPipeline.Effects.psConatusEnergy'
  639       -- and 'psBlanketViolationCount'). This is the
  640       -- single-source-of-truth for the pre-turn Conatus snapshot;
  641       -- the previous local 'computeSelfBlanket' \/
  642       -- 'checkInitialBlanket' \/ 'computeConatusEnergy' triple has
  643       -- been removed to eliminate the duplicate computation.
  644       conatusEnergy     = tiConatusEnergy ti
  645       conatusViolationCount = tiBlanketViolationCount ti
  646       conatusEvidence =
  647         [ "conatus_gate_fired"
  648         , "conatus_energy=" <> T.pack (show (ceScalar conatusEnergy))
  649         , "blanket_violations=" <> T.pack (show conatusViolationCount)
  650         ]
  651       candidate =
  652         case () of
  653           _
  654             | tiConatusGateFired ti ->
  655                 let gradient = computeConatusGradient (computeSelfBlanket ss)
  656                     strategy = selectConatusRecoveryStrategy gradient
  657                     evidence =
  658                       conatusEvidence
  659                         <> [ "conatus_gradient_m=" <> T.pack (show (cgMorphology gradient))
  660                            , "conatus_gradient_c=" <> T.pack (show (cgIdentity gradient))
  661                            , "conatus_gradient_t=" <> T.pack (show (cgTurns gradient))
  662                            , "conatus_strategy=" <> renderLocalRecoveryStrategy strategy
  663                            ]
  664                  in Just
  665                       ( RecoveryConatusGate
  666                       , strategy
  667                       , evidence
  668                       )
  669             -- WP3: learning-driven recovery.  Only triggered when a
  670             -- persistent need (already stored in state from previous
  671             -- turns) has a high deficit level.  This introduces one
  672             -- turn of natural latency, acting as a lightweight
  673             -- rate-limit / quarantine before escalating to an external
  674             -- request strategy.
  675             | learningNeedActive ss ->
  676                 let lns = ssLearningNeedState ss
  677                     need = lnsCurrentNeed lns
  678                     strategy = selectLearningRecoveryStrategy need
  679                     evidence =
  680                       [ "learning_need=" <> T.pack (show need)
  681                       , "learning_level=" <> T.pack (show (lnsLevel lns))
  682                       , "learning_persistence=" <> T.pack (show (lnsPersistence lns))
  683                       , "learning_strategy=" <> renderLocalRecoveryStrategy strategy
  684                       ]
  685                  in Just
  686                        ( RecoveryLearningNeed
  687                        , strategy
  688                        , evidence
  689                        )
  690             | tpShadowStatus tp == ShadowDiverged
  691                 && tpShadowDivergenceSeverity tp /= ShadowSeverityAdvisory ->
  692                 Just
  693                   ( RecoveryShadowDivergence
  694                   , StrategyNarrowScope
  695                   , [ "shadow_status=diverged"
  696                     , "shadow_kind=" <> T.pack (show (tpShadowDivergenceKind tp))
  697                     , "shadow_severity=" <> shadowDivergenceSeverityText (tpShadowDivergenceSeverity tp)
  698                     ]
  699                   )
  700             | tpShadowStatus tp == ShadowUnavailable ->
  701                 Just
  702                   ( RecoveryShadowUnavailable
  703                   , StrategyExposeUncertainty
  704                   , [ "shadow_status=unavailable"
  705                     , "shadow_snapshot=" <> T.pack (show (tpShadowSnapshotId tp))
  706                     ]
  707                   )
  708             | parserConfidence < parserLowConfidenceThreshold ->
  709                 Just
  710                   ( RecoveryParserLowConfidence
  711                   , parserStrategy
  712                   , parserEvidence
  713                   )
  714             | legitScore < lowLegitThreshold ->
  715                 Just
  716                   ( RecoveryLowLegitimacy
  717                   , StrategyExposeUncertainty
  718                   , ["legitimacy_score=" <> T.pack (show legitScore)]
  719                   )
  720             | runtimeMode == RuntimeDegraded ->
  721                 Just
  722                   ( RecoveryRuntimeDegraded
  723                   , StrategyNarrowScope
  724                   , ["runtime_mode=degraded"]
  725                   )
  726             | hasStructuredDialogueSurface (tiFrame ti) ->
  727                 Nothing
  728             | otherwise ->
  729                 case morphologyWarning of
  730                   Just topic ->
  731                     Just
  732                       ( RecoveryUnknownTopic
  733                       , StrategyDefineKnownTerms
  734                       , ["unknown_topic=" <> topic]
  735                       )
  736                   Nothing ->
  737                     Nothing
  738    in fmap
  739         (\(cause, strategy, evidence) ->
  740           let preferredLang = detectInputGfLang (ipfRawText (tiFrame ti))
  741           in
  742           LocalRecoveryPlan
  743             { lrpCause = cause
  744             , lrpStrategy = strategy
  745             , lrpEvidence = evidence
  746              , lrpSurface = renderLocalRecoverySurface preferredLang cause strategy (tiBestTopic ti) evidence
  747             })
  748          candidate

  750 -- | WP1 (GAP1): map the Conatus gradient to a recovery strategy.
  751 -- The component with the strictly largest partial derivative is
  752 -- the axis that yields the highest marginal gain in Conatus energy
  753 -- and therefore determines the recovery direction.
  754 -- If two or more components tie for the maximum (degenerate gradient),
  755 -- fall back to 'StrategySafeRecovery'.
  756 selectConatusRecoveryStrategy :: ConatusGradient -> LocalRecoveryStrategy
  757 selectConatusRecoveryStrategy g =
  758   let comps =
  759         [ (cgMorphology g, StrategyMorphologyExpansion)
  760         , (cgIdentity   g, StrategyIdentityReinforcement)
  761         , (cgTurns      g, StrategyTemporalDeepening)
  762         ]
  763       sorted = L.sortBy (\(a, _) (b, _) -> compare b a) comps
  764    in case sorted of
  765         ((v1, s1) : (v2, _) : _)
  766           | v1 > v2   -> s1
  767           | otherwise -> StrategySafeRecovery
  768         _ -> StrategySafeRecovery
  769 
  770 -- | WP3: predicate for whether a persisted learning need is active
  771 -- and severe enough to trigger an external-request recovery strategy.
  772 -- The threshold (0.6) is chosen so that only high-deficit needs
  773 -- escalate beyond local strategies; moderate deficits remain handled
  774 -- by the normal turn pipeline.
  775 learningNeedActive :: SystemState -> Bool
  776 learningNeedActive ss =
  777   let lns = ssLearningNeedState ss
  778   in lnsCurrentNeed lns /= NeedNone && lnsLevel lns >= 0.6

  780 -- | WP3: map an active 'LearningNeed' to the corresponding
  781 -- 'LocalRecoveryStrategy' that requests external help.
  782 selectLearningRecoveryStrategy :: LearningNeed -> LocalRecoveryStrategy
  783 selectLearningRecoveryStrategy NeedSalienceCalibration = StrategyRequestCalibration
  784 selectLearningRecoveryStrategy NeedKeywordEnrichment   = StrategyRequestConcept
  785 selectLearningRecoveryStrategy NeedLexiconExtension    = StrategyRequestConcept
  786 selectLearningRecoveryStrategy NeedNone                = StrategySafeRecovery

  788 -- | Parse Conatus gradient components (m, c, t) from recovery evidence.
  789 -- Graceful fallback: returns Nothing for missing or unparsable markers.
  790 parseGradient :: [Text] -> Maybe (Double, Double, Double)
  791 parseGradient evidence =
  792   let findMarker prefix =
  793         case T.stripPrefix prefix =<< L.find (T.isPrefixOf prefix) evidence of
  794           Nothing    -> Nothing
  795           Just rest  -> readMaybe (T.unpack (T.takeWhile (/= ' ') rest))
  796       m = findMarker "conatus_gradient_m="
  797       c = findMarker "conatus_gradient_c="
  798       t = findMarker "conatus_gradient_t="
  799   in case (m, c, t) of
  800        (Just m', Just c', Just t') -> Just (m', c', t')
  801        _                           -> Nothing

  804 renderGradientFragment Nothing           = ""

  806   T.concat [ "gradient(m=" , T.pack (show m)
  807            , ", c=" , T.pack (show c)
  808            , ", t=" , T.pack (show t) , ")"

  985   in
  986   case result of

  996                   , draFallbackReason = glrFallbackReason gfResult
  997                   , draContractProvenance = case glrAuthorityClass gfResult of
  998                       AuthorityShim -> ShimRoute
  999                       AuthorityDefault -> DefaultRoute
 1000                       AuthorityGeneratedArtifact -> GeneratedArtifactRoute
 1001                       AuthorityFallback -> FallbackRoute
 1002                       AuthorityRecovery -> RecoveryRoute
 1003                       AuthorityAssembled -> AssembledClaim
 1004                       AuthorityCanonical -> BuiltClaim
 1005                       AuthorityLegacyIncomplete -> AssembledClaim
 1006                   , draSurfaceProvenance = case glrAuthorityClass gfResult of
 1007                       AuthorityShim -> FromShim
 1008                       AuthorityDefault -> FromDefault
 1009                       AuthorityGeneratedArtifact -> FromGeneratedArtifact
 1010                       AuthorityFallback -> FromFallback
 1011                       AuthorityRecovery -> FromRecovery
 1012                       AuthorityAssembled -> FromOperator
 1013                       AuthorityCanonical -> FromDB
 1014                       AuthorityLegacyIncomplete -> FromHardFallback
 1015                   , draDerivationTags = draDerivationTags baseArtifact
 1016                       <> ["gf_authority=" <> T.pack (show (glrAuthorityClass gfResult))
 1017                          , "gf_assembly_path=" <> T.pack (show (glrAssemblyPath gfResult))
 1018                          ]
 1019                   }
 1020           in renderStatic
 1021                 { rsTemplateArtifact = updatedArtifact

 1088 normalizeGfLang :: Text -> Text
 1089 normalizeGfLang raw =
 1090   case T.toLower (T.strip raw) of
 1091     "ru" -> "QxFx0SyntaxRus"
 1092     "russian" -> "QxFx0SyntaxRus"
 1093     "en" -> "QxFx0SyntaxEng"
 1094     "english" -> "QxFx0SyntaxEng"
 1095     "qxfx0syntaxrus" -> "QxFx0SyntaxRus"
 1096     "qxfx0syntaxeng" -> "QxFx0SyntaxEng"
 1097     x | T.isPrefixOf "qxfx0syntax" x -> raw
 1098     _ -> "QxFx0SyntaxRus"

 1107 detectInputGfLang :: Text -> Text
 1108 -- Language routing policy:
 1109 -- - Pure Latin input (no Cyrillic) → English GF path (QxFx0SyntaxEng)
 1110 -- - Any Cyrillic present → Russian GF path (QxFx0SyntaxRus)
 1111 -- This conservative policy prevents RU leakage on EN input by default.
 1112 -- Override via QXFX0_GF_LANG environment variable.
 1113 detectInputGfLang input
 1114   | hasLatin && not hasCyrillic = "QxFx0SyntaxEng"
 1115   | otherwise = "QxFx0SyntaxRus"
 1116   where
 1117     letters = T.filter isAlpha input
 1118     hasLatin = T.any (\c -> ('a' <= c && c <= 'z') || ('A' <= c && c <= 'Z')) letters
 1119     hasCyrillic = T.any (\c -> ('а' <= c && c <= 'я') || ('А' <= c && c <= 'Я') || c == 'ё' || c == 'Ё') letters

 1121 rebuildRenderWithBg :: RenderStatic -> Text -> Text
 1122 rebuildRenderWithBg renderStatic claimText =
 1123   let renderWithContext =
 1124         T.intercalate
 1125           "\n"
 1126           ( filter
 1127               (not . T.null)
 1128               [ rsModePrefixText renderStatic
 1129               , rsAnchorPrefixText renderStatic
 1130               , claimText
 1131               ]
 1132           )
 1133       withNarrative =
 1134         if T.null (rsNarrativeFragmentText renderStatic)
 1135           then renderWithContext
 1136           else renderWithContext <> "\n" <> T.take 80 (rsNarrativeFragmentText renderStatic)
 1137   in if T.null (rsSurfacingFragmentText renderStatic)
 1138        then withNarrative
 1139        else withNarrative <> "\n" <> rsSurfacingFragmentText renderStatic
