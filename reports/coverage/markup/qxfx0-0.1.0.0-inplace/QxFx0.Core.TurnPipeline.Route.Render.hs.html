   99 planRenderEffectsForRuntime :: PipelineRuntimeMode -> LocalRecoveryPolicy -> SystemState -> TurnInput -> TurnSignals -> TurnPlan -> RenderEffectPlan
  100 planRenderEffectsForRuntime runtimeMode localRecoveryPolicy ss ti ts tp =
  101   let bestTopic = tiBestTopic ti
  102       mFlash = tsFlash ts
  103       consciousLoop' = tsConsciousLoop' ts
  104       rmpAfterLegit = tpRmpAfterLegit tp
  105       rcpFinal = tpRcpFinal tp
  106       semanticAnchor = tpSemanticAnchor tp
  107       (mPressure, principledModeResult) =
  108         case tpPrincipledMode tp of
  109           Just (p, pmr) -> (Just p, Just pmr)
  110           Nothing -> (Nothing, Nothing)
  111       identityClaims = integrateIdentityClaims (ssIdentityClaims ss) (tpFamily tp) bestTopic
  112       input = ipfRawText (tiFrame ti)
  113       structuredSurface = hasStructuredDialogueSurface (tiFrame ti)
  114       dialogueArtifact =
  115         renderDialogueArtifact (tiFrame ti) rmpAfterLegit rcpFinal bestTopic identityClaims (ssMorphology ss)
  116       forceFinalized =
  117         if structuredSurface
  118           then draRenderedText dialogueArtifact
  119           else finalizeForce (rmpForce rmpAfterLegit) (draRenderedText dialogueArtifact)
  120       finalRender =
  121         case mFlash of
  122           Just flash ->
  123             if ifOverridesAll flash
  124               then forceFinalized <> "\n[" <> ifDirective flash <> "]"
  125               else forceFinalized
  126           Nothing -> forceFinalized
  127       renderWithContext =
  128         let modePrefix =
  129               if structuredSurface
  130                 then ""
  131                 else maybe "" (renderPrincipledPrefix mPressure) principledModeResult
  132             stylePrefix =
  133               if structuredSurface
  134                 then ""
  135                 else renderStylePrefix (rcpStyle rcpFinal)
  136             anchorPrefixAllowed =
  137               not structuredSurface && tpFinalFamily tp == CMAnchor
  138             anchorPrefix =
  139               if not anchorPrefixAllowed
  140                 then ""
  141                 else maybe "" renderAnchorPrefix semanticAnchor
  142         in T.intercalate "\n" (filter (not . T.null) [stylePrefix, modePrefix, anchorPrefix, finalRender])
  143       narrativeFragment = maybe "" id (tsNarrativeFragment ts)
  144       narrativeEnriched =
  145         if structuredSurface || T.null narrativeFragment
  146           then renderWithContext
  147           else renderWithContext <> "\n" <> T.take 80 narrativeFragment
  148       surfacingFragment =
  149         case structuredSurface of
  150           True -> ""
  151           False ->
  152             case clLastSurfacing consciousLoop' of
  153               Just se -> surfacingToFragment se
  154               Nothing -> ""
  155       renderWithBg =
  156         if T.null surfacingFragment
  157           then narrativeEnriched
  158           else narrativeEnriched <> "\n" <> surfacingFragment
  159       structuredQuestion =
  160         ipfConfidence (tiFrame ti) >= parserLowConfidenceThreshold
  161           && ipfPropositionType (tiFrame ti) /= "PlainAssert"
  162       morphologyWarning =
  163         if T.any isSpace input
  164              && not structuredQuestion
  165              && not (T.null bestTopic)
  166              && not (hasKnownMorphologyForm (ssMorphology ss) bestTopic)
  167           then Just bestTopic
  168           else Nothing
  169       localRecoveryPlan =
  170         buildLocalRecoveryPlan runtimeMode localRecoveryPolicy ss ti tp morphologyWarning
  171   in RenderEffectPlan
  172       { repRenderStatic = RenderStatic
  173           { rsRenderWithBg = renderWithBg
  174           , rsTemplateArtifact = dialogueArtifact
  175           , rsModePrefixText =
  176               if structuredSurface
  177                 then ""
  178                 else
  179                   T.intercalate
  180                     "\n"
  181                     (filter
  182                       (not . T.null)
  183                       [ renderStylePrefix (rcpStyle rcpFinal)
  184                       , maybe "" (renderPrincipledPrefix mPressure) principledModeResult
  185                       ])
  186           , rsAnchorPrefixText =
  187               if not (not structuredSurface && tpFinalFamily tp == CMAnchor)
  188                 then ""
  189                 else maybe "" renderAnchorPrefix semanticAnchor
  190           , rsNarrativeFragmentText =
  191               if structuredSurface
  192                 then ""
  193                 else narrativeFragment
  194           , rsSurfacingFragmentText = surfacingFragment
  195           }
  196       , repLocalRecoveryPlan = localRecoveryPlan
  197       , repRenderMorphologyWarning = morphologyWarning
  198       }

  200 resolveRenderEffects :: PipelineIO -> RenderEffectPlan -> IO RenderEffectResults
  201 resolveRenderEffects pio effectPlan = do
  202   tRender0 <- resolveRenderCurrentTime pio
  203   warnMorphologyFallback <- shouldWarnMorphologyFallback pio
  204   case repRenderMorphologyWarning effectPlan of
  205     Just bestTopic | warnMorphologyFallback ->
  206       hPutStrLnWarning ("Morphology fallback: unknown topic lexeme: " <> T.unpack bestTopic)
  207     _ ->
  208       pure ()
  209 
  210   resolvedRenderStatic <- resolveRuntimeGfLinearization pio (repRenderStatic effectPlan)
  211   tRender1 <- resolveRenderCurrentTime pio
  212   pure RenderEffectResults
  213     { rerRenderTimeline = RenderTimeline
  214         { rtlRenderStart = tRender0
  215         , rtlRenderEnd = tRender1
  216         }
  217     , rerResolvedRenderStatic = resolvedRenderStatic
  218     }

  220 buildTurnArtifacts :: SystemState -> TurnInput -> TurnSignals -> TurnPlan -> RenderEffectPlan -> RenderEffectResults -> TurnArtifacts
  221 buildTurnArtifacts ss ti _ts tp effectPlan effectResults =
  222   let renderStatic = fromMaybe (repRenderStatic effectPlan) (rerResolvedRenderStatic effectResults)
  223       renderWithBg = rsRenderWithBg renderStatic
  224       localRecoveryPlan = repLocalRecoveryPlan effectPlan
  225       localRecoveryText = lrpSurface <$> localRecoveryPlan
  226       preSafetyRendered =
  227         case localRecoveryText of
  228           Just fb -> renderWithBg <> "\n" <> fb
  229           Nothing -> renderWithBg
  230       preSafetySurface =
  231         Guard.GuardSurface
  232           { Guard.gsRenderedText = preSafetyRendered
  233           , Guard.gsSegments =
  234               filter (not . T.null . Guard.rsText)
  235                 [ Guard.RenderSegment Guard.SegmentTemplate (draStylePrefixText (rsTemplateArtifact renderStatic))
  236                 , Guard.RenderSegment Guard.SegmentTemplate (rsModePrefixText renderStatic)
  237                 , Guard.RenderSegment Guard.SegmentTemplate (rsAnchorPrefixText renderStatic)
  238                 , Guard.RenderSegment Guard.SegmentTemplate (draTemplateBodyText (rsTemplateArtifact renderStatic))
  239                 , Guard.RenderSegment Guard.SegmentIdentityClaim (draClaimText (rsTemplateArtifact renderStatic))
  240                 , Guard.RenderSegment Guard.SegmentNarrative (rsNarrativeFragmentText renderStatic)
  241                 , Guard.RenderSegment Guard.SegmentSurfacing (rsSurfacingFragmentText renderStatic)
  242                 ]
  243                   ++ maybe [] (\fb -> [Guard.RenderSegment Guard.SegmentLocalRecovery fb]) localRecoveryText
  244           , Guard.gsQuestionLike = draQuestionLike (rsTemplateArtifact renderStatic)
  245           }
  246       timeline = rerRenderTimeline effectResults
  247       !metrics4 = addPhase (recordPhase "render" (rtlRenderStart timeline) (rtlRenderEnd timeline)) (tpMetrics tp)
  248       guardSafety = Guard.postRenderSafetyCheckSurface preSafetySurface (F.toList (ssHistory ss))
  249       (renderedSurface, surfaceProv) = finalizeOutput preSafetySurface (F.toList (ssHistory ss))
  250       rendered = Guard.gsRenderedText renderedSurface
  251       finalRendered = rendered
  252       (recoveryCause, recoveryStrategy, recoveryEvidence) =
  253         case surfaceProv of
  254           FromRecovery ->
  255             (Just RecoveryRenderBlocked, Just StrategySafeRecovery, ["render_guard=blocked"])
  256           _ ->
  257             case localRecoveryPlan of
  258               Just plan -> (Just (lrpCause plan), Just (lrpStrategy plan), lrpEvidence plan)
  259               Nothing -> (Nothing, Nothing, [])
  260       decision = TurnDecision
  261         { tdFamily = case surfaceProv of FromRecovery -> CMRepair; _ -> tpFinalFamily tp
  262         , tdForce = case surfaceProv of FromRecovery -> IFOffer; _ -> tpFinalForce tp
  263         , tdRenderStrategy = tpRenderStrategy tp
  264         , tdRenderStyle = rcpStyle (tpRcpFinal tp)
  265         , tdGuardStatus =
  266             if tpShadowGateTriggered tp
  267               then Blocked (tpShadowMessage tp)
  268               else case guardSafety of
  269                 Guard.InvariantBlock w -> Blocked w
  270                 _ -> tiNixStatus ti
  271         , tdGuardReport = tpGuardReport tp
  272         , tdLegitimacy = tpLegitScore tp
  273         , tdIdentity = snapshotIdentitySignal (tpIdentitySignal tp)
  274         , tdSemanticAnchor = tpSemanticAnchor tp
  275         }
  276   in TurnArtifacts
  277       { taPreSafetyRendered = preSafetyRendered
  278       , taGuardSurface = renderedSurface
  279       , taRendered = rendered
  280       , taSurfaceProv = surfaceProv
  281       , taFinalRendered = finalRendered
  282       , taClaimAst = draClaimAst (rsTemplateArtifact renderStatic)
  283       , taLinearizationLang = draLinearizationLang (rsTemplateArtifact renderStatic)
  284       , taLinearizationOk = draLinearizationOk (rsTemplateArtifact renderStatic)
  285       , taLinearizationFallbackReason = draFallbackReason (rsTemplateArtifact renderStatic)
  286       , taDecision = decision
  287       , taLocalRecoveryCause = recoveryCause
  288       , taLocalRecoveryStrategy = recoveryStrategy
  289       , taLocalRecoveryEvidence = recoveryEvidence
  290       , taMetrics = metrics4
  291       }

  293 buildLocalRecoveryPlan :: PipelineRuntimeMode -> LocalRecoveryPolicy -> SystemState -> TurnInput -> TurnPlan -> Maybe Text -> Maybe LocalRecoveryPlan
  294 buildLocalRecoveryPlan _ LocalRecoveryDisabled _ _ _ _ = Nothing
  295 buildLocalRecoveryPlan runtimeMode LocalRecoveryEnabled ss ti tp morphologyWarning =
  296   let parserConfidence = ipfConfidence (tiFrame ti)
  297       legitScore = tpLegitScore tp
  298       lowLegitThreshold = ctLocalRecoveryThreshold (obsConstitutionalThresholds (ssObservability ss))
  299       candidateFamilies = localRecoveryCandidateFamilies ti tp
  300       hasCandidateSplit = length candidateFamilies >= 2
  301       hasDisambiguationCue = mentionsDisambiguationCue (ipfRawText (tiFrame ti))
  302       parserStrategy =
  303         if hasCandidateSplit && hasDisambiguationCue
  304           then StrategyDistinguishCandidates
  305           else StrategyAskClarification
  306       parserEvidence =
  307         ["parser_confidence=" <> T.pack (show parserConfidence)]
  308           <> if hasCandidateSplit
  309             then ["candidate_families=" <> renderCandidateFamilies candidateFamilies]
  310             else []
  311       candidate =
  312         case () of
  313           _
  314             | tpShadowStatus tp == ShadowDiverged
  315                 && tpShadowDivergenceSeverity tp /= ShadowSeverityAdvisory ->
  316                 Just
  317                   ( RecoveryShadowDivergence
  318                   , StrategyNarrowScope
  319                   , [ "shadow_status=diverged"
  320                     , "shadow_kind=" <> T.pack (show (tpShadowDivergenceKind tp))
  321                     , "shadow_severity=" <> shadowDivergenceSeverityText (tpShadowDivergenceSeverity tp)
  322                     ]
  323                   )
  324             | tpShadowStatus tp == ShadowUnavailable ->
  325                 Just
  326                   ( RecoveryShadowUnavailable
  327                   , StrategyExposeUncertainty
  328                   , [ "shadow_status=unavailable"
  329                     , "shadow_snapshot=" <> T.pack (show (tpShadowSnapshotId tp))
  330                     ]
  331                   )
  332             | parserConfidence < parserLowConfidenceThreshold ->
  333                 Just
  334                   ( RecoveryParserLowConfidence
  335                   , parserStrategy
  336                   , parserEvidence
  337                   )
  338             | legitScore < lowLegitThreshold ->
  339                 Just
  340                   ( RecoveryLowLegitimacy
  341                   , StrategyExposeUncertainty
  342                   , ["legitimacy_score=" <> T.pack (show legitScore)]
  343                   )
  344             | runtimeMode == RuntimeDegraded ->
  345                 Just
  346                   ( RecoveryRuntimeDegraded
  347                   , StrategyNarrowScope
  348                   , ["runtime_mode=degraded"]
  349                   )
  350             | hasStructuredDialogueSurface (tiFrame ti) ->
  351                 Nothing
  352             | otherwise ->
  353                 case morphologyWarning of
  354                   Just topic ->
  355                     Just
  356                       ( RecoveryUnknownTopic
  357                       , StrategyDefineKnownTerms
  358                       , ["unknown_topic=" <> topic]
  359                       )
  360                   Nothing ->
  361                     Nothing
  362    in fmap
  363         (\(cause, strategy, evidence) ->
  364           LocalRecoveryPlan
  365             { lrpCause = cause
  366             , lrpStrategy = strategy
  367             , lrpEvidence = evidence
  368             , lrpSurface = renderLocalRecoverySurface cause strategy (tiBestTopic ti)
  369             })
  370         candidate

  372 renderLocalRecoverySurface :: LocalRecoveryCause -> LocalRecoveryStrategy -> Text -> Text
  373 renderLocalRecoverySurface _cause strategy topic =
  374   let topicText = if T.null topic then "этот вопрос" else topic
  375       header = "Локальный режим восстановления."
  376    in case strategy of
  377         StrategyAskClarification ->
  378           header <> " Уточни, тебе нужно определение, различение или пример по теме: " <> topicText <> "?"
  379         StrategyNarrowScope ->
  380           header <> " Я сужаю ответ до устойчивой части и не буду достраивать непроверенные выводы."
  381         StrategyDefineKnownTerms ->
  382           header <> " Я могу опереться только на известные локальные термины; для нового термина нужна рамка употребления: " <> topicText <> "."
  383         StrategyDistinguishCandidates ->
  384           header <> " Я различу возможные чтения и отмечу, где локальных данных недостаточно."
  385         StrategyExposeUncertainty ->
  386           header <> " Уверенность снижена; продолжу с явной пометкой неопределенности вместо внешней догадки."
  387         StrategySafeRecovery ->
  388           header <> " Ответ переведен в безопасную форму восстановления хода."

  401 mentionsDisambiguationCue :: Text -> Bool
  402 mentionsDisambiguationCue input =
  403   let lowered = " " <> T.toLower input <> " "
  404       cues =
  405         [ " или "
  406         , " либо "
  407         , "разниц"
  408         , "отлич"
  409         , " vs "
  410         , " versus "
  411         ]
  412   in any (`T.isInfixOf` lowered) cues

  422     TurnResReadEnv (Just "1") -> pure True
  423     TurnResReadEnv _ -> pure False

  426 resolveRenderCurrentTime :: PipelineIO -> IO UTCTime

  428   result <- resolveTurnEffect pio TurnReqCurrentTime

  433 resolveRuntimeGfLinearization :: PipelineIO -> RenderStatic -> IO (Maybe RenderStatic)
  434 resolveRuntimeGfLinearization pio renderStatic = do
  435   runtimeEnabled <- shouldUseGfRuntime pio
  436   if not runtimeEnabled
  437     then pure Nothing
  438     else
  439       case draClaimAst (rsTemplateArtifact renderStatic) of
  440         Nothing -> pure Nothing
  441         Just claimAst -> do
  442           mPgfPath <- resolveGfPgfPath pio
  443           result <- resolveTurnEffect pio (TurnReqLinearizeClaimAst mPgfPath claimAst)
  444           pure (Just (applyRuntimeGfResult renderStatic result))
  445 
  446 shouldUseGfRuntime :: PipelineIO -> IO Bool
  447 shouldUseGfRuntime pio = do
  448   result <- resolveTurnEffect pio (TurnReqReadEnv "QXFX0_GF_RUNTIME")
  449   case result of
  450     TurnResReadEnv (Just rawValue) ->
  451       pure (normalizeBool rawValue)
  452     _ ->
  453       pure False
  454 
  455 resolveGfPgfPath :: PipelineIO -> IO (Maybe FilePath)
  456 resolveGfPgfPath pio = do
  457   result <- resolveTurnEffect pio (TurnReqReadEnv "QXFX0_GF_PGF_PATH")
  458   case result of
  459     TurnResReadEnv (Just rawPath) ->
  460       let stripped = T.unpack (T.strip rawPath)
  461       in pure (if null stripped then Nothing else Just stripped)
  462     _ ->
  463       pure Nothing
  464 
  465 normalizeBool :: Text -> Bool
  466 normalizeBool rawValue =
  467   T.toLower (T.strip rawValue) `elem` ["1", "true", "yes", "on"]
  468 
  469 applyRuntimeGfResult :: RenderStatic -> TurnEffectResult -> RenderStatic
  470 applyRuntimeGfResult renderStatic result =
  471   case result of
  472     TurnResLinearizeClaimAst (Right gfText)
  473       | not (T.null (T.strip gfText)) ->
  474           let baseArtifact = rsTemplateArtifact renderStatic
  475               updatedArtifact =
  476                 baseArtifact
  477                   { draRenderedText = gfText
  478                   , draTemplateBodyText = gfText
  479                   , draLinearizationLang = Just "ru_GF_PGF"
  480                   , draLinearizationOk = True
  481                   , draFallbackReason = Nothing
  482                   }
  483           in renderStatic
  484                { rsTemplateArtifact = updatedArtifact
  485                , rsRenderWithBg = rebuildRenderWithBg renderStatic gfText
  486                }
  487     TurnResLinearizeClaimAst (Left err) ->
  488       let baseArtifact = rsTemplateArtifact renderStatic
  489       in renderStatic
  490            { rsTemplateArtifact =
  491                baseArtifact
  492                  { draLinearizationLang = Just "ru_GF_PGF"
  493                  , draLinearizationOk = False
  494                  , draFallbackReason = Just ("gf_runtime:" <> err)
  495                  }
  496            }
  497     _ ->
  498       let baseArtifact = rsTemplateArtifact renderStatic
  499       in renderStatic
  500            { rsTemplateArtifact =
  501                baseArtifact
  502                  { draLinearizationLang = Just "ru_GF_PGF"
  503                  , draLinearizationOk = False
  504                  , draFallbackReason = Just "gf_runtime:unexpected_effect_result"
  505                  }
  506            }

  508 rebuildRenderWithBg :: RenderStatic -> Text -> Text
  509 rebuildRenderWithBg renderStatic claimText =
  510   let renderWithContext =
  511         T.intercalate
  512           "\n"
  513           ( filter
  514               (not . T.null)
  515               [ rsModePrefixText renderStatic
  516               , rsAnchorPrefixText renderStatic
  517               , claimText
  518               ]
  519           )
  520       withNarrative =
  521         if T.null (rsNarrativeFragmentText renderStatic)
  522           then renderWithContext
  523           else renderWithContext <> "\n" <> T.take 80 (rsNarrativeFragmentText renderStatic)
  524   in if T.null (rsSurfacingFragmentText renderStatic)
  525        then withNarrative
  526        else withNarrative <> "\n" <> rsSurfacingFragmentText renderStatic
