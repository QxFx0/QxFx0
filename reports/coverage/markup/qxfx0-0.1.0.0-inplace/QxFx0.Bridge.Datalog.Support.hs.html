   45 createShadowTempFiles :: IO (FilePath, FilePath, IO ())
   46 createShadowTempFiles = do
   47   tmpDir <- getTemporaryDirectory
   48   (seedFile, handle) <- openTempFile tmpDir "qxfx0_shadow_rules.seed"
   49   hClose handle
   50   let workDir = seedFile <> ".dir"
   51       dlFile = workDir </> "qxfx0_shadow_rules.dl"
   52       outDir = workDir </> "out"
   53       dirMode = ownerReadMode `unionFileModes` ownerWriteMode `unionFileModes` ownerExecuteMode
   54       cleanupAction = cleanup seedFile workDir
   55   (do createDirectory workDir
   56       setFileMode workDir dirMode
   57       createDirectory outDir
   58       setFileMode outDir dirMode
   59       pure (dlFile, outDir, cleanupAction))
   60     `onException` cleanupAction
   61   where
   62     cleanup seedFile workDir = do
   63       ignoreCleanup (removeFile seedFile)
   64       ignoreCleanup (removeDirectoryRecursive workDir)
   65     ignoreCleanup action =
   66       catchIO action (\_ -> pure ())

   68 renderRuntimeFacts :: ShadowSnapshot -> Text
   69 renderRuntimeFacts snapshot =
   70   T.unlines $
   71     [ "RequestedFamily(\"" <> escapeSymbol (renderShow (ssRequestedFamily snapshot)) <> "\")."
   72     , "InputForce(\"" <> escapeSymbol (renderShow (ssInputForce snapshot)) <> "\")."
   73     ]
   74     ++ map (\atomName -> "InputAtom(\"" <> escapeSymbol atomName <> "\").") (ssInputAtoms snapshot)
   75     ++ map renderAtomDetail (ssInputAtomDetails snapshot)

   77 buildShadowSnapshot :: CanonicalMoveFamily -> IllocutionaryForce -> [AtomTag] -> ShadowSnapshot
   78 buildShadowSnapshot requestedFamily inputForce tags =
   79   let atomNames = sort (nub (map atomTagName tags))
   80       atomDetails = sort (nub (concatMap atomTagDetails tags))
   81   in ShadowSnapshot
   82       { ssRequestedFamily = requestedFamily
   83       , ssInputForce = inputForce
   84       , ssInputAtoms = atomNames
   85       , ssInputAtomDetails = atomDetails
   86       , ssSourceAtomTags = tags
   87       }

   89 renderAtomDetail :: (Text, Text) -> Text
   90 renderAtomDetail (tagName, detail) =
   91   "InputAtomDetail(\""
   92     <> escapeSymbol tagName
   93     <> "\", \""
   94     <> escapeSymbol detail
   95     <> "\")."

   97 atomTagName :: AtomTag -> Text
   98 atomTagName tag = case tag of
   99   Searching _ -> "Searching"
  100   Exhaustion _ -> "Exhaustion"
  101   Verification _ -> "Verification"
  102   Doubt _ -> "Doubt"
  103   NeedContact _ -> "NeedContact"
  104   NeedMeaning _ -> "NeedMeaning"
  105   AgencyLost _ -> "AgencyLost"
  106   AgencyFound _ -> "AgencyFound"
  107   Anchoring _ -> "Anchoring"
  108   Contradiction _ _ -> "Contradiction"
  109   CustomAtom label _ -> "CustomAtom:" <> compactDiagnostic label
  110   AffectiveAtom label _ -> "AffectiveAtom:" <> compactDiagnostic label
  111 
  112 atomTagDetails :: AtomTag -> [(Text, Text)]
  113 atomTagDetails tag = case tag of
  114   Searching payload -> [("Searching", compactDetail payload)]
  115   Exhaustion payload -> [("Exhaustion", compactDetail payload)]
  116   Verification payload -> [("Verification", compactDetail payload)]
  117   Doubt payload -> [("Doubt", compactDetail payload)]
  118   NeedContact payload -> [("NeedContact", compactDetail payload)]
  119   NeedMeaning payload -> [("NeedMeaning", compactDetail payload)]
  120   AgencyLost value -> [("AgencyLost", compactDetail (T.pack (show value)))]
  121   AgencyFound value -> [("AgencyFound", compactDetail (T.pack (show value)))]
  122   Anchoring payload -> [("Anchoring", compactDetail payload)]
  123   Contradiction left right -> [("ContradictionLeft", compactDetail left), ("ContradictionRight", compactDetail right)]
  124   CustomAtom label payload -> [("CustomAtom", compactDetail label), ("CustomPayload", compactDetail payload)]
  125   AffectiveAtom label value -> [("AffectiveAtom", compactDetail label), ("AffectiveValence", compactDetail (T.pack (show value)))]
  126 
  127 compactDetail :: Text -> Text
  128 compactDetail = T.take 24 . compactDiagnostic

  130 parseShadowOutput :: FilePath -> IO (Either Text DatalogExecution)
  131 parseShadowOutput outDir = do
  132   verdictRows <- readRelationRows outDir "R5Verdict.csv"
  133   alertRows <- readRelationRows outDir "ShadowAlert.csv"
  134   case parseVerdictRows verdictRows of
  135     Left err -> pure (Left err)
  136     Right verdict ->
  137       pure (Right (DatalogExecution verdict (map renderDiagnostic alertRows)))
  138 
  139 readRelationRows :: FilePath -> FilePath -> IO [[Text]]
  140 readRelationRows outDir fileName = do
  141   let fullPath = outDir </> fileName
  142   exists <- doesFileExist fullPath
  143   if not exists
  144     then pure []
  145     else do
  146       content <- TIO.readFile fullPath
  147       pure
  148         [ map unquoteCell (splitRelationLine line)
  149         | line <- T.lines content
  150         , not (T.null (T.strip line))
  151         ]
  152 
  153 splitRelationLine :: Text -> [Text]
  154 splitRelationLine line =
  155   let tabParts = map T.strip (T.splitOn "\t" line)
  156   in if length tabParts > 1
  157        then tabParts
  158        else map T.strip (T.splitOn "," line)
  159 
  160 parseVerdictRows :: [[Text]] -> Either Text R5Verdict
  161 parseVerdictRows [] = Left "shadow produced no R5Verdict rows"
  162 parseVerdictRows (row:_) = case row of
  163   [famText, forceText, clauseText, layerText, warrantedText] ->
  164     R5Verdict
  165       <$> parseEnumText "family" famText
  166       <*> parseEnumText "force" forceText
  167       <*> parseEnumText "clause" clauseText
  168       <*> parseEnumText "layer" layerText
  169       <*> parseEnumText "warranted" warrantedText
  170   _ ->
  171     Left ("unexpected R5Verdict row: " <> T.intercalate "|" row)
  172 
  173 parseEnumText :: Read a => Text -> Text -> Either Text a
  174 parseEnumText label raw =
  175   case readMaybe (T.unpack cleaned) of
  176     Just value -> Right value
  177     Nothing -> Left ("invalid " <> label <> " from shadow: " <> cleaned)
  178   where
  179     cleaned = T.strip raw
  180 
  181 renderDiagnostic :: [Text] -> Text
  182 renderDiagnostic [] = "shadow_alert"
  183 renderDiagnostic [kind] = kind
  184 renderDiagnostic [kind, detail] = kind <> ":" <> detail
  185 renderDiagnostic values = T.intercalate ":" values
  186 
  187 unquoteCell :: Text -> Text
  188 unquoteCell raw =
  189   let stripped = T.strip raw
  190   in case T.uncons stripped of
  191        Just ('"', rest) -> case T.unsnoc rest of
  192          Just (inner, '"') -> inner
  193          _ -> stripped
  194        _ -> stripped
  195 
  196 renderShow :: Show a => a -> Text
  197 renderShow = T.pack . show
  198 
  199 escapeSymbol :: Text -> Text
  200 escapeSymbol =
  201   T.concatMap $ \ch -> case ch of
  202     '\\' -> "\\\\"
  203     '"' -> "\\\""
  204     '(' -> "_"
  205     ')' -> "_"
  206     ';' -> "_"
  207     '.' -> "_"
  208     '%' -> "_"
  209     '\n' -> " "
  210     '\r' -> " "
  211     '\t' -> " "
  212     _ -> T.singleton ch
  213 
  214 compactDiagnostic :: Text -> Text
  215 compactDiagnostic =
  216   T.intercalate " " . filter (not . T.null) . T.words
