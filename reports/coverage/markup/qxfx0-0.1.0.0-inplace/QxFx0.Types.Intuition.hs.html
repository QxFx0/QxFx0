   25 import QxFx0.Types.Thresholds

   31 
   32 data IntuitiveState = IntuitiveState
   33   { isCooldown :: !Int
   34   , isFlashCount :: !Int
   35   , isLastTurn :: !Int
   36   , isPosterior :: !Double
   37   , isLongPosterior :: !Double
   38   } deriving stock (Show, Read, Eq, Generic)
   39     deriving anyclass (NFData)
   40 
   41 instance ToJSON IntuitiveState where
   42   toJSON state = object
   43     [ "cooldown" .= isCooldown state
   44     , "flashCount" .= isFlashCount state
   45     , "lastTurn" .= isLastTurn state
   46     , "posterior" .= isPosterior state
   47     , "longPosterior" .= isLongPosterior state
   48     ]
   49 
   50 instance FromJSON IntuitiveState where
   51   parseJSON = withObject "IntuitiveState" $ \o -> do
   52     posterior <- o .:? "posterior" .!= basePrior
   53     longPosterior <- o .:? "longPosterior" .!= posterior
   54     IntuitiveState
   55       <$> o .:? "cooldown" .!= 0
   56       <*> o .:? "flashCount" .!= 0
   57       <*> o .:? "lastTurn" .!= 0
   58       <*> pure posterior
   59       <*> pure longPosterior
   60 
   61 basePrior :: Double
   62 basePrior = intuitionBasePriorDefault
   63 
   64 defaultIntuitiveState :: IntuitiveState
   65 defaultIntuitiveState = IntuitiveState 0 0 0 basePrior basePrior
   66 
   67 effectivePosterior :: IntuitiveState -> Double
   68 effectivePosterior state =
   69   let blended =
   70         intuitionEffectivePosteriorShortWeight * isPosterior state
   71           + intuitionEffectivePosteriorLongWeight * isLongPosterior state
   72   in clamp01 (max (isPosterior state) blended)
