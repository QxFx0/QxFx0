   49 data IntuitiveState = IntuitiveState
   50   { isCooldown :: !Int
   51   , isFlashCount :: !Int
   52   , isLastTurn :: !Int
   53   , isPosterior :: !Double
   54   , isLongPosterior :: !Double
   55   } deriving stock (Show, Read, Eq, Generic)
   56     deriving anyclass (NFData)
   57 
   58 instance ToJSON IntuitiveState where
   59   toJSON state = object
   60     [ "cooldown" .= isCooldown state
   61     , "flashCount" .= isFlashCount state
   62     , "lastTurn" .= isLastTurn state
   63     , "posterior" .= isPosterior state
   64     , "longPosterior" .= isLongPosterior state
   65     ]
   66 
   67 instance FromJSON IntuitiveState where
   68   parseJSON = withObject "IntuitiveState" $ \o -> do
   69     posterior <- o .:? "posterior" .!= basePrior
   70     longPosterior <- o .:? "longPosterior" .!= posterior
   71     IntuitiveState
   72       <$> o .:? "cooldown" .!= 0
   73       <*> o .:? "flashCount" .!= 0
   74       <*> o .:? "lastTurn" .!= 0
   75       <*> pure posterior
   76       <*> pure longPosterior
   77 
   78 basePrior :: Double
   79 basePrior = intuitionBasePriorDefault
   80 
   81 defaultIntuitiveState :: IntuitiveState
   82 defaultIntuitiveState = IntuitiveState 0 0 0 basePrior basePrior
   83 
   84 effectivePosterior :: IntuitiveState -> Double
   85 effectivePosterior state =
   86   let blended =
   87         intuitionEffectivePosteriorShortWeight * isPosterior state
   88           + intuitionEffectivePosteriorLongWeight * isLongPosterior state
   89   in clamp01 (max (isPosterior state) blended)
