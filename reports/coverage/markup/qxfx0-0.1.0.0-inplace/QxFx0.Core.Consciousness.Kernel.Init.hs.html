   79 mkDesire :: (Text, Text, Text, Maybe Text) -> KernelDesire
   80 mkDesire (name, strength, vector, conflict) = KernelDesire
   81   { desireName = name
   82   , desireStrength = case strength of
   83       "Fundamental" -> Fundamental
   84       "Strong" -> Strong
   85       "Moderate" -> Moderate

   87   , desireVector = vector

   91 mkSkillSet :: [(Text, Text, Double, Double)] -> SkillSet
   92 mkSkillSet skillsRaw =
   93   let allSkills = map (\(n, d, a, c) -> Skill n d a c) skillsRaw
   94    in SkillSet
   95         { skills = allSkills
   96         , dominantSkill = case allSkills of
