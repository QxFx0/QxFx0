   20 buildIdentityGuardReportSimple
   21   :: IdentityGuardCalibration
   22   -> Double -> Double -> Double -> Double
   23   -> IdentityGuardReport
   24 buildIdentityGuardReportSimple calib prevAgency curAgency prevTension curTension =
   25   let agencyDelta = curAgency - prevAgency
   26       tensionDelta = curTension - prevTension
   27       absTensionDelta = abs tensionDelta
   28       warnings = [ GuardTransitionOutsideManifold | absTensionDelta > igcTensionDriftThreshold calib ]
   29                   ++ [ GuardHighTensionDrift          | curTension > igcTensionCeiling calib ]
   30                   ++ [ GuardAgencyCollapse             | curAgency < igcAgencyFloor calib ]
   31   in IdentityGuardReport
   32      { igrAgencyDelta  = agencyDelta
   33      , igrTensionDelta = tensionDelta
   34      , igrWithinBounds = null warnings
   35      , igrWarnings     = warnings
   36      }
