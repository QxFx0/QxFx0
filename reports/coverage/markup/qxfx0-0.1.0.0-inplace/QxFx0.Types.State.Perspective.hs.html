  268 instance FromJSON PerspectiveRegistry where
  269   parseJSON = withObject "PerspectiveRegistry" $ \o -> do
  270     threads <- o .:? "threads" .!= []
  271     profiles <- o .:? "normativeProfiles" .!= [defaultNormativeProfile]
  272     if hasDuplicateKeys (map ptScope threads) || hasDuplicateKeys (map npId profiles)
  273       then empty
  274       else pure ()
  275     activeProfileId <- o .:? "activeNormativeProfileId" .!= defaultNormativeProfileId
  276     PerspectiveRegistry
  277       <$> pure (M.fromList [(ptScope thread, thread) | thread <- threads])
  278       <*> pure (M.fromList [(npId profile, profile) | profile <- profiles])
  279       <*> pure activeProfileId
  280       <*> o .:? "nextPerspectiveOrdinal" .!= 1
  281       <*> o .:? "nextVersionOrdinal" .!= 1
  282       <*> o .:? "maxActivePerspectives" .!= 16
  283       <*> o .:? "maxActivePerScope" .!= 1
  284       <*> o .:? "maxRevisionsPerScope" .!= 12
  285       <*> o .:? "maxInactiveVersions" .!= 8
  286       <*> o .:? "maxLineageSlice" .!= 8
  287       <*> o .:? "lastUpdatedTurn" .!= 0

  343 latestEndorsedPerspective :: PerspectiveThread -> Maybe EndorsedPerspective
  344 latestEndorsedPerspective thread = case ptVersions thread of
  345   [] -> Nothing
  346   x:_ -> Just x
  347 
  348 activeEndorsedPerspective :: PerspectiveThread -> Maybe EndorsedPerspective
  349 activeEndorsedPerspective thread =
  350   case ptActiveVersion thread of
  351     Nothing -> Nothing
  352     Just versionId ->
  353       case filter (isActiveVersion versionId) (ptVersions thread) of
  354         (entry:_) -> Just entry
  355         [] -> Nothing
  356   where
  357     isActiveVersion versionId entry =
  358       epVersion entry == versionId && epStatus entry == PerspectiveActive
  359 
  360 activePerspectiveProjectionScope :: PerspectiveRegistry -> Maybe PerspectiveScope
  361 activePerspectiveProjectionScope registry =
  362   case activePerspectiveProjectionScopes registry of
  363     scope:_ -> Just scope
  364     [] -> Nothing
  365 
  366 activePerspectiveProjectionScopes :: PerspectiveRegistry -> [PerspectiveScope]
  367 activePerspectiveProjectionScopes registry =
  368   map ptScope
  369     . take (prMaxActivePerspectives registry)
  370     . sortOn (Down . ptLastUpdatedTurn)
  371     . filter hasProjection
  372     $ M.elems (prThreads registry)
  373   where
  374     hasProjection thread = activeEndorsedPerspective thread /= Nothing

  376 hasDuplicateKeys :: Ord a => [a] -> Bool
  377 hasDuplicateKeys values =
  378   M.size (M.fromList [(value, ()) | value <- values]) /= length values
