   50 routeFamily :: CanonicalMoveFamily -> InputPropositionFrame -> AtomSet -> UserState
   51             -> SystemState -> [Text] -> Text -> Bool -> Text
   52             -> Maybe ConsciousnessNarrative -> Double
   53             -> RoutingDecision
   54 routeFamily recommendedFamily frame atomSet nextUserState ss history input isNixBlocked currentTopic mNarrative intuitPosterior =
   55   let phase@RoutingPhase{..} = computeRoutingPhase recommendedFamily frame atomSet nextUserState ss history input
   56       cascade = runFamilyCascade phase ss nextUserState frame atomSet history input mNarrative intuitPosterior isNixBlocked
   57       FamilyCascade{..} = cascade
   58 
   59       newEgo = updateEgoFromTurn (ssEgo ss) fcFinalFamily (computeTensionDelta input ss)
   60       identitySignal = buildIdentitySignalSimple rpOrbitalPhase rpEncounterMode rpPrevDirective
   61                          (asRegister atomSet) (usNeedLayer nextUserState) fcFinalFamily (forceForFamily fcFinalFamily)
   62       guardReport = buildGuardReport (ssLastGuardReport ss) (ssEgo ss) newEgo
   63       semanticInput = buildSemanticInputSimple input atomSet frame fcFinalFamily (asRegister atomSet) (usNeedLayer nextUserState)
   64       semanticAnchor = deriveSemanticAnchor (ssSemanticAnchor ss) semanticInput currentTopic (ssTurnCount ss + 1)
   65       renderStrategy = rpChosenStrategy
   66       renderStyle = renderStyleFromDecision renderStrategy rpPrincipledModeResult identitySignal semanticAnchor semanticInput
   67   in RoutingDecision
   68        { rdFamily = fcFinalFamily
   69        , rdNewEgo = newEgo
   70        , rdIdentitySignal = identitySignal
   71        , rdGuardReport = guardReport
   72        , rdSemanticInput = semanticInput
   73        , rdSemanticAnchor = semanticAnchor
   74        , rdRenderStrategy = renderStrategy
   75        , rdRenderStyle = renderStyle
   76        , rdPrincipledMode = rpPrincipledModeResult
   77        , rdPressure = rpMPressure
   78        , rdUpdatedOrbital = rpUpdatedOrbital
   79        , rdFromMs = rpFromMs
   80        , rdToMs = rpToMs
   81        , rdStrategyFamily = rpStrategyFamily
   82        }
