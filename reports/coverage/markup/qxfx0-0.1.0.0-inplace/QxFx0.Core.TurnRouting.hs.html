   76 -- | Phase 6 (M6) + Phase 5.5d: 'routeFamily' takes the per-turn
   77 -- 'ConatusEnergy' and 'Field' precomputed by the Prepare stage
   78 -- (see 'QxFx0.Core.TurnPipeline.Effects.psConatusEnergy' \/
   79 -- 'psField', threaded through 'tiConatusEnergy' \/ 'tiField').
   80 -- The previous local computeSelfBlanket \/ checkInitialBlanket
   81 -- \/ salienceFromBlanket triple was already collapsed in M6;
   82 -- this commit additionally replaces the @emptyField@ default in
   83 -- 'salienceFromConatusEnergy' with the canonical pre-turn
   84 -- 'Field' carrying the runtime Resonance component.
   85 routeFamily :: CanonicalMoveFamily -> InputPropositionFrame -> AtomSet -> UserState
   86             -> SystemState -> [Text] -> Text -> Bool -> Text
   87             -> Maybe ConsciousnessNarrative -> Double -> ConatusEnergy -> Field
   88             -> Maybe (Plan -> Bool)
   89             -> RoutingDecision
   90 routeFamily recommendedFamily frame atomSet nextUserState ss history input isNixBlocked currentTopic mNarrative intuitPosterior conatusEnergy preparedField mCourtesy =
   91   let phase@RoutingPhase{..} = computeRoutingPhase recommendedFamily frame atomSet nextUserState ss history input
   92       routingSalience = computeSalience (ssSalienceWeights ss) conatusEnergy preparedField
   93       cascade = runFamilyCascade phase ss nextUserState frame atomSet history input mNarrative intuitPosterior isNixBlocked routingSalience
   94       FamilyCascade{..} = cascade
   95 
   96       -- Phase 8 (M1/M2): build hemispheric proposals and reconcile.
   97       -- Package D: family divergence is feature-flagged (default
   98       -- False) so baseline behaviour is preserved while the
   99       -- adjunction mapping is corrected.
  100       familyDivergenceEnabled = False
  101       formalFamily   = fcFinalFamily
  102       holisticFamily = if familyDivergenceEnabled then nearestHolistic fcFinalFamily else fcFinalFamily
  103       renderStrategy = rpChosenStrategy
  104       atmosphereTone =
  105         let a = fieldAtmosphere preparedField
  106             dm = defaultDeliberationModulation
  107         in if atmosphereArousal a > dmToneArousalFloor dm && atmosphereValence a >= dmToneValenceNeutral dm
  108              then NarrativeWarm
  109            else if atmosphereArousal a > dmToneArousalFloor dm && atmosphereValence a < dmToneValenceNeutral dm
  110              then NarrativeTerse
  111            else NarrativeNeutral
  112 
  113       baseStyle =
  114         let styleIdentitySignal = buildIdentitySignalSimple rpOrbitalPhase rpEncounterMode rpPrevDirective
  115                                   (asRegister atomSet) (usNeedLayer nextUserState) fcFinalFamily (forceForFamily fcFinalFamily)
  116             styleSemanticInput  = buildSemanticInputSimple input atomSet frame fcFinalFamily (asRegister atomSet) (usNeedLayer nextUserState)
  117             styleSemanticAnchor = deriveSemanticAnchor (ssSemanticAnchor ss) styleSemanticInput currentTopic (ssTurnCount ss + 1)
  118          in renderStyleFromDecision renderStrategy rpPrincipledModeResult atmosphereTone styleIdentitySignal styleSemanticAnchor styleSemanticInput
  119       salienceStyle = applySalienceToStyle (ssSalienceWeights ss) routingSalience preparedField baseStyle
  120 
  121       -- Contract status: bounded causal contour. Narrative tone is not
  122       -- rendered directly, but committed essence validation reads the
  123       -- reconciled tone and may fail closed on mismatch.
  124       -- Phase 8 Package C: observability-grade divergence.
  125       -- Both proposals share the same reconciled family and style
  126       -- (behaviour-preserving baseline), but the holistic proposal
  127       -- carries a narrative tone derived from field atmosphere.
  128       -- This produces trace-visible 'DivergeOnTone' without directly
  129       -- changing rendered output (§1 ADR-0011 Package C note).
  130       formalPlan = defaultPlan
  131         { planFamily      = formalFamily
  132         , planRenderStyle = salienceStyle
  133         }
  134       holisticPlan = defaultPlan
  135         { planFamily      = holisticFamily
  136         , planRenderStyle = salienceStyle
  137         , planNarrativeTone = atmosphereTone
  138         , planConfidence  = salienceConfidence routingSalience
  139         }
  140       -- Package D: formal hemisphere probes the field; holistic
  141       -- hemisphere carries atmosphere-derived tone already baked
  142       -- into the Plan, so the wrapper is field-independent.
  143       deliberation = reconcile mCourtesy routingSalience (holisticProposal holisticPlan emptyField) (formalProposal (\_fd -> formalPlan)) preparedField
  144       reconciledPlan = delibReconciled deliberation
  145       reconciledFamily = planFamily reconciledPlan
  146       reconciledStyle  = planRenderStyle reconciledPlan
  147 
  148       -- Recompute downstream context if the reconciled family
  149       -- diverges from the salience-escalated candidate.
  150       newEgo = updateEgoFromTurn (ssEgo ss) reconciledFamily (computeTensionDelta input ss)
  151       identitySignal = buildIdentitySignalSimple rpOrbitalPhase rpEncounterMode rpPrevDirective
  152                          (asRegister atomSet) (usNeedLayer nextUserState) reconciledFamily (forceForFamily reconciledFamily)
  153       guardReport = buildGuardReport (ssLastGuardReport ss) (ssEgo ss) newEgo
  154       semanticInput = buildSemanticInputSimple input atomSet frame reconciledFamily (asRegister atomSet) (usNeedLayer nextUserState)
  155       semanticAnchor = deriveSemanticAnchor (ssSemanticAnchor ss) semanticInput currentTopic (ssTurnCount ss + 1)
  156    in RoutingDecision
  157         { rdFamily = reconciledFamily
  158        , rdNewEgo = newEgo
  159        , rdIdentitySignal = identitySignal
  160        , rdGuardReport = guardReport
  161        , rdSemanticInput = semanticInput
  162        , rdSemanticAnchor = semanticAnchor
  163        , rdRenderStrategy = renderStrategy
  164        , rdRenderStyle = reconciledStyle
  165        , rdPrincipledMode = rpPrincipledModeResult
  166        , rdPressure = rpMPressure
  167        , rdUpdatedOrbital = rpUpdatedOrbital
  168        , rdFromMs = rpFromMs
  169        , rdToMs = rpToMs
  170        , rdStrategyFamily = rpStrategyFamily
  171        , rdDeliberation = Just deliberation
  172        }
