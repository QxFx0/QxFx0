   51 defaultConatusPrior :: ConatusPrior
   52 defaultConatusPrior conatus request =
   53   let conatusBias
   54         | ceScalar conatus < 0 = 100
   55         | ceScalar conatus < 0.5 = 10
   56         | otherwise = 0
   57   in conatusBias + intrinsicPriority request
   58   where
   59     intrinsicPriority req =
   60       case req of
   61         TurnReqApiHealth -> 90
   62         TurnReqReadEnv _ -> 85
   63         TurnReqSemanticIntrospectionEnv -> 85
   64         TurnReqCurrentTime -> 85
   65         TurnReqRequestId -> 85
   66         TurnReqAgdaVerify -> 80
   67         TurnReqShadow _ _ _ -> 75
   68         TurnReqConsciousness _ _ _ _ _ -> 70
   69         TurnReqIntuition _ _ _ _ _ _ _ -> 65
   70         TurnReqEmbedding _ -> 60
   71         TurnReqNixGuard _ _ _ -> 55
   72         TurnReqLinearizeClaimAst _ _ _ -> 50
   73         TurnReqLinearizeDialogAtoms _ _ _ -> 50
   74         TurnReqExternalQuery _ _ _ -> 45
   75         TurnReqCommitRuntimeState _ _ _ -> 40
   76         TurnReqSaveState _ _ _ -> 35
   77         TurnReqRollbackTurnProjections _ _ -> 30
   78         TurnReqCheckpoint _ -> 25
