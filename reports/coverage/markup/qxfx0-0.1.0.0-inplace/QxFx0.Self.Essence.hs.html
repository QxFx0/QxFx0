  155 -- | Coarse hash of a 'Field' into a 'FieldSignature' for
  156 -- trajectory storage. Components in [0, 1] bucket at
  157 -- @bandLow / bandHigh@; valence in [-1, 1] buckets at
  158 -- @valenceLowEdge / valenceHighEdge@.
  159 fieldSignature :: EssenceModulation -> Field -> FieldSignature
  160 fieldSignature em f = FieldSignature
  161   { fsResonance      = bucketUnit (emBandLowEdge em) (emBandHighEdge em) (unResonance       (fieldResonance      f))
  162   , fsArousal        = bucketUnit (emBandLowEdge em) (emBandHighEdge em) (atmosphereArousal (fieldAtmosphere     f))
  163   , fsValence        = bucketValence (emValenceLowEdge em) (emValenceHighEdge em) (atmosphereValence (fieldAtmosphere f))
  164   , fsConsolidation  = bucketUnit (emBandLowEdge em) (emBandHighEdge em) (unConsolidation   (fieldConsolidation  f))
  165   , fsCounterfactual = bucketUnit (emBandLowEdge em) (emBandHighEdge em) (unCounterfactual  (fieldCounterfactual f))
  166   }
  167   where
  168     bucketUnit lo hi x
  169       | x < lo    = BandLow
  170       | x > hi    = BandHigh
  171       | otherwise = BandMid
  172     bucketValence lo hi x
  173       | x < lo    = ValenceNegative
  174       | x > hi    = ValencePositive
  175       | otherwise = ValenceNeutral

  295 -- | Ingest one turn's deliberation into the trajectory.
  296 --
  297 -- Effects:
  298 --
  299 -- 1. Append a new 'EssenceWitness' to 'etWitnesses', truncating
  300 --    the front to keep length ≤ 'etCapacity'.
  301 -- 2. Update 'etAngstLevel':
  302 --      * RuleConatusOverride       → no change.
  303 --      * FullAgreement, divergence 0 → decay by 'emAngstDecayRate'.
  304 --      * RuleHolisticAdvantage/RuleFormalAdvantage with
  305 --        divergence ≥ 'emAngstAccrualDivergenceFloor'
  306 --                              → accrue by 'emAngstAccrualRate'.
  307 --      * otherwise (RuleSalienceLead, RuleTiedFallback,
  308 --        sub-floor divergence)
  309 --                              → no change.
  310 --    Clamp to @[0, 1]@.
  311 -- 3. Update 'etConatusFloor' to @min etConatusFloor ceScalar@.
  312 witness
  313   :: EssenceModulation
  314   -> Int          -- turn ordinal
  315   -> ConatusEnergy
  316   -> Field
  317   -> Deliberation
  318   -> EssenceTrajectory
  319   -> EssenceTrajectory
  320 witness em turnOrdinal ce field deliberation traj =
  321   let sig     = fieldSignature em field
  322       witness_ = EssenceWitness
  323         { ewTurnOrdinal     = turnOrdinal
  324         , ewSalienceDriver  = dtSalienceDriver (delibTrace deliberation)
  325         , ewReconcileRule   = dtRule (delibTrace deliberation)
  326         , ewAgreement       = dtAgreement (delibTrace deliberation)
  327         , ewDivergence      = dtDivergence (delibTrace deliberation)
  328         , ewConatusScalar   = ceScalar ce
  329         , ewFieldSignature  = sig
  330         }
  331       newWitnesses = trimSeq (etCapacity traj) (etWitnesses traj Seq.|> witness_)
  332       newAngst = clampUnit $ case dtRule (delibTrace deliberation) of
  333         RuleConatusOverride -> etAngstLevel traj
  334         RuleAgreement
  335           | dtDivergence (delibTrace deliberation) == 0.0
  336           -> max 0.0 (etAngstLevel traj - emAngstDecayRate em)
  337         RuleHolisticAdvantage
  338           | dtDivergence (delibTrace deliberation) >= emAngstAccrualDivergenceFloor em
  339           -> min 1.0 (etAngstLevel traj + emAngstAccrualRate em)
  340         RuleFormalAdvantage
  341           | dtDivergence (delibTrace deliberation) >= emAngstAccrualDivergenceFloor em
  342           -> min 1.0 (etAngstLevel traj + emAngstAccrualRate em)
  343         _ -> etAngstLevel traj
  344       newFloor = min (etConatusFloor traj) (ceScalar ce)
  345   in traj
  346        { etWitnesses    = newWitnesses
  347        , etAngstLevel   = newAngst
  348        , etConatusFloor = newFloor
  349        }
  350   where
  351     trimSeq cap sq
  352       | Seq.length sq > cap = case Seq.viewl sq of
  353           Seq.EmptyL  -> sq
  354           (_ Seq.:< rest) -> trimSeq cap rest
  355       | otherwise = sq

  357 -- | Returns 'Just' when the trajectory has crossed a commitment
  358 -- threshold. Priority: 'TriggerAngstThreshold' over
  359 -- 'TriggerConatusErosion' (see §0 Q2 of the spec).
  360 --
  361 -- Conatus erosion triggers when *every one* of the last
  362 -- 'emConatusFloorWindow' witnessed turns has 'ewConatusScalar'
  363 -- below 'emConatusStructuralFloor' (true sliding window, Phase 10).
  364 -- The 'etConatusFloor' field is retained for diagnostics and trace
  365 -- but is no longer the trigger signal.
  366 shouldCommit :: EssenceModulation -> EssenceTrajectory -> Maybe CommitmentTrigger
  367 shouldCommit em traj =
  368   let angstFires = etAngstLevel traj >= emAngstCommitmentThreshold em
  369       window     = emConatusFloorWindow em
  370       ws         = etWitnesses traj
  371       lastN      = Seq.drop (max 0 (Seq.length ws - window)) ws
  372       enoughWits = Seq.length lastN >= window
  373       allSubFloor =
  374         enoughWits
  375           && all (\w -> ewConatusScalar w < emConatusStructuralFloor em) lastN
  376   in case (angstFires, allSubFloor) of
  377        (True,  _)    -> Just TriggerAngstThreshold
  378        (False, True) -> Just TriggerConatusErosion
  379        _             -> Nothing

  381 -- | Deterministic mode extraction.  Never returns
  382 -- 'EssenceWitnessing'.  Selection rule:
  383 --
  384 -- @
  385 --   let n = length etWitnesses
  386 --       aRate = #{Agreement = FullAgreement} / n
  387 --       hRate = #{ReconcileRule = RuleHolisticAdvantage} / n
  388 --       fRate = #{ReconcileRule = RuleFormalAdvantage}   / n
  389 --   in case maxBy snd
  390 --        [(EssenceIntegrative,   aRate)
  391 --        ,(EssenceDialogical,    hRate)
  392 --        ,(EssenceContemplative, fRate)] of
  393 --        (m, r) | r > 0     -> m
  394 --               | otherwise -> EssenceContemplative   -- tie-break
  395 -- @
  396 --
  397 -- The tie-break is 'EssenceContemplative' because, in the absence
  398 -- of any signal, the formal hemisphere is the safer default
  399 -- (lower behavioural divergence post-commit).
  400 extractMode :: EssenceTrajectory -> EssenceMode
  401 extractMode traj =
  402   let ws = etWitnesses traj
  403       n  = fromIntegral (Seq.length ws) :: Double
  404       counts = foldl' tally (0, 0, 0) ws
  405       (aCount, hCount, fCount) = counts
  406       aRate = if n > 0 then fromIntegral aCount / n else 0.0
  407       hRate = if n > 0 then fromIntegral hCount / n else 0.0
  408       fRate = if n > 0 then fromIntegral fCount / n else 0.0
  409       candidates =
  410         [ (EssenceIntegrative,   aRate)
  411         , (EssenceDialogical,    hRate)
  412         , (EssenceContemplative, fRate)
  413         ]
  414       (bestMode, bestRate) = maximumBySnd candidates
  415   in if bestRate > 0.0 then bestMode else EssenceContemplative
  416   where
  417     tally (a, h, f) w =
  418       case (ewAgreement w, ewReconcileRule w) of
  419         (Agree, _)            -> (a + 1, h, f)
  420         (_, RuleHolisticAdvantage) -> (a, h + 1, f)
  421         (_, RuleFormalAdvantage)   -> (a, h, f + 1)
  422         _                   -> (a, h, f)
  423     maximumBySnd [] = (EssenceContemplative, 0.0)
  424     maximumBySnd (x:xs) = foldl' pick x xs
  425       where
  426         pick acc@(_, va) y@(_, vb) = if vb > va then y else acc

  428 -- | Construct the commitment.  Total over trajectories that have
  429 -- passed 'shouldCommit'; technically total over all trajectories
  430 -- but used only when 'shouldCommit' returned 'Just'.  The hash is
  431 -- a simple structural fold over the 'etWitnesses' sequence —
  432 -- enough to detect tampering across replay, not a cryptographic
  433 -- guarantee.
  434 commit
  435   :: Int                -- current turn ordinal
  436   -> CommitmentTrigger
  437   -> EssenceTrajectory
  438   -> EssenceCommitment
  439 commit turnOrdinal trigger traj =
  440   EssenceCommitment
  441     { ecMode        = extractMode traj
  442     , ecTrigger     = trigger
  443     , ecCommittedAt = turnOrdinal
  444     , ecWitnessHash = TrajectoryHash (sha256Witnesses (toList (etWitnesses traj)))
  445     }
  446   where
  447     sha256Witnesses witnesses =
  448       "sha256:" <> T.pack (concatMap byteToHex (BS.unpack (SHA256.hashlazy (encode witnesses))))
  449     byteToHex byte =
  450       let raw = showHex byte ""
  451       in case raw of
  452            [single] -> ['0', single]
  453            other -> other

  459 -- | A commitment violation.  Priority order: family → tone → style;
  460 -- the first mismatch is the one reported.
  461 data EssenceViolation
  462   = ViolationFamilyMismatch !EssenceMode !CanonicalMoveFamily
  463   | ViolationToneMismatch   !EssenceMode !NarrativeTone
  464   | ViolationStyleMismatch  !EssenceMode !RenderStyle
  465   | ViolationMissingDeliberation !EssenceMode
  466     -- ^ A committed essence cannot be validated without a reconciled
  467     --   plan. Missing deliberation is therefore a fail-closed state,
  468     --   not an implicit validation pass.
  469   | ViolationRefusedCommitment !CommitmentTrigger
  470     -- ^ WP1 (contour closure): 'shouldCommit' fired with this trigger,
  471     --   but the resulting state remained 'EssenceUncommitted'.
  472     --   This is a structural inconsistency, not a content mismatch.
  473   deriving stock (Eq, Show, Generic)
  474   deriving anyclass (NFData, ToJSON, FromJSON)

  478 renderEssenceViolation = \case

  533 -- | Validate a reconciled 'Plan' against a committed essence.
  534 -- Priority order: family → tone → style; first mismatch wins.
  535 validatePlan :: EssenceCommitment -> Plan -> Either EssenceViolation Plan
  536 validatePlan c p
  537   | not (planFamily        p `Set.member` admissibleFamilies (ecMode c))
  538       = Left (ViolationFamilyMismatch (ecMode c) (planFamily p))
  539   | not (planNarrativeTone p `Set.member` admissibleTones    (ecMode c))
  540       = Left (ViolationToneMismatch  (ecMode c) (planNarrativeTone p))
  541   | not (planRenderStyle   p `Set.member` admissibleStyles   (ecMode c))
  542       = Left (ViolationStyleMismatch (ecMode c) (planRenderStyle p))
  543   | otherwise = Right p
