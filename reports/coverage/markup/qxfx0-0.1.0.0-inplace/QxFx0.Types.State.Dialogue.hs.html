   83 appendHistoryBounded :: Int -> Seq Text -> Text -> Seq Text
   84 appendHistoryBounded maxLen hist item =
   85   let new = hist Seq.|> item
   86   in if Seq.length new > maxLen then Seq.drop (Seq.length new - maxLen) new else new
