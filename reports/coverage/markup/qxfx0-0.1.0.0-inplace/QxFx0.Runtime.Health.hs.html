   81 checkHealth :: RuntimeContext -> IO SystemHealth
   82 checkHealth ctx = do
   83   let dbPath = rcDbPath ctx
   84   readiness <- assessResourceReadiness dbPath
   85   runtimeMode <- resolveRuntimeMode
   86   backendResult <- tryIO (wireRuntimeReadiness ctx)
   87   let backend = either (const backendProbeFailedReadiness) id backendResult
   88   mkSystemHealth runtimeMode dbPath readiness backend
   89 
   90 probeRuntimeReadiness :: IO SystemHealth
   91 probeRuntimeReadiness = do
   92   dbPath <- resolveDbPath
   93   readiness <- assessResourceReadiness dbPath
   94   backendResult <- tryIO probeBackendReadiness
   95   let backend = either (const backendProbeFailedReadiness) id backendResult
   96   runtimeMode <- resolveRuntimeMode
   97   mkSystemHealth runtimeMode dbPath readiness backend
   98 
   99 backendProbeFailedReadiness :: BackendReadiness
  100 backendProbeFailedReadiness = BackendReadiness
  101   { brEmbeddingAlive = False
  102   , brEmbeddingOperational = False
  103   , brEmbeddingExplicit = False
  104   , brEmbeddingBackend = "unknown"
  105   , brEmbeddingQuality = "unknown"
  106   , brNixOperational = False
  107   , brNixIssues = ["backend_probe_failed"]
  108   , brDatalogReady = False
  109   , brDatalogIssues = ["backend_probe_failed"]
  110   , brAgdaStatus = AgdaInvalid
  111   , brAgdaWitnessPath = ""
  112   , brAgdaIssues = ["backend_probe_failed"]
  113   }

  115 mkSystemHealth :: RuntimeMode -> FilePath -> ReadinessStatus -> BackendReadiness -> IO SystemHealth
  116 mkSystemHealth runtimeMode dbPath readiness backend = do
  117   morphBackend <- resolveMorphBackend
  118   let readinessMode = computeReadinessMode readiness
  119       componentOk rc = maybe False (\(_, ok, _) -> ok) (find (\(c, _, _) -> c == rc) (rsComponents readiness))
  120       morpOk = componentOk RcMorphology
  121       nixPolicyPresent = componentOk RcNixPolicy
  122       nixOk = nixPolicyPresent && brNixOperational backend
  123       datalogOk = componentOk RcDatalogRules && brDatalogReady backend
  124       agdaSpecOk = componentOk RcAgdaSpec
  125       agdaStatus = brAgdaStatus backend
  126   dbHealth <- inspectDatabaseHealth dbPath
  127   contractResult <- inspectSchemaContract dbPath
  128   let dbReady = dhReady dbHealth && componentOk RcDatabase
  129       schemaOk = isSchemaContractOk contractResult
  130       schemaVersion = schemaContractVersion contractResult
  131       schemaReason = renderSchemaContractResult contractResult
  132       agdaOk = agdaSpecOk && agdaVerificationReady agdaStatus
  133       strictBackendRequired = isStrictRuntimeMode runtimeMode
  134       strictReadinessOk = case readinessMode of
  135         NotReady _ -> False
  136         Degraded _ -> not strictBackendRequired
  137         Ready -> True
  138       embedStrictReady = brEmbeddingAlive backend
  139       backendOk = embedStrictReady && agdaOk && datalogOk && nixOk
  140       morphBackendLocal = morphBackend == MorphBackendLocal
  141       morphBackendText =
  142         case morphBackend of
  143           MorphBackendLocal -> "local"
  144           MorphBackendRemote -> "remote"
  145       decisionPathLocalOnly = brEmbeddingBackend backend == "local_deterministic" && morphBackendLocal
  146       networkOptionalOnly = decisionPathLocalOnly
  147       nixIssues =
  148         (if nixPolicyPresent then [] else ["nix_policy_missing"])
  149           ++ brNixIssues backend
  150       readinessText = case readinessMode of
  151         Ready -> "ready"
  152         Degraded xs -> "degraded:" <> T.intercalate "," (map (T.pack . show) xs)
  153         NotReady xs -> "not_ready:" <> T.intercalate "," (map (T.pack . show) xs)
  154       strictDecisionPathOk = not strictBackendRequired || decisionPathLocalOnly
  155       ready = strictReadinessOk && dbReady && schemaOk && (not strictBackendRequired || backendOk) && strictDecisionPathOk
  156       degraded = ready && (readinessMode /= Ready || not embedStrictReady || not agdaOk || not datalogOk || not nixOk)
  157       status
  158         | not ready = "not_ready"
  159         | degraded = "degraded"
  160         | otherwise = "ok"
  161   pure SystemHealth
  162     { shStatus = status
  163     , shRuntimeMode = runtimeModeText runtimeMode
  164     , shReady = ready
  165     , shDbAlive = dhAlive dbHealth
  166     , shDbBootstrapable = dhBootstrapable dbHealth
  167     , shMorphoReady = morpOk
  168     , shNixPolicyPresent = nixPolicyPresent
  169     , shNixReady = nixOk
  170     , shNixIssues = nixIssues
  171     , shEmbeddingAlive = embedStrictReady
  172     , shEmbeddingOperational = brEmbeddingOperational backend
  173     , shEmbeddingExplicit = brEmbeddingExplicit backend
  174     , shEmbeddingBackend = brEmbeddingBackend backend
  175     , shEmbeddingQuality = brEmbeddingQuality backend
  176     , shMorphBackend = morphBackendText
  177     , shMorphBackendLocal = morphBackendLocal
  178     , shDecisionPathLocalOnly = decisionPathLocalOnly
  179     , shNetworkOptionalOnly = networkOptionalOnly
  180     , shLlmDecisionPath = not decisionPathLocalOnly
  181     , shAgdaReady = agdaOk
  182     , shAgdaStatus = agdaStatus
  183     , shAgdaWitnessPath = brAgdaWitnessPath backend
  184     , shAgdaIssues = brAgdaIssues backend
  185     , shDatalogReady = datalogOk
  186     , shDatalogIssues = brDatalogIssues backend
  187     , shSchemaOk = schemaOk
  188     , shSchemaVersion = schemaVersion
  189     , shSchemaReason = schemaReason
  190     , shReadinessMode = readinessText
  191     }

  193 isSchemaContractOk :: SchemaContractResult -> Bool
  194 isSchemaContractOk (SchemaContractOk _) = True
  195 isSchemaContractOk SchemaContractFreshBootstrapable = True
  196 isSchemaContractOk _ = False
  197 
  198 schemaContractVersion :: SchemaContractResult -> Int
  199 schemaContractVersion (SchemaContractOk v) = v
  200 schemaContractVersion SchemaContractFreshBootstrapable = 0
  201 schemaContractVersion (SchemaContractVersionBehind _ actual) = actual
  202 schemaContractVersion _ = 0
  203 
  204 inspectSchemaContract :: FilePath -> IO SchemaContractResult
  205 inspectSchemaContract dbPath = do
  206   dbExists <- doesFileExist dbPath
  207   if not dbExists
  208     then pure SchemaContractFreshBootstrapable
  209     else do
  210       result <- NSQL.withDatabase dbPath $ \db -> checkSchemaContract db
  211       pure (either (\err -> SchemaContractQueryFailed (T.pack (show err))) id result)

  213 data DatabaseHealth = DatabaseHealth
  214   { dhAlive         :: !Bool
  215   , dhBootstrapable :: !Bool
  216   , dhReady         :: !Bool
  217   }
  218 
  219 inspectDatabaseHealth :: FilePath -> IO DatabaseHealth
  220 inspectDatabaseHealth dbPath = do
  221   dbExists <- doesFileExist dbPath
  222   dbDirOk <- doesDirectoryExist (takeDirectory dbPath)
  223   dbAlive <-
  224     if dbExists
  225       then checkExistingDatabase dbPath
  226       else pure False
  227   pure DatabaseHealth
  228     { dhAlive = dbAlive
  229     , dhBootstrapable = dbDirOk
  230     , dhReady = if dbExists then dbAlive else dbDirOk
  231     }
  232 
  233 checkExistingDatabase :: FilePath -> IO Bool
  234 checkExistingDatabase dbPath = do
  235   result <- NSQL.withDatabase dbPath $ \db -> do
  236     NSQL.withStatement db "SELECT 1" $ \stmt -> do
  237       mStepResult <- NSQL.step stmt
  238       case mStepResult of
  239         Right rc | rc == 100 -> do
  240           _ <- NSQL.columnInt stmt 0
  241           pure True
  242         _ -> pure False
  243   case result of
  244     Left _   -> pure False
  245     Right (Left _) -> pure False
  246     Right (Right alive) -> pure alive
