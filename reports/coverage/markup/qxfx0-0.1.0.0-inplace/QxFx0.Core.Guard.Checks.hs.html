   31 postRenderSafetyCheck :: Text -> [Text] -> SafetyStatus
   32 postRenderSafetyCheck rendered history =
   33   postRenderSafetyCheckSurface
   34     GuardSurface
   35       { gsRenderedText = rendered
   36       , gsSegments = [RenderSegment SegmentLocalRecovery rendered]
   37       , gsQuestionLike = T.any (== '?') rendered
   38       }
   39     history
   40 
   41 postRenderSafetyCheckSurface :: GuardSurface -> [Text] -> SafetyStatus
   42 postRenderSafetyCheckSurface surface history =
   43   foldl chooseMoreSevere InvariantOK
   44     [ checkEmptyRealization rendered
   45     , checkMetadataLeak surface
   46     , checkStuckRealization rendered history
   47     , checkIdentityDrift rendered
   48     , checkToxicity rendered
   49     , checkLength rendered
   50     , checkInputInjection surface
   51     ]
   52   where
   53     rendered = gsRenderedText surface
   54 
   55 checkEmptyRealization :: Text -> SafetyStatus
   56 checkEmptyRealization text =
   57   let trimmed = T.strip text
   58       isEmpty = T.null trimmed || trimmed == "..." || trimmed == "?"
   59    in if isEmpty
   60         then InvariantBlock guardEmptyResponse
   61         else InvariantOK
   62 
   63 checkMetadataLeak :: GuardSurface -> SafetyStatus
   64 checkMetadataLeak surface =
   65   let leakPatterns = ["{topic}", "{left}", "{right}", "{style}", "{move}", "{slot}"]
   66       found = findInUntrustedSegments leakPatterns surface
   67    in if null found
   68         then InvariantOK
   69         else InvariantBlock (guardMetadataLeak <> T.intercalate ", " found)
   70 
   71 checkStuckRealization :: Text -> [Text] -> SafetyStatus
   72 checkStuckRealization rendered history =
   73   let normalizedRender = normalizeClaimText rendered
   74       matchCount = length $ filter (\historic -> normalizeClaimText historic == normalizedRender) (take 5 history)
   75    in if matchCount >= 3
   76         then InvariantWarn guardStuckRepetition
   77         else InvariantOK
   78 
   79 checkIdentityDrift :: Text -> SafetyStatus
   80 checkIdentityDrift text =
   81   let lowerText = T.toLower text
   82       found = filter (`T.isInfixOf` lowerText) driftPatterns
   83    in if null found
   84         then InvariantOK
   85         else InvariantBlock (guardMarkupLeak <> ": " <> T.intercalate ", " found)
   86 
   87 checkToxicity :: Text -> SafetyStatus
   88 checkToxicity text =
   89   let tokens = tokenizeKeywordText text
   90       found = filter (containsKeywordPhrase tokens) toxicPatterns
   91    in if null found
   92         then InvariantOK
   93         else InvariantWarn (guardToxicPatterns <> T.intercalate ", " found)
   94 
   95 checkLength :: Text -> SafetyStatus
   96 checkLength text =
   97   let len = T.length text
   98    in if len > 5000
   99         then InvariantBlock guardTooLong
  100         else if len == 0
  101           then InvariantBlock guardEmptyResponse
  102           else InvariantOK
  103 
  104 checkInputInjection :: GuardSurface -> SafetyStatus
  105 checkInputInjection surface =
  106   let found = findInUntrustedSegments injectionPatterns surface
  107    in if null found
  108         then InvariantOK
  109         else InvariantBlock (guardSuspiciousPatterns <> T.intercalate ", " found)
  110 
  111 findInUntrustedSegments :: [Text] -> GuardSurface -> [Text]
  112 findInUntrustedSegments patterns surface =
  113   filter present patterns
  114   where
  115     untrustedTexts =
  116       [ T.toLower (rsText segment)
  117       | segment <- gsSegments surface
  118       , rsKind segment /= SegmentTemplate
  119       ]
  120     present pattern =
  121       let needle = T.toLower pattern
  122        in any (T.isInfixOf needle) untrustedTexts
  123 
  124 chooseMoreSevere :: SafetyStatus -> SafetyStatus -> SafetyStatus
  125 chooseMoreSevere left right
  126   | severity right > severity left = right
  127   | otherwise = left
  128 
  129 severity :: SafetyStatus -> Int
  130 severity InvariantOK = 0
  131 severity (InvariantWarn _) = 1
  132 severity (InvariantBlock _) = 2
