    7 {-|
    8 Module      : QxFx0.Learning.TrainingCycle
    9 Description : Phase 10 — Offline training cycle: extract traces,
   10   generate bounded calibration candidates, evaluate offline,
   11   promote non-regressing winners, rollback-ready.
   12 
   13 This module implements the first real training cycle for QxFx0.
   14 It is pure and offline: no live-turn state is mutated.
   15 The cycle:
   16 
   17 1. Extracts a training dataset from 'SystemState' history
   18    ('CalibrationSnapshot's, 'KnowledgeTree' health, need trends).
   19 2. Generates a bounded pool of 'CalibrationCandidate's by
   20    perturbing 'SalienceWeights' and 'FieldHeuristics'.
   21 3. Evaluates each candidate against the dataset using proxy
   22    metrics (conatus trend, uncertainty, repair-loop frequency,
   23    reject-rate deltas).
   24 4. Accepts only non-regressing candidates; the best is promoted
   25    with a 'CalibrationId' and rollback linkage.
   26 5. Emits a 'TrainingCycleOutcome' with full telemetry.
   27 
   28 All evaluation is fail-closed: any dubious candidate is rejected
   29 with a typed 'TrainingRejectReason'.
   30 -}

  103 -- | A single trace extracted from system-state history.
  104 data TrainingTrace = TrainingTrace
  105   { ttTurn            :: !Int
  106     -- ^ Turn number.
  107   , ttSignalComponents :: !SignalComponents
  108     -- ^ Four feature values that fed the signal at this turn.
  109   , ttSignal          :: !Double
  110     -- ^ Final clamped signal value.
  111   , ttDecision        :: !CalibrationDecision
  112     -- ^ Decision taken by the pipeline.
  113   , ttNeedLevel       :: !Double
  114     -- ^ Learning-need level at this turn.
  115   , ttTreeHealth      :: !Double
  116     -- ^ Knowledge-tree health proxy at this turn.
  117   , ttRepairLoopCount :: !Int
  118     -- ^ Repair-loop count in the recent window.
  119   } deriving stock (Eq, Show, Generic)
  120     deriving anyclass (NFData, FromJSON, ToJSON)

  122 -- | Normalised training dataset split into train/eval subsets.
  123 data TrainingDataset = TrainingDataset
  124   { tdTraces :: ![TrainingTrace]
  125     -- ^ All traces, chronological order (oldest first).
  126   , tdTrain  :: ![TrainingTrace]
  127     -- ^ Training subset (~70 % of traces).
  128   , tdEval   :: ![TrainingTrace]
  129     -- ^ Evaluation subset (~30 % of traces).
  130   , tdStats  :: !DatasetStats
  131     -- ^ Aggregated statistics.
  132   } deriving stock (Eq, Show, Generic)

  135 -- | Aggregate statistics for a dataset.
  136 data DatasetStats = DatasetStats
  137   { dsTotalTurns          :: !Int
  138   , dsAcceptedProposals   :: !Int
  139   , dsRejectedProposals   :: !Int
  140   , dsTransportErrors     :: !Int
  141   , dsFallbackHeavySessions :: !Int
  142     -- ^ Sessions where >50 % of turns used recovery or fallback.
  143   } deriving stock (Eq, Show, Generic)
  144     deriving anyclass (NFData, FromJSON, ToJSON)

  146 -- | Extract a 'TrainingDataset' from a 'SystemState'.
  147 --
  148 -- Uses 'ssCalibrationSnapshots' as the primary trace source.
  149 -- If fewer than 5 snapshots exist, returns an empty dataset
  150 -- (the training cycle needs minimal history to be meaningful).
  151 extractTrainingDataset :: SystemState -> TrainingDataset
  152 extractTrainingDataset ss =
  153   let snaps = ssCalibrationSnapshots ss
  154       needState = ssLearningNeedState ss
  155       tree = ssKnowledgeTree ss
  156       traces = catMaybes (map (snapshotToTrace needState tree) snaps)
  157       total = length traces
  158       trainEnd = (total * 7) `div` 10
  159       (train, eval) = splitAt trainEnd traces
  160       stats = computeDatasetStats traces
  161   in if total < 5
  162        then TrainingDataset [] [] [] (DatasetStats 0 0 0 0 0)
  163        else TrainingDataset traces train eval stats
  164 
  165 snapshotToTrace :: LearningNeedState -> KnowledgeTree -> CalibrationSnapshot -> Maybe TrainingTrace
  166 snapshotToTrace needState tree snap =
  167   let comps = csComponents snap
  168       health = branchHealthTrend tree
  169       -- Repair-loop proxy: count of recent high-need turns.
  170       recentLevels = map snd (take 5 (reverse (lnsHistory needState)))
  171       loopCount = length (filter (> 0.6) recentLevels)
  172   in Just TrainingTrace
  173        { ttTurn            = fromIntegral (T.length (csRunId snap))  -- proxy; real turn not in snapshot
  174        , ttSignalComponents = comps
  175        , ttSignal          = csSignal snap
  176        , ttDecision        = csDecision snap
  177        , ttNeedLevel       = 0.0  -- not directly available in snapshot; filled from needState below
  178        , ttTreeHealth      = health
  179        , ttRepairLoopCount = loopCount
  180        }
  181 
  182 -- | Override need levels from the need-state history when available.
  183 -- Snapshots don't carry turn numbers, so we zip by position.
  184 mergeNeedLevels :: LearningNeedState -> [TrainingTrace] -> [TrainingTrace]
  185 mergeNeedLevels needState traces =
  186   let hist = lnsHistory needState
  187       histLen = length hist
  188       go [] _ = []
  189       go ts [] = ts
  190       go (t : ts) ((_, lvl) : ls) = t { ttNeedLevel = lvl } : go ts ls
  191   in if null hist then traces else go traces (reverse hist)
  192 
  193 computeDatasetStats :: [TrainingTrace] -> DatasetStats
  194 computeDatasetStats traces =
  195   let total = length traces
  196       accepted = length (filter (\t -> ttDecision t == CdApplySignal) traces)
  197       rejected = length (filter (\t -> ttDecision t `elem` [CdHoldLowConfidence, CdHoldGuardrails]) traces)
  198       errors = length (filter (\t -> abs (ttSignal t) < 0.01 && ttDecision t == CdHoldNoNeed) traces)
  199       -- Fallback-heavy proxy: turns where signal was near-zero but decision was not CdHoldNoNeed
  200       fallback = length (filter (\t -> abs (ttSignal t) < 0.05 && ttDecision t /= CdHoldNoNeed) traces)
  201       fallbackSessions = if total > 0 && (fromIntegral fallback / fromIntegral total :: Double) > 0.5 then 1 else 0
  202   in DatasetStats total accepted rejected errors fallbackSessions

  215 -- | A calibration candidate with full provenance.
  216 data CalibrationCandidate = CalibrationCandidate
  217   { ccId               :: !CalibrationId
  218   , ccType             :: !CandidateType
  219   , ccSalienceWeights  :: !(Maybe SalienceWeights)
  220   , ccFieldHeuristics  :: !(Maybe FieldHeuristics)
  221   , ccSourceRunId      :: !Text
  222   , ccGenerationSignal :: !Double
  223     -- ^ The adaptation signal used to produce this candidate.
  224   , ccTimestamp        :: !UTCTime
  225   } deriving stock (Eq, Show, Generic)
  226     deriving anyclass (NFData, FromJSON, ToJSON)

  228 -- | Default perturbation signals.  Seven values give 14 candidates
  229 -- total (7 salience + 7 field), well inside the 10–30 bound.
  230 defaultTrainingSignals :: [Double]
  231 defaultTrainingSignals = [-0.30, -0.20, -0.10, 0.10, 0.20, 0.30]
  232 
  233 -- | Generate a bounded pool of calibration candidates.
  234 --
  235 -- Each signal value produces one 'SalienceWeights' candidate and one
  236 -- 'FieldHeuristics' candidate by applying the existing bounded
  237 -- adaptation functions 'adaptSalienceWeights' / 'adaptFieldHeuristics'.
  238 generateCandidates
  239   :: CalibrationId     -- ^ starting version id
  240   -> [Double]          -- ^ perturbation signals
  241   -> Text              -- ^ source run id
  242   -> UTCTime           -- ^ generation timestamp
  243   -> [CalibrationCandidate]
  244 generateCandidates startId signals runId tstamp =
  245   let salienceBase = defaultSalienceWeights
  246       fieldBase  = defaultFieldHeuristics
  247       go _ [] = []
  248       go cid (s : ss) =
  249         mkSalience cid s : mkField (CalibrationId (unCalibrationId cid + 1)) s
  250         : go (CalibrationId (unCalibrationId cid + 2)) ss
  251       mkSalience cid s = CalibrationCandidate
  252         { ccId = cid
  253         , ccType = CandidateSalience
  254         , ccSalienceWeights = Just (adaptSalienceWeights s salienceBase)
  255         , ccFieldHeuristics = Nothing
  256         , ccSourceRunId = runId
  257         , ccGenerationSignal = s
  258         , ccTimestamp = tstamp
  259         }
  260       mkField cid s = CalibrationCandidate
  261         { ccId = cid
  262         , ccType = CandidateField
  263         , ccSalienceWeights = Nothing
  264         , ccFieldHeuristics = Just (adaptFieldHeuristics s fieldBase)
  265         , ccSourceRunId = runId
  266         , ccGenerationSignal = s
  267         , ccTimestamp = tstamp
  268         }
  269   in go startId signals

  282 -- | Typed reject reason for telemetry and audit.
  283 data TrainingRejectReason
  284   = TrRegressionConatus
  285     -- ^ Projected conatus trend worse than baseline.
  286   | TrRegressionUncertainty
  287     -- ^ Projected uncertainty (oscillation) higher than baseline.
  288   | TrRegressionRepair
  289     -- ^ Estimated repair-loop frequency would increase.
  290   | TrRegressionRejectRate
  291     -- ^ Estimated reject rate would increase.
  292   | TrUnstableVariance
  293     -- ^ Metrics show high variance across the eval window.
  294   | TrInsufficientSignal
  295     -- ^ Dataset too small or too weak to justify adaptation.
  296   deriving stock (Eq, Show, Generic)
  297     deriving anyclass (NFData, FromJSON, ToJSON)

  299 -- | Proxy metrics comparing candidate vs baseline on the eval subset.
  300 data EvaluationMetrics = EvaluationMetrics
  301   { emConatusTrendDelta    :: !Double
  302     -- ^ Negative = candidate improves conatus trend (need rising less fast).
  303   , emUncertaintyDelta     :: !Double
  304     -- ^ Negative = candidate reduces need-level oscillation.
  305   , emRepairLoopFreqDelta  :: !Double
  306     -- ^ Negative = candidate reduces estimated repair-loop frequency.
  307   , emRejectRateDelta      :: !Double
  308     -- ^ Negative = candidate reduces estimated calibration reject rate.
  309   , emNetScore             :: !Double
  310     -- ^ Weighted composite: higher = better.
  311   } deriving stock (Eq, Show, Generic)
  312     deriving anyclass (NFData, FromJSON, ToJSON)

  314 -- | Full evaluation result for one candidate.
  315 data CandidateEvaluation = CandidateEvaluation
  316   { ceCandidate    :: !CalibrationCandidate
  317   , ceMetrics      :: !EvaluationMetrics
  318   , ceVerdict      :: !CandidateVerdict
  319   , ceRejectReason :: !(Maybe TrainingRejectReason)
  320   } deriving stock (Eq, Show, Generic)
  321     deriving anyclass (NFData, FromJSON, ToJSON)

  323 -- | Evaluate a single candidate against a 'TrainingDataset'.
  324 --
  325 -- This is a lightweight proxy evaluation: it does NOT replay full
  326 -- turns. Instead it projects the candidate's parameter changes onto
  327 -- the historical 'TrainingTrace' window and computes delta metrics.
  328 evaluateCandidate :: TrainingDataset -> CalibrationCandidate -> CandidateEvaluation
  329 evaluateCandidate dataset candidate =
  330   let evalTraces = tdEval dataset
  331       baselineMetrics = computeBaselineMetrics evalTraces
  332       candidateMetrics = computeCandidateMetrics candidate evalTraces
  333       deltas = computeDeltas baselineMetrics candidateMetrics
  334       -- Acceptance policy (fail-closed)
  335       (verdict, mReason) = applyAcceptancePolicy deltas evalTraces
  336   in CandidateEvaluation
  337        { ceCandidate = candidate
  338        , ceMetrics = deltas
  339        , ceVerdict = verdict
  340        , ceRejectReason = mReason
  341        }
  342 
  343 -- | Baseline metrics from the eval subset using current defaults.
  344 computeBaselineMetrics :: [TrainingTrace] -> (Double, Double, Double, Double)
  345 computeBaselineMetrics traces =
  346   let conatusTrend = slopeOfNeedLevels traces
  347       uncertainty  = oscillationOfNeedLevels traces
  348       repairFreq   = repairLoopProxy traces
  349       rejectRate   = rejectRateProxy traces
  350   in (conatusTrend, uncertainty, repairFreq, rejectRate)
  351 
  352 computeCandidateMetrics :: CalibrationCandidate -> [TrainingTrace] -> (Double, Double, Double, Double)
  353 computeCandidateMetrics candidate traces =
  354   -- Proxy: apply candidate's conservatism factor to historical traces
  355   -- and recompute the metrics.
  356   let factor = candidateConservatism candidate
  357       -- Adjust each trace's need level by the conservatism factor
  358       -- (conservative candidate damps rising needs, amplifies falling needs)
  359       adjusted = map (adjustTrace factor) traces
  360   in computeBaselineMetrics adjusted
  361 
  362 -- | Conservatism factor: > 0 means more conservative, < 0 means more aggressive.
  363 candidateConservatism :: CalibrationCandidate -> Double
  364 candidateConservatism c = case ccType c of
  365   CandidateSalience ->
  366     case ccSalienceWeights c of
  367       Nothing -> 0.0
  368       Just w  ->
  369         -- Higher field-confidence weight and lower counterfactual = more conservative
  370         (weightFieldConfidence w - weightFieldConfidence defaultSalienceWeights)
  371         - (weightCounterfactual w - weightCounterfactual defaultSalienceWeights)
  372   CandidateField ->
  373     case ccFieldHeuristics c of
  374       Nothing -> 0.0
  375       Just fh ->
  376         -- Lower streak boost = more conservative = positive factor
  377         let defBoost = fhHolisticStreakBoostRate defaultFieldHeuristics
  378             candBoost = fhHolisticStreakBoostRate fh
  379         in defBoost - candBoost
  380 
  381 -- | Adjust a trace's need level by the candidate's conservatism factor.
  382 -- Conservative candidates damp positive trends and amplify negative trends.
  383 adjustTrace :: Double -> TrainingTrace -> TrainingTrace
  384 adjustTrace factor t =
  385   let lvl = ttNeedLevel t
  386       -- Damp the level toward stability based on conservatism
  387       adjusted = if lvl > 0.5
  388                    then max 0.0 (lvl - factor * 0.1)  -- conservative: reduce high need
  389                    else min 1.0 (lvl + factor * 0.05) -- conservative: boost low need slightly
  390   in t { ttNeedLevel = adjusted }
  391 
  392 -- | Compute metric deltas: candidate minus baseline.
  393 -- Negative delta = improvement (candidate is better).
  394 computeDeltas :: (Double, Double, Double, Double) -> (Double, Double, Double, Double) -> EvaluationMetrics
  395 computeDeltas (bCon, bUnc, bRep, bRej) (cCon, cUnc, cRep, cRej) =
  396   let dCon = cCon - bCon
  397       dUnc = cUnc - bUnc
  398       dRep = cRep - bRep
  399       dRej = cRej - bRej
  400       -- Net score: weighted composite (higher = better)
  401       net = negate (0.35 * dCon + 0.25 * dUnc + 0.25 * dRep + 0.15 * dRej)
  402   in EvaluationMetrics dCon dUnc dRep dRej net
  403 
  404 -- | Acceptance policy (fail-closed).
  405 applyAcceptancePolicy :: EvaluationMetrics -> [TrainingTrace] -> (CandidateVerdict, Maybe TrainingRejectReason)
  406 applyAcceptancePolicy metrics traces
  407   -- Insufficient signal: need at least 3 eval traces.
  408   | length traces < 3 =
  409       (CandidateReject, Just TrInsufficientSignal)
  410   -- Strict non-regression: conatus must not regress.
  411   | emConatusTrendDelta metrics > 0.05 =
  412       (CandidateReject, Just TrRegressionConatus)
  413   -- Uncertainty must not increase significantly.
  414   | emUncertaintyDelta metrics > 0.05 =
  415       (CandidateReject, Just TrRegressionUncertainty)
  416   -- Repair-loop frequency must not increase.
  417   | emRepairLoopFreqDelta metrics > 0.05 =
  418       (CandidateReject, Just TrRegressionRepair)
  419   -- Reject rate must not increase.
  420   | emRejectRateDelta metrics > 0.05 =
  421       (CandidateReject, Just TrRegressionRejectRate)
  422   -- Unstable variance: if all deltas are near-zero but net score is also near-zero,
  423   -- the candidate is effectively a no-op with noise.
  424   | abs (emNetScore metrics) < 0.001 =
  425       (CandidateReject, Just TrInsufficientSignal)
  426   -- Otherwise accept.
  427   | otherwise =
  428       (CandidateAccept, Nothing)
  429 
  430 -- | Slope of need levels over the trace window (last 3 points).
  431 slopeOfNeedLevels :: [TrainingTrace] -> Double
  432 slopeOfNeedLevels traces =
  433   let levels = map ttNeedLevel traces
  434       recent = take 3 (reverse levels)
  435   in case recent of
  436        (y2 : y1 : y0 : _) -> ((y2 - y1) + (y1 - y0)) / 2.0
  437        _ -> 0.0
  438 
  439 -- | Oscillation amplitude (max - min) over the trace window.
  440 oscillationOfNeedLevels :: [TrainingTrace] -> Double
  441 oscillationOfNeedLevels traces =
  442   let levels = map ttNeedLevel traces
  443   in if null levels then 0.0 else maximum levels - minimum levels
  444 
  445 -- | Proxy for repair-loop frequency: fraction of traces with repair-loop count > 0.
  446 repairLoopProxy :: [TrainingTrace] -> Double
  447 repairLoopProxy traces =
  448   let total = fromIntegral (max 1 (length traces))
  449       loops = fromIntegral (length (filter (\t -> ttRepairLoopCount t > 0) traces))
  450   in loops / total
  451 
  452 -- | Proxy for reject rate: fraction of traces with non-ApplySignal decisions.
  453 rejectRateProxy :: [TrainingTrace] -> Double
  454 rejectRateProxy traces =
  455   let total = fromIntegral (max 1 (length traces))
  456       rejects = fromIntegral (length (filter (\t -> ttDecision t /= CdApplySignal) traces))
  457   in rejects / total

  459 -- | Evaluate all candidates, returning them sorted by net score (best first).
  460 evaluateAllCandidates :: TrainingDataset -> [CalibrationCandidate] -> [CandidateEvaluation]
  461 evaluateAllCandidates dataset candidates =
  462   let evals = map (evaluateCandidate dataset) candidates
  463       accepted = filter (\e -> ceVerdict e == CandidateAccept) evals
  464   in sortOn (negate . emNetScore . ceMetrics) accepted ++ filter (\e -> ceVerdict e == CandidateReject) evals

  488 -- | Outcome of a complete offline training cycle.
  489 data TrainingCycleOutcome = TrainingCycleOutcome
  490   { tcoCycleId           :: !Text
  491   , tcoDatasetStats      :: !DatasetStats
  492   , tcoCandidates        :: ![CandidateEvaluation]
  493     -- ^ All evaluated candidates, best first.
  494   , tcoPromotedCandidate :: !(Maybe CalibrationCandidate)
  495     -- ^ The single best accepted candidate, if any.
  496   , tcoPreviousVersion   :: !(Maybe CalibrationId)
  497     -- ^ Version ID that was current before promotion.
  498   , tcoRollbackEntry     :: !(Maybe CalibrationEntry)

  503 -- | Run the full offline training cycle.
  504 --
  505 -- Pure function: takes 'SystemState' (for history) and config,
  506 -- returns 'TrainingCycleOutcome'.  Does NOT mutate any live state.
  507 runTrainingCycle :: SystemState -> TrainingCycleConfig -> UTCTime -> TrainingCycleOutcome
  508 runTrainingCycle ss cfg tstamp =
  509   let dataset = extractTrainingDataset ss
  510       candidates = generateCandidates (tccStartVersion cfg) (tccSignals cfg) (tccCycleId cfg) tstamp
  511       evaluations = evaluateAllCandidates dataset candidates
  512       promoted = case filter (\e -> ceVerdict e == CandidateAccept) evaluations of
  513                    (best : _) -> Just (ceCandidate best)
  514                    _          -> Nothing
  515   in TrainingCycleOutcome
  516        { tcoCycleId           = tccCycleId cfg
  517        , tcoDatasetStats      = tdStats dataset
  518        , tcoCandidates        = evaluations
  519        , tcoPromotedCandidate = promoted
  520        , tcoPreviousVersion   = Just (tccStartVersion cfg)
  521        , tcoRollbackEntry   = Nothing
  522        }

  524 -- | Promote an accepted candidate into the calibration log.
  525 -- Returns the new calibration entry and the next version ID.
  526 promoteCandidate
  527   :: CalibrationId        -- ^ next available version id
  528   -> CalibrationCandidate
  529   -> Int                  -- ^ current turn
  530   -> Maybe CalibrationId  -- ^ previous version for rollback
  531   -> Either Text (CalibrationEntry, CalibrationId)
  532 promoteCandidate nextId cand turn prevId =
  533   let proposal = case ccType cand of
  534         CandidateSalience ->
  535           case ccSalienceWeights cand of
  536             Just w  -> Right (ProposalSalienceWeights w)
  537             Nothing -> Left "training-cycle promotion missing salience weights"
  538         CandidateField ->
  539           case ccFieldHeuristics cand of
  540             Just fh -> Right (ProposalFieldHeuristics fh)
  541             Nothing -> Left "training-cycle promotion missing field heuristics"
  542   in fmap (\resolvedProposal -> acceptProposal nextId resolvedProposal turn prevId) proposal
  543 
  544 -- | Rollback a promoted training-cycle candidate.
  545 -- Returns the rolled-back entry and the previous version ID that is
  546 -- now current, if the promoted entry had a 'prevId'.
  547 rollbackTrainingCycle :: CalibrationEntry -> Int -> Maybe (CalibrationEntry, CalibrationId)
  548 rollbackTrainingCycle entry turn =
  549   case cePrevId entry of
  550     Nothing -> Nothing
  551     Just prevId ->
  552       let rolled = entry { ceStatus = RolledBack, ceDecidedTurn = Just turn }
  553       in Just (rolled, prevId)
