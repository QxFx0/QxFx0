   58 -- | Emit a stderr trace line with counts of persisted learning artefacts.
   59 logPersistenceCounts :: String -> SystemState -> IO ()
   60 logPersistenceCounts label ss = do
   61   let tree = ssKnowledgeTree ss
   62       branchesFruits = sum (map length (M.elems (ktBranches tree)))
   63       quarantineFruits = length (ktQuarantine tree)
   64       morph = ssMorphology ss
   65       calibCount = length (unCalibrationLog (ssCalibrationLog ss))
   66       guardQuarantine = length (gsQuarantine (ssGuardrailState ss))
   67   hPutStrLn stderr $ concat
   68     [ "[persistence_trace] ", label
   69     , " session_id=", T.unpack (ssSessionId ss)
   70     , " turn_count=", show (ssTurnCount ss)
   71     , " ktree_branches_fruits=", show branchesFruits
   72     , " ktree_quarantine=", show quarantineFruits
   73     , " ktree_grafted=", show (ktGraftedCount tree)
   74     , " ktree_pruned=", show (ktPrunedCount tree)
   75     , " morph_prep=", show (M.size (mdPrepositional morph))
   76     , " morph_gen=", show (M.size (mdGenitive morph))
   77     , " morph_nom=", show (M.size (mdNominative morph))
   78     , " morph_surface=", show (M.size (mdFormsBySurface morph))
   79     , " calib_log=", show calibCount
   80     , " guard_quarantine=", show guardQuarantine
   81     ]

   86 saveStateWithProjection :: DbRunner -> SystemState -> Text -> Maybe TurnProjection -> IO (Either PersistenceDiagnostic SystemState)
   87 saveStateWithProjection withDb ss sessionId mProjection = do
   88   logPersistenceCounts "pre_save" ss
   89   let persistenceSnapshot = preparePersistenceSnapshot sessionId ss
   90       persistedState = spsCanonicalState persistenceSnapshot
   91       runtimeState = spsRuntimeContinuation persistenceSnapshot
   92   result <- tryQxFx0 $ withDb $ \db -> do
   93     withImmediateTransaction db $ do
   94       touchRuntimeSessionActivity db sessionId
   95       persistSystemBlob db sessionId persistedState
   96       persistTurnArtifacts db sessionId mProjection
   97 
   98       pure runtimeState
   99   case result of
  100     Left (PersistenceTxError stage msg) -> do
  101       hPutStrLn stderr $ "[persistence_debug] save_tx_error session=" <> T.unpack sessionId <> " stage=" <> show stage <> " detail=" <> T.unpack msg
  102       pure (Left (diagnoseSave stage (Just msg)))
  103     Left other -> do
  104       hPutStrLn stderr $ "[persistence_debug] save_unknown_qxfx0_exception session=" <> T.unpack sessionId <> " detail=" <> T.unpack (renderQxFx0ExceptionForLog other)
  105       pure (Left (PdSaveFailed StageUnknown Nothing (Just (renderQxFx0ExceptionForLog other))))
  106     Right savedSs -> pure $ Right savedSs

  108 rollbackTurnProjections :: DbRunner -> Text -> Int -> IO (Either PersistenceDiagnostic ())
  109 rollbackTurnProjections withDb sessionId stableTurn = do
  110   result <- tryQxFx0 $ withDb $ \db -> do
  111     withImmediateTransaction db $ do
  112       deleteTurnQualityAbove db sessionId stableTurn
  113       deleteShadowDivergenceAbove db sessionId stableTurn
  114   case result of
  115     Left (PersistenceTxError stage msg) -> pure (Left (diagnoseRollback stage (Just msg)))
  116     Left other -> pure (Left (PdRollbackFailed StageUnknown Nothing (Just (renderQxFx0ExceptionForLog other))))
  117     Right () -> pure (Right ())

  119 withImmediateTransaction :: NSQL.Database -> IO a -> IO a
  120 withImmediateTransaction db action = mask $ \restore -> do
  121   beginResult <- NSQL.execSql db "BEGIN IMMEDIATE;"
  122   case beginResult of
  123     Left err ->
  124       throwQxFx0 (PersistenceTxError StageTxBegin ("tx_begin_failed: " <> err))
  125     Right _ ->
  126       pure ()
  127   result <- restore action `onException` rollbackBestEffort db
  128   commitResult <- NSQL.execSql db "COMMIT;"
  129   case commitResult of
  130     Right _ ->
  131       pure result
  132     Left err -> do
  133       rollbackResult <- NSQL.execSql db "ROLLBACK;"
  134       case rollbackResult of
  135         Right _ ->
  136           throwQxFx0 (PersistenceTxError StageTxCommit ("tx_commit_failed: " <> err))
  137         Left rbErr ->
  138           throwQxFx0
  139             (PersistenceTxError StageTxCommit ("tx_commit_and_rollback_failed: commit=" <> err <> " rollback=" <> rbErr))
  140 
  141 rollbackBestEffort :: NSQL.Database -> IO ()
  142 rollbackBestEffort db = do
  143   _ <- NSQL.execSql db "ROLLBACK;"
  144   pure ()

  162 loadState :: DbRunner -> Text -> IO LoadStateResult
  163 loadState withDb sessionId = withDb $ \db -> do
  164   mBlobResult <- try (loadKV db sessionId "__system_state__") :: IO (Either UnicodeException (Maybe Text))
  165   case mBlobResult of
  166     Left err -> do
  167       hPutStrLn stderr $ "[persistence_debug] load_unicode_exception session=" <> T.unpack sessionId <> " detail=" <> show err
  168       pure (LoadStateCorrupt [PdCorruptDecode])
  169     Right mBlob ->
  170       case mBlob of
  171         Just blob ->
  172           case Aeson.eitherDecodeStrict (TE.encodeUtf8 blob) of
  173             Right ss ->
  174               if persistedTruthIsAuthoritative (ssTruthContractStatus ss)
  175                 then do
  176                   case rebuildGovernedSystemState ss of
  177                     Right rebuilt -> do
  178                       let restored = runtimeContinuationState sessionId rebuilt
  179                       logPersistenceCounts "post_load" restored
  180                       pure (LoadStateRestored restored)
  181                     Left err -> do
  182                       hPutStrLn stderr $ "[persistence_debug] governance_rebuild_failed session=" <> T.unpack sessionId <> " detail=" <> T.unpack err
  183                       pure (LoadStateCorrupt [PdCorruptDecode, PdSchemaMissingFields ["governance_rebuild_failed:" <> err]])
  184                 else do
  185                   hPutStrLn stderr $ "[persistence_debug] non_authoritative_persisted_state session=" <> T.unpack sessionId
  186                   pure (LoadStateCorrupt [PdCorruptDecode, PdSchemaMissingFields ["non_authoritative_persisted_state"]])
  187             Left decodeErr -> do
  188               hPutStrLn stderr $ "[persistence_debug] decode_failed session=" <> T.unpack sessionId <> " detail=" <> decodeErr
  189               pure (LoadStateCorrupt (PdCorruptDecode : stateBlobDiagnostics blob))
  190         Nothing -> pure LoadStateMissing
  191 
  192 stateBlobDiagnostics :: Text -> [PersistenceDiagnostic]
  193 stateBlobDiagnostics blob =
  194   case Aeson.decode (BL.fromStrict (TE.encodeUtf8 blob)) :: Maybe Aeson.Object of
  195     Nothing -> []
  196     Just obj ->
  197       let optionalSystemFields =
  198             [ "lastGuardReport"
  199             , "dreamState"
  200             , "intuitionState"
  201             , "semanticAnchor"
  202             , "lastTurnDecision"
  203             ]
  204           missing = filter (\k -> not (KM.member (AK.fromText k) obj)) optionalSystemFields
  205       in if null missing
  206            then []
  207            else [PdSchemaMissingFields missing]

  209 loadKV :: NSQL.Database -> Text -> Text -> IO (Maybe Text)
  210 loadKV db sessionId k = do
  211   let sql = "SELECT value FROM dialogue_state WHERE session_id = ? AND key = ?"
  212   withPreparedStatement db sql ("loadKV key=" <> k <> ", session=" <> sessionId) $ \stmt -> do
  213     _ <- NSQL.bindText stmt 1 sessionId
  214     _ <- NSQL.bindText stmt 2 k
  215     hasRow <- NSQL.stepRow stmt
  216     if hasRow
  217       then do
  218         value <- NSQL.columnText stmt 0
  219         _ <- evaluate (T.length value)
  220         pure (Just value)
  221       else pure Nothing

  223 persistTurnQuality :: NSQL.Database -> Text -> TurnProjection -> IO ()
  224 persistTurnQuality db sessionId p = do
  225   let sql = "INSERT OR REPLACE INTO turn_quality(session_id, turn, parser_mode, parser_confidence, parser_errors, planner_mode, planner_decision, atom_register, atom_load, scene_pressure, scene_request, scene_stance, render_lane, render_style, legitimacy_status, legitimacy_reason, warranted_mode, decision_disposition, owner_family, owner_force, shadow_status, shadow_snapshot_id, shadow_divergence_kind, shadow_family, shadow_force, shadow_message, replay_trace_json, divergence) VALUES(?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)"
  226       replayTrace0 = tqpReplayTrace p
  227       replayAuthority = fromMaybe AuthorityLegacyIncomplete (trcAuthorityClass replayTrace0)
  228       replayTrace =
  229         replayTrace0
  230           { trcReplayProvenanceStatus =
  231               normalizeReplayProvenanceStatus (trcReplayProvenanceStatus replayTrace0) replayAuthority
  232           }
  233       replayTraceJson = TE.decodeUtf8 . BL.toStrict . Aeson.encode $ replayTrace
  234   ts <- prepareTx db "turn_quality" sql
  235   bindTextOrFail ts 1 sessionId
  236   bindInt64OrFail ts 2 (fromIntegral (tqpTurn p))
  237   bindTextOrFail ts 3 (parserModeText (tqpParserMode p))
  238   bindDoubleOrFail ts 4 (tqpParserConfidence p)
  239   bindTextOrFail ts 5 (T.intercalate "," (tqpParserErrors p))
  240   bindTextOrFail ts 6 (plannerModeText (tqpPlannerMode p))
  241   bindTextOrFail ts 7 (T.pack (show (tqpPlannerDecision p)))
  242   bindTextOrFail ts 8 (T.pack (show (tqpAtomRegister p)))
  243   bindDoubleOrFail ts 9 (tqpAtomLoad p)
  244   bindTextOrFail ts 10 (scenePressureText (tqpScenePressure p))
  245   bindTextOrFail ts 11 (tqpSceneRequest p)
  246   bindTextOrFail ts 12 (T.pack (show (tqpSceneStance p)))
  247   bindTextOrFail ts 13 (T.pack (show (tqpRenderLane p)))
  248   bindTextOrFail ts 14 (renderStyleText (tqpRenderStyle p))
  249   bindTextOrFail ts 15 (legitimacyStatusText (tqpLegitimacyStatus p))
  250   bindTextOrFail ts 16 (legitimacyReasonText (tqpLegitimacyReason p))
  251   bindTextOrFail ts 17 (T.pack (show (tqpWarrantedMode p)))
  252   bindTextOrFail ts 18 (decisionDispositionText (tqpDecisionDisposition p))
  253   bindTextOrFail ts 19 (T.pack (show (tqpOwnerFamily p)))
  254   bindTextOrFail ts 20 (T.pack (show (tqpOwnerForce p)))
  255   bindTextOrFail ts 21 (shadowStatusText (tqpShadowStatus p))
  256   bindTextOrFail ts 22 (shadowSnapshotIdText (tqpShadowSnapshotId p))
  257   bindTextOrFail ts 23 (shadowDivergenceKindText (tqpShadowDivergenceKind p))
  258   bindTextOrFail ts 24 (maybe "" (T.pack . show) (tqpShadowFamily p))
  259   bindTextOrFail ts 25 (maybe "" (T.pack . show) (tqpShadowForce p))
  260   bindTextOrFail ts 26 (tqpShadowMessage p)
  261   bindTextOrFail ts 27 replayTraceJson
  262   bindIntOrFail ts 28 (if tqpDivergence p then 1 else 0)
  263   stepOrFail ts

  309 withPreparedStatement :: NSQL.Database -> Text -> Text -> (NSQL.Statement -> IO a) -> IO a
  310 withPreparedStatement db sql context action = do
  311   mStmt <- NSQL.prepare db sql
  312   case mStmt of
  313     Left err ->
  314       throwQxFx0 (PersistenceTxError StageUnknown ("prepare failed for " <> context <> ": " <> err))
  315     Right stmt ->
  316       action stmt `finally` finalizeBestEffort stmt
  317 
  318 finalizeBestEffort :: NSQL.Statement -> IO ()
  319 finalizeBestEffort stmt = do
  320   _ <- tryQxFx0 (NSQL.finalize stmt)
  321   pure ()
