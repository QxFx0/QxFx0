   45 saveStateWithProjection :: DbRunner -> SystemState -> Text -> Maybe TurnProjection -> IO (Either PersistenceDiagnostic SystemState)
   46 saveStateWithProjection withDb ss sessionId mProjection = do
   47   let persistedState = ss { ssSessionId = sessionId }
   48   result <- tryQxFx0 $ withDb $ \db -> do
   49     withImmediateTransaction db $ do
   50       let jsonBlob = TE.decodeUtf8With lenientDecode . BL.toStrict . Aeson.encode $ persistedState
   51       touchRuntimeSessionActivity db sessionId
   52       saveKV db sessionId "__system_state__" jsonBlob
   53 
   54       case mProjection of
   55         Nothing -> pure ()
   56         Just projection -> do
   57           persistTurnQuality db sessionId projection
   58           when (tqpDivergence projection) $
   59             persistShadowDivergence db sessionId projection
   60 
   61       pure persistedState
   62   case result of
   63     Left (PersistenceTxError stage msg) -> pure (Left (diagnoseSave stage (Just msg)))
   64     Left other -> pure (Left (PdSaveFailed StageUnknown Nothing (Just (T.pack (show other)))))
   65     Right savedSs -> pure $ Right savedSs
   66 
   67 rollbackTurnProjections :: DbRunner -> Text -> Int -> IO (Either PersistenceDiagnostic ())
   68 rollbackTurnProjections withDb sessionId stableTurn = do
   69   result <- tryQxFx0 $ withDb $ \db -> do
   70     withImmediateTransaction db $ do
   71       deleteTurnQualityAbove db sessionId stableTurn
   72       deleteShadowDivergenceAbove db sessionId stableTurn
   73   case result of
   74     Left (PersistenceTxError stage msg) -> pure (Left (diagnoseRollback stage (Just msg)))
   75     Left other -> pure (Left (PdRollbackFailed StageUnknown Nothing (Just (T.pack (show other)))))
   76     Right () -> pure (Right ())

   78 withImmediateTransaction :: NSQL.Database -> IO a -> IO a
   79 withImmediateTransaction db action = mask $ \restore -> do
   80   beginResult <- NSQL.execSql db "BEGIN IMMEDIATE;"
   81   case beginResult of
   82     Left err ->
   83       throwQxFx0 (PersistenceTxError StageTxBegin ("tx_begin_failed: " <> err))
   84     Right _ ->
   85       pure ()
   86   result <- restore action `onException` rollbackBestEffort db
   87   commitResult <- NSQL.execSql db "COMMIT;"
   88   case commitResult of
   89     Right _ ->
   90       pure result
   91     Left err -> do
   92       rollbackResult <- NSQL.execSql db "ROLLBACK;"
   93       case rollbackResult of
   94         Right _ ->
   95           throwQxFx0 (PersistenceTxError StageTxCommit ("tx_commit_failed: " <> err))
   96         Left rbErr ->
   97           throwQxFx0
   98             (PersistenceTxError StageTxCommit ("tx_commit_and_rollback_failed: commit=" <> err <> " rollback=" <> rbErr))
   99 
  100 rollbackBestEffort :: NSQL.Database -> IO ()
  101 rollbackBestEffort db = do
  102   _ <- NSQL.execSql db "ROLLBACK;"
  103   pure ()

  121 loadState :: DbRunner -> Text -> IO LoadStateResult
  122 loadState withDb sessionId = withDb $ \db -> do
  123   mBlob <- loadKV db sessionId "__system_state__"
  124   case mBlob of
  125     Just blob ->
  126       case Aeson.eitherDecodeStrict (TE.encodeUtf8 blob) of
  127         Right ss -> pure (LoadStateRestored ss)
  128         Left _ -> pure (LoadStateCorrupt (PdCorruptDecode : stateBlobDiagnostics blob))
  129     Nothing -> pure LoadStateMissing
  130 
  131 stateBlobDiagnostics :: Text -> [PersistenceDiagnostic]
  132 stateBlobDiagnostics blob =
  133   case Aeson.decode (BL.fromStrict (TE.encodeUtf8 blob)) :: Maybe Aeson.Object of
  134     Nothing -> []
  135     Just obj ->
  136       let optionalSystemFields =
  137             [ "lastGuardReport"
  138             , "dreamState"
  139             , "intuitionState"
  140             , "semanticAnchor"
  141             , "lastTurnDecision"
  142             ]
  143           missing = filter (\k -> not (KM.member (AK.fromText k) obj)) optionalSystemFields
  144       in if null missing
  145            then []
  146            else [PdSchemaMissingFields missing]

  148 loadKV :: NSQL.Database -> Text -> Text -> IO (Maybe Text)
  149 loadKV db sessionId k = do
  150   let sql = "SELECT value FROM dialogue_state WHERE session_id = ? AND key = ?"
  151   withPreparedStatement db sql ("loadKV key=" <> k <> ", session=" <> sessionId) $ \stmt -> do
  152     _ <- NSQL.bindText stmt 1 sessionId
  153     _ <- NSQL.bindText stmt 2 k
  154     hasRow <- NSQL.stepRow stmt
  155     if hasRow
  156       then Just <$> NSQL.columnText stmt 0
  157       else pure Nothing

  226 withPreparedStatement :: NSQL.Database -> Text -> Text -> (NSQL.Statement -> IO a) -> IO a
  227 withPreparedStatement db sql context action = do
  228   mStmt <- NSQL.prepare db sql
  229   case mStmt of
  230     Left err ->
  231       throwQxFx0 (PersistenceTxError StageUnknown ("prepare failed for " <> context <> ": " <> err))
  232     Right stmt ->
  233       action stmt `finally` finalizeBestEffort stmt
  234 
  235 finalizeBestEffort :: NSQL.Statement -> IO ()
  236 finalizeBestEffort stmt = do
  237   _ <- tryQxFx0 (NSQL.finalize stmt)
  238   pure ()

  240 diagnoseSave :: PersistenceStage -> Maybe Text -> PersistenceDiagnostic
  241 diagnoseSave StageTxBegin _ = PdTransactionBeginFailed
  242 diagnoseSave StageTxCommit _ = PdTransactionCommitFailed
  243 diagnoseSave StageTxRollback _ = PdTransactionRollbackFailed
  244 diagnoseSave stage mMsg = PdSaveFailed stage Nothing mMsg

  246 diagnoseRollback :: PersistenceStage -> Maybe Text -> PersistenceDiagnostic
  247 diagnoseRollback StageTxBegin _ = PdTransactionBeginFailed
  248 diagnoseRollback StageTxCommit _ = PdTransactionCommitFailed
  249 diagnoseRollback StageTxRollback _ = PdTransactionRollbackFailed
  250 diagnoseRollback stage mMsg = PdRollbackFailed stage Nothing mMsg
