   11 import QxFx0.Types (R5Verdict)
   12 import QxFx0.Types.Decision (ShadowStatus(..))

   14   ( ShadowDivergence(..)

   21 
   22 data ShadowResult = ShadowResult
   23   { srStatus :: !ShadowStatus
   24   , srDivergence :: !ShadowDivergence
   25   , srDatalogVerdict :: !(Maybe R5Verdict)
   26   , srSnapshotId :: !ShadowSnapshotId
   27   , srDiagnostics :: ![Text]
   28   }
   29   deriving stock (Eq, Show)
   30 
   31 shadowUnavailableResult :: ShadowSnapshot -> ShadowDivergenceKind -> Text -> ShadowResult
   32 shadowUnavailableResult snapshot kind err =
   33   ShadowResult
   34     { srStatus = ShadowUnavailable
   35     , srDivergence = emptyShadowDivergence {sdKind = kind}
   36     , srDatalogVerdict = Nothing
   37     , srSnapshotId = mkShadowSnapshotId snapshot
   38     , srDiagnostics = [err]
   39     }
