   43 interpretOutput :: ConsciousnessModel -> KernelOutput -> Text -> ConsciousnessModel
   44 interpretOutput model output inputText =
   45   let consciousState = cmConscious model
   46       turn = csTurnCount consciousState + 1
   47       event = InterpretationEvent
   48         { ieWhat = interpretationEventWhatPrefix <> koSelectedSkill output
   49         , ieWhy = interpretWhy output
   50         , ieDesire = case koActiveDesires output of
   51             desire : _ -> desire
   52             [] -> interpretationNoDesire
   53         , ieTurn = turn
   54         }
   55       newSelfInterpretation = updateSelfInterpretation (csSelfInterp consciousState) output event
   56       newTrajectory = take consciousnessTrajectoryLimit (event : csTrajectory consciousState)
   57       newFocus = deriveFocus output inputText
   58       newConsciousState =
   59         consciousState
   60           { csSelfInterp = newSelfInterpretation
   61           , csTrajectory = newTrajectory
   62           , csFocus = newFocus
   63           , csTurnCount = turn
   64           }
   65    in model {cmConscious = newConsciousState}
   66 
   67 updateSelfInterpretation :: SelfInterpretation -> KernelOutput -> InterpretationEvent -> SelfInterpretation
   68 updateSelfInterpretation selfInterpretation output event =
   69   let activeDesireNames = koActiveDesires output
   70       newNarrative = buildNarrative output
   71       newPattern = case koSelectedSkill output of
   72         skill | skillAffinityFromName skill > consciousnessSkillAffinityPatternThreshold ->
   73           Just (observedPatternPrefix <> skill)
   74         _ -> Nothing
   75       newPatterns = take 5 $ maybe (siObservedPatterns selfInterpretation) (: siObservedPatterns selfInterpretation) newPattern
   76    in SelfInterpretation
   77         { siCurrentNarrative = newNarrative
   78         , siActiveDesires = take 3 activeDesireNames
   79         , siObservedPatterns = newPatterns
   80         , siConflicts = take 3 (koConflicts output)
   81         , siRecentEvents = take 5 (event : siRecentEvents selfInterpretation)
   82         }

   84 consciousnessToNarrative :: ConsciousnessModel -> KernelOutput -> ConsciousnessNarrative
   85 consciousnessToNarrative model output =
   86   let consciousState = cmConscious model
   87       selfInterpretation = csSelfInterp consciousState
   88       selfModel = ukSelfModel (cmKernel model)
   89    in ConsciousnessNarrative
   90         { cnKernelState = siCurrentNarrative selfInterpretation
   91         , cnActiveDesires = T.intercalate " / " (siActiveDesires selfInterpretation)
   92         , cnSkillInPlay = koSelectedSkill output
   93         , cnSelfView = smIdentity selfModel
   94         , cnConflict = case siConflicts selfInterpretation of
   95             [] -> ""
   96             conflict : _ -> consciousnessConflictPrefix <> conflict
   97         , cnLimitation = smLimitation selfModel
   98         }

  100 narrativeToPromptFragment :: ConsciousnessNarrative -> Text
  101 narrativeToPromptFragment narrative = T.unlines $ filter (not . T.null)
  102   [ consciousnessPromptHeader
  103   , ""
  104   , consciousnessKernelLabel <> cnKernelState narrative
  105   , consciousnessDesiresLabel <> cnActiveDesires narrative
  106   , consciousnessSkillLabel <> cnSkillInPlay narrative
  107   , if T.null (cnConflict narrative) then "" else cnConflict narrative
  108   , ""
  109   , consciousnessSelfModelLabel <> cnSelfView narrative
  110   , consciousnessLimitLabel <> cnLimitation narrative
  111   ]

  113 interpretWhy :: KernelOutput -> Text
  114 interpretWhy output =
  115   let desires = koActiveDesires output
  116       skill = koSelectedSkill output
  117       deep = trDeepContent (koSearchResult output)
  118    in interpretWhyDesiresPrefix
  119         <> T.intercalate ", " desires
  120         <> ". "
  121         <> interpretWhyDeepPrefix
  122         <> deep
  123         <> ". "
  124         <> interpretWhySkillLabel
  125         <> skill
  126         <> interpretWhySkillSuffix
  127 
  128 skillAffinityFromName :: Text -> Double
  129 skillAffinityFromName name = case lookup name skillAffinityMap of
  130   Just affinity -> affinity
  131   Nothing -> consciousnessDefaultSkillAffinity
  132 
  133 buildNarrative :: KernelOutput -> Text
  134 buildNarrative output =
  135   let skill = koSelectedSkill output
  136       desires = koActiveDesires output
  137       focus = trDeepContent (koSearchResult output)
  138       ontologicalQuestion = koOntologicalQuestion output
  139       framing = renderConsciousForm (oqForm ontologicalQuestion) <> " / " <> renderReturnMode (oqReturnForm ontologicalQuestion)
  140    in narrativeSearchPrefix
  141         <> focus
  142         <> ". "
  143         <> framing
  144         <> ". "
  145         <> narrativeActivePrefix
  146         <> T.intercalate ", " (take 2 desires)
  147         <> ". "
  148         <> consciousnessSkillLabel
  149         <> skill
  150         <> narrativeApplySuffix
  151 
  152 deriveFocus :: KernelOutput -> Text -> Text
  153 deriveFocus output inputText
  154   | T.null (koFocusHint output) = focusHumanPrefix <> T.take 30 inputText
  155   | otherwise = koFocusHint output
