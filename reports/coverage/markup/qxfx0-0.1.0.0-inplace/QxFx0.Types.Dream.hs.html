  125 dreamEpoch :: UTCTime
  126 dreamEpoch = UTCTime (fromGregorian 1970 1 1) 0

  128 emptyDreamState :: CoreVec -> DreamState
  129 emptyDreamState kv = initialDreamState dreamEpoch kv

  131 initialDreamState :: UTCTime -> CoreVec -> DreamState
  132 initialDreamState now kv = DreamState
  133   { dsR5State = R5State kv zeroVec zeroVec
  134   , dsBiasAttractor = zeroVec
  135   , dsLastWakeTime = now
  136   , dsLastDreamTime = now
  137   , dsDreamCycleCount = 0
  138   }
