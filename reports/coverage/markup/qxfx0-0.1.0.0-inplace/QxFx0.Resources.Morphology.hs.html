   25 {-# NOINLINE morphologyCache #-}
   26 morphologyCache :: MVar (Map.Map FilePath MorphologyData)
   27 morphologyCache = unsafePerformIO (newMVar Map.empty)
   28 
   29 loadMorphologyData :: IO MorphologyData
   30 loadMorphologyData = do
   31   mDirRaw <- getMorphologyDir
   32   mDir <- canonicalizePath mDirRaw
   33   modifyMVar morphologyCache $ \cache ->
   34     case Map.lookup mDir cache of
   35       Just md -> pure (cache, md)
   36       Nothing -> do
   37         prep <- loadMorphologyDict (mDir </> "prepositional.json")
   38         gen <- loadMorphologyDict (mDir </> "genitive.json")
   39         nom <- loadMorphologyDict (mDir </> "nominative.json")
   40         forms <- loadFormsBySurface (mDir </> "forms_by_surface.json")
   41         _ <- loadJsonValueStrict (mDir </> "lexicon_quality.json")
   42         let md = MorphologyData prep gen nom forms
   43         pure (Map.insert mDir md cache, md)
   44 
   45 validateMorphologyResources :: FilePath -> IO (Bool, String)
   46 validateMorphologyResources morphDir = do
   47   hasMorphDir <- doesDirectoryExist morphDir
   48   if not hasMorphDir
   49     then pure (False, "Morphology directory missing: " ++ morphDir)
   50     else do
   51       prep <- readMorphologyDict (morphDir </> "prepositional.json")
   52       gen <- readMorphologyDict (morphDir </> "genitive.json")
   53       nom <- readMorphologyDict (morphDir </> "nominative.json")
   54       -- forms_by_surface.json can be very large (>100 MB); validate
   55       -- presence only to keep readiness checks fast. Full parsing is
   56       -- deferred to loadMorphologyData at runtime.
   57       formsExist <- doesFileExist (morphDir </> "forms_by_surface.json")
   58       let forms = if formsExist then Right Map.empty else Left "forms_by_surface.json missing"
   59       quality <- loadJsonValueChecked (morphDir </> "lexicon_quality.json")
   60       let problems =
   61             [err | Left err <- [prep, gen, nom]]
   62               ++ either (:[]) (const []) forms
   63               ++ either (:[]) (const []) quality
   64       pure $
   65         if null problems
   66           then (True, "Morphology resources validated (forms_by_surface deferred to runtime)")
   67           else (False, unwordsWith "; " problems)
   68 
   69 loadMorphologyDict :: FilePath -> IO (Map.Map Text Text)
   70 loadMorphologyDict path = do
   71   result <- readMorphologyDict path
   72   case result of
   73     Right parsed -> pure parsed
   74     Left err -> throwMorphologyError err
   75 
   76 readMorphologyDict :: FilePath -> IO (Either String (Map.Map Text Text))
   77 readMorphologyDict path = do
   78   exists <- doesFileExist path
   79   if not exists
   80     then pure (Left ("Morphology file missing: " ++ path))
   81     else do
   82       content <- BL.readFile path
   83       case Aeson.eitherDecode content of
   84         Right parsed -> pure (Right parsed)
   85         Left err -> pure (Left ("Morphology JSON parse failed for " ++ path ++ ": " ++ err))
   86 
   87 loadJsonValueStrict :: FilePath -> IO Aeson.Value
   88 loadJsonValueStrict path = do
   89   result <- loadJsonValueChecked path
   90   case result of
   91     Right value -> pure value
   92     Left err -> throwMorphologyError err
   93 
   94 loadJsonValueChecked :: FilePath -> IO (Either String Aeson.Value)
   95 loadJsonValueChecked path = do
   96   exists <- doesFileExist path
   97   if not exists
   98     then pure (Left ("JSON resource missing: " ++ path))
   99     else do
  100       content <- BL.readFile path
  101       case Aeson.eitherDecode content of
  102         Right parsed -> pure (Right parsed)
  103         Left err -> pure (Left ("JSON parse failed for " ++ path ++ ": " ++ err))
  104 
  105 loadFormsBySurface :: FilePath -> IO (Map.Map Text [LexemeForm])
  106 loadFormsBySurface path = do
  107   result <- readFormsBySurface path
  108   case result of
  109     Right parsed -> pure parsed
  110     Left err -> throwMorphologyError err
  111 
  112 readFormsBySurface :: FilePath -> IO (Either String (Map.Map Text [LexemeForm]))
  113 readFormsBySurface path = do
  114   exists <- doesFileExist path
  115   if not exists
  116     then pure (Right Map.empty)
  117     else do
  118       content <- BL.readFile path
  119       case Aeson.eitherDecode content of
  120         Right parsed -> pure (Right (Map.map (map enforceQuality) parsed))
  121         Left err -> pure (Left ("forms_by_surface JSON parse failed for " ++ path ++ ": " ++ err))
  122   where
  123     enforceQuality :: LexemeForm -> LexemeForm
  124     enforceQuality lf = lf { lfQuality = max 0.0 (min 1.0 (lfQuality lf)) }
  125 
  126 unwordsWith :: String -> [String] -> String
  127 unwordsWith _ [] = ""
  128 unwordsWith sep (x:xs) = go x xs
  129   where
  130     go acc [] = acc
  131     go acc (y:ys) = go (acc ++ sep ++ y) ys
  132 
  133 throwMorphologyError :: String -> IO a
  134 throwMorphologyError = throwQxFx0 . RuntimeInitError . T.pack
