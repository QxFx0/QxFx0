  570 buildPerspectiveGovernanceEvent
  571   :: GovernanceActor
  572   -> Int
  573   -> Text
  574   -> Maybe GovernanceEvent
  575   -> PerspectiveRegistry
  576   -> PerspectiveRegistry
  577   -> PerspectiveInputBundle
  578   -> PerspectiveCandidate
  579   -> PerspectiveAdmissibility
  580   -> Maybe PerspectivePromotionDecision
  581   -> PerspectivePromotionDecision
  582   -> Maybe PerspectiveProjection
  583   -> GovernanceEvent
  584 buildPerspectiveGovernanceEvent actor sequenceNo sessionId previousEvent beforeRegistry afterRegistry bundle candidate admissibility requestedDecision resolvedDecision projection =
  585   normalizeGovernanceEvent rawEvent
  586   where
  587     turnId = isTurnCount (pibIdentitySlice bundle)
  588     scope = pcScope candidate
  589     (perspectiveId, perspectiveVersion, perspectiveStatus) = perspectiveIdentityForScope sequenceNo sessionId scope afterRegistry
  590     requestedGov = governanceDecisionFromPerspectivePromotion <$> requestedDecision
  591     resolvedGov = governanceDecisionFromPerspectiveOutcome admissibility resolvedDecision
  592     previousHash = previousEvent >>= geeEventHash . geEnvelope
  593     previousId = geeId . geEnvelope <$> previousEvent
  594     payload = GpPerspective PerspectivePayload
  595       { ppPerspectiveId = perspectiveId
  596       , ppPerspectiveScope = scope
  597       , ppPerspectiveVersion = perspectiveVersion
  598       , ppPerspectiveStatus = perspectiveStatus
  599       , ppPerspectiveInput = bundle
  600       , ppPerspectiveCandidate = candidate
  601       , ppPerspectiveAdmissibility = admissibility
  602       , ppRequestedDecision = requestedDecision
  603       , ppResolvedDecision = resolvedDecision
  604       , ppPerspectiveProjection = projection
  605       }
  606     eventId = GovernanceEventId
  607       ( T.intercalate ":"
  608           [ "gov"
  609           , safeEventText sessionId
  610           , "turn" <> T.pack (show turnId)
  611           , "seq" <> T.pack (show sequenceNo)
  612           , safeEventText (renderPerspectiveScope scope)
  613           , safeEventText (T.pack (show resolvedGov))
  614           ]
  615       )
  616     envelope = GovernanceEventEnvelope
  617       { geeId = eventId
  618       , geeSchemaVersion = currentGovernanceSchemaVersion
  619       , geePayloadVersion = currentGovernancePayloadVersion
  620       , geeLifecycleStatus = lifecycleForPerspective admissibility resolvedDecision
  621       , geeSubject = SubjectPerspective perspectiveId
  622       , geeActor = actor
  623       , geeTurnId = Just turnId
  624       , geeSessionId = if T.null sessionId then Nothing else Just sessionId
  625       , geeStreamId = "governance:" <> if T.null sessionId then "anonymous" else sessionId
  626       , geePartitionId = "perspective"
  627       , geeSequenceNo = sequenceNo
  628       , geeParentRefs = maybe [] pure previousId
  629       , geeDecision = resolvedGov
  630       , geeRequestedDecision = requestedGov
  631       , geeResolvedDecision = Just resolvedGov
  632       , geeReason = perspectiveGovernanceReason scope admissibility candidate resolvedGov
  633       , geeNormativeProfileVersion = Just (pcNormativeProfileVersion candidate)
  634       , geeProjectionVersion = currentProjectionVersion
  635       , geeReducerVersion = currentReducerVersion
  636       , geeNormativeEvaluatorVersion = currentEvaluatorVersion
  637       , geeConfidenceAlgebraVersion = currentEvaluatorVersion
  638       , geeCapabilityEvaluatorVersion = currentEvaluatorVersion
  639       , geeSandboxVersion = currentEvaluatorVersion
  640       , geeRollbackLink = if resolvedDecision == PpdRollbackPrior then previousId else Nothing
  641       , geeBeforeRef = Just (GovRefSnapshot ("perspective-registry:" <> perspectiveRegistryHash beforeRegistry))
  642       , geeAfterRef = Just (GovRefSnapshot ("perspective-registry:" <> perspectiveRegistryHash afterRegistry))
  643       , geeEffectRef = Just (GovRefPerspective perspectiveId)
  644       , geeEventHash = Nothing
  645       , geePrevHash = previousHash
  646       , geePayloadHash = Nothing
  647       }
  648     rawEvent = GovernanceEvent envelope payload

  650 appendGovernanceEventToHistory :: GovernanceEvent -> [GovernanceEvent] -> Either Text [GovernanceEvent]
  651 appendGovernanceEventToHistory event history = do
  652   normalized <- normalizeGovernanceEventChecked event
  653   let eventId = geeId (geEnvelope normalized)
  654       sameId existing = geeId (geEnvelope existing) == eventId
  655   normalizedHistory <- traverse normalizeGovernanceEventChecked history
  656   validateGovernanceHashChain normalizedHistory
  657   case filter sameId normalizedHistory of
  658     [] -> do
  659       validateGovernanceAppend normalized normalizedHistory
  660       pure (history <> [normalized])
  661     existing:_
  662       | existing == normalized -> pure history
  663       | otherwise -> Left ("duplicate governance event id with conflicting semantics: " <> governanceEventIdText eventId)

  665 canonicalizeGovernanceHistory :: [GovernanceEvent] -> Either Text [GovernanceEvent]
  666 canonicalizeGovernanceHistory events = do
  667   normalized <- traverse normalizeGovernanceEventChecked events
  668   let groups = groupBy ((==) `on` (geeId . geEnvelope)) (sortOn (geeId . geEnvelope) normalized)
  669   deduped <- traverse dedupeGovernanceGroup groups
  670   let ordered = sortOn governanceOrderKey deduped
  671   validateGovernanceHashChain ordered
  672   pure ordered

  674 normalizeGovernanceEventChecked :: GovernanceEvent -> Either Text GovernanceEvent
  675 normalizeGovernanceEventChecked event = do
  676   let envelope = geEnvelope event
  677   if geeSchemaVersion envelope /= currentGovernanceSchemaVersion
  678     then Left ("unsupported governance schema version: " <> T.pack (show (geeSchemaVersion envelope)))
  679     else pure ()
  680   if geePayloadVersion envelope /= currentGovernancePayloadVersion
  681     then Left ("unsupported governance payload version: " <> T.pack (show (geePayloadVersion envelope)))
  682     else pure ()
  683   if geeSequenceNo envelope <= 0
  684     then Left ("invalid governance sequence number: " <> T.pack (show (geeSequenceNo envelope)))
  685     else pure ()
  686   let scheme = detectGovernanceHashScheme envelope
  687       normalized = normalizeGovernanceEventWith scheme event
  688       normalizedEnvelope = geEnvelope normalized
  689   case geePayloadHash envelope of
  690     Just payloadHash | payloadHash /= fromMaybe "" (geePayloadHash normalizedEnvelope) ->
  691       Left ("governance payload hash mismatch: " <> governanceEventIdText (geeId envelope))
  692     _ -> pure ()
  693   case geeEventHash envelope of
  694     Just eventHash | eventHash /= fromMaybe "" (geeEventHash normalizedEnvelope) ->
  695       Left ("governance event hash mismatch: " <> governanceEventIdText (geeId envelope))
  696     _ -> pure ()
  697   validateGovernanceEventContract normalized
  698   pure normalized

  700 validateGovernanceEventContract :: GovernanceEvent -> Either Text ()
  701 validateGovernanceEventContract event = do
  702   let envelope = geEnvelope event
  703       eventId = geeId envelope
  704       failWith reason = Left (reason <> ": " <> governanceEventIdText eventId)
  705   validateGovernanceRefs event
  706   case geeResolvedDecision envelope of
  707     Just resolved | resolved /= geeDecision envelope ->
  708       failWith "resolved decision must equal canonical decision"
  709     _ -> pure ()
  710   if payloadSubject event /= geeSubject envelope
  711     then failWith "governance subject does not match typed payload"
  712     else pure ()
  713   if supportedGovernedSubject (geeSubject envelope)
  714     then pure ()
  715     else failWith "governed subject family is declared but not yet canonically replay-supported"
  716   case governancePermission (geeActor envelope) (geeSubject envelope) (geeDecision envelope) of
  717     GovernanceAllowed -> pure ()
  718     GovernanceDenied reason -> failWith ("governance permission denied: " <> reason)

  721   = RefRequired
  722   | RefOptional
  723   | RefForbidden

  728   , grcEffectRef :: !GovernanceRefRequirement
  729   , grcRollbackLink :: !GovernanceRefRequirement

  733 validateGovernanceRefs :: GovernanceEvent -> Either Text ()
  734 validateGovernanceRefs event = do
  735   let envelope = geEnvelope event
  736       eventId = governanceEventIdText (geeId envelope)
  737       contract = requiredRefsForEvent event
  738       failWith reason = Left (reason <> ": " <> eventId)
  739   validateRefField failWith "before_ref" (grcBeforeRef contract) (geeBeforeRef envelope)
  740   validateRefField failWith "after_ref" (grcAfterRef contract) (geeAfterRef envelope)
  741   validateRefField failWith "effect_ref" (grcEffectRef contract) (geeEffectRef envelope)
  742   validateRefField failWith "rollback_link" (grcRollbackLink contract) (geeRollbackLink envelope)
  743   validateRefField failWith "resolved decision" (grcResolvedDecision contract) (geeResolvedDecision envelope)

  745 requiredRefsForEvent :: GovernanceEvent -> GovernanceRefsContract
  746 requiredRefsForEvent event =
  747   subjectRefContract (geeSubject envelope) (geeLifecycleStatus envelope) (geeDecision envelope)

  749     envelope = geEnvelope event

  751 subjectRefContract :: GovernedSubject -> GovernanceLifecycleStatus -> GovernanceDecision -> GovernanceRefsContract
  752 subjectRefContract subject lifecycle decision =
  753   case subject of
  754     SubjectPerspective _ -> lifecycleRefContract lifecycle decision
  755     SubjectFreeze _ -> lifecycleRefContract lifecycle decision
  756     SubjectClaimStance _ -> lifecycleRefContract lifecycle decision
  757     SubjectCapability _ -> lifecycleRefContract lifecycle decision
  758     SubjectCrossSessionCarry _ -> lifecycleRefContract lifecycle decision
  759     SubjectNormativeProfile _ -> lifecycleRefContract lifecycle decision

  761 lifecycleRefContract :: GovernanceLifecycleStatus -> GovernanceDecision -> GovernanceRefsContract
  762 lifecycleRefContract lifecycle decision = GovernanceRefsContract
  763   { grcBeforeRef = RefRequired
  764   , grcAfterRef = afterEffectRefRequirement lifecycle decision
  765   , grcEffectRef = afterEffectRefRequirement lifecycle decision
  766   , grcRollbackLink = if decision == GovRollback || lifecycle == GlsRolledBack then RefRequired else RefOptional
  767   , grcResolvedDecision = RefRequired
  768   }

  770 afterEffectRefRequirement :: GovernanceLifecycleStatus -> GovernanceDecision -> GovernanceRefRequirement
  771 afterEffectRefRequirement lifecycle decision
  772   | decision == GovDeny = RefOptional
  773   | otherwise =
  774       case lifecycle of
  775         GlsEvaluated -> RefRequired
  776         GlsCommitted -> RefRequired
  777         GlsRolledBack -> RefRequired
  778         GlsDenied -> RefOptional
  779         GlsIntent -> RefOptional
  780         GlsArchived -> RefOptional

  782 validateRefField :: (Text -> Either Text ()) -> Text -> GovernanceRefRequirement -> Maybe a -> Either Text ()
  783 validateRefField failWith label requirement value =
  784   case (requirement, value) of
  785     (RefRequired, Nothing) -> failWith ("governance event missing " <> label)
  786     (RefForbidden, Just _) -> failWith ("governance event unexpected " <> label)
  787     _ -> pure ()

 1047 validateGovernanceHashChain :: [GovernanceEvent] -> Either Text ()
 1048 validateGovernanceHashChain = go Nothing
 1049   where
 1050     go _ [] = pure ()
 1051     go expectedPrev (event:rest) = do
 1052       let envelope = geEnvelope event
 1053       if geePrevHash envelope /= expectedPrev
 1054         then Left ("governance hash chain mismatch: " <> governanceEventIdText (geeId envelope))
 1055         else go (geeEventHash envelope) rest

 1057 validateGovernanceAppend :: GovernanceEvent -> [GovernanceEvent] -> Either Text ()
 1058 validateGovernanceAppend event history = do
 1059   validateGovernanceHashChain history
 1060   case history of
 1061     [] ->
 1062       if geePrevHash envelope == Nothing
 1063         then pure ()
 1064         else Left ("first governance event must not carry prev_hash: " <> eventId)
 1065     previous : rest -> do
 1066       let previousEvent = foldl (\_ current -> current) previous rest
 1067           previousEnvelope = geEnvelope previousEvent
 1068       if geeSequenceNo envelope > geeSequenceNo previousEnvelope
 1069         then pure ()
 1070         else Left ("governance sequence must increase on append: " <> eventId)
 1071       if geePrevHash envelope == geeEventHash previousEnvelope
 1072         then pure ()
 1073         else Left ("governance append prev_hash mismatch: " <> eventId)
 1074   where
 1075     envelope = geEnvelope event
 1076     eventId = governanceEventIdText (geeId envelope)

 1088 hashStringWith :: GovernanceHashScheme -> String -> Text
 1089 hashStringWith scheme input = case scheme of
 1090   GovernanceHashLegacyFNV -> legacyFNV input
 1091   GovernanceHashSha256 -> "sha256:" <> sha256Hex (SHA256.hashlazy (BL8.pack input))
 1092   where
 1093     sha256Hex = T.pack . concatMap byteToHex . BS.unpack
 1094     byteToHex byte =
 1095       let raw = showHex byte ""
 1096       in case raw of
 1097            [single] -> ['0', single]
 1098            other -> other

 1100 legacyFNV :: String -> Text
 1101 legacyFNV input = T.pack ("fnv1a64:" <> showHex (foldl fnvStep fnvOffset input) "")
 1102   where
 1103     fnvOffset :: Word64
 1104     fnvOffset = 14695981039346656037
 1105 
 1106     fnvStep :: Word64 -> Char -> Word64
 1107     fnvStep h ch = (h `xor` fromIntegral (fromEnum ch)) * 1099511628211
