   35 guessPartOfSpeech :: Text -> InputPartOfSpeech
   36 guessPartOfSpeech token =
   37   case posOverride (lemmaForToken token) of
   38     Just forcedPos -> forcedPos
   39     Nothing ->
   40       case generatedLemmaToPos (lemmaForToken token) of
   41         Just pos -> pos
   42         Nothing
   43           | token `elem` prepositions -> PosPreposition
   44           | token `elem` conjunctions -> PosConjunction
   45           | token `elem` particles -> PosParticle
   46           | token `elem` interjections -> PosInterjection
   47           | token `elem` pronouns -> PosPronoun
   48           | token `elem` numerals -> PosNumeral
   49           | isVerbLike token -> PosVerb
   50           | isAdjectiveLike token -> PosAdjective
   51           | isAdverbLike token -> PosAdverb
   52           | isNounLike token -> PosNoun
   53           | otherwise -> PosUnknown

   55 guessMorphFeatures :: Text -> [InputMorphFeature]
   56 guessMorphFeatures token =
   57   concat
   58     [ if token == "не" then [FeatNegated] else []
   59     , if token `elem` firstPerson then [FeatPerson1] else []
   60     , if token `elem` secondPerson then [FeatPerson2] else []
   61     , if token `elem` pluralHints then [FeatNumberPlur] else []
   62     , if token `elem` singularHints then [FeatNumberSing] else []
   63     ]

   65 semanticClassesForToken :: Text -> [InputSemanticClass]
   66 semanticClassesForToken token =
   67   semanticClassesForLemmaAndPos token (lemmaForToken token) (guessPartOfSpeech token)
   68 
   69 semanticClassesForLemmaAndPos :: Text -> Text -> InputPartOfSpeech -> [InputSemanticClass]
   70 semanticClassesForLemmaAndPos token lemma pos =
   71   let genSem = generatedLemmaToSem lemma
   72   in case semanticOverride lemma of
   73        Just forcedSem -> forcedSem
   74        Nothing ->
   75          if not (null genSem) then genSem else
   76            if token `elem` selfReferenceTokens then [SemSelfReference]
   77            else if token `elem` userReferenceTokens then [SemUserReference]
   78            else if isIdentityLemma lemma then [SemIdentity]
   79            else if isWorldNoun lemma || isWorldNoun token then [SemWorldObject]
   80            else if isMentalNoun lemma || isMentalNoun token then [SemMentalObject]
   81            else if lemma `elem` knowledgeTokens || token `elem` knowledgeTokens then [SemKnowledge]
   82            else if lemma `elem` comparisonTokens || token `elem` comparisonTokens then [SemComparison]
   83            else if lemma `elem` causeTokens || token `elem` causeTokens then [SemCause]
   84            else if lemma `elem` invitationTokens || token `elem` invitationTokens then [SemDialogueInvitation]
   85            else if lemma `elem` repairTokens || token `elem` repairTokens then [SemDialogueRepair]
   86            else if lemma `elem` contemplativeTokens || token `elem` contemplativeTokens then [SemContemplative]
   87            else case pos of
   88              PosVerb -> [SemAction]
   89              PosAdjective -> [SemState, SemQualityProperty]
   90              PosAdverb -> [SemState]
   91              PosNoun
   92                | isAbstractNounLike lemma -> [SemAbstractConcept]
   93                | isPhysicalNounLike lemma -> [SemPhysicalObject, SemWorldObject]
   94                | otherwise -> [SemUnknown]
   95              PosPronoun -> [SemUnknown]
   96              PosNumeral -> [SemUnknown]
   97              PosPreposition -> [SemRelation]
   98              PosConjunction -> [SemUnknown]
   99              PosParticle -> [SemUnknown]
  100              PosInterjection -> [SemDialogueInvitation]
  101              PosUnknown -> [SemUnknown]

  103 posOverride :: Text -> Maybe InputPartOfSpeech
  104 posOverride lemma =
  105   case lemma of
  106     "привет" -> Just PosInterjection
  107     "здравствуй" -> Just PosInterjection
  108     "здравствуйте" -> Just PosInterjection
  109     "салют" -> Just PosInterjection
  110     "хай" -> Just PosInterjection
  111     _ -> Nothing
  112 
  113 semanticOverride :: Text -> Maybe [InputSemanticClass]
  114 semanticOverride lemma
  115   | lemma `elem` ["привет", "здравствуй", "здравствуйте", "салют", "хай"] =
  116       Just [SemDialogueInvitation]
  117   | otherwise = Nothing

  119 discourseFunctionsForToken :: Text -> [InputDiscourseFunction]
  120 discourseFunctionsForToken token =
  121   concat
  122     [ if token == "не" then [DiscNegation] else []
  123     , if token `elem` questionMarkers then [DiscQuestion] else []
  124     , if token `elem` contrastMarkers then [DiscContrast] else []
  125     , if token `elem` conditionMarkers then [DiscCondition] else []
  126     , if token `elem` causeMarkers then [DiscCause] else []
  127     , if token `elem` resultMarkers then [DiscResult] else []
  128     , if token `elem` invitationTokens then [DiscInvitation] else []
  129     , if token `elem` clarifyMarkers then [DiscClarification] else []
  130     , if token `elem` emphasisMarkers then [DiscEmphasis] else []
  131     ]

  222 isFunctionWord :: Text -> Bool
  223 isFunctionWord token =
  224   token `elem` (prepositions <> conjunctions <> particles <> questionMarkers <> contrastMarkers <> causeMarkers)

  226 isWorldNoun :: Text -> Bool

  229 isMentalNoun :: Text -> Bool

  244 isVerbLike :: Text -> Bool
  245 isVerbLike token =
  246   token `elem` commonVerbs
  247     || any (`T.isSuffixOf` token)
  248       [ "ть", "ти", "чь"
  249       , "ю", "у", "ем", "ешь", "ет", "ют", "ут", "ешься", "ется", "емся", "етесь"
  250       , "ил", "ила", "или", "ал", "ала", "али", "ался", "алась", "алось", "ались"
  251       ]
  252 
  253 isAdjectiveLike :: Text -> Bool
  254 isAdjectiveLike token =
  255   any (`T.isSuffixOf` token)
  256     [ "ый", "ий", "ой", "ая", "яя", "ое", "ее", "ые", "ие"
  257     , "ого", "ему", "ыми", "ых"
  258     ]
  259 
  260 isAdverbLike :: Text -> Bool
  261 isAdverbLike token =
  262   token `elem` adverbs
  263     || any (`T.isSuffixOf` token) ["о", "е"]
  264 
  265 isNounLike :: Text -> Bool
  266 isNounLike token =
  267   token `elem` worldNouns
  268     || token `elem` mentalNouns
  269     || any (`T.isSuffixOf` token)
  270       [ "а", "я", "о", "е", "и", "ы", "ь", "ие", "ия", "ость", "ение", "изм" ]
  271 
  272 isAbstractNounLike :: Text -> Bool
  273 isAbstractNounLike lemma =
  274   any (`T.isSuffixOf` lemma)
  275     [ "ость", "есть", "ение", "ание", "изм", "ция", "ность", "ство", "смысл", "логика" ]
  276 
  277 isPhysicalNounLike :: Text -> Bool
  278 isPhysicalNounLike lemma =
  279   isWorldNoun lemma
  280     || any (`T.isSuffixOf` lemma)
  281       [ "дом", "город", "лес", "камень", "река", "море", "стол", "свет", "ветер" ]
