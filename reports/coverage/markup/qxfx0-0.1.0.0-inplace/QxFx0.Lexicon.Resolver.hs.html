    1 {-# LANGUAGE DerivingStrategies #-}
    2 {-# LANGUAGE OverloadedStrings #-}
    3 {-# LANGUAGE StrictData #-}
    4 {-| Lexeme candidate resolver policy: curated > brain-kb-reviewed > auto-verified > auto-coverage,
    5     exact match preferred, higher quality wins, dangerous ambiguity falls back.
    6 -}
    7 module QxFx0.Lexicon.Resolver
    8   ( resolveLexemeForm
    9   , resolveLexemeFormRawFallback
   10   , tierPriority
   11   ) where
   12 
   13 import Data.List (sortOn)
   14 import qualified Data.Map.Strict as M
   15 import Data.Text (Text)
   16 import qualified Data.Text as T
   17 
   18 import QxFx0.Types.Domain.Atoms
   19   ( LexemeForm(..)
   20   , LexemeCase(..)
   21   , LexemeNumber(..)
   22   , SourceTier(..)
   23   , MorphologyData(..)
   24   )
   25 
   26 tierPriority :: SourceTier -> Int
   27 tierPriority CuratedTier = 4
   28 tierPriority BrainKbReviewedTier = 3
   29 tierPriority AutoVerifiedTier = 2
   30 tierPriority AutoCoverageTier = 1
   31 
   32 resolveLexemeForm
   33   :: MorphologyData
   34   -> Text
   35   -> Maybe LexemeCase
   36   -> Maybe LexemeNumber
   37   -> Maybe LexemeForm
   38 resolveLexemeForm md surface mCase mNumber =
   39   let lower = T.toLower surface
   40       candidates = M.lookup lower (mdFormsBySurface md)
   41   in case candidates of
   42        Nothing -> Nothing
   43        Just forms ->
   44          let filtered = filterMatches mCase mNumber forms
   45              ranked = sortOn rankingKey filtered
   46          in pickUnambiguous ranked
   47 
   48 filterMatches :: Maybe LexemeCase -> Maybe LexemeNumber -> [LexemeForm] -> [LexemeForm]
   49 filterMatches Nothing Nothing forms = forms
   50 filterMatches (Just c) Nothing forms = [f | f <- forms, lfCase f == c]
   51 filterMatches Nothing (Just n) forms = [f | f <- forms, lfNumber f == n]
   52 filterMatches (Just c) (Just n) forms = [f | f <- forms, lfCase f == c, lfNumber f == n]
   53 
   54 rankingKey :: LexemeForm -> (Int, Double, Text)
   55 rankingKey f =
   56   ( negate (tierPriority (lfTier f))
   57   , negate (lfQuality f)
   58   , lfLemma f
   59   )
   60 
   61 -- | Pick the top candidate only if it is unambiguous at the tier level.
   62 -- If the top two candidates have the same tier and quality, treat as
   63 -- dangerous ambiguity and return Nothing (fallback to raw surface).
   64 pickUnambiguous :: [LexemeForm] -> Maybe LexemeForm
   65 pickUnambiguous [] = Nothing
   66 pickUnambiguous [f] = Just f
   67 pickUnambiguous (f1:f2:_)
   68   | lfTier f1 == lfTier f2 && lfQuality f1 == lfQuality f2 = Nothing
   69   | otherwise = Just f1
   70 
   71 -- | Convenience: resolve and return the raw surface on failure.
   72 resolveLexemeFormRawFallback
   73   :: MorphologyData
   74   -> Text
   75   -> Maybe LexemeCase
   76   -> Maybe LexemeNumber
   77   -> Text
   78 resolveLexemeFormRawFallback md surface mCase mNumber =
   79   maybe surface lfSurface (resolveLexemeForm md surface mCase mNumber)
