   34 --------------------------------------------------------------------------------
   35 -- Atom slot types
   36 --------------------------------------------------------------------------------
   37 
   38 {-| Semantic slot kind — what role does this piece of information play? -}
   39 data MeaningAtom
   40   = AtomSubject      -- ^ The entity being discussed (сердце, кровь, Солнце)
   41   | AtomPredicate    -- ^ Action or property (сокращается, течёт, излучает)
   42   | AtomQuantity     -- ^ Numeric value (100 000, 7 000, 2,5)
   43   | AtomUnit         -- ^ Unit of measurement (сокращений, литров, км/с)
   44   | AtomPeriod       -- ^ Time frame (в сутки, за секунду, миллионы лет)
   45   | AtomEffect       -- ^ Result/consequence (перекачивает кровь, освещает Землю)
   46   | AtomMechanism    -- ^ How it works (мышечное сокращение, ядерный синтез)
   47   | AtomContext      -- ^ Broader system (кровеносная система, Солнечная система)
   48   | AtomRelation     -- ^ Connection to other concepts (part_of, regulates, orbits)
   49   | AtomScale        -- ^ Comparison reference (35 полных ванн, как теннисный мяч)
   50   | AtomLocation     -- ^ Where (в грудной клетке, в центре Солнечной системы)
   51   | AtomProperty     -- ^ Attribute (полый, мышечный, красный, горячий)
   52   | AtomDiscoverer   -- ^ Who discovered/first described
   53   | AtomYear         -- ^ Historical moment (1901, XIV век)
   54    | AtomReason       -- ^ Causal explanation (почему это важно)
   55    | AtomCondition    -- ^ Under what conditions
   56    | AtomAnalogy      -- ^ Pre-computed analogy (работает как насос)
   57     | TVerb           -- ^ Verb/action in a fact: "распределяет", "сокращается"
   58     | TObject         -- ^ Object of action: "кровь", "органы", "кислород"
   59     | TCondition      -- ^ Condition/prerequisite: "без учёта", "при отсутствии"
   60     | TVerbReason     -- ^ Verb in a reason clause: "доставлять", "требовать"
   61     | TObjectReason   -- ^ Object of the reason verb: "кислород", "совместимость"
   62    | TModifier       -- ^ Adjectival modifier: "крупнейший", "красный", "насыщенный"
   63    | TQuantity       -- ^ Numeric quantity extracted from scale: "2,5", "1869"
   64    | TRelation       -- ^ Relation from decomposed scale: "длиннее экватора Земли"
   65     deriving stock (Eq, Ord, Show, Generic, Bounded, Enum, Read)
   66    deriving anyclass (NFData, ToJSON, FromJSON)
   67 
   68 instance ToJSONKey MeaningAtom where
   69   toJSONKey = ToJSONKeyValue toJSON toEncoding
   70 instance FromJSONKey MeaningAtom where
   71   fromJSONKey = FromJSONKeyTextParser $ \t ->
   72     case readMaybe (T.unpack t) of
   73       Just tag -> pure tag
   74       Nothing  -> fail ("Unknown MeaningAtom: " ++ T.unpack t)

   76 --------------------------------------------------------------------------------
   77 -- Concrete slot value
   78 --------------------------------------------------------------------------------
   79 
   80 {-| One slot: an atom kind plus its concrete text value.
   81   Additional case forms are stored for morphological variation. -}
   82 data AtomSlot = AtomSlot
   83   { asAtom :: !MeaningAtom
   84   , asValue :: !Text              -- ^ Primary form (usually nominative)
   85   , asGen :: !Text                -- ^ Genitive form
   86   , asPrep :: !Text               -- ^ Prepositional form
   87   , asAcc :: !Text                -- ^ Accusative form
   88   , asIns :: !Text                -- ^ Instrumental form
   89   , asPlural :: !Text             -- ^ Plural nominative
   90   } deriving stock (Eq, Show, Generic)
   91   deriving anyclass (NFData, ToJSON, FromJSON)
   92 
   93 {-| Simple slot with no case variants (for numbers, units, etc.) -}
   94 plainSlot :: MeaningAtom -> Text -> AtomSlot
   95 plainSlot atom val = AtomSlot atom val val val val val val
   96 
   97 {-| Slot with a noun/lemma and its full 5-case forms. -}
   98 nounSlot :: MeaningAtom -> Text -> Text -> Text -> Text -> Text -> Text -> AtomSlot
   99 nounSlot atom nom gen prep acc ins pl =
  100   AtomSlot atom nom gen prep acc ins pl
  101 
  102 --------------------------------------------------------------------------------
  103 -- FactAtoms: a decomposed fact
  104 --------------------------------------------------------------------------------
  105 
  106 {-| One decomposed fact = a set of AtomSlots keyed by MeaningAtom.
  107   Multiple slots of the same kind are possible (e.g. several properties). -}
  108 newtype FactAtoms = FactAtoms
  109   { faSlots :: Map MeaningAtom [AtomSlot]
  110   } deriving stock (Eq, Show, Generic)
  111   deriving anyclass (NFData, ToJSON, FromJSON)
  112 
  113 emptyFactAtoms :: FactAtoms
  114 emptyFactAtoms = FactAtoms M.empty
  115 
  116 slotValue :: MeaningAtom -> FactAtoms -> [AtomSlot]
  117 slotValue atom (FactAtoms m) = M.findWithDefault [] atom m
  118 
  119 hasAtom :: MeaningAtom -> FactAtoms -> Bool
  120 hasAtom atom = not . null . slotValue atom
