   32 buildTurnInput :: SystemState -> Text -> Text -> PrepareEffectPlan -> PrepareEffectResults -> TurnInput
   33 buildTurnInput ss requestId sessionId effectPlan effectResults =
   34   let prepareStatic = pepStatic effectPlan
   35       embResult = perEmbeddingResult effectResults
   36       emb = erEmbedding embResult
   37       embQuality = embeddingSourceQuality (erSource embResult)
   38       embSimilarity =
   39         case (embQuality, ssLastEmbedding ss) of
   40           (EmbeddingQualityModeled, Just prevEmb) -> realToFrac (cosineSimilarity emb prevEmb)
   41           _ -> 0.0
   42       timeline = perTimeline effectResults
   43       !metrics0 = emptyTurnMetrics requestId sessionId
   44       !metrics1 =
   45         addPhase
   46           (recordPhase "prepare_static" (ptlStartTime timeline) (ptlPrepareStaticDone timeline))
   47           metrics0
   48       !metrics2 =
   49         (addPhase
   50           (recordPhase "embedding" (ptlEmbeddingStart timeline) (ptlEmbeddingDone timeline))
   51           metrics1)
   52             { tmEmbeddingSource = embeddingSourceText (erSource embResult) }
   53       !metrics3 =
   54         addPhase
   55           (recordPhase "nix_check" (ptlNixStart timeline) (ptlNixDone timeline))
   56           metrics2
   57       !metrics4 =
   58         addPhase
   59           (recordPhase "consciousness" (ptlConsciousnessStart timeline) (ptlConsciousnessDone timeline))
   60           metrics3
   61       !metrics5 =
   62         addPhase
   63           (recordPhase "intuition" (ptlIntuitionStart timeline) (ptlIntuitionDone timeline))
   64           metrics4
   65       !metrics6 =
   66         addPhase
   67           (recordPhase "api_health" (ptlApiHealthStart timeline) (ptlApiHealthDone timeline))
   68           metrics5
   69       nixStatus = perNixStatus effectResults
   70       nixAvailable = case nixStatus of Unavailable _ -> False; _ -> True
   71       isNixBlocked = case nixStatus of Blocked _ -> True; _ -> False
   72   in TurnInput
   73       { tiStartTime = ptlStartTime timeline
   74       , tiEmbedding = emb
   75       , tiEmbeddingSource = erSource embResult
   76       , tiEmbeddingQuality = embQuality
   77       , tiEmbSimilarity = embSimilarity
   78       , tiAtomSet = psAtomSet prepareStatic
   79       , tiNewTrace = psNewTrace prepareStatic
   80       , tiNextUserState = psNextUserState prepareStatic
   81       , tiRecommendedFamily = psRecommendedFamily prepareStatic
   82       , tiFrame = psFrame prepareStatic
   83       , tiNixStatus = nixStatus
   84       , tiNixAvailable = nixAvailable
   85       , tiIsNixBlocked = isNixBlocked
   86       , tiConceptToCheck = psConceptToCheck prepareStatic
   87       , tiBestTopic = psBestTopic prepareStatic
   88       , tiMetrics = metrics6
   89       }
