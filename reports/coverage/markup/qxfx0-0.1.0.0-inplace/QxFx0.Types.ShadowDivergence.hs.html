   99 mkShadowSnapshotId :: ShadowSnapshot -> ShadowSnapshotId
  100 mkShadowSnapshotId snapshot =
  101   let payload =
  102         T.intercalate
  103           "|"
  104           [ T.pack (show (ssRequestedFamily snapshot))
  105           , T.pack (show (ssInputForce snapshot))
  106           , T.intercalate "," (ssInputAtoms snapshot)
  107           , T.intercalate "," (map (\(k, v) -> k <> "=" <> v) (ssInputAtomDetails snapshot))
  108           ]
  109       hashValue = fnv1a64 payload
  110       asHex = T.pack (padHex16 (showHex hashValue ""))
  111   in ShadowSnapshotId ("shadow:" <> asHex)
  112 
  113 fnv1a64 :: Text -> Word64
  114 fnv1a64 =
  115   T.foldl'
  116     (\acc ch -> (acc `xor` fromIntegral (ord ch)) * 1099511628211)
  117     1469598103934665603
  118 
  119 padHex16 :: String -> String
  120 padHex16 raw =
  121   let n = length raw
  122   in replicate (max 0 (16 - n)) '0' <> raw

  143 computeShadowLegitimacyPenalty :: ShadowDivergence -> Double
  144 computeShadowLegitimacyPenalty divergence =
  145   let penalty = 0.0
  146       penalty' = if sdFamilyMismatch divergence then penalty + shadowPenaltyFamilyMismatch else penalty
  147       penalty'' = if sdForceMismatch divergence then penalty' + shadowPenaltyForceMismatch else penalty'
  148       penalty''' = if sdClauseMismatch divergence then penalty'' + shadowPenaltyClauseMismatch else penalty''
  149       penalty'''' = if sdLayerMismatch divergence then penalty''' + shadowPenaltyLayerMismatch else penalty'''
  150       penalty''''' = if sdWarrantedMismatch divergence then penalty'''' + shadowPenaltyWarrantedMismatch else penalty''''
  151   in min 1.0 penalty'''''
  152 
  153 computeShadowLegitimacyPenaltyWithSeverity :: ShadowDivergenceSeverity -> ShadowDivergence -> Double
  154 computeShadowLegitimacyPenaltyWithSeverity severity divergence =
  155   case severity of
  156     ShadowSeverityClean -> 0.0
  157     ShadowSeverityAdvisory -> 0.0
  158     ShadowSeveritySafety -> computeShadowLegitimacyPenalty divergence
  159     ShadowSeverityContract -> computeShadowLegitimacyPenalty divergence
  160     ShadowSeverityUnavailable -> computeShadowLegitimacyPenalty divergence
