   92 mkShadowSnapshotId :: ShadowSnapshot -> ShadowSnapshotId
   93 mkShadowSnapshotId snapshot =
   94   let payload =
   95         T.intercalate
   96           "|"
   97           [ T.pack (show (ssRequestedFamily snapshot))
   98           , T.pack (show (ssInputForce snapshot))
   99           , T.intercalate "," (ssInputAtoms snapshot)
  100           , T.intercalate "," (map (\(k, v) -> k <> "=" <> v) (ssInputAtomDetails snapshot))
  101           ]
  102       hashValue = fnv1a64 payload
  103       asHex = T.pack (padHex16 (showHex hashValue ""))
  104   in ShadowSnapshotId ("shadow:" <> asHex)
  105 
  106 fnv1a64 :: Text -> Word64
  107 fnv1a64 =
  108   T.foldl'
  109     (\acc ch -> (acc `xor` fromIntegral (ord ch)) * 1099511628211)
  110     1469598103934665603
  111 
  112 padHex16 :: String -> String
  113 padHex16 raw =
  114   let n = length raw
  115   in replicate (max 0 (16 - n)) '0' <> raw

  136 computeShadowLegitimacyPenalty :: ShadowDivergence -> Double
  137 computeShadowLegitimacyPenalty divergence =
  138   let penalty = 0.0
  139       penalty' = if sdFamilyMismatch divergence then penalty + shadowPenaltyFamilyMismatch else penalty
  140       penalty'' = if sdForceMismatch divergence then penalty' + shadowPenaltyForceMismatch else penalty'
  141       penalty''' = if sdClauseMismatch divergence then penalty'' + shadowPenaltyClauseMismatch else penalty''
  142       penalty'''' = if sdLayerMismatch divergence then penalty''' + shadowPenaltyLayerMismatch else penalty'''
  143       penalty''''' = if sdWarrantedMismatch divergence then penalty'''' + shadowPenaltyWarrantedMismatch else penalty''''
  144   in min 1.0 penalty'''''
  145 
  146 computeShadowLegitimacyPenaltyWithSeverity :: ShadowDivergenceSeverity -> ShadowDivergence -> Double
  147 computeShadowLegitimacyPenaltyWithSeverity severity divergence =
  148   case severity of
  149     ShadowSeverityClean -> 0.0
  150     ShadowSeverityAdvisory -> 0.0
  151     ShadowSeveritySafety -> computeShadowLegitimacyPenalty divergence
  152     ShadowSeverityContract -> computeShadowLegitimacyPenalty divergence
  153     ShadowSeverityUnavailable -> computeShadowLegitimacyPenalty divergence
