   15 fromMorphologyData :: MorphologyData -> LexicalRuntimeData
   16 fromMorphologyData md = LexicalRuntimeData
   17   { lrdLanguage      = LanguageCode "ru"
   18   , lrdNominative    = mdNominative md
   19   , lrdGenitive      = mdGenitive md
   20   , lrdPrepositional = mdPrepositional md
   21   }

   23 buildLexicalRuntimeData :: LexicalRuntimeData
   24 buildLexicalRuntimeData = foldl' insertGroup (emptyLexicalRuntimeData (LanguageCode "ru")) (Map.toList grouped)
   25   where
   26     grouped = foldl' (\acc (surface, lemma, _pos, caseTag) ->
   27                         Map.insertWith (++) lemma [(surface, caseTag)] acc)
   28                      Map.empty generatedLexemeEntries
   29     insertGroup lrd (_lemma, forms) =
   30       let nom = lookupForm "nominative" forms
   31           gen = lookupForm "genitive" forms
   32           prep = lookupForm "prepositional" forms
   33       in case (nom, gen, prep) of
   34            (Just n, Just g, Just p) ->
   35              lrd { lrdNominative    = Map.insert n n $ Map.insert g n $ Map.insert p n $ lrdNominative lrd
   36                  , lrdGenitive      = Map.insert n g (lrdGenitive lrd)
   37                  , lrdPrepositional  = Map.insert n p (lrdPrepositional lrd)
   38                  }
   39            _ -> lrd
   40     lookupForm tag = lookup tag . map (\(sf, ct) -> (ct, sf))
   41 
   42 loadLexicalRuntimeData :: IO LexicalRuntimeData
   43 loadLexicalRuntimeData = do
   44   md <- loadMorphologyData
   45   pure (fromMorphologyData md)
