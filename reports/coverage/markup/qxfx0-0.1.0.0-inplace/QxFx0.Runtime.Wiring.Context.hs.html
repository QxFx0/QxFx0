  100 initRuntimeContext :: FilePath -> IO RuntimeContext
  101 initRuntimeContext path = mask $ \restore -> do
  102   runtimeMode <- resolveRuntimeMode
  103   healthCache <- newMVar Nothing
  104   nixCache <- newNixCache 300 1000
  105   dbPool <- newDBPool path 2
  106   httpManager <- newManager tlsManagerSettings
  107   timeSource <- resolveTimeSource
  108   restore (buildRuntimeContext path runtimeMode healthCache nixCache dbPool httpManager timeSource)
  109     `onException` (closeManager httpManager `finally` closeDBPool dbPool)
  110 
  111 resolveTimeSource :: IO TimeSource
  112 resolveTimeSource = do
  113   mFixed <- lookupEnv "QXFX0_TEST_FIXED_TIME"
  114   case mFixed of
  115     Nothing -> pure Clock.getCurrentTime
  116     Just epochStr ->
  117       case (reads epochStr :: [(Integer, String)]) of
  118         [(epochSeconds, "")] -> do
  119           ref <- newIORef epochSeconds
  120           pure $ do
  121             current <- readIORef ref
  122             writeIORef ref (current + 1)
  123             pure (Clock.addUTCTime (realToFrac current) (Clock.UTCTime (ModifiedJulianDay 0) 0))
  124         _ -> pure Clock.getCurrentTime
  125 
  126 releaseRuntimeContext :: RuntimeContext -> IO ()
  127 releaseRuntimeContext ctx =
  128   closeManager (rtwHttpManager (rcWorkers ctx))
  129     `finally` closeDBPool (rtwDbPool (rcWorkers ctx))

  131 buildRuntimeContext
  132   :: FilePath
  133   -> RuntimeMode
  134   -> APIHealthCache
  135   -> NixCache
  136   -> WorkerDBPool
  137   -> Manager
  138   -> TimeSource
  139   -> IO RuntimeContext
  140 buildRuntimeContext path runtimeMode healthCache nixCache dbPool httpManager timeSource = do
  141   sessionLock <- newSessionLockManager
  142   runtimeTurnState <- newMVar RuntimeTurnState
  143     { rtsConsciousLoop = initialLoop
  144     , rtsIntuition = defaultIntuitiveState
  145     }
  146   agdaWitness <- newMVar Nothing
  147   nixPathVar <- newMVar Nothing
  148   soufflePathVar <- newMVar Nothing
  149   pure RuntimeContext
  150     { rcDbPath = path
  151     , rcMode = runtimeMode
  152     , rcCaches = RuntimeCaches
  153         { rtcHealth = healthCache
  154         , rtcNix = nixCache
  155         }
  156     , rcWorkers = RuntimeWorkers
  157         { rtwDbPool = dbPool
  158         , rtwHttpManager = httpManager
  159         }
  160     , rcLocks = RuntimeLocks
  161         { rtlSession = sessionLock
  162         }
  163     , rcTurn = TurnRuntime
  164         { rtrState = runtimeTurnState
  165         , rtrAgdaWitness = agdaWitness
  166         , rtrNixPath = nixPathVar
  167         , rtrSoufflePath = soufflePathVar
  168         }
  169     , rcTimeSource = timeSource
  170     }

  172 withRuntimeSession :: RuntimeContext -> Text -> IO a -> IO a
  173 withRuntimeSession ctx sessionId action = do
  174   lockEnabled <- readSessionLockFlag (rcMode ctx)
  175   if lockEnabled
  176     then withSessionLock (rtlSession (rcLocks ctx)) sessionId action
  177     else action

  194 modifyConsciousLoop :: RuntimeContext -> (ConsciousnessLoop -> IO (ConsciousnessLoop, a)) -> IO a
  195 modifyConsciousLoop ctx f =
  196   modifyMVar (rtrState (rcTurn ctx)) $ \turnState -> do
  197     (nextLoop, result) <- f (rtsConsciousLoop turnState)
  198     pure (turnState { rtsConsciousLoop = nextLoop }, result)

  200 modifyIntuition :: RuntimeContext -> (IntuitiveState -> IO (IntuitiveState, a)) -> IO a
  201 modifyIntuition ctx f =
  202   modifyMVar (rtrState (rcTurn ctx)) $ \turnState -> do
  203     (nextIntuition, result) <- f (rtsIntuition turnState)
  204     pure (turnState { rtsIntuition = nextIntuition }, result)

  206 commitRuntimeTurnState :: RuntimeContext -> ConsciousnessLoop -> IntuitiveState -> ResponseObservation -> IO ()
  207 commitRuntimeTurnState ctx previewLoop previewIntuition observation =
  208   modifyMVar_ (rtrState (rcTurn ctx)) $ \turnState ->
  209     pure
  210       turnState
  211         { rtsIntuition = previewIntuition
  212         , rtsConsciousLoop = updateAfterResponse previewLoop observation
  213         }

  215 hydrateRuntimeTurnState :: RuntimeContext -> SystemState -> IO ()
  216 hydrateRuntimeTurnState ctx ss =
  217   -- Consciousness loop itself is ephemeral; on restore we re-seed it with persisted turn continuity.
  218   modifyMVar_ (rtrState (rcTurn ctx)) $ \turnState ->
  219     pure
  220       turnState
  221         { rtsConsciousLoop = initialLoop { clDialogueTurn = ssTurnCount ss }
  222         , rtsIntuition = maybe defaultIntuitiveState id (ssIntuitionState ss)
  223         }

  225 updateHistoryStrict :: Text -> Seq.Seq Text -> Seq.Seq Text
  226 updateHistoryStrict !newEntry !history =
  227   let !truncated = T.take maxHistoryEntryChars newEntry
  228       !updated = truncated Seq.<| history
  229       !bounded = Seq.take 50 updated
  230   in bounded
  231 
  232 maxHistoryEntryChars :: Int
  233 maxHistoryEntryChars = 4096

  235 resolveNixPath :: RuntimeContext -> IO (Maybe FilePath)
  236 resolveNixPath ctx = do
  237   mStored <- readMVar (rtrNixPath (rcTurn ctx))
  238   case mStored of
  239     Just p -> pure (Just p)
  240     Nothing -> do
  241       raw <- getNixGuardPath
  242       exists <- doesFileExist raw
  243       let result = if exists then Just raw else Nothing
  244       modifyMVar_ (rtrNixPath (rcTurn ctx)) $ \_ -> pure result
  245       pure result
  246 
  247 resolveSouffleExecutableCached :: RuntimeContext -> IO (Either Text FilePath)
  248 resolveSouffleExecutableCached ctx = do
  249   cached <- readMVar (rtrSoufflePath (rcTurn ctx))
  250   case cached of
  251     Just result -> pure result
  252     Nothing -> do
  253       result <- Datalog.resolveSouffleExecutable
  254       modifyMVar_ (rtrSoufflePath (rcTurn ctx)) $ \_ -> pure (Just result)
  255       pure result

  257 readSessionLockFlag :: RuntimeMode -> IO Bool
  258 readSessionLockFlag runtimeMode = do
  259   mVal <- lookupEnv "QXFX0_SESSION_LOCK"
  260   pure $ case mVal of
  261     Nothing -> True
  262     Just raw ->
  263       case map Char.toLower raw of
  264         "off" -> runtimeMode /= StrictRuntime
  265         "0" -> runtimeMode /= StrictRuntime
  266         "on" -> True
  267         "1" -> True
  268         _ -> True
