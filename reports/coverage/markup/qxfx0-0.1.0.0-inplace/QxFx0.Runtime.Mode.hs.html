    1 {-# LANGUAGE DerivingStrategies #-}

    4 module QxFx0.Runtime.Mode
    5   ( RuntimeMode(..)
    6   , resolveRuntimeMode
    7   , runtimeModeText
    8   , isStrictRuntimeMode
    9   ) where
   10 
   11 import Data.Text (Text)
   12 import qualified Data.Text as T
   13 import System.Environment (lookupEnv)
   14 
   15 data RuntimeMode
   16   = DegradedRuntime
   17   | StrictRuntime
   18   deriving stock (Eq, Show)
   19 
   20 resolveRuntimeMode :: IO RuntimeMode
   21 resolveRuntimeMode = do
   22   mMode <- lookupEnv "QXFX0_RUNTIME_MODE"
   23   pure $ case fmap (T.toLower . T.strip . T.pack) mMode of
   24     Just "degraded" -> DegradedRuntime
   25     Just "test-degraded" -> DegradedRuntime
   26     Just "strict" -> StrictRuntime
   27     Just "clockwork" -> StrictRuntime
   28     _ -> StrictRuntime
   29 
   30 runtimeModeText :: RuntimeMode -> Text
   31 runtimeModeText DegradedRuntime = "degraded"
   32 runtimeModeText StrictRuntime = "strict"
   33 
   34 isStrictRuntimeMode :: RuntimeMode -> Bool
   35 isStrictRuntimeMode StrictRuntime = True
   36 isStrictRuntimeMode DegradedRuntime = False
