   43 data EpistemicStatus
   44   = Known !Double
   45   | Probable !Double
   46   | Uncertain !Double
   47   | Unknown !Double
   48   | Speculative !Double
   49   deriving stock (Eq, Ord, Show, Read, Generic)
   50   deriving anyclass (NFData)
   51 
   52 epistemicConfidence :: EpistemicStatus -> Double
   53 epistemicConfidence (Known confidence) = confidence
   54 epistemicConfidence (Probable confidence) = confidence
   55 epistemicConfidence (Uncertain confidence) = confidence
   56 epistemicConfidence (Unknown confidence) = confidence
   57 epistemicConfidence (Speculative confidence) = confidence
   58 
   59 instance ToJSON EpistemicStatus where
   60   toJSON (Known confidence) = object ["tag" .= ("Known" :: Text), "confidence" .= confidence]
   61   toJSON (Probable confidence) = object ["tag" .= ("Probable" :: Text), "confidence" .= confidence]
   62   toJSON (Uncertain confidence) = object ["tag" .= ("Uncertain" :: Text), "confidence" .= confidence]
   63   toJSON (Unknown confidence) = object ["tag" .= ("Unknown" :: Text), "confidence" .= confidence]
   64   toJSON (Speculative confidence) = object ["tag" .= ("Speculative" :: Text), "confidence" .= confidence]
   65 
   66 instance FromJSON EpistemicStatus where
   67   parseJSON = withObject "EpistemicStatus" $ \objectValue -> do
   68     tag <- objectValue .: "tag"
   69     confidence <- objectValue .: "confidence"
   70     case tag :: Text of
   71       "Known" -> pure (Known confidence)
   72       "Probable" -> pure (Probable confidence)
   73       "Uncertain" -> pure (Uncertain confidence)
   74       "Unknown" -> pure (Unknown confidence)
   75       "Speculative" -> pure (Speculative confidence)
   76       _ -> fail ("unknown EpistemicStatus: " ++ T.unpack tag)
