   23 toPipelineIO :: RuntimeContext -> PipelineIO
   24 toPipelineIO ctx = PipelineIO
   25   { pioRuntimeMode = case rcMode ctx of
   26       StrictRuntime -> RuntimeStrict
   27       DegradedRuntime -> RuntimeDegraded
   28   , pioShadowPolicy = case rcMode ctx of
   29       StrictRuntime -> ShadowBlockOnUnavailableOrDivergence
   30       DegradedRuntime -> ShadowObserve
   31   , pioLocalRecoveryPolicy = LocalRecoveryEnabled
   32   , pioInterpreter = handleTurnEffect ctx
   33   , pioUpdateHistory = updateHistoryStrict
   34   , pioModifyConsciousLoop = modifyConsciousLoop ctx
   35   , pioModifyIntuition = modifyIntuition ctx
   36   , pioConatusPrior = defaultConatusPrior
   37   }
