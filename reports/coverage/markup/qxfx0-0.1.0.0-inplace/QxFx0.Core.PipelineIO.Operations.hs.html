   90 runShadowVerification :: PipelineIO -> CanonicalMoveFamily -> IllocutionaryForce -> [AtomTag] -> IO ShadowResult
   91 runShadowVerification pio family force atomTags = do
   92   result <- resolveTurnEffect pio (TurnReqShadow family force atomTags)
   93   case result of
   94     TurnResShadow datalogVerdict shadowStatus divergence snapshotId diagnostics ->
   95       pure ShadowResult
   96         { srDatalogVerdict = datalogVerdict
   97         , srStatus = shadowStatus
   98         , srDivergence = divergence
   99         , srSnapshotId = snapshotId
  100         , srDiagnostics = diagnostics
  101         }
  102     _ ->
  103       pure ShadowResult
  104         { srDatalogVerdict = Nothing
  105         , srStatus = ShadowUnavailable
  106         , srDivergence = emptyShadowDivergence { sdKind = ShadowBridgeSkew }
  107         , srSnapshotId = ShadowSnapshotId "shadow:unexpected_effect"
  108         , srDiagnostics = ["unexpected_shadow_effect_result"]
  109         }

  111 verifyPipelineAgda :: PipelineIO -> IO AgdaVerificationStatus
  112 verifyPipelineAgda pio = do
  113   result <- resolveTurnEffect pio TurnReqAgdaVerify
  114   case result of
  115     TurnResAgdaVerify agdaStatus -> pure agdaStatus
  116     _ -> pure AgdaInvalid
  117 
  118 savePipelineState :: PipelineIO -> SystemState -> Text -> Maybe TurnProjection -> IO (Either PersistenceDiagnostic SystemState)
  119 savePipelineState pio ss sid mProj = do
  120   result <- resolveTurnEffect pio (TurnReqSaveState ss sid mProj)
  121   case result of
  122     TurnResSaveState saved -> pure saved
  123     _ -> pure (Left (PdSaveFailed StageUnknown (Just "savePipelineState") (Just "unexpected_save_state_effect_result")))
  124 
  125 checkpointPipelineState :: PipelineIO -> Int -> IO ()
  126 checkpointPipelineState pio turnCount = do
  127   _ <- resolveTurnEffect pio (TurnReqCheckpoint turnCount)
  128   pure ()

  130 resolvePipelineNixPath :: PipelineIO -> IO (Maybe FilePath)
  131 resolvePipelineNixPath _ = pure Nothing

  133 runPipelineNixCheck :: PipelineIO -> Maybe FilePath -> Text -> Double -> Double -> IO NixGuardStatus
  134 runPipelineNixCheck pio _ concept agency tension = do
  135   result <- resolveTurnEffect pio (TurnReqNixGuard concept agency tension)
  136   case result of
  137     TurnResNixGuard nixStatus -> pure nixStatus
  138     _ -> pure (Unavailable "unexpected_nix_effect_result")

  140 modifyPipelineConsciousLoop :: PipelineIO -> (forall a. (ConsciousnessLoop -> IO (ConsciousnessLoop, a)) -> IO a)
  141 modifyPipelineConsciousLoop _ f = do
  142   let cl = initialLoop
  143   (_, result) <- f cl
  144   pure result

  146 modifyPipelineIntuition :: PipelineIO -> (forall a. (IntuitiveState -> IO (IntuitiveState, a)) -> IO a)
  147 modifyPipelineIntuition _ f = do
  148   let intuitive = defaultIntuitiveState
  149   (_, result) <- f intuitive
  150   pure result

  152 checkPipelineApiHealth :: PipelineIO -> IO Bool
  153 checkPipelineApiHealth pio = do
  154   result <- resolveTurnEffect pio TurnReqApiHealth
  155   case result of
  156     TurnResApiHealth ok -> pure ok
  157     _ -> pure False
