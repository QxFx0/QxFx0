   63 -- | Mutable guardrail counters kept in 'SystemState'.
   64 data GuardrailState = GuardrailState
   65   { gsLastProposalTurn      :: !Int
   66     -- ^ Turn of the most recent proposal submission.
   67   , gsProposalsThisWindow   :: !Int
   68     -- ^ Count of submissions within the current window.
   69   , gsWindowStart           :: !Int
   70     -- ^ Turn when the current window began.
   71   , gsConsecutiveRejections :: !Int
   72     -- ^ How many consecutive proposals were rejected.
   73   , gsCooldownExpiry        :: !Int
   74     -- ^ Turn after which the circuit breaker re-closes.
   75     --   0 = no active cooldown.
   76   , gsQuarantine            :: ![(Int, CalibrationId)]
   77     -- ^ (submittedTurn, proposalId) pairs still in quarantine.
   78   }
   79   deriving stock (Eq, Show, Generic)
   80     deriving anyclass (NFData)
   81 
   82 instance ToJSON GuardrailState where
   83   toJSON s = object
   84     [ "lastProposalTurn"      .= gsLastProposalTurn s
   85     , "proposalsThisWindow"   .= gsProposalsThisWindow s
   86     , "windowStart"           .= gsWindowStart s
   87     , "consecutiveRejections" .= gsConsecutiveRejections s
   88     , "cooldownExpiry"        .= gsCooldownExpiry s
   89     , "quarantine"            .= gsQuarantine s
   90     ]

  102 emptyGuardrailState :: GuardrailState

  104   { gsLastProposalTurn      = 0
  105   , gsProposalsThisWindow   = 0
  106   , gsWindowStart           = 0
  107   , gsConsecutiveRejections = 0
  108   , gsCooldownExpiry        = 0
  109   , gsQuarantine            = []
  110   }

  112 -- | Predicate: can a new proposal be submitted at the given turn?
  113 canSubmitProposal :: GuardrailState -> Int -> Bool
  114 canSubmitProposal gs turn =
  115   not (circuitBreakerOpen gs turn) && not (rateLimitExceeded gs turn)
  116 
  117 -- | Record a new proposal submission, updating counters and window.
  118 recordProposalSubmission :: GuardrailState -> Int -> CalibrationId -> GuardrailState
  119 recordProposalSubmission gs turn proposalId =
  120   let (newWindowStart, newWindowCount) =
  121         if turn - gsWindowStart gs > proposalWindowTurns
  122            then (turn, 1)
  123            else (gsWindowStart gs, gsProposalsThisWindow gs + 1)
  124       quarantine' = pruneQuarantine turn ((turn, proposalId) : gsQuarantine gs)
  125    in gs
  126         { gsLastProposalTurn    = turn
  127         , gsProposalsThisWindow = newWindowCount
  128         , gsWindowStart         = newWindowStart
  129         , gsQuarantine          = quarantine'
  130         }
  131 
  132 -- | Record a rejection, bumping the consecutive-rejection counter
  133 -- and potentially opening the circuit breaker.
  134 recordRejection :: GuardrailState -> Int -> GuardrailState
  135 recordRejection gs turn =
  136   let newRejections = gsConsecutiveRejections gs + 1
  137       newCooldown =
  138         if newRejections >= maxConsecutiveRejections
  139            then turn + cooldownTurns
  140            else gsCooldownExpiry gs
  141   in gs { gsConsecutiveRejections = newRejections, gsCooldownExpiry = newCooldown }
  142 
  143 -- | Record an acceptance, resetting the rejection streak.
  144 recordAcceptance :: GuardrailState -> GuardrailState
  145 recordAcceptance gs = gs { gsConsecutiveRejections = 0 }
  146 
  147 -- | Add a proposal to quarantine (done automatically by
  148 -- 'recordProposalSubmission').  This function is exposed for
  149 -- re-hydration from persisted state if needed.
  150 quarantineProposal :: GuardrailState -> Int -> CalibrationId -> GuardrailState
  151 quarantineProposal gs turn proposalId =
  152   gs { gsQuarantine = pruneQuarantine turn ((turn, proposalId) : gsQuarantine gs) }
  153 
  154 -- | Check whether a specific proposal has completed its quarantine.
  155 isQuarantineExpired :: GuardrailState -> Int -> CalibrationId -> Bool
  156 isQuarantineExpired gs turn proposalId =
  157   case lookup proposalId (map swap (gsQuarantine gs)) of
  158     Nothing    -> True
  159     Just subTurn -> turn - subTurn >= minQuarantineTurns
  160   where
  161     swap (a, b) = (b, a)
  162 
  163 pruneQuarantine :: Int -> [(Int, CalibrationId)] -> [(Int, CalibrationId)]
  164 pruneQuarantine turn =
  165   take maxQuarantineEntries
  166     . filter (\(submittedTurn, _) -> turn - submittedTurn <= minQuarantineTurns)
  167 
  168 -- Internal helpers
  169 
  170 circuitBreakerOpen :: GuardrailState -> Int -> Bool
  171 circuitBreakerOpen gs turn = gsCooldownExpiry gs > 0 && turn <= gsCooldownExpiry gs
  172 
  173 rateLimitExceeded :: GuardrailState -> Int -> Bool
  174 rateLimitExceeded gs turn =
  175   let inCurrentWindow = turn - gsWindowStart gs <= proposalWindowTurns
  176   in inCurrentWindow && gsProposalsThisWindow gs >= maxProposalsPerWindow
