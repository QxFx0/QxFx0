    1 {-# LANGUAGE DerivingStrategies, OverloadedStrings, StrictData #-}
    2 module QxFx0.Bridge.TxStatement
    3   ( TxStmt
    4   , prepareTx
    5   , bindTextOrFail
    6   , bindIntOrFail
    7   , bindDoubleOrFail
    8   , stepOrFail
    9   , rollbackAndFail
   10   ) where
   11 
   12 import qualified QxFx0.Bridge.NativeSQLite as NSQL
   13 import QxFx0.ExceptionPolicy (throwQxFx0, QxFx0Exception(PersistenceTxError))
   14 import QxFx0.Types.Persistence (PersistenceStage(..))
   15 import Data.Text (Text)
   16 import qualified Data.Text as T
   17 import Foreign.C.Types (CInt)
   18 
   19 data TxStmt = TxStmt
   20   { txsStmt :: !NSQL.Statement
   21   , txsDb   :: !NSQL.Database
   22   , txsCtx  :: !Text
   23   }
   24 
   25 prepareTx :: NSQL.Database -> Text -> Text -> IO TxStmt
   26 prepareTx db ctx sql = do
   27   mStmt <- NSQL.prepare db sql
   28   case mStmt of
   29     Left err -> do
   30       _ <- NSQL.execSql db "ROLLBACK;"
   31       throwQxFx0 (PersistenceTxError (stageFromContext ctx) err)
   32     Right s -> pure TxStmt { txsStmt = s, txsDb = db, txsCtx = ctx }
   33 
   34 rollbackAndFail :: TxStmt -> Text -> IO a
   35 rollbackAndFail ts msg = do
   36   NSQL.finalize (txsStmt ts)
   37   _ <- NSQL.execSql (txsDb ts) "ROLLBACK;"
   38   throwQxFx0 (PersistenceTxError (stageFromContext (txsCtx ts)) msg)
   39 
   40 bindTextOrFail :: TxStmt -> CInt -> Text -> IO ()
   41 bindTextOrFail ts ix v =
   42   NSQL.bindText (txsStmt ts) ix v >>= either
   43     (\err -> rollbackAndFail ts ("bindText[" <> T.pack (show ix) <> "] failed: " <> err))
   44     (const (pure ()))
   45 
   46 bindIntOrFail :: TxStmt -> CInt -> Int -> IO ()
   47 bindIntOrFail ts ix v =
   48   NSQL.bindInt (txsStmt ts) ix v >>= either
   49     (\err -> rollbackAndFail ts ("bindInt[" <> T.pack (show ix) <> "] failed: " <> err))
   50     (const (pure ()))
   51 
   52 bindDoubleOrFail :: TxStmt -> CInt -> Double -> IO ()
   53 bindDoubleOrFail ts ix v =
   54   NSQL.bindDouble (txsStmt ts) ix v >>= either
   55     (\err -> rollbackAndFail ts ("bindDouble[" <> T.pack (show ix) <> "] failed: " <> err))
   56     (const (pure ()))
   57 
   58 stepOrFail :: TxStmt -> IO ()
   59 stepOrFail ts = do
   60   stepResult <- NSQL.step (txsStmt ts)
   61   NSQL.finalize (txsStmt ts)
   62   case stepResult of
   63     Left err -> do
   64       _ <- NSQL.execSql (txsDb ts) "ROLLBACK;"
   65       throwQxFx0 (PersistenceTxError (stageFromContext (txsCtx ts)) err)
   66     Right _ -> pure ()
   67 
   68 stageFromContext :: Text -> PersistenceStage
   69 stageFromContext ctx
   70   | "turn_quality" `T.isPrefixOf` ctx = StageTurnQualityUpsert
   71   | "shadow_divergence_log" `T.isPrefixOf` ctx = StageShadowDivergenceInsert
   72   | "saveKV:" `T.isPrefixOf` ctx = StageStateBlobUpsert
   73   | "touchRuntimeSessionActivity" `T.isPrefixOf` ctx = StageSessionTouch
   74   | "delete_turn_quality_above" `T.isPrefixOf` ctx = StageRollbackTurnQuality
   75   | "delete_shadow_divergence_above" `T.isPrefixOf` ctx = StageRollbackShadowDivergence
   76   | otherwise = StageUnknown
