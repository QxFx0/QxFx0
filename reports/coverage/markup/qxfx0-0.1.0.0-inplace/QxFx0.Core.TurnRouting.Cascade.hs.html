   44 runFamilyCascade :: RoutingPhase -> SystemState -> UserState -> InputPropositionFrame -> AtomSet -> [Text] -> Text
   45                  -> Maybe ConsciousnessNarrative -> Double -> Bool -> Salience -> FamilyCascade
   46 runFamilyCascade RoutingPhase{..} systemState _nextUserState frame _atomSet _history _input narrative intuitionPosterior isNixBlocked salience =
   47   let parserLockedFamily =
   48         if ipfConfidence frame >= parserHighConfidenceThreshold
   49              && ipfPropositionType frame /= T.pack "PlainAssert"
   50           then Just (ipfCanonicalFamily frame)
   51           else Nothing
   52       familyAfterIdentity =
   53         case parserLockedFamily of
   54           Just parserFamily -> parserFamily
   55           Nothing ->
   56             maybe rpFamilyAfterStrategy (`preferFamily` rpFamilyAfterStrategy) (identityFamilyHint rpIdentitySignal0)
   57       narrativeHint = narrative >>= narrativeFamilyHint
   58       familyAfterNarrative =
   59         case parserLockedFamily of
   60           Just parserFamily -> parserFamily
   61           Nothing -> maybe familyAfterIdentity (`preferFamily` familyAfterIdentity) narrativeHint
   62       familyAfterIntuition =
   63         case parserLockedFamily of
   64           Just parserFamily -> parserFamily
   65           Nothing -> maybe familyAfterNarrative (`preferFamily` familyAfterNarrative) (intuitionFamilyHint intuitionPosterior)
   66       familyAfterPrincipled =
   67         case parserLockedFamily of
   68           Just parserFamily -> parserFamily
   69           Nothing -> applyPrincipledFamilyModulated familyDivergenceEnabled salience rpPrincipledModeResult familyAfterIntuition
   70       guardReportPre = buildGuardReport (ssLastGuardReport systemState) (ssEgo systemState) rpPreEgo
   71       familyAfterGuard = applyGuardGatingModulated familyDivergenceEnabled salience guardReportPre familyAfterPrincipled
   72       familyCascade = fromMaybe familyAfterGuard (antiStuck (ssConsecutiveReflect systemState) rpPreEgo familyAfterGuard)
   73       finalFamily = if isNixBlocked then CMRepair else familyCascade
   74       familyDivergenceEnabled = False
   75    in FamilyCascade
   76         { fcFamilyAfterIdentity = familyAfterIdentity
   77         , fcFamilyAfterNarrative = familyAfterNarrative
   78         , fcFamilyAfterIntuition = familyAfterIntuition
   79         , fcFamilyAfterPrincipled = familyAfterPrincipled
   80         , fcGuardReportPre = guardReportPre
   81         , fcFamilyAfterGuard = familyAfterGuard
   82         , fcFamilyCascade = familyCascade
   83         , fcFinalFamily = finalFamily
   84         }

   86 applyPrincipledFamily :: Maybe PrincipledMode -> CanonicalMoveFamily -> CanonicalMoveFamily
   87 applyPrincipledFamily mode family =
   88   case mode of
   89     Just HoldGround
   90       | family `elem` [CMConfront, CMHypothesis] -> CMGround
   91       | otherwise -> family
   92     Just OpenToUpdate
   93       | family == CMGround -> CMClarify
   94       | otherwise -> family
   95     Just AcknowledgeAndHold
   96       | family `elem` [CMConfront, CMDeepen] -> CMContact
   97       | otherwise -> family
   98     Nothing -> family

  100 -- | Salience-modulated principled-family application.
  101 -- When the controller strongly prefers Holistic (above
  102 -- 'smModulationHolisticBiasFloor'), principled constraints are
  103 -- relaxed so intuition/narrative signals retain more influence.
  104 -- Conatus-gate events force the strict formal path.
  105 --
  106 -- Threshold sourced from 'defaultSalienceModulation'; Phase 7
  107 -- calibration goes through that single record rather than this
  108 -- function.
  109 applyPrincipledFamilyModulated :: Bool -> Salience -> Maybe PrincipledMode -> CanonicalMoveFamily -> CanonicalMoveFamily
  110 applyPrincipledFamilyModulated familyDivergenceEnabled salience mode family =
  111   case salienceDriver salience of
  112     DrivenByConatusGate -> applyPrincipledFamily mode family
  113     _ | familyDivergenceEnabled && salienceHolisticBias salience > smModulationHolisticBiasFloor defaultSalienceModulation -> family
  114     _ -> applyPrincipledFamily mode family

  116 applyGuardGating :: IdentityGuardReport -> CanonicalMoveFamily -> CanonicalMoveFamily
  117 applyGuardGating guardReport family
  118   | igrWithinBounds guardReport = family
  119   | GuardAgencyCollapse `elem` igrWarnings guardReport = CMRepair
  120   | GuardHighTensionDrift `elem` igrWarnings guardReport =
  121       case family of
  122         CMConfront -> CMAnchor
  123         CMDeepen -> CMGround
  124         _ -> family
  125   | GuardTransitionOutsideManifold `elem` igrWarnings guardReport =
  126       case family of
  127         CMConfront -> CMAnchor
  128         CMHypothesis -> CMGround
  129         _ -> family
  130   | otherwise = family
  131 
  132 -- | Salience-modulated guard gating.
  133 -- Holistic-preferring turns (above
  134 -- 'smModulationHolisticBiasFloor') keep all families except
  135 -- agency-collapse (which is always repaired).  Formal/Conatus
  136 -- turns apply the full guard cascade.
  137 --
  138 -- Threshold sourced from 'defaultSalienceModulation'.
  139 applyGuardGatingModulated :: Bool -> Salience -> IdentityGuardReport -> CanonicalMoveFamily -> CanonicalMoveFamily
  140 applyGuardGatingModulated familyDivergenceEnabled salience guard family
  141   | salienceDriver salience == DrivenByConatusGate = applyGuardGating guard family
  142   | familyDivergenceEnabled && salienceHolisticBias salience > smModulationHolisticBiasFloor defaultSalienceModulation =
  143       if GuardAgencyCollapse `elem` igrWarnings guard then CMRepair else family
  144   | otherwise = applyGuardGating guard family

  146 buildGuardReport :: Maybe IdentityGuardReport -> EgoState -> EgoState -> IdentityGuardReport
  147 buildGuardReport lastGuard oldEgo newEgo =
  148   let (baseAgency, baseTension) =
  149         case lastGuard of
  150           Just _ -> (egoAgency oldEgo, egoTension oldEgo)
  151           Nothing -> (identityGuardDefaultAgencyBaseline, identityGuardDefaultTensionBaseline)
  152    in buildIdentityGuardReportSimple defaultIdentityGuardCalibration
  153         baseAgency (egoAgency newEgo) baseTension (egoTension newEgo)
