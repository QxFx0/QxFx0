   38 runFamilyCascade :: RoutingPhase -> SystemState -> UserState -> InputPropositionFrame -> AtomSet -> [Text] -> Text
   39                  -> Maybe ConsciousnessNarrative -> Double -> Bool -> FamilyCascade
   40 runFamilyCascade RoutingPhase{..} systemState _nextUserState frame _atomSet _history _input narrative intuitionPosterior isNixBlocked =
   41   let parserLockedFamily =
   42         if ipfConfidence frame >= parserHighConfidenceThreshold
   43              && ipfPropositionType frame /= T.pack "PlainAssert"
   44           then Just (ipfCanonicalFamily frame)
   45           else Nothing
   46       familyAfterIdentity =
   47         case parserLockedFamily of
   48           Just parserFamily -> parserFamily
   49           Nothing ->
   50             maybe rpFamilyAfterStrategy (`preferFamily` rpFamilyAfterStrategy) (identityFamilyHint rpIdentitySignal0)
   51       narrativeHint = narrative >>= narrativeFamilyHint
   52       familyAfterNarrative =
   53         case parserLockedFamily of
   54           Just parserFamily -> parserFamily
   55           Nothing -> maybe familyAfterIdentity (`preferFamily` familyAfterIdentity) narrativeHint
   56       familyAfterIntuition =
   57         case parserLockedFamily of
   58           Just parserFamily -> parserFamily
   59           Nothing -> maybe familyAfterNarrative (`preferFamily` familyAfterNarrative) (intuitionFamilyHint intuitionPosterior)
   60       familyAfterPrincipled =
   61         case parserLockedFamily of
   62           Just parserFamily -> parserFamily
   63           Nothing -> applyPrincipledFamily rpPrincipledModeResult familyAfterIntuition
   64       guardReportPre = buildGuardReport (ssLastGuardReport systemState) (ssEgo systemState) rpPreEgo
   65       familyAfterGuard = applyGuardGating guardReportPre familyAfterPrincipled
   66       familyCascade = fromMaybe familyAfterGuard (antiStuck (ssConsecutiveReflect systemState) rpPreEgo familyAfterGuard)
   67       finalFamily = if isNixBlocked then CMRepair else familyCascade
   68    in FamilyCascade
   69         { fcFamilyAfterIdentity = familyAfterIdentity
   70         , fcFamilyAfterNarrative = familyAfterNarrative
   71         , fcFamilyAfterIntuition = familyAfterIntuition
   72         , fcFamilyAfterPrincipled = familyAfterPrincipled
   73         , fcGuardReportPre = guardReportPre
   74         , fcFamilyAfterGuard = familyAfterGuard
   75         , fcFamilyCascade = familyCascade
   76         , fcFinalFamily = finalFamily
   77         }

   79 applyPrincipledFamily :: Maybe PrincipledMode -> CanonicalMoveFamily -> CanonicalMoveFamily
   80 applyPrincipledFamily mode family =
   81   case mode of
   82     Just HoldGround
   83       | family `elem` [CMConfront, CMHypothesis] -> CMGround
   84       | otherwise -> family
   85     Just OpenToUpdate
   86       | family == CMGround -> CMClarify
   87       | otherwise -> family
   88     Just AcknowledgeAndHold
   89       | family `elem` [CMConfront, CMDeepen] -> CMContact
   90       | otherwise -> family
   91     Nothing -> family
   92 
   93 applyGuardGating :: IdentityGuardReport -> CanonicalMoveFamily -> CanonicalMoveFamily
   94 applyGuardGating guardReport family
   95   | igrWithinBounds guardReport = family
   96   | GuardAgencyCollapse `elem` igrWarnings guardReport = CMRepair
   97   | GuardHighTensionDrift `elem` igrWarnings guardReport =
   98       case family of
   99         CMConfront -> CMAnchor
  100         CMDeepen -> CMGround
  101         _ -> family
  102   | GuardTransitionOutsideManifold `elem` igrWarnings guardReport =
  103       case family of
  104         CMConfront -> CMAnchor
  105         CMHypothesis -> CMGround
  106         _ -> family
  107   | otherwise = family

  109 buildGuardReport :: Maybe IdentityGuardReport -> EgoState -> EgoState -> IdentityGuardReport
  110 buildGuardReport lastGuard oldEgo newEgo =
  111   let (baseAgency, baseTension) =
  112         case lastGuard of
  113           Just _ -> (egoAgency oldEgo, egoTension oldEgo)
  114           Nothing -> (identityGuardDefaultAgencyBaseline, identityGuardDefaultTensionBaseline)
  115    in buildIdentityGuardReportSimple defaultIdentityGuardCalibration
  116         baseAgency (egoAgency newEgo) baseTension (egoTension newEgo)
