   69 detectPressure :: Text -> [Text] -> Maybe PressureSignal
   70 detectPressure raw history =
   71   let tokens = tokenizeKeywordText raw
   72       hasCorrection  = anyContains tokens correctionMarkers
   73       hasAuthority   = anyContains tokens authorityMarkers
   74       hasEmotional   = anyContains tokens emotionalMarkers
   75       hasInsistence  = anyContains tokens insistenceMarkers
   76       hasNewInfo     = anyContains tokens newInfoMarkers
   77       isRepetitive = case history of
   78         (prev:_) ->
   79           let ws1 = tokens
   80               ws2 = tokenizeKeywordText prev
   81               overlap = length (filter (`elem` ws2) ws1)
   82               total   = max 1 (length ws1)
   83               overlapRatio :: Double
   84               overlapRatio = fromIntegral overlap / fromIntegral total
   85           in overlapRatio > principledRepetitionOverlapThreshold && length ws1 > 4
   86         _ -> False
   87   in if hasAuthority
   88      then Just $ PressureSignal AuthorityCommand principledAuthorityPressureStrength False
   89      else if hasCorrection
   90           then Just $ PressureSignal
   91                DirectCorrection
   92                ( if hasNewInfo
   93                    then principledCorrectionPressureWithNewInfo
   94                    else principledCorrectionPressureWithoutNewInfo
   95                )
   96                hasNewInfo
   97           else if hasEmotional
   98                then Just $ PressureSignal EmotionalDemand principledEmotionalPressureStrength False
   99                else if hasInsistence || isRepetitive
  100                     then Just $ PressureSignal RepetitiveInsistence principledInsistencePressureStrength False
  101                     else Nothing
  102 
  103 anyContains :: [Text] -> [Text] -> Bool
  104 anyContains haystack needles = containsAnyKeywordPhrase haystack needles
  105 
  106 principledMode :: PressureSignal -> PrincipledMode
  107 principledMode signal
  108   | psHasNewInfo signal              = OpenToUpdate
  109   | psType signal == EmotionalDemand = AcknowledgeAndHold
  110   | otherwise                        = HoldGround
  111 
  112 classifyPressure :: Maybe PressureSignal -> [Text] -> PressureState
  113 classifyPressure Nothing _ = NoPressure
  114 classifyPressure (Just sig) _ =
  115   ActivePressure
  116     { pstIsRepeated = psType sig `elem` [RepetitiveInsistence, DirectCorrection]
  117     , pstStrength = psStrength sig
  118     }
  119 
  120 pressureBandFromState :: PressureState -> PressureBand
  121 pressureBandFromState NoPressure = PressNone
  122 pressureBandFromState ActivePressure { pstIsRepeated = True } = PressHeavy
  123 pressureBandFromState ActivePressure {} = PressLight

  125 principledToPromptSection :: PressureSignal -> PrincipledMode -> Text
  126 principledToPromptSection signal mode =
  127   let pTypeText = case lookup (renderPressureType (psType signal)) pressureTypeText of
  128                    Just t -> t; Nothing -> renderPressureType (psType signal)
  129       header = principledHeader <> pTypeText <> "]"
  130   in T.unlines $ [header, ""] ++ modeDirective mode
  131 
  132 modeDirective :: PrincipledMode -> [Text]
  133 modeDirective HoldGround = holdGroundDirective
  134 modeDirective OpenToUpdate = openToUpdateDirective
  135 modeDirective AcknowledgeAndHold = acknowledgeAndHoldDirective
  136 
  137 pressureDescription :: PressureSignal -> Text
  138 pressureDescription signal =
  139   renderPressureType (psType signal) <>
  140   pressureForceLabel <> fmtPct (psStrength signal) <> ")" <>
  141   if psHasNewInfo signal then newInfoTag else ""
