   23 isSafeChar :: Char -> Bool
   24 isSafeChar c = isAscii c && (isAlphaNum c || c == '-' || c == '_')
   25            || (not (isAscii c) && isLetter c)

   27 nixStringLiteral :: Text -> Text
   28 nixStringLiteral t =
   29   let escaped =
   30         t
   31           & T.replace "\\" "\\\\"
   32           & T.replace "\n" "\\n"
   33           & T.replace "\r" "\\r"
   34           & T.replace "\t" "\\t"
   35           & T.replace "\"" "\\\""
   36           & T.replace "${" "\\${"
   37   in "\"" <> escaped <> "\""

   39 checkConstitution :: FilePath -> Text -> Double -> Double -> IO NixGuardStatus
   40 checkConstitution nixPath concept agency tension =
   41   case normalizeConceptKey concept of
   42     Nothing
   43       | T.null (T.strip concept) -> return Allowed
   44       | otherwise -> do
   45           lenient <- isLenientMode
   46           if lenient
   47             then return (Unavailable "constitution concept unsupported; policy check skipped")
   48             else return (Blocked "constitution concept unsupported")
   49     Just conceptKey -> do
   50       let nixExpr = "let agency = " <> T.pack (show agency)
   51                    <> "; tension = " <> T.pack (show tension)
   52                    <> "; data = import " <> nixStringLiteral (T.pack nixPath)
   53                    <> "; key = " <> nixStringLiteral conceptKey
   54                     <> "; match = builtins.filter (c: builtins.toLower c.id == builtins.toLower key) data.concepts;"
   55                     <> "  concept = if builtins.length match > 0 then builtins.elemAt match 0 else null;"
   56                     <> "  agencyOk = concept != null && (concept.minAgency == null || agency >= concept.minAgency);"
   57                     <> "  tensionOk = concept != null && (concept.minTension == null || tension >= concept.minTension);"
   58                     <> " in if concept != null then agencyOk && tensionOk else true"
   59       result <- runNixEval nixExpr
   60       case result of
   61         Right "true"  -> return Allowed
   62         Right "false" -> return $ Blocked $ "constitution blocked: " <> conceptKey
   63         Right other   -> return $ Unavailable $ "nix evaluator unavailable: unexpected_result:" <> other
   64         Left err      -> return $ Unavailable $ "nix evaluator unavailable: " <> classifyNixEvalError err
   65 
   66 getNixGuardStatus :: IO (Either Text NixGuardStatus)
   67 getNixGuardStatus = do
   68   result <- runNixEval "true"
   69   case result of
   70     Right _    -> return (Right Allowed)
   71     Left  err  -> return (Left err)
   72 
   73 runNixEval :: Text -> IO (Either Text Text)
   74 runNixEval nixExpr = do
   75   restrictedResult <- runNixInstantiate True nixExpr
   76   case restrictedResult of
   77     Left err
   78       | isRestrictedFlagUnsupported err ->
   79           runNixInstantiate False nixExpr
   80     _ ->
   81       pure restrictedResult
   82 
   83 runNixInstantiate :: Bool -> Text -> IO (Either Text Text)
   84 runNixInstantiate restricted nixExpr = do
   85   let timeoutSec :: Int
   86       timeoutSec = 5
   87       modeArgs = if restricted then ["--restricted"] else []
   88       modeLabel = if restricted then "restricted" else "unrestricted"
   89   result <- catchIO
   90     (do (exitCode, stdout, stderr) <- readProcessWithExitCode
   91           "timeout" ([show timeoutSec, "nix-instantiate"] <> modeArgs <> ["--eval", "--expr", T.unpack nixExpr]) ""
   92         case exitCode of
   93           ExitSuccess ->
   94             let output = T.strip (T.pack stdout)
   95             in return (Right output)
   96           ExitFailure code
   97             | code == 124 -> return (Left "nix evaluation timed out")
   98             | otherwise   -> return (Left $ "nix-instantiate failed (" <> T.pack (show code) <> "): mode=" <> modeLabel <> ": " <> T.strip (T.pack stderr)))
   99     (\e -> return (Left $ "nix exception: " <> T.pack (show e)))
  100   return result

  102 normalizeConceptKey :: Text -> Maybe Text
  103 normalizeConceptKey raw =
  104   let normalized = T.toLower (T.strip raw)
  105   in if T.null normalized
  106        then Nothing
  107        else case conceptAlias normalized of
  108          Just canonical -> Just canonical
  109          Nothing -> if T.all isConceptChar normalized
  110            then Just normalized
  111            else Nothing

  113 isConceptChar :: Char -> Bool
  114 isConceptChar c =
  115   (isAscii c && (isAlphaNum c || c == '-' || c == '_'))
  116     || (not (isAscii c) && isLetter c)

  159 isLenientMode :: IO Bool
  160 isLenientMode = do
  161   mEnv <- lookupEnv "QXFX0_NIXGUARD_LENIENT_UNSUPPORTED"
  162   return (isJust mEnv && mEnv == Just "1")

  164 classifyNixEvalError :: Text -> Text
  165 classifyNixEvalError err
  166   | isRestrictedFlagUnsupported err = "restricted_eval_unsupported"
  167   | "nix evaluation timed out" `T.isInfixOf` err = "timeout"
  168   | "nix exception:" `T.isPrefixOf` err = "process_exception"
  169   | "nix-instantiate failed" `T.isPrefixOf` err = "evaluation_failed"
  170   | otherwise = "evaluation_failed"
  171 
  172 isRestrictedFlagUnsupported :: Text -> Bool
  173 isRestrictedFlagUnsupported err =
  174   ("--restricted" `T.isInfixOf` err)
  175     && (("unrecognised flag" `T.isInfixOf` err) || ("unrecognized flag" `T.isInfixOf` err))
