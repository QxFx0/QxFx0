   16 extractSenseVector :: UtteranceSemanticFrame -> SenseVector
   17 extractSenseVector frame =
   18   let anchorTxt = if T.null (T.strip (usfTopic frame)) then "unknown" else usfTopic frame
   19       evidenceNodes = map (SemanticNodeId . wmuLemma) (take 6 (usfWordUnits frame))
   20       axes = foldl accumulateAxis M.empty (usfWordUnits frame)
   21       operators = nub [inferOperator frame]
   22       polarity = case usfPolarity frame of
   23         PolarityPositive -> SpAffirm
   24         PolarityNegative -> SpDeny
   25       lexicalFamily =
   26         let roots = nub (map wmuLemma (filter ((/= PosUnknown) . wmuPartOfSpeech) (usfWordUnits frame)))
   27         in case roots of
   28              root:_ -> Just (FamilyId root)
   29              [] -> Nothing
   30       toNode txt = if T.null (T.strip txt) then Nothing else Just (SemanticNodeId txt)
   31   in emptySenseVector
   32       { svAnchor = SemanticNodeId anchorTxt
   33       , svLexicalFamily = lexicalFamily
   34       , svEvidenceNodes = evidenceNodes
   35       , svAxes = axes
   36       , svOperators = operators
   37       , svPolarity = polarity
   38       , svAgent = toNode =<< Just (fromMaybeText (usfAgent frame))
   39       , svTarget = toNode =<< Just (fromMaybeText (usfTarget frame))
   40       , svConfidence = usfConfidence frame
   41       }
   42   where
   43     fromMaybeText = maybe "" id
   44 
   45 accumulateAxis :: Map SenseAxis Double -> WordMeaningUnit -> Map SenseAxis Double
   46 accumulateAxis acc unit =
   47   foldl (\m axis -> M.insertWith (+) axis 1.0 m) acc (mapMaybe semanticClassAxis (wmuSemanticClasses unit) ++ discourseAxes)
   48   where
   49     discourseAxes = concatMap discourseAxis (wmuDiscourseFunctions unit)
   50 
   51 semanticClassAxis :: InputSemanticClass -> Maybe SenseAxis
   52 semanticClassAxis cls = case cls of
   53   SemIdentity -> Just AxIdentity
   54   SemCause -> Just AxCause
   55   SemPurposeFunction -> Just AxPurpose
   56   SemRelation -> Just AxContrast
   57   SemAction -> Just AxAction
   58   SemState -> Just AxState
   59   SemKnowledge -> Just AxKnowledge
   60   SemSelfReference -> Just AxSelf
   61   SemUserReference -> Just AxOther
   62   SemDialogueRepair -> Just AxRepair
   63   SemComparison -> Just AxContrast
   64   _ -> Nothing
   65 
   66 discourseAxis :: InputDiscourseFunction -> [SenseAxis]
   67 discourseAxis fn = case fn of
   68   DiscNegation -> [AxBoundary]
   69   DiscQuestion -> [AxKnowledge]
   70   DiscContrast -> [AxContrast, AxBoundary]
   71   DiscCause -> [AxCause]
   72   DiscClarification -> [AxRepair]
   73   _ -> []
   74 
   75 inferOperator :: UtteranceSemanticFrame -> SenseOperator
   76 inferOperator frame =
   77   case T.toLower (irhTag (usfRouteHint frame)) of
   78     "concept_knowledge" -> OpDefine
   79     "world_cause" -> OpExplainCause
   80     "comparison_plausibility" -> OpDistinguish
   81     "comparison_relation" -> OpDistinguish
   82     "misunderstanding" -> OpRepair
   83     "boundary_command" -> OpRepair
   84     "next_step" -> OpNextStep
   85     _ | usfSpeechAct frame == ActAsk -> OpClarify
   86       | otherwise -> OpGround
