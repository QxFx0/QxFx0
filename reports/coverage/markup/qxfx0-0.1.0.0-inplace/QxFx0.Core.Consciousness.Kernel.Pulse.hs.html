   70 kernelPulse :: UnconsciousKernel -> SemanticInput -> Double -> Double -> Int -> KernelOutput
   71 kernelPulse kernel semanticInput humanTheta resonance _turn =
   72   let tv = ukThinking kernel
   73       inputText = siRawInput semanticInput
   74       frame = siPropositionFrame semanticInput
   75       recommendedFamily = siRecommendedFamily semanticInput
   76       ontQ = applyOntologicalCore (ukOntology kernel) frame recommendedFamily resonance
   77       activeDesires = activateDesires (ukDesires kernel) humanTheta resonance frame
   78       selectedSkill = selectSkillWithOntology (ukSkills kernel) activeDesires frame recommendedFamily ontQ
   79       thinkResult = applyThinkingVectorWithOntology tv inputText frame recommendedFamily ontQ
   80       conflicts = findDesireConflicts activeDesires
   81       narrativeDrive = deriveNarrativeDrive activeDesires frame recommendedFamily selectedSkill
   82       focusHint = focusForDrive narrativeDrive frame inputText
   83    in KernelOutput
   84         { koActiveDesires = map desireName activeDesires
   85         , koSelectedSkill = skillName selectedSkill
   86         , koSearchResult = thinkResult
   87         , koConflicts = conflicts
   88         , koOntologicalQuestion = ontQ
   89         , koNarrativeDrive = narrativeDrive
   90         , koFocusHint = focusHint
   91         }

   93 selectSkillWithOntology :: SkillSet -> [KernelDesire] -> InputPropositionFrame -> CanonicalMoveFamily -> OntologicalQuestion -> Skill
   94 selectSkillWithOntology skillSet _desires frame recommendedFamily ontQ =
   95   let allSkills = skills skillSet
   96       ontBonus skill = case (skillName skill, oqForm ontQ, oqMeaning ontQ) of
   97         (n, FormExperience, _) | n == skillSilenceName -> consciousnessSilenceOntologyBonus
   98         (n, FormReasoning, MeaningFragmentMode) | n == skillListenName -> consciousnessListenOntologyBonus
   99         (n, FormCause, _) | n == skillNameName -> consciousnessNameOntologyBonus
  100         _ -> 0.0
  101       scored = map (\skill -> (skill, skillAffinity skill + ontBonus skill + skillContextBonus skill frame recommendedFamily)) allSkills
  102    in case sortBy (comparing (Down . snd)) scored of
  103         (skill, _) : _ -> skill
  104         [] -> dominantSkill skillSet
  105   where
  106     skillContextBonus skill propositionFrame family
  107       | skillName skill == skillSilenceName
  108           && (ipfEmotionalTone propositionFrame == ToneDistress
  109                 || family `elem` [CMContact, CMRepair, CMAnchor]) =
  110           consciousnessSilenceContextBonus
  111       | skillName skill == skillListenName
  112           && (ipfIsQuestion propositionFrame
  113                 || ipfRegisterHint propositionFrame == Search
  114                 || family `elem` [CMClarify, CMDeepen]) =
  115           consciousnessListenContextBonus
  116       | skillName skill == skillResistName
  117           && (ipfEmotionalTone propositionFrame == ToneConfrontational
  118                 || family == CMConfront
  119                 || ipfIsNegated propositionFrame) =
  120           consciousnessResistContextBonus
  121       | otherwise = 0.0

  123 applyThinkingVectorWithOntology :: ThinkingVector -> Text -> InputPropositionFrame -> CanonicalMoveFamily -> OntologicalQuestion -> ThinkingResult
  124 applyThinkingVectorWithOntology tv txt frame recommendedFamily ontQ =
  125   let base = applyThinkingVector tv txt frame recommendedFamily
  126    in base {trDeepContent = renderMeaningMode (oqMeaning ontQ)}
  127 
  128 applyThinkingVector :: ThinkingVector -> Text -> InputPropositionFrame -> CanonicalMoveFamily -> ThinkingResult
  129 applyThinkingVector tv txt frame recommendedFamily =
  130   let tokens = tokenizeKeywordText txt
  131       surface = T.take 100 txt
  132       deepMode
  133         | tvSearchDepth tv > consciousnessDeepSearchThreshold = findDeepContent frame recommendedFamily tokens
  134         | tvSearchDepth tv > consciousnessShallowSearchThreshold = MeaningLowConfidenceMode
  135         | otherwise = MeaningExtendedMode
  136       deep = renderMeaningMode deepMode
  137       temporal = detectTemporalFocus tokens
  138       humanAbstrLvl = estimateAbstractionLevel tokens
  139       abstractionGap = humanAbstrLvl - tvAbstractionLvl tv
  140    in ThinkingResult surface deep temporal abstractionGap

  142 findDeepContent :: InputPropositionFrame -> CanonicalMoveFamily -> [Text] -> MeaningMode
  143 findDeepContent frame recommendedFamily tokens
  144   | ipfConfidence frame < consciousnessLowConfidenceThreshold = MeaningLowConfidenceMode
  145   | ipfEmotionalTone frame == ToneDistress || ipfIsNegated frame = MeaningUncertaintyMode
  146   | recommendedFamily `elem` [CMDefine, CMDistinguish, CMPurpose, CMHypothesis, CMReflect] = MeaningGeneralizationMode
  147   | recommendedFamily `elem` [CMClarify, CMDeepen, CMNextStep] = MeaningNarrowingMode
  148   | recommendedFamily `elem` [CMContact, CMRepair, CMAnchor] = MeaningExperienceMode
  149   | containsAnyKeywordPhrase tokens deepContentMinimizationKeywords = MeaningExperienceMode
  150   | containsAnyKeywordPhrase tokens deepContentGeneralizationKeywords = MeaningGeneralizationMode
  151   | containsAnyKeywordPhrase tokens deepContentNarrowingKeywords = MeaningNarrowingMode
  152   | containsAnyKeywordPhrase tokens deepContentUncertaintyKeywords = MeaningUncertaintyMode
  153   | otherwise = MeaningExtendedMode
  154 
  155 detectTemporalFocus :: [Text] -> Text
  156 detectTemporalFocus tokens
  157   | containsAnyKeywordPhrase tokens temporalPastKeywords = temporalPast
  158   | containsAnyKeywordPhrase tokens temporalFutureKeywords = temporalFuture
  159   | otherwise = temporalPresent

  161 estimateAbstractionLevel :: [Text] -> Double
  162 estimateAbstractionLevel tokens =
  163   let aScore = fromIntegral (countKeywordPhraseHits tokens abstractKeywords)
  164       cScore = fromIntegral (countKeywordPhraseHits tokens concreteKeywords)
  165       total = aScore + cScore + consciousnessAbstractionEpsilon
  166    in aScore / total

  168 activateDesires :: [KernelDesire] -> Double -> Double -> InputPropositionFrame -> [KernelDesire]
  169 activateDesires desires _theta resonance frame =
  170   let fundamental = filter (\desire -> desireStrength desire == Fundamental) desires
  171       contextual =
  172         filter (isActivated frame resonance) $
  173           filter (\desire -> desireStrength desire /= Fundamental) desires
  174    in fundamental ++ contextual
  175 
  176 isActivated :: InputPropositionFrame -> Double -> KernelDesire -> Bool
  177 isActivated frame resonance desire =
  178   case desireName desire of
  179     name | name == desirePresenceName -> resonance > consciousnessDesirePresenceThreshold
  180     name | name == desireUnderstandName ->
  181       ipfIsQuestion frame || ipfRegisterHint frame == Search || ipfSemanticLayer frame == MetaLayer
  182     name | name == desirePreserveName -> resonance > consciousnessDesirePreserveThreshold
  183     _ -> False

  185 findDesireConflicts :: [KernelDesire] -> [Text]
  186 findDesireConflicts desires =
  187   [ desireName desire <> desireConflictSeparator <> conflict
  188   | desire <- desires
  189   , Just conflict <- [desireConflict desire]
  190   ]

  192 applyOntologicalCore :: OntologicalCore -> InputPropositionFrame -> CanonicalMoveFamily -> Double -> OntologicalQuestion
  193 applyOntologicalCore _core frame recommendedFamily resonance =
  194   let form = classifyForm frame recommendedFamily
  195       meaning = deriveMeaning resonance form frame recommendedFamily
  196       returnForm = selectReturnForm form frame resonance
  197    in OntologicalQuestion form meaning returnForm
  198 
  199 classifyForm :: InputPropositionFrame -> CanonicalMoveFamily -> ConsciousForm
  200 classifyForm frame recommendedFamily
  201   | ipfSemanticLayer frame == ContactLayer = FormExperience
  202   | recommendedFamily `elem` [CMContact, CMRepair, CMAnchor, CMDescribe] = FormExperience
  203   | ipfClauseForm frame == Imperative || recommendedFamily `elem` [CMNextStep, CMConfront] = FormAction
  204   | recommendedFamily `elem` [CMDefine, CMPurpose, CMGround] = FormCause
  205   | recommendedFamily `elem` [CMDistinguish, CMReflect, CMHypothesis, CMClarify, CMDeepen] = FormReasoning
  206   | T.length (ipfRawText frame) < consciousnessFragmentLengthThreshold = FormFragment
  207   | otherwise = FormExtended
  208 
  209 deriveMeaning :: Double -> ConsciousForm -> InputPropositionFrame -> CanonicalMoveFamily -> MeaningMode
  210 deriveMeaning resonance form frame recommendedFamily
  211   | ipfConfidence frame < consciousnessLowConfidenceThreshold = MeaningLowConfidenceMode
  212   | ipfEmotionalTone frame == ToneDistress || ipfIsNegated frame = MeaningUncertaintyMode
  213   | recommendedFamily `elem` [CMDefine, CMPurpose, CMGround] || form == FormCause = MeaningCauseMode
  214   | recommendedFamily `elem` [CMDistinguish, CMReflect, CMHypothesis] = MeaningGeneralizationMode
  215   | recommendedFamily `elem` [CMClarify, CMDeepen, CMNextStep] = MeaningNarrowingMode
  216   | recommendedFamily `elem` [CMContact, CMDescribe, CMRepair, CMAnchor] || form == FormExperience = MeaningExperienceMode
  217   | form == FormAction && resonance < consciousnessLowResonanceThreshold = MeaningActionLowResonanceMode
  218   | form == FormAction = MeaningActionHighResonanceMode
  219   | form == FormFragment = MeaningFragmentMode
  220   | otherwise = MeaningExtendedMode
  221 
  222 selectReturnForm :: ConsciousForm -> InputPropositionFrame -> Double -> ReturnMode
  223 selectReturnForm form frame resonance
  224   | form == FormExperience || ipfSemanticLayer frame == ContactLayer = ReturnExperienceMode
  225   | resonance > consciousnessHighResonanceReturnThreshold = ReturnDirectHighResonanceMode
  226   | resonance < consciousnessLowResonanceReturnThreshold || ipfConfidence frame < consciousnessLowConfidenceThreshold = ReturnStructuredLowResonanceMode
  227   | otherwise = ReturnMixedMode

  229 deriveNarrativeDrive :: [KernelDesire] -> InputPropositionFrame -> CanonicalMoveFamily -> Skill -> NarrativeDrive
  230 deriveNarrativeDrive activeDesires frame recommendedFamily selectedSkill
  231   | any ((== desirePreserveName) . desireName) activeDesires = DriveStability
  232   | recommendedFamily == CMRepair = DriveRepair
  233   | ipfSemanticLayer frame == ContactLayer || recommendedFamily == CMContact = DriveContact
  234   | recommendedFamily `elem` [CMDefine, CMDistinguish] = DriveDefinition
  235   | ipfIsQuestion frame || ipfRegisterHint frame == Search || recommendedFamily `elem` [CMClarify, CMDeepen, CMHypothesis] = DriveInquiry
  236   | recommendedFamily `elem` [CMNextStep, CMConfront] = DriveAction
  237   | skillName selectedSkill == skillSilenceName = DriveSpace
  238   | otherwise = DrivePresence

  240 focusForDrive :: NarrativeDrive -> InputPropositionFrame -> Text -> Text
  241 focusForDrive drive frame inputText =
  242   let focusEntity =
  243         if T.null (ipfFocusNominative frame)
  244           then ipfFocusEntity frame
  245           else ipfFocusNominative frame
  246       humanFocus = if T.null focusEntity then T.take 30 inputText else focusEntity
  247    in case drive of
  248         DriveStability -> focusStability
  249         DriveSpace -> focusSpace
  250         DriveInquiry -> focusHumanPrefix <> humanFocus
  251         DriveDefinition -> humanFocus
  252         DriveContact -> focusHumanPrefix <> humanFocus
  253         DriveAction -> humanFocus
  254         DriveRepair -> focusHumanPrefix <> humanFocus
  255         DrivePresence -> focusHumanPrefix <> humanFocus
