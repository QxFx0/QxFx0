   14 updateStateNixCache :: Text -> NixGuardStatus -> Map.Map Text NixGuardStatus -> Map.Map Text NixGuardStatus
   15 updateStateNixCache key status oldCache =
   16   let inserted = Map.insert key status oldCache
   17    in if Map.size inserted <= nixCacheMaxSize
   18         then inserted
   19         else Map.fromList (take nixCacheMaxSize (Map.toList inserted))
