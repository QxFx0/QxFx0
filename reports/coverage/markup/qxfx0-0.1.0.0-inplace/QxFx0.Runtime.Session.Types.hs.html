    6 {-| Public session types and output-mode codecs for the runtime shell. -}
    7 module QxFx0.Runtime.Session.Types
    8   ( RuntimeOutputMode(..)
    9   , Session(..)
   10   , StateOrigin(..)

   12   , runtimeToDialogueMode

   14   ) where
   15 
   16 import Data.Aeson (FromJSON(..), ToJSON(..), Value(..), withText)
   17 import Data.Text (Text)
   18 import qualified Data.Text as T
   19 import QxFx0.Resources (ReadinessMode)
   20 import QxFx0.Runtime.Context (RuntimeContext)
   21 import QxFx0.Types.Decision (DialogueOutputMode(..))
   22 import QxFx0.Types.State (SystemState)
   23 
   24 data RuntimeOutputMode
   25   = DialogueMode
   26   | SemanticIntrospectionMode
   27   deriving stock (Eq, Show)
   28 
   29 data StateOrigin
   30   = FreshOrigin
   31   | RestoredOrigin
   32   | RecoveredCorruptOrigin
   33   deriving stock (Eq, Show)
   34 
   35 instance ToJSON StateOrigin where
   36   toJSON FreshOrigin = String "fresh"
   37   toJSON RestoredOrigin = String "restored"
   38   toJSON RecoveredCorruptOrigin = String "recovered_corrupt"
   39 
   40 instance FromJSON StateOrigin where
   41   parseJSON = withText "StateOrigin" $ \t ->
   42     case T.toLower (T.strip t) of
   43       "fresh" -> pure FreshOrigin
   44       "restored" -> pure RestoredOrigin
   45       "recovered_corrupt" -> pure RecoveredCorruptOrigin
   46       _ -> fail ("Unknown StateOrigin: " <> T.unpack t)
   47 
   48 data Session = Session
   49   { sessSystemState :: !SystemState
   50   , sessOutputMode :: !RuntimeOutputMode
   51   , sessSessionId :: !Text
   52   , sessDbPath :: !FilePath
   53   , sessStateOrigin :: !StateOrigin
   54   , sessReadinessMode :: !ReadinessMode
   55   , sessRuntime :: !RuntimeContext
   56   }
   57 
   58 renderRuntimeOutputMode :: RuntimeOutputMode -> Text
   59 renderRuntimeOutputMode DialogueMode = "dialogue"
   60 renderRuntimeOutputMode SemanticIntrospectionMode = "semantic"
   61 
   62 runtimeToDialogueMode :: RuntimeOutputMode -> DialogueOutputMode
   63 runtimeToDialogueMode DialogueMode = DialogueOutput
   64 runtimeToDialogueMode SemanticIntrospectionMode = SemanticIntrospectionOutput
   65 
   66 dialogueToRuntimeMode :: DialogueOutputMode -> RuntimeOutputMode
   67 dialogueToRuntimeMode DialogueOutput = DialogueMode
   68 dialogueToRuntimeMode SemanticIntrospectionOutput = SemanticIntrospectionMode
