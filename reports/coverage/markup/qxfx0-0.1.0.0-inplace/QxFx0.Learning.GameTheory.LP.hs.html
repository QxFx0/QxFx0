    6 {-|
    7 Proper LP solver for zero-sum mixed-strategy games using the simplex method.
    8 
    9 Formulation (primal):
   10   minimise   sum_i w_i
   11   subject to sum_i w_i * B_{ij} >= 1   for all j
   12              w_i >= 0
   13 
   14 where B = A + C is the payoff matrix shifted so every entry is > 0.
   15 The optimal mixed strategy is p_i = w_i / sum_i w_i.
   16 
   17 We solve the dual problem via the revised simplex tableau,
   18 then recover the primal variables from the objective-row reduced costs.
   19 -}
   20 module QxFx0.Learning.GameTheory.LP
   21   ( solveMixedStrategy
   22   ) where
   23 
   24 import Data.List (minimumBy)
   25 import Data.Ord (comparing)
   26 
   27 epsilon :: Double
   28 epsilon = 1e-9
   29 
   30 -- | Solve a zero-sum mixed-strategy LP.
   31 -- Input: payoff matrix A (n rows x m columns).
   32 -- Returns: list of n probabilities summing to 1, or Nothing if degenerate.
   33 solveMixedStrategy :: [[Double]] -> Maybe [Double]
   34 solveMixedStrategy a =
   35   case a of
   36     [] -> Nothing
   37     (row0:_) | null row0 -> Nothing
   38     (row0:_) ->
   39       let n = length a
   40           m = length row0
   41           rectangular = all ((== m) . length) a
   42           -- Shift payoffs so every entry >= 1 (guarantees v > 0)
   43           minA = minimum (concat a)
   44           shift = if minA < 1.0 then 1.0 - minA else 0.0
   45           b = map (map (+ shift)) a
   46           -- Dual problem: max sum_j y_j
   47           -- s.t. sum_j b_{ij} y_j <= 1 for all i
   48           -- Build tableau with n constraints, m decision + n slack variables
   49           nVars = m + n
   50           -- Objective row: -z + sum y_j = 0  (maximisation convention)
   51           objRow = replicate m 1.0 ++ replicate n 0.0
   52           objRHS = 0.0
   53           -- Constraint rows: sum_j b_{ij} y_j + s_i = 1
   54           conRows = zipWith (\row i ->
   55                               take m row ++ replicate i 0.0 ++ [1.0] ++ replicate (n - i - 1) 0.0)
   56                             b [0..]
   57           conRHS = replicate n 1.0
   58           -- Initial basic variables are the slacks (indices m .. m+n-1)
   59           basic = [m .. m+n-1]
   60           -- Run simplex (fail-closed: validate tableau dimensions first)
   61           final = if rectangular && all (\row -> length row == nVars) conRows
   62                      then simplexLoop nVars conRows conRHS objRow objRHS basic 1000
   63                      else Nothing
   64        in case final of
   65             Nothing -> Nothing
   66             Just (_, objR, _, bas, _) ->
   67               let -- Primal variables w_i = - (objective coefficient of slack s_i)
   68                   w = [ let coeff = if (m + i) `elem` bas then 0.0 else atOrZero objR (m + i)
   69                         in max 0.0 (-coeff)
   70                       | i <- [0 .. n-1] ]
   71                   total = sum w
   72               in if total > epsilon
   73                  then Just (map (/ total) w)
   74                  else Nothing
   75 
   76 -- | Simplex tableau iteration.
   77 simplexLoop :: Int                          -- total variables
   78             -> [[Double]]                   -- constraint rows
   79             -> [Double]                     -- constraint RHS
   80             -> [Double]                     -- objective coefficients
   81             -> Double                       -- objective RHS
   82             -> [Int]                        -- basic variable per row
   83             -> Int                          -- max iterations
   84             -> Maybe ([[Double]], [Double], Double, [Int], [Double])
   85 simplexLoop nVars conRows conRHS objRow objRHS basic limit
   86   | limit <= 0 = Nothing
   87   | otherwise =
   88       let nonBasic = filter (`notElem` basic) [0 .. nVars-1]
   89           -- Optimality: for maximisation with -z + c^T x = RHS,
   90           -- optimal when all nonbasic objective coefficients <= 0
   91           entering = findEntering objRow nonBasic
   92       in case entering of
   93            Nothing -> Just (conRows, objRow, objRHS, basic, conRHS)
   94            Just e ->
   95               let -- Ratio test: for each row where coeff > 0, RHS / coeff
   96                   ratios =
   97                     [ (atOrZero conRHS i / coeff, i)
   98                     | i <- [0 .. length conRows - 1]
   99                     , Just row <- [safeIndex conRows i]
  100                     , let coeff = atOrZero row e
  101                     , coeff > epsilon
  102                     ]
  103               in if null ratios
  104                  then Nothing  -- unbounded
  105                  else
  106                    let (_, l) = minimumBy (comparing fst) ratios
  107                    in case safeIndex conRows l of
  108                         Nothing -> Nothing
  109                         Just pivotRowRaw ->
  110                           let pivotElem = atOrZero pivotRowRaw e
  111                           in if pivotElem <= epsilon
  112                                then Nothing
  113                                else
  114                                  let pivotRow = map (/ pivotElem) pivotRowRaw
  115                                      pivotRHS = atOrZero conRHS l / pivotElem
  116                                      updateRow row rhs =
  117                                        let coeff = atOrZero row e
  118                                            newRow = zipWith (-) row (map (* coeff) pivotRow)
  119                                            newRHS = rhs - coeff * pivotRHS
  120                                        in (replaceAt e 0.0 newRow, newRHS)
  121                                      rebuilt =
  122                                        [ if i == l
  123                                            then (pivotRow, pivotRHS)
  124                                            else updateRow row (atOrZero conRHS i)
  125                                        | (i, row) <- zip [0 ..] conRows
  126                                        ]
  127                                      (conRows', conRHS') = unzip rebuilt
  128                                      objCoeffE = atOrZero objRow e
  129                                      newObjRow = zipWith (-) objRow (map (* objCoeffE) pivotRow)
  130                                      newObjRHS = objRHS - objCoeffE * pivotRHS
  131                                      newObjRow' = replaceAt e 0.0 newObjRow
  132                                      basic' = take l basic ++ [e] ++ drop (l + 1) basic
  133                                  in simplexLoop nVars conRows' conRHS' newObjRow' newObjRHS basic' (limit - 1)
  134 
  135 findEntering :: [Double] -> [Int] -> Maybe Int
  136 findEntering objRow nonBasic =
  137   let candidates = filter (\j -> atOrZero objRow j > epsilon) nonBasic
  138   in case candidates of
  139        [] -> Nothing
  140        js -> Just (minimum js)  -- Bland's rule: smallest index with positive reduced cost
  141 
  142 atOrZero :: [Double] -> Int -> Double
  143 atOrZero xs idx = maybe 0.0 id (safeIndex xs idx)
  144 
  145 safeIndex :: [a] -> Int -> Maybe a
  146 safeIndex xs idx
  147   | idx < 0 = Nothing
  148   | otherwise =
  149       case drop idx xs of
  150         y:_ -> Just y
  151         [] -> Nothing
  152 
  153 replaceAt :: Int -> a -> [a] -> [a]
  154 replaceAt idx value xs
  155   | idx < 0 = xs
  156   | otherwise =
  157       case splitAt idx xs of
  158         (prefix, _ : suffix) -> prefix ++ value : suffix
  159         _ -> xs
