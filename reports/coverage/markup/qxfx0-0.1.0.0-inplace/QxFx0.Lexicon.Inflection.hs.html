   28 toNominative :: MorphologyData -> Text -> Text
   29 toNominative md w =
   30   case M.lookup w (mdNominative md) of
   31     Just f -> f
   32     Nothing -> case M.lookup (T.toLower w) (mdNominative md) of
   33       Just f -> f
   34       Nothing -> resolveCandidateNominative md w
   35 
   36 resolveCandidateNominative :: MorphologyData -> Text -> Text
   37 resolveCandidateNominative md surface =
   38   case resolveLexemeForm md surface (Just NominativeCase) Nothing of
   39     Just form -> lfLemma form
   40     Nothing -> surface
   41 
   42 genitiveForm :: MorphologyData -> Text -> Text
   43 genitiveForm md w = case M.lookup w (mdGenitive md) of
   44   Just f -> f
   45   Nothing -> case M.lookup (T.toLower w) (mdGenitive md) of
   46     Just f -> f
   47     Nothing -> resolveCandidateGenitive md w
   48 
   49 resolveCandidateGenitive :: MorphologyData -> Text -> Text
   50 resolveCandidateGenitive md surface =
   51   case resolveLexemeForm md surface (Just GenitiveCase) Nothing of
   52     Just form -> lfSurface form
   53     Nothing -> surface
   54 
   55 accusativeForm :: MorphologyData -> Text -> Text
   56 accusativeForm md w =
   57   case resolveCandidateAccusative md w of
   58     Just form -> form
   59     Nothing ->
   60       let gender = guessGender w
   61           animacy = guessAnimacy w
   62       in case (gender, animacy) of
   63             (Masculine, Inanimate) -> w
   64             (Neuter, _) -> w
   65             _ -> genitiveForm md w
   66 
   67 resolveCandidateAccusative :: MorphologyData -> Text -> Maybe Text
   68 resolveCandidateAccusative md surface =
   69   lfSurface <$> resolveLexemeForm md surface (Just AccusativeCase) Nothing
   70 
   71 data Animacy = Animate | Inanimate
   72   deriving stock (Eq, Show)
   73 
   74 data Gender = Masculine | Feminine | Neuter
   75   deriving stock (Eq, Show)
   76 
   77 guessAnimacy :: Text -> Animacy
   78 guessAnimacy w =
   79   let lower = T.toLower w
   80       animateSuffixes =
   81         [ morphFemSuffixA
   82         , morphFemSuffixYa
   83         , morphNounSuffixA
   84         , morphNounSuffixIya
   85         ]
   86   in if any (`T.isSuffixOf` lower) animateSuffixes then Animate else Inanimate
   87 
   88 guessGender :: Text -> Gender
   89 guessGender w
   90   | T.isSuffixOf morphFemSuffixA lower = Feminine
   91   | T.isSuffixOf morphFemSuffixYa lower = Feminine
   92   | T.isSuffixOf morphNounSuffixA lower = Feminine
   93   | T.isSuffixOf morphNounSuffixIya lower = Feminine
   94   | T.isSuffixOf morphNeutSuffixO lower = Neuter
   95   | T.isSuffixOf morphNeutSuffixE lower = Neuter
   96   | T.isSuffixOf morphGenSuffixI lower = Feminine
   97   | otherwise = Masculine
   98   where
   99     lower = T.toLower w

  101 prepositionalForm :: MorphologyData -> Text -> Text
  102 prepositionalForm md w = case M.lookup w (mdPrepositional md) of
  103   Just f -> f
  104   Nothing -> case M.lookup (T.toLower w) (mdPrepositional md) of
  105     Just f -> f
  106     Nothing -> resolveCandidatePrepositional md w
  107 
  108 resolveCandidatePrepositional :: MorphologyData -> Text -> Text
  109 resolveCandidatePrepositional md surface =
  110   case resolveLexemeForm md surface (Just PrepositionalCase) Nothing of
  111     Just form -> lfLemma form
  112     Nothing -> surface
