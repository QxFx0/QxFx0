  173 -- | Validate an endpoint URL using the strict default policy.
  174 --
  175 -- Rules (fail-closed):
  176 --   1. Scheme must be @https://@.
  177 --   2. Host must be in 'llmEndpointAllowlist'.
  178 --   3. A single untrusted-host override is not sufficient by itself.
  179 --      Use 'validateEndpointUrlWithContext' with dev/test or double opt-in
  180 --      context when this is intentional.
  181 --   3. Empty or malformed URI is rejected.
  182 --
  183 -- Returns 'Right ()' when allowed, 'Left reason' when blocked.
  184 validateEndpointUrl :: Text -> Maybe Text -> Either TransportFallbackReason ()
  185 validateEndpointUrl endpoint mAllowOverride =
  186   () <$ validateEndpointUrlWithContext endpoint mAllowOverride Nothing Nothing
  187 
  188 -- | Validate an endpoint URL with explicit override context.
  189 --
  190 -- Parameters after the endpoint are:
  191 --
  192 -- * @QXFX0_LLM_ALLOW_UNTRUSTED_HOST@ value.
  193 -- * dev/test mode marker, for example @QXFX0_TEST_MODE=1@ or
  194 --   @QXFX0_LLM_DEV_MODE=1@.
  195 -- * double-confirmation marker, currently
  196 --   @QXFX0_LLM_ALLOW_UNTRUSTED_HOST_CONFIRM=1@.
  197 --
  198 -- A successful untrusted-host override returns the warning tag that must be
  199 -- emitted to logs/trace before a bearer token is sent to that host.
  200 validateEndpointUrlWithContext
  201   :: Text
  202   -> Maybe Text
  203   -> Maybe Text
  204   -> Maybe Text
  205   -> Either TransportFallbackReason (Maybe Text)
  206 validateEndpointUrlWithContext endpoint mAllowOverride mDevOrTestMode mDoubleConfirm =
  207   let stripped = T.strip endpoint
  208   in if T.null stripped
  209        then Left TfrUnsafeEndpoint
  210        else case parseEndpointHost stripped of
  211               Nothing -> Left TfrUnsafeEndpoint
  212               Just host ->
  213                 if host `elem` llmEndpointAllowlist
  214                   then Right Nothing
  215                   else if mAllowOverride == Just "1"
  216                          then if isTruthy mDevOrTestMode || isTruthy mDoubleConfirm
  217                                 then Right (Just llmUntrustedHostOverrideWarningTag)
  218                                 else Left TfrUntrustedOverrideRejected
  219                          else Left TfrBlockedHost
  220 
  221 parseEndpointHost :: Text -> Maybe Text
  222 parseEndpointHost endpoint = do
  223   uri <- parseURI (T.unpack endpoint)
  224   auth <- uriAuthority uri
  225   let scheme = T.toLower (T.pack (uriScheme uri))
  226       rawHost = uriRegName auth
  227       rawPort = uriPort auth
  228       portAllowed = null rawPort || rawPort == ":443"
  229   if scheme == "https:" && null (uriUserInfo auth) && not (null rawHost) && portAllowed
  230     then Just (T.toLower (T.pack rawHost))
  231     else Nothing
  232 
  233 endpointHost :: Text -> Text
  234 endpointHost rawEndpoint =
  235   fromMaybe "invalid_endpoint" (parseEndpointHost (T.strip rawEndpoint))
  236 
  237 isTruthy :: Maybe Text -> Bool
  238 isTruthy raw =
  239   case T.toLower . T.strip <$> raw of
  240     Just "1" -> True
  241     Just "true" -> True
  242     Just "yes" -> True
  243     Just "on" -> True
  244     Just "dev" -> True
  245     Just "test" -> True
  246     _ -> False

  289 resolveTransportConfigFromEnv :: IO ExternalQueryConfig
  290 resolveTransportConfigFromEnv = do
  291   mTransport <- lookupEnv "QXFX0_LLM_TRANSPORT"
  292   let mode = fromMaybe "disabled" mTransport
  293   case mode of
  294     "mock" -> pure $ defaultExternalQueryConfig
  295       { eqcTransportMode = "mock"
  296       , eqcFallbackReason = Just TfrExplicitMock
  297       }
  298     "mistral" -> do
  299       mKey <- lookupEnv "QXFX0_MISTRAL_API_KEY"
  300       overrideContext <- readOverrideContext
  301       timeoutMs <- readTransportTimeoutMs ["QXFX0_MISTRAL_TIMEOUT_MS", "QXFX0_LLM_TIMEOUT_MS"]
  302       case fmap (T.strip . T.pack) mKey of
  303         Nothing -> pure $ defaultExternalQueryConfig
  304           { eqcTransportMode = "mistral"
  305           , eqcTimeoutMs = timeoutMs
  306           , eqcFallbackReason = Just TfrKeyMissing
  307           }
  308         Just key | T.null key -> pure $ defaultExternalQueryConfig
  309           { eqcTransportMode = "mistral"
  310           , eqcTimeoutMs = timeoutMs
  311           , eqcFallbackReason = Just TfrKeyMissing
  312           }
  313         Just key -> do
  314           model    <- fmap (T.pack . fromMaybe "mistral-small-latest") (lookupEnv "QXFX0_MISTRAL_MODEL")
  315           endpoint <- fmap (T.pack . fromMaybe "https://api.mistral.ai/v1/chat/completions") (lookupEnv "QXFX0_MISTRAL_ENDPOINT")
  316           case validateEndpointWithOverrideContext endpoint overrideContext of
  317             Left reason -> pure $ defaultExternalQueryConfig
  318               { eqcTransportMode = "mistral"
  319               , eqcApiKey = Nothing
  320               , eqcModel = model
  321               , eqcEndpoint = endpoint
  322               , eqcTimeoutMs = timeoutMs
  323               , eqcFallbackReason = Just reason
  324               }
  325             Right mWarningTag -> do
  326               emitOverrideWarning endpoint mWarningTag
  327               pure $ ExternalQueryConfig
  328                 { eqcTransportMode = "mistral"
  329                 , eqcApiKey = Just key
  330                 , eqcModel = model
  331                 , eqcEndpoint = endpoint
  332                 , eqcTimeoutMs = timeoutMs
  333                 , eqcFallbackReason = Nothing
  334                 }
  335     "fireworks" -> do
  336       mKey <- lookupEnv "QXFX0_FIREWORKS_API_KEY"
  337       overrideContext <- readOverrideContext
  338       timeoutMs <- readTransportTimeoutMs ["QXFX0_FIREWORKS_TIMEOUT_MS", "QXFX0_LLM_TIMEOUT_MS"]
  339       case fmap (T.strip . T.pack) mKey of
  340         Nothing -> pure $ defaultExternalQueryConfig
  341           { eqcTransportMode = "fireworks"
  342           , eqcTimeoutMs = timeoutMs
  343           , eqcFallbackReason = Just TfrKeyMissing
  344           }
  345         Just key | T.null key -> pure $ defaultExternalQueryConfig
  346           { eqcTransportMode = "fireworks"
  347           , eqcTimeoutMs = timeoutMs
  348           , eqcFallbackReason = Just TfrKeyMissing
  349           }
  350         Just key -> do
  351           model    <- fmap (T.pack . fromMaybe "accounts/fireworks/models/glm-5p1") (lookupEnv "QXFX0_FIREWORKS_MODEL")
  352           endpoint <- fmap (T.pack . fromMaybe "https://api.fireworks.ai/inference/v1/chat/completions") (lookupEnv "QXFX0_FIREWORKS_ENDPOINT")
  353           case validateEndpointWithOverrideContext endpoint overrideContext of
  354             Left reason -> pure $ defaultExternalQueryConfig
  355               { eqcTransportMode = "fireworks"
  356               , eqcApiKey = Nothing
  357               , eqcModel = model
  358               , eqcEndpoint = endpoint
  359               , eqcTimeoutMs = timeoutMs
  360               , eqcFallbackReason = Just reason
  361               }
  362             Right mWarningTag -> do
  363               emitOverrideWarning endpoint mWarningTag
  364               pure $ ExternalQueryConfig
  365                 { eqcTransportMode = "fireworks"
  366                 , eqcApiKey = Just key
  367                 , eqcModel = model
  368                 , eqcEndpoint = endpoint
  369                 , eqcTimeoutMs = timeoutMs
  370                 , eqcFallbackReason = Nothing
  371                 }
  372     _ -> pure $ defaultExternalQueryConfig
  373            { eqcTransportMode = T.pack mode

  377 data OverrideContext = OverrideContext
  378   { ocAllowUntrusted :: !(Maybe Text)
  379   , ocDevOrTestMode :: !(Maybe Text)
  380   , ocDoubleConfirm :: !(Maybe Text)
  381   }
  382 
  383 readOverrideContext :: IO OverrideContext
  384 readOverrideContext = do
  385   mAllowUntrusted <- fmap (fmap T.pack) (lookupEnv "QXFX0_LLM_ALLOW_UNTRUSTED_HOST")
  386   mTestMode <- fmap (fmap T.pack) (lookupEnv "QXFX0_TEST_MODE")
  387   mDevMode <- fmap (fmap T.pack) (lookupEnv "QXFX0_LLM_DEV_MODE")
  388   mDoubleConfirm <- fmap (fmap T.pack) (lookupEnv "QXFX0_LLM_ALLOW_UNTRUSTED_HOST_CONFIRM")
  389   let mDevOrTest = firstTruthy [mTestMode, mDevMode]
  390   pure OverrideContext
  391     { ocAllowUntrusted = mAllowUntrusted
  392     , ocDevOrTestMode = mDevOrTest
  393     , ocDoubleConfirm = mDoubleConfirm
  394     }

  396 validateEndpointWithOverrideContext :: Text -> OverrideContext -> Either TransportFallbackReason (Maybe Text)
  397 validateEndpointWithOverrideContext endpoint ctx =
  398   validateEndpointUrlWithContext endpoint (ocAllowUntrusted ctx) (ocDevOrTestMode ctx) (ocDoubleConfirm ctx)

  400 firstTruthy :: [Maybe Text] -> Maybe Text
  401 firstTruthy [] = Nothing
  402 firstTruthy (x:xs)
  403   | isTruthy x = x
  404   | otherwise = firstTruthy xs
  405 
  406 readTransportTimeoutMs :: [String] -> IO Int
  407 readTransportTimeoutMs names = do
  408   raw <- firstPresentEnv names
  409   pure (sanitizeTimeoutMs raw)
  410 
  411 firstPresentEnv :: [String] -> IO (Maybe String)
  412 firstPresentEnv [] = pure Nothing
  413 firstPresentEnv (name:rest) = do
  414   value <- lookupEnv name
  415   case value of
  416     Just raw | not (null raw) -> pure (Just raw)
  417     _ -> firstPresentEnv rest
  418 
  419 sanitizeTimeoutMs :: Maybe String -> Int
  420 sanitizeTimeoutMs raw =
  421   let parsed = fromMaybe llmDefaultTimeoutMs (raw >>= readMaybe)
  422   in max llmMinTimeoutMs (min llmMaxTimeoutMs parsed)
  423 
  424 emitOverrideWarning :: Text -> Maybe Text -> IO ()
  425 emitOverrideWarning _ Nothing = pure ()
  426 emitOverrideWarning endpoint (Just tag) =
  427   hPutStrLn stderr $ T.unpack $ T.concat
  428     [ "[security_warning] tag="
  429     , tag
  430     , " endpoint_host="
  431     , endpointHost (T.strip endpoint)
  432     ]

  434 buildTransportFromValidatedConfig :: ExternalQueryConfig -> IO LLMTransport
  435 buildTransportFromValidatedConfig cfg =
  436   case eqcFallbackReason cfg of
  437     Just reason -> pure (MockTransport defaultMockTable (Just cfg { eqcApiKey = Nothing, eqcFallbackReason = Just reason }))
  438     Nothing ->
  439       case eqcApiKey cfg of
  440         Nothing -> pure (MockTransport defaultMockTable (Just cfg { eqcFallbackReason = Just TfrKeyMissing }))

  442           mgr <- newTlsManager
  443           buildTransportFromValidatedConfigWithManager mgr cfg

  447   case eqcFallbackReason cfg of
  448     Just reason -> pure (MockTransport defaultMockTable (Just cfg { eqcApiKey = Nothing, eqcFallbackReason = Just reason }))
  449     Nothing ->
  450       case eqcApiKey cfg of
  451         Nothing -> pure (MockTransport defaultMockTable (Just cfg { eqcFallbackReason = Just TfrKeyMissing }))
  452         Just key -> pure (realTransportForKey mgr cfg key)

  470 -- | Build transport from an explicit configuration record.
  471 -- Useful for tests and for deterministic fallback paths.
  472 buildTransportFromConfig :: ExternalQueryConfig -> IO LLMTransport
  473 buildTransportFromConfig cfg =
  474   case eqcFallbackReason cfg of
  475     Just reason -> pure (MockTransport defaultMockTable (Just cfg { eqcApiKey = Nothing, eqcFallbackReason = Just reason }))
  476     Nothing ->
  477       case eqcApiKey cfg of
  478         Nothing -> pure (MockTransport defaultMockTable (Just cfg { eqcFallbackReason = Just TfrKeyMissing }))
  479         Just _ ->
  480           case validateEndpointUrl (eqcEndpoint cfg) Nothing of
  481             Left reason ->
  482               pure (MockTransport defaultMockTable (Just cfg { eqcApiKey = Nothing, eqcFallbackReason = Just reason }))
  483             Right () -> do
  484               mgr <- newTlsManager
  485               buildTransportFromValidatedConfigWithManager mgr cfg

  488 buildTransportFromConfigWithManager mgr cfg =

  536 -- | Execute a query against the chosen transport.
  537 --
  538 -- Returns 'Either ExternalQueryError ExternalQueryResponse' so the
  539 -- caller (learning loop) can decide what to do next.
  540 queryExternalTool
  541   :: LLMTransport
  542   -> ExternalTool
  543   -> LearningNeed
  544   -> Text        -- ^ user query / prompt
  545   -> IO (Either ExternalQueryError ExternalQueryResponse)
  546 queryExternalTool transport tool need query =
  547   let queryChars = T.length query
  548   in if queryChars > llmMaxQueryChars
  549        then pure (Left (queryTooLarge queryChars llmMaxQueryChars))
  550        else case transport of
  551          MockTransport table mCfg  ->
  552            case mCfg >>= eqcFallbackReason of
  553              Just reason | reason /= TfrExplicitMock -> pure (Left (EqeFallback reason))
  554              _ -> mockQuery table tool need query
  555          MistralTransport mgr cfg  -> mistralQuery mgr cfg tool need query
  556          FireworksTransport mgr cfg -> fireworksQuery mgr cfg tool need query
  557

  565 queryExternalToolWithConfig cfg tool need query = do

  569 -- | Deterministic mock query.
  570 mockQuery :: MockTable -> ExternalTool -> LearningNeed -> Text -> IO (Either ExternalQueryError ExternalQueryResponse)
  571 mockQuery table tool need query = do
  572   let needTag = renderNeedTag need
  573       qPrefix = T.takeWhile (not . isSpace) (T.strip query)
  574       match = mockTableLookup (etName tool, needTag, qPrefix) table
  575             <|> mockTableLookup (etName tool, needTag, "") table
  576             <|> mockTableLookup (etName tool, "", qPrefix) table
  577   pure $ case match of
  578     Just (Right body) ->
  579       Right ExternalQueryResponse
  580         { eqrRawBody    = body
  581         , eqrStructured = body
  582         , eqrToolName   = etName tool
  583         , eqrLatencyMs  = 0
  584         }
  585     Just (Left err) ->
  586       Left err
  587     Nothing ->
  588       Left (EqeNetworkUnavailable "mock_no_matching_response")

  611 chatCompletionQuery
  612   :: Manager
  613   -> Text
  614   -> Text
  615   -> Text
  616   -> Text
  617   -> Int
  618   -> ExternalTool
  619   -> Text
  620   -> IO (Either ExternalQueryError ExternalQueryResponse)
  621 chatCompletionQuery mgr _provider endpoint model apiKey timeoutMs tool query = do
  622   requestResult <- buildChatRequest endpoint model apiKey timeoutMs query
  623   case requestResult of
  624     Left err -> pure (Left err)
  625     Right req -> do
  626       start <- getCurrentTime
  627       httpResult <- runChatHttp mgr req
  628       end <- getCurrentTime
  629       let latencyMs = elapsedMs start end
  630       pure (handleChatHttpResult tool latencyMs httpResult)
  631 
  632 buildChatRequest :: Text -> Text -> Text -> Int -> Text -> IO (Either ExternalQueryError Request)
  633 buildChatRequest endpoint model apiKey timeoutMs query = do
  634   parsed <- try (parseRequest (T.unpack endpoint)) :: IO (Either HttpException Request)
  635   pure $ case parsed of
  636     Left err -> Left (classifyHttpException err)
  637     Right req0 ->
  638       let payload = encode $ object
  639             [ "model" .= model
  640             , "messages" .= [object ["role" .= ("user" :: Text), "content" .= query]]
  641             ]
  642           payloadBytes = fromIntegral (LBS.length payload)
  643       in if payloadBytes > llmMaxRequestBytes
  644            then Left (requestBodyTooLarge payloadBytes llmMaxRequestBytes)
  645            else Right req0
  646              { method = "POST"
  647              , requestHeaders =
  648                  [ ("Authorization", TE.encodeUtf8 (T.concat ["Bearer ", apiKey]))
  649                  , ("Content-Type", "application/json")
  650                  ]
  651              , requestBody = RequestBodyLBS payload
  652              , responseTimeout = responseTimeoutMicro (timeoutMicroseconds timeoutMs)
  653              , checkResponse = \_ _ -> pure ()
  654              }
  655 
  656 runChatHttp :: Manager -> Request -> IO (Either HttpException (Status, Either ExternalQueryError BS.ByteString))
  657 runChatHttp mgr req =
  658   try (withResponse req mgr $ \resp -> do
  659     bodyResult <- readResponseBodyLimited llmMaxResponseBytes (responseBody resp)
  660     pure (responseStatus resp, bodyResult))
  661 
  662 handleChatHttpResult
  663   :: ExternalTool
  664   -> Int
  665   -> Either HttpException (Status, Either ExternalQueryError BS.ByteString)
  666   -> Either ExternalQueryError ExternalQueryResponse
  667 handleChatHttpResult tool latencyMs httpResult =
  668   case httpResult of
  669     Left err -> Left (classifyHttpException err)
  670     Right (status, bodyResult) ->
  671       case bodyResult of
  672         Left err -> Left (classifyBodyReadFailure (statusCode status) err)
  673         Right rawBody ->
  674           case decodeLlmBodyLimited llmMaxResponseBytes rawBody of
  675             Left err -> Left err
  676             Right body -> responseFromStatus status body
  677   where
  678     responseFromStatus status body =
  679       let code = statusCode status
  680           diag = redactUpstreamError code body
  681       in if code >= 200 && code < 300
  682            then if T.null (T.strip body)
  683                   then Left EqeEmptyResponse
  684                   else Right ExternalQueryResponse
  685                          { eqrRawBody = T.take llmRawBodyTelemetryChars body
  686                          , eqrStructured = extractStructured body
  687                          , eqrToolName = etName tool
  688                          , eqrLatencyMs = latencyMs
  689                          }
  690            else case code of
  691              _ -> Left (classifyHttpStatusError code diag)
  692 
  693 classifyBodyReadFailure :: Int -> ExternalQueryError -> ExternalQueryError
  694 classifyBodyReadFailure code err =
  695   classifyHttpStatusError code $ T.concat
  696     [ "upstream_status="
  697     , T.pack (show code)
  698     , ":body_redacted:"
  699     , bodyReadFailureTag err
  700     ]
  701 
  702 classifyHttpStatusError :: Int -> Text -> ExternalQueryError
  703 classifyHttpStatusError code diag =
  704   case code of
  705     401 -> EqeAuthFailure diag
  706     403 -> EqeAuthFailure diag
  707     429 -> EqeRateLimited diag
  708     _ | code >= 500 && code < 600 -> EqeServerError diag
  709     _ | code >= 400 && code < 500 -> EqeInvalidResponse diag
  710     _ -> EqeInvalidResponse diag
  711 
  712 bodyReadFailureTag :: ExternalQueryError -> Text
  713 bodyReadFailureTag (EqeInvalidResponse msg) = msg
  714 bodyReadFailureTag EqeEmptyResponse = "empty_response"
  715 bodyReadFailureTag other = T.pack (show other)
  716 
  717 readResponseBodyLimited :: Int -> BodyReader -> IO (Either ExternalQueryError BS.ByteString)
  718 readResponseBodyLimited maxBytes reader = go 0 []
  719   where
  720     go total chunks = do
  721       chunk <- brRead reader
  722       if BS.null chunk
  723         then pure (Right (BS.concat (reverse chunks)))
  724         else do
  725           let total' = total + BS.length chunk
  726           if total' > maxBytes
  727             then pure (Left (responseTooLarge total' maxBytes))
  728             else go total' (chunk : chunks)
  729 
  730 decodeLlmBodyLimited :: Int -> BS.ByteString -> Either ExternalQueryError Text
  731 decodeLlmBodyLimited maxBytes raw
  732   | BS.length raw > maxBytes = Left (responseTooLarge (BS.length raw) maxBytes)
  733   | otherwise =
  734       case TE.decodeUtf8' raw of
  735         Left _ -> Left (EqeInvalidResponse "invalid_utf8_response")
  736         Right text -> Right text
  737 
  738 responseTooLarge :: Int -> Int -> ExternalQueryError
  739 responseTooLarge actual maxBytes =
  740   EqeInvalidResponse $ T.concat
  741     [ "response_too_large:max_bytes="
  742     , T.pack (show maxBytes)
  743     , ":actual_bytes="
  744     , T.pack (show actual)
  745     ]
  746 
  747 queryTooLarge :: Int -> Int -> ExternalQueryError
  748 queryTooLarge actual maxChars =
  749   EqeInvalidResponse $ T.concat
  750     [ "query_too_large:max_chars="
  751     , T.pack (show maxChars)
  752     , ":actual_chars="
  753     , T.pack (show actual)
  754     ]
  755 
  756 requestBodyTooLarge :: Int -> Int -> ExternalQueryError
  757 requestBodyTooLarge actual maxBytes =
  758   EqeInvalidResponse $ T.concat
  759     [ "request_too_large:max_bytes="
  760     , T.pack (show maxBytes)
  761     , ":actual_bytes="
  762     , T.pack (show actual)
  763     ]
  764 
  765 redactUpstreamError :: Int -> Text -> Text
  766 redactUpstreamError code body =
  767   T.concat
  768     [ "upstream_status="
  769     , T.pack (show code)
  770     , ":body_redacted:body_chars="
  771     , T.pack (show (T.length body))
  772     ]

  774 classifyHttpException :: HttpException -> ExternalQueryError
  775 classifyHttpException err =
  776   case err of
  777     InvalidUrlException _ _ -> EqeInvalidResponse "invalid_url"
  778     HttpExceptionRequest _ ResponseTimeout -> EqeTimeout "response_timeout"
  779     HttpExceptionRequest _ ConnectionTimeout -> EqeTimeout "connection_timeout"
  780     HttpExceptionRequest _ content -> EqeNetworkUnavailable (httpExceptionContentTag content)
  781 
  782 httpExceptionContentTag :: HttpExceptionContent -> Text
  783 httpExceptionContentTag content =
  784   "http_exception:" <> T.take 80 (T.takeWhile (\c -> c /= ' ' && c /= '\n') (T.pack (show content)))

  790 elapsedMs start end = floor ((realToFrac (diffUTCTime end start) :: Double) * 1000.0)

  792 -- | OpenAI-compatible chat-completion response envelope.
  793 -- Used to unwrap the assistant message content.
  794 data ChatCompletionResponse = ChatCompletionResponse
  795   { ccrChoices :: ![ChatCompletionChoice]
  796   }
  797   deriving stock (Generic)
  798 
  799 instance FromJSON ChatCompletionResponse where
  800   parseJSON = withObject "ChatCompletionResponse" $ \o ->
  801     ChatCompletionResponse <$> o .: "choices"

  803 data ChatCompletionChoice = ChatCompletionChoice
  804   { cccMessage :: !ChatCompletionMessage
  805   }
  806   deriving stock (Generic)

  808 instance FromJSON ChatCompletionChoice where
  809   parseJSON = withObject "ChatCompletionChoice" $ \o ->
  810     ChatCompletionChoice <$> o .: "message"

  813   { ccmContent :: !(Maybe Text)
  814   }
  815   deriving stock (Generic)

  817 instance FromJSON ChatCompletionMessage where
  818   parseJSON = withObject "ChatCompletionMessage" $ \o ->
  819     ChatCompletionMessage <$> o .:? "content"

  821 -- | Extract structured payload from a Mistral / Fireworks chat-completion
  822 -- response using a typed JSON decoder.
  823 --
  824 -- 1. Decode the body as an OpenAI-compatible 'ChatCompletionResponse'.
  825 -- 2. If successful, return the first choice's message content.
  826 -- 3. If decoding fails OR the content is empty/Null, return the raw
  827 --    body so downstream parsers can attempt direct JSON or legacy text.
  828 extractStructured :: Text -> Text
  829 extractStructured body =
  830   case decodeStrict (TE.encodeUtf8 body) :: Maybe ChatCompletionResponse of
  831     Nothing -> body
  832     Just ccr ->
  833       case ccrChoices ccr of
  834         [] -> body
  835         (choice : _) ->
  836           case ccmContent (cccMessage choice) of
  837             Nothing  -> body
  838             Just ""  -> body
  839             Just txt -> txt

  844 mockTableLookup _ [] = Nothing
