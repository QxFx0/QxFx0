   26 type RankedFamily = (CanonicalMoveFamily, Double)
   27 
   28 data LogicRule = LogicRule
   29   { lrFamily :: CanonicalMoveFamily
   30   , lrWeight :: Double
   31   , lrMatch :: MeaningAtom -> Bool
   32   }
   33 
   34 ruleTable :: [LogicRule]
   35 ruleTable =
   36   [ LogicRule CMRepair      semanticLogicRepairWeight      (\a -> case maTag a of Exhaustion _ -> True; _ -> False)
   37   , LogicRule CMContact     semanticLogicContactWeight     (\a -> case maTag a of NeedContact _ -> True; _ -> False)
   38   , LogicRule CMDefine      semanticLogicDefineWeight      (\a -> case maTag a of Searching _ -> True; _ -> False)
   39   , LogicRule CMReflect     semanticLogicReflectWeight     (\a -> case maTag a of NeedMeaning _ -> True; _ -> False)
   40   , LogicRule CMAnchor      semanticLogicAnchorWeight      (\a -> case maTag a of Anchoring _ -> True; _ -> False)
   41   , LogicRule CMClarify     semanticLogicClarifyWeight     (\a -> case maTag a of Verification _ -> True; _ -> False)
   42   , LogicRule CMDeepen      semanticLogicDeepenWeight      (\a -> case maTag a of AgencyFound _ -> True; _ -> False)
   43   , LogicRule CMConfront    semanticLogicConfrontWeight    (\a -> case maTag a of Contradiction _ _ -> True; _ -> False)
   44   , LogicRule CMDistinguish semanticLogicDistinguishWeight (\a -> case maTag a of Doubt _ -> True; _ -> False)
   45   , LogicRule CMHypothesis  semanticLogicHypothesisWeight  (\a -> case maTag a of Doubt _ -> True; _ -> False)
   46   ]
   47 
   48 runRule :: LogicRule -> [MeaningAtom] -> Maybe RankedFamily
   49 runRule rule atoms
   50   | any (lrMatch rule) atoms = Just (lrFamily rule, lrWeight rule)
   51   | otherwise = Nothing
   52 
   53 runSpecialRule :: CanonicalMoveFamily -> Double -> (Int -> Bool) -> [MeaningAtom] -> Maybe RankedFamily
   54 runSpecialRule family weight cond atoms
   55   | cond (length atoms) = Just (family, weight)
   56   | otherwise = Nothing
   57 
   58 runSemanticLogic :: AtomSet -> [RankedFamily]
   59 runSemanticLogic atoms =
   60   let rawAtoms = asAtoms atoms
   61       primaryResults = concatMap (\r -> maybe [] (:[]) (runRule r rawAtoms)) ruleTable
   62       specialResults = catMaybes
   63         [ runSpecialRule CMPurpose semanticSpecialPurposeWeight (> 3) rawAtoms
   64         , runSpecialRule CMDescribe semanticSpecialDescribeWeight (== 0) rawAtoms
   65         ]
   66       results = primaryResults ++ specialResults
   67       fallbacks = case results of
   68                     [] -> [(CMNextStep, semanticFallbackNextStepWeight), (CMGround, semanticFallbackGroundWeight)]
   69                     [(CMDescribe, _)] -> [(CMNextStep, semanticFallbackNextStepWeight), (CMGround, semanticFallbackGroundWeight)] ++ results
   70                     _ -> []
   71   in results ++ fallbacks
