  109 classifyWordUnitsMorph :: MorphBackend -> NormalizedInput -> IO [WordMeaningUnit]
  110 classifyWordUnitsMorph backend normalizedInput =
  111   fmap assignSyntacticRoles (mapM (classifyTokenMorph backend) (niTokens normalizedInput))

  113 classifyTokenMorph :: MorphBackend -> Text -> IO WordMeaningUnit
  114 classifyTokenMorph backend token = do
  115   morphToken <- analyzeMorphWithBackend backend token
  116   let lemma = mtLemma morphToken
  117       pos = mapInputPos (mtPOS morphToken)
  118       morphFeatures = guessMorphFeatures token <> morphFeaturesFromToken morphToken
  119       semanticClasses = semanticClassesForToken token
  120       discourseFunctions = discourseFunctionsForToken token
  121       confidence
  122         | pos == PosUnknown = 0.35
  123         | isFunctionWord token = 0.7
  124         | otherwise = 0.82
  125       ambiguityCandidates =
  126         case pos of
  127           PosUnknown -> [token]
  128           PosAdverb -> [token, lemma]
  129           PosAdjective -> [token, lemma]
  130           _ -> [lemma]
  131   pure WordMeaningUnit
  132       { wmuSurfaceForm = token
  133       , wmuLemma = lemma
  134       , wmuPartOfSpeech = pos
  135       , wmuMorphFeatures = morphFeatures
  136       , wmuSyntacticRole = SynUnknown
  137       , wmuSemanticClasses = semanticClasses
  138       , wmuDiscourseFunctions = discourseFunctions
  139       , wmuAmbiguityCandidates = ambiguityCandidates
  140       , wmuConfidence = confidence
  141       }

  143 morphFeaturesFromToken :: MorphToken -> [InputMorphFeature]
  144 morphFeaturesFromToken morphToken =
  145   caseFeatures <> numberFeatures <> personFeatures <> tenseFeatures <> moodFeatures
  146   where
  147     caseFeatures = case mtCase morphToken of
  148       Just Nominative -> [FeatCaseNom]
  149       Just Genitive -> [FeatCaseGen]
  150       Just Dative -> [FeatCaseDat]
  151       Just Accusative -> [FeatCaseAcc]
  152       Just Instrumental -> [FeatCaseIns]
  153       Just Prepositional -> [FeatCaseLoc]
  154       Nothing -> []
  155     numberFeatures = case mtNumber morphToken of
  156       Just Singular -> [FeatNumberSing]
  157       Just Plural -> [FeatNumberPlur]
  158       _ -> []
  159     personFeatures = case mtPerson morphToken of
  160       Just First -> [FeatPerson1]
  161       Just Second -> [FeatPerson2]
  162       Just Third -> [FeatPerson3]
  163       _ -> []
  164     tenseFeatures = case mtTense morphToken of
  165       Just Past -> [FeatTensePast]
  166       Just Present -> [FeatTensePres]
  167       Just Future -> [FeatTenseFut]
  168       _ -> []
  169     moodFeatures = case mtMood morphToken of
  170       Just Indicative -> [FeatMoodInd]
  171       Just ImperativeMood -> [FeatMoodImp]
  172       _ -> []

  174 assignSyntacticRoles :: [WordMeaningUnit] -> [WordMeaningUnit]
  175 assignSyntacticRoles units =
  176   zipWith assign [0 ..] units
  177   where
  178     rootIndex = findRootIndex units
  179     assign idx unit =
  180       unit {wmuSyntacticRole = inferRole rootIndex idx unit}
  181 
  182 inferRole :: Int -> Int -> WordMeaningUnit -> InputSyntacticRole
  183 inferRole rootIndex idx unit
  184   | idx == rootIndex = SynRoot
  185   | wmuPartOfSpeech unit == PosVerb = SynPredicate
  186   | idx < rootIndex && wmuPartOfSpeech unit `elem` [PosNoun, PosPronoun, PosNumeral] = SynSubject
  187   | idx > rootIndex && wmuPartOfSpeech unit `elem` [PosNoun, PosPronoun, PosNumeral] = SynObject
  188   | wmuPartOfSpeech unit == PosAdjective = SynAttribute
  189   | wmuPartOfSpeech unit `elem` [PosAdverb, PosPreposition] = SynCircumstance
  190   | wmuPartOfSpeech unit `elem` [PosConjunction, PosParticle, PosInterjection] = SynMarker
  191   | otherwise = SynUnknown
  192 
  193 findRootIndex :: [WordMeaningUnit] -> Int
  194 findRootIndex units =
  195   case findIndex ((== PosVerb) . wmuPartOfSpeech) units of
  196     Just idx -> idx
  197     Nothing ->
  198       case findIndex (\u -> wmuPartOfSpeech u `elem` [PosNoun, PosPronoun]) units of
  199         Just idx -> idx
  200         Nothing -> 0

  367 scoreRouteHint :: NormalizedInput -> [WordMeaningUnit] -> InputRouteHint -> InputRouteHint
  368 scoreRouteHint normalized units ruleHint =
  369   let clauseType = inferClauseType normalized
  370       normalizedText = niNormalizedText normalized
  371       scoredCandidates = map (buildRouteCandidate normalized units clauseType normalizedText ruleHint) routeCatalog
  372       bestCandidate =
  373         if null scoredCandidates
  374           then defaultCandidate ruleHint
  375           else maximumBy (comparing rcFinalScore) scoredCandidates
  376       ruleCandidate = fallbackRuleCandidate ruleHint scoredCandidates
  377       preserveRuleHint =
  378         irhTag ruleHint `elem`
  379           [ "misunderstanding"
  380           , "boundary_command"
  381           , "apology_repair"
  382           , "farewell_contact"
  383           , "gratitude_contact"
  384           , "affective_help"
  385           , "greeting_smalltalk"
  386           , "purpose_function"
  387           , "comparison_relation"
  388           , "disagreement_confront"
  389           , "next_step"
  390           , "short_dialogue_probe"
  391           , "concept_knowledge"
  392           , "self_knowledge"
  393           , "self_state"
  394           , "system_logic"
  395           , "operational_cause"
  396           , "world_cause"
  397           , "location_formation"
  398           , "generative_prompt"
  399           ]
  400       useBest =
  401         (not preserveRuleHint)
  402           && ( irhTag ruleHint == "unknown"
  403                 || rcFinalScore bestCandidate > rcFinalScore ruleCandidate + 0.05
  404              )
  405       chosen = if useBest then bestCandidate else ruleCandidate
  406   in InputRouteHint
  407       { irhType = rcRouteType chosen
  408       , irhTag = rcTag chosen
  409       , irhReason = rcReason chosen
  410       , irhRuleScore = rcRuleScore chosen
  411       , irhSemanticScore = rcSemanticScore chosen
  412       , irhSyntacticScore = rcSyntacticScore chosen
  413       , irhEmbeddingScore = rcEmbeddingScore chosen
  414       , irhEvidence = rcEvidence chosen
  415       , irhConfidence = rcFinalScore chosen
  416       }

  430 buildRouteCandidate
  431   :: NormalizedInput
  432   -> [WordMeaningUnit]
  433   -> InputClauseType
  434   -> Text
  435   -> InputRouteHint
  436   -> (InputRouteType, Text, Text, [Text])
  437   -> RouteCandidate
  438 buildRouteCandidate normalized units clauseType normalizedText ruleHint (routeType, tag, reason, prototypes) =
  439   let ruleScore = ruleAlignmentScore ruleHint routeType tag
  440       semanticScore = semanticScoreForTag normalized units tag
  441       syntacticScore = syntacticScoreForRoute clauseType routeType
  442       embeddingScore = embeddingScoreForTag normalizedText prototypes
  443       finalScore = clamp01 ((ruleScore * 0.42) + (semanticScore * 0.23) + (syntacticScore * 0.15) + (embeddingScore * 0.20))
  444       evidence =
  445         [ "rule=" <> showScore ruleScore
  446         , "semantic=" <> showScore semanticScore
  447         , "syntax=" <> showScore syntacticScore
  448         , "embedding=" <> showScore embeddingScore
  449         , "final=" <> showScore finalScore
  450         ]
  451   in RouteCandidate
  452       { rcRouteType = routeType
  453       , rcTag = tag
  454       , rcReason = reason
  455       , rcRuleScore = ruleScore
  456       , rcSemanticScore = semanticScore
  457       , rcSyntacticScore = syntacticScore
  458       , rcEmbeddingScore = embeddingScore
  459       , rcFinalScore = finalScore
  460       , rcEvidence = evidence
  461       }

  477 fallbackRuleCandidate :: InputRouteHint -> [RouteCandidate] -> RouteCandidate
  478 fallbackRuleCandidate ruleHint candidates =
  479   case findByTag (irhTag ruleHint) candidates of
  480     Just candidate -> candidate
  481     Nothing ->
  482       case findByType (irhType ruleHint) candidates of
  483         Just candidate -> candidate
  484         Nothing -> defaultCandidate ruleHint

  588 embeddingScoreForTag :: Text -> [Text] -> Double
  589 embeddingScoreForTag normalizedText prototypes =
  590   let inputVec = fallbackEmbedding (T.unpack normalizedText)
  591       similarities =
  592         [ (realToFrac (cosineSimilarity inputVec (fallbackEmbedding (T.unpack prototype))) + 1.0) / 2.0
  593         | prototype <- prototypes
  594         ]
  595   in case similarities of
  596       [] -> 0.0
  597       _ -> maximum similarities
