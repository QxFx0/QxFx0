   51 rebuildGovernedPerspectiveState :: [GovernanceEvent] -> Either Text PerspectiveRegistry
   52 rebuildGovernedPerspectiveState events = do
   53   ordered <- canonicalizeGovernanceHistory events
   54   foldGovernanceEvent emptyPerspectiveRegistry ordered
   55 
   56 verifyPerspectiveRegistryRebuild :: [GovernanceEvent] -> PerspectiveRegistry -> Either Text PerspectiveRegistry
   57 verifyPerspectiveRegistryRebuild events expectedRegistry = do
   58   rebuilt <- rebuildGovernedPerspectiveState events
   59   if rebuilt == expectedRegistry
   60     then Right rebuilt
   61     else Left "governance perspective registry rebuild mismatch"
   62 
   63 rebuildGovernanceProjection :: [GovernanceEvent] -> Either Text GovernanceProjection
   64 rebuildGovernanceProjection events = do
   65   ordered <- canonicalizeGovernanceHistory events
   66   registry <- foldGovernanceEvent emptyPerspectiveRegistry ordered
   67   let projections = buildActivePerspectiveProjections registry
   68       governedRefs = map governanceProjectionRef ordered
   69       meta = ProjectionMeta
   70         { pmProjectionVersion = currentProjectionVersion
   71         , pmReducerVersion = currentReducerVersion
   72         , pmSnapshotTurn = Just (prLastUpdatedTurn registry)
   73         }
   74       checksum = governanceProjectionChecksum meta registry projections governedRefs
   75   pure GovernanceProjection
   76     { gpMeta = meta
   77     , gpPerspectiveRegistry = registry
   78     , gpActivePerspectiveProjections = projections
   79     , gpGovernedRefs = governedRefs
   80     , gpProjectionChecksum = checksum
   81     }

   83 rebuildGovernedViews :: SystemState -> Either Text (PerspectiveRegistry, GovernanceProjection)
   84 rebuildGovernedViews ss = do
   85   if not (truthContractIsAuthoritative (ssTruthContractStatus ss))
   86     then Left "governance_replay_non_authoritative_truth_contract"
   87     else pure ()
   88   projection <- rebuildGovernanceProjection (ssGovernanceHistory ss)
   89   pure (gpPerspectiveRegistry projection, projection)
   90 
   91 rebuildGovernedSystemState :: SystemState -> Either Text SystemState
   92 rebuildGovernedSystemState ss = do
   93   (registry, projection) <- rebuildGovernedViews ss
   94   pure ss
   95     { ssPerspectiveRegistry = registry
   96     , ssGovernanceProjection = projection
   97     }

   99 governanceReplayProof :: [GovernanceEvent] -> PerspectiveRegistry -> Either Text [GovernanceProvenanceLink]
  100 governanceReplayProof events expectedRegistry = do
  101   _ <- verifyPerspectiveRegistryRebuild events expectedRegistry
  102   governanceProvenanceTrail events

  104 foldGovernanceEvent :: PerspectiveRegistry -> [GovernanceEvent] -> Either Text PerspectiveRegistry
  105 foldGovernanceEvent registry [] = Right registry
  106 foldGovernanceEvent registry (event:rest) = do
  107   next <- applyGovernanceEvent registry event
  108   foldGovernanceEvent next rest
  109 
  110 applyGovernanceEvent :: PerspectiveRegistry -> GovernanceEvent -> Either Text PerspectiveRegistry
  111 applyGovernanceEvent registry event =
  112   case gePayload event of
  113     GpPerspective payload -> applyPerspectiveGovernancePayload registry (geEnvelope event) payload
  114     _ ->
  115       Right registry { prLastUpdatedTurn = fromMaybe (prLastUpdatedTurn registry) (geeTurnId (geEnvelope event)) }
  116 
  117 applyPerspectiveGovernancePayload :: PerspectiveRegistry -> GovernanceEventEnvelope -> PerspectivePayload -> Either Text PerspectiveRegistry
  118 applyPerspectiveGovernancePayload registry envelope payload = do
  119   ensurePerspectiveReplayConsistency envelope payload
  120   case geeDecision envelope of
  121     GovObserveOnly -> Right registry { prLastUpdatedTurn = eventTurn }
  122     GovDeny -> Right registry { prLastUpdatedTurn = eventTurn }
  123     GovQuarantine -> applyResolved PpdQuarantine
  124     GovAcceptBounded -> applyResolved PpdAcceptBounded
  125     GovPromote -> applyResolvedPerspectivePromote
  126     GovSuspend -> applyResolved PpdSuspendActive
  127     GovRollback -> applyResolved PpdRollbackPrior
  128     GovFreeze -> Left "governance replay cannot apply GovFreeze to perspective payload"
  129   where
  130     eventTurn = fromMaybe (isTurnCount (pibIdentitySlice (ppPerspectiveInput payload))) (geeTurnId envelope)
  131 
  132     applyResolved :: PerspectivePromotionDecision -> Either Text PerspectiveRegistry
  133     applyResolved expectedDecision
  134       | ppResolvedDecision payload == expectedDecision =
  135           Right (applyPerspectiveDecision eventTurn registry (ppPerspectiveInput payload) (ppPerspectiveCandidate payload) expectedDecision)
  136       | otherwise =
  137           Left
  138             ( "governance replay payload decision mismatch: expected "
  139                 <> payloadDecisionTag expectedDecision
  140                 <> ", got "
  141                 <> payloadDecisionTag (ppResolvedDecision payload)
  142             )
  143 
  144     applyResolvedPerspectivePromote :: Either Text PerspectiveRegistry
  145     applyResolvedPerspectivePromote =
  146       case ppResolvedDecision payload of
  147         PpdPromoteEndorsed -> Right (applyPerspectiveDecision eventTurn registry (ppPerspectiveInput payload) (ppPerspectiveCandidate payload) PpdPromoteEndorsed)
  148         PpdReviseActive -> Right (applyPerspectiveDecision eventTurn registry (ppPerspectiveInput payload) (ppPerspectiveCandidate payload) PpdReviseActive)
  149         other ->
  150           Left
  151             ( "governance replay payload decision mismatch: GovPromote requires promote/revise payload, got "
  152                 <> payloadDecisionTag other
  153             )
  154 
  155 ensurePerspectiveReplayConsistency :: GovernanceEventEnvelope -> PerspectivePayload -> Either Text ()
  156 ensurePerspectiveReplayConsistency envelope payload = do
  157   let canonicalDecision = geeDecision envelope
  158       resolvedDecision = ppResolvedDecision payload
  159       sameSubject = geeSubject envelope == SubjectPerspective (ppPerspectiveId payload)
  160   if not sameSubject
  161     then Left "governance replay subject mismatch for perspective payload"
  162     else pure ()
  163   case geeResolvedDecision envelope of
  164     Just resolvedGov
  165       | resolvedGov /= canonicalDecision ->
  166           Left "governance replay envelope resolved decision mismatch"
  167     _ -> pure ()
  168   case canonicalDecision of
  169     GovDeny
  170       | resolvedDecision == PpdObserveOnly -> pure ()
  171       | governanceDecisionFromPerspectivePromotion resolvedDecision == GovDeny -> pure ()
  172       | otherwise ->
  173           Left
  174             ( "governance replay envelope/payload decision mismatch: envelope="
  175                 <> T.pack (show canonicalDecision)
  176                 <> ", payload="
  177                 <> payloadDecisionTag resolvedDecision
  178             )
  179     _
  180       | governanceDecisionFromPerspectivePromotion resolvedDecision == canonicalDecision -> pure ()
  181       | otherwise ->
  182           Left
  183             ( "governance replay envelope/payload decision mismatch: envelope="
  184                 <> T.pack (show canonicalDecision)
  185                 <> ", payload="
  186                 <> payloadDecisionTag resolvedDecision
  187             )
  188 
  189 payloadTag :: GovernancePayload -> Text
  190 payloadTag payload =
  191   case payload of
  192     GpPerspective _ -> "GpPerspective"
  193     GpClaimStance _ -> "GpClaimStance"
  194     GpCapability _ -> "GpCapability"
  195     GpFreeze _ -> "GpFreeze"
  196     GpCarry _ -> "GpCarry"
  197     GpNormativeRevision _ -> "GpNormativeRevision"
  198 
  199 payloadDecisionTag :: PerspectivePromotionDecision -> Text
  200 payloadDecisionTag = T.pack . show

  202 governanceProjectionRef :: GovernanceEvent -> GovernedProjectionRef
  203 governanceProjectionRef event =
  204   let envelope = geEnvelope event
  205   in GovernedProjectionRef
  206       { gprEventId = geeId envelope
  207       , gprSubject = geeSubject envelope
  208       , gprDecision = geeDecision envelope
  209       , gprLifecycleStatus = geeLifecycleStatus envelope
  210       , gprEffectRef = geeEffectRef envelope
  211       }
