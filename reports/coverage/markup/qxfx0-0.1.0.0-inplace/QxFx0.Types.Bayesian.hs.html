   29 data HiddenBelief
   30   = UserWantsDefine
   31   | UserWantsCompare
   32   | UserWantsConfront
   33   | UserSeeksSupport

   35   | UserIsDistressed

   37   deriving anyclass (NFData, ToJSON, FromJSON)
   38 
   39 instance ToJSONKey HiddenBelief where

   44 type BeliefState = Map HiddenBelief Double

   46 initialBeliefs :: BeliefState
