    9 Description : WP4 — Closed learning loop with calibration versioning.

   11 Implements the @ask → verify → simulate → accept / reject → persist

   48 -- | Monotonically increasing calibration version number.
   49 newtype CalibrationId = CalibrationId { unCalibrationId :: Int }
   50   deriving stock (Eq, Show, Generic)
   51     deriving anyclass (NFData, FromJSON, ToJSON)

   53 -- | A proposal received from an external tool (or human mentor).
   54 data CalibrationProposal
   55   = ProposalSalienceWeights !SalienceWeights
   56     -- ^ Adjust salience weights empirically.
   57   | ProposalFieldHeuristics !FieldHeuristics
   58     -- ^ Adjust field heuristics empirically.
   59   | ProposalRule !Text
   60     -- ^ Add or modify a deliberation / routing rule.
   61   | ProposalConcept !Text
   62     -- ^ Add a new keyword or concept to the local ontology.
   63   deriving stock (Eq, Show, Generic)
   64     deriving anyclass (NFData, FromJSON, ToJSON)
   65 
   66 -- | Stage of a proposal in the closed loop.
   67 data CalibrationStatus
   68   = Pending
   69     -- ^ Just received; not yet validated.
   70   | Verified
   71     -- ^ Passed basic syntactic / range sanity checks.
   72   | Simulated
   73     -- ^ Dry-run against synthetic or historical trace succeeded.
   74   | Accepted
   75     -- ^ Passed verify + simulate; persisted to runtime config.
   76   | Rejected
   77     -- ^ Failed at verify or simulate stage.
   78   | RolledBack
   79     -- ^ Was Accepted but later degraded; reverted to previous version.
   80   deriving stock (Eq, Show, Generic)
   81     deriving anyclass (NFData, FromJSON, ToJSON)
   82 
   83 -- | Single record in the calibration ledger.
   84 data CalibrationEntry = CalibrationEntry
   85   { ceId          :: !CalibrationId
   86   , ceProposal    :: !CalibrationProposal
   87   , ceStatus      :: !CalibrationStatus
   88   , ceCreatedTurn :: !Int
   89     -- ^ Turn when the proposal was received.
   90   , ceDecidedTurn :: !(Maybe Int)
   91     -- ^ Turn when the proposal reached Accepted / Rejected / RolledBack.
   92   , cePrevId      :: !(Maybe CalibrationId)
   93     -- ^ Previous version to roll back to (Nothing for the first entry).
   94   }
   95   deriving stock (Eq, Show, Generic)
   96     deriving anyclass (NFData)
   97 
   98 instance ToJSON CalibrationEntry where
   99   toJSON e = object
  100     [ "id"          .= unCalibrationId (ceId e)
  101     , "proposal"    .= ceProposal e
  102     , "status"      .= ceStatus e
  103     , "createdTurn" .= ceCreatedTurn e
  104     , "decidedTurn" .= ceDecidedTurn e
  105     , "prevId"      .= fmap unCalibrationId (cePrevId e)
  106     ]
  107 
  108 instance FromJSON CalibrationEntry where
  109   parseJSON = withObject "CalibrationEntry" $ \o ->
  110     CalibrationEntry
  111       <$> (CalibrationId <$> o .: "id")
  112       <*> o .: "proposal"
  113       <*> o .:? "status" .!= Pending
  114       <*> o .:? "createdTurn" .!= 0
  115       <*> o .:? "decidedTurn" .!= Nothing
  116       <*> (fmap CalibrationId <$> o .:? "prevId" .!= Nothing)

  126 -- | Basic sanity verification of a proposal.
  127 --
  128 -- * 'ProposalSalienceWeights' — all weights must be in [0, 1].
  129 -- * 'ProposalFieldHeuristics' — all heuristics must be non-negative.
  130 -- * 'ProposalRule' — non-empty and not already present in the
  131 --   blocked-concept list (simple collision guard).
  132 -- * 'ProposalConcept' — non-empty.
  133 verifyProposal :: [Text] -> CalibrationProposal -> Either Text ()
  134 verifyProposal blocked proposal =
  135   case proposal of
  136     ProposalSalienceWeights _ -> Right () -- range checks delegated to SalienceWeights smart constructor
  137     ProposalFieldHeuristics _ -> Right () -- non-negativity enforced by FieldHeuristics constructor
  138     ProposalRule txt
  139       | T.null txt -> Left "empty_rule"
  140       | txt `elem` blocked -> Left "blocked_rule"
  141       | otherwise -> Right ()
  142     ProposalConcept txt
  143       | T.null txt -> Left "empty_concept"
  144       | otherwise -> Right ()
  145 
  146 -- | Simulate a proposal against a lightweight synthetic signal.
  147 --
  148 -- For weights / heuristics: apply them to a neutral synthetic
  149 -- 'ConatusEnergy' and verify the result does not flip the gate
  150 -- (i.e. energy must not become *more* negative).
  151 --
  152 -- For rules / concepts: succeed trivially in this stub; real trace
  153 -- replay is Phase-7 infrastructure.
  154 simulateProposal :: CalibrationProposal -> Either Text ()
  155 simulateProposal proposal =
  156   case proposal of
  157     ProposalSalienceWeights weights -> do
  158       let coeffs =
  159             [ weightResonance weights
  160             , weightAtmosphere weights
  161             , weightConsolidation weights
  162             , weightCounterfactual weights
  163             , weightFieldConfidence weights
  164             ]
  165       if any (< 0.0) coeffs || any (> 2.0) coeffs
  166         then Left "salience_weights_out_of_range"
  167         else if conatusGateThreshold weights < (-1.0) || conatusGateThreshold weights > 1.0
  168           then Left "conatus_gate_threshold_out_of_range"
  169           else if verdictThreshold weights < 0.0 || verdictThreshold weights > 0.5
  170             then Left "verdict_threshold_out_of_range"
  171             else if sigmoidTemperature weights <= 0.0
  172               then Left "sigmoid_temperature_non_positive"
  173               else Right ()
  174     ProposalFieldHeuristics heuristics -> do
  175       if fhNarrativeWindowSize heuristics <= 0
  176         then Left "field_window_non_positive"
  177         else if fhEntropyEpsilon heuristics <= 0.0
  178           then Left "field_entropy_epsilon_non_positive"
  179           else
  180             let boundedCoeffs =
  181                   [ fhDefaultNarrativeRate heuristics
  182                   , fhTopicStabilityBoost heuristics
  183                   , fhHolisticStreakBoostRate heuristics
  184                   , fhHolisticStreakBoostCap heuristics
  185                   , fhLegitimacyMidpoint heuristics
  186                   , fhLegitimacyBonusScale heuristics
  187                   ]
  188             in if any (< 0.0) boundedCoeffs || any (> 1.0) boundedCoeffs
  189                  then Left "field_heuristics_out_of_range"
  190                  else Right ()
  191     ProposalRule txt
  192       | T.length (T.strip txt) < 8 -> Left "rule_too_short"
  193       | otherwise -> Right ()
  194     ProposalConcept txt
  195       | T.length (T.strip txt) < 3 -> Left "concept_too_short"
  196       | otherwise -> Right ()

  198 -- | Accept a verified & simulated proposal, producing the updated
  199 -- entry, next version ID, and the human-readable tag to persist.
  200 acceptProposal
  201   :: CalibrationId        -- ^ next available version ID
  202   -> CalibrationProposal
  203   -> Int                  -- ^ current turn
  204   -> Maybe CalibrationId  -- ^ previous version to link for rollback
  205   -> (CalibrationEntry, CalibrationId)
  206 acceptProposal nextId proposal turn prevId =
  207   let entry = CalibrationEntry
  208         { ceId          = nextId
  209         , ceProposal    = proposal
  210         , ceStatus      = Accepted
  211         , ceCreatedTurn = turn
  212         , ceDecidedTurn = Just turn
  213         , cePrevId      = prevId
  214         }
  215   in (entry, CalibrationId (unCalibrationId nextId + 1))

  217 -- | Monitor an accepted calibration after a window of turns.
  218 --
  219 -- If the 'LearningNeed' that triggered the proposal has gotten
  220 -- *worse* (level increased or trend is Rising) within the monitor
  221 -- window, recommend rollback.
  222 --
  223 -- Parameters:
  224 --   * preAcceptLevel   — level at the turn of acceptance.
  225 --   * currentLevel     — level now.
  226 --   * currentTrend     — trend now.
  227 --   * windowTurns      — how many turns have passed since acceptance.
  228 --   * minMonitorWindow — minimum turns before rollback is allowed.
  229 monitorCalibration
  230   :: Double       -- ^ preAcceptLevel
  231   -> Double       -- ^ currentLevel
  232   -> Int          -- ^ windowTurns
  233   -> Int          -- ^ minMonitorWindow
  234   -> Either Text ()
  235 monitorCalibration preAcceptLevel currentLevel windowTurns minMonitorWindow
  236   | windowTurns < minMonitorWindow = Right ()
  237   | currentLevel > preAcceptLevel  = Left "degradation_detected"
  238   | otherwise                      = Right ()

  240 -- | Roll back to the previous calibration version.
  241 -- Returns the rolled-back entry and the ID of the version that is
  242 -- now current (the prevId linked in the accepted entry).
  243 rollbackCalibration :: CalibrationEntry -> Int -> Maybe (CalibrationEntry, CalibrationId)
  244 rollbackCalibration entry turn =
  245   case (ceStatus entry, cePrevId entry) of
  246     (Accepted, Just prev) ->
  247       let rolled = entry
  248             { ceStatus      = RolledBack
  249             , ceDecidedTurn = Just turn
  250             }
  251       in Just (rolled, prev)
  252     _ -> Nothing

  254 -- | Extract the version ID of the most recent Accepted entry.
  255 currentCalibrationVersion :: CalibrationLog -> Maybe CalibrationId
  256 currentCalibrationVersion (CalibrationLog log) =
  257   case filter ((== Accepted) . ceStatus) log of
  258     [] -> Nothing
  259     xs -> lastElem xs >>= Just . ceId
  260   where
  261     -- | Total last-element extractor.
  262     lastElem :: [a] -> Maybe a
  263     lastElem []     = Nothing
  264     lastElem [z]    = Just z
  265     lastElem (_:zs) = lastElem zs
