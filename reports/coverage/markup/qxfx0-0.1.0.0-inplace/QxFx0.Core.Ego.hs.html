   42 readAgencyFromHistory :: Text -> Double
   43 readAgencyFromHistory t =
   44   maybe 0.0 id (readMaybe (T.unpack t))

   46 updateEgoFromTurn :: EgoState -> CanonicalMoveFamily -> Double -> EgoState
   47 updateEgoFromTurn ego fam tensionDelta =
   48   let newTension = egoTension ego * egoTensionDecayFactor + tensionDelta * egoTensionInputFactor
   49       agencyDelta = case fam of
   50         CMGround      -> egoAgencyDeltaGround
   51         CMDefine      -> egoAgencyDeltaDefine
   52         CMDistinguish -> egoAgencyDeltaDistinguish
   53         CMReflect     -> egoAgencyDeltaReflect
   54         CMDescribe    -> egoAgencyDeltaDescribe
   55         CMPurpose     -> egoAgencyDeltaPurpose
   56         CMHypothesis  -> egoAgencyDeltaHypothesis
   57         CMRepair      -> egoAgencyDeltaRepair
   58         CMContact     -> egoAgencyDeltaContact
   59         CMAnchor      -> egoAgencyDeltaAnchor
   60         CMClarify     -> egoAgencyDeltaClarify
   61         CMDeepen      -> egoAgencyDeltaDeepen
   62         CMConfront    -> egoAgencyDeltaConfront
   63         CMNextStep    -> egoAgencyDeltaNextStep
   64       newAgency = min 1.0 $ max 0.0 $ egoAgency ego * egoAgencyDecayFactor + (egoAgency ego + agencyDelta) * egoAgencyInputFactor
   65       mission = inferMission fam
   66       newDynamics = updateSubjectDynamics (egoSubjectDynamics ego) newAgency
   67       newUserState = (egoUserState ego) { usReadiness = newAgency }
   68   in ego
   69       { egoTension = min 1.0 $ max 0.0 newTension
   70       , egoAgency = newAgency
   71       , egoMission = mission
   72       , egoUserState = newUserState
   73       , egoSubjectDynamics = newDynamics
   74       }

   76 inferMission :: CanonicalMoveFamily -> Text
   77 inferMission fam = case lookup (textShow fam) missionTexts of
   78   Just t  -> t
   79   Nothing -> textShow fam

   81 detectTrend :: [Double] -> Trend
   82 detectTrend [] = Plateau
   83 detectTrend [_] = Plateau
   84 detectTrend vals =
   85   let recent = take egoTrendWindow vals
   86       len = length recent
   87       diffs = if len >= 2
   88                 then zipWith (-) (drop 1 recent) (take (len - 1) recent)
   89                 else []
   90       avgDiff = if null diffs then 0.0 else sum diffs / fromIntegral (length diffs)
   91   in if avgDiff > egoTrendRisingThreshold then Rising
   92      else if avgDiff < egoTrendFallingThreshold then Falling
   93      else Plateau
   94 
   95 updateSubjectDynamics :: SubjectDynamics -> Double -> SubjectDynamics
   96 updateSubjectDynamics sd agency =
   97   let histTexts = sdSemanticHistory sd
   98       histDoubles = map readAgencyFromHistory histTexts
   99       newHistDoubles = agency : take egoHistoryRetention histDoubles
  100       newHistTexts = take egoHistoryLimit $ map (T.pack . show) newHistDoubles
  101       trend = detectTrend newHistDoubles
  102       shiftTurn = if trend /= sdTrend sd then 0 else sdLastShiftTurn sd + 1
  103   in sd
  104       { sdTrend = trend
  105       , sdSemanticHistory = newHistTexts
  106       , sdLastShiftTurn = shiftTurn
  107       }
