   82 -- | Classify orbital phase from collapse/freeze risk signals.
   83 classifyOrbitalPhaseSimple :: Double -> Double -> OrbitalPhase
   84 classifyOrbitalPhaseSimple collapseRisk freezeRisk
   85   | collapseRisk > orbitalHighRiskThreshold = OrbitCollapseRisk
   86   | freezeRisk   > orbitalHighRiskThreshold = OrbitFreezeRisk
   87   | collapseRisk > orbitalCounterpressureThreshold = OrbitCounterpressure
   88   | freezeRisk   > orbitalCounterpressureThreshold = OrbitRecovery
   89   | otherwise           = OrbitStable

   91 -- | Classify encounter mode from orbital phase and current tension.
   92 classifyEncounterModeSimple :: OrbitalPhase -> Double -> EncounterMode
   93 classifyEncounterModeSimple OrbitCollapseRisk _    = EncounterPressure
   94 classifyEncounterModeSimple OrbitFreezeRisk _       = EncounterHolding
   95 classifyEncounterModeSimple OrbitCounterpressure _  = EncounterCounterweight
   96 classifyEncounterModeSimple OrbitRecovery _         = EncounterRecovery
   97 classifyEncounterModeSimple OrbitStable tension
   98   | tension > encounterMirroringTensionThreshold = EncounterMirroring
   99   | otherwise        = EncounterExploration

  101 -- | Update rolling orbital memory with the latest phase/encounter outcome.
  102 updateOrbitalMemorySimple :: OrbitalMemory -> OrbitalPhase -> EncounterMode -> CoreDirective -> OrbitalMemory
  103 updateOrbitalMemorySimple om phase encounter directive =
  104   let addPhase p hist = take orbitalHistoryLimit (p : hist)
  105       addEnc e hist = take orbitalHistoryLimit (e : hist)
  106       streakFor = case phase of
  107         OrbitStable          -> omStableStreak om + 1
  108         OrbitCollapseRisk    -> omCollapseStreak om + 1
  109         OrbitFreezeRisk      -> omFreezeStreak om + 1
  110         OrbitCounterpressure -> omCounterpressureStreak om + 1
  111         OrbitRecovery        -> omRecoveryStreak om + 1
  112       resetOther = case phase of
  113         OrbitStable          -> om { omCollapseStreak = 0, omFreezeStreak = 0, omCounterpressureStreak = 0, omRecoveryStreak = 0 }
  114         OrbitCollapseRisk    -> om { omStableStreak = 0, omFreezeStreak = 0, omCounterpressureStreak = 0, omRecoveryStreak = 0 }
  115         OrbitFreezeRisk      -> om { omStableStreak = 0, omCollapseStreak = 0, omCounterpressureStreak = 0, omRecoveryStreak = 0 }
  116         OrbitCounterpressure -> om { omStableStreak = 0, omCollapseStreak = 0, omFreezeStreak = 0, omRecoveryStreak = 0 }
  117         OrbitRecovery        -> om { omStableStreak = 0, omCollapseStreak = 0, omFreezeStreak = 0, omCounterpressureStreak = 0 }
  118       om' = resetOther
  119       distEst = case phase of
  120         OrbitCollapseRisk    -> orbitalDistEstCollapseRisk
  121         OrbitFreezeRisk      -> orbitalDistEstFreezeRisk
  122         OrbitCounterpressure -> orbitalDistEstCounterpressure
  123         OrbitRecovery        -> orbitalDistEstRecovery
  124         OrbitStable          -> orbitalDistEstStable
  125       alpha = orbitalEmaAlpha
  126       newDist = alpha * distEst + (1.0 - alpha) * omAvgDistance om
  127       newAttr = alpha * cdContactBias directive + (1.0 - alpha) * omAvgAttraction om
  128       newRepl = alpha * cdBoundaryBias directive + (1.0 - alpha) * omAvgRepulsion om
  129       setPhaseStreak = case phase of
  130         OrbitStable          -> om' { omStableStreak = streakFor }
  131         OrbitCollapseRisk    -> om' { omCollapseStreak = streakFor }
  132         OrbitFreezeRisk      -> om' { omFreezeStreak = streakFor }
  133         OrbitCounterpressure -> om' { omCounterpressureStreak = streakFor }
  134         OrbitRecovery        -> om' { omRecoveryStreak = streakFor }
  135   in setPhaseStreak
  136      { omCurrentPhase     = phase
  137      , omPhaseHistory     = addPhase phase (omPhaseHistory om)
  138      , omEncounterHistory = addEnc encounter (omEncounterHistory om)
  139      , omAvgDistance       = newDist
  140      , omAvgAttraction     = newAttr
  141      , omAvgRepulsion      = newRepl
  142      , omLastContactBias   = cdContactBias directive
  143      , omLastBoundaryBias  = cdBoundaryBias directive
  144      }

  146 -- | Steer directive contact/boundary biases from current orbital phase.
  147 steerDirectiveWithOrbitalSimple :: OrbitalMemory -> CoreDirective -> CoreDirective
  148 steerDirectiveWithOrbitalSimple om d =
  149   let contactBoost = case omCurrentPhase om of
  150         OrbitRecovery        -> orbitalContactBoostRecovery
  151         OrbitFreezeRisk      -> orbitalContactBoostFreezeRisk
  152         OrbitCounterpressure -> orbitalContactPenaltyCounterpressure
  153         _                    -> 0.0
  154       boundaryBoost = case omCurrentPhase om of
  155         OrbitCollapseRisk    -> orbitalBoundaryBoostCollapseRisk
  156         OrbitCounterpressure -> orbitalBoundaryBoostCounterpressure
  157         OrbitRecovery        -> orbitalBoundaryPenaltyRecovery
  158         _                    -> 0.0
  159   in d { cdContactBias = clamp01 (cdContactBias d + contactBoost)
  160         , cdBoundaryBias = clamp01 (cdBoundaryBias d + boundaryBoost)
  161         }
