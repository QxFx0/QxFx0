   11 
   12 import Control.DeepSeq (NFData)
   13 import Data.Aeson (FromJSON(..), ToJSON(..), withText)
   14 import Data.Text (Text)
   15 import qualified Data.Text as T
   16 import GHC.Generics (Generic)
   17 
   18 data AgdaVerificationStatus
   19   = AgdaVerified
   20   | AgdaMissingWitness
   21   | AgdaDecodeFailed
   22   | AgdaVersionMismatch
   23   | AgdaMissingInput
   24   | AgdaHashMismatch
   25   | AgdaUnexpectedInput
   26   | AgdaInvalid
   27   deriving stock (Eq, Ord, Show, Read, Generic)
   28   deriving anyclass (NFData)
   29 
   30 agdaVerificationReady :: AgdaVerificationStatus -> Bool
   31 agdaVerificationReady AgdaVerified = True
   32 agdaVerificationReady _ = False
   33 
   34 agdaVerificationStatusText :: AgdaVerificationStatus -> Text
   35 agdaVerificationStatusText status =
   36   case status of
   37     AgdaVerified -> "verified"
   38     AgdaMissingWitness -> "missing_witness"
   39     AgdaDecodeFailed -> "decode_failed"
   40     AgdaVersionMismatch -> "version_mismatch"
   41     AgdaMissingInput -> "missing_input"
   42     AgdaHashMismatch -> "hash_mismatch"
   43     AgdaUnexpectedInput -> "unexpected_input"
   44     AgdaInvalid -> "invalid"
   45 
   46 instance ToJSON AgdaVerificationStatus where
   47   toJSON = toJSON . agdaVerificationStatusText
   48 
   49 instance FromJSON AgdaVerificationStatus where
   50   parseJSON = withText "AgdaVerificationStatus" $ \raw ->
   51     case T.toLower (T.strip raw) of
   52       "verified" -> pure AgdaVerified
   53       "missing_witness" -> pure AgdaMissingWitness
   54       "decode_failed" -> pure AgdaDecodeFailed
   55       "version_mismatch" -> pure AgdaVersionMismatch
   56       "missing_input" -> pure AgdaMissingInput
   57       "hash_mismatch" -> pure AgdaHashMismatch
   58       "unexpected_input" -> pure AgdaUnexpectedInput
   59       "invalid" -> pure AgdaInvalid
   60       other -> fail ("Unknown AgdaVerificationStatus: " <> T.unpack other)
