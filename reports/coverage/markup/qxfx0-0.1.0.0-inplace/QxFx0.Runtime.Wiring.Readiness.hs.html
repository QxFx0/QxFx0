   43 data BackendReadiness = BackendReadiness
   44   { brEmbeddingAlive       :: !Bool
   45   , brEmbeddingOperational :: !Bool
   46   , brEmbeddingExplicit    :: !Bool
   47   , brEmbeddingBackend     :: !Text
   48   , brEmbeddingQuality     :: !Text
   49   , brNixOperational       :: !Bool
   50   , brNixIssues            :: ![Text]
   51   , brDatalogReady         :: !Bool
   52   , brDatalogIssues        :: ![Text]
   53   , brAgdaStatus           :: !AgdaVerificationStatus
   54   , brAgdaWitnessPath      :: !Text
   55   , brAgdaIssues           :: ![Text]
   56   } deriving stock (Eq, Show)

   58 wireVerifyAgda :: RuntimeContext -> IO AgdaVerificationStatus
   59 wireVerifyAgda ctx = awrStatus <$> readAgdaWitnessReportCached ctx
   60 
   61 readAgdaWitnessReportCached :: RuntimeContext -> IO AgdaWitnessReport
   62 readAgdaWitnessReportCached ctx = do
   63   report <- readAgdaWitnessReport
   64   modifyMVar_ (rtrAgdaWitness (rcTurn ctx)) $ \_ -> pure (Just report)
   65   pure report
   66 
   67 probeBackendReadiness :: IO BackendReadiness
   68 probeBackendReadiness = do
   69   healthCache <- newMVar Nothing
   70   embeddingHealth <- checkEmbeddingHealth healthCache
   71   nixProbe <- probeNixEvaluator
   72   datalogProbe <- probeDatalogBackend
   73   agdaReport <- readAgdaWitnessReport
   74   pure (mkBackendReadiness embeddingHealth nixProbe datalogProbe agdaReport)
   75 
   76 wireRuntimeReadiness :: RuntimeContext -> IO BackendReadiness
   77 wireRuntimeReadiness ctx = do
   78   embeddingHealth <- readEmbeddingHealth ctx
   79   nixProbe <- probeNixEvaluator
   80   datalogProbe <- probeDatalogBackendCached ctx
   81   agdaReport <- readAgdaWitnessReportCached ctx
   82   pure (mkBackendReadiness embeddingHealth nixProbe datalogProbe agdaReport)
   83 
   84 mkBackendReadiness
   85   :: EmbeddingHealth
   86   -> Either Text ()
   87   -> Either Text ()
   88   -> AgdaWitnessReport
   89   -> BackendReadiness
   90 mkBackendReadiness embeddingHealth nixProbe datalogProbe agdaReport =
   91   BackendReadiness
   92     { brEmbeddingAlive = ehStrictReady embeddingHealth
   93     , brEmbeddingOperational = ehOperational embeddingHealth
   94     , brEmbeddingExplicit = ehExplicit embeddingHealth
   95     , brEmbeddingBackend = embeddingBackendText (ehBackend embeddingHealth)
   96     , brEmbeddingQuality = embeddingQualityText (ehQuality embeddingHealth)
   97     , brNixOperational = either (const False) (const True) nixProbe
   98     , brNixIssues = either (:[]) (const []) nixProbe
   99     , brDatalogReady = either (const False) (const True) datalogProbe
  100     , brDatalogIssues = either (:[]) (const []) datalogProbe
  101     , brAgdaStatus = awrStatus agdaReport
  102     , brAgdaWitnessPath = T.pack (awrPath agdaReport)
  103     , brAgdaIssues = awrIssues agdaReport
  104     }

  106 wireNixEval :: NixCache -> Maybe FilePath -> Text -> Double -> Double -> IO NixGuardStatus
  107 wireNixEval cache mNixPath concept agency tension =
  108   case mNixPath of
  109     Just nixPath -> cachedNixEval cache nixPath concept agency tension
  110     Nothing -> pure (Unavailable "No Nix guard path")
  111 
  112 checkNixWithCache :: NixCache -> Maybe FilePath -> Text -> Double -> Double -> IO NixGuardStatus
  113 checkNixWithCache = wireNixEval
  114 
  115 verifyAgdaCached :: RuntimeContext -> IO AgdaVerificationStatus
  116 verifyAgdaCached = wireVerifyAgda

  118 getNixPathCached :: RuntimeContext -> IO (Maybe FilePath)
  119 getNixPathCached = resolveNixPath

  121 probeNixEvaluator :: IO (Either Text ())
  122 probeNixEvaluator = do
  123   status <- NixGuard.getNixGuardStatus
  124   pure $
  125     case status of
  126       Right _ -> Right ()
  127       Left err -> Left err
  128 
  129 probeDatalogBackend :: IO (Either Text ())
  130 probeDatalogBackend = do
  131   execResult <- Datalog.resolveSouffleExecutable
  132   case execResult of
  133     Left err -> pure (Left err)
  134     Right executable -> do
  135       probe <- Datalog.compileAndRunDatalogWithExecutable executable "" CMGround
  136       pure $ case probe of
  137         Left err -> Left err
  138         Right _ -> Right ()
  139 
  140 probeDatalogBackendCached :: RuntimeContext -> IO (Either Text ())
  141 probeDatalogBackendCached ctx = do
  142   execResult <- resolveSouffleExecutableCached ctx
  143   case execResult of
  144     Left err -> pure (Left err)
  145     Right executable -> do
  146       probe <- Datalog.compileAndRunDatalogWithExecutable executable "" CMGround
  147       pure $ case probe of
  148         Left err -> Left err
  149         Right _ -> Right ()
