   50 planRouteEffects :: SystemState -> TurnInput -> TurnSignals -> RouteEffectPlan
   51 planRouteEffects ss ti ts =
   52   let frame = tiFrame ti
   53       atomSet = tiAtomSet ti
   54       recommendedFamily = tiRecommendedFamily ti
   55       intuitPosterior = tsIntuitPosterior ts
   56       mCommitment =
   57         case tiEssence ti of
   58           EssenceCommitted _ c -> Just c
   59           EssenceUncommitted _ -> Nothing
   60       courtesyPred = fmap (\c p -> case validatePlan c p of Right _ -> True; Left _ -> False) mCommitment
   61       rd =
   62         routeFamily
   63           recommendedFamily
   64           frame
   65           atomSet
   66           (tiNextUserState ti)
   67           ss
   68           (F.toList (ssHistory ss))
   69           (ipfRawText frame)
   70           (tiIsNixBlocked ti)
   71           (tiBestTopic ti)
   72           (tsCurrentNarrative ts)
   73           intuitPosterior
   74           (tiConatusEnergy ti)
   75           (tiField ti)
   76           courtesyPred
   77       family = rdFamily rd
   78       atomTags = map maTag (asAtoms atomSet)
   79    in RouteEffectPlan
   80         { repStatic = RouteStatic {rsRoutingDecision = rd}
   81         , repRouteTurnInput = ti
   82         , repShadowRequest = RouteReqShadow family (forceForFamily family) atomTags
   83         , repAgdaRequest = RouteReqAgdaVerify
   84         }

   86 resolveRouteEffects :: PipelineIO -> RouteEffectPlan -> IO RouteEffectResults
   87 resolveRouteEffects pio effectPlan = do
   88   let scheduledRequests :: [(Text, TurnEffectRequest)]
   89       scheduledRequests =
   90         scheduleTurnEffects pio (tiConatusEnergy (repRouteTurnInput effectPlan))
   91           ( [ ("shadow", TurnReqShadow family force atomTags)
   92             | RouteReqShadow family force atomTags <- [repShadowRequest effectPlan]
   93             ]
   94          <> [ ("agda", TurnReqAgdaVerify)
   95             | RouteReqAgdaVerify <- [repAgdaRequest effectPlan]
   96             ]
   97           )
   98   resolved <- forConcurrently scheduledRequests $ \(label, request) -> do
   99     result <- resolveTurnEffect pio request
  100     pure (label, result)
  101   let shadowResult =
  102         fromMaybe unexpectedShadowResult $ do
  103           (_, result) <- firstMatch (\(label, _) -> label == "shadow") resolved
  104           shadowResultFromTurnResult result
  105       agdaStatus =
  106         fromMaybe AgdaInvalid $ do
  107           (_, result) <- firstMatch (\(label, _) -> label == "agda") resolved
  108           agdaStatusFromTurnResult result
  109   let agdaReady = agdaVerificationReady agdaStatus
  110       strictMode = pipelineRuntimeMode pio == RuntimeStrict
  111       agdaMsg = "agda_status=" <> agdaVerificationStatusText agdaStatus
  112   when (strictMode && not agdaReady) $
  113     throwQxFx0 (AgdaGateError agdaMsg)
  114   when (not agdaReady) $
  115     hPutStrLnWarning ("Agda R5 verification: " <> agdaVerificationStatusText agdaStatus)
  116   pure RouteEffectResults
  117     { rerShadowResult = shadowResult
  118     , rerAgdaStatus = agdaStatus
  119     }
  120 
  121 shadowResultFromTurnResult :: TurnEffectResult -> Maybe ShadowResult
  122 shadowResultFromTurnResult result =
  123   case result of
  124     TurnResShadow datalogVerdict shadowStatus divergence snapshotId diagnostics ->
  125       Just ShadowResult
  126         { srDatalogVerdict = datalogVerdict
  127         , srStatus = shadowStatus
  128         , srDivergence = divergence
  129         , srSnapshotId = snapshotId
  130         , srDiagnostics = diagnostics
  131         }
  132     _ ->
  133       Nothing
  134 
  135 agdaStatusFromTurnResult :: TurnEffectResult -> Maybe AgdaVerificationStatus
  136 agdaStatusFromTurnResult result =
  137   case result of
  138     TurnResAgdaVerify agdaStatus -> Just agdaStatus
  139     _ -> Nothing
  140 
  141 firstMatch :: (a -> Bool) -> [a] -> Maybe a
  142 firstMatch predicate = go
  143   where
  144     go [] = Nothing
  145     go (x:xs)
  146       | predicate x = Just x
  147       | otherwise = go xs
  148 
  149 unexpectedShadowResult :: ShadowResult
  150 unexpectedShadowResult =
  151   ShadowResult
  152     { srDatalogVerdict = Nothing
  153     , srStatus = ShadowUnavailable
  154     , srDivergence = emptyShadowDivergence {sdKind = ShadowBridgeSkew}
  155     , srSnapshotId = ShadowSnapshotId "shadow:route_unexpected_effect"
  156     , srDiagnostics = ["unexpected_route_shadow_request"]
  157     }
