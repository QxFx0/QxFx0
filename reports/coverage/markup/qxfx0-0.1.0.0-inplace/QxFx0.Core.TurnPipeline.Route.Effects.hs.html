   47 planRouteEffects :: SystemState -> TurnInput -> TurnSignals -> RouteEffectPlan
   48 planRouteEffects ss ti ts =
   49   let frame = tiFrame ti
   50       atomSet = tiAtomSet ti
   51       recommendedFamily = tiRecommendedFamily ti
   52       intuitPosterior = tsIntuitPosterior ts
   53       rd =
   54         routeFamily
   55           recommendedFamily
   56           frame
   57           atomSet
   58           (tiNextUserState ti)
   59           ss
   60           (F.toList (ssHistory ss))
   61           (ipfRawText frame)
   62           (tiIsNixBlocked ti)
   63           (tiBestTopic ti)
   64           (tsCurrentNarrative ts)
   65           intuitPosterior
   66       family = rdFamily rd
   67       atomTags = map maTag (asAtoms atomSet)
   68    in RouteEffectPlan
   69         { repStatic = RouteStatic {rsRoutingDecision = rd}
   70         , repShadowRequest = RouteReqShadow family (forceForFamily family) atomTags
   71         , repAgdaRequest = RouteReqAgdaVerify
   72         }

   74 resolveRouteEffects :: PipelineIO -> RouteEffectPlan -> IO RouteEffectResults
   75 resolveRouteEffects pio effectPlan = do
   76   (shadowResult, agdaStatus) <-
   77     concurrently
   78       (resolveShadowEffect pio (repShadowRequest effectPlan))
   79       (resolveAgdaEffect pio (repAgdaRequest effectPlan))
   80   let agdaReady = agdaVerificationReady agdaStatus
   81       strictMode = pipelineRuntimeMode pio == RuntimeStrict
   82       agdaMsg = "agda_status=" <> agdaVerificationStatusText agdaStatus
   83   when (strictMode && not agdaReady) $
   84     throwQxFx0 (AgdaGateError agdaMsg)
   85   when (not agdaReady) $
   86     hPutStrLnWarning ("Agda R5 verification: " ++ T.unpack (agdaVerificationStatusText agdaStatus))
   87   pure RouteEffectResults
   88     { rerShadowResult = shadowResult
   89     , rerAgdaStatus = agdaStatus
   90     }
   91 
   92 resolveShadowEffect :: PipelineIO -> RouteEffectRequest -> IO ShadowResult
   93 resolveShadowEffect pio request =
   94   case routeRequestToTurnEffect request of
   95     Just turnRequest -> do
   96       result <- resolveTurnEffect pio turnRequest
   97       case result of
   98         TurnResShadow datalogVerdict shadowStatus divergence snapshotId diagnostics ->
   99           pure ShadowResult
  100             { srDatalogVerdict = datalogVerdict
  101             , srStatus = shadowStatus
  102             , srDivergence = divergence
  103             , srSnapshotId = snapshotId
  104             , srDiagnostics = diagnostics
  105             }
  106         _ ->
  107           pure unexpectedShadowResult
  108     Nothing ->
  109       pure unexpectedShadowResult
  110 
  111 resolveAgdaEffect :: PipelineIO -> RouteEffectRequest -> IO AgdaVerificationStatus
  112 resolveAgdaEffect pio request =
  113   case routeRequestToTurnEffect request of
  114     Just turnRequest -> do
  115       result <- resolveTurnEffect pio turnRequest
  116       case result of
  117         TurnResAgdaVerify agdaStatus -> pure agdaStatus
  118         _ -> pure AgdaInvalid
  119     Nothing ->
  120       pure AgdaInvalid
  121 
  122 routeRequestToTurnEffect :: RouteEffectRequest -> Maybe TurnEffectRequest
  123 routeRequestToTurnEffect request =
  124   case request of
  125     RouteReqShadow family force atomTags ->
  126       Just (TurnReqShadow family force atomTags)
  127     RouteReqAgdaVerify ->
  128       Just TurnReqAgdaVerify
  129 
  130 unexpectedShadowResult :: ShadowResult
  131 unexpectedShadowResult =
  132   ShadowResult
  133     { srDatalogVerdict = Nothing
  134     , srStatus = ShadowUnavailable
  135     , srDivergence = emptyShadowDivergence {sdKind = ShadowBridgeSkew}
  136     , srSnapshotId = ShadowSnapshotId "shadow:route_unexpected_effect"
  137     , srDiagnostics = ["unexpected_route_shadow_request"]
  138     }
