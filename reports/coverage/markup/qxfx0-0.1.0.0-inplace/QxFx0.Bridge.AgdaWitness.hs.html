   59 witnessVersion :: Int

   62 resolveAgdaWitnessPath :: IO FilePath
   63 resolveAgdaWitnessPath = do
   64   mExplicit <- lookupEnv "QXFX0_AGDA_WITNESS"
   65   case fmap normalise mExplicit of
   66     Just path -> pure path
   67     Nothing -> do
   68       stateDir <- resolveQxFx0StateDir
   69       pure (stateDir </> "agda-witness.json")

   71 writeAgdaWitness :: IO FilePath
   72 writeAgdaWitness = do
   73   verifyResult <- verifyR5WithAgda
   74   case verifyResult of
   75     AgdaPass -> do
   76       paths <- resolveResourcePaths
   77       hashes <- currentWitnessHashes paths
   78       witnessPath <- resolveAgdaWitnessPath
   79       createDirectoryIfMissing True (takeDirectory witnessPath)
   80       BL.writeFile witnessPath (encode AgdaWitness { awVersion = witnessVersion, awFiles = hashes })
   81       pure witnessPath
   82     other ->
   83       throwQxFx0 (RuntimeInitError ("Cannot write Agda witness: " <> T.pack (renderAgdaFailure other)))
   84 
   85 writeStubAgdaWitness :: FilePath -> IO ()
   86 writeStubAgdaWitness witnessPath = do
   87   paths <- resolveResourcePaths
   88   hashes <- currentWitnessHashes paths
   89   createDirectoryIfMissing True (takeDirectory witnessPath)
   90   BL.writeFile witnessPath (encode AgdaWitness { awVersion = witnessVersion, awFiles = hashes })

   92 verifyAgdaWitnessFresh :: IO Bool
   93 verifyAgdaWitnessFresh = awrFresh <$> readAgdaWitnessReport
   94 
   95 readAgdaWitnessReport :: IO AgdaWitnessReport
   96 readAgdaWitnessReport = do
   97   witnessPath <- resolveAgdaWitnessPath
   98   witnessExists <- doesFileExist witnessPath
   99   if not witnessExists
  100     then pure AgdaWitnessReport
  101       { awrPath = witnessPath
  102       , awrStatus = AgdaMissingWitness
  103       , awrFresh = False
  104       , awrIssues = ["missing_witness"]
  105       }
  106     else do
  107       body <- BL.readFile witnessPath
  108       case eitherDecode body of
  109         Left err ->
  110           pure AgdaWitnessReport
  111             { awrPath = witnessPath
  112             , awrStatus = AgdaDecodeFailed
  113             , awrFresh = False
  114             , awrIssues = ["decode_failed:" <> T.pack err]
  115             }
  116         Right witness
  117           | awVersion witness /= witnessVersion ->
  118               pure AgdaWitnessReport
  119                 { awrPath = witnessPath
  120                 , awrStatus = AgdaVersionMismatch
  121                 , awrFresh = False
  122                 , awrIssues =
  123                     [ "version_mismatch:expected="
  124                         <> T.pack (show witnessVersion)
  125                         <> ",got="
  126                         <> T.pack (show (awVersion witness))
  127                     ]
  128                 }
  129           | otherwise -> do
  130               paths <- resolveResourcePaths
  131               currentHashes <- currentWitnessHashes paths
  132               let issues = compareWitnessHashes (awFiles witness) currentHashes
  133                   status =
  134                     if null issues
  135                       then AgdaVerified
  136                       else classifyAgdaIssues issues
  137               pure AgdaWitnessReport
  138                 { awrPath = witnessPath
  139                 , awrStatus = status
  140                 , awrFresh = agdaVerificationReady status
  141                 , awrIssues = issues
  142                 }
  143 
  144 classifyAgdaIssues :: [Text] -> AgdaVerificationStatus
  145 classifyAgdaIssues issues
  146   | any (== "missing_witness") issues = AgdaMissingWitness
  147   | any (T.isPrefixOf "decode_failed:") issues = AgdaDecodeFailed
  148   | any (T.isPrefixOf "version_mismatch:") issues = AgdaVersionMismatch
  149   | any (T.isPrefixOf "missing_input:") issues = AgdaMissingInput
  150   | any (T.isPrefixOf "hash_mismatch:") issues = AgdaHashMismatch
  151   | any (T.isPrefixOf "unexpected_input:") issues = AgdaUnexpectedInput
  152   | otherwise = AgdaInvalid

  154 resolveQxFx0StateDir :: IO FilePath
  155 resolveQxFx0StateDir = do
  156   mStateDir <- lookupEnv "QXFX0_STATE_DIR"
  157   case fmap normalise mStateDir of
  158     Just dir -> pure dir
  159     Nothing -> do
  160       mXdgStateHome <- lookupEnv "XDG_STATE_HOME"
  161       mHome <- lookupEnv "HOME"
  162       pure $
  163         case fmap normalise mXdgStateHome of
  164           Just xdgStateHome -> xdgStateHome </> "qxfx0"
  165           Nothing -> fromMaybe "." ((</> ".local/state/qxfx0") . normalise <$> mHome)

  167 currentWitnessHashes :: ResourcePaths -> IO (Map.Map Text Text)
  168 currentWitnessHashes paths = do
  169   hashedEntries <- mapM hashWitnessInput (witnessInputs paths)
  170   pure (Map.fromList hashedEntries)
  171 
  172 witnessInputs :: ResourcePaths -> [(Text, FilePath)]
  173 witnessInputs paths =
  174   let root = rpResourceDir paths
  175       specDir = takeDirectory (rpAgdaSpec paths)
  176       specLabel file = T.pack ("spec" </> file)
  177   in sortOn fst
  178       [ (specLabel "R5Core.agda", rpAgdaSpec paths)
  179       , (specLabel "Sovereignty.agda", specDir </> "Sovereignty.agda")
  180       , (specLabel "Legitimacy.agda", specDir </> "Legitimacy.agda")
  181       , (specLabel "LexiconContract.agda", specDir </> "LexiconContract.agda")
  182       , (specLabel "LexiconData.agda", specDir </> "LexiconData.agda")
  183       , (specLabel "LexiconProof.agda", specDir </> "LexiconProof.agda")
  184       , (specLabel "r5-snapshot.tsv", rpAgdaSnapshot paths)
  185       , ("src/QxFx0/Types/Domain.hs", root </> "src" </> "QxFx0" </> "Types" </> "Domain.hs")
  186       ]

  188 hashWitnessInput :: (Text, FilePath) -> IO (Text, Text)
  189 hashWitnessInput (label, path) = do
  190   exists <- doesFileExist path
  191   if not exists
  192     then throwQxFx0 (RuntimeInitError ("Witness input missing: " <> T.pack path))
  193     else do
  194       contents <- BL.readFile path
  195       pure (label, hashBytes contents)
  196 
  197 hashBytes :: BL.ByteString -> Text
  198 hashBytes bytes =
  199   T.pack . concatMap byteToHex . BS.unpack $ SHA256.hashlazy bytes
  200 
  201 byteToHex :: Word8 -> String
  202 byteToHex byte =
  203   let raw = showHex byte ""
  204   in if length raw == 1 then '0' : raw else raw

  206 compareWitnessHashes :: Map.Map Text Text -> Map.Map Text Text -> [Text]
  207 compareWitnessHashes recorded current =
  208   missingIssues ++ mismatchIssues ++ unexpectedIssues
  209   where
  210     missingIssues =
  211       [ "missing_input:" <> label
  212       | label <- Map.keys current
  213       , Map.notMember label recorded
  214       ]
  215     mismatchIssues =
  216       [ "hash_mismatch:" <> label
  217       | (label, currentHash) <- Map.toList current
  218       , Just recordedHash <- [Map.lookup label recorded]
  219       , recordedHash /= currentHash
  220       ]
  221     unexpectedIssues =
  222       [ "unexpected_input:" <> label
  223       | label <- Map.keys recorded
  224       , Map.notMember label current
  225       ]

  227 renderAgdaFailure :: AgdaVerificationResult -> String
  228 renderAgdaFailure AgdaPass = "unexpected success state"
  229 renderAgdaFailure (AgdaTypeCheckFailed detail) = T.unpack detail
  230 renderAgdaFailure (AgdaNotAvailable detail) = T.unpack detail
  231 renderAgdaFailure (AgdaSnapshotMismatch mismatches) =
  232   "snapshot mismatch: "
  233     <> T.unpack
  234       (T.intercalate "; " [fam <> " -> " <> detail | (fam, detail) <- mismatches])
