   15 fallbackEmbedding :: String -> Embedding
   16 fallbackEmbedding input =
   17   let ws = words (filter (`notElem` (".,!?:;\"'" :: String)) input)
   18       rawVec = V.create $ do
   19         v <- MV.replicate embeddingDim 0.0
   20         forM_ (zip [(0 :: Int) ..] ws) $ \(i, w) -> do
   21           let h = stableHash w
   22               idx = abs h `mod` embeddingDim
   23               val = if h > 0 then 1.0 else -1.0
   24           curr <- MV.read v idx
   25           MV.write v idx (curr + val + fromIntegral (i `mod` 3) * 0.1)
   26         pure v
   27       norm = magnitude rawVec
   28    in if norm > 0
   29         then V.map (/ norm) rawVec
   30         else V.replicate embeddingDim 0.0
   31 
   32 stableHash :: String -> Int
   33 stableHash = go 0
   34   where
   35     go acc [] = if acc == 0 then 1 else acc
   36     go acc (c : cs) = go ((acc * 31 + fromEnum c) `mod` (maxBound `div` 2)) cs

   38 cosineSimilarity :: Embedding -> Embedding -> Float
   39 cosineSimilarity v1 v2
   40   | m1 == 0 || m2 == 0 = 0.0
   41   | otherwise =
   42       let sim = dotProduct v1 v2 / (m1 * m2)
   43        in if isNaN sim || isInfinite sim then 0.0 else sim
   44   where
   45     dotProduct a b = V.sum $ V.zipWith (*) a b
   46     m1 = magnitude v1
   47     m2 = magnitude v2
   48 
   49 magnitude :: Embedding -> Float
   50 magnitude v = sqrt $ V.sum $ V.map (\x -> x * x) v
