   40 instance ToJSON SemanticConfig where
   41   toJSON = genericToJSON defaultOptions
   42     { fieldLabelModifier = \lbl -> case lbl of
   43         ('s':'c':rest) -> drop 1 (camelToSnake rest)
   44         _              -> map toLower lbl
   45     }
   46 
   47 instance FromJSON SemanticConfig where
   48   parseJSON = genericParseJSON defaultOptions
   49     { fieldLabelModifier = \lbl -> case lbl of
   50         ('s':'c':rest) -> drop 1 (camelToSnake rest)
   51         _              -> map toLower lbl
   52     }
   53 
   54 camelToSnake :: String -> String
   55 camelToSnake [] = []
   56 camelToSnake (c:cs)
   57   | c `elem` ['A'..'Z'] = '_' : toLower c : camelToSnake cs
   58   | otherwise = c : camelToSnake cs
