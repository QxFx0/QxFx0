   23 truthContractIsAuthoritative :: TruthContractStatus -> Bool
   24 truthContractIsAuthoritative truthStatus =
   25   truthStatus `elem` [CanonicalSurfacePreserved, AssembledSurfacePreserved]
   26 
   27 truthContractAllowsStrongMutation :: TruthContractStatus -> Bool
   28 truthContractAllowsStrongMutation = truthContractIsAuthoritative
   29 
   30 truthContractAllowsHardKnowledgeTone :: TruthContractStatus -> Bool
   31 truthContractAllowsHardKnowledgeTone CanonicalSurfacePreserved = True
   32 truthContractAllowsHardKnowledgeTone _ = False
   33 
   34 authorityClassIsAuthoritative :: AuthorityClass -> Bool
   35 authorityClassIsAuthoritative authority =
   36   authority `elem` [AuthorityCanonical, AuthorityAssembled]
   37 
   38 applyTruthContractCeiling :: TruthContractStatus -> EpistemicStatus -> EpistemicStatus
   39 applyTruthContractCeiling = capEpistemicByTruthContract
   40 
   41 capByTruthContract :: TruthContractStatus -> EpistemicStatus -> EpistemicStatus
   42 capByTruthContract = capEpistemicByTruthContract
   43 
   44 capEpistemicByTruthContract :: TruthContractStatus -> EpistemicStatus -> EpistemicStatus
   45 capEpistemicByTruthContract truthStatus epistemic
   46   | truthContractIsAuthoritative truthStatus = epistemic
   47   | otherwise =
   48       case epistemic of
   49         Known confidence -> Probable confidence
   50         Probable confidence -> Uncertain confidence
   51         other -> other
   52 
   53 capCommitmentStrengthByTruthContract :: TruthContractStatus -> Double -> Double
   54 capCommitmentStrengthByTruthContract truthStatus strength =
   55   case truthStatus of
   56     CanonicalSurfacePreserved -> strength
   57     AssembledSurfacePreserved -> min 0.75 strength
   58     _ -> min 0.45 strength
   59 
   60 replayProvenanceStatusForAuthority :: AuthorityClass -> ReplayProvenanceStatus
   61 replayProvenanceStatusForAuthority authority =
   62   case authority of
   63     AuthorityCanonical -> ReplayProvenanceComplete
   64     AuthorityAssembled -> ReplayProvenanceComplete
   65     AuthorityGeneratedArtifact -> ReplayProvenanceLegacyIncomplete
   66     AuthorityLegacyIncomplete -> ReplayProvenanceLegacyIncomplete
   67     AuthorityRecovery -> ReplayProvenanceLegacyIncomplete
   68     AuthorityFallback -> ReplayProvenanceLegacyIncomplete
   69     AuthorityShim -> ReplayProvenanceLegacyIncomplete
   70     AuthorityDefault -> ReplayProvenanceLegacyIncomplete
   71 
   72 replayProvenanceStatusForOutcome :: ExecutedTurnOutcome -> ReplayProvenanceStatus
   73 replayProvenanceStatusForOutcome = replayProvenanceStatusForAuthority . etoAuthorityClass
   74 
   75 normalizedReplayProvenanceStatus :: ReplayProvenanceStatus -> AuthorityClass -> ReplayProvenanceStatus
   76 normalizedReplayProvenanceStatus replayStatus authority
   77   | authorityClassIsAuthoritative authority = replayStatus
   78   | otherwise = ReplayProvenanceLegacyIncomplete
   79 
   80 truthContractRebindRenderedText :: TruthContractStatus -> AuthorityClass -> Text -> Text
   81 truthContractRebindRenderedText truthStatus authority rendered
   82   | truthContractIsAuthoritative truthStatus = rendered
   83   | authority == AuthorityGeneratedArtifact = rebind "Локально сгенерированный вариант: observational only."
   84   | authority == AuthorityLegacyIncomplete = rebind "Трасса неполная: authoritative replay unavailable."
   85   | authority == AuthorityFallback = rebind "Использован fallback-маршрут: поверхность удержана в осторожном режиме."
   86   | authority == AuthorityRecovery = rebind "Включён non-expansive recovery: ответ удержан в режиме безопасного восстановления."
   87   | authority == AuthorityShim = rebind "Использован compatibility shim: поверхность не считается канонической."
   88   | authority == AuthorityDefault = rebind "Использован default-маршрут: поверхность не усиливает semantic authority."
   89   | otherwise = rendered
   90   where
   91     rebind note = T.intercalate "\n" [rendered, note]
