    6 {-|
    7 Module      : QxFx0.Self.Field
    8 Description : Phase-4 right-hemispheric observation summary.
    9 
   10 A typed, pure realisation of the five-component @Field@ named in
   11 @docs\/adr\/0007-dual-mode-conatus.md@ and pinned by
   12 @docs\/adr\/0009-right-hemisphere-field.md@.
   13 
   14 Phase 3 (commit @20d5611@, @QxFx0.Self.Adjunction@) shipped a
   15 single-'Double' stub for 'Field'. This module replaces that stub
   16 with the substantive five-component record:
   17 
   18 * 'Resonance'       — peak similarity of the current turn to its
   19   recent context.
   20 * 'Atmosphere'      — two-dimensional valence\/arousal affect.
   21 * 'FieldConfidence' — internal-coherence score, derived from the
   22   rest.
   23 * 'Consolidation'   — narrative-integration scalar over the
   24   recent window (genuinely temporal).
   25 * 'Counterfactual'  — diversity of plausible alternative parses.
   26 
   27 Each component is a 'newtype' around a refined numeric value,
   28 with a documented range and a documented operational meaning.
   29 The 'Field' record is strict in every field.
   30 
   31 == What this module is /not/
   32 
   33 This module is the algebraic /shape/ of the right-hemispheric
   34 observation summary. It does /not/ source values from the runtime,
   35 does /not/ touch the turn pipeline, and does /not/ know about
   36 @Holistic@\/@Formal@ from @QxFx0.Self.Adjunction@ — that module
   37 imports /this/ one, not the other way round.
   38 
   39 Sourcing of values from runtime signals (semantic embedding,
   40 consciousness-loop narrative tone, Bayesian posterior diversity,
   41 etc.) is Phase-5 work and lives outside this module.
   42 
   43 == Combination laws
   44 
   45 Per ADR-0009 §2.2, 'Field' is /not/ a global 'Monoid'. The natural
   46 combination law differs by component:
   47 
   48 * 'combineResonance'       uses 'max'  — \"strongest echo found\".
   49 * 'combineAtmosphere'      uses weighted average of valence and
   50   arousal — affect drifts; combining two snapshots is interpolation.
   51 * 'combineFieldConfidence' uses 'min'  — combined confidence is at
   52   most each individual.
   53 * 'combineConsolidation'   uses additive (clipped at @1.0@) —
   54   consolidation accumulates over a window.
   55 * 'combineCounterfactual'  uses 'max'  — preserve highest
   56   counterfactual signal.
   57 
   58 Field-level combination is exposed as 'combineField' parameterised
   59 by an explicit 'CombineMode'.
   60 
   61 == History stub

   65 buffer with a most-recent-wins summariser; this is enough to let
   66 downstream modules import the type and the operations, and lets
   67 Phase 5 grow the substantive sliding-window summary without
   68 breaking imports.

  129 -- * 'atmosphereValence' in @[-1, 1]@: @-1@ negative … @1@ positive.

  189 -- | The right-hemispheric observation summary at one moment in
  190 -- time. A snapshot, not a history; trajectories live in
  191 -- 'FieldHistory'.
  192 data Field = Field
  193   { fieldResonance      :: !Resonance
  194   , fieldAtmosphere     :: !Atmosphere
  195   , fieldConfidence     :: !FieldConfidence
  196   , fieldConsolidation  :: !Consolidation
  197   , fieldCounterfactual :: !Counterfactual

  200 -- | The \"no observation yet\" zero. Per ADR-0009 §4.4,
  201 -- 'fieldConfidence' is set to @1.0@: a system that has not yet
  202 -- observed anything is uninformed, not unconfident.
  203 emptyField :: Field
  204 emptyField =
  205   Field
  206     { fieldResonance      = Resonance       0.0
  207     , fieldAtmosphere     = Atmosphere      0.0 0.0
  208     , fieldConfidence     = FieldConfidence 1.0
  209     , fieldConsolidation  = Consolidation   0.0
  210     , fieldCounterfactual = Counterfactual  0.0
  211     }

  213 -- | Derive 'FieldConfidence' from the other four components.
  214 --
  215 -- The default formula reduces 'Atmosphere' to its arousal
  216 -- magnitude (the @[0, 1]@-valued dimension) and computes
  217 -- @1 - dispersion@, where dispersion is the variance of the four
  218 -- scalarised components, normalised so a uniform field gives
  219 -- @1.0@ and a maximally split field (two zeros, two ones) gives
  220 -- @0.0@.
  221 --
  222 -- Phase 5 may override this default with a richer derivation; the
  223 -- formula here is documented to be transparent and total, not
  224 -- canonical.
  225 deriveFieldConfidence :: Field -> FieldConfidence
  226 deriveFieldConfidence f =
  227   FieldConfidence (1.0 - clampUnit normalised)
  228   where
  229     xs =
  230       [ unResonance       (fieldResonance      f)
  231       , atmosphereArousal (fieldAtmosphere     f)
  232       , unConsolidation   (fieldConsolidation  f)
  233       , unCounterfactual  (fieldCounterfactual f)
  234       ]
  235     n          = fromIntegral (length xs) :: Double
  236     mu         = sum xs / n
  237     var        = sum [(x - mu) * (x - mu) | x <- xs] / n
  238     -- Maximum variance for n=4 samples in [0,1] is 0.25
  239     -- (achieved when half are 0 and half are 1).
  240     normalised = var / 0.25

  246 -- | Tunable parameters for the four Phase-5.5d heuristics that
  247 -- populate a 'Field' from runtime signals.  Previously these were
  248 -- magic constants embedded in 'buildPrepareEffectPlan'; Phase 7
  249 -- extracts them into a single calibration record so that
  250 -- property-based lifeness gates can vary the parameters and
  251 -- verify boundary behaviour.
  252 data FieldHeuristics = FieldHeuristics
  253   { fhNarrativeWindowSize     :: !Int
  254     -- ^ Window over 'ssRecentNarrativeSuccess' used for the
  255     --   consolidation narrative-rate computation. Default: 5.
  256   , fhDefaultNarrativeRate    :: !Double
  257     -- ^ Consolidation fallback when the window is empty.
  258     --   Default: 0.2.
  259   , fhTopicStabilityBoost     :: !Double
  260     -- ^ Floor applied to consolidation when the current topic
  261     --   matches the previous turn's topic. Default: 0.5.
  262   , fhEntropyEpsilon          :: !Double
  263     -- ^ Small additive guard against division-by-zero when
  264     --   normalising family weights for entropy. Default: 1e-9.
  265   , fhHolisticStreakBoostRate :: !Double
  266     -- ^ Increment added to counterfactual per unbroken holistic
  267     --   turn in the streak. Default: 0.05.
  268   , fhHolisticStreakBoostCap  :: !Double
  269     -- ^ Maximum total streak boost. Default: 0.2.
  270   , fhLegitimacyMidpoint      :: !Double
  271     -- ^ Legitimacy score treated as neutral for atmosphere
  272     --   valence modulation. Default: 0.5.
  273   , fhLegitimacyBonusScale    :: !Double
  274     -- ^ Multiplier on (legitimacy − midpoint) that is added
  275     --   to the ego-derived valence base. Default: 0.4.
  276   }
  277   deriving stock (Eq, Show, Generic)
  278   deriving anyclass (NFData, ToJSON, FromJSON)

  280 -- | Phase-7 default heuristic parameters.  These reproduce the
  281 -- behaviour of the hardcoded constants that shipped in Phase 5.5d.
  282 defaultFieldHeuristics :: FieldHeuristics
  283 defaultFieldHeuristics = FieldHeuristics
  284   { fhNarrativeWindowSize     = 5
  285   , fhDefaultNarrativeRate    = 0.2
  286   , fhTopicStabilityBoost   = 0.5
  287   , fhEntropyEpsilon        = 1e-9
  288   , fhHolisticStreakBoostRate = 0.05
  289   , fhHolisticStreakBoostCap  = 0.2
  290   , fhLegitimacyMidpoint      = 0.5
  291   , fhLegitimacyBonusScale    = 0.4
  292   }

  294 -- | Compute 'Consolidation' from a window of recent narrative
  295 -- success flags and a topic-stability indicator.
  296 --
  297 -- Lifeness gate: result is always in @[0, 1]@ because the raw
  298 -- value is routed through 'mkConsolidation'.
  299 computeConsolidation :: FieldHeuristics -> [Bool] -> Bool -> Consolidation
  300 computeConsolidation fh recentSuccess sameTopic =
  301   let window = take (fhNarrativeWindowSize fh) recentSuccess
  302       rate   = if null window
  303                  then fhDefaultNarrativeRate fh
  304                  else fromIntegral (length (filter id window))
  305                       / fromIntegral (length window)
  306       raw    = if sameTopic then max (fhTopicStabilityBoost fh) rate else rate
  307   in mkConsolidation raw
  308 
  309 -- | Compute 'Counterfactual' from a list of semantic-logic family
  310 -- weights and the current holistic streak length.
  311 --
  312 -- Lifeness gate: result is always in @[0, 1]@ because the raw
  313 -- value is routed through 'mkCounterfactual'.
  314 computeCounterfactual :: FieldHeuristics -> [Double] -> Int -> Counterfactual
  315 computeCounterfactual fh weights streak =
  316   let totalWeight  = sum weights
  317       epsilon      = fhEntropyEpsilon fh
  318       normWeights  = map (/ max epsilon totalWeight) weights
  319       entropy      = negate $ sum [ p * log (max epsilon p) | p <- normWeights ]
  320       maxEntropy   = log (fromIntegral (max 1 (length normWeights)))
  321       ambiguity    = if maxEntropy > 0 then entropy / maxEntropy else 0.0
  322       boost        = min (fromIntegral streak * fhHolisticStreakBoostRate fh)
  323                          (fhHolisticStreakBoostCap fh)
  324       raw          = min 1.0 (ambiguity + boost)
  325   in mkCounterfactual raw
  326 
  327 -- | Compute 'Atmosphere' from ego agency, ego tension, and the
  328 -- last legitimacy score.
  329 --
  330 -- Lifeness gate: the returned 'Atmosphere' is always well-formed
  331 -- because 'mkAtmosphere' clamps both axes.
  332 computeAtmosphere :: FieldHeuristics -> Double -> Double -> Double -> Atmosphere
  333 computeAtmosphere fh egoAgency egoTension legitScore =
  334   let valenceBase = egoAgency - egoTension
  335       legitBonus  = (legitScore - fhLegitimacyMidpoint fh)
  336                     * fhLegitimacyBonusScale fh
  337       valence     = valenceBase + legitBonus
  338       arousal     = egoTension
  339   in mkAtmosphere valence arousal

  341 -- | Bounded adaptation of field-heuristic parameters.
  342 --
  343 -- Only the 'Double' fields are adapted; 'Int' fields (window size)
  344 -- are left unchanged to avoid brittle integer drift.  Each adapted
  345 -- value is clamped to @[0.0, 1.0]@ and further constrained to stay
  346 -- within @±0.5@ of its default, providing an anti-drift guardrail.
  347 --
  348 -- Signal in @[-1, 1]@; learning rate capped at @0.02@ per turn.
  349 -- Empirical signal generation is deferred to Phase 7; this function
  350 -- wires the bounded mechanics.
  351 adaptFieldHeuristics :: Double -> FieldHeuristics -> FieldHeuristics
  352 adaptFieldHeuristics rawSignal fh =
  353   let lr        = 0.02
  354       signal    = max (-1.0) (min 1.0 rawSignal)
  355       delta     = signal * lr
  356       -- Anti-drift only applies when we are actually adapting;
  357       -- signal=0 must be identity to respect the gating contract.
  358       bounded target x =
  359         let adapted = max 0.0 (min 1.0 (x + delta))
  360         in if abs delta < 1e-12
  361              then x
  362              else max (target - 0.5) (min (target + 0.5) adapted)
  363       def       = defaultFieldHeuristics
  364   in fh
  365        { fhDefaultNarrativeRate    = bounded (fhDefaultNarrativeRate    def) (fhDefaultNarrativeRate    fh)
  366        , fhTopicStabilityBoost     = bounded (fhTopicStabilityBoost     def) (fhTopicStabilityBoost     fh)
  367        , fhHolisticStreakBoostRate = bounded (fhHolisticStreakBoostRate def) (fhHolisticStreakBoostRate fh)
  368        , fhHolisticStreakBoostCap  = bounded (fhHolisticStreakBoostCap  def) (fhHolisticStreakBoostCap  fh)
  369        , fhLegitimacyMidpoint      = bounded (fhLegitimacyMidpoint      def) (fhLegitimacyMidpoint      fh)
  370        , fhLegitimacyBonusScale    = bounded (fhLegitimacyBonusScale    def) (fhLegitimacyBonusScale    fh)
  371        }

  377 -- | Combine two 'Resonance' values by 'max'.
  378 -- Commutative, associative, idempotent.
  379 combineResonance :: Resonance -> Resonance -> Resonance
  380 combineResonance (Resonance a) (Resonance b) = Resonance (max a b)
  381 
  382 -- | Combine two 'Atmosphere' values by weighted linear interpolation
  383 -- on each axis. Weight is clamped to @[0, 1]@; a weight of @1@
  384 -- returns the first argument unchanged, a weight of @0@ returns
  385 -- the second.
  386 combineAtmosphere :: Double -> Atmosphere -> Atmosphere -> Atmosphere
  387 combineAtmosphere w a1 a2 =
  388   Atmosphere
  389     { atmosphereValence = w' * atmosphereValence a1 + (1 - w') * atmosphereValence a2
  390     , atmosphereArousal = w' * atmosphereArousal a1 + (1 - w') * atmosphereArousal a2
  391     }
  392   where
  393     w' = clampUnit w
  394 
  395 -- | Combine two 'FieldConfidence' values by 'min'.
  396 -- Commutative, associative, idempotent.
  397 combineFieldConfidence :: FieldConfidence -> FieldConfidence -> FieldConfidence
  398 combineFieldConfidence (FieldConfidence a) (FieldConfidence b) =
  399   FieldConfidence (min a b)
  400 
  401 -- | Combine two 'Consolidation' values additively, clipped at
  402 -- @1.0@. Commutative, associative; @Consolidation 0@ is the
  403 -- identity.
  404 combineConsolidation :: Consolidation -> Consolidation -> Consolidation
  405 combineConsolidation (Consolidation a) (Consolidation b) =
  406   Consolidation (min 1.0 (a + b))

  408 -- | Combine two 'Counterfactual' values by 'max'.
  409 -- Commutative, associative, idempotent.
  410 combineCounterfactual :: Counterfactual -> Counterfactual -> Counterfactual
  411 combineCounterfactual (Counterfactual a) (Counterfactual b) =
  412   Counterfactual (max a b)

  418 -- | Strategy for combining two 'Field' snapshots.
  419 data CombineMode
  420   = CombineMaxima      -- ^ Per-component max where it makes sense;
  421                        --   atmosphere by 50\/50 average; confidence by 'min'.
  422   | CombineAverage     -- ^ Per-component arithmetic mean of every scalar.
  423   | CombineAccumulate  -- ^ Additive on consolidation, max on the other
  424                        --   non-confidence components, 'min' on confidence.
  425   deriving stock (Eq, Show)
  426 
  427 -- | Combine two 'Field' snapshots according to a 'CombineMode'.
  428 combineField :: CombineMode -> Field -> Field -> Field
  429 combineField mode f1 f2 = case mode of
  430   CombineMaxima ->
  431     Field
  432       { fieldResonance      = combineResonance       (fieldResonance      f1) (fieldResonance      f2)
  433       , fieldAtmosphere     = combineAtmosphere 0.5  (fieldAtmosphere     f1) (fieldAtmosphere     f2)
  434       , fieldConfidence     = combineFieldConfidence (fieldConfidence     f1) (fieldConfidence     f2)
  435       , fieldConsolidation  =
  436           mkConsolidation
  437             (max
  438                (unConsolidation (fieldConsolidation f1))
  439                (unConsolidation (fieldConsolidation f2)))
  440       , fieldCounterfactual = combineCounterfactual  (fieldCounterfactual f1) (fieldCounterfactual f2)
  441       }
  442   CombineAverage ->
  443     Field
  444       { fieldResonance      = mkResonance       (avgUnit (fieldResonance      f1) (fieldResonance      f2) unResonance)
  445       , fieldAtmosphere     = combineAtmosphere 0.5  (fieldAtmosphere     f1) (fieldAtmosphere     f2)
  446       , fieldConfidence     = mkFieldConfidence (avgUnit (fieldConfidence     f1) (fieldConfidence     f2) unFieldConfidence)
  447       , fieldConsolidation  = mkConsolidation   (avgUnit (fieldConsolidation  f1) (fieldConsolidation  f2) unConsolidation)
  448       , fieldCounterfactual = mkCounterfactual  (avgUnit (fieldCounterfactual f1) (fieldCounterfactual f2) unCounterfactual)
  449       }
  450   CombineAccumulate ->
  451     Field
  452       { fieldResonance      = combineResonance       (fieldResonance      f1) (fieldResonance      f2)
  453       , fieldAtmosphere     = combineAtmosphere 0.5  (fieldAtmosphere     f1) (fieldAtmosphere     f2)
  454       , fieldConfidence     = combineFieldConfidence (fieldConfidence     f1) (fieldConfidence     f2)
  455       , fieldConsolidation  = combineConsolidation   (fieldConsolidation  f1) (fieldConsolidation  f2)
  456       , fieldCounterfactual = combineCounterfactual  (fieldCounterfactual f1) (fieldCounterfactual f2)
  457       }
  458   where
  459     avgUnit :: a -> a -> (a -> Double) -> Double
  460     avgUnit x y un = (un x + un y) / 2.0

  473 -- | The empty history.

  477 -- | Prepend a new 'Field' onto the history, dropping older entries
  478 -- to keep the buffer at most 32 deep.
  479 recordFieldOnto :: Field -> FieldHistory -> FieldHistory
  480 recordFieldOnto f (FieldHistory xs) = FieldHistory (f : take 31 xs)

  482 -- | Summarise a 'FieldHistory' to a single 'Field'. Phase-4 stub:

  485 summariseFrom :: FieldHistory -> Field

  487 summariseFrom (FieldHistory (f : _)) = f

  493 clamp :: Ord a => a -> a -> a -> a
  494 clamp lo hi = max lo . min hi

  496 clampUnit :: Double -> Double
