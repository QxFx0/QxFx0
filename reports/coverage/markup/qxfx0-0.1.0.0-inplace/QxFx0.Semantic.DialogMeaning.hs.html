   26 buildDialogAtoms :: InputPropositionFrame -> ResponseMeaningPlan -> SystemState
   27                   -> MorphologyData -> ParsedInput -> Maybe ConsciousnessNarrative
   28                   -> DialogAtoms
   29 buildDialogAtoms frame rmp ss morph parsed mnarr =
   30   let raw = ipfRawText frame
   31       topic = nonEmptyOr (rmpTopic rmp) (nonEmptyOr (ipfFocusEntity frame) "тема")
   32       turn = ssTurnCount ss + 1
   33       cappedEpistemic = capByTruthContract (rmpTruthContractStatus rmp) (rmpEpistemic rmp)
   34       slots = [ userIntentSlot frame
   35               , topicSlot topic morph
   36               , stanceSlot (rmpStance rmp)
   37               , emotionSlot frame
   38               , knowledgeSlot cappedEpistemic
   39               , phaseSlot ss
   40               , turnGoalSlot (rmpFamily rmp)
   41                 , identitySlot
   42                 , constraintSlot (rmpTruthContractStatus rmp) frame
   43               , greetingSlot frame
   44               , farewellSlot frame
   45               , hedgeSlot cappedEpistemic
   46               , connectorSlot
   47               , softenerSlot frame
   48                , parseSubjectSlot parsed
   49                , parseObjectSlot parsed
   50                 , parseNegationSlot (ssSemanticConfig ss) parsed
   51                 , parseModifierSlot (ssSemanticConfig ss) parsed
   52                 , parseVerbSlot parsed
   53                 , parseAgreementSlot (ssSemanticConfig ss) parsed
   54                 , modalSlot
   55                 , fillerSlot
   56                  , emphaticSlot
   57                  , numberSlot
   58                  , quantifierSlot
   59                  , dreamAxiomSlot ss
   60                  , narrativeSlot mnarr
   61                 ]
   62   in DialogAtoms (mergeSlots (concat slots)) raw turn
   63 
   64 mergeSlots :: [(AtomTag, [AtomSlot])] -> Map AtomTag [AtomSlot]
   65 mergeSlots = foldr (\(tag, vals) acc ->
   66   M.insertWith (++) tag vals acc) M.empty

   68 -- | Tag user intent: one of define, clarify, ground, reflect, complain, greet, leave...
   69 userIntentSlot :: InputPropositionFrame -> [(AtomTag, [AtomSlot])]
   70 userIntentSlot frame = case ipfCanonicalFamily frame of
   71   CMGround    -> [(TUserIntent, [plainSlot TUserIntent "ground"])]
   72   CMDefine    -> [(TUserIntent, [plainSlot TUserIntent "define"])]
   73   CMClarify   -> [(TUserIntent, [plainSlot TUserIntent "clarify"])]
   74   CMReflect   -> [(TUserIntent, [plainSlot TUserIntent "reflect"])]
   75   CMContact   -> [(TUserIntent, [plainSlot TUserIntent "connect"])]
   76   CMPurpose   -> [(TUserIntent, [plainSlot TUserIntent "ask_purpose"])]
   77   CMHypothesis ->[(TUserIntent, [plainSlot TUserIntent "hypothesize"])]
   78   CMDistinguish->[(TUserIntent, [plainSlot TUserIntent "distinguish"])]
   79   CMRepair    -> [(TUserIntent, [plainSlot TUserIntent "complain"])]
   80   CMDeepen    -> [(TUserIntent, [plainSlot TUserIntent "deepen"])]
   81   CMConfront  -> [(TUserIntent, [plainSlot TUserIntent "confront"])]
   82   CMNextStep  -> [(TUserIntent, [plainSlot TUserIntent "next"])]
   83   CMAnchor    -> [(TUserIntent, [plainSlot TUserIntent "anchor"])]
   84   CMDescribe  -> [(TUserIntent, [plainSlot TUserIntent "describe"])]

   86 -- | Topic — the conversation subject
   87 topicSlot :: Text -> MorphologyData -> [(AtomTag, [AtomSlot])]
   88 topicSlot raw morph =
   89   let t = T.strip raw
   90       nom = t
   91       gen = genitiveForm morph t
   92       prep = prepositionalForm morph t
   93       acc = accusativeForm morph t
   94       ins = t
   95       pl = t
   96   in if T.null t then [] else
   97        [(TTopic, [nounSlot TTopic nom gen prep acc ins pl])]

  127 -- | Dialog phase
  128 phaseSlot :: SystemState -> [(AtomTag, [AtomSlot])]
  129 phaseSlot ss
  130   | turn <= 1 = [(TPhase, [plainSlot TPhase "opening"])]
  131   | consecReflect >= 3 = [(TPhase, [plainSlot TPhase "stuck"])]
  132   | consecReflect >= 1 = [(TPhase, [plainSlot TPhase "reflect"])]
  133   | otherwise = [(TPhase, [plainSlot TPhase "dialogue"])]
  134   where
  135     turn = ssTurnCount ss
  136     consecReflect = ssConsecutiveReflect ss

  159 -- | Constraint: what limits apply
  160 constraintSlot :: TruthContractStatus -> InputPropositionFrame -> [(AtomTag, [AtomSlot])]
  161 constraintSlot truthStatus frame =
  162   case (truthStatus, ipfSemanticTarget frame) of
  163     (CanonicalSurfacePreserved, "user") -> [(TConstraint, [plainSlot TConstraint "я знаю только то, что проявлено в этой сессии"])]
  164     (_, "user") -> [(TConstraint, [plainSlot TConstraint "я удерживаю только локально проявленный контекст этой сессии"])]
  165     (CanonicalSurfacePreserved, _) -> [(TConstraint, [plainSlot TConstraint "я работаю в рамках текущего диалога"])]
  166     _ -> [(TConstraint, [plainSlot TConstraint "я держусь локальной рамки текущего диалога и не усиливаю её сверх фактической опоры"])]

  256   | otherwise =

  262 isConfrontationalLike :: InputPropositionFrame -> Bool
  263 isConfrontationalLike frame =
  264   any (`T.isInfixOf` T.toLower (ipfRawText frame))
  265     ["почему","зачем","ты не","ты никогда","ты всегда"]

  279 isGreetingLike :: InputPropositionFrame -> Bool
  280 isGreetingLike frame =
  281   any (`T.isInfixOf` T.toLower (ipfRawText frame))

  284 isFarewellLike :: InputPropositionFrame -> Bool

  289 nonEmptyOr :: Text -> Text -> Text
  290 nonEmptyOr preferred fallback
  291   | T.null (T.strip preferred) = fallback
  292   | otherwise = preferred

  298 parseSubjectSlot :: ParsedInput -> [(AtomTag, [AtomSlot])]
  299 parseSubjectSlot parsed =
  300   case filter (\t -> ptDep t `elem` ["nsubj", "nsubj:pass"]) (piTokens parsed) of
  301     (t:_) -> [(TTopic, [plainSlot TTopic (ptLemma t)])]
  302     []    -> []
  303 
  304 parseObjectSlot :: ParsedInput -> [(AtomTag, [AtomSlot])]
  305 parseObjectSlot parsed =
  306   case filter (\t -> ptDep t `elem` ["obj", "iobj", "obl"]) (piTokens parsed) of
  307     (t:_) -> [(TProperty, [plainSlot TProperty (ptLemma t)])]
  308     []    -> []
  309 
  310 parseNegationSlot :: SemanticConfig -> ParsedInput -> [(AtomTag, [AtomSlot])]
  311 parseNegationSlot cfg parsed =
  312   case filter (\t -> ptDep t == "advmod" && ptLemma t `elem` scDialogNegationKeywords cfg) (piTokens parsed) of
  313     (_:_) -> [(TEmotion, [plainSlot TEmotion "negated"])]
  314     []    -> []
  315 
  316 parseModifierSlot :: SemanticConfig -> ParsedInput -> [(AtomTag, [AtomSlot])]
  317 parseModifierSlot cfg parsed =
  318   let known = scDialogModifierKeywords cfg
  319       mods = filter (\t ->
  320         (ptDep t `elem` ["advmod","amod"]) &&
  321         ptLemma t `elem` known) (piTokens parsed)
  322   in if null mods then [] else
  323        [(TModifier, map (\t -> plainSlot TModifier (ptLemma t)) mods)]
  324 
  325 parseVerbSlot :: ParsedInput -> [(AtomTag, [AtomSlot])]
  326 parseVerbSlot parsed =
  327   case filter (\t -> ptDep t == "ROOT") (piTokens parsed) of
  328     (t:_) -> [(TUserIntent, [plainSlot TUserIntent (ptLemma t)])]
  329     []    -> []
  330 
  331 parseAgreementSlot :: SemanticConfig -> ParsedInput -> [(AtomTag, [AtomSlot])]
  332 parseAgreementSlot cfg parsed =
  333   let words_ = map ptLemma (piTokens parsed)
  334   in if any (`elem` scDialogAgreementKeywords cfg) words_
  335      then [(TAgreement, [plainSlot TAgreement "agree"])]
  336      else if any (`elem` scDialogDisagreementKeywords cfg) words_
  337            then [(TDisagreement, [plainSlot TDisagreement "disagree"])]
  338            else []
  339 
  340 dreamAxiomSlot :: SystemState -> [(AtomTag, [AtomSlot])]
  341 dreamAxiomSlot ss =
  342   let axiom = ssDreamAxiom ss
  343   in if T.null axiom
  344      then []
  345      else [(TConnector, [plainSlot TConnector ("кстати, связь: " <> axiom)])]

  347 narrativeSlot :: Maybe ConsciousnessNarrative -> [(AtomTag, [AtomSlot])]
  348 narrativeSlot Nothing = []
  349 narrativeSlot (Just narr) =
  350   let conflictAtom = if T.null (cnConflict narr) then [] else [(TEmphatic, [plainSlot TEmphatic (cnConflict narr)])]
  351       limitAtom    = if T.null (cnLimitation narr) then [] else [(THedge, [plainSlot THedge (cnLimitation narr)])]
  352       -- TModifier / TConnector are not consumed by DialogAssembly;
  353       -- emitting them here would be dishonest (the assembler silently drops them).
  354   in conflictAtom ++ limitAtom
