   65 bootstrapSession :: Bool -> Text -> IO Session
   66 bootstrapSession quiet sessionId = do
   67   dbPath <- resolveDbPath
   68   runtimeMode <- resolveRuntimeMode
   69   createDirectoryIfMissing True (takeDirectory dbPath)
   70   readiness <- assessResourceReadiness dbPath
   71   let readinessMode = computeReadinessMode readiness
   72   case evaluateBootstrapReadiness runtimeMode readinessMode of
   73     Left failure ->
   74       throwQxFx0 (RuntimeInitError (renderBootstrapGateFailure failure))
   75     Right _ ->
   76       case readinessMode of
   77         Degraded failed ->
   78           unless quiet $ hPutStrLn stderr $ "[degraded] optional components unavailable: " ++ show failed
   79         _ ->
   80           pure ()
   81   schemaInitResult <- try (withDB dbPath $ \db -> do
   82     ensureSchemaMigrations db
   83     mStmt <- NSQL.prepare db "INSERT OR IGNORE INTO runtime_sessions(id, agency, tension, status) VALUES(?, 0.5, 0.3, 'active')"
   84     case mStmt of
   85       Left err -> throwQxFx0 (SQLiteError err)
   86       Right stmt -> do
   87         _ <- NSQL.bindText stmt 1 sessionId
   88         _ <- NSQL.step stmt
   89         NSQL.finalize stmt
   90         pure ()
   91     ) :: IO (Either QxFx0Exception (Either Text ()))
   92   case schemaInitResult of
   93     Left err -> throwQxFx0 (RuntimeInitError $ "Cannot initialize schema: " <> T.pack (show err))
   94     Right (Left err) -> throwQxFx0 (RuntimeInitError $ "Cannot initialize schema: " <> err)
   95     Right (Right _) -> pure ()
   96 
   97   morphologyResult <- tryQxFx0 loadMorphologyData
   98   morphology <- case morphologyResult of
   99     Left err -> throwQxFx0 (RuntimeInitError $ "Cannot load morphology data: " <> T.pack (show err))
  100     Right md -> pure md
  101   runtime <- initRuntimeContext dbPath
  102   health <- checkHealth runtime
  103   case evaluateStrictHealth runtimeMode health of
  104     Left failure ->
  105       throwQxFx0 (RuntimeInitError (renderBootstrapGateFailure failure))
  106     Right _ ->
  107       pure ()
  108 
  109   idClaims <- withRuntimeDb runtime $ \db ->
  110     queryIdentityClaimsByFocus db ["identity", "agency", "meaning", "consciousness", "truth"]
  111 
  112   clusters <- withRuntimeDb runtime loadClusters
  113   scenes <- withRuntimeDb runtime loadScenes
  114 
  115   let firstScene = case (scenes ++ defaultScenes) of
  116         s : _ -> s
  117         [] -> ssActiveScene emptySystemState
  118       freshState = emptySystemState
  119         { ssDialogue = (ssDialogue emptySystemState) {dsActiveScene = firstScene}
  120         , ssMorphology = morphology
  121         , ssIdentity = (ssIdentity emptySystemState) {idsIdentityClaims = idClaims}
  122         , ssSemantic = (ssSemantic emptySystemState) {semClusters = clusters}
  123         , ssSessionId = sessionId
  124         }
  125   (stateOrigin, restored) <- do
  126     mSs <- tryIO (loadState (withRuntimeDb runtime) sessionId)
  127     case mSs of
  128       Left err -> do
  129         unless quiet $ hPutStrLn stderr $ "[warn] cannot restore state, starting fresh: " ++ show err
  130         pure (FreshOrigin, freshState)
  131       Right LoadStateMissing ->
  132         pure (FreshOrigin, freshState)
  133       Right (LoadStateCorrupt diagnostics) -> do
  134         unless quiet $
  135           hPutStrLn stderr $
  136             "[warn] persisted state is corrupt, entering recovery bootstrap: " <> T.unpack (renderPersistenceDiagnostics diagnostics)
  137         pure (RecoveredCorruptOrigin, freshState)
  138       Right (LoadStateRestored ss) ->
  139         if ssTurnCount ss == 0 && null (ssHistory ss)
  140           then pure (FreshOrigin, freshState)
  141           else pure (RestoredOrigin, ss
  142             { ssDialogue = (ssDialogue ss) {dsActiveScene = firstScene}
  143             , ssMorphology = morphology
  144             , ssIdentity = (ssIdentity ss)
  145               { idsIdentityClaims = if null (ssIdentityClaims ss) then idClaims else ssIdentityClaims ss
  146               }
  147             , ssSemantic = (ssSemantic ss)
  148               { semClusters = if null (ssClusters ss) then clusters else ssClusters ss
  149               }
  150             , ssSessionId = sessionId
  151             })
  152   hydrateRuntimeTurnState runtime restored
  153   pure Session
  154     { sessSystemState = restored
  155     , sessOutputMode = DialogueMode
  156     , sessSessionId = sessionId
  157     , sessDbPath = dbPath
  158     , sessStateOrigin = stateOrigin
  159     , sessReadinessMode = readinessMode
  160     , sessRuntime = runtime
  161     }
  162 
  163 withBootstrappedSession :: Bool -> Text -> (Session -> IO a) -> IO a
  164 withBootstrappedSession quiet sessionId =
  165   bracket (bootstrapSession quiet sessionId) closeSession
  166 
  167 closeSession :: Session -> IO ()
  168 closeSession = releaseRuntimeContext . sessRuntime
  169 
  170 checkSessionReadiness :: Session -> IO ReadinessMode
  171 checkSessionReadiness session = do
  172   readiness <- assessResourceReadiness (sessDbPath session)
  173   pure (computeReadinessMode readiness)
