   72 runtimeInitSchemaMessage :: QxFx0Exception -> Text
   73 runtimeInitSchemaMessage err =
   74   case err of
   75     SQLiteError msg -> "Cannot initialize schema: " <> msg
   76     PersistenceTxError _ msg -> "Cannot initialize schema: " <> msg
   77     _ -> "Cannot initialize schema: " <> renderQxFx0ExceptionForLog err

   79 bootstrapSession :: Bool -> Text -> IO Session
   80 bootstrapSession quiet sessionId = do
   81   dbPath <- resolveDbPath
   82   runtimeMode <- resolveRuntimeMode
   83   createDirectoryIfMissing True (takeDirectory dbPath)
   84   readiness <- assessResourceReadiness dbPath
   85   let readinessMode = computeReadinessMode readiness
   86   case evaluateBootstrapReadiness runtimeMode readinessMode of
   87     Left failure ->
   88       throwQxFx0 (RuntimeInitError (renderBootstrapGateFailure failure))
   89     Right _ ->
   90       case readinessMode of
   91         Degraded failed ->
   92           unless quiet $ hPutStrLn stderr $ "[degraded] optional components unavailable: " ++ show failed
   93         _ ->
   94           pure ()
   95   schemaInitResult <- try (withDB dbPath $ \db -> do
   96     ensureSchemaMigrations db
   97     ts <- prepareTx db "bootstrap_runtime_session" "INSERT OR IGNORE INTO runtime_sessions(id, agency, tension, status) VALUES(?, 0.5, 0.3, 'active')"
   98     bindTextOrFail ts 1 sessionId
   99     stepOrFail ts
  100     pure ()
  101     ) :: IO (Either QxFx0Exception (Either Text ()))
  102   case schemaInitResult of
  103     Left err -> do
  104       hPutStrLn stderr $ "[runtime_init_debug] schema_init_qxfx0_exception db=" <> dbPath <> " detail=" <> T.unpack (renderQxFx0ExceptionForLog err)
  105       throwQxFx0 (RuntimeInitError (runtimeInitSchemaMessage err))
  106     Right (Left err) -> do
  107       hPutStrLn stderr $ "[runtime_init_debug] schema_init_sql_error db=" <> dbPath <> " detail=" <> T.unpack err
  108       throwQxFx0 (RuntimeInitError $ "Cannot initialize schema: " <> err)
  109     Right (Right _) -> pure ()
  110 
  111   morphologyResult <- tryQxFx0 loadMorphologyData
  112   morphology <- case morphologyResult of
  113     Left err -> do
  114       hPutStrLn stderr $ "[runtime_init_debug] morphology_load_failed detail=" <> T.unpack (renderQxFx0ExceptionForLog err)
  115       throwQxFx0 (RuntimeInitError $ "Cannot load morphology data: " <> renderQxFx0ExceptionForLog err)
  116     Right md -> pure md
  117   runtime <- initRuntimeContext dbPath
  118   health <- checkHealth runtime
  119   case evaluateStrictHealth runtimeMode health of
  120     Left failure ->
  121       throwQxFx0 (RuntimeInitError (renderBootstrapGateFailure failure))
  122     Right _ ->
  123       pure ()
  124 
  125   idClaims <- withRuntimeDb runtime $ \db ->
  126     queryIdentityClaimsByFocus db ["identity", "agency", "meaning", "consciousness", "truth"]
  127 
  128   clusters <- withRuntimeDb runtime loadClusters
  129   scenes <- withRuntimeDb runtime loadScenes
  130 
  131   let firstScene = case (scenes ++ defaultScenes) of
  132         s : _ -> s
  133         [] -> ssActiveScene emptySystemState
  134       freshState = emptySystemState
  135         { ssDialogue = (ssDialogue emptySystemState) {dsActiveScene = firstScene}
  136         , ssMorphology = morphology
  137         , ssIdentity = (ssIdentity emptySystemState) {idsIdentityClaims = idClaims}
  138         , ssSemantic = (ssSemantic emptySystemState) {semClusters = clusters}
  139         , ssSessionId = sessionId
  140         }
  141   (stateOrigin, restored) <- do
  142     mSs <- tryIO (loadState (withRuntimeDb runtime) sessionId)
  143     case mSs of
  144       Left err -> do
  145         unless quiet $ hPutStrLn stderr $ "[warn] cannot restore state, starting fresh: " ++ show err
  146         pure (FreshOrigin, freshState)
  147       Right LoadStateMissing ->
  148         pure (FreshOrigin, freshState)
  149       Right (LoadStateCorrupt diagnostics) -> do
  150         let rendered = renderPersistenceDiagnostics diagnostics
  151         unless quiet $
  152           hPutStrLn stderr $
  153             "[runtime_init_debug] persisted_state_corrupt session=" <> T.unpack sessionId <> " detail=" <> T.unpack rendered
  154         case runtimeMode of
  155           StrictRuntime ->
  156             throwQxFx0 (RuntimeInitError ("Persisted state is corrupt: " <> rendered))
  157           DegradedRuntime -> do
  158             unless quiet $
  159               hPutStrLn stderr $
  160                 "[warn] persisted state is corrupt, entering non-authoritative recovery bootstrap: " <> T.unpack rendered
  161             pure
  162               ( RecoveredCorruptOrigin
  163               , freshState { ssGovernanceRuntimeFault = Just (GrfRecoveredCorruptBootstrap rendered) }
  164               )
  165       Right (LoadStateRestored ss) ->
  166         let restored0 = ss
  167               { ssDialogue = (ssDialogue ss) {dsActiveScene = firstScene}
  168               , ssMorphology = mergeMorphology morphology (ssMorphology ss)
  169               , ssIdentity = (ssIdentity ss)
  170                 { idsIdentityClaims = if null (ssIdentityClaims ss) then idClaims else ssIdentityClaims ss
  171                 }
  172               , ssSemantic = (ssSemantic ss)
  173                 { semClusters = if null (ssClusters ss) then clusters else ssClusters ss
  174                 }
  175               , ssSessionId = sessionId
  176               }
  177         in case rebuildGovernedSystemState restored0 of
  178              Right restored1 ->
  179                pure
  180                  ( RestoredOrigin
  181                  , restored1
  182                  )
  183              Left err -> pure
  184                 ( RecoveredCorruptOrigin
  185                 , freshState { ssGovernanceRuntimeFault = Just (GrfRecoveredCorruptBootstrap err) }
  186                 )
  187   -- Phase 1: verify that the freshly bootstrapped state forms a
  188   -- structurally coherent self (see docs/THEORY.md §4.1 and
  189   -- docs/adr/0007-dual-mode-conatus.md). Failure here is categorical:
  190   -- the session was unable to come into being as /this system/, and
  191   -- there is nothing to recover.
  192   case checkInitialBlanket (computeSelfBlanket restored) of
  193     [] -> pure ()
  194     vs -> throwQxFx0 (IdentityRupture ("bootstrap: " <> renderBlanketViolations vs))
  195   hydrateRuntimeTurnState runtime restored
  196   pure Session
  197     { sessSystemState = restored
  198     , sessOutputMode = DialogueMode
  199     , sessSessionId = sessionId
  200     , sessDbPath = dbPath
  201     , sessStateOrigin = stateOrigin
  202     , sessReadinessMode = readinessMode
  203     , sessRuntime = runtime
  204     }

  206 -- | Merge persisted morphology with resource-loaded morphology.
  207 -- Resource morphology takes precedence; persisted morphology may fill
  208 -- only gaps and must not outrun the current provenance boundary.
  209 mergeMorphology :: MorphologyData -> MorphologyData -> MorphologyData
  210 mergeMorphology resource persisted = MorphologyData
  211   { mdPrepositional = M.union (mdPrepositional resource) (mdPrepositional persisted)
  212   , mdGenitive      = M.union (mdGenitive resource)      (mdGenitive persisted)
  213   , mdNominative    = M.union (mdNominative resource)    (mdNominative persisted)
  214   , mdFormsBySurface = M.union (mdFormsBySurface resource) (mdFormsBySurface persisted)
  215   }

  217 withBootstrappedSession :: Bool -> Text -> (Session -> IO a) -> IO a
  218 withBootstrappedSession quiet sessionId =
  219   bracket (bootstrapSession quiet sessionId) closeSession

  221 closeSession :: Session -> IO ()
  222 closeSession = releaseRuntimeContext . sessRuntime

  224 checkSessionReadiness :: Session -> IO ReadinessMode
  225 checkSessionReadiness session = do
  226   readiness <- assessResourceReadiness (sessDbPath session)
  227   pure (computeReadinessMode readiness)
