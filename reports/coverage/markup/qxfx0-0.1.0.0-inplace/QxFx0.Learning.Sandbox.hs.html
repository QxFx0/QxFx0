    7 {-|
    8 Module      : QxFx0.Learning.Sandbox
    9 Description : WP5 — Sandbox simulation gate before graft.
   10 
   11 Before a 'KnowledgeFruitPayload' is promoted to the live
   12 'KnowledgeTree', the sandbox projects its impact onto a *copy* of the
   13 relevant state slice and computes delta metrics:
   14 
   15 - Conatus trend (improvement vs. degradation)
   16 - Uncertainty / counterfactual trend
   17 - Repair-loop frequency over a simulated window
   18 
   19 Acceptance rule (fail-closed):
   20 - 'SandboxAccept' if net score >= 0  (non-regression)
   21 - 'SandboxAccept' with improvement bonus if both deltas positive
   22 - 'SandboxReject' if projected conatus drops below safety floor
   23   or net score is clearly negative.
   24 
   25 The sandbox is intentionally lightweight: it does NOT run a full
   26 N-turn replay (that would require the whole pipeline).  Instead it
   27 projects deltas onto the existing 'LearningNeedState' history curve
   28 and checks monotonicity constraints.
   29 -}

   51 -- | Outcome of the sandbox simulation.
   52 data SandboxResult
   53   = SandboxAccept !SandboxMetrics
   54     -- ^ Fruit passes non-regression / improvement test.
   55   | SandboxReject !SandboxMetrics !SandboxRejectReason
   56     -- ^ Fruit would degrade system state; reject with telemetry reason.
   57   deriving stock (Eq, Show, Generic)
   58     deriving anyclass (NFData, FromJSON, ToJSON)
   59 
   60 -- | Metrics captured during sandbox projection.
   61 data SandboxMetrics = SandboxMetrics
   62   { sbConatusTrend      :: !Double
   63     -- ^ Projected conatus trend after applying fruit delta.
   64   , sbUncertaintyTrend  :: !Double
   65     -- ^ Projected uncertainty (counterfactual) trend.
   66   , sbRepairLoopFreq    :: !Double
   67     -- ^ Estimated repair-loop frequency over simulated window.
   68   , sbNetScore          :: !Double
   69     -- ^ Weighted composite: 0.5*conatusDelta + 0.5*predictiveDelta.
   70   , sbWindowSize        :: !Int
   71     -- ^ Simulated window turns (N).
   72   }
   73   deriving stock (Eq, Show, Generic)
   74     deriving anyclass (NFData, FromJSON, ToJSON)

   76 -- | Human-readable reject reason for telemetry and audit.
   77 data SandboxRejectReason
   78   = SbrDegradingConatus

   80   | SbrHighRepairLoopRisk
   81   | SbrNegativeNetScore
   82   deriving stock (Eq, Show, Generic)
   83     deriving anyclass (NFData, FromJSON, ToJSON)

   85 -- | Configurable sandbox parameters.
   86 -- Low-RAM safe: small window, tight timeout budget, strict floor.
   87 data SandboxConfig = SandboxConfig
   88   { scWindowSize        :: !Int
   89     -- ^ History window for trend projection (default 5).
   90   , scSafetyFloor       :: !Double
   91     -- ^ Minimum acceptable projected conatus trend (default -0.3).
   92   , scMinNetScore       :: !Double
   93     -- ^ Minimum weighted composite for non-regression (default -0.05).
   94   , scMaxUncertaintyIncrease :: !Double
   95     -- ^ Maximum allowed projected uncertainty increase over the current
   96     --   baseline (default 0.20).
   97   }
   98   deriving stock (Eq, Show, Generic)
   99     deriving anyclass (NFData, FromJSON, ToJSON)

  112

  114 minNetScore :: Double
  115 minNetScore = -0.05

  121 -- | Run the sandbox gate with default configuration.
  122 runSandboxGate :: SystemState -> KnowledgeFruitPayload -> SandboxResult
  123 runSandboxGate = runSandboxGateWithConfig defaultSandboxConfig
  124 
  125 -- | Run the sandbox gate with explicit configuration.
  126 --
  127 -- Acceptance policy (fail-closed):
  128 -- 1. Strict non-regression: projected conatus >= safetyFloor AND netScore >= minNetScore.
  129 -- 2. Improvement bonus: if both conatusDelta and predictiveDelta are > 0,
  130 --    the netScore threshold is relaxed slightly.
  131 -- 3. Bounded uncertainty increase: if the projected uncertainty rises too far
  132 --    above the current baseline, reject with 'SbrRisingUncertainty'.
  133 runSandboxGateWithConfig :: SandboxConfig -> SystemState -> KnowledgeFruitPayload -> SandboxResult
  134 runSandboxGateWithConfig cfg ss payload =
  135   let needState = ssLearningNeedState ss
  136       -- Current conatus trend from the last 3 history points (or 0 if insufficient)
  137       currentConatusTrend = conatusTrendFromHistory (lnsHistory needState)
  138       -- Current uncertainty proxy: amplitude of history oscillation
  139       currentUncertainty  = historyOscillationWithWindow (scWindowSize cfg) (lnsHistory needState)
  140       -- Projected values after applying the fruit deltas
  141       projectedConatus  = currentConatusTrend + kfpConatusDelta payload
  142       projectedUncertainty = max 0 (currentUncertainty - kfpPredictiveDelta payload)
  143       -- Repair-loop frequency proxy: how often history levels exceed a soft threshold
  144       loopFreq            = repairLoopProxyWithWindow (scWindowSize cfg) (lnsHistory needState)
  145       -- Net score: weighted composite with improvement bonus
  146       rawNetScore         = 0.5 * kfpConatusDelta payload + 0.5 * kfpPredictiveDelta payload
  147       -- Improvement bonus: both deltas positive -> relax threshold by 0.05
  148       improvementBonus    = if kfpConatusDelta payload > 0 && kfpPredictiveDelta payload > 0 then 0.05 else 0.0
  149       netScore            = rawNetScore + improvementBonus
  150 
  151       metrics = SandboxMetrics
  152         { sbConatusTrend     = projectedConatus
  153         , sbUncertaintyTrend = projectedUncertainty
  154         , sbRepairLoopFreq   = loopFreq
  155         , sbNetScore         = netScore
  156         , sbWindowSize       = scWindowSize cfg
  157         }
  158 
  159       effectiveMinScore = scMinNetScore cfg
  160 
  161   in if projectedConatus < scSafetyFloor cfg
  162        then SandboxReject metrics SbrDegradingConatus
  163       else if projectedUncertainty > currentUncertainty + scMaxUncertaintyIncrease cfg
  164         then SandboxReject metrics SbrRisingUncertainty
  165       else if netScore < effectiveMinScore
  166         then SandboxReject metrics SbrNegativeNetScore
  167       else if loopFreq > 0.8
  168         then SandboxReject metrics SbrHighRepairLoopRisk
  169      else
  170        -- Non-regression or improvement: accept
  171        SandboxAccept metrics
  172 
  173 -- | Compute conatus trend as average slope over last 3 points.
  174 conatusTrendFromHistory :: [(Int, Double)] -> Double
  175 conatusTrendFromHistory hist =
  176   case reverse hist of
  177     (_, y2) : (_, y1) : (_, y0) : _ -> ((y2 - y1) + (y1 - y0)) / 2.0
  178     _ -> 0.0
  179 
  180 -- | Simple oscillation amplitude (max - min) over last windowSize points.
  181 historyOscillationWithWindow :: Int -> [(Int, Double)] -> Double
  182 historyOscillationWithWindow configuredWindow hist =
  183   let recent = take (max 1 configuredWindow) (map snd (reverse hist))
  184   in if null recent then 0.0 else maximum recent - minimum recent
  185 
  186 -- | Proxy for repair-loop frequency: fraction of recent points above 0.6.
  187 repairLoopProxyWithWindow :: Int -> [(Int, Double)] -> Double
  188 repairLoopProxyWithWindow configuredWindow hist =
  189   let recent = take (max 1 configuredWindow) (map snd (reverse hist))
  190       count  = fromIntegral (length (filter (> 0.6) recent))
  191       total  = fromIntegral (max 1 (length recent))
  192   in count / total

  194 renderSandboxRejectReason :: SandboxRejectReason -> Text
  195 renderSandboxRejectReason SbrDegradingConatus     = "degrading_conatus"
  196 renderSandboxRejectReason SbrRisingUncertainty    = "rising_uncertainty"
  197 renderSandboxRejectReason SbrHighRepairLoopRisk   = "high_repair_loop_risk"
  198 renderSandboxRejectReason SbrNegativeNetScore     = "negative_net_score"
