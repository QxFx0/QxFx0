   30 data RuntimeOutputMode = DialogueMode | SemanticIntrospectionMode
   31   deriving stock (Eq, Show)
   32 
   33 data WorkerCommand
   34   = WorkerShutdown
   35   | WorkerPing
   36   | WorkerHealth !Text
   37   | WorkerState !Text
   38   | WorkerTurn !Text !RuntimeOutputMode !Text
   39   deriving stock (Eq, Show)
   40 
   41 decodeWorkerCommand :: Text -> Either Text WorkerCommand
   42 decodeWorkerCommand line =
   43   case parseJsonStringArray line of
   44     Just values -> parseWorkerArgs values
   45     Nothing ->
   46       case readMaybe (T.unpack line) :: Maybe [String] of
   47         Just legacy -> parseWorkerArgs (map T.pack legacy)
   48         Nothing -> Left "Malformed worker command: expected JSON array command"
   49 
   50 parseWorkerArgs :: [Text] -> Either Text WorkerCommand
   51 parseWorkerArgs values =
   52   case values of
   53     ["shutdown"] -> Right WorkerShutdown
   54     ["ping"] -> Right WorkerPing
   55     ["health", sid] -> Right (WorkerHealth sid)
   56     ["state", sid] -> Right (WorkerState sid)
   57     ["turn", sid, modeTxt, inputTxt] ->
   58       case parseMode modeTxt of
   59         Right mode -> Right (WorkerTurn sid mode inputTxt)
   60         Left err -> Left err
   61     _ -> Left "Malformed worker command: unsupported shape"
   62 
   63 parseMode :: Text -> Either Text RuntimeOutputMode
   64 parseMode modeTxt =
   65   case T.toLower (T.strip modeTxt) of
   66     "semantic" -> Right SemanticIntrospectionMode
   67     "dialogue" -> Right DialogueMode
   68     other -> Left ("Unsupported worker mode: " <> other)
   69 
   70 parseJsonStringArray :: Text -> Maybe [Text]
   71 parseJsonStringArray raw =
   72   case readP_to_S parser (T.unpack raw) of
   73     [] -> Nothing
   74     xs -> case [values | (values, rest) <- xs, null rest] of
   75             [] -> Nothing
   76             (c:_) -> Just c
   77   where
   78     parser = skipSpaces *> jsonStringArrayP <* skipSpaces <* eof
   79 
   80 jsonStringArrayP :: ReadP [Text]
   81 jsonStringArrayP = do
   82   _ <- char '['
   83   skipSpaces
   84   values <- jsonStringP `sepBy` jsonCommaP
   85   skipSpaces
   86   _ <- char ']'
   87   pure values
   88 
   89 jsonCommaP :: ReadP ()
   90 jsonCommaP = skipSpaces *> char ',' *> skipSpaces
   91 
   92 jsonStringP :: ReadP Text
   93 jsonStringP = do
   94   _ <- char '"'
   95   chars <- many jsonStringCharP
   96   _ <- char '"'
   97   pure (T.pack chars)
   98 
   99 jsonStringCharP :: ReadP Char
  100 jsonStringCharP = escaped <++ plain
  101   where
  102     plain = satisfy (\c -> c /= '"' && c /= '\\' && c >= ' ')
  103     escaped = char '\\' *> parseEscape
  104     parseEscape =
  105       (char '"' >> pure '"')
  106       <++ (char '\\' >> pure '\\')
  107       <++ (char '/' >> pure '/')
  108       <++ (char 'b' >> pure '\b')
  109       <++ (char 'f' >> pure '\f')
  110       <++ (char 'n' >> pure '\n')
  111       <++ (char 'r' >> pure '\r')
  112       <++ (char 't' >> pure '\t')
  113       <++ parseUnicodeEscape
  114     parseUnicodeEscape = do
  115       _ <- char 'u'
  116       a <- hexDigitP
  117       b <- hexDigitP
  118       c <- hexDigitP
  119       d <- hexDigitP
  120       pure (chr ((((a * 16) + b) * 16 + c) * 16 + d))
  121     hexDigitP = digitToInt <$> satisfy isHexDigit
  122 
  123 extractSessionArgs :: Text -> [String] -> (Text, [String])
  124 extractSessionArgs defaultSid args = case args of
  125   ("--session-id":sid:rest) | not (null sid) -> (T.pack sid, rest)
  126   ("--session":sid:rest) | not (null sid)     -> (T.pack sid, rest)
  127   _ -> (defaultSid, args)
