   39 data RuntimeOutputMode = DialogueMode | SemanticIntrospectionMode
   40   deriving stock (Eq, Show)
   41 
   42 data WorkerCommand
   43   = WorkerHello !Text ![Text]
   44   | WorkerShutdown
   45   | WorkerPing
   46   | WorkerHealth !Text
   47   | WorkerState !Text
   48   | WorkerTurn !Text !RuntimeOutputMode !Text
   49   deriving stock (Eq, Show)
   50 
   51 data WorkerProtocolError
   52   = WorkerMalformedCommand !Text
   53   | WorkerUnknownCommand !Text
   54   | WorkerUnsupportedWorkerMode !Text
   55   deriving stock (Eq, Show)
   56 
   57 workerProtocolErrorCode :: WorkerProtocolError -> Text
   58 workerProtocolErrorCode err = case err of
   59   WorkerMalformedCommand _ -> "malformed_command"
   60   WorkerUnknownCommand _ -> "unknown_command"
   61   WorkerUnsupportedWorkerMode _ -> "unsupported_output_mode"
   62 
   63 workerProtocolErrorMessage :: WorkerProtocolError -> Text
   64 workerProtocolErrorMessage err = case err of
   65   WorkerMalformedCommand detail -> detail
   66   WorkerUnknownCommand cmd -> "Unsupported worker command: " <> cmd
   67   WorkerUnsupportedWorkerMode modeTxt -> "Unsupported worker mode: " <> modeTxt
   68 
   69 decodeWorkerMessage :: Text -> Either WorkerProtocolError WorkerCommand
   70 decodeWorkerMessage line =
   71   case eitherDecodeStrict' (TE.encodeUtf8 line) of
   72     Right (Object obj) -> parseWorkerObject obj
   73     _ -> decodeWorkerArray line
   74 
   75 decodeWorkerCommand :: Text -> Either Text WorkerCommand
   76 decodeWorkerCommand = first workerProtocolErrorMessage . decodeWorkerMessage
   77 
   78 decodeWorkerArray :: Text -> Either WorkerProtocolError WorkerCommand
   79 decodeWorkerArray line =
   80   case parseJsonStringArray line of
   81     Just values -> parseWorkerArgs values
   82     Nothing ->
   83       case readMaybe (T.unpack line) :: Maybe [String] of
   84         Just legacy -> parseWorkerArgs (map T.pack legacy)
   85         Nothing -> Left (WorkerMalformedCommand "Malformed worker command: expected JSON array command or hello object")
   86 
   87 parseWorkerObject :: AesonKeyMap.KeyMap Value -> Either WorkerProtocolError WorkerCommand
   88 parseWorkerObject obj =
   89   case lookupText "command" of
   90     Nothing -> Left (WorkerMalformedCommand "Malformed worker command: object is missing command")
   91     Just "hello" ->
   92       case lookupText "protocol_version" of
   93         Nothing -> Left (WorkerMalformedCommand "Malformed hello command: missing protocol_version")
   94         Just version -> Right (WorkerHello version (lookupTextList "capabilities"))
   95     Just other -> Left (WorkerUnknownCommand other)
   96   where
   97     lookupText key =
   98       case AesonKeyMap.lookup (AesonKey.fromText key) obj of
   99         Just (String txt) -> Just txt
  100         _ -> Nothing
  101     lookupTextList key =
  102       case AesonKeyMap.lookup (AesonKey.fromText key) obj of
  103         Just (Array values) ->
  104           [ txt | String txt <- V.toList values ]
  105         _ -> []
  106 
  107 parseWorkerArgs :: [Text] -> Either WorkerProtocolError WorkerCommand
  108 parseWorkerArgs values =
  109   case values of
  110     ["hello", version] -> Right (WorkerHello version [])
  111     ["shutdown"] -> Right WorkerShutdown
  112     ["ping"] -> Right WorkerPing
  113     ["health", sid] -> Right (WorkerHealth sid)
  114     ["state", sid] -> Right (WorkerState sid)
  115     ["turn", sid, modeTxt, inputTxt] ->
  116       case parseMode modeTxt of
  117         Right mode -> Right (WorkerTurn sid mode inputTxt)
  118         Left err -> Left err
  119     command : _ -> Left (WorkerUnknownCommand command)
  120     [] -> Left (WorkerMalformedCommand "Malformed worker command: empty command payload")
  121 
  122 parseMode :: Text -> Either WorkerProtocolError RuntimeOutputMode
  123 parseMode modeTxt =
  124   case T.toLower (T.strip modeTxt) of
  125     "semantic" -> Right SemanticIntrospectionMode
  126     "dialogue" -> Right DialogueMode
  127     other -> Left (WorkerUnsupportedWorkerMode other)
  128 
  129 parseJsonStringArray :: Text -> Maybe [Text]
  130 parseJsonStringArray raw =
  131   case readP_to_S parser (T.unpack raw) of
  132     [] -> Nothing
  133     xs -> case [values | (values, rest) <- xs, null rest] of
  134             [] -> Nothing
  135             (c:_) -> Just c
  136   where
  137     parser = skipSpaces *> jsonStringArrayP <* skipSpaces <* eof
  138 
  139 jsonStringArrayP :: ReadP [Text]
  140 jsonStringArrayP = do
  141   _ <- char '['
  142   skipSpaces
  143   values <- jsonStringP `sepBy` jsonCommaP
  144   skipSpaces
  145   _ <- char ']'
  146   pure values
  147 
  148 jsonCommaP :: ReadP ()
  149 jsonCommaP = skipSpaces *> char ',' *> skipSpaces
  150 
  151 jsonStringP :: ReadP Text
  152 jsonStringP = do
  153   _ <- char '"'
  154   chars <- many jsonStringCharP
  155   _ <- char '"'
  156   pure (T.pack chars)
  157 
  158 jsonStringCharP :: ReadP Char
  159 jsonStringCharP = escaped <++ plain
  160   where
  161     plain = satisfy (\c -> c /= '"' && c /= '\\' && c >= ' ')
  162     escaped = char '\\' *> parseEscape
  163     parseEscape =
  164       (char '"' >> pure '"')
  165       <++ (char '\\' >> pure '\\')
  166       <++ (char '/' >> pure '/')
  167       <++ (char 'b' >> pure '\b')
  168       <++ (char 'f' >> pure '\f')
  169       <++ (char 'n' >> pure '\n')
  170       <++ (char 'r' >> pure '\r')
  171       <++ (char 't' >> pure '\t')
  172       <++ parseUnicodeEscape
  173     parseUnicodeEscape = do
  174       _ <- char 'u'
  175       a <- hexDigitP
  176       b <- hexDigitP
  177       c <- hexDigitP
  178       d <- hexDigitP
  179       pure (chr ((((a * 16) + b) * 16 + c) * 16 + d))
  180     hexDigitP = digitToInt <$> satisfy isHexDigit
  181 
  182 extractSessionArgs :: Text -> [String] -> (Text, [String])
  183 extractSessionArgs defaultSid args = case args of
  184   ("--session-id":sid:rest) | not (null sid) -> (T.pack sid, rest)
  185   ("--session":sid:rest) | not (null sid)     -> (T.pack sid, rest)
  186   _ -> (defaultSid, args)
