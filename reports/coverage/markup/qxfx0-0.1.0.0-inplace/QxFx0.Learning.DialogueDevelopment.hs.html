   61 applyDialogueDevelopment
   62   :: SystemState  -- ^ pre-turn state, used for repeated-input evidence
   63   -> SystemState  -- ^ post-base-finalize state, receives persistent updates
   64   -> TurnInput
   65   -> TurnPlan
   66   -> TurnArtifacts
   67   -> SystemState
   68 applyDialogueDevelopment previousState postState ti tp ta =
   69   let sample = assessDialogueOutcome previousState postState ti tp ta
   70       style = tdRenderStyle (taDecision ta)
   71       outcomeState = updateDialogueOutcomeLearning sample (ssDialogueOutcomeLearning postState)
   72       speechPolicy = updateSpeechPolicy sample style (ssSpeechPolicyState postState)
   73       beliefStore = updateBeliefStore previousState sample ti tp (ssBeliefStore postState)
   74       updated = postState
   75         { ssDialogueOutcomeLearning = outcomeState
   76         , ssSpeechPolicyState = speechPolicy
   77         , ssBeliefStore = beliefStore
   78         }
   79   in appendAdaptiveMutationRecords (adrMutationRecords (dosDecisionRecord sample)) updated

   81 assessDialogueOutcome
   82   :: SystemState
   83   -> SystemState
   84   -> TurnInput
   85   -> TurnPlan
   86   -> TurnArtifacts
   87   -> DialogueOutcomeSample
   88 assessDialogueOutcome previousState postState ti tp ta =
   89   let frame = tiFrame ti
   90       raw = normalizeDialogueText (ipfRawText frame)
   91       propositionType = ipfPropositionType frame
   92       truthStatus = taTruthContractStatus ta
   93       previousInputs = map normalizeDialogueText (F.toList (ssRawInputHistory previousState))
   94       repeatedInput = not (T.null raw) && raw `elem` take 3 (reverse previousInputs)
   95       localRecovery = taSurfaceProv ta == FromRecovery || isJust (taLocalRecoveryCause ta) || not (truthContractAllowsStrongMutation truthStatus)
   96       postRenderRetightened = taRendered ta /= taPreSafetyRendered ta || taFinalRendered ta /= taRendered ta
   97       decisionRepair = tdFamily (taDecision ta) == CMRepair
   98       renderEmpty = T.null (T.strip (taFinalRendered ta))
   99       userRepair = propositionType `elem` ["RepairSignal", "MisunderstandingReport"]
  100       userClarification = propositionType `elem` ["ClarifyQ", "EpistemicQ", "RequestQ"]
  101       successfulContinuation = propositionType == "NextStepQ"
  102       userConflict = propositionType == "ConfrontQ" || matchesAnyCue raw conflictPhrases
  103       strongPositiveConfirmation = matchesAnyCue raw strongPositiveConfirmationPhrases
  104       weakAcknowledgement = isWeakAcknowledgementText raw
  105       (kind0, strong0, strength0) =
  106         if userConflict
  107           then (DialogueOutcomeConflict, True, EvidenceStrong)
  108         else if userRepair
  109           then (DialogueOutcomeRepairRequested, True, EvidenceStrong)
  110         else if repeatedInput
  111           then (DialogueOutcomeRepeatedQuestion, True, EvidenceStrong)
  112         else if strongPositiveConfirmation
  113           then (DialogueOutcomeSuccess, True, EvidenceStrong)
  114         else if weakAcknowledgement
  115           then (DialogueOutcomePartialSuccess, False, EvidenceWeak)
  116         else if localRecovery || decisionRepair || renderEmpty
  117           then (DialogueOutcomeDegraded, True, EvidenceStrong)
  118         else if successfulContinuation || userClarification
  119           then (DialogueOutcomePartialSuccess, False, EvidenceModerate)
  120         else (DialogueOutcomeUncertain, False, EvidenceWeak)
  121       (kind, strong, strength) =
  122         if truthContractAllowsStrongMutation truthStatus
  123           then (kind0, strong0, strength0)
  124           else (DialogueOutcomeDegraded, False, max strength0 EvidenceModerate)
  125       topic = firstNonEmpty
  126         [ if strongPositiveConfirmation || weakAcknowledgement then ssLastTopic previousState else ""
  127         , tiBestTopic ti
  128         , ipfSemanticSubject frame
  129         , rmpTopic (tpRmpAfterLegit tp)
  130         ]
  131       signals = outcomeSignals propositionType repeatedInput localRecovery decisionRepair renderEmpty userConflict strongPositiveConfirmation weakAcknowledgement successfulContinuation userClarification
  132         <> flag "post_render_truth_retightened" postRenderRetightened
  133       decision = adaptiveDecisionRecord (ssTurnCount postState) kind signals strong strength
  134   in DialogueOutcomeSample
  135        { dosTurn = ssTurnCount postState
  136        , dosKind = kind
  137        , dosTopic = topic
  138        , dosSignals = signals
  139        , dosEvidenceStrength = strength
  140        , dosStrongUpdate = strong
  141        , dosDecisionRecord = decision
  142        }

  144 updateDialogueOutcomeLearning :: DialogueOutcomeSample -> DialogueOutcomeLearningState -> DialogueOutcomeLearningState
  145 updateDialogueOutcomeLearning sample state0 =
  146   let state = state0 { dolRecentOutcomes = take maxRecentOutcomeSamples (sample : dolRecentOutcomes state0) }
  147   in case dosKind sample of
  148        DialogueOutcomeSuccess -> state { dolSuccessCount = dolSuccessCount state + 1 }
  149        DialogueOutcomePartialSuccess -> state { dolPartialSuccessCount = dolPartialSuccessCount state + 1 }
  150        DialogueOutcomeRepairRequested -> state { dolRepairRequestCount = dolRepairRequestCount state + 1 }
  151        DialogueOutcomeRepeatedQuestion -> state { dolRepeatedQuestionCount = dolRepeatedQuestionCount state + 1 }
  152        DialogueOutcomeConflict -> state { dolConflictCount = dolConflictCount state + 1 }
  153        DialogueOutcomeDegraded -> state { dolDegradedCount = dolDegradedCount state + 1 }
  154        DialogueOutcomeUncertain -> state { dolUncertainCount = dolUncertainCount state + 1 }

  156 adaptiveDecisionRecord
  157   :: Int
  158   -> DialogueOutcomeKind
  159   -> [Text]
  160   -> Bool
  161   -> EvidenceStrength
  162   -> AdaptiveDecisionRecord
  163 adaptiveDecisionRecord turn kind signals strong strength =
  164   let cause = "dialogue_outcome:" <> renderDialogueOutcomeKind kind
  165       confidence = evidenceConfidence strength
  166       boundedDelta = boundedDeltaFor kind strong
  167       decision = if strong then AdaptiveAccepted else AdaptiveObserved
  168       targets = mutationTargetsFor kind strong
  169   in AdaptiveDecisionRecord
  170        { adrTurn = turn
  171        , adrCause = cause
  172        , adrEvidence = signals
  173        , adrConfidence = confidence
  174        , adrBoundedDelta = boundedDelta
  175        , adrDecision = decision
  176        , adrTargets = targets
  177        , adrMutationRecords = mutationRecordsForDialogue turn cause signals strength confidence decision targets
  178        }

  180 evidenceConfidence :: EvidenceStrength -> Double
  181 evidenceConfidence strength =
  182   case strength of
  183     EvidenceObservational -> 0.10
  184     EvidenceWeak -> 0.25
  185     EvidenceModerate -> 0.50
  186     EvidenceStrong -> 0.80
  187 
  188 boundedDeltaFor :: DialogueOutcomeKind -> Bool -> [Text]
  189 boundedDeltaFor kind strong =
  190   [ "recent_outcomes<=12" ]
  191     <> if not strong
  192          then []
  193          else case kind of
  194            DialogueOutcomeSuccess -> ["speech_patterns<=8", "claim_stance_entries<=64"]
  195            DialogueOutcomeConflict -> ["speech_patterns<=8", "claim_stance_entries<=64", "claim_revisions<=20"]
  196            DialogueOutcomeRepairRequested -> ["speech_patterns<=8"]
  197            DialogueOutcomeRepeatedQuestion -> ["speech_patterns<=8"]
  198            DialogueOutcomeDegraded -> ["speech_patterns<=8"]
  199            DialogueOutcomePartialSuccess -> []
  200            DialogueOutcomeUncertain -> []
  201 
  202 mutationTargetsFor :: DialogueOutcomeKind -> Bool -> [AdaptiveMutationKind]
  203 mutationTargetsFor kind strong =
  204   MutDialogueOutcome :
  205     if not strong
  206       then []
  207       else case kind of
  208         DialogueOutcomeSuccess -> [MutSpeechPolicy, MutClaimStance]
  209         DialogueOutcomeConflict -> [MutSpeechPolicy, MutClaimStance]
  210         DialogueOutcomeRepairRequested -> [MutSpeechPolicy]
  211         DialogueOutcomeRepeatedQuestion -> [MutSpeechPolicy]
  212         DialogueOutcomeDegraded -> [MutSpeechPolicy]
  213         DialogueOutcomePartialSuccess -> []
  214         DialogueOutcomeUncertain -> []

  216 mutationRecordsForDialogue
  217   :: Int
  218   -> Text
  219   -> [Text]
  220   -> EvidenceStrength
  221   -> Double
  222   -> AdaptiveDecision
  223   -> [AdaptiveMutationKind]
  224   -> [AdaptiveMutationRecord]
  225 mutationRecordsForDialogue turn cause signals strength confidence decision targets =
  226   map targetRecord targets
  227   where
  228     targetRecord target = AdaptiveMutationRecord
  229       { amrTurnId = turn
  230       , amrKind = target
  231       , amrCause = cause
  232       , amrEvidence = signals
  233       , amrEvidenceStrength = strength
  234       , amrConfidence = confidence
  235       , amrBoundedDelta = boundedDeltaForTarget target
  236       , amrDecision = decision
  237       }

  239 boundedDeltaForTarget :: AdaptiveMutationKind -> Maybe Double
  240 boundedDeltaForTarget target =
  241   case target of
  242     MutDialogueOutcome -> Just (fromIntegral maxRecentOutcomeSamples)
  243     MutSpeechPolicy -> Just 0.12
  244     MutClaimStance -> Just (fromIntegral maxClaimStanceEntries)
  245     _ -> Nothing

  247 decisionAllowsBoundedMutation :: DialogueOutcomeSample -> Bool
  248 decisionAllowsBoundedMutation sample =
  249   case adrDecision (dosDecisionRecord sample) of
  250     AdaptiveAccepted -> dosStrongUpdate sample
  251     _ -> False

  253 updateSpeechPolicy :: DialogueOutcomeSample -> RenderStyle -> SpeechPolicyState -> SpeechPolicyState
  254 updateSpeechPolicy sample style state
  255   | not (decisionAllowsBoundedMutation sample) = state
  256   | otherwise =
  257       case dosKind sample of
  258         DialogueOutcomeSuccess ->
  259           let next = state
  260                 { spsSuccessfulPatterns = bump styleTag (spsSuccessfulPatterns state)
  261                 , spsRepairBias = clamp01 (spsRepairBias state - 0.05)
  262                 , spsLastUpdatedTurn = dosTurn sample
  263                 }
  264           in boundSpeechPolicy next
  265         DialogueOutcomePartialSuccess ->
  266           tune 0.04 0.04 (-0.02) 0.08 True
  267         DialogueOutcomeRepairRequested ->
  268           tune 0.12 0.12 (-0.10) 0.30 False
  269         DialogueOutcomeRepeatedQuestion ->
  270           tune 0.12 0.15 (-0.10) 0.30 False
  271         DialogueOutcomeConflict ->
  272           tune 0.05 0.03 (-0.07) 0.18 False
  273         DialogueOutcomeDegraded ->
  274           tune 0.10 0.12 (-0.12) 0.30 False
  275         DialogueOutcomeUncertain -> state
  276   where
  277     styleTag = renderStyleText style
  278     bump key = capCounterMap maxSpeechPatternEntries . M.insertWith (+) key 1
  279     tune directDelta compressionDelta ambiguityDelta repairDelta positive =
  280       let next = state
  281             { spsDirectness = clamp01 (spsDirectness state + directDelta)
  282             , spsCompression = clamp01 (spsCompression state + compressionDelta)
  283             , spsAmbiguityTolerance = clamp01 (spsAmbiguityTolerance state + ambiguityDelta)
  284             , spsRepairBias = clamp01 (spsRepairBias state + repairDelta)
  285             , spsSuccessfulPatterns = if positive then bump styleTag (spsSuccessfulPatterns state) else spsSuccessfulPatterns state
  286             , spsFailedPatterns = if positive then spsFailedPatterns state else bump styleTag (spsFailedPatterns state)
  287             , spsLastUpdatedTurn = dosTurn sample
  288             }
  289       in boundSpeechPolicy next

  291 updateBeliefStore :: SystemState -> DialogueOutcomeSample -> TurnInput -> TurnPlan -> BeliefStore -> BeliefStore
  292 updateBeliefStore previousState sample ti tp store
  293   | not (decisionAllowsBoundedMutation sample) = store
  294   | T.null claimKey = store
  295   | otherwise =
  296       case dosKind sample of
  297         DialogueOutcomeSuccess -> upsertPositive
  298         DialogueOutcomeConflict -> upsertConflict
  299         _ -> store
  300   where
  301     rmp = tpRmpAfterLegit tp
  302     priorTopic = ssLastTopic previousState
  303     claimText =
  304       if not (T.null (T.strip priorTopic))
  305         then priorTopic
  306         else firstNonEmpty [rmpPrimaryClaim rmp, tiBestTopic ti]
  307     claimKey = normalizeDialogueText claimText
  308     evidenceTag = T.concat ["turn=", T.pack (show (dosTurn sample)), ":", renderDialogueOutcomeKind (dosKind sample)]
  309     stancePolarity = stanceToPolarity (rmpStance rmp)
  310     existing = M.lookup claimKey (bsClaims store)
  311     basePositive = fromMaybe
  312       BeliefRecord
  313         { brClaim = claimText
  314         , brPolarity = stancePolarity
  315         , brConfidence = 0.50
  316         , brEvidence = []
  317         , brCounterEvidence = []
  318         , brLastUpdatedTurn = 0
  319         , brRevisionCount = 0
  320         }
  321       existing
  322     upsertPositive =
  323       let polarity = case brPolarity basePositive of
  324             BeliefContested -> BeliefTentative
  325             BeliefRejected -> BeliefTentative
  326             _ -> stancePolarity
  327           entry = basePositive
  328             { brPolarity = polarity
  329             , brConfidence = clamp01 (brConfidence basePositive + 0.08)
  330             , brEvidence = take maxClaimEvidence (evidenceTag : brEvidence basePositive)
  331             , brLastUpdatedTurn = dosTurn sample
  332             }
  333       in store { bsClaims = insertBoundedClaim claimKey entry (bsClaims store) }
  334     upsertConflict =
  335       let base = fromMaybe
  336             BeliefRecord
  337               { brClaim = claimText
  338               , brPolarity = BeliefContested
  339               , brConfidence = 0.45
  340               , brEvidence = []
  341               , brCounterEvidence = []
  342               , brLastUpdatedTurn = 0
  343               , brRevisionCount = 0
  344               }
  345             existing
  346           entry = base
  347             { brPolarity = BeliefContested
  348             , brConfidence = clamp01 (brConfidence base - 0.15)
  349             , brCounterEvidence = take maxClaimEvidence (evidenceTag : brCounterEvidence base)
  350             , brLastUpdatedTurn = dosTurn sample
  351             , brRevisionCount = brRevisionCount base + 1
  352             }
  353       in store
  354            { bsClaims = insertBoundedClaim claimKey entry (bsClaims store)
  355            , bsRecentRevisions = take maxClaimRevisionsPerStore (claimKey : bsRecentRevisions store)
  356            }

  358 adjustRenderStyleForSpeechPolicy :: SpeechPolicyState -> RenderStyle -> RenderStyle
  359 adjustRenderStyleForSpeechPolicy policy original
  360   | original == StyleRecovery = StyleRecovery
  361   | spsRepairBias policy >= 0.60 = StyleRecovery
  362   | spsDirectness policy >= 0.65 && spsCompression policy >= 0.65 = StyleDirect
  363   | spsAmbiguityTolerance policy <= 0.30 = StyleClinical
  364   | otherwise = original

  366 outcomeSignals :: Text -> Bool -> Bool -> Bool -> Bool -> Bool -> Bool -> Bool -> Bool -> Bool -> [Text]
  367 outcomeSignals propositionType repeatedInput localRecovery decisionRepair renderEmpty userConflict strongPositiveConfirmation weakAcknowledgement successfulContinuation userClarification =
  368   [ "proposition_type=" <> propositionType ]
  369     <> flag "repeated_question" repeatedInput
  370     <> flag "local_recovery" localRecovery
  371     <> flag "decision_repair" decisionRepair
  372     <> flag "empty_render" renderEmpty
  373     <> flag "user_conflict" userConflict
  374     <> flag "strong_positive_confirmation" strongPositiveConfirmation
  375     <> flag "weak_acknowledgement" weakAcknowledgement
  376     <> flag "successful_continuation" successfulContinuation
  377     <> flag "clarification_request" userClarification

  379 flag :: Text -> Bool -> [Text]

  382 matchesAnyCue :: Text -> [Text] -> Bool
  383 matchesAnyCue text = any (matchesCue text)
  384 
  385 matchesCue :: Text -> Text -> Bool
  386 matchesCue text cue
  387   | T.any isSpace cue = cue `T.isInfixOf` text
  388   | T.length cue < 5 = cue `elem` tokenizeDialogueText text
  389   | otherwise = cue `T.isInfixOf` text

  430 stanceToPolarity :: StanceMarker -> BeliefPolarity

  435     Curated -> BeliefAffirmed
  436     HoldBack -> BeliefTentative

  439     Explore -> BeliefTentative

  442 boundSpeechPolicy :: SpeechPolicyState -> SpeechPolicyState
  443 boundSpeechPolicy state =
  444   state
  445     { spsSuccessfulPatterns = capCounterMap maxSpeechPatternEntries (spsSuccessfulPatterns state)
  446     , spsFailedPatterns = capCounterMap maxSpeechPatternEntries (spsFailedPatterns state)
  447     }

  449 capCounterMap :: Int -> M.Map Text Int -> M.Map Text Int
  450 capCounterMap limit = M.fromList . take limit . sortBy compareEntry . M.toList
  451   where
  452     compareEntry (k1, v1) (k2, v2) = compare v2 v1 <> compare k1 k2
  453 
  454 insertBoundedClaim :: Text -> BeliefRecord -> M.Map Text BeliefRecord -> M.Map Text BeliefRecord
  455 insertBoundedClaim key entry claims =
  456   let next = M.insert key entry claims
  457   in if M.size next <= maxClaimStanceEntries
  458        then next
  459        else pruneClaimMap key next
  460 
  461 pruneClaimMap :: Text -> M.Map Text BeliefRecord -> M.Map Text BeliefRecord
  462 pruneClaimMap preservedKey claims =
  463   let sorted = sortBy compareRecord (M.toList claims)
  464       keepCount = max 0 (maxClaimStanceEntries - 1)
  465       kept = take keepCount (filter ((/= preservedKey) . fst) sorted)
  466   in M.fromList (kept <> maybeToList (fmap (\value -> (preservedKey, value)) (M.lookup preservedKey claims)))
  467 
  468 compareRecord :: (Text, BeliefRecord) -> (Text, BeliefRecord) -> Ordering
  469 compareRecord (k1, r1) (k2, r2) =
  470   compare (brLastUpdatedTurn r2) (brLastUpdatedTurn r1)
  471     <> compare (brRevisionCount r2) (brRevisionCount r1)
  472     <> compare k1 k2

  478 renderDialogueOutcomeKind :: DialogueOutcomeKind -> Text
  479 renderDialogueOutcomeKind kind =

  481     DialogueOutcomeSuccess -> "success"
  482     DialogueOutcomePartialSuccess -> "partial_success"
  483     DialogueOutcomeRepairRequested -> "repair_requested"
  484     DialogueOutcomeRepeatedQuestion -> "repeated_question"
  485     DialogueOutcomeConflict -> "conflict"
  486     DialogueOutcomeDegraded -> "degraded"
  487     DialogueOutcomeUncertain -> "uncertain"

  489 firstNonEmpty :: [Text] -> Text
  490 firstNonEmpty = fromMaybe "" . safeHead . filter (not . T.null . T.strip)
  491   where
  492     safeHead [] = Nothing
  493     safeHead (x:_) = Just (T.strip x)

  495 normalizeDialogueText :: Text -> Text
  496 normalizeDialogueText = T.unwords . T.words . T.toLower . T.replace "ё" "е" . T.strip
  497 
  498 tokenizeDialogueText :: Text -> [Text]
  499 tokenizeDialogueText = filter (not . T.null) . T.words . T.map replaceNonToken
  500   where
  501     replaceNonToken c
  502       | isAlphaNum c || c == '-' = c
  503       | otherwise = ' '

  505 clamp01 :: Double -> Double
  506 clamp01 = max 0.0 . min 1.0
