    8 Module      : QxFx0.Learning.Signal
    9 Description : WP4 — Bounded Phase-7 calibration signal.
   10 
   11 Replaces the hardcoded @signal = 0.0@ in
   12 'buildNextSystemState' with an explainable, bounded composite
   13 computed from:
   14 
   15 1. Conatus trend (from 'LearningNeedState' history slope),
   16 2. Field uncertainty trend (counterfactual trajectory),
   17 3. Repair-loop risk (frequency of recovery over a window),
   18 4. Branch health trend (from 'KnowledgeTree').
   19 
   20 The signal is clamped to [-1, 1].  Positive = empirical evidence
   21 suggests the current weights/heuristics should shift in the direction
   22 of the observed deficit.  Negative = evidence is contradictory or
   23 improving; adaptation should be conservative.

   47 -- | Bounded calibration signal in [-1, 1].

   52 -- | Raw components that feed into the composite signal.
   53 -- Each component is independently normalised to [-1, 1] before
   54 -- weighting so the formula remains explainable.
   55 data SignalComponents = SignalComponents
   56   { scConatusTrend      :: !Double
   57     -- ^ Slope of need level over the recent 3 history points.
   58     --   Positive = deficit worsening.
   59   , scUncertaintyTrend  :: !Double
   60     -- ^ Normalised counterfactual magnitude.  Positive = higher
   61     --   ambiguity / more alternative parses.
   62   , scLoopRisk          :: !Double
   63     -- ^ Normalised repair-loop frequency.  Positive = frequent
   64     --   recovery-driven turns.
   65   , scBranchHealthTrend :: !Double
   66     -- ^ Average branch health from the knowledge tree.
   67     --   Negative = tree is decaying.
   68   }
   69   deriving stock (Eq, Show, Generic)
   70     deriving anyclass (NFData, FromJSON, ToJSON)

   80 -- | Persisted snapshot of a calibration signal computation.
   81 -- Captures the full feature vector and decision so the pipeline is
   82 -- reproducible and auditable.
   83 data CalibrationSnapshot = CalibrationSnapshot
   84   { csTimestamp   :: !UTCTime
   85     -- ^ When the snapshot was taken (turn resolution time).
   86   , csRunId       :: !Text

   88   , csComponents  :: !SignalComponents
   89     -- ^ The four raw feature values that fed the signal.
   90   , csSignal      :: !Double

   92   , csDecision    :: !CalibrationDecision

   94   }
   95   deriving stock (Eq, Show, Generic)
   96     deriving anyclass (NFData, FromJSON, ToJSON)

   98 -- | Decision made by the calibration pipeline for a given snapshot.
   99 data CalibrationDecision
  100   = CdApplySignal
  101     -- ^ Signal confidence and guardrails passed; apply to weights/heuristics.
  102   | CdHoldLowConfidence
  103     -- ^ Signal magnitude below threshold or confidence insufficient.
  104   | CdHoldGuardrails
  105     -- ^ Guardrails blocked application (e.g. rate limit, circuit breaker open).
  106   | CdHoldNoNeed
  107     -- ^ No active learning need; signal computed but not actionable.
  108   deriving stock (Eq, Show, Generic)
  109     deriving anyclass (NFData, FromJSON, ToJSON)

  111 -- | Configuration for the calibration signal pipeline.
  112 -- Keeps the pipeline gated and conservative by default.
  113 data SignalPipelineConfig = SignalPipelineConfig

  116   , spcApplyRateLimit     :: !Int
  117     -- ^ Max applications per N turns (default 1 per 5 turns).
  118   , spcApplyWindow        :: !Int
  119     -- ^ Window size for rate limit (default 5 turns).
  120   , spcConatusWeight      :: !Double
  121     -- ^ Weight for conatus trend component (default 0.30).
  122   , spcUncertaintyWeight  :: !Double
  123     -- ^ Weight for uncertainty trend component (default 0.30).
  124   , spcLoopRiskWeight     :: !Double
  125     -- ^ Weight for loop risk component (default 0.20).
  126   , spcBranchHealthWeight :: !Double
  127     -- ^ Weight for branch health component (default 0.20).
  128   }
  129   deriving stock (Eq, Show, Generic)
  130     deriving anyclass (NFData, FromJSON, ToJSON)

  143 -- | Gated application of a calibration signal.
  144 -- Returns 'True' if the signal should be applied to weights/heuristics,
  145 -- 'False' if held back by confidence threshold or rate limit.
  146 applyCalibrationGated
  147   :: SignalPipelineConfig
  148   -> CalibrationSignal
  149   -> [CalibrationSnapshot]  -- ^ prior snapshots (for rate-limit check)
  150   -> (Bool, CalibrationDecision)
  151 applyCalibrationGated cfg signal snapshots =
  152   let s = unCalibrationSignal signal
  153       nonFinite = isNaN s || isInfinite s
  154       absS = abs s
  155       -- Rate limit: count recent CdApplySignal within window
  156       recentApplies = length (filter (\cs -> csDecision cs == CdApplySignal)
  157                               (take (spcApplyWindow cfg) snapshots))
  158       rateLimitOk = recentApplies < spcApplyRateLimit cfg
  159   in if nonFinite
  160        then (False, CdHoldLowConfidence)
  161      else if absS < spcMinConfidence cfg
  162         then (False, CdHoldLowConfidence)
  163       else if not rateLimitOk
  164         then (False, CdHoldGuardrails)
  165       else (True, CdApplySignal)
  166 
  167 -- | Compute the bounded calibration signal.
  168 --
  169 -- Formula (fixed weights for interpretability):
  170 --
  171 -- @
  172 -- signal = clamp [-1,1] (
  173 --     0.30 * conatusTrend
  174 --   + 0.30 * uncertaintyTrend
  175 --   + 0.20 * loopRisk
  176 --   + 0.20 * (-branchHealthTrend)   -- decaying tree pushes signal up
  177 -- )
  178 -- @
  179 --
  180 -- Rationale:
  181 -- * Conatus and uncertainty each get 30 % because they are the most
  182 --   direct indicators of structural drift.
  183 -- * Loop risk gets 20 % as a lagging but robust proxy.
  184 -- * Branch health gets 20 % with inverted sign: a decaying
  185 --   knowledge tree means the current calibration is failing to
  186 --   retain useful structure, so adaptation pressure rises.
  187 computeCalibrationSignal
  188   :: LearningNeedState
  189   -> Double       -- ^ current field counterfactual (raw [0,1])
  190   -> Int          -- ^ repair-loop count in recent window
  191   -> Int          -- ^ window size for loop normalisation
  192   -> KnowledgeTree
  193   -> (CalibrationSignal, SignalComponents)
  194 computeCalibrationSignal needState counterfactual loopCount windowSize tree =
  195   let       -- 1. Conatus trend: slope over recent 3 points
  196       conatusTrend = case lnsHistory needState of
  197         (_, y2) : (_, y1) : (_, y0) : _ ->
  198           let avgSlope = ((y2 - y1) + (y1 - y0)) / 2.0
  199           in clampUnit (avgSlope * 5.0)  -- scale: 0.2 level diff -> 1.0
  200         _ -> 0.0
  201 
  202       -- 2. Uncertainty trend: raw counterfactual scaled to [-1,1]
  203       -- centred at 0.5; >0.5 positive, <0.5 negative
  204       uncertaintyTrend = clampUnit ((counterfactual - 0.5) * 4.0)
  205 
  206       -- 3. Loop risk: normalised by window size
  207       loopRisk =
  208         if windowSize <= 0
  209            then 0.0
  210            else clampUnit (fromIntegral loopCount / fromIntegral windowSize * 2.0 - 1.0)
  211 
  212       -- 4. Branch health trend: inverted so decay -> positive signal
  213       avgHealth = branchHealthTrend tree
  214       branchHealthTrend' =
  215         if isNaN avgHealth || isInfinite avgHealth
  216           then 0.0
  217           else clampUnit (-avgHealth * 2.0)
  218 
  219       comps = SignalComponents
  220         { scConatusTrend = conatusTrend
  221         , scUncertaintyTrend = uncertaintyTrend
  222         , scLoopRisk = loopRisk
  223         , scBranchHealthTrend = branchHealthTrend'
  224         }
  225 
  226       rawSignal =
  227           0.30 * conatusTrend
  228         + 0.30 * uncertaintyTrend
  229         + 0.20 * loopRisk
  230         + 0.20 * branchHealthTrend'
  231 
  232       signal = CalibrationSignal (clampRange (-1.0) 1.0 rawSignal)
  233 
  234   in (signal, comps)
  235 
  236 clampUnit :: Double -> Double
  237 clampUnit x
  238   | isNaN x || isInfinite x = 0.0
  239   | otherwise = max (-1.0) (min 1.0 x)
  240 
  241 clampRange :: Double -> Double -> Double -> Double
  242 clampRange lo hi x
  243   | isNaN x || isInfinite x = 0.0
  244   | otherwise = max lo (min hi x)
