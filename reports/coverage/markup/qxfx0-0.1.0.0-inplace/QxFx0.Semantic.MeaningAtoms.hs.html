   24 collectAtoms :: Text -> [ClusterDef] -> AtomSet
   25 collectAtoms input clusters =
   26   let inputLower = T.toLower input
   27       inputTokens = tokenizeKeywordText input
   28       suppressedExhaustion = shouldSuppressExhaustion inputTokens
   29       foundAtoms0 = concatMap (matchCluster inputLower inputTokens) clusters
   30       foundAtoms = if suppressedExhaustion then filter (not . isExhaustionAtom) foundAtoms0 else foundAtoms0
   31       lexical = lexicalAtoms inputLower inputTokens
   32       structural = if containsAnyKeywordPhrase inputTokens ["\1095\1090\1086", "\1082\1072\1082", "\1087\1086\1095\1077\1084\1091"] || T.isSuffixOf "?" (T.strip input)
   33                    then [MeaningAtom (extractObject input) (Searching (extractObject input)) (fallbackEmbedding inputLower)]
   34                    else []
   35       allFound = foundAtoms ++ lexical ++ structural
   36       load = L.foldl' (\acc a -> acc + atomIntensity a) 0.0 allFound
   37   in AtomSet
   38     { asAtoms    = allFound
   39     , asLoad     = min 1.0 load
   40     , asRegister = inferRegister allFound
   41     }

   43 atomIntensity :: MeaningAtom -> Double

   46 isExhaustionAtom :: MeaningAtom -> Bool
   47 isExhaustionAtom atom =
   48   case maTag atom of
   49     Exhaustion _ -> True
   50     _ -> False

   52 matchCluster :: Text -> [Text] -> ClusterDef -> [MeaningAtom]
   53 matchCluster inp inpTokens cd =
   54   let keywords = map T.toLower (cdKeywords cd)
   55       hits = filter (containsKeywordPhrase inpTokens) keywords
   56   in if not (null hits)
   57      then [MeaningAtom (T.intercalate ", " hits) (tagFromCluster (T.toLower $ cdName cd) (T.intercalate ", " hits)) (fallbackEmbedding inp)]
   58      else []

   60 lexicalAtoms :: Text -> [Text] -> [MeaningAtom]
   61 lexicalAtoms inputLower inputTokens =
   62   concat
   63     [ detectUnless suppressedExhaustion (Exhaustion "\1083\1077\1082\1089\1080\1082\1072") exhaustionLexemes
   64     , detect (NeedContact "\1083\1077\1082\1089\1080\1082\1072") contactLexemes
   65     , detect (NeedMeaning "\1083\1077\1082\1089\1080\1082\1072") meaningLexemes
   66     , detect (AgencyLost semanticLexicalAgencyLostStrength) agencyLostLexemes
   67     ]
   68   where
   69     suppressedExhaustion = shouldSuppressExhaustion inputTokens
   70 
   71     detect :: AtomTag -> [Text] -> [MeaningAtom]
   72     detect tag lexemes =
   73       if containsAnyKeywordPhrase inputTokens lexemes
   74         then [MeaningAtom "\1083\1077\1082\1089\1080\1082\1072" tag (fallbackEmbedding inputLower)]
   75         else []
   76 
   77     detectUnless :: Bool -> AtomTag -> [Text] -> [MeaningAtom]
   78     detectUnless suppressed tag lexemes =
   79       if suppressed then [] else detect tag lexemes

  121 tagFromCluster :: Text -> Text -> AtomTag
  122 tagFromCluster t val
  123   | t == "exhaustion"   = Exhaustion val
  124   | t == "need_contact" = NeedContact val
  125   | t == "need_meaning" = NeedMeaning val
  126   | t == "anchoring"    = Anchoring val
  127   | t == "verification" = Verification val
  128   | t == "doubt"        = Doubt val
  129   | t == "logicalinference" = Verification val
  130   | t == "proofrequest" = Searching val
  131   | t == "distinctionrequest" = Doubt val
  132   | t == "counterexamplerequest" = Contradiction val val
  133   | t == "obligationduty" = Verification val
  134   | t == "permissionright" = Verification val
  135   | t == "temporalordering" = Verification val
  136   | t == "contrastcorrection" = Doubt val
  137   | otherwise           = CustomAtom t val

  139 inferRegister :: [MeaningAtom] -> Register
  140 inferRegister as
  141   | any (\a -> case maTag a of Exhaustion _ -> True; _ -> False) as = Exhaust
  142   | any (\a -> case maTag a of NeedContact _ -> True; _ -> False) as = Contact
  143   | any (\a -> case maTag a of Anchoring _ -> True; _ -> False) as = Anchor
  144   | any (\a -> case maTag a of Searching _ -> True; _ -> False) as = Search
  145   | otherwise = Neutral

  147 updateTrace :: AtomTrace -> Int -> AtomSet -> AtomTrace
  148 updateTrace old turn atoms =
  149   let newHistory = take 20 $ (fromIntegral turn, asLoad atoms) : atHistory old
  150       alpha = atAlpha old
  151       newLoad = if null (atHistory old)
  152                 then asLoad atoms
  153                 else alpha * asLoad atoms + (1.0 - alpha) * atCurrentLoad old
  154   in old { atHistory = newHistory, atCurrentLoad = newLoad }

  156 extractObject :: Text -> Text
  157 extractObject t =
  158   let ws = filter (\w -> T.length w > 3 && not (isStopWord w)) (T.words t)
  159   in case ws of
  160        (x:_) -> T.filter (`notElem` ("?!" :: String)) x
  161        []    -> "\1101\1090\1086"

  177   Searching x     -> x
  178   Verification x  -> x
  179   Doubt x         -> x
  180   AgencyFound x   -> textShow x

  183   Contradiction x _ -> x
  184   Exhaustion x    -> x
  185   NeedContact x   -> x
