   54 computeShadowContext :: ShadowResult -> InputPropositionFrame -> AtomTrace -> Double -> EmbeddingQuality -> Double -> Bool -> ShadowContext
   55 computeShadowContext shadowResult frame newTrace intuitConfidence embeddingQuality embSimilarity apiHealthy =
   56   let divergence = srDivergence shadowResult
   57       shadowStatus = srStatus shadowResult
   58       mismatchTags = concat
   59         [ ["family" | sdFamilyMismatch divergence]
   60         , ["force" | sdForceMismatch divergence]
   61         , ["clause" | sdClauseMismatch divergence]
   62         , ["layer" | sdLayerMismatch divergence]
   63         , ["warranted" | sdWarrantedMismatch divergence]
   64         ]
   65       diagnosticTags = srDiagnostics shadowResult
   66       shadowHasDivergence = shadowStatus == ShadowDiverged || not (null mismatchTags)
   67       shadowFamily = fst <$> srDatalogVerdict shadowResult
   68       shadowMessage =
   69         case (shadowStatus, mismatchTags, diagnosticTags) of
   70           (ShadowUnavailable, _, []) -> "shadow_unavailable"
   71           (ShadowUnavailable, _, diags) -> "shadow_unavailable:" <> T.intercalate "|" diags
   72           (ShadowMatch, [], []) -> "shadow_match"
   73           (ShadowMatch, [], diags) -> "shadow_match:" <> T.intercalate "|" diags
   74           (ShadowMatch, mismatches, []) -> "shadow_match_inconsistent:" <> T.intercalate "," mismatches
   75           (ShadowMatch, mismatches, diags) ->
   76             "shadow_match_inconsistent:" <> T.intercalate "," mismatches <> "|diag:" <> T.intercalate "|" diags
   77           (ShadowDiverged, mismatches, []) -> "shadow_diverged:" <> T.intercalate "," mismatches
   78           (ShadowDiverged, mismatches, diags) ->
   79             "shadow_diverged:" <> T.intercalate "," mismatches <> "|diag:" <> T.intercalate "|" diags
   80       severity = classifyShadowDivergenceSeverity shadowStatus shadowFamily divergence diagnosticTags
   81       baseLegit =
   82         legitimacyScoreWithPenalty
   83           (ipfConfidence frame)
   84           (computeShadowLegitimacyPenaltyWithSeverity severity divergence)
   85           (atCurrentLoad newTrace)
   86           apiHealthy
   87       embBonus =
   88         if embeddingQuality == EmbeddingQualityModeled && embSimilarity > embSimilarityBonusThreshold
   89           then embSimilarityLegitimacyBonus
   90           else 0.0
   91       intuitBonus =
   92         if intuitConfidence > intuitConfidenceBonusThreshold
   93           then intuitConfidenceLegitimacyBonus
   94           else 0.0
   95       adjustedBaseLegit = min 1.0 (baseLegit + embBonus + intuitBonus)
   96   in ShadowContext
   97       { scAdjustedBaseLegit = adjustedBaseLegit
   98       , scShadowStatus = shadowStatus
   99       , scShadowHasDivergence = shadowHasDivergence
  100       , scShadowDivergenceKind = sdKind divergence
  101       , scShadowDivergenceSeverity = severity
  102       , scShadowSnapshotId = srSnapshotId shadowResult
  103       , scShadowFamily = shadowFamily
  104       , scShadowForce = snd <$> srDatalogVerdict shadowResult
  105       , scShadowMessage = shadowMessage
  106       }

  108 resolveShadowFamily :: ShadowPolicy -> CanonicalMoveFamily -> ShadowContext -> ShadowResolution
  109 resolveShadowFamily policy requestedFamily sc =
  110   case policy of
  111     ShadowObserve ->
  112       ShadowResolution requestedFamily False
  113     ShadowPreferVerified ->
  114       if scShadowHasDivergence sc
  115         then ShadowResolution (maybe requestedFamily id (scShadowFamily sc)) False
  116         else ShadowResolution requestedFamily False
  117     ShadowBlockOnUnavailableOrDivergence ->
  118       case scShadowDivergenceSeverity sc of
  119         ShadowSeverityUnavailable -> ShadowResolution CMRepair True
  120         ShadowSeveritySafety -> ShadowResolution (maybe CMRepair id (scShadowFamily sc)) True
  121         ShadowSeverityContract -> ShadowResolution (maybe CMRepair id (scShadowFamily sc)) True
  122         ShadowSeverityAdvisory -> ShadowResolution (maybe requestedFamily id (scShadowFamily sc)) False
  123         ShadowSeverityClean -> ShadowResolution requestedFamily False

  125 legitimacyScoreWithPenalty :: Double -> Double -> Double -> Bool -> Double
  126 legitimacyScoreWithPenalty parserConfidence shadowPenalty emaLoad apiHealthy =
  127   let fullPenaltyScore = legitimacyScore parserConfidence ShadowDivergence
  128         { sdKind = ShadowNoDivergence
  129         , sdFamilyMismatch = False
  130         , sdForceMismatch = False
  131         , sdClauseMismatch = False
  132         , sdLayerMismatch = False
  133         , sdWarrantedMismatch = False
  134         } emaLoad apiHealthy
  135   in max 0.0 (min 1.0 (fullPenaltyScore - shadowPenalty))

  137 classifyShadowDivergenceSeverity :: ShadowStatus -> Maybe CanonicalMoveFamily -> ShadowDivergence -> [Text] -> ShadowDivergenceSeverity
  138 classifyShadowDivergenceSeverity status shadowFamily divergence diagnostics =
  139   case status of
  140     ShadowUnavailable -> ShadowSeverityUnavailable
  141     ShadowMatch
  142       | not (hasDivergence divergence) -> ShadowSeverityClean
  143       | advisoryShift -> ShadowSeverityAdvisory
  144       | safetyShift -> ShadowSeveritySafety
  145       | otherwise -> ShadowSeverityContract
  146     ShadowDiverged
  147       | safetyShift -> ShadowSeveritySafety
  148       | advisoryShift -> ShadowSeverityAdvisory
  149       | otherwise -> ShadowSeverityContract
  150   where
  151     hasDivergence d =
  152       or
  153         [ sdFamilyMismatch d
  154         , sdForceMismatch d
  155         , sdClauseMismatch d
  156         , sdLayerMismatch d
  157         , sdWarrantedMismatch d
  158         ]
  159     safetyShift =
  160       shadowFamily `elem` map Just [CMRepair, CMContact, CMConfront]
  161         || any isSafetyDiagnostic diagnostics
  162     advisoryShift =
  163       shadowFamily `elem` map Just [CMDeepen, CMClarify, CMDefine, CMDistinguish, CMReflect, CMAnchor, CMGround]
  164         && any isAdvisoryDiagnostic diagnostics
  165         && not safetyShift
  166     isSafetyDiagnostic d =
  167       any (`T.isPrefixOf` d)
  168         [ "repair_signal:"
  169         , "repair_detail:"
  170         , "contact_signal:"
  171         , "contact_detail:"
  172         , "confront_signal:"
  173         , "agency_loss"
  174         ]
  175     isAdvisoryDiagnostic d =
  176       any (`T.isPrefixOf` d)
  177         [ "requested_family_shift:"
  178         , "input_force_mismatch:"
  179         , "deepen_signal:"
  180         , "deepen_detail:"
  181         , "search_detail:"
  182         , "clarify_signal:"
  183         , "clarify_detail:"
  184         , "anchor_signal:"
  185         , "anchor_detail:"
  186         ]
