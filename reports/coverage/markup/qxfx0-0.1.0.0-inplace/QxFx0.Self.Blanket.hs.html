   31 -- | Project a 'SystemState' onto its structural self-identity
   32 -- snapshot. Pure and total. The result depends only on the four
   33 -- fields that participate in the structural invariants
   34 -- ('QxFx0.Self.Types.SelfBlanket') and is otherwise independent
   35 -- of dialogue history, embeddings, observability, or recovery
   36 -- traces.
   37 computeSelfBlanket :: SystemState -> SelfBlanket
   38 computeSelfBlanket ss = SelfBlanket
   39   { sbSessionId           = ssSessionId ss
   40   , sbMorphologyTotalSize = morphologyTotalSize (ssMorphology ss)
   41   , sbIdentityClaimsCount = length (idsIdentityClaims (ssIdentity ss))
   42   , sbTurnCount           = ssTurnCount ss
   43   }
   44 
   45 -- | Sum of entry counts across all morphology dictionaries.
   46 -- A 'MorphologyData' with zero total entries cannot drive
   47 -- generation and so cannot represent /this system/.
   48 morphologyTotalSize :: MorphologyData -> Int
   49 morphologyTotalSize md =
   50   Map.size (mdPrepositional md)
   51     + Map.size (mdGenitive md)
   52     + Map.size (mdNominative md)
   53     + Map.size (mdFormsBySurface md)
