  102 data SchemaContractResult
  103   = SchemaContractOk !Int
  104   | SchemaContractFreshBootstrapable
  105   | SchemaContractVersionBehind !Int !Int   -- ^ expected, actual
  106   | SchemaContractMissingTable !Text
  107   | SchemaContractMissingColumns !Text ![Text]
  108   | SchemaContractMissingIndex !Text
  109   | SchemaContractMissingTrigger !Text
  110   | SchemaContractMissingFTS !Text
  111   | SchemaContractInconsistent !Text
  112   | SchemaContractQueryFailed !Text
  113   deriving stock (Eq, Show)
  114 
  115 data SchemaVersionStatus
  116   = SchemaVersionMissingTable
  117   | SchemaVersionEmpty
  118   | SchemaVersionPresent !Int
  119   deriving stock (Eq, Show)
  120 
  121 renderSchemaContractResult :: SchemaContractResult -> Text
  122 renderSchemaContractResult (SchemaContractOk v) =
  123   "schema_ok version=" <> T.pack (show v)
  124 renderSchemaContractResult SchemaContractFreshBootstrapable =
  125   "schema_bootstrapable_fresh_db"
  126 renderSchemaContractResult (SchemaContractVersionBehind expected actual) =
  127   "schema_version_behind expected=" <> T.pack (show expected)
  128   <> " actual=" <> T.pack (show actual)
  129 renderSchemaContractResult (SchemaContractMissingTable t) =
  130   "schema_missing_table " <> t
  131 renderSchemaContractResult (SchemaContractMissingColumns t cols) =
  132   "schema_missing_columns table=" <> t
  133   <> " columns=" <> T.intercalate "," cols
  134 renderSchemaContractResult (SchemaContractMissingIndex ix) =
  135   "schema_missing_index " <> ix
  136 renderSchemaContractResult (SchemaContractMissingTrigger tr) =
  137   "schema_missing_trigger " <> tr
  138 renderSchemaContractResult (SchemaContractMissingFTS fts) =
  139   "schema_missing_fts " <> fts
  140 renderSchemaContractResult (SchemaContractInconsistent reason) =
  141   "schema_inconsistent " <> reason
  142 renderSchemaContractResult (SchemaContractQueryFailed msg) =
  143   "schema_query_failed " <> msg

  145 -- | Read-only schema contract check.  Does NOT mutate the database.
  146 checkSchemaContract :: NSQL.Database -> IO SchemaContractResult
  147 checkSchemaContract db = do
  148   versionStatus <- readSchemaVersion db
  149   case versionStatus of
  150     SchemaVersionMissingTable -> do
  151       anyTableExists <- anyM (tableExists db) schemaContractTables
  152       if not anyTableExists
  153         then pure SchemaContractFreshBootstrapable
  154         else do
  155           v1 <- isV1Shape db
  156           if v1
  157             then pure (SchemaContractVersionBehind currentSchemaVersion 1)
  158             else pure (SchemaContractInconsistent "tables exist but schema_version missing and shape not v1")
  159     SchemaVersionEmpty -> do
  160       v1 <- isV1Shape db
  161       if v1
  162         then pure (SchemaContractVersionBehind currentSchemaVersion 1)
  163         else do
  164           v2 <- isV2Shape db
  165           if v2
  166             then pure (SchemaContractOk currentSchemaVersion)
  167             else pure (SchemaContractInconsistent "schema_version empty and shape not v1 or v2")
  168     SchemaVersionPresent v
  169       | v < currentSchemaVersion ->
  170           pure (SchemaContractVersionBehind currentSchemaVersion v)
  171       | otherwise ->
  172           checkTablesAndColumns db
  173 
  174 -- | Check tables and columns only, ignoring schema_version.
  175 validateSchemaContractTablesAndColumns :: NSQL.Database -> IO SchemaContractResult
  176 validateSchemaContractTablesAndColumns db = checkTablesAndColumns db
  177 
  178 -- | Check that all required tables exist, then required columns, indexes,
  179 --   triggers, and FTS virtual tables.
  180 checkTablesAndColumns :: NSQL.Database -> IO SchemaContractResult
  181 checkTablesAndColumns db = do
  182   missingTables <- filterM (fmap not . tableExists db) schemaContractTables
  183   case missingTables of
  184     (t:_) -> pure (SchemaContractMissingTable t)
  185     [] -> do
  186       let checkOne (tbl, cols) = do
  187             missing <- filterM (columnMissing db tbl) cols
  188             pure (if null missing then Nothing else Just (tbl, missing))
  189       badCols <- mapMaybe id <$> mapM checkOne (Map.toList schemaContractColumns)
  190       case badCols of
  191         ((tbl, cols):_) -> pure (SchemaContractMissingColumns tbl cols)
  192         [] -> do
  193           missingIndexes <- filterM (fmap not . indexExists db) schemaContractIndexes
  194           case missingIndexes of
  195             (ix:_) -> pure (SchemaContractMissingIndex ix)
  196             [] -> do
  197               missingTriggers <- filterM (fmap not . triggerExists db) schemaContractTriggers
  198               case missingTriggers of
  199                 (tr:_) -> pure (SchemaContractMissingTrigger tr)
  200                 [] -> do
  201                   missingFTS <- filterM (fmap not . ftsTableExists db) schemaContractFTS
  202                   case missingFTS of
  203                     (fts:_) -> pure (SchemaContractMissingFTS fts)
  204                     [] -> pure (SchemaContractOk currentSchemaVersion)

  206 readSchemaVersion :: NSQL.Database -> IO SchemaVersionStatus
  207 readSchemaVersion db = do
  208   schemaVersionTableExists <- tableExists db "schema_version"
  209   if not schemaVersionTableExists
  210     then pure SchemaVersionMissingTable
  211     else do
  212       mVersion <- readSchemaVersionRow db
  213       case mVersion of
  214         Nothing -> pure SchemaVersionEmpty
  215         Just v  -> pure (SchemaVersionPresent v)
  216 
  217 readSchemaVersionRow :: NSQL.Database -> IO (Maybe Int)
  218 readSchemaVersionRow db = do
  219   mStmt <- NSQL.prepare db "SELECT version FROM schema_version ORDER BY version DESC LIMIT 1"
  220   case mStmt of
  221     Left _ -> pure Nothing
  222     Right stmt -> do
  223       hasRow <- NSQL.stepRow stmt
  224       version <- if hasRow then NSQL.columnInt stmt 0 else pure 0
  225       _ <- tryQxFx0 (NSQL.finalize stmt)
  226       if hasRow && version > 0
  227         then pure (Just version)
  228         else pure Nothing

  230 -- | True if the database has the turn_quality table but is missing any v2 column.
  231 --   Used to infer a v1-shaped legacy database when the schema_version marker
  232 --   is missing or empty.
  233 isV1Shape :: NSQL.Database -> IO Bool
  234 isV1Shape db = do
  235   tqExists <- tableExists db "turn_quality"
  236   if not tqExists
  237     then pure False
  238     else do
  239       missing <- filterM (columnMissing db "turn_quality") v2TurnQualityColumns
  240       pure (not (null missing))
  241 
  242 -- | True if the database has all required tables and all required columns.
  243 --   Used to recognise a v2-complete database that may be missing its version marker.
  244 isV2Shape :: NSQL.Database -> IO Bool
  245 isV2Shape db = do
  246   result <- checkTablesAndColumns db
  247   pure $ case result of
  248     SchemaContractOk _ -> True
  249     _ -> False

  251 tableExists :: NSQL.Database -> Text -> IO Bool
  252 tableExists db tableName = do
  253   let sql = "SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = ? LIMIT 1"
  254   mStmt <- NSQL.prepare db sql
  255   case mStmt of
  256     Left _ -> pure False
  257     Right stmt -> do
  258       _ <- NSQL.bindText stmt 1 tableName
  259       hasRow <- NSQL.stepRow stmt
  260       _ <- tryQxFx0 (NSQL.finalize stmt)
  261       pure hasRow
  262 
  263 columnMissing :: NSQL.Database -> Text -> Text -> IO Bool
  264 columnMissing db tableName columnName = do
  265   let sql = "SELECT 1 FROM pragma_table_info('" <> tableName <> "') WHERE name = ? LIMIT 1"
  266   mStmt <- NSQL.prepare db sql
  267   case mStmt of
  268     Left _ -> pure True
  269     Right stmt -> do
  270       _ <- NSQL.bindText stmt 1 columnName
  271       hasRow <- NSQL.stepRow stmt
  272       _ <- tryQxFx0 (NSQL.finalize stmt)
  273       pure (not hasRow)
