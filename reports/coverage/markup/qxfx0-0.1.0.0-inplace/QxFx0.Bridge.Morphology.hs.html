   26 data MorphRemoteToken = MorphRemoteToken
   27   { mrtWord   :: !Text
   28   , mrtLemma  :: !Text
   29   , mrtPos    :: !(Maybe Text)
   30   , mrtCase   :: !(Maybe Text)
   31   , mrtNumber :: !(Maybe Text)
   32   , mrtGender :: !(Maybe Text)
   33   , mrtTense  :: !(Maybe Text)
   34   , mrtMood   :: !(Maybe Text)
   35   , mrtPerson :: !(Maybe Text)
   36   } deriving stock (Eq, Show)
   37 
   38 instance Aeson.FromJSON MorphRemoteToken where
   39   parseJSON = Aeson.withObject "MorphRemoteToken" $ \o ->
   40     MorphRemoteToken
   41       <$> o Aeson..:  "word"
   42       <*> o Aeson..:  "lemma"
   43       <*> o Aeson..:! "pos"
   44       <*> o Aeson..:! "case"

   51 data MorphBackend
   52   = MorphBackendLocal
   53   | MorphBackendRemote
   54   deriving stock (Eq, Show)
   55 
   56 resolveMorphBackend :: IO MorphBackend
   57 resolveMorphBackend = do
   58   mBackend <- fmap (fmap (T.toLower . T.pack)) (lookupEnv "QXFX0_MORPH_BACKEND")
   59   pure $ case mBackend of
   60     Just "remote" -> MorphBackendRemote
   61     _ -> MorphBackendLocal
   62 
   63 morphApiUrl :: IO (Maybe String)
   64 morphApiUrl = lookupEnv "MORPH_API_URL"
   65 
   66 analyzeMorphRemote :: Text -> IO (Either Text [MorphRemoteToken])
   67 analyzeMorphRemote text = do
   68   mUrl <- morphApiUrl
   69   case mUrl of
   70     Nothing -> pure (Left "MORPH_API_URL not set")
   71     Just url -> fetchRemoteMorph url text
   72 
   73 fetchRemoteMorph :: String -> Text -> IO (Either Text [MorphRemoteToken])
   74 fetchRemoteMorph url txt = do
   75   requestResult <- try @HttpException (HTTP.parseRequest url)
   76   case requestResult of
   77     Left err -> pure (Left ("invalid morph url: " <> showText err))
   78     Right baseRequest -> do
   79       let request =
   80             HTTP.setRequestIgnoreStatus $
   81               HTTP.setRequestResponseTimeout (HTTPClient.responseTimeoutMicro 5000000) $
   82                 HTTP.setRequestBodyJSON (Aeson.object ["text" Aeson..= txt]) $
   83                   HTTP.setRequestMethod "POST" baseRequest
   84       responseResult <- try @HttpException (HTTP.httpBS request)
   85       case responseResult of
   86         Left err -> pure (Left ("morph request failed: " <> showText err))
   87         Right response
   88           | not (successfulStatus response) ->
   89               pure (Left ("unexpected morph status: " <> showText (HTTP.getResponseStatusCode response)))
   90           | not (jsonContentType response) ->
   91               pure (Left ("unexpected morph content-type: " <> renderContentTypes response))
   92           | otherwise ->
   93               pure (parseMorphResponse (HTTP.getResponseBody response))
   94 
   95 checkMorphHealth :: IO Bool
   96 checkMorphHealth = do
   97   mUrl <- morphApiUrl
   98   case mUrl of
   99     Nothing -> pure False
  100     Just url -> do
  101       requestResult <- try @HttpException (HTTP.parseRequest (url <> "/health"))
  102       case requestResult of
  103         Left _ -> pure False
  104         Right baseRequest -> do
  105           let request = HTTP.setRequestResponseTimeout (HTTPClient.responseTimeoutMicro 2000000) $
  106                         HTTP.setRequestMethod "GET" baseRequest
  107           responseResult <- try @HttpException (HTTP.httpBS request)
  108           pure $ case responseResult of
  109             Right response -> successfulStatus response
  110             Left _ -> False
  111 
  112 successfulStatus :: HTTP.Response a -> Bool
  113 successfulStatus response =
  114   let code = HTTP.getResponseStatusCode response
  115    in code >= 200 && code < 300
  116 
  117 jsonContentType :: HTTP.Response a -> Bool
  118 jsonContentType response =
  119   any isJsonHeader (HTTP.getResponseHeader "Content-Type" response)
  120   where
  121     isJsonHeader raw =
  122       let lowered = B8.map Char.toLower raw
  123        in "application/json" `B8.isPrefixOf` B8.dropWhile (== ' ') lowered
  124 
  125 renderContentTypes :: HTTP.Response a -> Text
  126 renderContentTypes response =
  127   case HTTP.getResponseHeader "Content-Type" response of
  128     [] -> "<missing>"
  129     headers -> T.intercalate ", " (map (T.pack . B8.unpack) headers)
  130 
  131 parseMorphResponse :: B8.ByteString -> Either Text [MorphRemoteToken]
  132 parseMorphResponse body = do
  133   value <- case Aeson.eitherDecodeStrict' body of
  134     Left err -> Left ("invalid morph json: " <> showText err)
  135     Right parsed -> Right parsed
  136   case Aeson.parseMaybe (Aeson.withObject "resp" (Aeson..: "tokens")) value of
  137     Just tokens -> Right tokens
  138     Nothing -> Left "invalid morph response format: missing 'tokens' field"
  139 
  140 showText :: Show a => a -> Text
  141 showText = T.pack . show
