   65 handleTurnEffect :: RuntimeContext -> TurnEffectRequest -> IO TurnEffectResult
   66 handleTurnEffect ctx request =
   67   case request of
   68     TurnReqEmbedding inputText ->
   69       TurnResEmbedding <$> textToEmbeddingResultWithManager (rtwHttpManager (rcWorkers ctx)) inputText
   70     TurnReqNixGuard concept agency tension -> do
   71       nixPath <- resolveNixPath ctx
   72       status <- checkNixWithCache (rtcNix (rcCaches ctx)) nixPath concept agency tension
   73       pure (TurnResNixGuard status)
   74     TurnReqConsciousness semanticInput humanTheta resonance conatusEnergy salienceWeights -> do
   75       cl <- readConsciousLoop ctx
   76       let salience = computeSalience salienceWeights conatusEnergy (emptyField { fieldResonance = mkResonance resonance })
   77           (nextLoop, fragment) =
   78             runConsciousnessLoopWithSalience salience cl semanticInput humanTheta resonance
   79           currentNarrative = clLastNarrative nextLoop
   80           narrativeFragment = if T.null fragment then Nothing else Just fragment
   81       pure (TurnResConsciousness nextLoop currentNarrative narrativeFragment)
   82     TurnReqIntuition inputText resonance tension turnNumber conatusEnergy salienceWeights semanticConfig -> do
   83       intuitive <- readIntuition ctx
   84       let salience = computeSalience salienceWeights conatusEnergy (emptyField { fieldResonance = mkResonance resonance })
   85           (mFlash, intuitionState) =
   86             checkIntuitionWithInputAndSalience
   87               salience semanticConfig inputText resonance tension turnNumber intuitive
   88       pure (TurnResIntuition mFlash (effectivePosterior intuitionState) intuitionState)
   89     TurnReqCommitRuntimeState previewLoop previewIntuition observation -> do
   90       commitRuntimeTurnState ctx previewLoop previewIntuition observation
   91       pure TurnResCommitRuntimeState
   92     TurnReqApiHealth ->
   93       TurnResApiHealth <$> readApiHealth ctx
   94     TurnReqReadEnv key ->
   95       if isAllowedReadEnvKey key
   96         then TurnResReadEnv . fmap T.pack <$> lookupEnv (T.unpack key)
   97         else pure (TurnResReadEnv Nothing)
   98     TurnReqTestMarkOnceFile pathText -> do
   99       mPath <- resolveTrustedMarkerPath (T.unpack (T.strip pathText))
  100       case mPath of
  101         Nothing ->
  102           pure (TurnResTestMarkOnceFile False)
  103         Just path -> do
  104           TurnResTestMarkOnceFile <$> markMarkerOnce path
  105     TurnReqShadow family force tags -> do
  106       execResult <- resolveSouffleExecutableCached ctx
  107       sr <- case execResult of
  108         Right executable -> Datalog.runDatalogShadowWithExecutable executable family force tags
  109         Left err ->
  110           pure Datalog.ShadowResult
  111             { Datalog.srStatus = ShadowUnavailable
  112             , Datalog.srDivergence = emptyShadowDivergence { sdKind = ShadowUnavailableDivergence }
  113             , Datalog.srDatalogVerdict = Nothing
  114             , Datalog.srSnapshotId = ShadowSnapshotId "shadow:runtime_souffle_unavailable"
  115             , Datalog.srDiagnostics = [err]
  116             }
  117       pure
  118         (TurnResShadow
  119           ((\v -> (r5Family v, r5Force v)) <$> Datalog.srDatalogVerdict sr)
  120           (Datalog.srStatus sr)
  121           (Datalog.srDivergence sr)
  122           (Datalog.srSnapshotId sr)
  123           (Datalog.srDiagnostics sr))
  124     TurnReqAgdaVerify ->
  125       TurnResAgdaVerify <$> wireVerifyAgda ctx
  126     TurnReqCurrentTime ->
  127       TurnResCurrentTime <$> rcTimeSource ctx
  128     TurnReqRequestId ->
  129       TurnResRequestId <$> generateRequestId (rcTimeSource ctx)
  130     TurnReqSemanticIntrospectionEnv -> do
  131       envResult <- handleTurnEffect ctx (TurnReqReadEnv "QXFX0_SEMANTIC_INTROSPECTION")
  132       case envResult of
  133         TurnResReadEnv value -> pure (TurnResSemanticIntrospectionEnv (maybe False (const True) value))
  134         _ -> pure (TurnResSemanticIntrospectionEnv False)
  135     TurnReqSaveState ss sid mProj ->
  136       TurnResSaveState <$> saveStateWithProjection (withRuntimeDb ctx) ss sid mProj
  137     TurnReqRollbackTurnProjections sid stableTurn ->
  138       TurnResRollbackTurnProjections <$> rollbackTurnProjections (withRuntimeDb ctx) sid stableTurn
  139     TurnReqCheckpoint turnCount -> do
  140       withRuntimeDb ctx $ \db -> maybeCheckpoint db turnCount
  141       pure TurnResCheckpointCompleted
  142     TurnReqLinearizeClaimAst mPgfPath lang claimAst ->
  143       TurnResLinearizeClaimAst <$> linearizeClaimAstGfLang mPgfPath lang claimAst
  144     TurnReqLinearizeDialogAtoms mPgfPath lang da ->
  145       TurnResLinearizeDialogAtoms <$> linearizeDialogAtomsGfLang mPgfPath lang da
  146     TurnReqExternalQuery tool need query -> do
  147       transport <- buildTransportFromEnvWithManager (rtwHttpManager (rcWorkers ctx))
  148       result <- queryExternalTool transport tool need query
  149       pure (TurnResExternalQuery result)

  151 generateRequestId :: TimeSource -> IO T.Text
  152 generateRequestId timeSource = do
  153   requestUuid <- UUIDv4.nextRandom
  154   now <- timeSource
  155   let ts = formatTime defaultTimeLocale "%Y%m%d%H%M%S" now
  156   pure (T.pack ts <> "-" <> UUID.toText requestUuid)

  158 resolveTrustedMarkerPath :: FilePath -> IO (Maybe FilePath)
  159 resolveTrustedMarkerPath rawPath = do
  160   let normalized = normalise (dropWhile (== ' ') rawPath)
  161   stateDir <- resolveStateDir
  162   createDirectoryIfMissing True stateDir
  163   canonicalStateDir <- canonicalizePath stateDir
  164   canonicalTmp <- canonicalizePath "/tmp"
  165   if null normalized
  166     then pure Nothing
  167     else
  168       if isAbsolute normalized
  169         then resolveAbsoluteMarkerPath canonicalTmp canonicalStateDir normalized
  170         else
  171           pure $
  172             if isSafeRelativeMarker normalized
  173               then Just (canonicalStateDir </> "test-hooks" </> takeFileName normalized)
  174               else Nothing
  175 
  176 resolveStateDir :: IO FilePath
  177 resolveStateDir = do
  178   mStateDir <- lookupEnv "QXFX0_STATE_DIR"
  179   pure $ case fmap normalise mStateDir of
  180     Just path | not (null path) -> path
  181     _ -> "/tmp/qxfx0"
  182 
  183 resolveAbsoluteMarkerPath :: FilePath -> FilePath -> FilePath -> IO (Maybe FilePath)
  184 resolveAbsoluteMarkerPath canonicalTmp canonicalStateDir absoluteCandidate = do
  185   let parentDir = takeDirectory absoluteCandidate
  186       markerName = takeFileName absoluteCandidate
  187   parentExists <- doesDirectoryExist parentDir
  188   if not parentExists || not (isSafeRelativeMarker markerName)
  189     then pure Nothing
  190     else do
  191       canonicalParent <- canonicalizePath parentDir
  192       let canonicalCandidate = canonicalParent </> markerName
  193       withinTmp <- isPathWithin canonicalTmp canonicalCandidate
  194       withinState <- isPathWithin canonicalStateDir canonicalCandidate
  195       pure $
  196         if withinTmp || withinState
  197           then Just canonicalCandidate
  198           else Nothing
  199 
  200 isSafeRelativeMarker :: FilePath -> Bool
  201 isSafeRelativeMarker relPath =
  202   relPath == takeFileName relPath
  203     && not (null relPath)
  204     && all isMarkerChar relPath
  205 
  206 isMarkerChar :: Char -> Bool
  207 isMarkerChar c = isAlphaNum c || c `elem` ("._-" :: String)
  208 
  209 markMarkerOnce :: FilePath -> IO Bool
  210 markMarkerOnce path = do
  211   createDirectoryIfMissing True (takeDirectory path)
  212   catchIO
  213     (do fd <- openFd path WriteOnly defaultFileFlags
  214           { exclusive = True
  215           , nofollow = True
  216           , creat = Just markerFileMode
  217           }
  218         handle <- fdToHandle fd
  219         hPutStr handle "triggered\n"
  220         hClose handle
  221         pure True)
  222     (\_ -> pure False)
  223 
  224 markerFileMode :: FileMode
  225 markerFileMode = ownerReadMode `unionFileModes` ownerWriteMode
  226 
  227 isAllowedReadEnvKey :: T.Text -> Bool
  228 isAllowedReadEnvKey key = key `Set.member` allowedReadEnvKeys
