  180 parseProposition :: Text -> InputPropositionFrame
  181 parseProposition rawText =
  182   let tokens = tokenizeKeywordText rawText
  183       semanticFrame = buildUtteranceSemanticFrame rawText
  184       isQ = T.isSuffixOf "?" (T.strip rawText)
  185       propType = fromMaybe (detectPropositionType rawText tokens) (propositionTypeHintFromFrame semanticFrame)
  186       family = propositionToFamily propType
  187       focus = fromMaybe (extractFocusEntity rawText) (specialFocusEntity propType)
  188       focusNom = toNominative (MorphologyData M.empty M.empty M.empty M.empty) focus
  189       (semanticSubject, semanticTarget, semanticCandidates, semanticEvidence) =
  190         inferSemanticSlotsWithFrame rawText tokens propType semanticFrame
  191       force = forceForFamily family
  192       clause = if isQ then Interrogative else clauseFormForIF force
  193       layer = layerForFamily family
  194       negated = containsKeywordPhrase tokens propositionNegationFragment
  195       reg = inferRegisterHint semanticFrame tokens
  196       keyPhrases = extractKeyPhrases tokens
  197       emotion = detectEmotion tokens
  198       confidence = computeConfidence propType keyPhrases semanticFrame
  199   in emptyInputPropositionFrame
  200     { ipfRawText = rawText
  201     , ipfPropositionType = textShow propType
  202     , ipfFocusEntity = focus
  203     , ipfFocusNominative = focusNom
  204     , ipfSemanticSubject = semanticSubject
  205     , ipfSemanticTarget = semanticTarget
  206     , ipfSemanticCandidates = semanticCandidates
  207     , ipfSemanticEvidence = semanticEvidence
  208     , ipfCanonicalFamily = family
  209     , ipfIllocutionaryForce = force
  210     , ipfClauseForm = clause
  211     , ipfSemanticLayer = layer
  212     , ipfKeyPhrases = keyPhrases
  213     , ipfEmotionalTone = emotion
  214     , ipfConfidence = confidence
  215     , ipfIsQuestion = isQ
  216     , ipfIsNegated = negated
  217     , ipfRegisterHint = reg
  218     }

  312 inferSemanticSlotsWithFrame :: Text -> [Text] -> PropositionType -> UtteranceSemanticFrame -> (Text, Text, [Text], [Text])
  313 inferSemanticSlotsWithFrame rawText tokens propositionType semanticFrame =
  314   let (subjectBase, targetBase, candidatesBase, evidenceBase) =
  315         inferSemanticSlots rawText tokens propositionType
  316       subjectFromFrame =
  317         if T.null (T.strip (usfTopic semanticFrame))
  318           then ""
  319           else usfTopic semanticFrame
  320       targetFromFrame = fromMaybe "" (usfTarget semanticFrame)
  321       subject =
  322         case propositionType of
  323           ContemplativeTopic -> pickNonEmpty subjectFromFrame subjectBase
  324           DialogueInvitationQ -> pickNonEmpty subjectFromFrame subjectBase
  325           PurposeQ ->
  326             if isDeicticTopic subjectFromFrame
  327               then subjectBase
  328               else pickNonEmpty subjectFromFrame subjectBase
  329           WorldCauseQ ->
  330             if isLikelyAdjectiveTopic subjectFromFrame
  331               then subjectBase
  332               else pickNonEmpty subjectFromFrame subjectBase
  333           LocationFormationQ -> pickNonEmpty subjectFromFrame subjectBase
  334           _ -> pickIfBaseEmpty subjectFromFrame subjectBase
  335       target = pickIfBaseEmpty targetFromFrame targetBase
  336       safeFrameCandidates = filter isSemanticCandidateSurface (usfSemanticCandidates semanticFrame)
  337       candidates =
  338         if null candidatesBase
  339           then take 8 (dedupeNormalized safeFrameCandidates)
  340           else take 8 (dedupeNormalized candidatesBase)
  341       -- Keep frame-level route diagnostics first so trace consumers always see
  342       -- route scores even when proposition-level evidence is long.
  343       evidence = take 16 (dedupeEvidence (frameEvidence semanticFrame <> evidenceBase))
  344   in (subject, target, candidates, evidence)

  346 frameEvidence :: UtteranceSemanticFrame -> [Text]
  347 frameEvidence semanticFrame =
  348   [ "frame.route_tag=" <> irhTag (usfRouteHint semanticFrame)
  349   , "frame.route_reason=" <> irhReason (usfRouteHint semanticFrame)
  350   , "frame.route_rule_score=" <> showRouteScore (irhRuleScore (usfRouteHint semanticFrame))
  351   , "frame.route_semantic_score=" <> showRouteScore (irhSemanticScore (usfRouteHint semanticFrame))
  352   , "frame.route_syntactic_score=" <> showRouteScore (irhSyntacticScore (usfRouteHint semanticFrame))
  353   , "frame.route_embedding_score=" <> showRouteScore (irhEmbeddingScore (usfRouteHint semanticFrame))
  354   , "frame.route_final_score=" <> showRouteScore (irhConfidence (usfRouteHint semanticFrame))
  355   , "frame.ambiguity=" <> usfAmbiguityLevel semanticFrame
  356   ]
  357   <> map ("frame.route_evidence=" <>) (take 2 (irhEvidence (usfRouteHint semanticFrame)))
  358   <> map unitEvidence (take 4 (usfWordUnits semanticFrame))
  359   where
  360     unitEvidence unit =
  361       "frame.token=" <> wmuSurfaceForm unit
  362         <> "|pos=" <> T.pack (show (wmuPartOfSpeech unit))
  363         <> "|role=" <> T.pack (show (wmuSyntacticRole unit))
  364     showRouteScore value =
  365       let scaled :: Integer
  366           scaled = round (value * 1000)
  367       in T.pack (show ((fromIntegral scaled / 1000.0) :: Double))

  369 pickNonEmpty :: Text -> Text -> Text
  370 pickNonEmpty preferred fallback
  371   | T.null (T.strip preferred) = fallback
  372   | otherwise = preferred

  374 pickIfBaseEmpty :: Text -> Text -> Text

  376   | T.null (T.strip base) = pickNonEmpty preferred base
  377   | otherwise = base

  468 detectContactSignal :: Text -> [Text] -> Maybe PropositionType
  469 detectContactSignal rawText tokens
  470   | containsKeywordPhrase tokens "как ты работаешь" = Nothing
  471   | containsKeywordPhrase tokens "как вы работаете" = Nothing
  472   | containsKeywordPhrase tokens "как ты устроен" = Nothing
  473   | containsKeywordPhrase tokens "как дела" = Just ContactSignal
  474   | containsKeywordPhrase tokens "как жизнь" = Just ContactSignal
  475   | containsKeywordPhrase tokens "как ты" && isShortHowYouContact tokens
  476       && not (containsKeywordPhrase tokens "как ты устроен")
  477       && not (containsKeywordPhrase tokens "как ты работаешь") = Just ContactSignal
  478   | T.toLower (T.strip rawText) `elem` ["привет", "здравствуй", "здравствуйте", "салют", "хай", "hello", "hi"]
  479       = Just ContactSignal
  480   | any (`T.isInfixOf` T.toLower rawText) ["добрый день", "доброе утро", "добрый вечер", "рад видеть"]
  481       = Just ContactSignal
  482   | containsKeywordPhrase tokens "до свидания" = Just ContactSignal
  483   | containsKeywordPhrase tokens "до встречи" = Just ContactSignal
  484   | containsKeywordPhrase tokens "всего доброго" = Just ContactSignal
  485   | containsKeywordPhrase tokens "всего хорошего" = Just ContactSignal
  486   | any (`elem` tokens) ["пока", "прощай", "бывай", "увидимся", "спасибо", "благодарю", "thanks"]
  487       = Just ContactSignal
  488   | containsKeywordPhrase tokens "как сам" = Just ContactSignal
  489   | containsKeywordPhrase tokens "как настроение" = Just ContactSignal
  490   | containsKeywordPhrase tokens "рад тебя видеть" = Just ContactSignal
  491   | containsKeywordPhrase tokens "рад вас видеть" = Just ContactSignal
  492   | containsKeywordPhrase tokens "начнем разговор" = Just ContactSignal
  493   | containsKeywordPhrase tokens "начнём разговор" = Just ContactSignal
  494   | containsKeywordPhrase tokens "можем пообщаться" = Just ContactSignal
  495   | containsKeywordPhrase tokens "я вернулся" = Just ContactSignal
  496   | containsKeywordPhrase tokens "снова привет" = Just ContactSignal
  497   | containsKeywordPhrase tokens "готов к разговору" = Just ContactSignal
  498   | containsKeywordPhrase tokens "контакт есть" = Just ContactSignal
  499   | containsKeywordPhrase tokens "есть контакт" = Just ContactSignal
  500   | containsKeywordPhrase tokens "мы на связи" = Just ContactSignal
  501   | containsKeywordPhrase tokens "слышишь меня" = Just ContactSignal
  502   | containsKeywordPhrase tokens "ты онлайн" = Just ContactSignal
  503   | shortDialogueProbe tokens && any (`elem` tokens) ["поговорим", "обсудим"] = Just ContactSignal
  504   | otherwise = Nothing
  505   where
  506     shortDialogueProbe ts = length ts <= 2

  634 inferSemanticSlots :: Text -> [Text] -> PropositionType -> (Text, Text, [Text], [Text])
  635 inferSemanticSlots rawText tokens propType =
  636   case propType of
  637     OperationalStatusQ ->
  638       ("система", "работа", [], semanticEvidenceFor rawText tokens propType)
  639     OperationalCauseQ ->
  640       ("система", "работа", [], semanticEvidenceFor rawText tokens propType)
  641     SystemLogicQ ->
  642       ("система", "логика", [], semanticEvidenceFor rawText tokens propType)
  643     SelfKnowledgeQ ->
  644       let lowered = T.toLower rawText
  645           target
  646             | asksAboutUser rawText = "user"
  647             | T.isInfixOf "помоч" lowered = "user_help"
  648             | any (`T.isInfixOf` lowered) ["намерени"] = "self_intentions"
  649             | any (`T.isInfixOf` lowered) ["важно", "ценност", "послание миру"] = "self_values"
  650             | any (`T.isInfixOf` lowered) ["будущ"] = "self_future"
  651             | any (`T.isInfixOf` lowered) ["свобод"] = "self_freedom"
  652             | any (`T.isInfixOf` lowered) ["промт", "prompt", "субъект", "субьект", "умный", "сложная система"] = "self_reflection"
  653             | any (`T.isInfixOf` lowered) ["свой же вопрос", "свой вопрос"] = "self_reflection"
  654             | any (`T.isInfixOf` lowered) ["умеешь", "можешь"] = "self_capability"
  655             | otherwise = "self"
  656           subject =
  657             case target of
  658               "user" -> "пользователь"
  659               "user_help" -> fromMaybe "помощь" (capabilitySubject tokens)
  660               "self_capability" -> fromMaybe "способность" (capabilitySubject tokens)
  661               "self_intentions" -> "намерения"
  662               "self_values" -> "принципы"
  663               "self_future" -> "будущее"
  664               "self_freedom" -> "свобода"
  665               "self_reflection" -> "саморефлексия"
  666               _ -> "система"
  667       in (subject, target, [], semanticEvidenceFor rawText tokens propType)
  668     DialogueInvitationQ ->
  669       (invitationTopic rawText, "dialogue", [], semanticEvidenceFor rawText tokens propType)
  670     ConceptKnowledgeQ ->
  671       (conceptSubject rawText tokens, "concept", [], semanticEvidenceFor rawText tokens propType)
  672     WorldCauseQ ->
  673       (fromMaybe "мир" (firstConcreteWorldNoun tokens), "причина", [], semanticEvidenceFor rawText tokens propType)
  674     LocationFormationQ ->
  675       (fromMaybe "мысль" (firstMentalNoun tokens), "источник", [], semanticEvidenceFor rawText tokens propType)
  676     SelfStateQ ->
  677       ("система", "внутренний_фокус", [], semanticEvidenceFor rawText tokens propType)
  678     PurposeQ ->
  679       let subject
  680             | any (`elem` tokens) ["ты", "вы", "система"] = "система"
  681             | any (`elem` tokens) ["тут", "здесь", "там"] = "объект"
  682             | otherwise = fromMaybe "объект" (extractTopicAfterMarkers rawText ["для", "у", "в", "о", "об", "про"])
  683       in (subject, "назначение", [], semanticEvidenceFor rawText tokens propType)
  684     ComparisonPlausibilityQ ->
  685       let candidates = comparisonCandidates rawText
  686       in ("сравнение", "логичность", candidates, semanticEvidenceFor rawText tokens propType)
  687     DistinctionQ ->
  688       let candidates = comparisonCandidates rawText
  689       in ("сравнение", "различение", candidates, semanticEvidenceFor rawText tokens propType)
  690     MisunderstandingReport ->
  691       ("диалог", "взаимопонимание", [], semanticEvidenceFor rawText tokens propType)
  692     GenerativePrompt ->
  693       ("", "порождение", [], semanticEvidenceFor rawText tokens propType)
  694     ContemplativeTopic ->
  695       (contemplativeTopic rawText, "созерцание", [], semanticEvidenceFor rawText tokens propType)
  696     _ ->
  697       ("", "", [], semanticEvidenceFor rawText tokens propType)

  998 comparisonCandidates :: Text -> [Text]
  999 comparisonCandidates rawText =
 1000   case splitByEither normalized of
 1001     pair@(_:_:_) -> take 2 pair
 1002     _ -> splitByFrom normalized
 1003   where
 1004     normalized = T.unwords (T.words (T.toLower (T.replace "\n" " " rawText)))
 1005     splitByEither txt =
 1006       case T.breakOn "или" txt of
 1007         (left, rightRaw)
 1008           | T.null rightRaw -> []
 1009           | otherwise ->
 1010               let right = T.drop 3 rightRaw
 1011                   leftCandidate = cleanCandidate left
 1012                   rightCandidate = cleanCandidate (trimAtQuestion right)
 1013               in filter (not . T.null) [leftCandidate, rightCandidate]
 1014     splitByFrom txt =
 1015       case T.breakOn " от " txt of
 1016         (left, rightRaw)
 1017           | T.null rightRaw -> []
 1018           | otherwise ->
 1019               let right = T.drop 4 rightRaw
 1020                   leftCandidate = cleanCandidate (dropDistinctionPrefix left)
 1021                   rightCandidate = cleanCandidate (trimAtQuestion right)
 1022               in filter (not . T.null) [leftCandidate, rightCandidate]
 1023     dropDistinctionPrefix txt =
 1024       let trimmed = T.strip txt
 1025           prefixes =
 1026             [ "как отличить "
 1027             , "чем отличается "
 1028             , "чем отличить "
 1029             , "как различить "
 1030             , "отличить "
 1031             , "различить "
 1032             ]
 1033       in stripKnownPrefix trimmed prefixes
 1034     stripKnownPrefix txt [] = txt
 1035     stripKnownPrefix txt (p:ps)
 1036       | p `T.isPrefixOf` txt = T.strip (T.drop (T.length p) txt)
 1037       | otherwise = stripKnownPrefix txt ps
 1038     trimAtQuestion = fst . T.breakOn "что "
 1039     cleanCandidate =
 1040       T.dropAround (\c -> Char.isSpace c || c `elem` ['.', ',', ';', ':', '?', '!'])
 1041         . T.replace "или" ""
