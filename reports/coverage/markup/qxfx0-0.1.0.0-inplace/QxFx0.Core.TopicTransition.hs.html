   19 buildAdjacency :: MeaningGraph -> Map MeaningStateId (Map MeaningStateId Double)
   20 buildAdjacency mg =
   21   let outgoing = L.foldl' addEdge M.empty (mgEdges mg)
   22   in outgoing
   23   where
   24     addEdge acc e =
   25       let w = fromIntegral (meCount e)
   26       in M.insertWith (M.unionWith (+)) (meFromId e) (M.singleton (meToId e) w) acc
   27 
   28 shortestPath :: Map MeaningStateId (Map MeaningStateId Double) -> MeaningStateId -> MeaningStateId -> Double
   29 shortestPath adj start goal = go M.empty (M.singleton start 0.0)
   30   where
   31     go visited dists
   32       | M.null unvisited = 1.0 / 0.0
   33       | current == goal  = currentDist
   34       | otherwise =
   35           let nbrs = M.findWithDefault M.empty current adj
   36               visited' = M.insert current currentDist visited
   37               dists' = M.delete current dists
   38               dists'' = M.foldlWithKey' updateDist dists' nbrs
   39           in go visited' dists''
   40       where
   41         unvisited = M.difference dists visited
   42         (current, currentDist) = L.minimumBy (comparing snd) (M.toList unvisited)
   43         updateDist d neighbor weight =
   44           let edgeDist = 1.0 / max weight 1e-10
   45               newDist = currentDist + edgeDist
   46           in M.insertWith min neighbor newDist d
   47 
   48 geodesicDistance :: MeaningGraph -> Text -> Text -> Double
   49 geodesicDistance mg fromTopic toTopic
   50   | fromTopic == toTopic = 0.0
   51   | otherwise =
   52       let adj = buildAdjacency mg
   53           fromId = T.unpack fromTopic
   54           toId   = T.unpack toTopic
   55       in if M.member fromId adj && M.member toId adj
   56          then shortestPath adj fromId toId
   57          else 1.0 / 0.0
   58 
   59 geodesicRouter :: MeaningGraph -> Text -> Text -> [Text] -> GeodesicPlan
   60 geodesicRouter mg fromTopic toTopic recentTopics =
   61   let dist = geodesicDistance mg fromTopic toTopic
   62       threshold = 3.0
   63   in if dist <= threshold || isInfinite dist
   64      then DirectJump
   65      else let adj = buildAdjacency mg
   66               fromId = T.unpack fromTopic
   67               toId   = T.unpack toTopic
   68               fromNeighbors = M.keys (M.findWithDefault M.empty fromId adj)
   69               bridges = filter (\bridge -> M.member toId (M.findWithDefault M.empty bridge adj)) fromNeighbors
   70               recentIds = map T.unpack recentTopics
   71               fresh = filter (`notElem` recentIds) bridges
   72            in if null fresh
   73               then DirectJump
   74               else BridgedJump (take 2 (map T.pack fresh))
